import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_init, n_points):
    """Ensure h is in [0,1] and sums to n_points/2"""
    h = np.array(h_init)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    """Initialize coarse grid"""
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    return h

def upscale(h_coarse, n_points):
    """Upsample coarse solution to fine grid"""
    # Repeat each coarse value to fill fine grid
    upsample_factor = n_points // len(h_coarse)
    h_fine = np.repeat(h_coarse, upsample_factor)
    # Adjust to match exact sum
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h, dx=2.0/len(h)):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h_init, n_points, max_iter=100, tol=1e-8):
    """Optimize h using SLSQP"""
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
        
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    h = h_init.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        result = minimize(
            lambda x: compute_c5(x, dx),
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'ftol': tol}
        )
        
        if result.success:
            h = result.x
        else:
            # Small random perturbation
            h += np.random.normal(0, 0.005, n_points)
            
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    return best_h, best_c5
