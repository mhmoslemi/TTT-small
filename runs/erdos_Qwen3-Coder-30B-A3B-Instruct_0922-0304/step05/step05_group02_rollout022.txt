import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * 0.5
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute all cross-correlations for shifts
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(overlap) * dx

    # Project h to feasible region
    def project_h(h_vec):
        h_proj = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_proj)
        if h_sum > 0:
            h_proj = h_proj * (n_points / 2) / h_sum
        return h_proj
    
    # Interior point optimization to minimize max overlap
    def optimize_step(h_current):
        # Create variables
        h = Variable(n_points)
        C = Variable(1)
        
        # Objective
        objective = Minimize(C)
        
        # Constraints
        constraints = [sum(h) == n_points / 2, 
                       h >= 0, 
                       h <= 1]
        
        # Add max overlap constraints
        h_ref = np.array(h_current)
        corr_ref = correlate(h_ref, 1 - h_ref, mode="full")
        
        # Add all shift constraints
        for k in range(len(corr_ref)):
            shift_corr = corr_ref[k]
            constraints.append(C >= shift_corr * dx)
        
        # Problem
        prob = Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=ECOS, verbose=False, feastol=1e-6, reltol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_result = h.value
                return project_h(h_result)
        except:
            pass
        return h_current

    iter_count = 0
    perturbation_decay = 0.95
    perturbation_magnitude = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Optimization step
        h_new = optimize_step(h)
        
        # Local search step using gradient-based optimization
        c5_new = compute_c5(h_new)
        
        # Update best if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Perturbation
        if iter_count % 50 == 0 and iter_count > 0:
            perturbation_magnitude *= perturbation_decay
            
        if perturbation_magnitude > 1e-6:
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h_new = h_new + noise
            h_new = project_h(h_new)
        
        h = h_new
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
