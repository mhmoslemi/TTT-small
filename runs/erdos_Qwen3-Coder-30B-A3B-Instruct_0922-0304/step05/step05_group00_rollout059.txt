import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

        h[n_points//2:] = 0

    h = h * (target_sum / np.sum(h))
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps to select important lags
        overlap_full = correlate(h, 1 - h, mode="full")
        # Get indices of top 10% lags by overlap value
        top_k = max(1, n_points // 10)
        top_lags = np.argsort(overlap_full)[-top_k:]
        
        # Create variables for optimization
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints for selected lags
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1,
            C5 >= 0
        ]
        
        # Add linearized constraints for top lags
        for k in top_lags:
            # This is an approximation: using bilinear terms for linearization

            # Linearized as: sum_i h_vars[i] * (1 - h[k]) + sum_i (h[k] - h[i+k]) * h_vars[i]
            # But proceed with exact computation for fewer terms

            # Dual approximation: h[k] * (sum(h) - h[k]) + h[i] * (1 - h[i+k]) - h[k] * (h[i] - h[i])
            # This is still complex, we'll compute it exactly through processing
            
            # Better strategy: materialize all lag-dependent terms explicitly
            # Let's compute overlap exactly via explicit sum for selected lags
            # Overlap_k = sum(h[i] * (1 - h[i+k]) for i in range(n_points-k) if i+k < n_points)
            
            # So terms for i = 0 to n_points-k-1:
            # sum_i h[i] * (1 - h[i+k]) = sum_i h[i] - sum_i h[i] * h[i+k]
            # We linearize: - sum_i h[i] * h[i+k] ≈  sum_i h[i] * (1 - h[i+k]) - h[i] 
            # Approximate: 
            # Let C[k] = sum_i h[i+k] 
            # Then -sum_i h[i] * h[i+k] + h[i] ≈ h[i]* (1 - h[i+k]) - h[i]*h[i+k] + h[i]

    return (best_h.tolist(), best_c5, n_points)
