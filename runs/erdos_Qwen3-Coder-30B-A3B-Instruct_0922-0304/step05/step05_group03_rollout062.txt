import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to each h[i]."""
    n = len(h)
    grad = np.zeros(n)
    
    # For each position, compute gradient contribution
    # This simulates the gradient of the max overlap term
    # We'll linearize by using the most critical lag
    overlap = correlate(h, 1 - h, mode="full")
    max_lag_idx = np.argmax(overlap)
    
    # The gradient computation is simplified due to the non-smoothness
    # We use finite difference approximation for the gradient
    eps = 1e-5
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = normalize_h(h_plus, np.sum(h))
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = normalize_h(h_minus, np.sum(h))
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Parameters for optimization
    learning_rate = 0.1
    max_iter = 10000
    patience = 100
    patience_counter = 0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise optimization steps
        for _ in range(5):  # Do several coordinate updates
            if time.time() - start_time > budget_s - 1:
                break
            i = np.random.randint(0, n_points)
            h_old = h[i]
            
            # Try small perturbations
            perturbations = [-0.01, 0.01]
            best_h_i = h_old
            best_c5_i = current_c5
            
            for delta in perturbations:
                h_test = h.copy()
                h_test[i] += delta
                h_test = normalize_h(h_test, target_sum)
                
                c5_test = evaluate_c5(h_test, dx)
                if c5_test < best_c5_i:
                    best_c5_i = c5_test
                    best_h_i = h_test[i]
                    
            h[i] = best_h_i
            
        # Compute gradient and update
        if time.time() - start_time < budget_s - 1:
            grad = compute_gradient(h, dx)
            h = h - learning_rate * grad
            h = normalize_h(h, target_sum)
            
        # Adjust learning rate
        if iteration % 50 == 0:
            # Reduce learning rate if no improvement for a while
            if iteration > 0 and iteration % 100 == 0:
                patience_counter += 1
                if patience_counter > patience:
                    learning_rate *= 0.5
                    patience_counter = 0

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
