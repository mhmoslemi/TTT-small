import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def halton_sequence(n, base=2):
    """Generate n points of Halton sequence in [0,1)^n."""
    result = np.zeros((n, 1))
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while i > 0:
            f /= base
            result[i] = (result[i] + (i % base) * f)
            i //= base
    return result

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Generate multiple initial points using Halton sequence
    halton_points = halton_sequence(50, 2)
    initial_points = []
    initial_points.append(h.copy())
    
    for i in range(1, 50):
        # Use Halton points and adjust for constraints
        h_init = halton_points[i].flatten()
        h_init = normalize_h(h_init, target_sum)
        initial_points.append(h_init.copy())
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with SLSQP
    for i, h_start in enumerate(initial_points):
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # SLSQP optimization with bounds and constraints
        try:
            result = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
                h_opt = normalize_h(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    
        except:
            pass
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
