import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient with respect to h for the overlap at lag k_max."""
    n = len(h)
    grad = np.zeros(n)
    # For each position i, compute how change in h[i] affects the overlap at k_max
    for i in range(n):
        # Contribution to overlap at k_max is h[i] * (1 - h[i + k_max])
        # When k_max is positive (i.e., lag is in forward direction)
        if i + k_max < n:
            grad[i] += 1 - h[i + k_max]
        # When k_max is negative (i.e., lag is in backward direction)
        if i - k_max >= 0:
            grad[i] += h[i - k_max]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = normalize_h(h, target_sum)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize learning rate and related parameters
    alpha = 0.1
    no_improvement_count = 0
    max_no_improvement = 50
    
    iters = 0
    while time.time() - start_time < budget_s - 1:
        iters += 1
        
        # Compute current overlap and find maximum lag
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Track best
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Reduce learning rate if no improvement for a while
                alpha *= 0.5
                no_improvement_count = 0
                if alpha < 1e-6:
                    break  # Stop if learning rate gets too small
                    
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient for max overlap lag
        grad = compute_gradient(h, dx, k_max)
        
        # Update h
        h_new = h - alpha * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to maintain sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Add small random perturbation to escape local minima occasionally
        if np.random.random() < 0.05 and iters > 10:
            h_new += np.random.normal(0, 0.001, n_points)
            h_new = normalize_h(h_new, target_sum)
        
        h = h_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
