```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] with sum = target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial if available, otherwise create a good starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = project_to_feasible(h, target_sum)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute C5 for current h
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Optimization using CVXPY for convex relaxation
        # Create variables
        h_var = Variable(n_points)
        C = Variable()
        
        # Build constraint matrix
        # Each constraint is C >= (h * (1-h))[k] 
        # For simplicity, we'll use the full correlation and fix one projection
        # Manually construct the constraints
        
        # Compute cross correlation for h*(1-h) at all shifts
        cross_corr = correlate(h, 1-h, mode="full")
        limits = len(cross_corr) // 2
        
        # Constraints: C >= [cross_corr[k] * dx for k in range(-limits, limits+1)]
        constraints = [C >= cross_corr[k] * dx for k in range(len(cross_corr))]
        constraints.append(sum(h_var) == target_sum)
        constraints.extend([0 <= h_var[i], h_var[i] <= 1 for i in range(n_points)])
        
        # Solve optimization problem
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        # Try different solvers
        try:
            prob.solve(solver=CP_ECS, verbose=False, max_iters=100)
        except Exception:
            try:
                prob.solve(solver=OSQP, verbose=False, max_iter=100)
            except Exception:
                try:
                    prob.solve(solver=SCS, verbose=False, max_iter=100)
                except Exception:
                    # Fall back to inexact approach or skip
                    pass

        # Update h if successful
        if prob.status == 'optimal':
            h_new = np.array(h_var.value).flatten()
            h = project_to_feasible(h_new, target_sum)
        else:
            # Fallback: local search with small perturbation
            h = h + np.random.normal(0, 0.001, n_points)
            h = project_to_feasible(h, target_sum)

        # Occasionally perturb to escape local minima
        if iteration % 50 == 0:
            h = h + np.random.normal(0, 0.01 * (1 - iteration/(budget_s/10)), n_points)
            h = project_to_feasible(h, target_sum)
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
