import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        
        while len(population) < population_size:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
        
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_index].copy())
        return selected

    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2)
        return child

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Random perturbation
                if np.random.rand() < 0.5:
                    h[i] += np.random.normal(0, 0.05)
                else:
                    h[i] = np.random.rand()
                
                # Ensure feasibility
                h[i] = np.clip(h[i], 0, 1)
        
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Initialize
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = [compute_c5(h) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Time check
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_h.copy())
        
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
