import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with adaptive step size
    learning_rate = 0.1
    patience = 0
    patience_limit = 10
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap correlation to find the maximum lag
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        
        # Compute gradient numerically
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = normalize_h(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = normalize_h(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_lag_idx]
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_lag_idx]
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Gradient step
        h_new = h - learning_rate * grad
        
        # Project to feasible region
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Update if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
            last_improvement = iteration
        else:
            patience += 1
            
        # Adapt learning rate
        if patience > patience_limit and iteration - last_improvement > patience_limit:
            learning_rate *= 0.5
            patience = 0
            
        # Coordinate-wise local search on top 10% of indices
        if iteration % 5 == 0:
            top_indices = np.argsort(np.abs(grad))[-n_points//10:]
            for idx in top_indices:
                original = h[idx]
                for perturbation in [-0.01, 0.01]:
                    h_try = h.copy()
                    h_try[idx] += perturbation
                    h_try = normalize_h(h_try, target_sum)
                    c5_try = evaluate_c5(h_try, dx)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        break
                else:
                    h[idx] = original
        
        h = h_new
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
