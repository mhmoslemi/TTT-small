import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create a simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)

    # Gradient descent parameters
    max_iter = 10000
    lr = 0.01
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient for the active shift
        # We need to find the shift corresponding to max overlap
        # The shift k corresponds to the position of max_k in the full correlation
        # For a correlation of length 2*n-1, the shift k (0 to 2*n-2) means:
        # k=0 corresponds to no shift, k=1 to shift by 1, ..., k=n-1 to shift by n-1
        # and then k=n to shift by -1, ..., k=2*n-2 to shift by -(n-1)
        # But since we're dealing with wraparound, we can just use the full correlations
        # and find the maximum

        # Special case for finding maximum overlap shift
        k_actual = max_k - (n_points - 1)  # Convert full index to shift value
        # Handle wraparound
        if k_actual < 0:
            k_actual = n_points + k_actual
        
        # Compute gradient information
        # The gradient for shift k is (1 - h_{i+k}) where h_{i+k} is the shifted version
        # We're gradient of the overlap wrt to h, and we want to minimize it
        # d/dh_i [ sum_j h_j (1-h_{j+k}) ] = 1 - h_{i+k}
        # Fixing a consistent derivative: should be based on 1-h_{i-k}  (position j in sum over i)
        # gradient = 1 - h_{i-k}
        # if k=1, h[i-1], so gradient[idx] = (1-h[idx-1])  
        # But need to carefully define what we shift, and how to sample
        # Accessing hshifted = h(connecting with wraparound), generally h[idx + k]
        # Since we care about the overlaps, let's correctly compute the return for the overlap shift
        
        # Simplified approach: compute actual gradient for the max overlap shift
        if k_actual < n_points:
            # This is shift to the right. Actual shift index: j = i + k_actual
            gradient = np.zeros(n_points)
            for j in range(n_points):
                i = (j - k_actual) % n_points  # j = i + k_actual => i = j - k_actual (mod n)
                gradient[j] = 1 - h[i]
        else:
            # Negative shift
            shift = 2*n_points - 1 - k_actual
            if shift >= n_points:
                shift = k_actual - n_points  # This path seems wrong, must recompute
            k_actual = (2*n_points - 1) - k_actual
            gradient = np.zeros(n_points)
            for j in range(n_points):
                i = (j + shift) % n_points
                gradient[j] = 1 - h[i]
        
        # Update h using gradient descent
        h = h - lr * gradient
        h = project_h(h, target_sum)
        
    # Final check on best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
