import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Binary search for optimal C5
    C5_low = 0.0
    C5_high = 0.38085
    tolerance = 1e-5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Track consecutive failures
    no_improvement_count = 0
    max_no_improvement = 30
    
    # Main binary search loop
    while C5_high - C5_low > tolerance and time.time() - start_time < budget_s - 1:
        C5_mid = (C5_low + C5_high) / 2.0
        
        # Define constraint for overlap
        def max_overlap_constraint(h):
            overlap = correlate(h, 1 - h, mode="full")
            return C5_mid * dx - np.max(overlap) * dx
        
        # Add the overlap constraint
        constraints_with_overlap = constraints + [{'type': 'ineq', 'fun': max_overlap_constraint}]
        
        # Try to find a feasible solution
        result = minimize(
            lambda x: np.sum(x**2),  # Objective to avoid infeasible regions
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints_with_overlap,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_candidate = result.x
            # Ensure bounds are respected
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = compute_c5(h_candidate)
            
            if c5_candidate <= C5_mid + 1e-8:
                # Valid solution found
                C5_high = C5_mid
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                # Perturb slightly to escape local minima
                h = h_candidate + np.random.normal(0, 0.005, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                no_improvement_count = 0
            else:
                C5_low = C5_mid
                no_improvement_count += 1
        else:
            C5_low = C5_mid
            no_improvement_count += 1
            
        # Perturb if no improvement for a while
        if no_improvement_count > 15:
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0

    # Final refinement
    if time.time() - start_time < budget_s - 1:
        # Try local optimization on best solution
        def objective(h):
            return compute_c5(h)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use trust-constr method for better local refinement
        result = minimize(
            objective,

            method='trust-constr',

            constraints=constraints,
            options={'maxiter': 30, 'gtol': 1e-8, 'barrier_tol': 1e-8}

        if result.success:

            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)

            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()

    return (best_h.tolist(), best_c5, n_points)
