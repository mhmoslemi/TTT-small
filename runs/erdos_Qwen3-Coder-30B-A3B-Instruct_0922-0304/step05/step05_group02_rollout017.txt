import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h = project_h(h, target_sum)
    
    # Optimization parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations and find the worst case
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Gradient computation: derivative of max correlation w.r.t. h
        # This is roughly (1 - h_shifted) where h_shifted is shifted h
        # For shift k=max_k, we compute the gradient
        k = max_k - (n_points - 1)  # this maps full correlation index to actual shift
        if k >= 0:
            # h[i] contributes to correlation[i+k] 
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + k < len(h):
                    grad[i] = 1 - h[i + k]
        else:
            # k < 0, reverse shift
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i - k < len(h):
                    grad[i] = 1 - h[i - k]
        
        # Update with gradient descent
        h = h - eta * grad * dx
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Adaptive learning rate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            eta = min(eta * 1.01, 1.0)
        else:
            eta = max(eta * 0.9, 0.001)
        
        # Periodic noise injection for escape from local optima
        if iteration % 100 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
