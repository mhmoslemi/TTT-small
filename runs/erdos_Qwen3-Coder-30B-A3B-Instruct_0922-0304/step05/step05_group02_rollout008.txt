import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while clipping to [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient for the k_max shift."""
    n = len(h)
    grad = np.zeros(n)
    # For shift k_max, gradient is (1 - h_{k_max:])
    # Direct correlation: when we correlate h with (1-h) at shift k_max,
    # the gradient w.r.t. h_i is (1 - h_{i+k_max})
    # So at position i, we add (1 - h_{i+k_max})
    # Check if k_max is positive: indexing in reverse
    # When the shift is positive, we shift h to the right, so 
    # gradient in position i is (1 - h[i-k_max]) when i>=k_max
    # When k_max is negative, we shift the mask to the right, 
    # position i gets contribution if i+k_max >= 0, so h[i+k_max] 
    # we correct for the range ambiguity:
    # when k_max is the index in the full correlation, it corresponds to
    # a left shift of |k_max| positions (i -> i + |k_max| )
    # in cross-correlation correlate(h, 1-h) where h is indexed from 0,
    # 1-h is shifted and matched, giving correlation at shift k
    # a shift of k means we match h[i] with (1-h)[i+k]
    # so gradient at i is (1-h[i+k])
    # but in the final result we need to shift the gradient negatively so
    # to have the shifted version inside: (1-h)[i+k] where shifting
    # k=i+k' implies i+k' = (i + k) mod n
    # For a right shift k (positive), we should match h[i] with (1-h)[i+k]
    # So gradient is (1-h[i+k]) for i from 0 to (n-k-1) and 0 elsewhere
    # For left shift k (negative), we match h[i] with (1-h)[i+k] 
    # where i+k non-negative and k negative (i+k <= i)
    # so it's h[i+k] for i from -k to n-1 
    # Correlation returns indices from -n+1 to n-1
    # index k_max represents a shift of k_max positions (0 is no shift)
    # if k_max >= 0, we match h[i] with (1-h)[i+k_max]
    # if k_max < 0, we match h[i] with (1-h)[i+k_max]
    
    # Converting k_max from full indices to shift direction:
    # full correlation index k means shift k = k - (n-1) 
    # so k_max = k + (n-1) so k = k_max - (n-1)
    shift = k_max - (n - 1)
    if shift >= 0:
        # right shift of shift
        # for i in range(n-shift), gradient += (1-h[i+shift])
        grad[shift:] += (1 - h[:n-shift]) 
    else:
        # left shift of -shift
        # for i in range(-shift, n), gradient += (1-h[i+shift]) 
        grad[:n+shift] += (1 - h[-shift:])
    return grad * dx

def line_search(h, grad, step_size, dx, max_iter=10):
    """Perform line search for step size."""
    old_val = compute_c5(h, dx)
    for i in range(max_iter):
        h_new = project_h(h - step_size * grad, len(h) / 2)
        new_val = compute_c5(h_new, dx)
        if new_val < old_val:
            return step_size
        step_size *= 0.5
    return step_size

        h = np.ones(n_points) * 0.5
        # slightly perturb it to avoid symmetry
        h += np.random.normal(0, 0.02, n_points)
        h = project_h(h, n_points / 2)
        
    # Parameters
    momentum = 0.9
    step_size = 0.05
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    last_improvement = 0
    
    # Accelerated coordinate ascents
    v = np.zeros(n_points)
    prev_grad = np.zeros(n_points)

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute all correlations
        cross_corr = correlate(h, 1 - h, mode="full")
        # Find k that maximizes correlation 
        k_max = np.argmax(cross_corr)
        # Get gradient direction
        grad = compute_gradient(h, k_max, dx)
        
        # Momentum-based update
        v = momentum * v + grad
        # Projected gradient descent step
        h_new = project_h(h - step_size * v, n_points / 2)
        
        # Optimistic line search
        step_size = line_search(h, v, step_size, dx)
        h = h_new
        
        # Update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            if best_c5 <= 0.3807:

        # Adaptive step-size

            if iteration - last_improvement > 50:
                step_size *= 0.8
                last_improvement = iteration
