import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 20000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: alternating pattern
        h = np.ones(n_points)
        h[::2] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simple gradient descent approach with annealing
    iteration = 0
    learning_rate = 0.01
    min_lr = 1e-5
    decay_factor = 0.999
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Current overlap computation
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        current_c5 = overlap[max_k] * dx
        
        # Store if better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Clamp and renormalize
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
        
        # Gradient step
        new_h = h - learning_rate * grad
        
        # Clamp and renormalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Update h with a small probability of random perturbation
        if np.random.random() < 0.05:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + 0.2 * perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = new_h
            
        # Anneal learning rate
        if learning_rate > min_lr:
            learning_rate *= decay_factor
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
