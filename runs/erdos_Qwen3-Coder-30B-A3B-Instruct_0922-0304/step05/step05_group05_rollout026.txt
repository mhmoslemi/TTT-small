import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # SA parameters
    T0 = 0.1 * current_c5
    T_min = 1e-8
    alpha = 0.95
    max_iter = 10000
    iter_per_temp = 50
    
    T = T0
    iteration = 0
    
    while T > T_min and time.time() - start_time < budget_s - 1:
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Create neighbor by perturbing current solution
            h_new = current_h + np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new objective
            c5_new = compute_c5(h_new)
            
            # Accept or reject
            delta = c5_new - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = h_new
                current_c5 = c5_new
                
                # Update best
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
                    if best_c5 <= 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
            
            iteration += 1
            
        # Cool down
        T *= alpha
        
    return (best_h.tolist(), best_c5, n_points)
