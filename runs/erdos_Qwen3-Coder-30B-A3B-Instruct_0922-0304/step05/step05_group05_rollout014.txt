import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization approach
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Solve convex relaxation using CVXPY
        try:
            # Create CVXPY variables
            variables = cp.Variable(n_points)
            constraints = [
                variables >= 0,
                variables <= 1,
                cp.sum(variables) == n_points / 2
            ]
            
            # The overlap constraint is tricky in convex form,
            # we approximate by using a linearization heuristic
            # For rough approximation: just solve with linearized objective
            
            # Objective: minimize max cross-correlation
            # This is non-convex, but we'll solve an easier approximant
            # Approximate max cross-correlation with a scalar variable
            max_overlap = cp.Variable()
            objective = cp.Minimize(max_overlap)
            
            # Constraints for overlap tracking
            # This is a complex constraint - it's simpler to work with the actual computation
            prob = cp.Problem(objective, constraints)
            
            # Since full convex reformulation is complex, 
            # we do a simple gradient-descent based optimization
            
        except:
            pass  # Fall back to simple update
            
        # Simple Gradient-based update
        # Compute gradient direction (approximate)
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Create small perturbation
        perturbation = np.zeros(n_points)
        # Perturb around the max overlap position
        if max_idx < n_points:
            # Make sure we are within valid index range
            i = max_idx
            # Simple negative gradient step
            if i < n_points - 1:
                perturbation[i] = 0.01
                perturbation[i+1] = -0.01
                
        # Apply gradient update and project
        h_new = h - 0.01 * perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Add small random noise to avoid local minima
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Early exit if we have a perfect solution (rare)
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), final_c5, n_points)
