import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Add initial solution as first particle
    particles.append(h.copy())
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * swarm_size
    
    # Initialize global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute overlap for current particle
            overlap = correlate(h, 1 - h, mode="full")
            current_score = np.max(overlap) * dx
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = h.copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = h.copy()
            
            # Early exit if target reached
            if global_best_score <= 0.3807:
                break
                
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            new_velocity = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - h) +
                c2 * r2 * (global_best - h)
            )
            
            # Clip velocity
            new_velocity = np.clip(new_velocity, -v_max, v_max)
            
            # Update position
            new_h = h + new_velocity
            
            # Apply boundary constraints
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize to maintain sum constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Update particle
            particles[i] = new_h
            velocities[i] = new_velocity
            
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_overlap = correlate(global_best, 1 - global_best, mode="full")
    final_c5 = np.max(final_overlap) * dx
    
    return (global_best.tolist(), final_c5, n_points)
