import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Fallback initialization
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial swarm
    particles = [h0.copy()]
    velocities = [np.random.uniform(-0.01, 0.01, n_points)]
    
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track best solutions
    swarm_best = [p.copy() for p in particles]
    swarm_best_scores = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute C5 for this particle
            score = compute_c5(h)
            
            # Update particle's best
            if score < swarm_best_scores[i]:
                swarm_best_scores[i] = score
                swarm_best[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
            
            # Early exit if target achieved
            if global_best_score <= 0.3807:
                break
                
        # Update particles
        for i in range(swarm_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            p_best = swarm_best[i]
            g_best = global_best
            
            v = w * v + c1 * r1 * (p_best - h) + c2 * r2 * (g_best - h)
            
            # Velocity clipping
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to sum to n/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
            # Check if we've hit target
            if global_best_score <= 0.3807:
                break
                
    return (global_best.tolist(), global_best_score, n_points)
