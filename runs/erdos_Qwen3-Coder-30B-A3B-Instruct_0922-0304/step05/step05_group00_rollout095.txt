import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        # Create a simple initial construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Track the best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Early exit if already good enough
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute current overlap to find worst lags
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        max_overlap = overlap[max_lag] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Select a few high-overlap lags to focus on
        # Use a window of ±10 lags around max_lag
        window_size = min(10, len(overlap) // 2)
        start_idx = max(0, max_lag - window_size)
        end_idx = min(len(overlap), max_lag + window_size + 1)
        
        # Sample lags to add to our linearization
        # Let's select the top 10% of lags in this window
        window_overlaps = overlap[start_idx:end_idx]
        top_indices = np.argsort(window_overlaps)[-max(2, len(window_overlaps)//10):]
        selected_lags = [start_idx + i for i in top_indices]
        
        # Now, create a local optimization problem for these lags
        # Using a heuristic gradient-like approach: move towards lower overlap
        step_size = 0.005
        temp_h = h.copy()
        
        # For each selected lag, adjust h to reduce that overlap
        for k in selected_lags:
            # This is a simplified approach
            # We compute the gradient element-wise w.r.t. h
            # For k lag, overlap contribution is h[i] * (1 - h[i+k])  
            # Gradient w.r.t. h[i] is (1 - h[i+k]) - h[i] * 0 = 1 - h[i+k]
            # We move h[i] towards decreasing this gradient (i.e., reduce h[i+k])
            
            # Adjust according to the influence of h[i+k] on overlap
            if k > 0 and k < len(h):
                for i in range(len(h) - k):
                    # Gradient from the overlap term: subtract 1 on violation
                    if h[i] > 0 and h[i+k] < 1:
                        temp_h[i] = np.clip(temp_h[i] - step_size * (1 - h[i+k]), 0, 1)
                        temp_h[i+k] = np.clip(temp_h[i+k] + step_size * h[i], 0, 1)
        
        # Apply projection to maintain feasibility
        temp_h = project_to_feasible(temp_h, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(temp_h, dx)
        if new_c5 < best_c5:
            h = temp_h
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
