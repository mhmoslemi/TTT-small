import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def halton_sequence(n, base=2):
    """Generate n points of 1D Halton sequence."""
    sequence = []
    for i in range(n):
        x = 0
        f = 1.0 / base
        while i > 0:
            i, r = divmod(i, base)
            x += r * f
            f /= base
        sequence.append(x)
    return np.array(sequence)

    target_sum = n_points / 2

    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Generate multiple initial guesses using Halton sequence
    population_size = 50
    population = []
    halton_points = halton_sequence(population_size, 2)
    
    # Add the current solution to the population
    population.append(h.copy())
    
    # Generate Halton-based initial guesses
    for i in range(population_size - 1):
        # Mix Halton with some random perturbation
        h_initial = np.interp(np.linspace(0, 1, n_points), 
                             np.linspace(0, 1, n_points), 
                             halton_points[i] * np.ones(n_points))
        # Smooth out the step function
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = normalize_h(h_initial, target_sum)
        population.append(h_initial)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start local search
    for i, h_start in enumerate(population):
        if time.time() - start_time > budget_s - 1:
            break
            
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_final = result.x
            h_final = normalize_h(h_final, target_sum)
            c5_value = compute_c5(h_final)
            
            if c5_value < best_c5:
                best_c5 = c5_value
                best_h = h_final.copy()

    # Final refinement using a simple gradient descent approach
    if time.time() - start_time < budget_s - 1:

            # Simple gradient descent on a subset of points to improve smoothness
            h = h.copy()
            for _ in range(50):

                delta = 1e-4

        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
