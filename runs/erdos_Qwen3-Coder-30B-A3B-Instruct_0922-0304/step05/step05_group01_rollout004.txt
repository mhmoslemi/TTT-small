import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    elite_size = population_size // 4
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def is_feasible(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) * dx - 1.0) < 1e-10
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def mutate(h, sigma):
        mutated = h + np.random.normal(0, sigma, h.shape)
        return repair(mutated)
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population[0] = repair(population[0])
    else:
        population = [np.ones(n_points)]
        population[0] = repair(population[0])
    
    # Fill population with mutated versions of initial
    while len(population) < population_size:
        mutant = mutate(population[0], 0.05)
        population.append(mutant)
    
    best_c5 = float('inf')
    best_h = None
    
    max_generations = 1000
    no_improvement_count = 0
    max_no_improvement = 50
    mutation_sigma = 0.01
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(h) for h in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Adjust mutation sigma if no improvement
        if no_improvement_count > 20:
            mutation_sigma *= 0.8
            no_improvement_count = 0
            
        # Create new population from elite
        new_population = []
        elite = population[:elite_size]
        
        for individual in elite:
            # Create two new individuals
            for _ in range(2):
                new_individual = mutate(individual, mutation_sigma)
                new_population.append(new_individual)
                
        # Ensure population size
        population = new_population[:population_size]
        
        # Add some random individuals occasionally
        while len(population) < population_size:
            random_h = np.random.rand(n_points)
            random_h = repair(random_h)
            population.append(random_h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
