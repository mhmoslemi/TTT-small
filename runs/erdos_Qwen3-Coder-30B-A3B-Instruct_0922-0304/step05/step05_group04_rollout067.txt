import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    # Compute C5 for an individual
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Tournament selection
    def tournament_selection(pop, fitnesses):
        tournament_indices = np.random.choice(len(pop), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        return pop[winner_index]
    
    # Uniform crossover
    def crossover(parent1, parent2):
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        # Renormalize
        child = child * (n_points / 2) / np.sum(child)
        return np.clip(child, 0, 1)
    
    # Mutation
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    # Add noise
                    h[i] += np.random.normal(0, 0.1)
                else:
                    # Set extreme value
                    h[i] = np.random.choice([0, 1])
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Main genetic algorithm
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = [compute_c5(individual) for individual in population]
        
        # Track best individual
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:population_size//2]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Final check and return
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
