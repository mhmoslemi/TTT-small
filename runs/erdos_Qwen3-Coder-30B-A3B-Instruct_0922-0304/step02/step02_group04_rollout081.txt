import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = project_to_feasible(h, n_points)

    # Initialize
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temperature = 0.1 * current_c5
    cooling_rate = 0.999
    noise_std = 0.01
    max_iterations = 10000
    
    # Gradient descent parameters
    grad_step_size = 0.01
    grad_steps = 5
    grad_decay = 0.9
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate via random perturbation
        candidate = h + np.random.normal(0, noise_std, n_points)
        candidate = project_to_feasible(candidate, n_points)
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject candidate
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            temperature *= cooling_rate
        else:
            # Accept with probability
            if random.random() < np.exp(-(candidate_c5 - current_c5) / max(temperature, 1e-10)):
                h = candidate
                current_c5 = candidate_c5
            temperature *= cooling_rate
            
        # Occasionally apply gradient descent refinement
        if iteration % 50 == 0 and iteration > 0:
            for _ in range(grad_steps):
                if time.time() - start_time > budget_s - 1:
                    break
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    c5_minus = np.max(corr_minus) * dx
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                h = h - grad_step_size * grad
                h = project_to_feasible(h, n_points)
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
                grad_step_size *= grad_decay
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
