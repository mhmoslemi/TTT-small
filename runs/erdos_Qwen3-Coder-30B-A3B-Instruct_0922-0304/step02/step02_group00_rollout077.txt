import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback construction
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative Frank-Wolfe style approach
    max_iter = 1000
    patience = 0
    patience_limit = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > patience_limit:
            break
            
        # Formulate and solve convex subproblem for k_max shift
        # Objective: minimize sum_i h[i] * (1 - h[(i + k_max) % n_points])
        h_new = h.copy()
        
        # Use CVXPY for convex relaxation
        h_var = Variable(n_points)
        # Linear objective for overlap at k_max
        objective = cp_sum(cp.multiply(h_var, 1 - h_var[(np.arange(n_points) + k_max) % n_points]))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve convex problem
        prob = Problem(Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
            else:
                h_new = h.copy()  # fallback to previous
        except:
            h_new = h.copy()  # fallback to previous
            
        # Maintain feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
