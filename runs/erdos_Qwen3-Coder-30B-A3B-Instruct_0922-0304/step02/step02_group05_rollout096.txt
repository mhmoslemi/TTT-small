import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        particles.append(h.copy())
    
    # Fill remaining particles with random valid solutions
    while len(particles) < num_particles:
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(num_particles):
        velocities.append(np.random.normal(0, 0.05, n_points))
        
    # Evaluate initial fitness
    fitness_scores = []
    for i, h in enumerate(particles):
        c5 = evaluate_c5(h, dx)
        fitness_scores.append((c5, i))
        
    # Set initial best particles
    fitness_scores.sort(key=lambda x: x[0])
    p_best_indices = [i for _, i in fitness_scores]
    g_best_idx = p_best_indices[0]
    best_c5 = fitness_scores[0][0]
    best_h = particles[g_best_idx].copy()
    
    # PSO parameters
    w = 1.0  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    max_vel = 0.2
    
    # Track stagnation
    stagnation_count = 0
    last_improvement = 0
    last_best_c5 = best_c5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        iteration += 1
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (particles[p_best_indices[i]] - particles[i]) +
                           c2 * r2 * (particles[g_best_idx] - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < fitness_scores[i][0]:
                fitness_scores[i] = (c5, i)
                p_best_indices[i] = i
                
                # Update global best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particles[i].copy()
                    last_improvement = iteration
                    
        # Update global best
        fitness_scores.sort(key=lambda x: x[0])
        if fitness_scores[0][0] < best_c5:
            best_c5 = fitness_scores[0][0]
            g_best_idx = fitness_scores[0][1]
            best_h = particles[g_best_idx].copy()
            last_improvement = iteration
            
        # Adaptive inertia
        if iteration % 5 == 0:
            w = max(0.8, w * 0.99)
            
        # Stagnation check and reset
        if iteration - last_improvement > 100:
            for i in range(num_particles):
                if fitness_scores[i][0] > best_c5 + 1e-4:
                    particles[i] = np.random.uniform(0, 1, n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    velocities[i] = np.random.normal(0, 0.05, n_points)
                    c5 = evaluate_c5(particles[i], dx)
                    fitness_scores[i] = (c5, i)
                    
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
