import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    # Compute overlap integral directly
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def project_feasible(h, n_points):
    # Project onto feasible region: h in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        scaling = (n_points / 2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, max_k):
    # We want the gradient of overlap_k with respect to h
    # overlap_k = sum_i h[i] * (1 - h[i+k])
    # grad overlap_k[i] = (1 - h[i+k]) - h[i+k]
    # Use zero-padding for indices beyond the array size
    n = len(h)
    grad = np.zeros_like(h)
    
    # For overlap_k = sum_i h[i] * (1 - h[i+k])
    for i in range(n):
        j = i + max_k
        if 0 <= j < n:
            grad[i] += (1 - h[j]) - h[j]
        else:
            # This case does not contribute to the gradient (zero padding)
            pass
    return grad

def line_search(h, grad, max_k, step_size=0.01, iterations=10):
    # Simple backtracking line search
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for _ in range(iterations):
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        step_size *= 0.5
    return best_h, best_c5

        h[:n_points//2] = 1.0

    # Initialize tracking variables
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    current_time = time.time()
    
    # Optimization loop
    for iteration in range(max_iterations):
        # Early exit if we're close
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_c5 = overlap[max_k] * dx
        
        # Compute approximate gradient
        grad = compute_gradient(h, max_k)
        
        # Line search to pick step size
        h_new, c5_new = line_search(h, grad, max_k)
        
        # Update h and check improvement
        h = h_new.copy()
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        
        # Periodic convex relaxation every 50 iterations
        if iteration % 50 == 0:
            # Add small L2 regularization to encourage exploration
            lambda_reg = 1e-5
            # Regularized optimization: minimize overlap + lambda * L2 norm
            # We do a simple relaxation step here with a small perturbation
            pert = np.random.normal(0, 1e-4, n_points)
            h_pert = h + pert
            h_pert = project_feasible(h_pert, n_points)
