import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h onto feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Run particle swarm optimization for Erdős minimum overlap problem."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    inertia_weight = 1.0
    cognitive_coeff = 2.0
    social_coeff = 2.0
    velocity_limit = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if it exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    else:
        # Random initialization
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h)
    
    # Fill remaining particles
    for _ in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h)
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial fitness
    fitness = []
    for i, h in enumerate(particles):
        c5 = evaluate_c5(h, dx)
        fitness.append(c5)
    
    # Set personal bests
    p_best = particles[:]
    p_best_fitness = fitness[:]
    
    # Set global best
    global_best_idx = np.argmin(fitness)
    g_best = particles[global_best_idx].copy()
    g_best_fitness = fitness[global_best_idx]
    
    # Track best solution found
    best_h = g_best.copy()
    best_c5 = g_best_fitness
    
    # PSO iterations
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] +
                            cognitive_coeff * r1 * (p_best[i] - particles[i]) +
                            social_coeff * r2 * (g_best - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate new fitness
            new_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_c5 < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = new_c5
                
                # Update global best if needed
                if new_c5 < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = new_c5
                    
                    # Update tracking
                    best_h = g_best.copy()
                    best_c5 = g_best_fitness
                    
                    last_improvement = iteration
        
        # Adaptive inertia
        if iteration % 50 == 0:
            inertia_weight = max(0.8, inertia_weight - 0.02)
            
            # Reset stagnant particles
            if iteration - last_improvement > 50:
                for i in range(n_particles):
                    if i != global_best_idx:
                        particles[i] = np.random.uniform(0, 1, n_points)
                        particles[i] = project_feasible(particles[i], n_points, dx)
                        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                        p_best[i] = particles[i].copy()
                        p_best_fitness[i] = evaluate_c5(particles[i], dx)
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
