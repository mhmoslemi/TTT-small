import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a step function that we'll refine
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add small noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    rho = 1.0
    lambda_k = np.zeros(2 * n_points - 1)  # For all shifts in full correlation
    penalty_factor = 2.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        current_c5 = overlap[k_max] * dx
        
        # Update best solution if better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
            
        # Using gradient descent with projected updates for optimization
        step_size = 0.01
        inner_iter = 100
        
        # Compute finite difference gradient of overlap w.r.t h
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update h
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        # Scale to maintain integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
