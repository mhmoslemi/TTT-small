import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, else create a good starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 30
    max_iter = 1000
    inertia = 0.7
    inertia_decay = 1e-4
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    
    # Initialize particles
    particles = []
    velocities = []
    
    # One particle from initial_h_values if available
    if 'initial_h_values' in globals():
        p = project_to_feasible(np.array(initial_h_values), n_points)
        particles.append(p.copy())
        velocities.append(np.zeros(n_points))
    else:
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill remaining particles with diverse structures
    for _ in range(n_particles - len(particles)):
        # Use blocky random structure
        p = np.ones(n_points)
        # Random split into 2 or more blocks
        n_blocks = 2
        block_start = np.sort(np.random.choice(n_points, n_blocks-1, replace=False))
        block_start = np.concatenate([[0], block_start, [n_points]])
        
        # Initialize each block with random intensity
        for i in range(n_blocks):
            start, end = block_start[i], block_start[i+1]
            intensity = np.random.uniform(0, 1)
            p[start:end] = intensity 
            
        # Add small Gaussian noise
        noise = np.random.normal(0, 0.05, n_points)
        p += noise
        p = project_to_feasible(p, n_points)
        
        particles.append(p.copy())
        velocities.append(np.zeros(n_points))
    
    # Best positions
    best_positions = [p.copy() for p in particles]
    best_c5s = [evaluate_c5(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(best_c5s)
    best_global_c5 = best_c5s[best_idx]
    best_global_pos = best_positions[best_idx].copy()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        inertia = max(0.3, inertia * (1 - inertia_decay))
        
        for i in range(n_particles):
            p = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = alpha * r1 * (best_positions[i] - p)
            social = beta * r2 * (best_global_pos - p)
            
            v = inertia * v + cognitive + social
            
            # Limit velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            p_new = p + v
            
            # Apply constraints
            p_new = project_to_feasible(p_new, n_points)
            
            # Check if improved
            c5_new = evaluate_c5(p_new, dx)
            particles[i] = p_new.copy()
            
            if c5_new < best_c5s[i]:
                best_positions[i] = p_new.copy()
                best_c5s[i] = c5_new
                
                if c5_new < best_global_c5:
                    best_global_c5 = c5_new
                    best_global_pos = p_new.copy()
                    
                    # Early termination
                    if best_global_c5 <= 0.3807:
                        break
                        
        # Early stopping if no improvement for a while
        if iteration % 50 == 0:
            if iteration > 0 and abs(best_global_c5 - evaluate_c5(best_global_pos, dx)) < 1e-6:
                break
                
    return (best_global_pos.tolist(), best_global_c5, n_points)
