import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a symmetric 5-fold pattern
        h = np.zeros(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.7
            else:
                h[start_idx:end_idx] = 0.3
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]

    def objective(h):
        return evaluate_c5(h, dx)

    def callback(x):
        nonlocal best_c5, best_h
        c5 = evaluate_c5(x, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()

    # Optimization using L-BFGS-B with multiple restarts
    max_restarts = 10
    restart_count = 0

    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Create a better starting solution
        h_mod = h + np.random.normal(0, 0.01, n_points)
        h_mod = np.clip(h_mod, 0, 1)
        h_mod = h_mod * (n_points / 2) / np.sum(h_mod)
        
        # Optimize using L-BFGS-B
        try:
            res = minimize(
                objective,
                h_mod,
                method='L-BFGS-B',
                bounds=bounds,
                callback=callback,
                options={'maxiter': 100}
            )
            
            if res.success:
                h = res.x.copy()
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                if evaluate_c5(h, dx) < best_c5:
                    best_c5 = evaluate_c5(h, dx)
                    best_h = h.copy()
                    
        except Exception:
            pass
            
        restart_count += 1
        
        # Early exit if target is reached
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
