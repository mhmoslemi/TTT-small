import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible space: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default: random construction
        h = np.random.random(n_points)
        h = project_feasible(h, n_points, dx)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    lambda_reg = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_k = np.argmax(overlap)
        
        # Compute gradient for shift k
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        current_c5 = evaluate_erdos_solution(h)
        best_candidate_c5 = current_c5
        best_candidate_h = h.copy()
        
        for step_size in line_search_steps:
            h_candidate = h - step_size * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # Accept better solution
        if best_candidate_c5 < current_c5:
            h = best_candidate_h
            current_c5 = best_candidate_c5
        
        # Periodic convex relaxation
        if iteration % relaxation_interval == 0:
            # Convex relaxation: min max_overlap + lambda * sum(h log h)
            try:
                h_relax = cp.Variable(n_points)
                constraints = [
                    h_relax >= 0,
                    h_relax <= 1,
                    cp.sum(h_relax) == n_points / 2
                ]
                obj = cp.Minimize(
                    max_overlap + lambda_reg * cp.sum(cp.entr(h_relax) + h_relax * np.log(n_points/2))
                )
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_relax_sol = h_relax.value
                    h_relax_proj = project_feasible(h_relax_sol, n_points, dx)
                    c5_relax = evaluate_erdos_solution(h_relax_proj)
                    if c5_relax < best_c5:
                        h = h_relax_proj.copy()
                        best_c5 = c5_relax
                        best_h = h.copy()
                    lambda_reg *= 0.5
                else:
                    lambda_reg *= 0.5
            except:
                lambda_reg *= 0.5
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
