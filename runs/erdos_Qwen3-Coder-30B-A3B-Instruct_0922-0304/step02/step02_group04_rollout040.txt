import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution_step(population, fitness, F=0.7, CR=0.9):
    new_population = []
    for i in range(len(population)):
        # Select three distinct individuals
        candidates = list(range(len(population)))
        candidates.remove(i)
        a, b, c = random.sample(candidates, 3)
        
        # Mutation
        mutant = population[a] + F * (population[b] - population[c])
        
        # Crossover
        crossover_mask = np.random.rand(len(population[i])) < CR
        trial = np.where(crossover_mask, mutant, population[i])
        
        new_population.append(trial)
        
    return new_population

def gradient_descent_refinement(h, dx, iterations=50, step_size=0.01, epsilon=1e-4):
    h = h.copy()
    for _ in range(iterations):
        current_c5 = evaluate_c5(h, dx)
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, len(h))
            c5_minus = evaluate_c5(h_minus, dx)
            
            gradient[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        h = h - step_size * gradient
        h = project_to_feasible(h, len(h))
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    F = 0.7
    CR = 0.9
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create default starting individual
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals to population to maintain diversity
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Differential evolution step
        new_population = differential_evolution_step(population, fitness, F, CR)
        
        # Keep top individuals and add new ones
        new_population = population[:population_size//2] + new_population[population_size//2:]
        
        # Apply feasibility projection
        for i in range(len(new_population)):
            new_population[i] = project_to_feasible(new_population[i], n_points)
        
        # Replace population
        population = new_population
        
        # Apply gradient descent refinement every 50 generations
        if gen % 50 == 0 and gen > 0:
            refined_solution = gradient_descent_refinement(best_solution, dx)
            refined_c5 = evaluate_c5(refined_solution, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_solution = refined_solution.copy()
    
    return (best_solution.tolist(), best_c5, n_points)
