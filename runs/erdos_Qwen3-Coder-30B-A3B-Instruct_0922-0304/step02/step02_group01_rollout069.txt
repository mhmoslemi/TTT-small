import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer settings
    max_iter = 1000
    patience = 100
    patience_counter = 0
    last_improvement = 0
    
    # Precompute shifted versions for gradient computation
    def compute_shifted_h(h_vals, dx):
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i] = np.roll(h_vals, i)
        return shifted_h

    # Gradient-based optimization using CVXPY
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = iteration
        else:
            patience_counter += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if patience_counter > patience:
            break
        
        # Try a local search using convex relaxation
        h_cvx = Variable(n_points)
        constraints = [0 <= h_cvx, h_cvx <= 1, cp_sum(h_cvx) == n_points / 2]
        
        # Convex approximation of the maximum overlap
        # Approximation using auxiliary variables for squared magnitude
        fft_h = cp.fft(h_cvx)
        mag_squared = cp.abs(fft_h)**2
        # For each shift k, the overlap is 1 - int h(x)h(x+k) dx
        # Approximate max by using a convex relaxation
        objective = Minimize(cp.max(mag_squared))
        
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_cvx.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new.copy()
            else:
                break
        except:
            break
            
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
