```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import log

def evaluate_erdos_solution(h):
    # Compute the overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Scale to make sum equal to n_points/2
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_shift):
    # For shift k, compute gradient for overlap_k
    # shift h to get h_shifted
    h_shifted = np.roll(h, k_shift)
    # gradient = (1 - h_shifted) - h_shifted = 1 - 2 * h_shifted
    gradient = 1 - 2 * h_shifted
    return gradient

def line_search(h, grad, dx, n_points):
    # Try various step sizes
    step_sizes = [0.1, 0.5, 1.0]
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for alpha in step_sizes:
        h_candidate = h - alpha * grad
        h_candidate = project_feasible(h_candidate, n_points, dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < best_c5:
            best_h = h_candidate
            best_c5 = candidate_c5
    return best_h, best_c5

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Convex relaxation
    h_var = cp.Variable(n_points)
    # Objective: min overlap + lambda * sum(h * log(h))
    # For the convex relaxation, approximate overlap with gradient descent
    # This is a simplification.
    # For k=z, overlap = sum(h(i) * (1-h(i+z)))
    # Also minimize entropy regularization
    entropy_reg = cp.sum(cp.multiply(h_var, cp.log(h_var + 1e-12)))
    
    # Constraint h >= 0, h <= 1, sum(h) = n_points/2
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
    
    # The objective function will be an approximation based on current h
    # Use linear approximation for simplicity in this relaxed version
    
    # Simple approach: just use entropy regularization to respect constraints
    objective = cp.Minimize(-entropy_reg + lambda_reg * cp.sum(h_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_relaxed = h_var.value
            h_relaxed = np.clip(h_relaxed, 0, 1)
            # Re-scale appropriately
            h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
            return h_relaxed
    except:
        pass
    return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
    else:
        # Create a simple starting vector if no initial construction
        h = np.ones(n_points) * 0.5
        h = h * (n_points/2) / np.sum(h)
    
    # Current best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate current overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap_index = np.argmax(np.abs(overlaps))
        # The k shift associated with this max overlap
        k_shift = max_overlap_index - (len(overlaps) - 1) // 2
        
        # Compute gradient
        gradient = compute_gradient(h, k_shift)
        
        # Perform line search
        h_new, c5_new = line_search(h, gradient, dx, n_points)
        
        # Update if improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update h
        h = h_new.copy()
        
        # Periodically solve convex relaxation to escape local minima
        if iteration % 20 == 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx)
            c5_relaxed = evaluate_erdos_solution(h_relaxed)
            if c5_relaxed < best_c5:
                best_h = h_relaxed.copy()
                best_c5 = c5_relaxed
                h = h_relaxed.copy()
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
```
