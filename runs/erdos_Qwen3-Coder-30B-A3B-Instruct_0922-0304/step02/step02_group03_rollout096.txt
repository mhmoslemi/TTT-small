import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with coarse discretization for rapid exploration
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize from given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = enforce_feasibility(h, n_points)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = enforce_feasibility(h, n_points)
    
    # Objective function for optimization
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Constraint function (sum must equal n_points/2)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimize with L-BFGS-B
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=[(0, 1) for _ in range(n_points)], 
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': 1000, 'ftol': 1e-6})
    
    h_opt = res.x
    h_opt = enforce_feasibility(h_opt, n_points)
    c5 = evaluate_overlap(h_opt, dx)
    
    # Upgrade to fine discretization for better solution
    if time.time() - start_time < budget_s - 2:
        n_points = 70
        dx = 2.0 / n_points
        h = enforce_feasibility(h_opt, n_points)
        
        # Evaluate overlap with fine grid
        c5_prev = c5
        
        # Gradient-based optimization with finer resolution
        def objective_fine(h):
            return evaluate_overlap(h, dx)
        
        def constraint_fine(h):
            return np.sum(h) - n_points / 2
        
        # Try with different method if still room
        res_fine = minimize(objective_fine, h, method='L-BFGS-B', 
                            bounds=[(0, 1) for _ in range(n_points)], 
                            constraints={'type': 'eq', 'fun': constraint_fine},
                            options={'maxiter': 500, 'ftol': 1e-6})
        
        h_fine = res_fine.x
        h_fine = enforce_feasibility(h_fine, n_points)
        c5_fine = evaluate_overlap(h_fine, dx)
        
        # Take the better of the two
        if c5_fine < c5_prev:
            h_opt = h_fine
            c5 = c5_fine
    
    return (h_opt.tolist(), c5, n_points)
