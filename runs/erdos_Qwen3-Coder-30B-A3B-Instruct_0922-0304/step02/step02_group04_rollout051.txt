import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute gradient of C5 with respect to h using finite differences
    eps = 1e-4
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def particle_swarm_optimization(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Add random particles
    while len(swarm) < n_particles:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Initialize personal bests and global best
    personal_bests = [swarm[i].copy() for i in range(n_particles)]
    personal_best_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_bests[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate current particle
            current_score = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_bests[i] = swarm[i].copy()
                personal_best_scores[i] = current_score
                
                # Update global best
                if current_score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = current_score
                    
        # Local refinement every 20 iterations
        if iteration % 20 == 0 and iteration > 0:
            # Apply gradient ascent to top 10% particles
            top_indices = np.argsort(personal_best_scores)[:n_particles//10]
            for idx in top_indices:
                h = personal_bests[idx].copy()
                step_size = 0.01
                for _ in range(50):
                    grad = compute_gradient(h, dx)
                    h = h - step_size * grad
                    h = project_to_feasible(h, n_points)
                # Update personal best if improved
                new_score = evaluate_c5(h, dx)
                if new_score < personal_best_scores[idx]:
                    personal_bests[idx] = h
                    personal_best_scores[idx] = new_score
                    if new_score < global_best_score:
                        global_best = h
                        global_best_score = new_score
                        
        # Early exit if target reached
        if global_best_score <= 0.3807:
            break
    
    return global_best, global_best_score

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = particle_swarm_optimization(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
