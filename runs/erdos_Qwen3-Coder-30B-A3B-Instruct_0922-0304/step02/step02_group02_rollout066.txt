import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Ensure h is within bounds and sum equals target_sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # This ensures sum(h) * dx = 1

    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a symmetric pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO Parameters
    n_particles = 50
    n_iterations = 500
    omega = 0.7
    c1 = c2 = 2.0
    max_vel = 0.1
    
    # Initialize PSO swarm
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        # Start with modified initial_h_values + noise
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            noise = np.random.normal(0, 0.1 * np.max(h_init), n_points)
            h_particle = h_init + noise
        else:
            h_particle = np.ones(n_points)
            h_particle[n_points//2:] = 0
            h_particle = h_particle * (n_points / 2) / np.sum(h_particle)
            noise = np.random.normal(0, 0.1 * np.max(h_particle), n_points)
            h_particle += noise
            
        h_particle = repair_h(h_particle, target_sum)
        particles.append(h_particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h_particle.copy())
        pbest_c5.append(evaluate_c5(h_particle, dx))
    
    gbest_c5 = min(pbest_c5)
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    
    # PSO main loop
    for iter_ in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair
            particles[i] = repair_h(particles[i], target_sum)
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
            
            # Update global best
            if current_c5 < gbest_c5:
                gbest_c5 = current_c5
                gbest = particles[i].copy()
        
        # Early exit
        if gbest_c5 <= 0.3807:
            break
    
    # Local refinement of top particles using gradient descent
    sorted_indices = np.argsort(pbest_c5)
    top_k = max(1, n_particles // 20)  # Top 5%
    
    for i in range(top_k):
        idx = sorted_indices[i]
        h_refine = pbest[idx].copy()
        for _ in range(30):  # Limited iterations
            if time.time() - start_time > budget_s - 1:
                break
            # Gradient approximation via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            base_c5 = evaluate_c5(h_refine, dx)
            
            for j in range(n_points):
                h_plus = h_refine.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, target_sum)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
                
            # Update
            h_refine -= 0.01 * grad
            h_refine = repair_h(h_refine, target_sum)
            
            new_c5 = evaluate_c5(h_refine, dx)
            if new_c5 < pbest_c5[idx]:
                pbest_c5[idx] = new_c5
                pbest[idx] = h_refine.copy()
        
        # Update global best
        if pbest_c5[idx] < gbest_c5:
            gbest_c5 = pbest_c5[idx]
            gbest = pbest[idx].copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(gbest, dx)
    
    # If the final is better than the best, update
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = gbest.copy()
        
    return (best_h.tolist(), best_c5, n_points)
