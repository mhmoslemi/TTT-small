import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # Clip to [0,1] and rescale to sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Generate initial solution
        h = np.random.rand(n_points)
    
    h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    iteration = 0
    relax_freq = 20
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx

        # Compute gradient for this shift
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search for step size
        step_sizes = [0.1, 0.5, 1.0]
        best_h_candidate = h
        best_candidate_c5 = best_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_c5(h_candidate, dx)
            
            if candidate_c5 < best_candidate_c5:
                best_h_candidate = h_candidate
                best_candidate_c5 = candidate_c5
                break
        
        # Update if improvement
        if best_candidate_c5 < best_c5:
            h = best_h_candidate
            best_c5 = best_candidate_c5
            
            if best_c5 < best_c5:
                best_h = h.copy()
                
        # Periodically solve convex relaxation
        if iteration % relax_freq == 0:
            # Create relaxed problem
            h_relax = cp.Variable(n_points)
            obj = cp.Maximize(cp.sum(cp.multiply(h_relax, 1 - h_relax)))
            constraints = [
                cp.sum(h_relax) == n_points/2,
                h_relax >= 0,
                h_relax <= 1
            ]
            prob = cp.Problem(cp.Maximize(obj), constraints)
            
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                if h_relax.value is not None:
                    h_new = np.clip(h_relax.value, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    h = h_new
                    current_c5 = evaluate_c5(h, dx)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
            except:
                pass  # Continue with current solution if optimization fails
    
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
