import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # == 35
    
    # Initialize swarm
    P = 50  # swarm size
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        # Simple initialization
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    # Create initial population with noise on base_h
    for _ in range(P):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, target_sum)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 2.0
    max_iter = 500
    
    # Initialize global best
    best_c5 = float('inf')
    best_h = None
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        particles_c5 = []
        for i, h in enumerate(swarm):
            c5 = evaluate_c5(h, dx)
            particles_c5.append(c5)
            
            # Update personal best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Find global best index
        global_best_idx = np.argmin(particles_c5)

        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            if i == global_best_idx:
                # No change if already best
                pass
            else:
                # Update velocity
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (swarm[i] - swarm[i]) + 
                                c2 * r2 * (swarm[global_best_idx] - swarm[i]))
                
                # Update position
                swarm[i] = swarm[i] + velocities[i]
                
                # Repair to feasible space
                swarm[i] = repair_h(swarm[i], target_sum)
    
    # Refinement with local search on top particles
    num_refine = max(1, P // 20)  # Top 5%
    sorted_idx = np.argsort(particles_c5)
    
    for i in range(num_refine):
        idx = sorted_idx[i]
        h_refine = swarm[idx].copy()
        
        # Gradient descent on top results
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            eps = 1e-4
            grad = np.zeros_like(h_refine)
            base_c5 = evaluate_c5(h_refine, dx)
            
            for j in range(n_points):
                h_plus = h_refine.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, target_sum)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
                
            # Update
            h_refine = h_refine - 0.01 * grad
            h_refine = repair_h(h_refine, target_sum)
            
            # Re-evaluate
            current_c5 = evaluate_c5(h_refine, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refine.copy()
        
        if best_c5 <= 0.3807:
            break
            
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
