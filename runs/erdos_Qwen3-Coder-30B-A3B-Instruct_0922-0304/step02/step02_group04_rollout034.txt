import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize as rectangular pulse
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Add noise to initial solution
    noise = np.random.normal(0, 0.05, n_points)
    h = h + noise
    h = project_to_feasible(h, n_points)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temporal annealing parameters
    initial_temp = 0.1 * current_c5
    temperature = initial_temp
    cooling_rate = 0.999
    
    # Gradient descent parameters
    grad_step = 0.01
    grad_decay = 0.9
    grad_steps = 5
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        
        # Generate candidate by perturbing current solution
        candidate = h + np.random.normal(0, 0.01, n_points)
        candidate = project_to_feasible(candidate, n_points)
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject based on simulated annealing
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            temperature *= cooling_rate
        else:
            # Accept with probability
            delta = candidate_c5 - current_c5
            if random.random() < np.exp(-delta / temperature):
                h = candidate
                current_c5 = candidate_c5
            temperature *= cooling_rate
        
        # Periodically refine with gradient descent
        if iteration % 50 == 0:
            for _ in range(grad_steps):
                # Compute gradient via finite differences
                eps = 1e-4
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    c5_minus = evaluate_c5(h_minus, dx)
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                # Update with gradient step
                h = h - grad_step * grad
                h = project_to_feasible(h, n_points)
                
                # Update temperature and step size
                grad_step *= grad_decay
        
        iteration += 1
        
        # Early exit if target hit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
