import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    overlaps = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlaps) * dx
    return c5

def project_feasible(h, n_points):
    """Ensure h is in [0,1] and sums to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def solve_convex_relaxation(h, n_points, lambda_reg=0.01):
    """Solve convex relaxation to escape local minima."""
    h0 = h.copy()
    h_cvx = cp.Variable(n_points)
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
    
    # Compute correlation matrix and find max correlation
    corr_matrix = np.zeros((n_points, n_points))
    for i in range(n_points):
        for j in range(n_points):
            k = (j - i) % n_points
            corr_matrix[i, j] = h0[i] * (1 - h0[(i + k) % n_points])
    
    # Regularized term
    entropic = cp.sum(cp.entr(h_cvx))  # log(h) term
    objective = cp.Minimize(cp.sum(corr_matrix @ h_cvx) - lambda_reg * entropic)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return np.array(h_cvx.value).flatten()
    except:
        pass
    return h0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():

        h = project_feasible(h, n_points)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Gradient descent with convex relaxation
    iteration = 0
    step_size = 1e-3
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate overlap
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Gradients via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, n_points)
            overlap_plus = np.max(correlate(h_plus, 1 - h_plus, mode="full"))
            grad[i] = (overlap_plus - max_overlap) / eps
        
        # Test line search
        found_improvement = False
        step_sizes = [1e-3, 0.5e-3, 1e-2]
        for alpha in step_sizes:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate, n_points)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < best_c5:
                h = h_candidate
                best_c5 = c5_candidate
                best_h = h.copy()
                found_improvement = True
                break
        
        # Convex relaxation every 50 iterations
        if iteration % 50 == 0:
            h_relax = solve_convex_relaxation(h, n_points, lambda_reg)
            c5_relax = evaluate_erdos_solution(h_relax)
            if c5_relax < best_c5:

                best_c5 = c5_relax
                best_h = h.copy()
            else:
                lambda_reg *= 0.5
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    best_h = project_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h.tolist())
    
    return (best_h.tolist(), final_c5, n_points)
