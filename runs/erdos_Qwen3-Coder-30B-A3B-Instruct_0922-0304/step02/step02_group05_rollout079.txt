import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, n_points, dx):
    # Create shifted versions for gradient computation
    h_shifted = np.zeros(n_points)
    h_shifted[k_max:] = h[:-k_max]
    h_shifted[:k_max] = h[-k_max:]
    
    # Gradient estimate: (1 - h)_{shifted} - h_{shifted}
    grad = (1 - h_shifted) - h_shifted
    return grad

def line_search(h, grad, n_points, dx, max_alpha=1.0):
    alpha = max_alpha
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Try several alpha values
    for _ in range(10):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points, dx)
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        alpha *= 0.5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a simple construction
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_size = 0.01
    relaxation_interval = 50
    
    step_count = 0
    while time.time() - start_time < budget_s - 1:
        step_count += 1
        
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = np.max(overlap) * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute approximate gradient
        grad = compute_gradient(h, k_max, n_points, dx)
        
        # Line search for optimal step size
        h_new, c5_new = line_search(h, grad, n_points, dx, step_size)
        
        h = h_new
        current_c5 = c5_new
        
        # Convex relaxation every few steps
        if step_count % relaxation_interval == 0:
            # Add regularization
            reg_strength = 0.001
            # Simple entropy regularization
            entropy_reg = np.log(h + 1e-10) + np.log(1 - h + 1e-10)
            entropy_reg = reg_strength * entropy_reg
            
            # Apply regularization to gradient
            current_grad = compute_gradient(h, k_max, n_points, dx)
            regularized_grad = current_grad + entropy_reg
            
            h = h - step_size * regularized_grad
            h = project_feasible(h, n_points, dx)
            current_c5 = evaluate_c5(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Reset step size after relaxation
            step_size = 0.01
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
