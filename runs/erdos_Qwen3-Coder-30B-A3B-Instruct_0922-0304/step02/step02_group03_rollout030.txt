import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Create initial particles
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * target_sum / np.sum(h_base)
        particles.append(h_base.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Fallback initialization
        h_base = np.ones(n_points)
        h_base[n_points//2:] = 0
        particles.append(h_base.copy())
        velocities.append(np.zeros(n_points))
    
    # Create diverse initial particles with block structures
    for i in range(1, n_particles):
        # Start with blocky pattern
        h = np.zeros(n_points)
        block_len = n_points // 3
        h[:block_len] = 1.0
        h[block_len:2*block_len] = 0.0
        h[2*block_len:] = 1.0
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Initialize bests
    best_individual = [p.copy() for p in particles]
    best_individual_c5 = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_individual_c5)
    global_best = particles[global_best_idx].copy()
    global_best_c5 = best_individual_c5[global_best_idx]
    
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        # Update inertia
        inertia = max(0.3, inertia * (1 - inertia_decay))
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (
                inertia * velocities[i] + 
                alpha * r1 * (best_individual[i] - particles[i]) + 
                beta * r2 * (global_best - particles[i])
            )
            # Clip velocity
            vel_update = np.clip(vel_update, -max_velocity, max_velocity)
            velocities[i] = vel_update
            
            # Update particle position
            new_pos = particles[i] + velocities[i]
            
            # Project to feasible region
            new_pos = project_to_feasible(new_pos, target_sum)
            
            particles[i] = new_pos
            
            # Evaluate new position
            c5 = evaluate_c5(new_pos, dx)
            
            # Update personal best
            if c5 < best_individual_c5[i]:
                best_individual[i] = new_pos.copy()
                best_individual_c5[i] = c5
                
                # Update global best
                if c5 < global_best_c5:
                    global_best = new_pos.copy()
                    global_best_c5 = c5
                    
        # Early stopping condition
        if global_best_c5 <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_c5, n_points)
