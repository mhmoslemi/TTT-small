import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import math

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
        
    return grad

def sqp_step(h, dx, max_iter=50, tol=1e-6):
    n = len(h)
    h = h.copy()
    x_prev = h.copy()
    dx = dx
    
    # Initialize BFGS approximation of Hessian
    H = np.eye(n)
    
    for i in range(max_iter):
        # Compute gradient
        grad = compute_gradient(x_prev, dx)
        
        # Check convergence
        if np.linalg.norm(grad) < tol:
            break
            
        # Compute search direction: H * d = -grad
        try:
            d = -np.linalg.solve(H, grad)
        except:
            d = -grad
            
        # Line search
        alpha = 1.0
        max_alpha = 10.0
        c1 = 1e-4
        c5_prev = evaluate_c5(x_prev, dx)
        
        while alpha > 1e-10:
            x_new = x_prev + alpha * d
            x_new = project_to_feasible(x_new, n)
            c5_new = evaluate_c5(x_new, dx)
            
            # Armijo condition
            if c5_new <= c5_prev + c1 * alpha * np.dot(grad, d):
                break
            alpha *= 0.5
            
        # Update h
        h = x_new.copy()
        
        # Update Hessian using BFGS
        if i > 0:
            s = h - x_prev
            y = compute_gradient(h, dx) - compute_gradient(x_prev, dx)
            
            if np.dot(s, y) > 1e-10:
                rho = 1.0 / np.dot(s, y)
                I = np.eye(n)
                H = (I - rho * np.outer(s, y)) @ H @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
        
        x_prev = h.copy()
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if it exists

    else:

        h = project_to_feasible(h, n_points)
    
    # SQP optimization
    h = sqp_step(h, dx, max_iter=500)
    c5_bound = evaluate_c5(h, dx)
    
    # Ensure final feasibility
    h = project_to_feasible(h, n_points)
    c5_bound = evaluate_c5(h, dx)
    
    # Early exit if we meet target
    if c5_bound <= 0.3807:
        return (h.tolist(), c5_bound, n_points)
    
    # Refine for a bit more with additional local search
    for _ in range(20):
        if time.time() - start_time > budget_s - 5:
            break
        h = sqp_step(h, dx, max_iter=50)
        c5_bound = evaluate_c5(h, dx)
        if c5_bound <= 0.3807:
            break
    
    return (h.tolist(), c5_bound, n_points)
