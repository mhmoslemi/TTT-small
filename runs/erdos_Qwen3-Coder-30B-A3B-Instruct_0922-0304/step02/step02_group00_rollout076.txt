import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, multiply
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create sawtooth-like initial guess
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    no_improvement_count = 0
    patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > patience:
            # If no improvement for a while, try a different init
            h = np.linspace(0, 1, n_points)
            h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            continue
            
        # Convex optimization to reduce overlap at k_max
        h_var = Variable(n_points)
        
        # Create a linearized version of the overlap at shift k_max
        # This approximates the overlap at shift k_max as a linear function of h
        # Linear term: sum_i h[i] * (1 - h[i + k_max])
        # The current values of h are fixed. 
        # We approximate the overlap as a linear function of h.
        
        # Compute the indices involved in this overlap
        indices = np.arange(n_points)
        shifted_indices = (indices + k_max) % n_points
        
        # Create a simple linear objective based on the current shift
        # This is the linear approximation to the overlap score around current h
        # It's a linear function with fixed coefficients
        objective = 0
        for i in range(n_points):
            objective += h_var[i] * (1 - h[shifted_indices[i]])
            
        # We minimize this linear objective with constraints
        # ensuring h meets the sum and bound constraints
        constraints = [cp_sum(h_var) == n_points / 2,
                       h_var >= 0,
                       h_var <= 1]
        
        # Create and solve the problem 
        try:
            prob = Problem(Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
                h = h_new
            else:
                # Fallback to a gradient approximation if not optimal
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
        except Exception:
            # Fallback if CVXPY fails
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            
        # Maintain sum constraint exactly
        h = h * (n_points / 2) / np.sum(h)
        
        # Ensure feasible domain
        h = np.clip(h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
