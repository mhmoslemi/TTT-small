import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Coarse optimization (n=35)
    n_points_coarse = 35
    dx_coarse = 2.0 / n_points_coarse
    
    # Initialize with two blocks
    h_coarse = np.ones(n_points_coarse)
    h_coarse[n_points_coarse//2:] = 0
    h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
    
    # Ensure bounds and integral constraint
    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
    
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    # Use L-BFGS-B for faster convergence
    bounds = [(0, 1) for _ in range(n_points_coarse)]
    
    def obj(h_vec):
        h = h_vec.copy()
        h = h * (n_points_coarse / 2) / np.sum(h)
        return evaluate_c5(h, dx_coarse)
    
    def constr(h_vec):
        return np.sum(h_vec) - n_points_coarse / 2
    
    cons = {'type': 'eq', 'fun': constr}
    
    # Optimize with bounds and constraint
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        res = minimize(obj, h_coarse, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                       options={'ftol': 1e-9, 'gtol': 1e-5})
        if res.success:
            h_coarse = res.x
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
            current_c5 = evaluate_c5(h_coarse, dx_coarse)
            if current_c5 < best_c5_coarse:
                best_c5_coarse = current_c5
                best_h_coarse = h_coarse.copy()
        else:
            break
    
    # Stage 2: Refinement (n=70)
    n_points = 70
    dx = 2.0 / n_points
    
    # Upsample coarse solution
    h_refined = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, n_points_coarse), 
                          best_h_coarse)
    
    # Add small random perturbations
    perturbation = 0.1 * (1 - 0)
    noise = np.random.uniform(-perturbation, perturbation, n_points)
    h_refined = h_refined + noise
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    # Maintain feasibility and optimize
    best_c5 = float('inf')
    best_h = h_refined.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj_fine(h_vec):
        h = h_vec.copy()
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    def constr_fine(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons_fine = {'type': 'eq', 'fun': constr_fine}
    
    # More iterations for refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        res = minimize(obj_fine, h_refined, method='L-BFGS-B', bounds=bounds, constraints=cons_fine, 
                       options={'ftol': 1e-9, 'gtol': 1e-5})
        if res.success:
            h_refined = res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            current_c5 = evaluate_c5(h_refined, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined.copy()
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
