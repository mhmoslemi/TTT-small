import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Initialize with given construction if available

    else:

        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure initial feasibility
    h = enforce_feasibility(h, n_points)
    
    # Define objective and constraints
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint_fun(h):
        return np.sum(h) - n_points / 2
    
    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_fun}
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use L-BFGS-B with bounds and constraint
    try:
        res = minimize(objective, h, method='L-BFGS-B', 
                       bounds=bounds, 
                       constraints=constraint,
                       options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6})
        
        if res.success:
            h_opt = res.x
            h_opt = enforce_feasibility(h_opt, n_points)
            c5_opt = evaluate_overlap(h_opt, dx)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Refine if needed
    if time.time() - start_time < budget_s - 1:
        # Try a simple gradient-based refinement
        h_refined = best_h.copy()
        step_size = 0.01
        max_iter = 1000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Finite differences to approximate gradient
            grad = np.zeros(n_points)
            eps = 1e-5
            base_c5 = evaluate_overlap(h_refined, dx)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = evaluate_overlap(h_plus, dx)
                grad[j] = (c5_plus - base_c5) / eps
            
            # Update
            h_new = h_refined - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_overlap(h_new, dx)
            if new_c5 < base_c5:
                h_refined = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_refined.copy()
            else:
                step_size *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
