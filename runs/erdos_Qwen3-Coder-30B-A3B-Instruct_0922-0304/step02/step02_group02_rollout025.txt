import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with symmetric construction
    h = np.ones(n_points)
    # Create 5-fold symmetric pattern with alternating high/low values
    segment_size = n_points // 5
    for i in range(5):
        start = i * segment_size
        end = start + segment_size
        if i % 2 == 0:
            h[start:end] = 0.6
        else:
            h[start:end] = 0.4
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h_vec):
        # Normalize h_vec to satisfy sum constraint
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        h_norm = np.clip(h_norm, 0, 1)
        return evaluate_c5(h_norm, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define optimization problem
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization using L-BFGS-B with bounds
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with L-BFGS-B
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-6})
        
        if result.success:
            h = result.x
        else:
            # Fallback to gradient-based approach
            h = h - 0.001 * np.gradient(h)
        
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
