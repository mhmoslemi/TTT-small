```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    """Solve a convex relaxation to escape local minima"""
    h_relax = cp.Variable(n_points)
    h_values = np.array(h)
    
    # Convex relaxation: minimize overlap + regularization
    # Using first-order approximation of the non-convex term
    # We approximate overlap using linear terms
    # Simplified: regularize entropy and maintain constraints
    
    # Create a proper convex objective (simplified version)
    # Minimize: overlap + lambda_reg * sum(h * log(h))
    # Since we don't have true gradient, we'll just solve a simple convex problem

    # For now, use: minimize sum(h_i * log(h_i)) + lambda_reg * sum(h_i)
    # Subject to sum(h) = n_points/2, h in [0,1]
    
    # We'll use this approximation: minimize sum(h * log(h)) + lambda_reg * sum(h)
    objective = cp.Minimize(cp.sum(cp.entr(h_relax)) + lambda_reg * cp.sum(h_relax))
    constraints = [
        cp.sum(h_relax) == n_points/2,
        h_relax >= 0,
        h_relax <= 1
    ]
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.SCS, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return np.array(h_relax.value)
    except:
        pass
    # Fallback to original
    return h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with either initial_h_values or a random construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    current_c5 = best_c5
    
    lambda_reg = 0.01
    iteration = 0
    max_iterations = 5000  # Prevent infinite loops in case nothing works
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute overlaps for all lags
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        max_idx = np.argmax(overlaps)
        
        # Compute approximate gradient for the max overlap shift
        # Overlap at k = sum_{i} h[i] * (1 - h[(i+k)%n])
        # Gradient w.r.t. h[i] is (1 - h[(i+k)%n]) - h[(i+k)%n] = 1 - 2*h[(i+k)%n]
        k = max_idx - (n_points - 1)  # Convert full indexing to shift
        
        # Rolling the array to get h shifted by k from the beginning
        h_shifted = np.roll(h, k)
        gradient = h_shifted - (1 - h_shifted)  # This is d/dh overlap_k
        
        # Simple line search
        step_sizes = [0.1, 0.5, 1.0]
        best_step = 0
        for step in step_sizes:
            h_candidate = h - step * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < current_c5:
                current_c5 = candidate_c5
                h = h_candidate
                best_step = step
                break
        
        # Periodically solve convex relaxation
        if iteration % 20 == 0:
            try:
                h_relax = solve_convex_relaxation(h, n_points, dx, lambda_reg)
                h_relax = project_feasible(h_relax, n_points, dx)
                c5_relax = evaluate_erdos_solution(h_relax)
                if c5_relax < current_c5:
                    h = h_relax
                    current_c5 = c5_relax
                    # Adjust regularization if needed
                    if lambda_reg > 0.001:
                        lambda_reg *= 0.5
                else:
                    lambda_reg *= 1.1
            except:
                pass
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit the target
        if best_c5 <= 0.3807:
            break
    
    # Final recheck
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
```
