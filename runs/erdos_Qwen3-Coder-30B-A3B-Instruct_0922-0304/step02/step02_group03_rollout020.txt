import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    else:

    dx = 2.0 / n_points
    
    # Ensure initial feasibility
    h = enforce_feasibility(h, n_points)
    
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using L-BFGS-B
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=[(0, 1) for _ in range(n_points)], 
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': 1000, 'ftol': 1e-6})
    
    best_h = res.x
    best_h = enforce_feasibility(best_h, n_points)
    c5 = evaluate_overlap(best_h, dx)
    
    # Final check and refinement
    if time.time() - start_time < budget_s - 1:
        # Try a simple local search refinement
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb and check
            perturbation = np.random.normal(0, 0.01, n_points)
            h_try = best_h + perturbation
            h_try = enforce_feasibility(h_try, n_points)
            c5_try = evaluate_overlap(h_try, dx)
            if c5_try < c5:
                best_h = h_try
                c5 = c5_try
    
    return (best_h.tolist(), c5, n_points)
