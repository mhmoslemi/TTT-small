import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use convex relaxation with homotopy continuation
    mu_schedule = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    learning_rate = 0.1
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define optimization variables
        h_var = cp.Variable(n_points)
        
        # Objective: weighted combination of C5 and regularizer
        # For the overlap, we compute it directly in the convex framework
        # This simplified version uses gradient descent with projected updates
        max_overlap = 0
        
        # Project h to ensure sum equals 35 and bounds are satisfied
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Local search using gradient descent
        max_iters = 30
        for _ in range(max_iters):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current value
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Estimate gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            # Update with momentum and projection
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # If no improvement from this mu step, try faster decay
        if current_c5 >= best_c5:
            learning_rate = max(0.01, learning_rate * 0.9)
    
    # Final validation and adjustment  
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h, dx)
    
    # If result is slightly worse, run final fine-tuning
    if final_c5 > best_c5:
        final_h = best_h.copy()
        final_c5 = best_c5
    
    return (final_h.tolist(), final_c5, n_points)
