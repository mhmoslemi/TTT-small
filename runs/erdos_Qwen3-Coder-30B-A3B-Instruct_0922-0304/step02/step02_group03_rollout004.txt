import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Coarse optimization (n=35)
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize with two uniform blocks
    h = np.ones(n_points)
    h[n_points//2:] = 0
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Callback function for tracking progress
    def callback(x):
        nonlocal best_c5, best_h
        if time.time() - start_time > budget_s - 1:
            return True
        h_test = x.copy()
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5 = evaluate_c5(h_test, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
        return False

    # Optimization parameters
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
    
    # L-BFGS-B optimization for coarse phase
    options = {'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-5}
    
    try:
        res = minimize(
            lambda x: evaluate_c5(np.clip(x, 0, 1) * (n_points / 2) / np.sum(np.clip(x, 0, 1)), dx),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            callback=callback,
            options=options,
            tol=1e-9
        )
        if not res.success:
            h = best_h
        else:
            h = np.clip(res.x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        h = best_h
    
    # Stage 2: Refinement (n=70)
    n_points = 70
    dx = 2.0 / n_points
    
    # Upsample from coarse solution
    h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, 35), h)
    
    # Perturb with small noise
    perturbation = 0.1 * (np.random.rand(n_points) - 0.5)
    h = np.clip(h_upsampled + perturbation, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Adjust for refinement phase
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]

    # Optimization for refined phase
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-5}
    
    try:
        res = minimize(
            lambda x: evaluate_c5(np.clip(x, 0, 1) * (n_points / 2) / np.sum(np.clip(x, 0, 1)), dx),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            callback=callback,
            options=options,
            tol=1e-9
        )
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Final evaluation at best solution found
    final_c5 = evaluate_c5(h, dx)
    
    # Last safety check
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    return (h.tolist(), final_c5, n_points)
