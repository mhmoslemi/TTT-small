import numpy as np
from scipy.signal import correlate
import time
from math import floor, ceil

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraint_space(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping h in [0,1]"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale and clip
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    # Adjust to exactly match target sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find indices where we can add the difference
        # Add to smallest elements
        idx = np.argsort(h)
        i = 0
        while abs(diff) > 1e-10 and i < len(h):
            if h[idx[i]] < 1:
                add = min(diff, 1 - h[idx[i]])
                h[idx[i]] += add
                diff -= add
            i += 1
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.1):
    """Apply Gaussian mutation with scaling to maintain constraints"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, sigma)
    return project_to_constraint_space(h, len(h) / 2)

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    if n <= 1:
        return parent1.copy()
    point = np.random.randint(1, n)
    child = np.concatenate([parent1[:point], parent2[point:]])
    return child

def select_parents(population, fitnesses, num_parents):
    """Tournament selection"""
    parents = []
    for _ in range(num_parents):
        tournament_size = 3
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        parents.append(population[winner_idx].copy())
    return parents

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraint_space(h, n_points / 2)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_c5(h, dx) for h in population]
    
    best_c5 = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    # Main GA loop
    no_improvement_count = 0
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection
        num_parents = population_size // 2
        parents = select_parents(population, fitnesses, num_parents)
        
        # Create new population
        new_population = [best_individual.copy()]  # Elitism
        
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate_individual(child)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Update best solution
        min_f = min(fitnesses)
        if min_f < best_c5:
            best_c5 = min_f
            best_individual = population[np.argmin(fitnesses)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if best_c5 <= 0.3807 or no_improvement_count > 100:
            break
            
    return (best_individual.tolist(), best_c5, n_points)
