import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def create_initial_particles(n_points, num_particles, initial_h_values=None):
    """Create initial particle positions"""
    particles = []
    
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    else:
        # Create random starting point
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, 2.0/n_points)
        particles.append(h)
    
    # Fill remaining particles with random feasible points
    while len(particles) < num_particles:
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, 2.0/n_points)
        particles.append(h)
    
    return particles

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    num_particles = 50
    max_iterations = int(budget_s * 5)  # Rough estimate
    
    # Create particles
    particles = create_initial_particles(n_points, num_particles, 
                                          globals().get('initial_h_values', None))
    
    # Initialize velocities
    velocities = [np.random.normal(0, 0.01, n_points) for _ in range(num_particles)]
    
    # Initialize personal bests
    p_best_positions = [p.copy() for p in particles]
    p_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_scores)
    g_best_position = p_best_positions[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO parameters
    w = 1.0  # Inertia weight
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    max_velocity = 0.2
    
    iteration = 0
    last_improvement = 0
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best_positions[i] - particles[i]) +
                           c2 * r2 * (g_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            current_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_score < p_best_scores[i]:
                p_best_scores[i] = current_score
                p_best_positions[i] = particles[i].copy()
                
                # Update global best
                if current_score < g_best_score:
                    g_best_score = current_score
                    g_best_position = particles[i].copy()
                    last_improvement = iteration
                
        # Adaptive inertia
        if iteration % 50 == 0:
            w = max(0.8, w * 0.99)
        
        # Reset stagnated particles
        if iteration - last_improvement > 100:
            idx = random.randint(0, num_particles-1)
            particles[idx] = np.random.uniform(0, 1, n_points)
            particles[idx] = project_feasible(particles[idx], n_points, dx)
            p_best_positions[idx] = particles[idx].copy()
            p_best_scores[idx] = evaluate_c5(particles[idx], dx)
            last_improvement = iteration
            
        # Early exit if target reached
        if g_best_score <= 0.3807:
            break
    
    # Return the best solution found
    return (g_best_position.tolist(), g_best_score, n_points)
