import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint while keeping values in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # equivalent to integral = 1
    
    # Initialize swarm
    n_particles = 50
    swarm = []
    velocities = []
    
    # Use initialization if available, otherwise create symmetric pattern
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        # Create a 5-fold symmetric pattern as base
        base_h = np.zeros(n_points)
        chunk_size = n_points // 5
        for i in range(5):
            start_idx = i * chunk_size
            end_idx = (i + 1) * chunk_size
            base_h[start_idx:end_idx] = 1.0 if i % 2 == 0 else 0.0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    # Initialize particles with noise
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, target_sum)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    inertia = 0.7
    c1, c2 = 2.0, 2.0
    max_iter = 500
    
    # Track best solutions
    global_best_h = None
    global_best_c5 = float('inf')
    
    # PSO iteration
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            h = swarm[i]
            current_c5 = evaluate_c5(h, dx)
            
            # Update particle best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = h.copy()
                
            # Early exit if target reached
            if global_best_c5 <= 0.3807:
                break
                
        # Local search on top 5% particles
        if iter_count % 20 == 0 and iter_count > 0:
            # Sort particles by fitness
            fitnesses = [evaluate_c5(h, dx) for h in swarm]
            sorted_indices = np.argsort(fitnesses)
            top_p = max(1, n_particles // 20)  # Top 5%
            
            for i in sorted_indices[:top_p]:
                h = swarm[i].copy()
                # Simple gradient ascent for local refinement
                eps = 1e-4
                grad = np.zeros(n_points)
                base_c5 = evaluate_c5(h, dx)
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, target_sum)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[j] = (plus_c5 - base_c5) / eps
                
                # Update with learning rate
                h = h - 0.01 * grad
                h = repair_h(h, target_sum)
                swarm[i] = h.copy()
                
        # Update swarm
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (h - h)  # Will update with particle best
            social = c2 * r2 * (global_best_h - h)
            v = inertia * v + cognitive + social
            
            # Update position
            h = h + v
            h = repair_h(h, target_sum)
            
            swarm[i] = h
            velocities[i] = v
            
    # Final optimization with gradient descent on global best
    best_h = global_best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        eps = 1e-4
        grad = np.zeros(n_points)
        base_c5 = evaluate_c5(best_h, dx)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, target_sum)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[j] = (plus_c5 - base_c5) / eps
        
        best_h = best_h - 0.005 * grad
        best_h = repair_h(best_h, target_sum)
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
