import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Clip values to [0,1] and rescale to target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # 35
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
    else:
        # Simple initialization: half 1s, half 0s (adjusted to ensure integral constraint)
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = repair_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_iterations = 500
    omega = 0.7
    c1 = c2 = 2.0
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_c5 = []
    pbest_h = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            part_h = np.array(initial_h_values)
            # Add noise
            noise_scale = 0.1 * np.max(part_h)
            part_h += np.random.normal(0, noise_scale, n_points)
            part_h = repair_h(part_h, target_sum)
        else:
            part_h = np.ones(n_points)
            part_h[n_points//2:] = 0
            part_h = repair_h(part_h, target_sum)
        particles.append(part_h)
        velocities.append(np.zeros(n_points))
        pbest_c5.append(float('inf'))
        pbest_h.append(part_h.copy())
    
    gbest_c5 = float('inf')
    gbest_h = h.copy()
    
    # PSO main loop
    for it in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Evaluate particle
            c5 = evaluate_c5(particles[i], dx)
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest_h[i] = particles[i].copy()
                
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_h = particles[i].copy()
                
            # Early exit if target reached
            if gbest_c5 <= 0.3807:
                break

        if gbest_c5 <= 0.3807:
            break
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (pbest_h[i] - particles[i]) + 
                           c2 * r2 * (gbest_h - particles[i]))
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            # Repair
            particles[i] = repair_h(particles[i], target_sum)
            
        # Apply local search on top 5% particles
        if it > 100:  # Start local search after 100 iterations
            sorted_indices = np.argsort(pbest_c5)
            n_local = max(1, n_particles // 20)  # Top 5%
            for i in range(n_local):
                idx = sorted_indices[i]
                local_h = pbest_h[idx].copy()
                # Simple gradient step
                eps = 1e-4
                grad = np.zeros(n_points)
                base_c5 = evaluate_c5(local_h, dx)
                for j in range(n_points):
                    h_plus = local_h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, target_sum)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[j] = (plus_c5 - base_c5) / eps
                # Update
                local_h = local_h - 0.01 * grad
                local_h = repair_h(local_h, target_sum)
                # Check improvement
                new_c5 = evaluate_c5(local_h, dx)
                if new_c5 < pbest_c5[idx]:
                    pbest_c5[idx] = new_c5
                    pbest_h[idx] = local_h.copy()
                    if new_c5 < gbest_c5:
                        gbest_c5 = new_c5
                        gbest_h = local_h.copy()
                        
        # Early exit
        if gbest_c5 <= 0.3807:
            break

    # Final evaluation and return
    final_c5 = evaluate_c5(gbest_h, dx)
    
    return (gbest_h.tolist(), final_c5, n_points)
