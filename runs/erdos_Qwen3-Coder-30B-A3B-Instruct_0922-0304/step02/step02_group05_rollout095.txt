import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    P = 50  # number of particles
    particles = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    else:
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
        h += np.random.normal(0, 0.05, n_points)
        particles.append(project_feasible(h, n_points, dx))
    
    # Fill up particles
    for _ in range(P - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(P)]
    
    # Evaluate initial fitness
    fitness = []
    for h in particles:
        c5 = evaluate_c5(h, dx)
        fitness.append(c5)
    
    # Initialize best positions
    p_best = particles[:]
    p_best_fitness = fitness[:]
    
    # Global best
    best_idx = np.argmin(fitness)
    g_best = particles[best_idx].copy()
    g_best_fitness = fitness[best_idx]
    
    # PSO parameters
    w_start = 1.0
    w_end = 0.8
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    perturb = 0.05
    
    # Main loop
    iteration = 0
    stagnant_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0:
            # Adjust inertia
            w = w_start - (w_start - w_end) * (iteration / 3000)
            if w < w_end:
                w = w_end
                
            # Reset stagnant particles
            if iteration > 0:
                if stagnant_count > 20:
                    for i in range(P):
                        if abs(fitness[i] - g_best_fitness) > 0.001:
                            particles[i] = np.random.uniform(0, 1, n_points)
                            particles[i] = project_feasible(particles[i], n_points, dx)
                            velocities[i] = np.random.normal(0, 0.05, n_points)
                stagnant_count = 0
        else:
            w = w_start - (w_start - w_end) * (iteration / 3000)
            if w < w_end:
                w = w_end
        
        # Update each particle
        for i in range(P):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Add random perturbation
            velocities[i] += np.random.normal(0, perturb, n_points)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            new_fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = new_fitness
                    last_improvement = iteration
                    stagnant_count = 0
            else:
                stagnant_count += 1
        
        iteration += 1
        
        # Early termination if target reached
        if g_best_fitness <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_fitness, n_points)
