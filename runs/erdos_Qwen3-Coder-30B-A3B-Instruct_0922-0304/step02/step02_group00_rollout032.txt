import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Use 25 particles around the initial solution
        for i in range(25):
            h = h_init + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
        # Use 25 random particles
        for i in range(25):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
    else:
        # All random particles
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.8  # inertia weight
    c1 = 1.2  # cognitive parameter
    c2 = 1.2  # social parameter
    mutation_rate = 0.05  # mutation rate
    pbest = [particles[i].copy() for i in range(n_particles)]
    pbest_scores = [float('inf')] * n_particles
    gbest = None
    gbest_score = float('inf')
    
    # Initialize personal bests and global best
    for i in range(n_particles):
        score = evaluate_c5(particles[i], dx)
        pbest_scores[i] = score
        if score < gbest_score:
            gbest_score = score
            gbest = particles[i].copy()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Scale inertia weight
        if iteration > 100:
            w = 0.6
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
            
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest = particles[i].copy()
                
            # Apply mutation with small probability
            if np.random.rand() < mutation_rate:
                mutation_strength = 0.05
                noise = np.random.normal(0, mutation_strength, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                score = evaluate_c5(particles[i], dx)
                
                if score < pbest_scores[i]:
                    pbest_scores[i] = score
                    pbest[i] = particles[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
        
        # Early exit condition
        if gbest_score <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
