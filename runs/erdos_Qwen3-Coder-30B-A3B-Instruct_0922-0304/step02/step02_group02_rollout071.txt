import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T0 = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 10000
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    T = T0
    iteration = 0
    
    while T > T_min and (time.time() - start_time) < budget_s - 1:
        # Perturb h
        h_new = h.copy()
        
        # Select random indices to perturb (10% of points)
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Compute sum of selected elements
        sum_selected = np.sum(h_new[indices])
        
        # Adjust selected elements and distribute change
        delta = np.random.uniform(-0.1, 0.1)
        scale = delta / n_perturb
        
        for i in indices:
            h_new[i] += scale
        
        # Redistribute to maintain sum constraint
        current_sum = np.sum(h_new)
        correction = (n_points / 2) - current_sum
        if correction != 0:
            # Distribute correction among non-selected indices
            non_selected = [i for i in range(n_points) if i not in indices]
            if len(non_selected) > 0:
                correction_per = correction / len(non_selected)
                for i in non_selected:
                    h_new[i] += correction_per
        
        # Clip values and rescale to sum = n_points / 2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Calculate new C5
        current_c5 = evaluate_c5(h, dx)
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject
        delta_E = new_c5 - current_c5
        
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature
        T = T * alpha
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
