import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    # Returns the maximum overlap value for the given h
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def project_feasible(h, n_points):
    # Project h to feasible region: [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from existing solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap and find maximum
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Approximate gradient for the maximum overlap
        # This is a rough approximation
        h_shifted = np.roll(h, k_max)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        best_candidate_h = h.copy()
        best_candidate_c5 = best_c5
        
        for step in line_search_steps:
            h_candidate = h - step * gradient
            h_candidate = project_feasible(h_candidate, n_points)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # Update with best candidate
        h = best_candidate_h
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = h.copy()
        
        # Occasionally solve convex relaxation
        if iteration % relaxation_interval == 0:
            # Solve relaxed problem to escape local minima
            try:
                h_relax = cp.Variable(n_points)
                # Objective: minimize overlap with regularization
                # We minimize the max overlap, but approximate it
                # Use a soft constraint
                over = correlate(h, 1 - h, mode="full")
                max_over = cp.max(over)
                
                # KL divergence regularization to prevent extreme values
                # We want to avoid h close to 0 or 1
                entropy_term = cp.sum(cp.entr(h_relax) + cp.entr(1 - h_relax))
                objective = cp.Minimize(max_over + lambda_reg * entropy_term)
                
                constraints = [0 <= h_relax, h_relax <= 1, cp.sum(h_relax) == n_points/2]
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_relax.value
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new.copy()
                    new_c5 = evaluate_erdos_solution(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    
                    # Decrease regularization if no improvement
                    if lambda_reg > 1e-6: 
                        lambda_reg *= 0.9
            except:
                pass  # If optimization fails, continue with current solution
                
        # Early exit if threshold is met
        if best_c5 <= 0.3807:
            break
            
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
