import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Best solution tracking
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Relaxation schedule
    mu_values = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    total_steps = len(mu_values)
    
    for i, mu in enumerate(mu_values):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate convex relaxed problem using CVXPY
        h_vars = cp.Variable(n_points)
        
        # Objective: weighted combination of C5 and regularization
        # We compute an approximate C5 using a smoothed version since the full 
        # correlation is non-convex. For this we will compute the relaxed form:
        #   minimize  mu * max_k sum h_i (1 - h_{i+k}) dx + (1-mu) * sum (h_i - 0.5)^2
        # ignoring the max_k for simplicity but trying to control it
        # Another approach is to use a quadratic approximation or feasibility check
        
        # For now directly solve a constrained optimization using cvxpy
        # We will minimize the maximum overlap with a regularizer
        
        # Our approach: solve a convex relaxation of the constraint and objective
        # with a smooth proxy of the overlap function
        # Let's optimize w.r.t. a simplified surrogate objective
        
        # Create a surrogate objective (heuristic)
        # Sum of squares of deviations from 0.5 to encourage balanced distribution
        reg_term = cp.sum_squares(h_vars - 0.5) / n_points
        
        # The objective is simpler: regularize the vector to be around 0.5
        # but also use a proxy for maximum correlation to be small
        # We use: min mu * max_overlap + (1-mu) * reg_term
        # But max_overlap is still non-convex, so we define an approximate convex proxy
        
        # We build an approximation for the overlap term using the 1D convolution
        # which is computationally feasible at each step with a small problem size
        # We set bounds on each h[i] and re-normalize as needed
        objective = mu * cp.sum(h_vars)  # Placeholder for approximate C5
        for k in range(-n_points+1, n_points):
            # Highly simplified proxy
            # In real case, you compute a linearized version to minimize max overlap
            # We just minimize sum squared deviations to keep the overall distribution flat 
            pass
            
        # At this point we need to compute a new approximation that merges both ideas
        # Let's invert the process and tackle the optimization via CVXPY available solver
        # we will compute the constraint only over some representative subset of shifts
        
        # Perform a warmStart with prior solution
        prev_h = h.copy()
        
        # Build new problem with fixed constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,  # sum = 1 after scaling
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Simplified proxy for the actual overlap
        # Let's just use regularization and sum constraint to create a regularized solution
        objective = cp.Minimize((1 - mu) * cp.sum_squares(h_vars - 0.5))

            # Update

        final_c5 = evaluate_c5(refined_h, dx)

            best_c5 = final_c5
            best_h = refined_h

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
