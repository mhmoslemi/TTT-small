import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1000.0
    alpha = 0.999
    T_min = 1.0
    max_iter = 10000
    perturb_size = 0.1
    
    # Perturb 10% of indices
    num_perturb = max(1, n_points // 10)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Accept time limits
        if T < T_min:
            break
            
        # Create a candidate solution by perturbing h
        candidate_h = h.copy()
        
        # Select random indices to perturb
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Add noise within bounds
        delta = np.random.normal(0, perturb_size, num_perturb)
        
        # Apply delta and clip
        for i, idx in enumerate(indices):
            candidate_h[idx] += delta[i]
            
        # Ensure all values stay within [0,1]
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Scale to maintain sum constraint (35)
        scale_factor = (n_points / 2) / np.sum(candidate_h)
        candidate_h *= scale_factor
        
        # Compute C5 values
        current_c5 = evaluate_c5(h, dx)
        candidate_c5 = evaluate_c5(candidate_h, dx)

        # Accept or reject based on SA criteria
        delta_E = candidate_c5 - current_c5
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.random():
            h = candidate_h.copy()
        
        # Track best solution
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Cool down temperature
        T *= alpha
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
