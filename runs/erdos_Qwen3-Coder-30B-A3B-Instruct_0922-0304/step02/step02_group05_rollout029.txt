import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

    # Approximate gradient: d/dh_i of sum_j h_j (1 - h_{j+k})

    # This is a simplified gradient estimate
    grad = np.zeros_like(h)
    # Simplified: gradient for the k that maximizes overlap
    # Use finite difference approximation for a better estimation
    eps = 1e-5
    current_overlap = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points, dx)
        overlap_plus = evaluate_c5(h_plus, dx)
        grad[i] = (overlap_plus - current_overlap) / eps
    return grad

def line_search(h, grad, n_points, dx, alpha_max=0.1, max_steps=10):
    # Simple backtracking line search
    alpha = alpha_max
    current_c5 = evaluate_c5(h, dx)
    for _ in range(max_steps):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points, dx)
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            return h_new, new_c5, alpha
        alpha *= 0.5
    return h, current_c5, alpha
