import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or create a pulse-like starting point
    if 'initial_h_values' in globals():

    else:

        h[:n_points//2] = 1.0
        
    # Add noise to break symmetry
    noise = np.random.normal(0, 0.05, n_points)
    h += noise
    h = project_to_feasible(h, n_points)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temperature = 0.1 * current_c5
    cooling_rate = 0.999
    noise_std = 0.01
    
    # Main loop
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Generate candidate solution
        candidate = h + np.random.normal(0, noise_std, n_points)
        candidate = project_to_feasible(candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject based on temperature
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Probabilistic acceptance
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temperature):
                h = candidate
                current_c5 = candidate_c5
        
        # Update temperature
        temperature *= cooling_rate
        
        # Occasionally refine with gradient descent
        if iterations % 50 == 0:
            # Simple gradient estimation via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, n_points)
                c5_minus = evaluate_c5(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Gradient descent step
            step_size = 0.01 * (0.99 ** (iterations // 100))
            h = h - step_size * grad
            h = project_to_feasible(h, n_points)
            
            # Update current evaluation
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation to make sure we have the true best
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
