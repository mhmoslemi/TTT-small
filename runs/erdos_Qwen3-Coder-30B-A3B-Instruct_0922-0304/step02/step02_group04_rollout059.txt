import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
    # Add noise to start
    h += np.random.normal(0, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    
    # Compute initial C5
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Set initial temperature and cooling parameters
    temperature = 0.1 * current_c5
    cooling_rate = 0.999
    min_temperature = 1e-8
    step_size = 0.01
    
    # Main simulated annealing loop
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Generate candidate by adding noise
        candidate = h + np.random.normal(0, 0.01, n_points)
        candidate = project_to_feasible(candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Accept with probability
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temperature):
                h = candidate
                current_c5 = candidate_c5
        
        # Cool down temperature
        temperature = max(temperature * cooling_rate, min_temperature)
        
        # Occasionally refine with gradient descent
        if iterations % 50 == 0:
            # Perform gradient descent refinement
            for _ in range(5):
                if time.time() - start_time > budget_s - 1:
                    break
                # Finite difference gradient estimation
                eps = 1e-4
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    c5_minus = evaluate_c5(h_minus, dx)
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                # Update step
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                # Adjust step size
                step_size *= 0.98
                
                if step_size < 1e-6:
                    step_size = 1e-6
            
            # Early exit if target reached
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
