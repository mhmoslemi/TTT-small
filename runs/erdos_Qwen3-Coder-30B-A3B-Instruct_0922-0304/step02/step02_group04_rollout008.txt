import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum=h, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Compute overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with rectangular pulse and noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
    
    h = project_to_feasible(h, n_points)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Set up annealing parameters
    temp = 0.1 * current_c5
    cooling_rate = 0.999
    noise_std = 0.01
    gd_steps = 5
    gd_step_size = 0.01
    refinement_freq = 50
    
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate via random perturbation
        candidate = h + np.random.normal(0, noise_std, n_points)
        candidate = project_to_feasible(candidate, n_points)
        
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            temp *= cooling_rate
        else:
            # Accept with probability
            if random.random() < np.exp(-(candidate_c5 - current_c5) / max(temp, 1e-10)):
                h = candidate
                current_c5 = candidate_c5
            temp *= cooling_rate
            
        # Gradient descent refinement every few iterations
        if i % refinement_freq == 0:
            # Simplified gradient descent
            eps = 1e-4
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = project_to_feasible(h_plus, n_points)
                h_minus = project_to_feasible(h_minus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                c5_minus = evaluate_c5(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h = h - gd_step_size * grad
            h = project_to_feasible(h, n_points)
            current_c5 = evaluate_c5(h, dx)
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
            gd_step_size *= 0.9
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
