import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    F = 0.7
    CR = 0.9
    gd_interval = 50
    gd_iterations = 50
    gd_step_size = 0.01
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)

    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Create new population through DE mutation and crossover
        new_population = population[:population_size//2].copy()
        
        # DE mutation: for each individual, create mutant
        for i in range(population_size//2, population_size):
            # Select three random individuals different from i
            a, b, c = random.sample(range(population_size//2), 3)
            a, b, c = sorted([a, b, c])
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            child = np.where(crossover_mask, mutant, population[i])
            
            # Apply feasibility projection
            child = project_to_feasible(child, n_points)
            new_population.append(child)
        
        population = new_population
        
        # Apply gradient descent refinement periodically
        if gen % gd_interval == 0 and gen > 0:
            # Refine the best solution found so far
            h_best = best_solution.copy()
            
            # Gradient descent with finite differences
            for _ in range(gd_iterations):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                corr = correlate(h_best, 1 - h_best, mode="full")
                c5_val = np.max(corr) * dx
                
                # Compute gradient via finite differences
                grad = np.zeros_like(h_best)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_best.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    
                    h_minus = h_best.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    c5_minus = np.max(corr_minus) * dx
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                # Gradient descent update
                h_best = h_best - gd_step_size * grad
                h_best = project_to_feasible(h_best, n_points)
            
            # Update best solution
            c5_refined = evaluate_c5(h_best, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_solution = h_best.copy()
        
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = differential_evolution(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
