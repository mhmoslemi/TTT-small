import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex relaxation approach
    n_steps = 50
    mu_schedule = np.logspace(np.log10(0.5), 0, n_steps)  # from 0.5 to 1.0
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Formulate the convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective: weighted combination of C5 and regularization
        # We approximate C5 with a linearized version for convexity
        # The actual maximization is hard, so we use a direct approach
        objective = cp.Minimize(
            mu * cp.sum_squares(h_var - 0.5) / n_points  
            + (1 - mu) * cp.sum(cp.multiply(h_var, 1 - h_var))
        )
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve the convex problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                # Renormalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate current solution
                current_c5 = evaluate_c5(h_new, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
                    
                # Use as new starting point for next iteration
                h = h_new
            else:
                # If failing, fallback to local search
                h = h * (n_points / 2) / np.sum(h)
        except:
            # fallback to previous h if any
            h = h * (n_points / 2) / np.sum(h)
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation and return
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
