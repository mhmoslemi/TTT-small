import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: clip to [0,1] and rescale to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    """Compute gradient of C5 using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_to_feasible(h_plus, n)
        h_minus = project_to_feasible(h_minus, n)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def line_search(h, d, dx, max_alpha=1.0, c1=1e-4):
    """Armijo backtracking line search"""
    alpha = max_alpha
    c5_base = evaluate_c5(h, dx)
    
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new <= c5_base + c1 * alpha * np.dot(d, compute_gradient(h, dx)):
            return alpha
        alpha *= 0.5
        
    return alpha

def solve_quadratic_subproblem(H, g, n_points):
    """Solve quadratic subproblem with bounds and equality constraint"""
    # Direct projection approach for simplicity
    # We'll solve the unconstrained problem and then project
    try:
        d = -np.linalg.solve(H, g)
    except:
        d = -g
    
    # As an approximation, we'll just use the negative gradient
    # and project to satisfy constraints
    return d

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable initial configuration
        h = np.ones(n_points) * 0.5
        # Add some structure
        h[:n_points//3] = 0.8
        h[n_points//3:2*n_points//3] = 0.2
        h[2*n_points//3:] = 0.7
    
    h = project_to_feasible(h, n_points)
    
    # Initialize Hessian approximation as identity
    H = np.eye(n_points)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Solve quadratic subproblem
        d = solve_quadratic_subproblem(H, g, n_points)
        
        # Line search
        alpha = line_search(h, d, dx)
        
        # Update h
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Update Hessian (BFGS update)
        # Simplified approximation
        y = compute_gradient(h_new, dx) - g
        s = h_new - h
        if np.dot(s, y) > 1e-10 and np.linalg.norm(s) > 1e-10:
            # BFGS update
            try:
                H = H + (np.outer(s, s) / np.dot(s, y)) - (np.outer(H @ y, H @ y) / np.dot(y, H @ y))
            except:
                pass  # If BFGS fails, continue with old H
        
        h = h_new
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early exit criterion
        if best_c5 <= 0.3807:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
