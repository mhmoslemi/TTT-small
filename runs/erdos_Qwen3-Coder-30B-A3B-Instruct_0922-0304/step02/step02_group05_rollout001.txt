import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    # Calculate the overlap integral as described
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Project h onto feasible region [0,1] and ensure sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, n_points, dx):
    # Compute an approximate gradient of overlap_k with respect to h
    # overlap_k = sum_i h[i](1 - h[i+k]) 
    # grad_i overlap_k = (1 - h[i+k]) - h[i+k] 
    gradient = np.zeros_like(h)
    for i in range(n_points):
        j = (i + k_max) % n_points
        gradient[i] = (1 - h[j]) - h[j]
    return gradient

def line_search(h, grad, max_iterations=10, step_size=0.01):
    # Perform simple line search to find the best step
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    current_h = h.copy()
    
    for _ in range(max_iterations):
        new_h = current_h - step_size * grad
        new_h = project_feasible(new_h, len(h), 2.0 / len(h))
        new_c5 = evaluate_erdos_solution(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            step_size *= 0.5
        current_h = new_h
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    # Initialize from the parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization if no parent
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_feasible(h, n_points, dx)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    max_steps = 1200
    step = 0
    relaxation_interval = 50
    lambda_reg = 0.01  # Regularization parameter for convex relaxation

    while step < max_steps and (time.time() - start_time) < budget_s - 1:
        # Compute overlaps for all shifts
        full_corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(full_corr)
        max_overlap = full_corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(h, k_max, n_points, dx)
        
        # Gradient descent step
        new_h, new_c5 = line_search(h, grad, max_iterations=10, step_size=0.01)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        h = new_h.copy()
        step += 1
        
        # Periodic relaxation with entropy regularization
        if step % relaxation_interval == 0:
            # Add entropy regularization to escape local minima
            h_reg = h + lambda_reg * (h * np.log(h + 1e-10) + (1 - h) * np.log(1 - h + 1e-10))
            h = project_feasible(h_reg, n_points, dx)
            
            # Update best if this improves solution
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
