import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    # Compute the maximum overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Clamp to [0,1] and scale to sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Default initialization: uniform random with some structure
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    # Track best solution
    best_h = h.copy()
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    
    # Gradient descent parameters
    alpha = 1e-3
    max_iter = int(budget_s / 0.1) if budget_s > 1 else 1000
    relaxation_interval = 50
    regularization_lambda = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find worst case
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_k = np.argmax(overlap)
        
        # Estimate gradient using finite differences
        eps = 1e-4
        gradient = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, n_points, dx)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus)
            gradient[i] = (c5_plus - max_overlap) / eps
        
        # Line search to find step size
        step_found = False
        for step_size in [1e-3, 1e-2, 1e-1]:
            h_candidate = h - step_size * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < current_c5:
                h = h_candidate
                current_c5 = c5_candidate
                step_found = True
                break
                
        if not step_found:
            # If no improvement, decrease step size
            alpha *= 0.9
            
        # Periodic convex relaxation
        if iteration % relaxation_interval == 0 and iteration > 0:
            # Convex relaxation approach
            try:
                h_cvx = cp.Variable(n_points)
                # Minimize max overlap plus regularization
                objective = cp.Minimize(
                    overlap[max_k] * dx +
                    regularization_lambda * cp.sum(cp.entr(h_cvx))
                )
                constraints = [
                    h_cvx >= 0,
                    h_cvx <= 1,
                    cp.sum(h_cvx) == n_points / 2
                ]
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relaxed = h_cvx.value
                    h_relaxed = project_feasible(h_relaxed, n_points, dx)
                    c5_relaxed = evaluate_erdos_solution(h_relaxed)
                    if c5_relaxed < current_c5:
                        h = h_relaxed
                        current_c5 = c5_relaxed
                else:
                    regularization_lambda *= 0.5
            except:
                pass  # If we fail, continue with gradient descent
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early return if target met
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
