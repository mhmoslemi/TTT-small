import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using symmetric pattern (5-fold)
    h = np.zeros(n_points)
    segment_size = n_points // 5
    for i in range(5):
        start_idx = i * segment_size
        end_idx = start_idx + segment_size
        if i % 2 == 0:
            h[start_idx:end_idx] = 0.7
        else:
            h[start_idx:end_idx] = 0.3
    
    # Adjust to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using L-BFGS-B
    def objective_and_gradient(h_vec):
        # Reshape
        h = h_vec.reshape(-1)
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute C5
        c5_val = evaluate_c5(h, dx)
        
        # For gradient, use finite differences with small perturbation
        eps = 1e-5
        grad = np.zeros_like(h)
        base_c5 = c5_val
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
            
        return c5_val, grad
    
    # Optimization parameters
    max_iter = 1000
    opt_iter = 0
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # L-BFGS-B optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    current_h = h.copy()
    
    while opt_iter < max_iter and time.time() - start_time < budget_s - 1:
        # Flatten
        current_flat = current_h.flatten()
        
        # Optimize
        try:
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                current_flat,
                method='L-BFGS-B',
                bounds=bounds,
                jac=lambda x: objective_and_gradient(x)[1],
                options={'maxiter': 10, 'ftol': 1e-6}
            )
            
            if result.success:
                current_h = result.x.reshape(-1)
                # Feasibility repair
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
                
                # Evaluate again
                current_c5 = evaluate_c5(current_h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                
                # Early exit if target reached
                if best_c5 <= 0.3807:
                    break
        except:
            # Fallback if optimization fails
            pass

        opt_iter += 1

    # Final evaluation to be sure
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
