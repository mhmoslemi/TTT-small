import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T0 = 1000.0
    T_min = 1.0
    alpha = 0.999
    T = T0
    
    # Best solution tracking
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Perturbation parameters
    perturb_frac = 0.1
    perturb_count = max(1, int(n_points * perturb_frac))
    
    iteration = 0
    while T > T_min and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Copy current solution
        h_current = h.copy()
        current_c5 = evaluate_c5(h_current, dx)
        
        # Generate candidate solution by perturbing 10% of points
        h_candidate = h_current.copy()
        indices = np.random.choice(n_points, size=perturb_count, replace=False)
        
        # Add small random changes
        delta = 0.1
        for i in indices:
            h_candidate[i] += np.random.uniform(-delta, delta)
        
        # Ensure sum constraint (maintain sum = 35)
        current_sum = np.sum(h_candidate)
        scale_factor = 35.0 / current_sum
        h_candidate = h_candidate * scale_factor
        
        # Ensure bounds [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Re-scale to exactly match sum constraint
        current_sum = np.sum(h_candidate)
        h_candidate = h_candidate * (35.0 / current_sum)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Metropolis acceptance criterion
        delta_e = candidate_c5 - current_c5
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h = h_candidate.copy()
        
        # Update best solution if improved
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h.copy()
        
        # Decrease temperature
        T = alpha * T
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
