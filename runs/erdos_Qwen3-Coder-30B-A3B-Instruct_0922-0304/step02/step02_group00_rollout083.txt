import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        # Use initial construction with some random perturbation
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles = []
        for i in range(n_particles//2):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
        # Add random particles
        for i in range(n_particles//2):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
    else:
        # Initialize random particles
        particles = []
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
    
    # Initialize velocities and best positions
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbests = [particle.copy() for particle in particles]
    pbest_scores = [evaluate_c5(particle, dx) for particle in particles]
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbests[best_idx].copy()
    best_c5 = pbest_scores[best_idx]
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            v = velocities[i]
            pbest = pbests[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (pbest - particle) + c2 * r2 * (gbest - particle)
            
            # Update position
            particle = particle + v
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Mutation with small probability
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                particle = particle + noise
                particle = np.clip(particle, 0, 1)
                particle = particle * (n_points / 2) / np.sum(particle)
            
            # Store updated particle
            particles[i] = particle
            
            # Evaluate and update personal best
            new_score = evaluate_c5(particle, dx)
            if new_score < pbest_scores[i]:
                pbest_scores[i] = new_score
                pbests[i] = particle.copy()
                
                # Update global best if needed
                if new_score < best_c5:
                    best_c5 = new_score
                    gbest = particle.copy()
        
        # Adaptive weight decay (optional)
        if iteration > 100:
            w = 0.7
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
    return (gbest.tolist(), best_c5, n_points)
