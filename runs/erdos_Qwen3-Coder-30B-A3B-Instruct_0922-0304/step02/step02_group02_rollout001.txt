import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        # Create symmetric structure that is likely to be better than simple step
        # Use a 5-fold symmetric pattern with alternating high/low values
        h = np.zeros(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            # Alternate between 0.6 and 0.4 in each segment
            for j in range(start_idx, end_idx):
                if (j - start_idx) % 2 == 0:
                    h[j] = 0.6
                else:
                    h[j] = 0.4
        h = project_to_feasible(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h_flat):
        h = h_flat.copy()
        h = project_to_feasible(h, target_sum)
        return evaluate_c5(h, dx)

    def callback(x):
        nonlocal best_c5, best_h
        h = project_to_feasible(x, target_sum)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 <= 0.3807:
            return True
        return False

    # Use L-BFGS-B with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    # Convert to flattened array for scipy minimize
    h_flat = h.copy()
    
    try:
        result = minimize(
            objective, 
            h_flat,
            method='L-BFGS-B',
            bounds=bounds,
            callback=callback,
            options=options,
            tol=1e-9
        )
        
        if result.success:
            h_opt = project_to_feasible(result.x, target_sum)
            final_c5 = evaluate_c5(h_opt, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_opt
    except:
        pass
    
    # Final refinement with local search if time permits
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        for _ in range(10000):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Small perturbations to escape local minima
            h_perturb = best_h.copy() + np.random.normal(0, 0.001, n_points)
            h_perturb = project_to_feasible(h_perturb, target_sum)
            
            c5_perturb = evaluate_c5(h_perturb, dx)
            if c5_perturb < best_c5:
                best_c5 = c5_perturb
                best_h = h_perturb.copy()
                
            if best_c5 <= 0.3807:
                break

    return (best_h.tolist(), best_c5, n_points)
