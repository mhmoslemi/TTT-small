import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with 5-fold symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create 5-fold symmetric pattern
        h = np.zeros(n_points)
        segment_size = n_points // 5
        pattern = [0.7, 0.3, 0.7, 0.3, 0.7]
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            h[start_idx:end_idx] = pattern[i]
        
    # Normalize to sum to 35 (n_points/2)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use L-BFGS-B for optimization
        def objective(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_scaled = h_clipped * (n_points / 2) / np.sum(h_clipped)
            return evaluate_c5(h_scaled, dx)
            
        def gradient(h_vec):
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            base_val = objective(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                grad[i] = (objective(h_plus) - base_val) / eps
            return grad
            
        # Apply constraints using bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize with L-BFGS-B
        result = minimize(
            objective,
            h,
            method="L-BFGS-B",
            bounds=bounds,
            jac=gradient,
            options={"maxiter": 20}
        )
        
        if result.success:
            h = result.x
        else:
            # Fallback to simpler update if optimization fails
            h = h * 0.99 + 0.01 * np.random.rand(n_points)
            
        # Maintain the constraint sum(h) = 35
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
