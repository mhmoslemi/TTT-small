import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two uniform blocks
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    max_iter = 1000
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap_val = overlap[max_overlap_idx] * dx
        
        # Early termination if target met
        if max_overlap_val <= 0.3807:
            break
            
        # Use CVXPY to solve convex relaxation
        # Variables
        h_vars = cp.Variable(n_points, name="h")
        # Objective: minimize maximum overlap through relaxation
        # Instead, gradient descent with finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        h_temp = h.copy()
        base_c5 = max_overlap_val
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update h using gradient descent
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if improvement
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < max_overlap_val:
            h = h_new
        else:
            # Reduce step size if no improvement
            step_size *= 0.95
            
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
