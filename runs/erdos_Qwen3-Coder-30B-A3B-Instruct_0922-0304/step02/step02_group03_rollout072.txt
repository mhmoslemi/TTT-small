import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: simple step function
        n_points = 70
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Enforce feasibility
    h = enforce_feasibility(h, n_points)
    dx = 2.0 / n_points
    
    # Objective function for optimization
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Constraint to maintain sum = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Run optimization with L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    constraint_eq = {'type': 'eq', 'fun': constraint}
    
    # Use multiple starting points to avoid local minima
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try with current h and a few random perturbations
    for i in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use current h or slightly perturb it
        if i == 0:
            h_start = h.copy()
        else:
            # Perturb the current best
            h_start = best_h + np.random.normal(0, 0.01, n_points)
            h_start = enforce_feasibility(h_start, n_points)
        
        # Optimize
        try:
            res = minimize(
                objective, 
                h_start, 
                method='L-BFGS-B', 
                bounds=bounds,
                constraints=constraint_eq,
                options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            if res.success:
                h_opt = enforce_feasibility(res.x, n_points)
                c5_opt = evaluate_overlap(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final check
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
