from scipy.signal import correlate

    best_h = h.copy()

    # ADMM-like optimization with convex subproblems
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Solve convex subproblem using CVXPY
        h_var = Variable(n_points)
        
        # Objective: minimize max overlap using augmented Lagrangian
        # h ⋆ (1-h) <= tau for all k, with penalization
        # We take the maximum overlap as objective and create an AL subproblem
        
        # Create lambda constraints

        # Simple convex relaxation via CVXPY
        # Minimize: max_k (h ⋆ (1-h))[k] + penalty * constraint_violation
        obj = cp.Minimize(cp_sum(h_var) * dx - 1)  # This will be constrained

        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp_sum(h_var) * dx == 1.0
        ]
        
        # Create problem and solve
        prob = Problem(obj, constraints)

        h = h_new
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    
    # Return best feasible solution found
    return (best_h.tolist(), final_c5, n_points)
