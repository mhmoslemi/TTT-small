import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        mid = n_points // 2
        h[:mid] = 1.0
        # Add some random noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 500
    learning_rate = 0.01
    max_inner_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap to find maximum
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx

        # Gradient descent to reduce the specific overlap at k_max
        h_temp = h.copy()

            corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
            c5_temp = np.max(corr_temp) * dx
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h_temp)

        if best_c5 <= 0.3807:

    # Check total computation time to avoid taking the last call
    if time.time() - start_time > budget_s - 0.1:
