import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 100000
    
    # Keep track of best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create candidate solution by perturbing current h
        h_candidate = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        delta = 0.1
        
        # Adjust selected indices
        for i in indices:
            h_candidate[i] += delta
        # Normalize to keep sum = 35
        sum_ratio = 35.0 / np.sum(h_candidate)
        h_candidate = h_candidate * sum_ratio
        # Clip and renormalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (35.0 / np.sum(h_candidate))
        
        # Evaluate current and candidate
        current_c5 = evaluate_c5(h, dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = h_candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta_e = candidate_c5 - current_c5
            if np.exp(-delta_e / T) > random.random():
                h = h_candidate.copy()
        
        # Update best if needed
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h_candidate.copy()
        
        # Cooling
        T = max(T_min, alpha * T)
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
