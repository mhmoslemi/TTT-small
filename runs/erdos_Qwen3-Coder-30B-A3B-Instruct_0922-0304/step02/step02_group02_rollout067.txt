import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Clip and rescale to meet sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # equivalent to integral constraint
    
    # Initialize swarm
    P = 50  # swarm size
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        # fallback: simple initialization
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        
    for i in range(P):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, target_sum)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 2.0
    max_iter = 500
    
    # Track best solution
    global_best_h = None
    global_best_c5 = float('inf')
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(P):
            h = swarm[i]
            v = velocities[i]
            
            # Evaluate current fitness
            current_c5 = evaluate_c5(h, dx)
            
            # Update personal best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = h.copy()
                
            # Early exit if target reached
            if global_best_c5 <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest = swarm[i]  # personal best
            gbest = global_best_h  # global best
            
            v = omega * v + c1 * r1 * (pbest - h) + c2 * r2 * (gbest - h)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = repair_h(h_new, target_sum)
            swarm[i] = h_new
            
        if global_best_c5 <= 0.3807:
            break
    
    # Local search refinement on top 5% of particles
    top_k = max(1, P // 20)
    candidates = [(evaluate_c5(swarm[i], dx), swarm[i]) for i in range(P)]
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:top_k]
    
    for _, h in top_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient descent refinement
        learning_rate = 0.01
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute finite differences for gradient
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, target_sum)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
                
            h = h - learning_rate * grad
            h = repair_h(h, target_sum)
            
        # Update best if improved
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < global_best_c5:
            global_best_c5 = current_c5
            global_best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
