import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, n_points, dx):
    # Estimate gradient for shift k_max
    # This is a simplified gradient approximation for overlap_k_max
    # We compute the derivative with respect to h[i] for the max overlap
    grad = np.zeros_like(h)
    h_shifted = np.roll(h, k_max)
    
    # Approximate gradient for overlap in mode full (zero padded)
    # Change in overlap due to change in h[i] due to shift k_max
    for i in range(n_points):
        # For overlap(k_max), contribution of h[i] is 1-h[i+k_max]
        # But we need to account for the fact that correlation is not symmetric
        # We do it simply by checking which values contribute
        if 0 <= i + k_max < n_points:
            # h[i] contributes: (1 - h[i + k_max])
            # Since we added the derivative of (1 - h[i+k_max]) * h[i],
            # we get: -h[i+k_max] + (1 - h[i+k_max]) = 1 - 2*h[i+k_max]
            grad[i] = 1 - 2 * h[i+k_max]
    
    return grad

def line_search(h, grad, step_size, n_points, dx):
    # Simple backtracking line search
    alpha = step_size
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for _ in range(20):  # max iterations for line search
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points, dx)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            return best_h, best_c5
        alpha *= 0.5
    return best_h, best_c5

def convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Introduce entropy regularization to escape local minima
    # Minimize: C5 + lambda * entropy(h)
    # But for demonstration, we'll try a simple L2 regularization approach
    
    # Construct a simple regularization approach:
    # Add a small L2 penalty to avoid overfitting to a single peak
    # Not the full convex relaxation but enough to encourage exploration
    h = h.copy()
    mean_h = np.mean(h)
    h += (mean_h - h) * lambda_reg * 0.1  # Reduce to balance
    h = project_feasible(h, n_points, dx)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the available parent solution or a heuristic
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Fallback heuristic
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_size = 0.01
    relaxation_interval = 50
    lambda_reg = 0.01
    
    step_count = 0
    while time.time() - start_time < budget_s - 1:
        step_count += 1
        
        # Compute all overlaps and get current max
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_c5 = overlap[k_max] * dx
        
        # Compute gradient estimate
        grad = compute_gradient(h, k_max, n_points, dx)
        
        # Update h and project
        h = h - step_size * grad
        h = project_feasible(h, n_points, dx)
        
        # Evaluate new overlap
        c5_new = evaluate_c5(h, dx)
        
        # Update best solution seen
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Periodic convex relaxation
        if step_count % relaxation_interval == 0:
            h = convex_relaxation(h, n_points, dx, lambda_reg)
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
