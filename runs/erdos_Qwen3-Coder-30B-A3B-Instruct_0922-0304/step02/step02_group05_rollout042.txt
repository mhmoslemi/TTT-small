import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    lambda_reg = 0.01
    
    # Initialize from the existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Approximate gradient for the largest overlap
        h_shifted = np.roll(h, max_k)
        grad = h_shifted - (1 - h_shifted)
        
        # Line search with step sizes
        step_sizes = [0.1, 0.5, 1.0]
        best_candidate_c5 = best_c5
        best_candidate_h = h.copy()
        
        for alpha in step_sizes:
            h_candidate = project_feasible(h - alpha * grad, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # Update solution if improvement
        if best_candidate_c5 < best_c5:
            h = best_candidate_h
            best_c5 = best_candidate_c5
            
            # Update best solution found so far
            if best_c5 < evaluate_erdos_solution(best_h):
                best_h = h.copy()
        
        # Every 20 iterations, try convex relaxation for escape from local minima
        if iteration % 20 == 0:
            try:
                # Construct convex relaxation
                h_relax = cp.Variable(n_points, nonneg=True)
                objective = cp.Maximize(cp.sum(cp.multiply(h_relax, 1 - h_relax)))
                constraints = [
                    cp.sum(h_relax) == n_points/2,
                    h_relax >= 0,
                    h_relax <= 1
                ]
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax_sol = h_relax.value
                    # Project to feasible space and evaluate
                    h_relax_projected = project_feasible(h_relax_sol, n_points, dx)
                    c5_relax = evaluate_erdos_solution(h_relax_projected)
                    
                    if c5_relax < best_c5:
                        h = h_relax_projected
                        best_c5 = c5_relax
                        if best_c5 < evaluate_erdos_solution(best_h):
                            best_h = h.copy()
            except:
                pass
        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
