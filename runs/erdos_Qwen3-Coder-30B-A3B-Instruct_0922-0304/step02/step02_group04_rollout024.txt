import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, epsilon=1e-4):
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project_to_feasible(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    return grad

def solve_quadratic_subproblem(H, g, n_points, dx):
    # Solve min 0.5 * d^T * H * d + g^T * d
    # Subject to: sum(d) = 0, 0 <= h[i] + d[i] <= 1
    # Use projected gradient descent on the quadratic subproblem
    
    d = -np.linalg.solve(H, g)
    
    # Ensure d sums to zero
    d_sum = np.sum(d)
    d = d - d_sum / n_points
    
    # Project to feasible region
    for i in range(n_points):
        if d[i] < -h[i]:
            d[i] = -h[i]
        elif d[i] > 1 - h[i]:
            d[i] = 1 - h[i]
    
    return d

def line_search(h, d, dx, alpha_max=1.0, beta=0.5):
    alpha = alpha_max
    c5_current = evaluate_c5(h, dx)
    
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < c5_current:
            return alpha
        alpha *= beta
    
    return alpha_max

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.uniform(0.3, 0.7, n_points)
        h = project_to_feasible(h, n_points)
    
    # Initialize Hessian approximation
    H = np.eye(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Update Hessian (BFGS-like update - simplified)
        # In full SQP we would compute actual Hessian but for this case we approximate
        H = np.eye(n_points)  # Using identity as quick approximation
        
        # Solve quadratic subproblem
        d = solve_quadratic_subproblem(H, g, n_points, dx)
        
        # Line search
        alpha = line_search(h, d, dx)
        
        # Update h
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Compute new objective
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept new solution if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Update h for next iteration
        h = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
