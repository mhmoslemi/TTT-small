import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # L1-ball projection to maintain sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, dx, max_shift):
    # For the max shift k, compute the gradient of overlap_k w.r.t h
    # overlap_k = sum h[i] * (1 - h[i+k])
    # d/dh[i] overlap_k = (1-h[i+k]) if i < N-k else 0
    # We compute d/dh[i] overlap_{max_shift}
    grad = np.zeros_like(h)
    # For the lag k in correlation, we have that:
    # overlap_k = sum_{i=0}^{N-1-k} h[i] * (1 - h[i+k])
    # so for i, gradient w.r.t to h[i] is (1 - h[i+k]) and to h[i+k] is -h[i]
    # Therefore we consider the shifted sum:
    # For terms in overlap_k that involve h[i], we get 
    # d/dh[i] = (1 - h[i+k])
    
    # So for i, we look at contributions from:
    # h[i] * (1 - h[i+k]), so grad[i] += (1 - h[i+k]) where i+k < N
    # h[i-k] * (1 - h[i]) so grad[i] -= h[i-k] where i-k >=0

    # Replace the recurrence to create the rank 1 matrix:
    # we get that gradient for index i is sum over k of (1-h[i+k] - h[i-k]) for valid indices
    # But actually for main contribution we'll use convolution approach with d/dh terms:
    # Note: overlap_k[i] = h[i] * (1-h[i+k])
    # For gradient of overlap_k w.r.t. h[j]: 
    # if j == i: 1-h[i+k]
    # if j == i+k: -h[i]
    # So we sum over all valid i: for fixed j, j=i or j=i+k
    
    # let's vectorize computing exact gradient for a given shift k
    # overlap_k = sum h[i] * (1 - h[i+k]), 
    # so grad h[j] = sum over i where (i+k)==j of (1 - h[j]) 
    # + sum over i where (i==j) of (1 - h[i+k]) -> (1 - h[j+k])
    
    # But since modulo is not used, it's zero padded:
    # Plain shift logic:
    # First compute indices of valid convolution: [0, N-k-1] for shift k -> h[i] * (1-h[i+k])
    # grad = 1 - h[i+k] for i+k < N, and 0 elsewhere  
    # This is zero-padded on the right; a bit tricky. Let's use a full correlation
    # Actually, use a simple matrix approach:
    
    # We compute: overlap_k = sum_{i=0}^{N} h[i] * (1 - h[i+k])  (zero padding means <=i+N-1)
    # That's equivalent as: overlap_k[i] = h[i] * (1 - h[i+k]) for i=0..N-k-1
    # Gradients:
    # For i and j in range N:
    # d/dh[j] (h[i] * (1 - h[i+k])) 
    # If i == j: (1 - h[i+k])
    # If i+k == j: -h[i]
    # For all other terms: 0
    # So for i, j, where i+j (shift by k) together, we get:
    
    # This is correct by definition in correlation
    # If `k` is the shift in correlation of h with (1-h), 
    # overlap_k = sum_{i=0}^{N-1} h[i] * (1 - h[i+k])  
    # for i+k < N
    # d/dh[j] overlap_k = (1 - h[j+k]) if j == i and i+k < N
    #                     - h[i] if j == i+k and i < N
    # Actually, consider 
    # overlap_k = sum_{i=0}^{N-1} h[i] * (1 - h[i+k]) 
    # Then d/dh_j overlap_k = (1 - h_j+k) if j == i for sum and i+k < N
    # -h_i if j = i+k for sum and i < N
    # Now over all terms in sum, 
    # => Say there are i's in sum. For a fixed j, 
    # - If j = i+k for some i (so i = j-k), we get contribution -h_i = -h[j-k]
    # - If j = i for some i (so i = j), we get (1 - h_i) = (1 - h[j])
    # but only sum when i, i+k < N
    
    # Let's compute this using a convolution (with compact indexing)
    N = len(h)
    
    # We have 0 shift: i, i+k for h[i] * (1 - h[i+k]), so for every k there's a shift
    # Overlap_k = forward indices i (0 ... N-1), with reverse indices (i+k) 
    # We need gradients with respect to h at each position: 
    # i -> contribute (1 - h[i+k]) when there is such contribution (i+k < N)
    # i+k -> contribute -h[i] when there is such contribution (i < N)
    
    # Create two parts manually on shifted left and shifted right:
    grad = np.zeros_like(h)
    
    # Where i range goes and contribution: 1 - h[i+k]
    for i in range(N):
        if i+k < N:
            grad[i] += 1.0 - h[i+k]
    # Where i range goes and contribution: -h[i-k] or h[i] but as contribution to i-k
    # i-k = j where i = j+k comes from the right side
    for i in range(N):
        if i-k >= 0 and i < N:
            grad[i] -= h[i-k]  
    
    # The CASE to handle every time is: 
    # Continue with vectorized corruption to clean this:
    # @Simpler: Create index arrays...
    # easy implementation:
    # shift_h = np.zeros_like(h)
    # shift_h[1:] = h[:-1]
    # gradient = 1-shift_h - np.roll(h, k) # This is of different doses

    # If shifting by +k in index (i+k) 
    # forward correlation: overlap_k = h[i] * (1-h[i+k])
    # So to diff by h[i], we get:  
    # d/dh[i] = (1-h[i+k]) (when exists)
    # Also d/dh[i+k] contributes as -h[i] if exists
    
    # vector version:
    shift_indices = np.arange(k, len(h))
    h_shifted = np.zeros_like(h)
    h_shifted[shift_indices] = h[:len(h) - k]
    grad = 1.0 - h_shifted
    if k < len(h): 
        # Handle negative shift
        shift_indices_neg = np.arange(k)
        h_shifted_neg = np.zeros_like(h)
        h_shifted_neg[shift_indices_neg] = h[k:]
        grad -= h_shifted_neg
    
    # postponed: handle reverse contribution:
    # Recursive definition.
    # For valid overlap_k (i+k < N):
    # For index i: contribution (1 - h[i+k])
    # For index i+k: contribution -h[i]
    # Translate this to scalar gradients.
    return grad

def compute_and_apply_gradient(h, dx, step_size):
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    max_val = overlap[max_k]
    N = len(h)
    
    # Approximate gradient in index space all:
    # Sparse updates for contribution from convolved indices
    # use 1-(h shifted) and -helement shifted to position j:
    # Needs i, i+k index access for contribution:
    # For j: 
    #    if exists i for i==j: 1-h[i+k] 
    #    if exists i for i+k==j: -h[i]
    #    otherwise 0
    
    grad = np.zeros(N)
    # contrib1 ... 1-h[i+k] for i = 0 to N-k-1
    for i in range(N - max_k):
        grad[i] += 1.0 - h[i + max_k]
    # contrib2: -h[i] for i+k = j: so i = j-k, j = i+k
    for j in range(max_k, N):
        i = j - max_k
        grad[j] -= h[i]
        
    new_h = h - step_size * grad
    return project_feasible(new_h, N, dx)
