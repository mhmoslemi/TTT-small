import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize particles with random variations of h0
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    
    for _ in range(n_particles - 1):
        h = h0.copy()
        # Add some block structure with perturbations
        block1 = np.random.randint(0, n_points//3)
        block2 = np.random.randint(2*n_points//3, n_points)
        h[block1:block2] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia_weight = 0.7
    inertia_decay = 1e-4
    
    # Track best positions
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = inertia_weight * velocities[i] \
                + alpha * r1 * (personal_best[i] - particles[i]) \
                + beta * r2 * (global_best - particles[i])
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[i] = v
            
            # Update position
            new_pos = particles[i] + v
            new_pos = project_to_feasible(new_pos, n_points)
            
            # Evaluate new position
            c5 = evaluate_c5(new_pos, dx)
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best[i] = new_pos.copy()
                personal_best_scores[i] = c5
                
                # Update global best if needed
                if c5 < global_best_score:
                    global_best = new_pos.copy()
                    global_best_score = c5
                    
            particles[i] = new_pos
            
        # Update inertia weight
        inertia_weight = max(0.4, inertia_weight * (1 - inertia_decay))
        
        # Early termination if target met
        if global_best_score <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
