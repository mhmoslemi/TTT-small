import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum, Minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Use a sawtooth pattern as initial guess
        h = np.linspace(0, 1, n_points)  
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Solve convex subproblem to reduce overlap at shift k_max
        h_var = Variable(n_points)
        
        # Create the linear objective
        # overlap at shift k_max is sum_{i} h[i] * (1 - h[(i+k_max)%n]) 
        objective = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            objective += h_var[i] * (1 - h_var[j])
        
        objective *= dx
            
        # Constraints
        constraints = [
            h_var[i] >= 0 for i in range(n_points)
        ] + [
            h_var[i] <= 1 for i in range(n_points)
        ] + [
            cp_sum(h_var) == n_points / 2
        ]
        
        # Formulate and solve the convex problem
        try:
            prob = Problem(Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array([h_var[i].value for i in range(n_points)])
                h = h_new
            else:
                # If optimization fails, use previous h and recompute
                pass
        except Exception:
            # If optimization fails for some reason, continue with current h
            pass
            
        # Maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
