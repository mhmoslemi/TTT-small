import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Settings
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    relaxation_interval = 20
    lambda_reg = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find max
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Compute gradient for the shift that achieves max
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        step_sizes = [0.1, 0.5, 1.0]
        best_candidate_h = h.copy()
        best_candidate_c5 = best_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_candidate_c5:
                best_candidate_h = h_candidate.copy()
                best_candidate_c5 = candidate_c5
                
        h = best_candidate_h
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = h.copy()
            
        # Occasionally apply convex relaxation
        if iteration % relaxation_interval == 0:
            # Solve convex relaxation to escape local minimum
            h_relax = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.max(cp.convolve(h_relax, 1 - h_relax, mode='full')) + 
                lambda_reg * cp.sum(cp.entr(h_relax) + cp.log(n_points/2) * h_relax)
            )
            constraints = [
                h_relax >= 0,
                h_relax <= 1,
                cp.sum(h_relax) == n_points/2
            ]
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == 'optimal':
                    h_relax_sol = h_relax.value
                    h_relax_sol = project_feasible(h_relax_sol, n_points, dx)
                    relax_c5 = evaluate_erdos_solution(h_relax_sol)
                    if relax_c5 < best_c5:
                        best_c5 = relax_c5
                        best_h = h_relax_sol.copy()
                        h = h_relax_sol.copy()
            except:
                pass  # Skip relaxation if solver fails
            
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
