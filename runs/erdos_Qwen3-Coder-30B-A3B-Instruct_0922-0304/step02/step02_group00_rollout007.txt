import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        # Seed with initial solution and random perturbations
        for i in range(n_particles // 2):
            h = initial_h.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
        for i in range(n_particles // 2, n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    else:
        # Fully random initialization
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    # Track best positions
    pbest = [p.copy() for p in particles]
    pbest_fitness = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Ensure sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
        
        # Apply mutation occasionally
        for i in range(n_particles):
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                
                # Update personal best if improved
                fitness = evaluate_c5(particles[i], dx)
                if fitness < pbest_fitness[i]:
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = fitness
                    
                    if fitness < gbest_fitness:
                        gbest = particles[i].copy()
                        gbest_fitness = fitness
                        
        # Decrease inertia weight over time
        if iteration > 100:
            w = 0.5
            
        # Early exit if target reached
        if gbest_fitness <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_fitness, n_points)
