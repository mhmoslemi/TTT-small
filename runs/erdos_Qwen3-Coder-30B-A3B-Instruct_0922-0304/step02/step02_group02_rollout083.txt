import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create initial symmetric construction
        h = np.zeros(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.7
            else:
                h[start_idx:end_idx] = 0.3
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective_and_gradient(h):
        # Scale h to meet constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute C5
        c5 = evaluate_c5(h, dx)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
        
        return c5, grad

    # Optimization using L-BFGS-B
    options = {'maxiter': 1000, 'disp': False}
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                h,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                jac=lambda x: objective_and_gradient(x)[1],
                options=options
            )
            
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
        except:
            # Fallback to simple gradient descent if L-BFGS fails
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Additional local refinement
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            h = h - 0.005 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
