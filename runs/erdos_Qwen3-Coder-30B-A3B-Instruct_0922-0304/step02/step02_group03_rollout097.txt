import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    h = enforce_feasibility(h, n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Objective function for scipy.optimize
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Constraint function for sum(h) = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each variable: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with L-BFGS-B
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use a simple loop with multiple restarts to get better results
    max_restarts = 10
    restart_count = 0
    
    while restart_count < max_restarts and time.time() - start_time < budget_s - 1:
        try:
            res = minimize(objective, h, method='L-BFGS-B', 
                           bounds=bounds, 
                           constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-6})
            
            if res.success:
                h_new = res.x
                h_new = enforce_feasibility(h_new, n_points)
                c5_new = evaluate_overlap(h_new, dx)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                
                # Early termination if target met
                if best_c5 <= 0.3807:
                    break
                    
            # Perturb starting point for next restart
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = enforce_feasibility(h, n_points)
            
        except Exception:
            # Fallback if optimization fails
            h = np.random.rand(n_points)
            h = enforce_feasibility(h, n_points)
            
        restart_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
