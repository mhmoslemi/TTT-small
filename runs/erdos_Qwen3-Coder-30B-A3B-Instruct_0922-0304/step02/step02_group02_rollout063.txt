import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex relaxation approach
    # Use log-scale for relaxation parameter mu
    mu_schedule = np.logspace(np.log10(0.5), 0.0, 50, endpoint=True)
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define convex optimization problem
        h_vars = cp.Variable(n_points)
        # Objective: mu * C5 + (1 - mu) * regularization
        # Regularization term to encourage centered distribution
        regularizer = cp.sum_squares(h_vars - 0.5) / n_points
        # Approximate C5 as maximum of a few key correlations
        # Simplify by optimizing a single representative correlation
        # We compute correlation with multiple shifts and pick max
        
        # For better numerical stability, we fix some constraints in optimization
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Use a convex approximation to avoid computing full correlation
        # This is approximate but should help escape local minima
        objective = cp.Minimize(mu * cp.max(cp.conv(h_vars, 1 - h_vars)) + (1 - mu) * regularizer)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                new_h = np.array(h_vars.value).reshape(-1)
                new_h = np.clip(new_h, 0, 1)
                # Renormalize to sum = n_points / 2
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
            else:
                # Fallback to previous solution if optimization fails
                pass
        except:
            # If solver fails, continue with previous solution
            pass
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
