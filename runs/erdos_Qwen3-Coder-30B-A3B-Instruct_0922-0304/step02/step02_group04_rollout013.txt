def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def local_improve(h, n_points, dx, iterations=50):
    # Simple hill climbing with gradient-like projection
    h = h.copy()
    for it in range(iterations):
        # Compute current C5
        c5_val = evaluate_c5(h, dx)
        
        # Estimate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_val) / eps
        
        # Move along gradient direction
        step_size = 0.005
        h = h - step_size * grad
        
        h = project_to_feasible(h, n_points)
        
    return h
