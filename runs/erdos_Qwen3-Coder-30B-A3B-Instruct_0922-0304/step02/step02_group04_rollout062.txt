def evaluate_c5(h, dx):
    """Evaluate the overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum=h, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def local_refinement(h, n_points, dx, max_iter=30):
    """Simple gradient descent or coordinate descent refinement"""
    h = h.copy()
    
    # Simple gradient based refinement loop with fixed step size
    step_size = 0.005
    
    for _ in range(max_iter):
        current_c5 = evaluate_c5(h, dx)
        
        # Compute approximate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update h with gradient descent step
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Check if improvement and update h
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            h = h_new
        else:
            step_size *= 0.9  # adaptive step size

    return h
