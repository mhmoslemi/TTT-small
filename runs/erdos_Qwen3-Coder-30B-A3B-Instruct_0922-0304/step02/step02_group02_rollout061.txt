import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use homotopy method with convex relaxation
    n_steps = 50
    mu_values = np.logspace(np.log10(0.5), 0, n_steps)
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create optimization variables
        h_vars = cp.Variable(n_points)
        
        # Objective: weighted combination of C5 and smoothing term
        # We'll compute C5 via relaxed quadratic approximation
        
        # Smoothing term: (h_i - 0.5)^2, averaged over all points
        smoothing_term = cp.sum_squares(h_vars - 0.5) / n_points
        
        # Simplified approach: use gradient descent for the normalized constraint

        # For larger mu, we trust it more and reduce smoothing
        objective = mu * cp.Minimize(cp.maximum(cp.sum(h_vars) - n_points/2, 0))  # Not quite right, but close
        
        # For now, just an efficient gradient descent based refinement
        # with convex approximate relaxation: We'll do better with simpler approach
        
        # Use previous solution as warm start
        h_vars.value = best_h.copy()
        h_vars.value = np.clip(h_vars.value, 0, 1)
        
        # Approximate the C5 minimization with simple gradient descent
        max_iter = 100
        step_size = 0.01
        
        for _ in range(max_iter):
            current_c5 = evaluate_c5(h_vars.value, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vars.value.copy()
                
            if time.time() - start_time > budget_s - 1:
                break
            
            # Compute gradient with finite difference
            eps = 1e-4
            grad = np.zeros(n_points)
            base_c5 = evaluate_c5(h_vars.value, dx)
            for i in range(n_points):
                h_plus = h_vars.value.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            
            # Update and project
            h_vars.value -= step_size * grad
            h_vars.value = np.clip(h_vars.value, 0, 1)
            h_vars.value = h_vars.value * (n_points / 2) / np.sum(h_vars.value)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)

    return (best_h.tolist(), final_c5, n_points)
