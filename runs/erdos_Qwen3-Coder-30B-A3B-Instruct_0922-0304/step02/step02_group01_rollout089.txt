import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.05):
    h_mutated = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mutated[i] += np.random.normal(0, sigma)
    h_mutated = np.clip(h_mutated, 0, 1)
    return h_mutated

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = scale_to_target_sum(h_init, target_sum)
        population.append(h_init)
    
    # Fill rest of population
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Main GA loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        # Elitism: keep best 10%
        elite_count = max(1, population_size // 10)
        elite_indices = np.argsort(fitness)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate_individual(child1)
            child1 = np.clip(child1, 0, 1)
            child1 = scale_to_target_sum(child1, target_sum)
            
            child2 = mutate_individual(child2)
            child2 = np.clip(child2, 0, 1)
            child2 = scale_to_target_sum(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
