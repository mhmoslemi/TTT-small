import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping h in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.05):
    """Gaussian mutation with boundary handling"""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += random.gauss(0, sigma)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_constraint(h, target_sum)
        population.append(h)
    else:
        # Create a simple initial solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_constraint(h, target_sum)
        population.append(h)
    
    # Add random individuals to the population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_constraint(h, target_sum)
        population.append(h)
    
    # GA loop
    best_c5 = float('inf')
    best_h = None
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, h))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness (lower is better)
        fitness.sort(key=lambda x: x[0])
        
        # Keep top 50% as parents
        parents = [h for _, h in fitness[:len(fitness)//2]]
        
        # Generate new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            # Select two parents
            parent1, parent2 = random.sample(parents, 2)
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            # Project to constraint
            child1 = project_to_constraint(child1, target_sum)
            child2 = project_to_constraint(child2, target_sum)
            # Add to population
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Replace population
        population = new_population
        
        # Check stagnation
        stagnation_count += 1
        if stagnation_count > 100:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
