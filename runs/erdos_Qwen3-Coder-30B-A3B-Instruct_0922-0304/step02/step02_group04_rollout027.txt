import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * 0.5
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-4
    h_base = h.copy()
    base_c5 = evaluate_c5(h_base, dx)
    
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm
    swarm_size = 30
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Create additional particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize velocities and personal bests
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    personal_best_positions = [p.copy() for p in population]
    personal_best_values = [evaluate_c5(p, dx) for p in population]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # Main PSO loop
    max_iterations = 500
    local_refinement_freq = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            h = population[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = random.random(), random.random()
            cognitive = c1 * r1 * (personal_best_positions[i] - h)
            social = c2 * r2 * (global_best_position - h)
            v = w * v + cognitive + social
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_h = h + v
            new_h = project_to_feasible(new_h, n_points)
            
            # Evaluate new position
            new_c5 = evaluate_c5(new_h, dx)
            
            # Update personal best
            if new_c5 < personal_best_values[i]:
                personal_best_positions[i] = new_h.copy()
                personal_best_values[i] = new_c5
                
                # Check if global best
                if new_c5 < global_best_value:
                    global_best_position = new_h.copy()
                    global_best_value = new_c5
            
            # Update swarm
            population[i] = new_h
            velocities[i] = v
            
        # Occasionally refine top particles
        if iteration % local_refinement_freq == 0 and iteration > 0:
            # Sort particles by fitness
            sorted_indices = np.argsort(personal_best_values)
            top_particles = sorted_indices[:swarm_size//10]
            
            for idx in top_particles:
                h = personal_best_positions[idx]
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h, dx)
                    h = h - step_size * grad
                    h = project_to_feasible(h, n_points)
                personal_best_positions[idx] = h
                personal_best_values[idx] = evaluate_c5(h, dx)
                
                # Check if global best
                if personal_best_values[idx] < global_best_value:
                    global_best_position = h.copy()
                    global_best_value = personal_best_values[idx]
                    
        # Early exit
        if global_best_value <= 0.3807:
            break
    
    return (global_best_position.tolist(), global_best_value, n_points)
