import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a symmetric 5-fold structured initialization
        h = np.zeros(n_points)
        segment_size = n_points // 5  # 14 points per segment
        for i in range(5):
            start = i * segment_size
            end = start + segment_size
            # Alternate high and low values within segment
            for j in range(start, end):
                if (j - start) % 2 == 0:
                    h[j] = 0.6
                else:
                    h[j] = 0.4
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        # Ensure h_vec sums to n_points/2
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return evaluate_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization with L-BFGS-B
    max_iter = 1000
    iter_count = 0
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    # Optimization using L-BFGS-B with bounds
    try:
        # Make sure optimization starts from a feasible point
        h_feasible = np.clip(h, 0, 1)
        h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
        
        # Use scipy.optimize.minimize with L-BFGS-B
        result = minimize(
            objective,
            h_feasible,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-9},
            callback=lambda x: None
        )
        
        if result.success or not result.failed:
            final_h = np.clip(result.x, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = evaluate_c5(final_h, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass  # Fall back to gradient descent if L-BFGS fails
    
    # Local gradient-based refinement with line search
    def line_search(h_vec, grad, alpha=0.1, beta=0.5):
        # Simple backtracking line search
        h_new = h_vec - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Gradient-based local search
    max_local_iter = 500
    for iteration in range(max_local_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Finite difference gradient estimation
        eps = 1e-4
        grad = np.zeros_like(best_h)
        base_c5 = evaluate_c5(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
            
        # Update with step size adaptation
        step_size = 0.01
        h_new = best_h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if the new value is better
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            best_h = h_new.copy()
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
