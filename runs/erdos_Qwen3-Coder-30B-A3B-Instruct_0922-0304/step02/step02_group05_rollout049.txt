import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        scaling = (n_points / 2) / current_sum
        h = h * scaling
    return h

def initialize_particles(n_points, n_particles):
    """Initialize particle swarm with some diversity"""
    particles = []
    
    # Use initial_h_values if available as first particle
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create random initial particle
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points)
        particles.append(h)
    
    # Fill remaining particles with random distributions
    for _ in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points)
        particles.append(h)
        
    return particles

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_particles = 50
    dx = 2.0 / n_points
    
    # PSO parameters
    w_start = 1.0  # Inertia weight
    w_min = 0.8
    c1 = 2.0  # Cognitive coefficient
    c2 = 2.0  # Social coefficient
    v_max = 0.2
    perturb_factor = 0.05
    
    # Initialize particles and velocities
    particles = initialize_particles(n_points, n_particles)
    velocities = [np.random.normal(0, 0.01, n_points) for _ in range(n_particles)]
    
    # Evaluate initial fitness
    fitness = [evaluate_erdos_solution(p) for p in particles]
    
    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_fitness = fitness[:]
    
    # Global best
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_fitness
    
    # PSO iterations
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Update inertia weight
        w = max(w_min, w_start - (w_start - w_min) * iteration / (budget_s / 2))
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply random perturbation occasionally
            if random.random() < 0.1:
                particles[i] += np.random.normal(0, perturb_factor, n_points)
            
            # Project feasible
            particles[i] = project_feasible(particles[i], n_points)
            
            # Evaluate fitness
            current_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = current_fitness
                
                # Update global best
                if current_fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = current_fitness
                    last_improvement = iteration
                    if current_fitness < best_c5:
                        best_c5 = current_fitness
                        best_h = g_best.copy()
                        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Reset stagnated particles
        if iteration - last_improvement > 50 and iteration % 50 == 0:
            for i in range(n_particles):
                if random.random() < 0.3:
                    particles[i] = np.random.uniform(0, 1, n_points)
                    particles[i] = project_feasible(particles[i], n_points)
                    velocities[i] = np.random.normal(0, 0.01, n_points)
                    fitness[i] = evaluate_erdos_solution(particles[i])
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = fitness[i]
    
    return (best_h.tolist(), best_c5, n_points)
