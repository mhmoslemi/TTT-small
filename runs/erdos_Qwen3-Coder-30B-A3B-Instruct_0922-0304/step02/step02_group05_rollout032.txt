import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import log

def evaluate_erdos_solution(h):
    # Using the full overlap computation with zero-padding
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap)

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Solve convex relaxation minimizing overlap_k + lambda * sum(h * log(h))
    h_vars = cp.Variable(n_points)
    # Constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points/2]
    
    # Objective: Maximize overlap (minimize -overlap) with regularization
    # Using approximate max overlap computation
    max_k = np.argmax(correlate(h, 1 - h, mode="full")) - (n_points-1)
    # We'll use a fixed lag for now to compute a gradient-like term in convex relaxation
    
    # Construct overlap expression for relaxation
    # The detailed pyomo/Hessian based relaxation is complex and we approximate
    # a simpler relaxation: minimize overlap_k + lambda*entropy
    objective = cp.Minimize(
        cp.sum(cp.multiply(h_vars, 1 - h_vars)) + lambda_reg * cp.sum(cp.multiply(h_vars, cp.log(h_vars + 1e-10)))
    )
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        return h_vars.value
    except:
        # Fallback to original
        return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    # Initialize from pre-existing solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Uniform starting point
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    step_size = 1e-3
    reg_lambda = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlaps and find max
        overlap = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlap) * dx
        max_k = np.argmax(overlap)
        
        # Compute gradient using finite differences in a small neighborhood
        grad = np.zeros_like(h)
        eps = 1e-5
        h_plus = h.copy()
        h_plus[max_k % n_points] += eps
        h_plus = project_feasible(h_plus, n_points, dx)
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        grad[max_k % n_points] = (np.max(overlap_plus) - current_c5) / eps
        
        # Line search for step size
        found_improvement = False
        for alpha in [1e-3, 5e-4, 1e-4]:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < current_c5:
                h = h_candidate
                current_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h.copy()
                found_improvement = True
                break
        
        # Apply convex relaxation every few iterations
        if iteration % 50 == 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx, reg_lambda)
            c5_relaxed = evaluate_erdos_solution(h_relaxed)
            if c5_relaxed < best_c5:
                best_c5 = c5_relaxed
                best_h = h_relaxed.copy()
                
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
