import numpy as np
from scipy.signal import correlate
import time

def evaluate_fitness(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def generate_random_step(n_points):
    h = np.random.rand(n_points)
    return enforce_feasibility(h, n_points)

def two_point_crossover(parent1, parent2):
    p1, p2 = sorted(np.random.choice(len(parent1), 2, replace=False))
    child = np.copy(parent1)
    child[p1:p2] = parent2[p1:p2]
    return child

def gaussian_mutation(child, mutation_rate, sigma):
    mut_indices = np.random.rand(len(child)) < mutation_rate
    child[mut_indices] += np.random.normal(0, sigma, sum(mut_indices))
    return child

def tournament_selection(population, fitness, tournament_size=3):
    selected = []
    for _ in range(len(population)):
        candidates = np.random.choice(len(population), tournament_size)
        winner_idx = candidates[np.argmin([fitness[i] for i in candidates])]
        selected.append(population[winner_idx])
    return selected

def initialize_population(n_points, pop_size, initial_h=None):
    if initial_h is not None:
        initial_h = np.array(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        return [initial_h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    else:
        return [generate_random_step(n_points) for _ in range(pop_size)]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.1
    cx_prob = 0.7
    best_fitness = float('inf')
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = initialize_population(n_points, pop_size, initial_h=initial_h_values)
    else:
        population = initialize_population(n_points, pop_size)
    
    # Fitness evaluation
    fitness = [evaluate_fitness(individual, dx) for individual in population]
    best = np.argmin(fitness)
    best_h = population[best]
    best_fitness = fitness[best]
    
    for gen in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and Mutation
        children = []
        for i in range(0, len(selected)-1, 2):
            if np.random.rand() < cx_prob:
                child1 = two_point_crossover(selected[i], selected[i+1])
                child2 = two_point_crossover(selected[i], selected[i+1])
                child1 = gaussian_mutation(child1, mutation_rate, 0.1)
                child2 = gaussian_mutation(child2, mutation_rate, 0.1)
                child1 = enforce_feasibility(child1, n_points)
                child2 = enforce_feasibility(child2, n_points)
                children.extend([child1, child2])
            else:
                children.extend([selected[i], selected[i+1]])
        
        # Elitism: keep best solution
        children.append(best_h)
        population = children
        
        # Fitness evaluation
        fitness = [evaluate_fitness(individual, dx) for individual in population]
        
        # Update best
        current_best = np.min(fitness)
        if current_best < best_fitness:
            best_fitness = current_best
            best_h = population[np.argmin(fitness)]
            
        # Adaptive mutation rate
        if gen % 50 == 0:
            if current_best == best_fitness:
                mutation_rate = min(mutation_rate * 1.1, 0.3)
            else:
                mutation_rate = max(mutation_rate * 0.9, 0.01)
                
        # Early exit
        if best_fitness <= 0.3807:
            break

    return (best_h.tolist(), best_fitness, n_points)
