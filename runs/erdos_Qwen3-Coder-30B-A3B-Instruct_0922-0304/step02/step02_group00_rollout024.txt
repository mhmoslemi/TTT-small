import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    rho = 1.0
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Create gradient approximation for the specific overlap term
        step = 0.01
        iterations = 100
        
        h_temp = h.copy()
        for _ in range(iterations):
            # Evaluate current overlap
            corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
            c5_temp = np.max(corr_temp) * dx
            k_max_temp = np.argmax(corr_temp)
            
            # Compute gradient w.r.t. h using finite differences
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_temp) / eps
                
                # Add penalty to prevent violating constraints
                if h_temp[i] <= 0 or h_temp[i] >= 1:
                    grad[i] += -1000 * (1 if h_temp[i] <= 0 else -1)

            # Update h using gradient descent
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            
            # Maintain integral constraint
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp
        
        # Evaluate new result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
