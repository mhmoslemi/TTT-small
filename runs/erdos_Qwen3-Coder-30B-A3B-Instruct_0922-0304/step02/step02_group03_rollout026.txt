import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    max_iter_coarse = 500
    max_iter_fine = 1000
    lbfgs_factr = 1e7
    lbfgs_pgtol = 1e-5
    perturbation_ratio = 0.1
    
    best_c5 = float('inf')
    best_h = None
    n_points = 35
    
    # Initial phase: coarse discretization
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with two uniform blocks
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization for coarse grid
    def obj_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        overlap = correlate(h, 1 - h, mode="full")
        return -np.max(overlap) * dx
    
    def constraint_h(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_h}
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(obj_h, h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter_coarse, 'ftol': 1e-10, 'gtol': lbfgs_pgtol, 'factr': lbfgs_factr})
    
    if result.success:
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = -result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Refinement phase: increase resolution to 70 points
    n_points = 70
    dx = 2.0 / n_points
    
    # Upsample and perturb
    if best_h is not None:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        # Add small random pertubation
        h += np.random.normal(0, perturbation_ratio * 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization for fine grid
    def obj_h_fine(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        overlap = correlate(h, 1 - h, mode="full")
        return -np.max(overlap) * dx
    
    def constraint_h_fine(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    cons_fine = {'type': 'eq', 'fun': constraint_h_fine}
    bounds_fine = [(0, 1) for _ in range(n_points)]
    
    result = minimize(obj_h_fine, h, method='L-BFGS-B', bounds=bounds_fine, constraints=cons_fine,
                      options={'maxiter': max_iter_fine, 'ftol': 1e-10, 'gtol': lbfgs_pgtol, 'factr': lbfgs_factr})
    
    if result.success:
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = -result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation
    if best_h is None:
        # Fallback if no valid solution found
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        best_h = h.copy()
        overlap = correlate(best_h, 1 - best_h, mode="full")
        best_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), best_c5, n_points)
