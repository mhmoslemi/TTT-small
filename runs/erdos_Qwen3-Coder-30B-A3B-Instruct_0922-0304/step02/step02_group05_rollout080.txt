import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, dx, max_k):
    # Approximate gradient using finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    h_orig = h.copy()
    for i in range(len(h)):
        h_plus = h_orig.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, len(h_plus), dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        c5_plus = np.max(corr_plus) * dx
        grad[i] = (c5_plus - evaluate_c5(h_orig, dx)) / eps
    return grad

def line_search(h, grad, dx, max_k, step_sizes=[1e-3, 1e-2, 1e-1]):
    current_c5 = evaluate_c5(h, dx)
    for alpha in step_sizes:
        h_candidate = h - alpha * grad
        h_candidate = project_feasible(h_candidate, len(h), dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        if candidate_c5 < current_c5:
            return h_candidate, candidate_c5
    return h, current_c5

def convex_relaxation_step(h, n_points, dx, lambda_reg=0.01):
    # Solve relaxed problem using CVXPY
    h_vars = cp.Variable(n_points)
    # Relaxation term: entropy regularization
    entropy_term = cp.sum(cp.entr(h_vars))  # h * log(h) term
    
    # Set up the constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points/2
    ]
    
    # Compute overlap term (we'll use a representative shift)
    corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(corr)
    
    # Construct overlap term
    overlap_sum = 0
    for i in range(n_points):
        for j in range(n_points):
            k = (j - i) % n_points
            if k == max_k:
                overlap_sum += h_vars[i] * (1 - h_vars[j])
    
    obj = cp.Minimize(overlap_sum + lambda_reg * entropy_term)
    
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == 'optimal':
            new_h = h_vars.value
            # Make sure values are in [0,1]
            new_h = np.clip(new_h, 0, 1)
            # Rescale to sum = n_points/2
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            return new_h
    except:
        pass
    # If CVXPY fails, fall back to gradient step
    return h.copy()

def optimize_with_relaxations(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Fallback to a simple construction
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = project_feasible(h, n_points, dx)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    step_size = 1e-3
    lambda_reg = 0.01
    
    for i in range(10000):  # More iterations
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlaps) * dx
        max_k = np.argmax(overlaps)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        if time.time() - start_time > budget_s - 1:
            break

        # Periodic convex relaxation
        if i % 50 == 0 and i > 0:
            h = convex_relaxation_step(h, n_points, dx, lambda_reg)
            # Adjust regularization term if needed
            lambda_reg *= 0.9  # Decrease lambda slowly
            continue

        # Compute gradient
        grad = compute_gradient(h, dx, max_k)
        
        # Line search
        new_h, new_c5 = line_search(h, grad, dx, max_k)
        
        if new_c5 < current_c5:
            h = new_h
            current_c5 = new_c5
        else:
            # If no improvement, reduce step size
            step_size *= 0.8

        # Keep elapsed time under control
        if time.time() - start_time > budget_s - 1:
            break

        # Early exit if target met
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return optimize_with_relaxations(seed, budget_s)
