import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # L1-ball projection to maintain sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap and max
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx] * dx
        
        # Compute gradient estimate for shift that gives max overlap
        shift = max_overlap_idx - (len(h) - 1)
        h_shifted = np.roll(h, shift)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        best_h_candidate = h.copy()
        best_c5_candidate = best_c5
        
        for alpha in line_search_steps:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            
            if c5_candidate < best_c5_candidate:
                best_h_candidate = h_candidate
                best_c5_candidate = c5_candidate
        
        # Update
        h = best_h_candidate
        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Periodic convex relaxation
        if iteration % relaxation_interval == 0:
            try:
                # Simple convex relaxation approach: 
                # minimize sum(overlap_k) + lambda_reg * sum(h * log(h + 1e-8))
                # subject to h in [0,1], sum(h) = n_points/2
                
                # For implementation simplicity, we will use a weighted average of current h and a uniform distribution
                # This is a pragmatic approach to escape local optima
                uniform_h = np.ones(n_points) * 0.5
                h = 0.8 * h + 0.2 * uniform_h
                h = project_feasible(h, n_points, dx)
                
                # Update best if better
                c5_current = evaluate_c5(h, dx)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h.copy()
                    
            except:
                pass
        
        # Ensure we don't exceed budget
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
