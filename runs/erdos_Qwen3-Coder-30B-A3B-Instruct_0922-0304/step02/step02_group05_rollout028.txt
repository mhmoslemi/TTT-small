import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_shift, dx):
    # For shift k, the gradient of overlap_k w.r.t h[i] is:
    # (1 - h[i+k]) for i, and -(h[i-k]) for i+k < N 
    # But we can compute it as:
    # delta_overlap = (1 - h) shifted by k minus h shifted by -k
    # Actually simpler: we compute the full convolution 
    # then extract the gradient direction for shift k
    h_padded = np.pad(h, (0, len(h)), mode='constant', constant_values=0)
    h_shifted = np.roll(h_padded, k_shift)
    h_shifted = h_shifted[:len(h)]
    
    diff_h = 1 - h_shifted
    
    # The gradient of h[i] in overlap_k is (1 - h[i+k]) - h[i-k]
    # but we're computing a simplified gradient:
    # we know that for fixed k, when we vary h[i], we affect the term
    # h[i](1 - h[i+k]) 
    # So we approximate gradient as (1 - h[i+k]) for a fixed k
    # Provide the gradient with respect to shift k now
    dh = np.zeros_like(h)
    for i in range(len(h)):
        if i + k_shift < len(h):
            dh[i] = 1 - h[i + k_shift]
        else:
            dh[i] = 0  # artificial
    return dh

def line_search(h, grad, dx, max_alpha=1.0):
    alpha = max_alpha
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    for step in range(10):
        new_h = h - alpha * grad
        new_h = project_feasible(new_h, len(h), dx)
        new_c5 = evaluate_c5(new_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
        alpha *= 0.5
    return best_h, best_c5

def convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Add entropy regularization to avoid getting trapped in local opt
    # Not a full convex optimization but a heuristic greedy improvement
    h_new = h.copy()
    max_iter = 50
    for i in range(max_iter):
        # Evaluate maximum overlap
        overlaps = correlate(h_new, 1 - h_new, mode="full")
        k_max = np.argmax(overlaps)
    
        # Compute gradient with respect to shift k_max
        grad = compute_gradient(h_new, k_max, dx)
        # Add entropy gradient: - log(h[i]) for h[i] > 0
        ent_grad = np.zeros_like(h_new)
        for j in range(len(h_new)):
            if h_new[j] > 1e-10:
                ent_grad[j] = -1.0 / h_new[j]
        
        # Combined gradient with entropy term
        total_grad = grad - lambda_reg * ent_grad
        
        # Simple update with step size
        step_size = 0.01
        h_new = h_new - step_size * total_grad
        h_new = project_feasible(h_new, n_points, dx)
        
        if i % 20 == 0:
            # Check if we return to sufficiently good
            pass
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization 
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    # Optimizer
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step = 0
    while time.time() - start_time < budget_s - 1:
        # Look at all shifts and find the maximum overlap
        overlaps = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute gradient with respect to shift k_max
        grad = compute_gradient(h, k_max, dx)
        
        # Line search for optimal step
        h_new, c5_new = line_search(h, grad, dx)
        h = h_new
        
        # Sometimes apply convex relaxation
        if step % 50 == 0:
            h = convex_relaxation(h, n_points, dx)
        
        # Update best if better
        c5_new = evaluate_c5(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        step += 1
        
        if time.time() - start_time > budget_s - 1:
            break

    # Final validation
    final_h = project_feasible(best_h, n_points, dx)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
