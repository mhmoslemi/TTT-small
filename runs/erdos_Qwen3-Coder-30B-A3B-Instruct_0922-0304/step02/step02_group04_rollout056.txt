import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Add noise
    h += np.random.normal(0, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    
    # Initialize best solution
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing parameters
    temp = 0.1 * current_c5
    cooling_rate = 0.999
    min_temp = 1e-8
    noise_std = 0.01
    grad_step = 0.01
    grad_decay = 0.9
    grad_steps = 5
    refresh_interval = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidate
        candidate = h + np.random.normal(0, noise_std, n_points)
        candidate = project_to_feasible(candidate, n_points)
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Apply cooling
        temp = max(temp * cooling_rate, min_temp)
        
        # Periodic gradient refinement
        if iteration % refresh_interval == 0:
            # Simple gradient descent rough approximation
            eps = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, n_points)
                c5_minus = evaluate_c5(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            h = h - grad_step * grad
            h = project_to_feasible(h, n_points)
            current_c5 = evaluate_c5(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            grad_step *= grad_decay
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
