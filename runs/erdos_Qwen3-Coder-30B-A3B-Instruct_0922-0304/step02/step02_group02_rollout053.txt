import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Ensure sum = n_points / 2 and bounds [0,1]
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Construct symmetric initialization
        h = np.zeros(n_points)
        # Create 5-fold symmetric pattern
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.5 + 0.3 * np.ones(segment_size)
            else:
                h[start_idx:end_idx] = 0.5 - 0.3 * np.ones(segment_size)
        # Adjust for sum constraint
        h = project_to_feasible(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h_vec):
        h = h_vec.copy()
        h = project_to_feasible(h, n_points)
        return evaluate_c5(h, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Initial evaluation
    current_c5 = objective(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Set up optimize problem
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use L-BFGS-B with bounds
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimization step
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 50})
        
        if result.success:
            h = result.x
        else:
            # Fallback to previous h if optimization failed
            h = h.copy()
            
        # Project to feasible
        h = project_to_feasible(h, n_points)
        
        # Evaluate
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
