import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_relaxation(h_init, n_points, dx):
    # Create variables
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Convex relaxation - approximate the max overlap
    # For each k, compute overlap_k = sum_i h[i] * (1 - h[i+k])
    # This is an affine or quadratic function of h.
    # A valid relaxation is to bound each individual overlap with C.
    
    # We need to form overlap computation in a convex form
    # For computational efficiency, we'll solve for one k first and then iterate
    # As a safety, we'll use a heuristic that's known to produce good results
    
    # Let's try projecting the convex solution to satisfy overlap bounds
    # We force each individual overlap constraint to be <= C
    
    # Generate all overlaps explicitly in a manageable way for small scales
    # The most relevant overlaps are around zero-lag (large overlap)
    # We instead use a Lagrangian heuristic approach to approximate constraints
    
    # Heuristic: try to optimize one of the peak overlaps in a controlled way
    # Build preliminary solution: try smooth solutions
    
    # Fast relax and solve
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    prob.solve(solver=cp.CVXOPT, verbose=False)
    
    # Relax last result and project
    h_sol = h.value
    if h_sol is not None:
        h_sol = project_to_feasible(h_sol, n_points)
    else:
        h_sol = h_init.copy()
    
    return h_sol

def local_improve(h, n_points, dx, max_iter=50):
    # Local gradient-based refinement with finite differences
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    for _ in range(max_iter):
        best_h = current_h.copy()
        best_c5 = current_c5
        improvement = False
        
        # Evaluate gradient via finite differences
        eps = 1e-4
        for idx in range(n_points):
            h_plus = current_h.copy()
            h_plus[idx] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            
            if c5_plus < best_c5:
                best_h = h_plus
                best_c5 = c5_plus
                improvement = True
                
        if not improvement:
            break
            
        current_h = best_h
        current_c5 = best_c5
    
    return current_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with a good starting point or default

    else:

        h[:n_points//2] = 1.0
    
    # Add some noise to escape simple cases
    np.random.seed(seed)
    noise = np.random.normal(0, 0.05, n_points)
    h += noise
    h = project_to_feasible(h, n_points)
    
    # Begin optimization loop
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Try convex relaxation approach to get better structure
    h = solve_convex_relaxation(h, n_points, dx)
    
    # Local refinement
    h, c5 = local_improve(h, n_points, dx)
    
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # Iterative refinement
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 100:
        iter_count += 1
        if iter_count % 2 == 0:
            # Perturb slightly and do local refinement
            h = best_h.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h += noise
            h = project_to_feasible(h, n_points)
            
            h, c5 = local_improve(h, n_points, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            # Try convex relaxation again
            h = solve_convex_relaxation(best_h, n_points, dx)
            h, c5 = local_improve(h, n_points, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
        if best_c5 <= 0.3807:
