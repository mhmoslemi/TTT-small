import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_individual(h, mutation_rate=0.1, noise_std=0.1):
    h_new = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_new[i] += np.random.normal(0, noise_std)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover(parent1, parent2):
    point = random.randint(1, len(parent1) - 1)
    child = np.concatenate([parent1[:point], parent2[point:]])
    return child

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(c5)
            
        # Update best
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection (tournament selection)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness_scores)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
            
        # Crossover and mutation
        new_population = []
        elite_count = population_size // 10
        elite_indices = np.argsort(fitness_scores)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child = crossover(parent1, parent2)
            child = mutate_individual(child)
            child = np.clip(child, 0, 1)
            child = scale_to_target_sum(child, target_sum)
            new_population.append(child)
            
        population = new_population
        
    if best_individual is None:
        # Fallback: use the best from last generation
        fitness_scores = [evaluate_c5(h, dx) for h in population]
        min_fitness_idx = np.argmin(fitness_scores)
        best_individual = population[min_fitness_idx]
        best_c5 = fitness_scores[min_fitness_idx]
        
    return best_individual, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
