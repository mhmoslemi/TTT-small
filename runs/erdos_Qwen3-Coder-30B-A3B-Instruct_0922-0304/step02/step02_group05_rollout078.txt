import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, dx):
    # Compute approximate gradient for shift k_max
    # This computes ∇h overlap_k_max
    n = len(h)
    grad = np.zeros(n)
    
    # For overlap_k = sum_i h[i] * (1 - h[i+k]), 
    # the gradient w.r.t. h[i] is (1 - h[i+k]) if k = 0,
    # and -h[i-k] if k != 0.
    # We'll compute the gradient for the shift k_max
    
    # Using zero-padding for boundary condition
    # we have overlap_k = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) 
    # when k >= 0, and the shift is in the forward direction
    
    # For i = 0 to n-1, 
    # ∂/∂h[i] overlap_k = 1 - h[i+k] + (if k != 0) terms from other components
    # But the site i contributes 1 - h[i+k] to gradient of h[i]
    
    # This computes directional derivatives for the gradient
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1.0 - h[i + k_max]
        else:
            # Assume extrapolation (0 for out of bounds due to zero pad)
            grad[i] = 1.0
            
    # Now correct to account for forward differences due to convolution structure 
    # This should be just a standard finite difference in reverse
    
    # We also account for h[i+k] such that gradient looks like
    # it's computing df/dh where f = h^T (1 - h_{shifted}) 
    # so gradients of each component are:
    # d/dh[i] = (1 - h[i+k]) if k >= 0 and k < n
    
    # Since we're doing correlation overlap_k = sum h[i] * (1 - h[i+k]) 
    # gradient of overlap_k wrt h[i] = 1 - h[i+k]
    # For finite size, if i+k >= n, then we project to 0
    # But manufacturing a finite difference of a tri-diagonal approx 
    # will be computationally simpler.
    
    # Simplified: this works as below:
    # Only contributions to gradient of overlap_k are:
    # ∇_i (h[i] * (1 - h[i+k])) = (1 - h[i+k]) * 1 (locally)
    # But we computed a transpose version, so actual gradient should be:
    # ∇(overlap_k) = - (h_{-k}) + (1 - h)_{-k} 
    # Which is not right. Simpler:
    # Hence for overlap_k = sum_i h[i] * (1 - h[i+k]),  
    # ∂/∂h[i] = (1 - h[i+k]) if i+k < n, 0 otherwise.
    return grad

def line_search(h, grad, dx, step_size, n_points):
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Try a few step sizes in step_size * [0.5, 1.0, 2.0]
    for alpha in [0.5*step_size, step_size, 2*step_size]:
        trial_h = h - alpha * grad
        trial_h = project_feasible(trial_h, n_points, dx)
        trial_c5 = evaluate_c5(trial_h, dx)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial solution if available
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent solution if exists; else create default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple construction: boxcar function
        h = np.zeros(n_points)
        half_len = n_points // 2
        h[:half_len] = 1.0
        h = project_feasible(h, n_points, dx)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    step_size = 0.01
    slack = 1.0  # For robustness  
    iter_count = 0

    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx

        # If found better solution, update
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Compute gradient
        grad = compute_gradient(h, max_k, dx)
        # Ensure actual change is in direction of descent, but project first
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n_points, dx)

        # Line search for best step size
        h_new, c5_new = line_search(h, grad, dx, step_size, n_points)

        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Periodically do relaxation / restart with warm start
        if iter_count % 50 == 0:
            # Simple relaxation: add entropy term via small noise
            # This is heuristic to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_feasible(h, n_points, dx)
            
    return (best_h.tolist(), best_c5, n_points)
