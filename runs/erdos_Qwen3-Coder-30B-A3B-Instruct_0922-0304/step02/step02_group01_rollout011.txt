import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        return h
    return h * target_sum / current_sum

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = scale_to_target_sum(h, target_sum)
        return np.clip(h, 0, 1)
    
    population_size = 50
    population = [create_individual() for _ in range(population_size)]
    
    # Evaluate initial population
    def fitness(individual):
        return evaluate_c5(individual, dx)
    
    fitness_scores = [fitness(ind) for ind in population]
    best_idx = np.argmin(fitness_scores)
    best_individual = population[best_idx]
    best_fitness = fitness_scores[best_idx]
    
    # Parameters
    generations = 1000
    elite_size = population_size // 10
    mutation_rate = 0.1
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Check for improvement
        current_best = fitness_scores[0]
        if current_best < best_fitness:
            best_fitness = current_best
            best_individual = population[0].copy()
        
        # Early exit if target achieved
        if best_fitness <= 0.3807:
            break
            
        # Create next generation
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            def tournament_selection(k=3):
                selected_indices = np.random.choice(population_size, k, replace=False)
                selected_fitness = [fitness_scores[i] for i in selected_indices]
                winner_idx = selected_indices[np.argmin(selected_fitness)]
                return population[winner_idx].copy()
            
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                
                # Apply constraints
                child = np.clip(child, 0, 1)
                child = scale_to_target_sum(child, target_sum)
                
                new_population.append(child)
            
            if len(new_population) >= population_size:
                break
        
        # Update population
        population = new_population[:population_size]
        fitness_scores = [fitness(ind) for ind in population]
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
