import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial_h_values as seed for some particles
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for i in range(n_particles // 2):
            # Perturb initial values
            h = h0 + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Random initialization
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best positions
    pbest = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * n_particles
    gbest = None
    gbest_fitness = float('inf')
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            fitness = evaluate_c5(h, dx)
            
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = h.copy()
                
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest = h.copy()
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
        # Adapt inertia weight
        if iteration > 100:
            w = 0.5 + 0.3 * (1.0 - iteration / max_iter)
            
        # Early exit if target reached
        if gbest_fitness <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_fitness, n_points)
