import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
    
    h = enforce_feasibility(h, n_points)
    
    # Define objective
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Define constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimize
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=bounds, 
                   constraints=constraints,
                   options={'maxiter': 1000, 'ftol': 1e-6})
    
    if not res.success:
        # Fallback to simple gradient descent if optimization fails
        h = enforce_feasibility(h, n_points)
        best_c5 = evaluate_overlap(h, dx)
        best_h = h.copy()
        
        # Simple gradient descent with adaptive step size
        step_size = 0.01
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient using finite difference
            grad = np.zeros(n_points)
            eps = 1e-5
            base_c5 = evaluate_overlap(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = enforce_feasibility(h_plus, n_points)
                c5_plus = evaluate_overlap(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            
            # Update
            h_new = h - step_size * grad
            h_new = enforce_feasibility(h_new, n_points)
            
            new_c5 = evaluate_overlap(h_new, dx)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
            else:
                step_size *= 0.95
                
            if best_c5 <= 0.3807:
                break
        return (h.tolist(), best_c5, n_points)
    
    # Return best solution
    final_h = enforce_feasibility(res.x, n_points)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
