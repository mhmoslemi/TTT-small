import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with sawtooth pattern
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative majorization-minimization approach
    max_iter = 1000
    patience = 0
    min_improvement = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # If we've improved significantly, reset patience
        if current_c5 < best_c5 - min_improvement:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # If no improvement in a while, try a different approach
        if patience > 100:
            # Try random restart
            h = np.linspace(0, 1, n_points)
            h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            continue
            
        # Formulate convex subproblem to minimize overlap at k_max 
        h_var = Variable(n_points)
        
        # For the overlap at shift k_max
        # sum_{i=0}^{n-1} h_var[i] * (1 - h_var[(i + k_max) % n])
        
        # Objective: linear in h_var
        obj = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            obj += h_var[i] * (1 - h_var[j])
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Minimize overlap at k_max
        prob = Problem(Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If solver fails, use small gradient step 
                grad = np.zeros_like(h)
                eps = 1e-4
                h_temp = h.copy()
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - current_c5) / eps
                
                step = 0.01
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback to gradient descent approach
            grad = np.zeros_like(h)
            eps = 1e-4
            h_temp = h.copy()
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps
            
            step = 0.01
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    
    # Make sure final result is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
