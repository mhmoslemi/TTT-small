import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, dx, max_k):
    h_padded = np.concatenate([h, h[:max_k]])
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad[i] = (1 - h_padded[i + max_k]) - h_padded[i]
    return grad

def line_search(h, grad, dx, max_k, step_sizes=[1e-3, 1e-2, 1e-1]):
    current_c5 = evaluate_c5(h, dx)
    for alpha in step_sizes:
        h_candidate = h - alpha * grad
        h_candidate = project_feasible(h_candidate, len(h), dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        if candidate_c5 < current_c5:
            return h_candidate, candidate_c5
    return h, current_c5

def convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    h_cvx = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(cp.conv(matrix=1-h, vector=h)) + lambda_reg * cp.sum(cp.entr(h_cvx)))
    constraints = [0 <= h_cvx, h_cvx <= 1, cp.sum(h_cvx) == n_points/2]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    if prob.status == cp.OPTIMAL:
        return np.array(h_cvx.value).flatten()
    else:
        return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the available construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization
    iteration = 0
    lambda_reg = 0.01
    while time.time() - start_time < budget_s - 1 and best_c5 > 0.3807:
        iteration += 1
        
        # Evaluate current overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        max_k_indices = np.where(overlaps == max_overlap)[0]
        max_k = max_k_indices[0] if len(max_k_indices) > 0 else 0
        
        # Compute gradient
        grad = compute_gradient(h, dx, max_k)
        
        # Line search for step size
        h_new, c5_new = line_search(h, grad, dx, max_k)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h = h_new
            
        # Periodic convex relaxation
        if iteration % 50 == 0:
            h_relaxed = convex_relaxation(h, n_points, dx, lambda_reg)
            c5_relaxed = evaluate_c5(h_relaxed, dx)
            if c5_relaxed < best_c5:
                best_c5 = c5_relaxed
                best_h = h_relaxed.copy()
                h = h_relaxed
            else:
                lambda_reg *= 0.5  # Reduce regularization if no improvement
    
    return (best_h.tolist(), best_c5, n_points)
