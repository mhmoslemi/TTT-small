import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute gradient via finite differences
    n_points = len(h)
    eps = 1e-4
    grad = np.zeros(n_points)
    c5_val = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_val) / eps
    
    return grad

def particle_swarm_optimization(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests and global best
    personal_bests = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_scores = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update swarm
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] +
                           c1 * r1 * (personal_bests[i] - swarm[i]) +
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            new_score = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_bests[i] = swarm[i].copy()
                
                # Update global best if needed
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best = swarm[i].copy()
        
        # Periodically apply local refinement
        if iteration % 20 == 0 and iteration > 0:
            # Refine top 10% of particles
            top_indices = np.argsort(personal_best_scores)[:swarm_size//10]
            for idx in top_indices:
                h_refine = personal_bests[idx].copy()
                # Apply gradient ascent
                step_size = 0.01
                for _ in range(50):
                    grad = compute_gradient(h_refine, dx)
                    h_refine = h_refine - step_size * grad
                    h_refine = project_to_feasible(h_refine, n_points)
                    
                    # Check if improved
                    new_score = evaluate_c5(h_refine, dx)
                    if new_score >= personal_best_scores[idx]:
                        break
                    else:
                        personal_best_scores[idx] = new_score
                        personal_bests[idx] = h_refine.copy()
                        
                        if new_score < global_best_score:
                            global_best_score = new_score
                            global_best = h_refine.copy()
        
        # Early exit
        if global_best_score <= 0.3807:
            break
    
    return global_best, global_best_score

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = particle_swarm_optimization(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
