from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with adaptive step size
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        overlap = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlap) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the shift that gives maximum overlap
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Finite difference gradient computation
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            
            # Ensure sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_overlap_plus = np.max(corr_plus) * dx
            
            grad[i] = (max_overlap_plus - current_c5) / eps
            
        # Update with momentum
        velocity = momentum * velocity - step_size * grad
        h = h + velocity
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_overlap = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
