```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region with sum = n_points/2 and h in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with best known construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create an initial feasible solution
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iteration = 0
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap values
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap
        h_shifted = np.roll(h, k_max)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search with different step sizes
        step_sizes = [0.1, 0.5, 1.0]
        best_candidate_c5 = best_c5
        best_candidate_h = h.copy()
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # Update if improvement found
        if best_candidate_c5 < best_c5:
            h = best_candidate_h
            best_c5 = best_candidate_c5
        
        # Occasionally solve convex relaxation
        if iteration % 20 == 0:
            try:
                # Convex relaxation: min sum over k overlap_k + lambda * sum(h * log(h))
                h_relax = cp.Variable(n_points)
                constraints = [h_relax >= 0, h_relax <= 1, cp.sum(h_relax) == n_points/2]
                
                # Approximate the max overlap term as a quadratic penalty
                objective = cp.Minimize(cp.sum(cp.multiply(h_relax, 1 - h_relax)) + 
                                       lambda_reg * cp.sum(cp.neg(cp.entr(h_relax))))
                
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax_val = h_relax.value
                    h_relax_val = np.clip(h_relax_val, 0, 1)
                    h_relax_val = h_relax_val * (n_points/2) / np.sum(h_relax_val)
                    
                    candidate_c5 = evaluate_erdos_solution(h_relax_val)
                    if candidate_c5 < best_c5:
                        h = h_relax_val
                        best_c5 = candidate_c5
                        lambda_reg *= 0.5
                else:
                    lambda_reg *= 0.5
                    
            except Exception:
                lambda_reg *= 0.5
        
        # Early exit if we have a good enough solution
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
