import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, n_points, dx):
    # Compute gradient for shift k_max
    # Shifted versions of h
    h_shifted = np.zeros(n_points)
    h_shifted[k_max:] = h[:-k_max]
    h_shifted[:k_max] = h[-k_max:]
    
    # Gradient components: d/dh_i overlap_k_max
    # overlap_k = sum_j h_j * (1 - h_{j+k}) where j+k can exceed n_points
    # So we use zero-padding for simplicity
    grad = np.zeros(n_points)
    
    # Simpler approximation:
    # For each i, we compute how much changing h[i] changes the overlap
    # Using each position's contribution over all k where h[i] contributes
    # Or just the gradient of the specific max_k component
    
    for i in range(n_points):
        # Contribution depends on the shift k_max
        # For simplicity, assume gradient ∝ (1 - h_{i+k})
        # Compute convolution gradient contribution for specific k_max
        # Since it's just the derivative of h_j * (1 - h_i) where j = i - k
        # This is complex without exact formula, so we just use a reasonable heuristic:
        # Approximate gradient for maximum overlap term
        grad[i] = 1.0 - h[i]
        
    return grad

def line_search(h, grad, n_points, dx, max_alpha=1.0):
    # Simple backtracking line search
    alpha = max_alpha
    original_c5 = evaluate_erdos_solution(h)
    while alpha > 1e-6:
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points, dx)
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < original_c5:
            return h_new, c5_new, alpha
        alpha *= 0.5
    return h, original_c5, 0.0

def convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # A simple L2 regularization to escape local minima
    # We create a smooth version that is easier to optimize
    # This is used as warm-start for a more refined search
    
    # Just apply some smoothing
    h_smooth = np.zeros(n_points)
    for i in range(n_points):
        h_smooth[i] = np.mean(h[max(0,i-2):min(n_points,i+3)])
    h_smooth = project_feasible(h_smooth, n_points, dx)
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from parent solution or create an initial vector
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a reasonable starting point
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = project_feasible(h, n_points, dx)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    iterations = 0
    max_iter = int(budget_s / 0.5)  # Rough estimate
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlaps to identify max
        overlap = correlate(best_h, 1 - best_h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute approximate gradient
        # The gradient is not directly computable due to nonlinearity,
        # but we can perform a simple update based on shift k_max
        grad = np.zeros(n_points)
        # For shift k_max (this can be better, but for now, use simple update)
        for i in range(n_points):
            next_i = (i + k_max) % n_points
            # Gradient of h[i] * (1 - h[i+k]) with respect to h[i]
            grad[i] = 1 - best_h[next_i] 
            # Also q-checks which terms involve h[i], but it's not linear...
            # So we use a reasonable approximation:
            grad[i] = 1 - best_h[next_i]
            
        # Apply learning rate
        h_new = best_h - learning_rate * grad
        
        # Project back to feasible set
        h_new = project_feasible(h_new, n_points, dx)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # Occasionally perform convex relaxation to avoid local minima
        if iteration % 50 == 0 and iteration > 0:
            h_for_relax = convex_relaxation(best_h, n_points, dx)
            # Set a tiny step size for relaxation
            c5_relax = evaluate_erdos_solution(h_for_relax)
            if c5_relax < best_c5:
                best_c5 = c5_relax
                best_h = h_for_relax.copy()
                
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
        
    return (best_h.tolist(), best_c5, n_points)
