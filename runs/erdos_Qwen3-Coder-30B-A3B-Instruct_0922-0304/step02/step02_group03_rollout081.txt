```python
import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Projects h to feasible region: h[i] in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluates C5 = max_k int h(x)(1-h(x+k)) dx"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 30
    max_iter = 1000
    inertia = 0.7
    inertia_decay = 1e-4
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        particles.append(np.ones(n_points))
        velocities.append(np.zeros(n_points))
    
    # Fill remaining particles with blocky patterns + noise
    for i in range(n_particles - len(particles)):
        # Create two blocks
        h = np.zeros(n_points)
        split1 = n_points // 3
        split2 = 2 * n_points // 3
        h[:split1] = 1
        h[split2:] = 1
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO variables
    best_positions = [p.copy() for p in particles]
    best_scores = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_scores)
    global_best_score = best_scores[global_best_idx]
    global_best_pos = best_positions[global_best_idx].copy()
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        inertia *= (1 - inertia_decay)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            personal_best = best_positions[i]
            global_best = global_best_pos
            
            velocities[i] = (inertia * velocities[i] + 
                           alpha * r1 * (personal_best - particles[i]) + 
                           beta * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < best_scores[i]:
                best_scores[i] = score
                best_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_pos = particles[i].copy()
                    global_best_idx = i
    
    return (global_best_pos.tolist(), global_best_score, n_points)
```
