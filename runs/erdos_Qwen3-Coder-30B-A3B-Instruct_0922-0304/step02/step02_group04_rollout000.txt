import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute gradient of C5 with respect to h using finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        # Use initial construction
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
        swarm.append(h_init)
    else:
        # Create base from uniform distribution
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_feasible(h_init, n_points)
        swarm.append(h_init)
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_vals = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            new_c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if new_c5 < personal_best_vals[i]:
                personal_best_vals[i] = new_c5
                personal_best[i] = swarm[i].copy()
            
            # Update global best
            if new_c5 < global_best_val:
                global_best_val = new_c5
                global_best = swarm[i].copy()
                
        # Occasionally apply local refinement to top 20% particles
        if iteration % 20 == 0 and iteration > 0:
            top_indices = np.argsort(personal_best_vals)[:swarm_size // 5]
            for idx in top_indices:
                h = personal_best[idx].copy()
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h, dx)
                    h -= step_size * grad
                    h = project_to_feasible(h, n_points)
                
                # Update if better
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < personal_best_vals[idx]:
                    personal_best_vals[idx] = new_c5
                    personal_best[idx] = h.copy()
                    if new_c5 < global_best_val:
                        global_best_val = new_c5
                        global_best = h.copy()
        
        # Early exit if target achieved
        if global_best_val <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_val, n_points)
