import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    # Computes C5 = max_k int h(x)(1-h(x+k)) dx
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Project h to feasible region: h in [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Parameters
    max_iterations = 2000
    lambda_reg = 0.01  # Regularization weight for convex relaxation
    ls_steps = [0.1, 0.5, 1.0]  # Line search step sizes
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations to find max overlap
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Compute gradient w.r.t. shift k
        # For shift k, gradient = h_shifted - (1 - h_shifted) where h_shifted is h shifted by k
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        current_c5 = max_overlap
        best_h_candidate = h.copy()
        best_c5_candidate = current_c5
        
        for alpha in ls_steps:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            
            if c5_candidate < best_c5_candidate:
                best_c5_candidate = c5_candidate
                best_h_candidate = h_candidate.copy()
        
        # Accept the best candidate
        h = best_h_candidate
        current_c5 = best_c5_candidate
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Use convex relaxation every 20 iterations
        if iteration % 20 == 0 and iteration > 0:
            # Solve convex relaxation
            h_relax = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                h_relax >= 0,
                h_relax <= 1,
                cp.sum(h_relax) == n_points/2
            ]
            
            # Objective: min sum(h_relax * (1 - h_relax)) + lambda_reg * sum(h_relax * log(h_relax))
            # But we need to avoid log(0) so we handle boundary cases
            # We'll simplify by minimizing simple quadratic approximation
            # Here we'll do a simpler convex approximation 
            # Minimize sum over all shifts the approximate overlap + regularization term
            # This is a rough approach since direct convex formulation is complex
            
            # Now solve relaxation using cvxpy with a deterministic quadratic model
            # A simplified version: regularization + quadratic approximation
            obj = cp.Minimize(
                cp.sum_squares(h_relax - 0.5) + lambda_reg * cp.sum(-h_relax * cp.log(h_relax + 1e-12))
            )
            
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_relax_val = h_relax.value
                    h_relax_val = np.clip(h_relax_val, 0, 1)
                    # Rescale to sum = n_points/2
                    h_relax_val = h_relax_val * (n_points/2) / np.sum(h_relax_val)
                    
                    # Check if relaxation improved
                    relaxation_c5 = evaluate_erdos_solution(h_relax_val)
                    if relaxation_c5 < best_c5:
                        best_h = h_relax_val.copy()
                        best_c5 = relaxation_c5
                        h = h_relax_val.copy()
                        
                # Adjust lambda if no improvement
                if prob.status != cp.OPTIMAL:
                    lambda_reg *= 0.5
            except Exception:
                lambda_reg *= 0.5  # Reduce regularization if CVXPY fails
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
