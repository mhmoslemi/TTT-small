import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
    return grad

def solve_quadratic_subproblem(H, g, n_points, eps=1e-6):
    # Solve: min 0.5 * d^T * H * d + g^T * d subject to constraints
    # This is a convex quadratic programming problem with bound constraints
    d = cp.Variable(n_points)
    
    # Objective: minimize quadratic
    objective = cp.Minimize(0.5 * cp.quad_form(d, H) + g.T @ d)
    
    # Constraints
    constraints = [d >= -1e-10, d <= 1 + 1e-10]  # Allow slight violations
    # Sum constraint
    constraints.append(cp.sum(d) == 0)
    
    # Solve QP
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            return d.value
    except:
        pass
    
    # Fallback to basic projection if solver fails
    return np.clip(-g, -1, 1)

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from existing solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create starting individual
        h = np.random.uniform(0.2, 0.8, n_points)
    
    h = project_to_feasible(h, n_points)
    
    # Initialize Hessian approximation (identity for simplicity)
    H = np.eye(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iterations = 1000
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Solve quadratic subproblem
        step = solve_quadratic_subproblem(H, g, n_points)
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        tolerance = 1e-6
        
        for _ in range(20):  # Max backtracking steps
            h_new = h + alpha * step
            h_new = project_to_feasible(h_new, n_points)
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 <= best_c5:
                h = h_new
                break
            else:
                alpha *= 0.5
                if alpha < tolerance:
                    break
        else:
            # If no good alpha found, use last one
            h = h_new
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Update Hessian (approximate using BFGS) - simplification
        # Actually this is a complex approximation; here we simplify
        # For practical purposes, keep identity or use simple update
        if iteration % 10 == 0:
            H = np.eye(n_points)  # Reset or modify based on actual updates if needed
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
