import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try multiple random starting points
    for restart in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize with random perturbation
        h_perturbed = h.copy()
        if restart > 0:
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed = h_perturbed + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
        # Convex optimization approach
        try:
            # Create variable with bounds
            h_var = cp.Variable(n_points, lower_bound=0, upper_bound=1)
            
            # Constraint for integral
            integral_constraint = cp.Variable()
            
            # Objective
            objective = cp.Minimize(integral_constraint)
            
            # Build constraints
            constraints = []
            constraints.append(cp.sum(h_var) == n_points / 2)
            
            # For each lag, add constraint
            for k in range(-n_points + 1, n_points):
                # Align h_var and 1 - h_var with lag k
                if k >= 0:
                    # h[i] * (1 - h[i+k]) for i = 0, ..., n_points-k-1
                    left_indices = np.arange(n_points - k)
                    right_indices = np.arange(k, n_points)
                    # Compute overlap sum
                    if k == 0:
                        overlap_expr = cp.sum(cp.multiply(h_var[left_indices], 1 - h_var[right_indices]))
                    else:
                        overlap_expr = cp.sum(cp.multiply(h_var[left_indices], 1 - h_var[right_indices]))
                    constraints.append(overlap_expr <= integral_constraint)
                else:
                    # h[i+k] * (1 - h[i]) for i = -k, ..., n_points-1
                    left_indices = np.arange(-k, n_points)
                    right_indices = np.arange(n_points + k)
                    overlap_expr = cp.sum(cp.multiply(h_var[left_indices], 1 - h_var[right_indices]))
                    constraints.append(overlap_expr <= integral_constraint)
            
            # Solve with a fast LP solver
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_new = h_var.value
                    if h_new is not None:
                        # Re-normalize to exact constraint
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                        
                        current_c5 = evaluate_c5(h_new, dx)
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = h_new.copy()
            except:
                pass  # Skip if solving failed
            
        except Exception as e:
            pass  # Skip if problem construction failed
            
        # Gradient descent refinement for best solution so far
        if time.time() - start_time > budget_s - 1:
            break
            
        h_refine = best_h.copy()
        for refine_iter in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate overlaps
            corr = correlate(h_refine, 1 - h_refine, mode="full")
            max_overlap = np.max(corr) * dx
            
            # Compute gradient (finite difference)
            grad = np.zeros_like(h_refine)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update step
            step = 0.1
            h_refine -= step * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            
        final_c5 = evaluate_c5(h_refine, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refine.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # One final polishing step
    if best_c5 > 0.3807:
        h_final = best_h.copy()
        original_c5 = best_c5
        num_iter = 1000
        for i in range(num_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Simple local search with random perturbations
            perturb = np.random.normal(0, 0.005, n_points)
            h_candidate = h_final + perturb
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                h_final = h_candidate.copy()
                
            if best_c5 <= 0.3807:
                break

    # Final check if initial solution is better
    if 'initial_h_values' in globals():
        init_c5 = evaluate_c5(np.array(initial_h_values), dx)
        if init_c5 < best_c5:
            best_c5 = init_c5
            best_h = np.array(initial_h_values)
            
    return (best_h.tolist(), best_c5, n_points)
