import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initial construction: rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Add noise to break symmetry
    h += np.random.normal(0, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temp = 0.1 * current_c5
    cooling_rate = 0.999
    min_temp = 1e-6
    
    iteration = 0
    max_iterations = 10000
    gradient_refinement_interval = 50
    gradient_steps = 5
    gradient_step_size = 0.01
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Generate candidate by perturbing current solution
        candidate_h = h + np.random.normal(0, 0.01, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject based on simulated annealing criteria
        if candidate_c5 < current_c5:
            h = candidate_h
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = candidate_h
                current_c5 = candidate_c5
        
        # Cooling schedule
        temp = max(temp * cooling_rate, min_temp)
        
        # Every few iterations, do gradient refinement
        if iteration % gradient_refinement_interval == 0 and iteration > 0:
            # Gradient descent refinement
            for _ in range(gradient_steps):
                eps = 1e-4
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    
                    c5_plus = evaluate_c5(h_plus, dx)
                    c5_minus = evaluate_c5(h_minus, dx)
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                # Update step
                h = h - gradient_step_size * grad
                h = project_to_feasible(h, n_points)
                
                # Update current C5
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
            # Reduce gradient step size
            gradient_step_size *= 0.9
            
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
