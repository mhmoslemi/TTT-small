import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Two-stage optimization: coarse to fine
    n_points_list = [35, 70]
    max_iter_list = [500, 1000]
    step_sizes = [0.1, 0.05]
    
    best_c5 = float('inf')
    best_h = None
    
    for stage, (n_points, max_iter, step_size) in enumerate(zip(n_points_list, max_iter_list, step_sizes)):
        dx = 2.0 / n_points
        
        # Initialize h
        if stage == 0:
            # Coarse stage: simple two-block initialization
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Refinement stage: upsample previous solution
            if best_h is not None:
                h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            
            # Add small random perturbations
            perturbation = np.random.normal(0, 0.05, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        def constraint_bounds(x):
            return np.concatenate([x, 1 - x])
            
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Optimization routine
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # For small n_points, direct optimization with gradient
            # Use L-BFGS-B for better convergence
            # Define objective function
            def obj(h):
                return evaluate_c5(h, dx)
                
            # Define bounds 
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimize
            result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                              options={'maxiter': 1, 'ftol': 1e-10, 'gtol': 1e-5})
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                # Rescale to satisfy integral constraint
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = h * (n_points / 2) / np.sum(h)
                
            # Evaluate current solution
            current_c5 = evaluate_c5(h, dx)
            
            # Early termination
            if current_c5 <= 0.3807:
                break
                
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        if time.time() - start_time > budget_s - 1:
            break
    
    if best_h is None:
        # Fallback
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = evaluate_c5(h, 2.0/n_points)
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, len(best_h))
