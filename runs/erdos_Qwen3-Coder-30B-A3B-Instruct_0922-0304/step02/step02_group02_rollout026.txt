import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum constraint and bounds"""
    # Force sum to be n_points/2
    h = h * (n_points / 2) / np.sum(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using structured symmetric approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Symmetric initialization pattern
        h = np.zeros(n_points)
        segment_size = n_points // 5  # 14 points per segment
        
        for i in range(5):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size
            # Alternate values within segment
            for j in range(start_idx, end_idx):
                if (j - start_idx) % 2 == 0:
                    h[j] = 0.6
                else:
                    h[j] = 0.4
        h = project_to_feasible(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define objective function for optimization
    def objective(h_vec):
        h = h_vec.copy()
        h = project_to_feasible(h, n_points)
        return evaluate_c5(h, dx)
    
    # Define constraint function for sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    max_iter = 500
    lr = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Perform optimization step
        try:
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                            options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8})
            if result.success:
                h = result.x
            else:
                h = h * 0.999
        except:
            h = h * 0.999
            
        h = project_to_feasible(h, n_points)
        
        # Additional local search
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            h_try = h + np.random.normal(0, 0.01, n_points)
            h_try = project_to_feasible(h_try, n_points)
            try:
                c5_try = evaluate_c5(h_try, dx)
                if c5_try < current_c5:
                    h = h_try
                    current_c5 = c5_try
            except:
                pass

    # Final evaluation
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
