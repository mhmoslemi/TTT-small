import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraints(h, target_sum):
    # Scale to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-scale to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h using structured approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_constraints(h, target_sum)
    else:
        # Structured initialization
        h = np.ones(n_points)
        # Use 5-fold symmetry and alternating pattern
        segment_size = n_points // 5
        for i in range(5):
            start = i * segment_size
            end = start + segment_size
            if i % 2 == 0:
                h[start:end] = 0.6
            else:
                h[start:end] = 0.4
        h = project_to_constraints(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with L-BFGS-B
    def objective(h_vec):
        h = h_vec.copy()
        h = project_to_constraints(h, target_sum)
        return evaluate_c5(h, dx)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = project_to_constraints(h, target_sum)
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_c5 = evaluate_c5(h, dx)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
        return grad
    
    # Line search parameters
    max_iter = 1000
    learning_rate = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Optimize using L-BFGS-B
        # Construct bounds [0,1] for all variables
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use minimize with L-BFGS-B
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                          jac=gradient, options={'maxiter': 10})
        
        if result.success:
            h = result.x.copy()
            h = project_to_constraints(h, target_sum)
        else:
            # Fallback: basic gradient descent with projection
            grad = gradient(h)
            h = h - learning_rate * grad
            h = project_to_constraints(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
