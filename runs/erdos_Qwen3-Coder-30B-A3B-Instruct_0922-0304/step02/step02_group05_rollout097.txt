import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    # Evaluate the C5 constant for given h
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize PSO parameters
    n_particles = 50
    w = 1.0  # inertia weight
    c1 = 2.0  # cognitive coefficient
    c2 = 2.0  # social coefficient
    max_velocity = 0.2
    perturb_std = 0.05
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create initial solution
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Add random particles
    while len(particles) < n_particles:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for p in particles:
        v = np.random.normal(0, 0.01, n_points)
        velocities.append(v.copy())
    
    # Evaluate initial solutions
    fitness = []
    for h in particles:
        c5 = evaluate_erdos_solution(h)
        fitness.append(c5)
    
    # Particle best
    p_best = [p.copy() for p in particles]
    p_best_fitness = fitness.copy()
    
    # Global best
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_fitness
    
    # PSO iterations
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i])
            
            # Update particle best
            if c5 < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = c5
            
            # Update global best
            if c5 < g_best_fitness:
                g_best = particles[i].copy()
                g_best_fitness = c5
                last_improvement = iteration
                if g_best_fitness < best_c5:
                    best_c5 = g_best_fitness
                    best_h = g_best.copy()
        
        # Adaptive inertia
        if iteration % 50 == 0:
            w = max(0.8, w - 0.02)
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
        
        # Reset stagnant particles (if no improvement for 50 iterations)
        if iteration - last_improvement > 50:
            for i in range(n_particles):
                if np.random.rand() < 0.3:  # Randomly reset 30% of particles
                    h = np.random.rand(n_points)
                    h = project_feasible(h, n_points, dx)
                    particles[i] = h.copy()
                    velocities[i] = np.random.normal(0, 0.01, n_points)
                    c5 = evaluate_erdos_solution(particles[i])
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = c5
                    if c5 < g_best_fitness:
                        g_best = particles[i].copy()
                        g_best_fitness = c5
                        last_improvement = iteration
        
        # Occasionally add random perturbation
        if iteration % 10 == 0:
            for i in range(n_particles):
                velocities[i] += np.random.normal(0, perturb_std, n_points)
        
        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Ensure best_h is feasible
    best_h = project_feasible(best_h, n_points, dx)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
