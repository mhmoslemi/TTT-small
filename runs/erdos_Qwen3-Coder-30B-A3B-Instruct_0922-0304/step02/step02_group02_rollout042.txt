import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    max_iter = 500
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
        
    for _ in range(pop_size):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(pop_size)]
    
    # PSO parameters
    w = 0.7
    c1, c2 = 2.0, 2.0
    
    # Track global best
    global_best_c5 = float('inf')
    global_best_h = None
    
    # Track particle bests
    particle_best_h = [h.copy() for h in swarm]
    particle_best_c5 = [evaluate_c5(h, dx) for h in swarm]
    
    # Local search indices
    local_search_fraction = 0.05
    local_search_count = max(1, int(pop_size * local_search_fraction))
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (particle_best_h[i] - swarm[i]) + 
                            c2 * r2 * (global_best_h - swarm[i]) if global_best_h is not None else velocities[i])
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Repair and evaluate
            swarm[i] = repair_h(swarm[i], n_points)
            current_c5 = evaluate_c5(swarm[i], dx)
            
            # Update particle best
            if current_c5 < particle_best_c5[i]:
                particle_best_c5[i] = current_c5
                particle_best_h[i] = swarm[i].copy()
            
            # Update global best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = swarm[i].copy()
                
            # Early exit
            if global_best_c5 <= 0.3807:
                break
        
        # Local search on top particles
        if iteration % 10 == 0:
            sorted_indices = np.argsort(particle_best_c5)
            top_particles = sorted_indices[:local_search_count]
            for idx in top_particles:
                h = particle_best_h[idx]
                # Gradient descent step with finite difference
                eps = 1e-4
                grad = np.zeros_like(h)
                base_c5 = particle_best_c5[idx]
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[j] = (plus_c5 - base_c5) / eps
                h = h - 0.01 * grad
                h = repair_h(h, n_points)
                particle_best_h[idx] = h.copy()
                particle_best_c5[idx] = evaluate_c5(h, dx)
    
    # Final evaluation
    if global_best_h is not None:
        final_c5 = global_best_c5
        final_h = global_best_h
    else:
        final_h = base_h
        final_c5 = evaluate_c5(base_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
