import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and bound constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.clip(h, 0, 1)
    
    h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Get the shift corresponding to maximum overlap
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        if shift >= 0:
            for i in range(n_points - shift):
                grad[i] += (1 - h[i + shift])
            for i in range(shift, n_points):
                grad[i] -= h[i - shift]
        else:
            shift_abs = -shift
            for i in range(n_points - shift_abs):
                grad[i] += (1 - h[i + shift_abs])
            for i in range(shift_abs, n_points):
                grad[i] -= h[i - shift_abs]
        
        # Update with adaptive step size
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project_h(h_new, target_sum)
        
        # Check if descent is beneficial
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            h = h_new
        else:
            # Try smaller step
            step_size *= 0.5
            h_new = h - step_size * grad
            h_new = project_h(h_new, target_sum)
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
