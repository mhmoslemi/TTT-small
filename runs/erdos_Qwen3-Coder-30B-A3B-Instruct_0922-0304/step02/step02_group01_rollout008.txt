import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Attempt convex optimization approach
        try:
            # Define convex relaxation variables
            h_vars = cp.Variable(n_points)
            
            # Constraint: sum(h) = n_points/2
            sum_constraint = cp.sum(h_vars) == n_points / 2
            
            # Constraint: 0 <= h <= 1
            bound_constraints = [h_vars >= 0, h_vars <= 1]
            
            # Create auxiliary variables for overlap terms
            # We use the convex relaxation:
            # h[i] * (1 - h[j]) <= t_ij, where t_ij are auxiliary variables
            
            # For our specific case, we want to bound the maximum overlap
            # Use a variable t to bound all overlaps
            
            t = cp.Variable()
            
            # For the overlap integral, we can model it with relaxation by Lagrangian approach
            # But for simplicity and efficiency, we will solve a constrained optimization:
            
            # Just reformulate the problem as an LP with a convex hull approximation
            # Let's do a hybrid approach: use convex optimization for first few steps
            # Then local optimization if needed
            
            # This part will just be constrained to enroll in the solver
            # Create constraints using cvxpy and solve to see the improvement
            constraints = bound_constraints + [sum_constraint]
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=cp.GLOP, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                if h_new is not None:
                    # Clip and normalize
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
                    continue  # Restart loop to get next best
                
        except:
            pass  # Fall back to gradient descent
            
        # Local improvement using gradient descent
        step = 0.01
        iterations = 100
        h_temp = h.copy()
        
        for _ in range(iterations):
            # Compute overlaps
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update h
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp
        
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
