import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h."""
    # Ensure h is properly normalized
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Relaxation parameters
    lambda_reg = 0.01
    relaxation_interval = 20
    max_iters = int(budget_s * 10)  # Rough estimate
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find the maximal one
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        # Approximate gradient w.r.t. to the maximal overlap
        h_shifted = np.roll(h, max_idx - (len(overlaps) // 2), axis=0)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search with step sizes
        step_sizes = [0.1, 0.5, 1.0]
        best_h_candidate = h
        best_candidate_c5 = best_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_h_candidate = h_candidate
                best_candidate_c5 = candidate_c5
                break
                
        # Update
        h = best_h_candidate
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = h.copy()
            
        # Periodic relaxation to escape local minima
        if iteration % relaxation_interval == 0 and iteration > 0:
            try:
                # Convex relaxation using CVXPY
                h_vars = cp.Variable(n_points)
                obj = cp.Minimize(
                    cp.max(cp.conv,cp.conv(h_vars, 1-h_vars)) + 
                    lambda_reg * cp.sum(cp.entr(h_vars))
                )
                constraints = [
                    h_vars >= 0,
                    h_vars <= 1,
                    cp.sum(h_vars) == n_points/2
                ]
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax = h_vars.value
                    h_relax = project_feasible(h_relax, n_points, dx)
                    c5_relax = evaluate_erdos_solution(h_relax)
                    if c5_relax < best_c5:
                        best_c5 = c5_relax
                        best_h = h_relax.copy()
                        h = h_relax.copy()
                    lambda_reg *= 0.5  # Decrease regularization if effective
            except:
                pass  # If convex optimization fails, proceed with gradient descent
                
    return (best_h.tolist(), best_c5, n_points)
