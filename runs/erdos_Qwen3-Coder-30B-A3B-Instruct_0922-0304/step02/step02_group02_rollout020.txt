import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing Parameters
    T0 = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 10000
    perturb_frac = 0.1
    delta = 0.1
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    T = T0
    iteration = 0
    
    while T > T_min and (time.time() - start_time) < budget_s - 1:
        if iteration >= max_iter:
            break
            
        # Create candidate solution
        h_candidate = h.copy()
        
        # Select indices to perturb
        n_perturb = max(1, int(perturb_frac * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Add delta to selected indices, subtract proportionally from others
        total_add = delta * n_perturb
        h_candidate[indices] += delta
        # Distribute reduction among non-perturbed indices
        non_perturb_indices = np.setdiff1d(np.arange(n_points), indices)
        if len(non_perturb_indices) > 0:
            reduction_per_idx = total_add / len(non_perturb_indices)
            h_candidate[non_perturb_indices] -= reduction_per_idx
            h_candidate[non_perturb_indices] = np.clip(h_candidate[non_perturb_indices], 0, 1)
        
        # Normalize to maintain sum constraint
        sum_h = np.sum(h_candidate)
        h_candidate = h_candidate * (n_points / 2) / sum_h
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Evaluate
        current_c5 = evaluate_c5(h, dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Metropolis criterion
        delta_E = candidate_c5 - current_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_candidate.copy()
            
        # Update best
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Cooling schedule
        T = alpha * T
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
