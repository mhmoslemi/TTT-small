import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import CVXOPT

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create starting individual
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
    
    h = project_to_feasible(h_init, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Convex optimization approach
    solver = CVXOPT
    relaxation_iters = 100
    local_iters = 50
    
    for iter_count in range(relaxation_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve relaxed problem
        h_var = cp.Variable(n_points)
        C = cp.Variable(1)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, C >= 0]
        
        # Add overlap constraints
        corr = correlate(h_var.value, 1 - h_var.value, mode="full")
        for i, val in enumerate(corr):
            constraints.append(val <= C)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver)
            if prob.status == 'optimal':
                h_relaxed = h_var.value
                h_projected = project_to_feasible(h_relaxed, n_points)
                
                # Local refinement
                for _ in range(local_iters):
                    if time.time() - start_time > budget_s - 1:
                        break
                    corr = correlate(h_projected, 1 - h_projected, mode="full")
                    c5_val = np.max(corr) * dx
                    
                    # Simple gradient estimation using finite differences
                    grad = np.zeros_like(h_projected)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_projected.copy()
                        h_plus[i] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_val) / eps
                    
                    # Gradient descent step
                    h_projected = h_projected - 0.005 * grad
                    h_projected = project_to_feasible(h_projected, n_points)
                
                c5_refined = evaluate_c5(h_projected, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_projected.copy()
        except:
            pass
            
        # Continue with regular step size for diversity
        # Simulated annealing like step
        h_new = h + np.random.normal(0, 0.01, n_points)
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
