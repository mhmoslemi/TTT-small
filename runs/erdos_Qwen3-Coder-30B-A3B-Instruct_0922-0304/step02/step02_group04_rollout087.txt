import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, norm

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_relaxation(h0, n_points, dx):
    # Create decision variables
    h = Variable(n_points)
    C = Variable()
    
    # Build overlap constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # Simulate full convolution constraints (simplified)
    # For each possible delay k, add constraint
    h_full = np.concatenate([h0, np.zeros(n_points-1)])
    h_full_rev = np.concatenate([np.zeros(n_points-1), h0])
    corr_full = correlate(h_full, 1 - h_full_rev, mode='full')
    
    # Only add constraints for key positions to reduce problem size
    valid_positions = range(len(corr_full) - n_points + 1)
    for k in valid_positions[::10]:  # Subsample for efficiency
        overlap_val = corr_full[k]
        constraints.append(overlap_val <= C)
    
    # Objective: minimize C
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
    except:
        return h0.copy()
    
    if prob.status == 'optimal':
        h_new = np.array(h.value).reshape(-1)
        return project_to_feasible(h_new, n_points)
    return h0.copy()

def local_search(h, n_points, dx, iterations=50):
    h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    for i in range(iterations):
        h_trial = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.02)
        h_trial[idx] += delta
        h_trial = project_to_feasible(h_trial, n_points)
        
        c5_trial = evaluate_c5(h_trial, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
    
    h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Iterate with convex relaxation + local search
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and best_c5 > 0.3807:
        iter_count += 1
        # Convex relaxation
        h_relaxed = solve_convex_relaxation(h, n_points, dx)
        
        # Local search on relaxed solution
        h_refined, c5_refined = local_search(h_relaxed, n_points, dx)
        
        # Update if found better
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
        
        # Next h is current refined solution
        h = h_refined
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
