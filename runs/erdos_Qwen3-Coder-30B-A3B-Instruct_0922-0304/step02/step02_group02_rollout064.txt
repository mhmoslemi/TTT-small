import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a smoothed step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define optimization variables
    h_var = cp.Variable(n_points)
    
    # Relaxation schedule
    mu_values = np.logspace(np.log10(0.5), 0, 50)  # From 0.5 to 1.0
    
    for mu in mu_values:
        # Updated constraint with relaxation
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Compute correlation terms using CVXPY
        # For optimization, this is a complex reformulation
        # We'll solve instructions in a more efficient fashion
        # Since rearranging complex objective might be difficult, 
        # use the same simple gradient descent with soft restarts
        
        # Direct gradient based optimization with few steps and LOGIC restarts
        current_h = h_var.value if h_var.value is not None else h.copy()
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Local gradient descent
        learning_rate = 0.01
        max_iter = 100
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            
            current_c5 = evaluate_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            if best_c5 <= 0.3807:
                break
            
            # Gradient estimation
            eps = 1e-4
            grad = np.zeros_like(current_h)
            base_c5 = evaluate_c5(current_h, dx)
            
            for j in range(n_points):
                h_plus = current_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
            
            # Update with learning rate
            current_h = current_h - learning_rate * grad
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        # update h for next run
        h = current_h.copy()

        # warmth for next relaxation level
        h = h * (n_points / 2) / np.sum(h)
        
        # Logical early termination
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
