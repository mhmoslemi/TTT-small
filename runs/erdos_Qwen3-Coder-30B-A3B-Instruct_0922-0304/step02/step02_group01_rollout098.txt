import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the maximum overlap shift
        overlap = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient for the maximum overlap shift
        grad = np.zeros_like(h)
        if max_shift >= 0:
            # h[i] * (1 - h[i + max_shift]) for i = 0..n_points-max_shift-1
            for i in range(n_points - max_shift):
                grad[i] += 1 - h[i + max_shift]
                if i + max_shift < n_points:
                    grad[i + max_shift] -= h[i]
        else:
            # h[i - max_shift] * (1 - h[i]) for i = -max_shift..n_points-1
            for i in range(-max_shift, n_points):
                grad[i] -= h[i + max_shift]
                if i + max_shift >= 0:
                    grad[i + max_shift] += 1 - h[i]
        
        # Update with momentum and step size
        step = 0.01
        h = h - step * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
