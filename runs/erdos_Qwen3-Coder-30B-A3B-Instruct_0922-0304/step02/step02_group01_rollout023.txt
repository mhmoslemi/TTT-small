import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and bounds"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, k_star, dx):
    """Compute gradient for the maximum overlap shift k_star"""
    n = len(h)
    grad = np.zeros(n)
    
    # For shift k_star, compute gradient of overlap
    if k_star >= 0:
        # h[i] * (1 - h[i + k_star]) terms
        for i in range(n - k_star):
            grad[i] += 1 - h[i + k_star]
            if i + k_star < n:
                grad[i + k_star] -= h[i]
    else:
        # k_star < 0, so h[i + k_star] * (1 - h[i])
        for i in range(-k_star, n):
            grad[i] -= 1 - h[i + k_star]
            if i + k_star >= 0:
                grad[i + k_star] += h[i]
    
    return grad * dx

    start_time = time.time()

        h = project_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    moment = 0.9
    step_size = 0.01
    
    recent_improvements = 0
    last_improvement_iter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            recent_improvements = 0
            last_improvement_iter = iteration
        else:
            recent_improvements += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Stop if no improvement in a while
        if recent_improvements > 100:
            break
            
        # Compute overlaps for all shifts and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        # Convert from full correlation index to actual shift k
        k_star = max_idx - (n_points - 1)
        
        # Compute gradient at the worst shift
        grad = compute_gradient(h, k_star, dx)
        
        # Update h with momentum
        h_new = h - step_size * grad
        h_new = project_h(h_new, target_sum)
        h = moment * h + (1 - moment) * h_new
        
        # Project again to ensure feasibility
        h = project_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
