import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Equality constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    # Optimization function
    def objective(h):
        return evaluate_c5(h, dx)
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use scipy.optimize.minimize with SLSQP
        try:
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            if result.success:
                h = result.x
                # Ensure h stays in bounds
                h = np.clip(h, 0, 1)
                # Correct the sum constraint
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If optimization fails, restart with perturbed solution
                h = best_h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback to random perturbation
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
