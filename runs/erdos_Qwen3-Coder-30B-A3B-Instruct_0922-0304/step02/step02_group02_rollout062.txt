```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Relaxation parameters
    n_steps = 50
    mu_values = np.logspace(np.log10(0.5), 0, n_steps)  # From 0.5 to 1.0
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create and solve convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective: weighted sum of C5 and a penalty term
        # We'll compute C5 in a way that's different from the direct approach
        # Use relaxation to handle the max operation
        # Create constraints for each shift k
        C5_terms = []
        for k in range(-n_points+1, n_points):  # all valid shifts
            start_idx = max(0, -k)
            end_idx = min(n_points, n_points - k)
            if end_idx > start_idx:
                # Compute the term for shift k
                term = cp.sum(cp.multiply(h_var[start_idx:end_idx], 1 - h_var[start_idx+k:end_idx+k]))
                C5_terms.append(term)
        
        # We want to minimize the maximum of all terms
        max_term = cp.Variable()
        constraints = [
            max_term >= term for term in C5_terms
        ]
        
        # Add constraints to ensure sum = 35 and bounds
        constraints.extend([
            cp.sum(h_var) == 35,
            h_var >= 0,
            h_var <= 1
        ])
        
        # Define objective
        obj = mu * max_term + (1 - mu) * cp.sum((h_var - 0.5)**2) / n_points
        
        # Formulate and solve the problem
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h = h_sol
            else:
                # If no solution or infeasible, fallback to previous
                pass
        except:
            pass
            
        # Normalize and clip
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
