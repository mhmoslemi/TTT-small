import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    # Ensure h[i] in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    P = 50  # population size
    max_iter = 500
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
    base_h = base_h * (n_points / 2) / np.sum(base_h)

    for _ in range(P):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, n_points)
        swarm.append(h.copy())
    
    velocities = [np.zeros(n_points) for _ in range(P)]
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 2.0
    best_individual = [None] * P
    best_individual_c5 = [float('inf')] * P
    best_global = None
    best_global_c5 = float('inf')
    
    # Initialize bests
    for i, h in enumerate(swarm):
        c5 = evaluate_c5(h, dx)
        best_individual[i] = h.copy()
        best_individual_c5[i] = c5
        if c5 < best_global_c5:
            best_global = h.copy()
            best_global_c5 = c5
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(P):
            # Update velocity
            r1, r2 = np.random.random(2)
            if best_global is not None:
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (best_individual[i] - swarm[i]) + 
                                c2 * r2 * (best_global - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Repair
            swarm[i] = repair_h(swarm[i], n_points)
            
            # Evaluate
            c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if c5 < best_individual_c5[i]:
                best_individual[i] = swarm[i].copy()
                best_individual_c5[i] = c5
                
            # Update global best
            if c5 < best_global_c5:
                best_global = swarm[i].copy()
                best_global_c5 = c5
                
            # Early exit if target reached
            if best_global_c5 <= 0.3807:
                break

        # Apply local search on best 5%  
        if iteration > 100 and time.time() - start_time < budget_s - 1:
            sorted_indices = np.argsort(best_individual_c5)
            top_count = max(1, P // 20)
            for i in range(top_count):
                idx = sorted_indices[i]
                h = best_individual[idx]
                # Gradient descent on top particles
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    eps = 1e-4
                    grad = np.zeros_like(h)
                    base_c5 = evaluate_c5(h, dx)
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        plus_c5 = evaluate_c5(h_plus, dx)
                        grad[j] = (plus_c5 - base_c5) / eps
                    h = h - 0.01 * grad
                    h = repair_h(h, n_points)
                    c5 = evaluate_c5(h, dx)
                    best_individual[idx] = h.copy()
                    best_individual_c5[idx] = c5
                    if c5 < best_global_c5:
                        best_global = h.copy()
                        best_global_c5 = c5
                    if c5 <= 0.3807:
                        break

    return (best_global.tolist(), best_global_c5, n_points)
