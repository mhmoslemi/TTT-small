import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute gradient of C5 with respect to h
    # This is a simplified finite difference approximation
    eps = 1e-5
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def particle_swarm_optimization(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    personal_best_positions = []
    personal_best_scores = []
    global_best_position = None
    global_best_score = float('inf')
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        personal_best_positions.append(h.copy())
        score = evaluate_c5(h, dx)
        personal_best_scores.append(score)
        if score < global_best_score:
            global_best_score = score
            global_best_position = h.copy()
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        personal_best_positions.append(h.copy())
        score = evaluate_c5(h, dx)
        personal_best_scores.append(score)
        if score < global_best_score:
            global_best_score = score
            global_best_position = h.copy()
    
    # Add more random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        personal_best_positions.append(h.copy())
        score = evaluate_c5(h, dx)
        personal_best_scores.append(score)
        if score < global_best_score:
            global_best_score = score
            global_best_position = h.copy()
    
    # Initialize velocities
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(swarm_size)]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best_position - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new position
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = particles[i].copy()
                
        # Local refinement every 20 iterations
        if iter_num % 20 == 0 and iter_num > 0:
            top_particles = np.argsort(personal_best_scores)[:swarm_size // 5]
            for i in top_particles:
                h_ref = personal_best_positions[i].copy()
                step_size = 0.01
                for _ in range(50):
                    grad = compute_gradient(h_ref, dx)
                    h_ref -= step_size * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                new_score = evaluate_c5(h_ref, dx)
                if new_score < personal_best_scores[i]:
                    personal_best_scores[i] = new_score
                    personal_best_positions[i] = h_ref.copy()
                    if new_score < global_best_score:
                        global_best_score = new_score
                        global_best_position = h_ref.copy()
                        
        # Early exit if target is reached
        if global_best_score <= 0.3807:
            break
    
    return global_best_position, global_best_score

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = particle_swarm_optimization(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
