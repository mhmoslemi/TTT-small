import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian parameters
    rho = 1.0
    lambda_k = np.zeros(2 * n_points - 1)  # Lagrange multipliers
    tau = np.zeros(2 * n_points - 1)       # Constraint thresholds
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Solve convex subproblem using cvxpy
        h_var = cp.Variable(n_points)
        
        # Create the augmented Lagrangian objective
        # We want to minimize the overlap subject to constraints
        # The AL term: sum over k of [overlap_k + (rho/2) * (overlap_k - tau_k)^2] + lambda_k * (overlap_k - tau_k)
        # This is a reformulation to make it convex in h_var
        
        # For simplicity, we introduce a reference constraint that limits overlap
        # We can directly penalize max overlap in a simpler convex way
        objective = cp.Minimize(cp.max(cp.multiply(corr, dx)) + 
                                rho * cp.sum_squares(cp.multiply(corr, dx) - tau)) + \
                    cp.sum(cp.multiply(lambda_k, cp.multiply(corr, dx) - tau))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve the convex problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == "optimal":
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Update h
                h = h_new
                
                # Update multipliers and thresholds based on constraint violation
                corr_new = correlate(h, 1 - h, mode="full")
                current_overlap = corr_new * dx
                
                # Update Lagrange multipliers
                lambda_k += rho * (current_overlap - tau)
                
                # Update thresholds (this keeps the constraints tight)
                tau = current_overlap
                
                # Increase penalty if necessary
                if np.max(np.abs(current_overlap - tau)) > 1e-3:
                    rho *= 2.0
            else:
                # If solver fails, try a gradient decent approach
                # This step is necessary to make the algorithm robust
                step = 0.001
                h_temp = h.copy()
                eps = 1e-4
                grad = np.zeros_like(h_temp)
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_temp = np.max(corr_temp) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
            
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except Exception:
            # Fallback: gradient descent approach for robustness
            step = 0.001
            h_temp = h.copy()
            eps = 1e-4
            grad = np.zeros_like(h_temp)
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                grad[i] = (c5_plus - c5_temp) / eps

            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp

    return (best_h.tolist(), best_c5, n_points)
