import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as starting point for some particles
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    for i in range(n_particles):
        if i < n_particles // 2:
            # Start from initial solution with small noise
            h = base_h + np.random.normal(0, 0.05, n_points)
        else:
            # Random initialization
            h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Initialize best positions
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w = 0.8  # inertia weight
    c1 = c2 = 1.2  # cognitive and social weights
    
    max_iter = 200
    mutation_rate = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            v = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            particles[i] = h
            velocities[i] = v
            
            # Evaluate
            score = evaluate_c5(h, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h.copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = h.copy()
                    gbest_score = score
        
        # Mutation to escape local optima
        for i in range(n_particles):
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = np.clip(particles[i] + noise, 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                
        # Early stopping
        if gbest_score <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
