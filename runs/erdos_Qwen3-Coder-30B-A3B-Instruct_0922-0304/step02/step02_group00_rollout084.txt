import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial construction for some particles
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        for i in range(n_particles // 2):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    else:
        # Random initialization
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w_init = 0.8
    c1, c2 = 1.2, 1.2
    mutation_rate = 0.05
    
    # Track bests
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    best_c5 = gbest_score
    best_h = gbest.copy()
    
    # Iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_init * (1 - iteration / max_iter)  # dynamic inertia
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v +
                 c1 * r1 * (p - h) +
                 c2 * r2 * (gbest - h))
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            particles[i] = h_new
            
            # Evaluate new position
            score = evaluate_c5(h_new, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = score
                
                # Check global best
                if score < gbest_score:
                    gbest = h_new.copy()
                    gbest_score = score
                    
                    if gbest_score < best_c5:
                        best_c5 = gbest_score
                        best_h = gbest.copy()

            # Early exit
            if best_c5 <= 0.3807:
                break
                
    return (best_h.tolist(), best_c5, n_points)
