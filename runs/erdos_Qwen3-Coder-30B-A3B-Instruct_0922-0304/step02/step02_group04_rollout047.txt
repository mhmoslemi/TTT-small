import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n_points = len(h)
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def line_search_step(h, d, dx, max_alpha=1.0, armijo_constant=1e-4, beta=0.5):
    # Armijo backtracking line search
    alpha = max_alpha
    base_c5 = evaluate_c5(h, dx)
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 <= base_c5 - armijo_constant * alpha * np.dot(d, d):
            return alpha
        alpha *= beta
    return alpha

def trust_region_step(h, dx, grad, H, delta=0.1):
    # Simple trust region step - search within a small region

    # Try a few steps in gradient direction
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    best_step = np.zeros_like(h)
    
    # Try a few candidates in direction of gradient
    for step_size in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        candidate_h = h - step_size * grad
        candidate_h = project_to_feasible(candidate_h, n_points)
        c5_candidate = evaluate_c5(candidate_h, dx)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = candidate_h.copy()
            best_step = -step_size * grad
    
    return best_h, best_c5, best_step
