import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    F = 0.7
    CR = 0.9
    gradient_step_size = 0.01
    gradient_freq = 50
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Create new population through DE
        new_population = population[:population_size//2]  # Elitism
        
        while len(new_population) < population_size:
            # Select three distinct individuals
            a, b, c = random.sample(range(population_size//2), 3)
            a, b, c = population[a], population[b], population[c]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = project_to_feasible(mutant, n_points)
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, population[0])
            
            # Selection
            trial = project_to_feasible(trial, n_points)
            trial_c5 = evaluate_c5(trial, dx)
            old_c5 = evaluate_c5(population[0], dx)
            
            if trial_c5 < old_c5:
                new_population.append(trial)
            else:
                new_population.append(population[0])
        
        population = new_population[:population_size]
        
        # Apply gradient descent refinement periodically
        if gen % gradient_freq == 0 and gen > 0:
            h_refined = best_solution.copy()
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                c5_old = evaluate_c5(h_refined, dx)
                
                # Gradient via finite differences
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    h_minus = h_refined.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    c5_minus = evaluate_c5(h_minus, dx)
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                # Update with gradient descent
                h_refined = h_refined - gradient_step_size * grad
                h_refined = project_to_feasible(h_refined, n_points)
                
                # Check if we've improved
                c5_new = evaluate_c5(h_refined, dx)
                if c5_new < c5_old:
                    best_solution = h_refined.copy()
                    if c5_new < best_c5:
                        best_c5 = c5_new
            
            # Ensure we're tracking the best solution
            if evaluate_c5(best_solution, dx) < best_c5:
                best_c5 = evaluate_c5(best_solution, dx)
            
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = differential_evolution(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
