import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Two-stage optimization: coarse then fine
    n_points_list = [35, 70]
    max_iter_list = [500, 1000]
    step_sizes = [0.1, 0.05]
    
    best_c5 = float('inf')
    best_h = None
    
    for stage, (n_points, max_iter, step_size) in enumerate(zip(n_points_list, max_iter_list, step_sizes)):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h with two blocks if available, otherwise uniform blocks
        if 'initial_h_values' in globals() and stage == 0:
            h = np.array(initial_h_values)
            h = h[:n_points]
            if len(h) < n_points:
                h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
            h = np.clip(h, 0, 1)
        else:
            # Uniform blocks approach
            h = np.ones(n_points)
            h[n_points//2:] = 0
            
        # Rescale to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # For coarse stage, add slight randomness
        if stage == 0:
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Optimization using L-BFGS-B
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_c5(h, dx)
            
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                              options={'ftol': 1e-9, 'gtol': 1e-5, 'maxiter': 10})
            
            if result.success:
                h = result.x.copy()
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                
                # Early termination if target met
                if current_c5 <= 0.3807:
                    break
                    
            # Adapt step-size towards the end of iteration
            if _ > max_iter * 0.8:
                step_size *= 0.95
                
        # Perturb and refine for fine phase
        if stage == 0 and n_points == 35:
            # Upsample to 70 points
            h_upsampled = np.interp(np.linspace(0, 2, 70), np.linspace(0, 2, 35), h)
            h = h_upsampled
            
            # Add small random perturbations
            perturb = np.random.normal(0, 0.05, 70)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (70 / 2) / np.sum(h)
    
    # Final refinement of best solution
    if best_h is not None:
        n_points_final = len(best_h)
        dx_final = 2.0 / n_points_final
        h_final = best_h.copy()
        
        def objective_final(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_final / 2) / np.sum(h)
            return evaluate_c5(h, dx_final)
            
        def constraint_final(h_vec):
            return np.sum(h_vec) - n_points_final / 2
            
        cons_final = {'type': 'eq', 'fun': constraint_final}
        bounds_final = [(0, 1) for _ in range(n_points_final)]
        
        # Final optimization
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(objective_final, h_final, method='L-BFGS-B', bounds=bounds_final, constraints=cons_final, 
                              options={'ftol': 1e-9, 'gtol': 1e-5, 'maxiter': 10})
            
            if result.success:
                h_final = result.x.copy()
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points_final / 2) / np.sum(h_final)
                
                current_c5 = evaluate_c5(h_final, dx_final)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    
            # Early termination if target met
            if current_c5 <= 0.3807:
                break
    
    return (h_final.tolist(), best_c5, n_points_final)
