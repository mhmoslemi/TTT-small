import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    # Compute gradient via finite differences
    n_points = len(h)
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def simulated_annealing(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    cooling_rate = 0.999
    initial_temp = 0.1
    noise_std = 0.01
    grad_steps = 5
    grad_step_size = 0.01
    
    # Load or create initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
    
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    temperature = initial_temp * current_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate solution
        candidate_h = h + np.random.normal(0, noise_std, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject candidate
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / max(temperature, 1e-10)):
            h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
                # Early exit if target reached
                if best_c5 <= 0.3807:
                    break
        
        # Gradual cooling
        temperature *= cooling_rate
        
        # Periodic gradient descent refinement
        if iteration % 50 == 0:
            for _ in range(grad_steps):
                grad = compute_gradient(h, dx)
                h = h - grad_step_size * grad
                h = project_to_feasible(h, n_points)
                current_c5 = evaluate_c5(h, dx)
                
            # Check if improvement
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = simulated_annealing(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
