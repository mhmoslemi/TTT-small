import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian parameters
    rho = 1.0
    lambda_k = np.zeros(2 * n_points - 1)  # Lagrange multipliers
    
    max_iter = 500
    
    # ADMM-like optimization with convex subproblems
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find max
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Formulate convex subproblem using CVXPY
        h_var = Variable(n_points)
        
        # Objective: minimize maximum overlap with augmented Lagrangian
        # The AL term is quadratic in the constraint violation
        objective = cp.Minimize(
            cp_sum(cp.multiply(lambda_k, corr)) +
            0.5 * rho * cp_sum(cp.square(corr - np.max(corr)))
        )
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve convex subproblem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
            if h_new is not None:
                h = h_new
        except:
            # If optimization fails, keep current h 
            pass
            
        # Update Lagrange multipliers
        corr_new = correlate(h, 1 - h, mode="full")
        lambda_k += rho * (corr_new - np.max(corr_new))
        
        # Increase penalty if needed
        if np.max(corr_new) > np.max(corr):
            rho *= 1.1
            
        # Maintain sum and bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new result
        corr_final = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr_final) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
