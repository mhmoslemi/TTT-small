import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    max_iter = 1000
    patience = 0
    min_patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Formulate convex subproblem to minimize overlap at k_max
        h_var = cp.Variable(n_points)
        
        # Build constraint matrix for overlap term at shift k_max
        # This is a linear function of h
        # overlap at shift k_max = sum_{i} h[i] * (1 - h[i + k_max])  
        # We can linearize it for the convex subproblem by fixing the current h
        # But for Benders-type approach we can formulate directly as a linear program
        # Objective: minimize sum over i of h_var[i] * (1 - h[i + k_max])
        
        # Build linear approximation 
        # We will compute overlap from 0 to n_points-1 directly
        # In our model, h[i + k_max] wraps around, 
        # so we worry about i + k_max mod n_points
        idxs = (np.arange(n_points) + k_max) % n_points
        obj = cp.sum(cp.multiply(h_var, 1 - h[idxs]))
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Formulate and solve
        problem = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                if h_new is not None:
                    h = h_new.copy()
            else:
                # Fall back to gradient ascent
                pass
        except:
            pass

        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience >= min_patience:
            # Reseed with better starting point
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.linspace(0, 1, n_points)
                h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            patience = 0
            
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
