import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, n_points, dx):
    # Approximate gradient: d/dh[i] overlap_k_max
    # overlap_k = sum h[i](1 - h[i+k]) = sum h[i] - sum h[i] h[i+k]
    # d/dh[i] overlap_k = 1 - h[i+k] - h[i+k] = 1 - 2*h[i+k] (when i+k < n_points)
    # But we don't have i+k < n_points check, so to be safe: zero padding
    # So it's more correct to compute all overlap shifts and take max
    # For specific k_shift = k_max, compute gradient
    # gradient[i] = d/dh[i] overlap_k_max
    # If k is such that i+k < n_points, then derivative is (1 - h[i+k]) - h[i+k] = (1 - 2*h[i+k])
    # But the derivative also considers earlier terms. Better: convolution gradient
    # The overlap with shift k is sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k]) 
    # Note: when we zero-pad, h[i+k] = 0 if i+k >= n_points
    # For shift k, derivative w.r.t. h[i] = 1 - h[i+k] if i+k < n_points, else 0
    # So i can see h[i+k] in the sum over index i, so derivative is (1 - h[i+k])
    # But that's not right either. Let's compute it properly by finite differences
    # Gradient at i = derivative of max overlap for that k with respect to h[i]
    # Per shift k, overlap = sum h[i] * (1-h[i+k]) = sum h[i] - sum h[i]*h[i+k]
    # d/dh[i] of this = with respect to h[i], so it becomes:
    # if k == 0: (1 - h[i]) from first sum, -h[i] from second sum = 1 - 2*h[i] (wrong)
    # For better clarity: Let's just compute the gradient by noting that:
    # For a shift k:
    # overlap_k = sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k]) 
    # d/dh[i] overlap_k = 1 - h[i+k] if i+k < n_points, else 0.
    # We are interested in the gradient at the k achieving max overlap
    
    # describe gradient for shift k:
    # Only valid if i+k < n_points (at the end we get zero padding)
    # So the gradient with respect to h[i] of overlap_k is:
    # if (i+k) is in [0, n_points), then 1 - h[i+k], else 0.
    # So we compute the gradient, normalized
    grad = np.zeros_like(h)
    # Equivalent to (1 - h) shifted right by k
    shifted = np.roll(1 - h, k_max)
    grad = shifted[:len(grad)]
    return grad

def line_search(h, grad, k_max, n_points, dx, max_alpha=1.0):
    # Try different alpha values
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for alpha in np.logspace(-3, 0, 10):
        if alpha > max_alpha:
            continue
        h_trial = h - alpha * grad
        h_trial = project_feasible(h_trial, n_points, dx)
        c5 = evaluate_c5(h_trial, dx)
        if c5 < best_c5:
            best_h = h_trial
            best_c5 = c5
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_size = 0.01
    step_count = 0
    relaxation_interval = 50
    
    while time.time() - start_time < budget_s - 1:
        # Incremental gradient descent approach
        # Evaluate all overlaps and find max
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k]
        max_c5 = max_overlap * dx
        
        # Compute gradient at max_k
        grad = compute_gradient(h, max_k, n_points, dx)
        
        # Apply update step
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n_points, dx)
        
        # Perform line search
        h_best, c5_best = line_search(h, grad, max_k, n_points, dx, max_alpha=0.2)
        
        step_count += 1
        if c5_best < best_c5:
            best_h = h_best
            best_c5 = c5_best
            h = h_best.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Periodic relaxation steps
        if step_count % relaxation_interval == 0:
            # Apply convex relaxation (small entropy penalty)
            # This will make it smoother, help escape local minima
            try:
                # Add small entropy penalty: minimize overlap + 0.01 * entropy(h)
                # Gradient of entropy = -np.log(h) - 1 
                # But h might be 0. Do 0 log 0 = 0, so safe to compute
                smooth_h = h + 1e-6  # avoid pole
                entropy_grad = -np.log(smooth_h) - 1
                h_new = h - step_size * (grad + 0.01 * entropy_grad)
                h_new = project_feasible(h_new, n_points, dx)
                test_c5 = evaluate_c5(h_new, dx)
                if test_c5 < best_c5:
                    h = h_new
                    best_c5 = test_c5
            except:
                pass  # ignore relax errors  
                
        # Check budget timeout
        if time.time() - start_time > budget_s - 1:
            break
            
    return (best_h.tolist(), best_c5, n_points)
