import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, multiply
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style majorization-minimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Formulate convex subproblem to reduce overlap at k_max
        h_var = Variable(n_points)
        
        # Build the linear objective for the specific k_max overlap
        # The overlap at shift k_max is sum h[i] * (1 - h[(i + k_max) % n_points])
        objective = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            objective += h_var[i] * (1 - h_var[j])
            
        # Constraints for subproblem
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve with ECOS default solver
        try:
            prob = Problem(Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = np.array(h_var.value).flatten()
        except:
            # If solver fails, take a small gradient step
            grad = np.zeros_like(h)
            eps = 1e-4
            corr_current = correlate(h, 1 - h, mode="full")
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                grad[i] = (np.max(corr_plus) * dx - current_c5) / eps
            step = 0.01
            h_new = h - step * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new.copy()
        
        # Maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
