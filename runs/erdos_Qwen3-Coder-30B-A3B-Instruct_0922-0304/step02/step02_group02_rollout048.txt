import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 300000
    perturb_size = 0.1
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Record current state
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Generate candidate solution
        h_new = h.copy()
        # Randomly select indices to perturb
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Add small perturbation
        delta = np.random.normal(0, perturb_size)
        h_new[indices] += delta
        
        # Ensure sum constraint (scale to match sum = 35)
        total_sum = np.sum(h_new)
        if total_sum > 0:
            scale_factor = 35.0 / total_sum
            h_new = h_new * scale_factor
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Recompute overlap for new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        delta_E = new_c5 - current_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new.copy()
        
        # Update temperature
        T = max(T_min, alpha * T)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
