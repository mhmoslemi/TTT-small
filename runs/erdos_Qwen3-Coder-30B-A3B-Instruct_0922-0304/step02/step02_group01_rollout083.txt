import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, Maximize

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try convex relaxation approach for better global search
        try:
            # Use a more systematic convex optimization
            h_vars = Variable(n_points)
            t = Variable(1)
            
            # Constraints
            constraints = [0 <= h_vars, h_vars <= 1, cp_sum(h_vars) == n_points / 2, t >= 0]
            
            # Add overlap constraints
            for k in range(-n_points+1, n_points):
                if k == 0:
                    overlap_expr = 0
                    for i in range(n_points):
                        overlap_expr += h_vars[i] * (1 - h_vars[i])
                elif k > 0:
                    overlap_expr = 0
                    for i in range(n_points - k):
                        overlap_expr += h_vars[i] * (1 - h_vars[i + k])
                else:  # k < 0
                    overlap_expr = 0
                    for i in range(-k, n_points):
                        overlap_expr += h_vars[i] * (1 - h_vars[i + k])
                
                # Relax bilinear terms
                # Use convex hull relaxation
                constraints.append(overlap_expr <= t)
            
            # Objective
            objective = Minimize(t)
            prob = Problem(objective, constraints)
            
            # Try multiple solvers
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.PDLP, verbose=False)
                except:
                    prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = np.array(h_vars.value).flatten()
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
        except Exception as e:
            # If convex optimization fails, fall back to gradient method
            step = 0.01
            iterations = 50
            h_temp = h.copy()
            
            for _ in range(iterations):
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Update h
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
            
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
