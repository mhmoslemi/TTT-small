import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Homotopy continuation parameters
    mu_values = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    max_iter_per_mu = 20
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create and solve relaxed convex problem
        h_var = cp.Variable(n_points)
        h_obj = cp.Minimize(
            mu * cp.max(cp.multiply(cp.conv(h_var, 1 - h_var, mode="full"), dx)) +
            (1 - mu) * cp.sum_squares(h_var - 0.5) / n_points
        )
        h_constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        h_prob = cp.Problem(h_obj, h_constraints)
        
        try:
            h_prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if h_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Ensure constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # Fallback to gradient descent if optimization fails
                eps = 1e-4
                grad = np.zeros_like(h)
                base_c5 = evaluate_c5(h, dx)
                
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[i] = (plus_c5 - base_c5) / eps
                
                h = h - 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient descent if optimization fails
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Local optimization steps
        for _ in range(max_iter_per_mu):
            if time.time() - start_time > budget_s - 1:
                break
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            
            h = h - 0.5 * 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate and update best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
