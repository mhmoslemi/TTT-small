import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5

        if best_c5 <= 0.3807:

        try:
            # Create variables
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # For each lag, add overlap constraints
            max_k = n_points - 1
            t = cp.Variable()
            
            # Iterate through lags
            for k in range(-max_k, max_k + 1):
                if k == 0:
                    continue
                # Compute the overlap sum for specific lag k
                # For positive k: sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
                # For negative k: sum_{i=0}^{n+k-1} h[i+k] * (1-h[i])
                if k > 0:

                    expr = 0

                        expr += h_vars[i] * (1 - h_vars[i + k])
                else:

                    expr = 0
                    for i in range(n_points + k):
                        expr += h_vars[i + k] * (1 - h_vars[i])
                
                # Add constraint expr <= t * dx
                constraints.append(expr <= t)
            
            # Objective
            objective = cp.Minimize(t)
            prob = cp.Problem(objective, constraints)

            try:
                prob.solve(solver=cp.GLPK, verbose=False)

        step = 0.01

            corr = correlate(h_temp, 1 - h_temp, mode="full")
