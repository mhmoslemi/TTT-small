import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create an initial guess - linear ramp
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing parameters
    T = 1.0
    beta = 0.995
    epsilon = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cooling schedule tracking
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Current C5
        current_c5 = evaluate_c5(h, dx)
        
        if iteration % 100 == 0:
            T = max(0.001, T * beta)
        
        # Generate candidate
        h_candidate = h + np.random.normal(0, epsilon * T, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Rescale to correct sum
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C5
        new_c5 = evaluate_c5(h_candidate, dx)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate
        
        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Early exit when target reached
        if best_c5 <= 0.3807:
            break
            
        # Reset temperature if no improvement for 1000 iterations
        if iteration - last_improvement > 1000:
            T = 1.0
            last_improvement = iteration
    
    return (best_h.tolist(), best_c5, n_points)
