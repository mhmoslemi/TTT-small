import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hyperparameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    patience = 100
    best_iter = 0
    
    # Initialize momentum
    velocity = np.zeros_like(h)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_iter = iteration
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the most constraining shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient with respect to h for this shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Check if we can shift k from position i
            if 0 <= i + max_k < n_points and i + max_k >= 0:
                if max_k >= 0:
                    grad[i] = -(1 - h[i + max_k])
                else:
                    grad[i] = -(1 - h[i + max_k])
            else:
                grad[i] = 0
            
        # Apply momentum
        velocity = momentum * velocity - step_size * grad
        h = h + velocity
        
        # Project h onto feasible region: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check for early stopping based on patience
        if iteration - best_iter > patience:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
