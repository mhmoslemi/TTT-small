import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import log

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h."""
    # Correlation with 1-h
    overlap = correlate(h, 1 - h, mode="full")
    # Multiply by dx to get the actual integral
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def solve_convex_relaxation(h, n_points):
    """Solve a convex relaxation to escape local minima."""
    # Use CVXPY to minimize a regularized version of overlap
    h_var = cp.Variable(n_points)
    # Simple convex objective with entropy regularization
    obj = cp.Minimize(
        cp.max(cp.convolve(h_var, 1 - h_var, mode="full")) +
        0.01 * cp.sum(cp.entr(h_var) + cp.log(1 - h_var))
    )
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h_var.value
    except:
        pass
    # If optimization fails, return original
    return h
