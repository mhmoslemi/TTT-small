import numpy as np
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant C5 for given step function h"""
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Initialize temperature and best solution
    T = T_initial
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement_count = 0

    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Generate candidate solution
        h_candidate = h + np.random.normal(0, epsilon * T, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate overlap
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject the candidate
        delta_c5 = candidate_c5 - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
            
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Decrease temperature periodically
        if iteration % 100 == 0:
            T = beta * T

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)

    return (best_h.tolist(), final_c5, n_points)
