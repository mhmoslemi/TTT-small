import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial_h_values to seed 25 particles
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        for _ in range(25):
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Initialize with random values
        for _ in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Best positions
    pbest = [p.copy() for p in particles]
    pbest_vals = [float('inf')] * n_particles
    gbest = None
    gbest_val = float('inf')
    
    # PSO parameters
    inertia_w = 0.8
    cognitive_w = 1.2
    social_w = 1.2
    mutation_rate = 0.05
    
    # Iterate
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            c5 = evaluate_c5(h, dx)
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = h.copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest = h.copy()
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            v = (inertia_w * v + 
                 cognitive_w * r1 * (pbest[i] - h) + 
                 social_w * r2 * (gbest - h))
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
        # Update inertia weight for later iterations
        if iteration > 100:
            inertia_w = 0.5 + (0.8 - 0.5) * (1 - iteration / max_iter)
        
        # Early exit
        if gbest_val <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_val, n_points)
