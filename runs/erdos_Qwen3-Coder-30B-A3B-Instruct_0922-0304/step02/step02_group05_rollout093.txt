import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        scaling = (n_points / 2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Main optimization loop
    iteration = 0
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current objective and find max overlap shift
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_k = np.argmax(overlap)
        
        # Compute gradient estimate
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search with step sizes
        step_sizes = [0.1, 0.5, 1.0]
        current_c5 = best_c5
        
        for step_size in step_sizes:
            h_candidate = h - step_size * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < current_c5:
                h = h_candidate
                current_c5 = candidate_c5
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                break
        
        # Periodically improve using convex relaxation
        if iteration % 20 == 0:
            # Solve relaxed problem using cvxpy
            h_relax = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.max(cp.convolve(h_relax, 1 - h_relax, mode='full')) + 
                lambda_reg * cp.sum(cp.entr(h_relax) + cp.log(2) - h_relax)
            )
            constraints = [
                h_relax >= 0,
                h_relax <= 1,
                cp.sum(h_relax) == n_points / 2
            ]
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_relax_sol = h_relax.value
                    h_candidate = project_feasible(h_relax_sol, n_points, dx)
                    candidate_c5 = evaluate_erdos_solution(h_candidate)
                    
                    if candidate_c5 < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = candidate_c5
                        h = h_candidate
                        
                    lambda_reg *= 0.8
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
