import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop with more sophisticated approach
    max_iter = 5000
    momentum = 0.9
    step_size = 0.01
    patience = 100
    no_improvement_count = 0
    
    # Store previous solution for momentum
    prev_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > patience:
            # Reset with a better diversification
            h = best_h.copy()
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            continue
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        
        # Find the shift that maximizes overlap
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (len(h) - 1)
        
        # Compute gradient with finite differences more carefully
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute gradient w.r.t. each component
        for i in range(n_points):
            # Perturb h[i]
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            
            # Recompute overlap with the perturbed value
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            # Find max overlap in shifted version
            max_idx_plus = np.argmax(overlap_plus)
            max_shift_plus = max_idx_plus - (len(h_plus) - 1)
            
            # The gradient is based on the sensitivity of max overlap to small changes
            # Compute the maximum overlap difference for this point
            max_val_plus = overlap_plus[max_idx_plus] * dx
            max_val_current = overlap[max_idx] * dx
            
            # Only use finite differences when shift is still relevant
            if max_shift_plus == max_shift:
                grad[i] = (max_val_plus - max_val_current) / eps
            else:
                # Gradient is zero if the maximum shift changes with perturbation
                grad[i] = 0.0
        
        # Update with momentum
        delta_h = momentum * (h - prev_h) - step_size * grad
        h = h + delta_h
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update previous for momentum
        prev_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
