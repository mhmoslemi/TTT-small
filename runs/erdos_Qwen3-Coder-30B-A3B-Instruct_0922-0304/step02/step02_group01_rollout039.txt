import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def normalize(h, target_sum):
    """Normalize h to have sum equal to target_sum"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    return h * target_sum / current_sum

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize(h, target_sum)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Parameters
    generations = 1000
    elite_size = population_size // 10
    mutation_rate = 0.1
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (lower C5 is better)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, h))
            
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        
        # Update best solution
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected = [h for _, h in fitness[:population_size//2]]
        
        # Elitism: keep best individuals
        elites = [h for _, h in fitness[:elite_size]]
        
        # Create new population
        new_population = elites[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            
            # Crossover (single point)
            crossover_point = random.randint(0, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Enforce bounds
            child = np.clip(child, 0, 1)
            
            # Normalize
            child = normalize(child, target_sum)
            
            new_population.append(child)
            
        population = new_population[:population_size]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global start_time
    start_time = time.time()
    
    n_points = 70
    best_h, best_c5 = genetic_algorithm(seed, budget_s, n_points)
    
    # Final validation
    dx = 2.0 / n_points
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
