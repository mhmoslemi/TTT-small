import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using 5-fold symmetry if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create 5-fold symmetric initial guess
        h = np.zeros(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.7
            else:
                h[start_idx:end_idx] = 0.3
        # Ensure sum is exactly 35
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Define objective function for optimization
        def obj_func(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_scaled = h_clipped * (n_points / 2) / np.sum(h_clipped)
            return evaluate_c5(h_scaled, dx)
        
        # Define constraint function for sum constraint
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        # Create constraint dictionary
        constraint = {'type': 'eq', 'fun': constraint_func}
        
        # Perform optimization step
        result = minimize(
            obj_func,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 10, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        h = result.x
        
        # Fix constraints after optimization
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
