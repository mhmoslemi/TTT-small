import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Convex optimization approach for better convergence
        # Relax the bilinear terms using convex hull relaxation
        h_cvx = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_cvx >= 0, h_cvx <= 1, cp_sum(h_cvx) == n_points / 2, t >= 0]
        
        # Add overlap constraints with convex hull relaxation
        # For each time lag k, we approximate h[i] * (1 - h[j]) with linear constraints
        for k in range(-n_points+1, n_points):
            # Calculate overlap at lag k
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0 to n_points-k-1
                for i in range(n_points - k):
                    # Relax h[i] * (1 - h[i+k]) <= t
                    # Using conic relaxation: h[i] * (1 - h[i+k]) <= h[i] + h[i+k] - h[i]*h[i+k]
                    # But since we want h[i] * (1 - h[i+k]) <= t, we need to formulate properly
                    # h[i] * (1 - h[i+k]) <= 1 - h[i+k]
                    # h[i] * (1 - h[i+k]) <= h[i]
                    # H[i] * (1 - H[j]) >= 0
                    # Actually, the simpler form: 1 - h[i+k] >= h[i] * (1 - h[i+k]) 
                    # and h[i] >= h[i] * (1 - h[i+k]) 
                    # OR use the multiplication relaxation:
                    # h[i]*h[j] >= 0, h[i] - h[i]*h[j] >= h[i] * (1 - h[j])
                    # h[j] - h[i]*h[j] >= h[j] * (1 - h[i])
                    # We're more interested in h[i] * (1 - h[i+k]) <= t for the max
                    # This means h[i] * (1 - h[i+k]) <= 1 - h[i+k] (if h[i+k]<=1)
                    # Also h[i] * (1 - h[i+k]) <= h[i] (if h[i]>=0)
                    # Also h[i] * (1 - h[i+k]) >= 0 (bounds)
                    if k == 0:  # This case isn't relevant, since h[i] * (1 - h[i]) appears in correlations
                        continue
                    # Add linear inequality h[i] - h[i]*h[i+k] <= t
                    # Which is same as h[i] - h[i+k] - (h[i] - h[i]) * h[i+k] <= t  
                    # Relaxation:
                    constraints.append(h_cvx[i] - h_cvx[i+k] <= t)
        
        # Add all unique constraints for validation and use
        # Use a different way: approximate all correlation terms
        # Actually, no need to relax all of them. Approximate just the ones 
        # that bound the max overlap.
        # Solve convex problem that approximates the optimization
        # Keep it simple with just the sum constraint, and all snapped to valid solutions
        # Approach: solve using the specific constraint that is likely critical 
        problem = Problem(Minimize(t), constraints)
        try:
            # Use a solver that is likely to be available
            problem.solve(solver=cp.GLPK, verbose=False)
            if problem.status in ["optimal", "optimal_inaccurate"]:
                # Updated h from solution
                h_new = np.array(h_cvx.value)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new.copy()
            else:
                # Fall back to gradient descent step if optimization fails
                step = 0.01
                iterations = 50
                h_temp = h.copy()
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback if convex optimization fails
            step = 0.01
            iterations = 50
            h_temp = h.copy()
            for _ in range(iterations):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
