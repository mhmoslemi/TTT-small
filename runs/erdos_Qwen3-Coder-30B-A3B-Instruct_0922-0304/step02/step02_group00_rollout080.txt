import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    if 'initial_h_values' in globals():
        # Use initial construction as seed for some particles
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        for i in range(n_particles // 2):
            particles.append(h_init.copy())
            # Add small noise
            noise = np.random.normal(0, 0.05, n_points)
            h_noisy = h_init + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
            particles.append(h_noisy)
    else:
        for i in range(n_particles):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
    
    # PSO parameters
    inertia = 0.8
    c1 = c2 = 1.2
    mutation_rate = 0.05
    
    # Initialize personal bests
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight (diminishing)
        if iteration > 100:
            inertia = 0.6
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            # Scale to correct sum
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    
        # Apply mutation
        for i in range(n_particles):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                score = evaluate_c5(particles[i], dx)
                if score < pbest_scores[i]:
                    pbest[i] = particles[i].copy()
                    pbest_scores[i] = score
                    if score < gbest_score:
                        gbest = particles[i].copy()
                        gbest_score = score
                        
        # Early exit condition
        if gbest_score <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
