import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with 5-fold symmetric construction
    h = np.zeros(n_points)
    segment_size = n_points // 5
    for i in range(5):
        start_idx = i * segment_size
        end_idx = start_idx + segment_size
        if i % 2 == 0:
            h[start_idx:end_idx] = 0.7
        else:
            h[start_idx:end_idx] = 0.3
    
    # Adjust the sum to exactly match the constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use L-BFGS-B optimization
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perform optimization step
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                         constraints=cons, options={'maxiter': 10})
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Fallback to simple gradient descent on max component
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Gradient approximation with finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            h = h - 0.005 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check for termination
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
