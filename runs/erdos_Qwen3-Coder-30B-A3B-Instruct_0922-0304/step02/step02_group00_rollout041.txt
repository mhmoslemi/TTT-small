import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Use a simple initial construction
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian parameters
    rho = 1.0
    lambda_mult = np.zeros(2 * n_points - 1)
    tau = np.zeros(2 * n_points - 1)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Convex optimization subproblem
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Objective is the overlap w.r.t. current k_max
        # This is approximate since we optimize one constraint at a time
        # Use AL approach with fixed lambda for now
        overlap_expr = cp.Variable(2 * n_points - 1)
        
        # Create a simpler objective: minimize the maximum overlap
        # Solve convex relaxation using barrier method
        obj = cp.Minimize(
            cp.max(cp.multiply(lambda_mult, corr)) + 
            0.5 * rho * cp.sum_squares(h_var - h)
        )
        
        # Simplified approach with just one constraint for now
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            pass
        
        # Update Lagrange multipliers
        current_overlap = correlate(h, 1 - h, mode="full")
        lambda_mult += rho * (current_overlap - tau)
        
        # Update penalty if needed
        if np.max(np.abs(current_overlap - tau)) > 1e-3:
            rho *= 1.1
            
        # Maintain sum and bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
