import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    h = enforce_feasibility(h, n_points)
    
    # Define the objective function
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Define the constraint for the integral
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimize using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    constraint_obj = {'type': 'eq', 'fun': constraint}
