import time

        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Stochastic gradient descent approach
        step_size = 0.01
        inner_iter = 50
        
        h_candidate = h.copy()
        for _ in range(inner_iter):
            # Compute overlap gradient via finite differences
            grad = np.zeros_like(h_candidate)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_candidate.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps
                
            # Update h_candidate
            h_candidate -= step_size * grad
            h_candidate = np.clip(h_candidate, 0, 1)

        # Accept new solution if better
        corr_new = correlate(h_candidate, 1 - h_candidate, mode="full")

            best_h = h.copy()
            
        # Early termination if target achieved
