import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    F = 0.7
    CR = 0.9
    gd_freq = 50
    gd_iterations = 50
    gd_step_size = 0.01
    eps = 1e-4
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Create new population through DE
        new_population = population[:population_size//4]  # Elite preservation
        
        for i in range(population_size//4, population_size):
            # DE/rand-to-best/1/bin
            a, b, c = random.sample(range(population_size//4), 3)
            a, b, c = population[a], population[b], population[c]
            
            # Mutation
            mutant = a + F * (population[0] - c)  # Best individual as target
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, population[i])
            
            # Ensure feasibility
            trial = project_to_feasible(trial, n_points)
            
            # Selection
            trial_c5 = evaluate_c5(trial, dx)
            old_c5 = fitness[i]
            
            if trial_c5 < old_c5:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Apply gradient descent to best individual periodically
        if gen % gd_freq == 0 and gen > 0:
            h_best = population[0].copy()
            for _ in range(gd_iterations):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                c5_val = evaluate_c5(h_best, dx)
                grad = np.zeros_like(h_best)
                
                for i in range(n_points):
                    h_plus = h_best.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    h_neg = h_best.copy()
                    h_neg[i] -= eps
                    h_neg = project_to_feasible(h_neg, n_points)
                    c5_neg = evaluate_c5(h_neg, dx)
                    
                    grad[i] = (c5_plus - c5_neg) / (2 * eps)
                
                h_best = h_best - gd_step_size * grad
                h_best = project_to_feasible(h_best, n_points)
            
            # Replace worst individual with refined solution
            population[-1] = h_best
            
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = differential_evolution(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
