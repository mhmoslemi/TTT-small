import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_shift):

    h_shifted = np.roll(h, k_shift)
    # gradient[i] = (1 - h[(i-k) % N]) - h[(i-k) % N]

    # Let's do actual derivative w.r.t h:
    # For each i: partial / partial h[i]  (sum_j h[j] * (1 - h[j+k])) 
    # = (1 - h[i+k])
    # BONUS: Actually need to consider two terms:
    # \sum_j h[j] (1-h[j+k])  
    # For i: partial = (1 - h[i+k])  (this is consistent)
    # gradient[i] for overlap_k = (1 - h[i+k]) where +k gives offset
    # where k = k_shift in correlation indices
    # We are told correlation of h, 1-h, mode full.

    # The center for full is 0, so shift k is peak at k+1 in full.
    # That overlaps with shifting h by k.  
    # Thus at h[i], it contributes to overlap at shift k due to h[i]*1-h[i+k].

    # Gradient at each i: (1 - h[(i+k)%n] ) - (0 + h[i+k]) = 1 - 2 * h[i+k]
    # No wait again:
    # two terms 1*h[i] and - h[i]*h[i+k]
    # derivative of first term gives 1. derivative of second - h[i+k]
    # So gradient is indeed 1 - h[i+k]

    # For overlap at shift k, we compute gradient at index k in full convolution.
    # correlation of h with (1-h) at full, peak occurs at index k+1
    # i corresponds to i in h
    # such that h[i]* (1-h[i-(k+1)]) contributes to overlap
    # This is incorrect logics 
    # Let's recompute from scratch.
    # Given overlap_k = sum_i h[i] * (1 - h[i+k]) 
    # By index i, derivative w.r.t h[i] = (1 - h[i+k])
    # so gradient = (1 - h[i+k]) for i in range(n)
    # Problem! This assumes we can vary all h[i], but:
    # 1. Correlation is cyclic, so h[i+k] may wrap around
    # 2. For full: h[0],...,h[n-1], and 1-h[0],...,1-h[n-1] (zero-padded to 2n-1 length)
    # overlap_k(i) = sum_{j} h[j] * (1 - h[j+k]) if (j+k) is valid 
    # That is, if we pick h[i], it will contribute to overlap_k via i+k to h.
    # Let's wrap using one-hot to simplify.
    # Actually, let's just compute it as Scala/Python code and test correctness uding finite diff.

    # There is only 1 non-zero entry in the final gradient stack
    # Gradient at index i = (1 - h[(i+k_shift) % n]) where the overlap shift k_shift corresponds to

    h_shifted = np.roll(h, -k_shift)

def line_search(h, gradient, c5_before, dx, n_points):
    steps = [0.1, 0.5, 1.0]
    for alpha in steps:
        h_candidate = h - alpha * gradient
        h_candidate = project_feasible(h_candidate, n_points, dx)
        c5_candidate = evaluate_c5(h_candidate, dx)
        if c5_candidate < c5_before:
            h = h_candidate
            c5_before = c5_candidate
            break
    return h, c5_before

        max_k = np.argmax(h_corr)
