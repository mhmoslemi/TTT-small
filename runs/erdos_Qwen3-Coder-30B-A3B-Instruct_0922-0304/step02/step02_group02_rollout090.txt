import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy the integral constraint"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def initialize_swarm(n_points, n_particles, initial_h):
    swarm = []
    for _ in range(n_particles):
        # Start from initial_h and add small noise
        h = initial_h + np.random.normal(0, 0.05 * np.max(initial_h), n_points)
        h = repair_h(h, n_points)
        swarm.append(h)
    return swarm

def pso_optimization(seed, budget_s, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize PSO parameters
    n_particles = 50
    max_iter = 500
    inertia_w = 0.7
    c1, c2 = 2.0, 2.0
    learning_rate = 0.01
    
    # Get initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Initialize swarm
    swarm = initialize_swarm(n_points, n_particles, initial_h)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    personal_best = [h.copy() for h in swarm]
    personal_best_scores = [evaluate_c5(h, 2.0/n_points) for h in swarm]
    
    global_best = initial_h.copy()
    global_best_score = evaluate_c5(initial_h, 2.0/n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = repair_h(swarm[i], n_points)
            
            # Evaluate and update personal best
            current_score = evaluate_c5(swarm[i], 2.0/n_points)
            if current_score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = current_score
                
            # Update global best
            if current_score < global_best_score:
                global_best = swarm[i].copy()
                global_best_score = current_score
                
        # Early exit
        if global_best_score <= 0.3807:
            break
    
    # Local search on top particles
    top_particles = sorted(range(n_particles), key=lambda i: personal_best_scores[i])[:n_particles//10]
    for i in top_particles:
        h = personal_best[i].copy()
        dx = 2.0 / n_points
        
        # Simple gradient-based refinement
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            eps = 1e-4
            grad = np.zeros(n_points)
            base_c5 = evaluate_c5(h, dx)
            
            # Compute finite difference gradient
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
                
            # Update step
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            
        # Update personal best if better
        new_score = evaluate_c5(h, dx)
        if new_score < personal_best_scores[i]:
            personal_best[i] = h.copy()
            personal_best_scores[i] = new_score
            if new_score < global_best_score:
                global_best = h.copy()
                global_best_score = new_score
    
    # Final check
    final_score = evaluate_c5(global_best, 2.0/n_points)
    return (global_best.tolist(), final_score, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return pso_optimization(seed, budget_s)
