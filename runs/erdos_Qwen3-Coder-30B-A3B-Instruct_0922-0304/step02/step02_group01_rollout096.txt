import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop with adaptive step sizes and momentum
    max_iter = 5000
    momentum = 0.9
    step_size = 0.01
    
    v = np.zeros(n_points)  # Velocity for momentum
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_full)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for the maximum overlap shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient contribution from shift max_k
            if 0 <= i + max_k < n_points and i + max_k >= 0:
                grad[i] += (1 - h[i + max_k])
            if 0 <= i - max_k < n_points and i - max_k >= 0:
                grad[i] += h[i - max_k]
        
        # Update with momentum and adaptive step size
        v = momentum * v - step_size * grad
        h_new = h + v
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Line search to find reasonable step size
        for _ in range(10):
            h_test = h_new.copy()
            test_c5 = evaluate_c5(h_test, dx)
            if test_c5 < current_c5:
                h = h_test.copy()
                break
            step_size *= 0.5
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # If no improvement, reduce step size
        if time.time() - start_time > budget_s - 2:
            break
            
        h = h_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
