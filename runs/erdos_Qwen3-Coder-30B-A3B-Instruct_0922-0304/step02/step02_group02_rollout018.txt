import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Clip values and rescale to meet sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # equivalent to integral constraint
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        # Create symmetric base
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    # Create initial swarm
    for _ in range(n_particles):
        h = base_h.copy()
        # Add noise with amplitude 0.1 * max(h)
        noise = np.random.normal(0, 0.1 * np.max(h), n_points)
        h += noise
        h = repair_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 2.0
    max_iter = 500
    
    # Track best positions
    personal_best_h = [p.copy() for p in particles]
    personal_best_c5 = [evaluate_c5(p, dx) for p in particles]
    
    global_best_h = None
    global_best_c5 = float('inf')
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate and update bests
        for i in range(n_particles):
            current_c5 = evaluate_c5(particles[i], dx)
            if current_c5 < personal_best_c5[i]:
                personal_best_c5[i] = current_c5
                personal_best_h[i] = particles[i].copy()
                
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = particles[i].copy()
                
        # Early exit if target met
        if global_best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_h[i] - particles[i]) +
                           c2 * r2 * (global_best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair feasibility
            particles[i] = repair_h(particles[i], target_sum)
    
    # Local search on top particles (top 10%)
    # Sort by fitness
    sorted_indices = np.argsort(personal_best_c5)
    top_particles = sorted_indices[:n_particles//10]
    
    for idx in top_particles:
        h = personal_best_h[idx].copy()
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, target_sum)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            
            # Update with learning rate 0.01
            h = h - 0.01 * grad
            h = repair_h(h, target_sum)
            
        # Update best solution if improved
        final_c5 = evaluate_c5(h, dx)
        if final_c5 < global_best_c5:
            global_best_c5 = final_c5
            global_best_h = h.copy()
    
    # Final evaluation and return
    final_c5 = evaluate_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
