import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    n_particles = 30
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available as one particle
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        particles.append(project_to_feasible(h, n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Fill other particles with different patterns
    for i in range(n_particles - len(particles)):
        # Alternate between block patterns
        h = np.zeros(n_points)
        if i % 2 == 0:
            # Two blocks
            split = n_points // 2
            h[:split] = 1.0
        else:
            # Three blocks
            split1 = n_points // 3
            split2 = 2 * n_points // 3
            h[:split1] = 1.0
            h[split2:] = 1.0
        # Add noise
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        particles.append(project_to_feasible(h, n_points))
    
    # Initialize velocities
    for i in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia_start = 0.7
    inertia_decay = 1e-4
    
    # Track best positions
    best_positions = [particles[i].copy() for i in range(n_particles)]
    best_c5_values = [evaluate_c5(particles[i], dx) for i in range(n_particles)]
    global_best_idx = np.argmin(best_c5_values)
    global_best_c5 = best_c5_values[global_best_idx]
    global_best_h = best_positions[global_best_idx].copy()
    
    # Optimization loop
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        inertia = inertia_start * np.exp(-inertia_decay * iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia * velocities[i] +
                alpha * r1 * (best_positions[i] - particles[i]) +
                beta * r2 * (global_best_h - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Enforce constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < best_c5_values[i]:
                best_c5_values[i] = c5
                best_positions[i] = particles[i].copy()
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best_h = particles[i].copy()
        
        # Early termination
        if global_best_c5 <= 0.3807:
            break
    
    return (global_best_h.tolist(), global_best_c5, n_points)
