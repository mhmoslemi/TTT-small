import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    """Solve convex relaxation using CVXPY"""
    # Create CVXPY variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize maximum overlap + regularization
    # We compute overlap for one lag and use it as proxy
    # This is a greedy approximation to introduce convexity
    
    # Overlap for lag 0 (center overlap term) 
    overlap_obj = cp.sum(cp.multiply(h_var, 1 - h_var))  # Diagonal terms
    # Add regularization to encourage uniformity
    reg_term = lambda_reg * cp.sum(cp.entr(h_var) + cp.log(n_points/2) * h_var)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points/2
    ]
    
    # Solve the problem
    prob = cp.Problem(cp.Minimize(overlap_obj + reg_term), constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            # Return the solution
            return np.array(h_var.value)
        else:
            # Return original if optimization fails
            return h.copy()
    except:
        # Return original if optimization fails
        return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 3000
    
    # Get initial solution or create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Create initial solution
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent parameters
    step_size = 0.005
    line_search_steps = [0.1, 0.05, 0.01]
    relax_freq = 50
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap with current h
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_k = np.argmax(overlap)
        
        # Consider small gradient estimation
        # Approximate gradient using finite differences
        eps = 1e-5
        gradient = np.zeros(n_points)
        h_plus = h.copy()
        
        # Compute gradient estimate
        for i in range(n_points):
            h_plus[i] = min(h[i] + eps, 1.0)
            h_plus = project_feasible(h_plus, n_points, dx)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus)
            gradient[i] = (c5_plus - max_overlap) / eps
            h_plus[i] = h[i]  # reset
        
        # Perform line search
        new_h = h - step_size * gradient
        new_h = project_feasible(new_h, n_points, dx)
        new_c5 = evaluate_erdos_solution(new_h)
        
        # If improves, accept
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            h = new_h.copy()
        elif new_c5 < max_overlap:
            h = new_h.copy()
            
        # Every few iterations, do convex relaxation to escape local minima
        if iter_count % relax_freq == 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx)
            c5_relaxed = evaluate_erdos_solution(h_relaxed)
            if c5_relaxed < best_c5:
                best_c5 = c5_relaxed
                best_h = h_relaxed.copy()
                h = h_relaxed.copy()
            
        # Early termination if we pass target
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    h_final = project_feasible(best_h, n_points, dx)
    c5_final = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), c5_final, n_points)
