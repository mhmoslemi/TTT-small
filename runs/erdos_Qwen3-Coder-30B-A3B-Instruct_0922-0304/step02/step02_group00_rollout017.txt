import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # AL parameters
    rho = 1.0
    lambda_multiplier = np.zeros(2 * n_points - 1)
    tau = np.zeros(2 * n_points - 1)
    
    max_iter = 500
    
    # ADMM with Augmented Lagrangian
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute overlap to get current values
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        current_overlap = corr * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 <= 0.3807:
            break

        # Solve convex optimization subproblem
        h_var = cp.Variable(n_points)
        # AL objective: minimize max overlap + penalty
        obj = 0
        for i in range(2 * n_points - 1):
            # Augmented Lagrangian term
            obj += lambda_multiplier[i] * (current_overlap[i] - tau[i])
            obj += 0.5 * rho * cp.square(current_overlap[i] - tau[i])
            
        # Constraint: sum must equal n_points/2
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Minimize AL objective (convex problem)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If convex solve fails, fallback to gradient descent
                step = 0.01
                iterations = 200
                h_temp = h.copy()
                for _ in range(iterations):
                    corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_temp = np.max(corr_temp) * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_temp) / eps
                        
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback to simple gradient descent if cvxpy fails
            step = 0.01
            iterations = 200
            h_temp = h.copy()
            for _ in range(iterations):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                    
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp

        # Update Lagrange multipliers
        corr = correlate(h, 1 - h, mode="full")
        current_overlap = corr * dx
        
        lambda_multiplier += rho * (current_overlap - tau)
        
        # Update penalty if needed
        if np.all(np.abs(current_overlap - tau) < 1e-3):
            rho *= 1.2
        else:
            rho *= 1.02

    # Final validation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
