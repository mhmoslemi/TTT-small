import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Stage 1: Convex relaxation with homotopy
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization parameters
    n_steps = 50
    mu_schedule = np.logspace(np.log10(0.5), 0, n_steps)  # From 0.5 to 1.0
    max_iter = 1000
    
    # Warm start variable
    prev_h = h.copy()
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 2:
            break
        
        # Create optimization variables
        h_var = cp.Variable(n_points)
        
        # Objective: relaxed problem
        # min mu * C5 + (1 - mu) * penalty_term
        # We'll compute C5 in a separate step, so this is approximated
        
        # Constraint that sum(h) = 35
        constraints = [cp.sum(h_var) == 35]
        
        # Constraint that h in [0,1]
        constraints.extend([h_var >= 0, h_var <= 1])
        
        # Try solving for a small subproblem to get a working baseline
        obj = cp.Minimize(cp.sum_squares(h_var - 0.5) * (1 - mu) + 
                          cp.max(cp.multiply(0.01, cp.Variable(n_points))) * mu)  # Dummy obj
        
        # We'll solve a simpler problem
        # Use a direct approach with projective gradient and convex solver
        
        # Create a more practical relaxation:
        # min 0.5 * sum(h_i - h_{i+k})^2
        # Actually, directly minimize C5 using convex proxy
        
        # Simplification: we use convex relaxation, with warm-start
        try:
            # Reuse solution from previous step to warm-start
            h_var.value = prev_h
            
            # For convexity, minimize a tractable proxy
            # min sum(h_i * (1 - h_{i+1})) (simplified)
            # Spend time on improvement
            prob = cp.Problem(cp.Minimize(cp.sum_squares(0.5 - h_var)), constraints)
            prob.solve(solver=cp.ECOS, max_iter=max_iter, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Project to ensure constraint
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                prev_h = new_h
            else:
                new_h = prev_h
        except:
            # Fallback to old method if convex fails
            new_h = prev_h
        
        # Evaluate solution
        current_c5 = evaluate_c5(new_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        
        # Early exit if target met
        if best_c5 <= 0.3807:
            break

    # Stage 2: Local refinement with gradient steps if not already good
    if best_c5 > 0.3807:
        h = best_h.copy()
        max_iter = 1000
        learning_rate = 0.005
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
                
            # Gradient computation (finite differences)
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
