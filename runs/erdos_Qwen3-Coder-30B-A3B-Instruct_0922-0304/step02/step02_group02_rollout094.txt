import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Equivalent to integral constraint
    
    # Initialize swarm
    n_particles = 50
    particles = []
    velocities = []
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        
    # Create swarm around the base solution
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 2.0
    max_iter = 500
    
    # Track best solutions
    global_best_idx = 0
    global_best_c5 = float('inf')
    particle_best_c5 = [float('inf')] * n_particles
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            current_c5 = evaluate_c5(particles[i], dx)
            if current_c5 < particle_best_c5[i]:
                particle_best_c5[i] = current_c5
                
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_idx = i
                
            if global_best_c5 <= 0.3807:
                break
                
        # Apply PSO updates
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            
            # Update velocity
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (particles[i] - particles[i]) + 
                           c2 * r2 * (particles[global_best_idx] - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair and rescale
            particles[i] = repair_h(particles[i], target_sum)
            
        # Local search on top 5% particles
        if iteration > 100 and iteration % 10 == 0:
            # Get top 5% particles by fitness
            sorted_indices = np.argsort(particle_best_c5)[:n_particles//20]
            
            # Apply gradient descent to these particles
            for i in sorted_indices:
                if time.time() - start_time > budget_s - 1:
                    break
                h = particles[i]
                eps = 1e-4
                grad = np.zeros_like(h)
                base_c5 = evaluate_c5(h, dx)
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, target_sum)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[j] = (plus_c5 - base_c5) / eps
                
                # Update using gradient
                h = h - 0.01 * grad
                particles[i] = repair_h(h, target_sum)
    
    # Return best solution
    final_h = particles[global_best_idx]
    final_c5 = global_best_c5
    
    return (final_h.tolist(), final_c5, n_points)
