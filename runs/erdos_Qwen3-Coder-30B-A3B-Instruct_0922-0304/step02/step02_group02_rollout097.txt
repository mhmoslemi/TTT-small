import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1000.0
    Tmin = 1.0
    alpha = 0.999
    T = T0
    max_iter = 100000
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Generate candidate solution by perturbing h
        h_new = h.copy()
        # Select random indices to modify (about 10% of points)
        n_modify = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        # Perturb selected indices
        delta = 0.1
        for i in indices:
            h_new[i] += np.random.randn() * delta
        # Normalize the sum to be exactly 35
        sum_h = np.sum(h_new)
        scale = 35.0 / sum_h
        h_new = h_new * scale
        # Clamp values to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Compute candidate C5
        candidate_c5 = evaluate_c5(h_new, dx)
        
        # Acceptance probability
        delta_e = candidate_c5 - current_c5
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h = h_new
        
        # Cool down temperature
        T = max(Tmin, alpha * T)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
