import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0)
    else:
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        particles.append(h0)
    
    # Fill remaining particles
    for _ in range(num_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    
    # Initialize personal bests
    p_best = [h.copy() for h in particles]
    p_best_scores = [evaluate_c5(h, dx) for h in particles]
    
    # Initialize global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    best_c5 = p_best_scores[best_idx]
    
    # PSO parameters
    omega = 1.0
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    perturb = 0.05
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Add random perturbation
            velocities[i] += np.random.normal(0, perturb, n_points)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = current_c5
                
                # Update global best
                if current_c5 < best_c5:
                    g_best = particles[i].copy()
                    best_c5 = current_c5
                    last_improvement = iteration
            
        # Adaptive inertia
        if iteration % 50 == 0:
            omega = max(0.8, omega - 0.02)
            
        # Reset stagnant particles
        if iteration - last_improvement > 50:
            for i in range(num_particles):
                if np.random.random() < 0.3:
                    particles[i] = np.random.uniform(0, 1, n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    velocities[i] = np.random.normal(0, 0.1, n_points)
        
        # Check early termination
        if best_c5 <= 0.3807:
            break
    
    return (g_best.tolist(), best_c5, n_points)
