import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    # Using full convolution to compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        scaling = (n_points / 2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Initialize timing
    start_time = time.time()
    
    # Set up parameters
    n_points = 70
    dx = 2.0 / n_points
    np.random.seed(seed)
    random.seed(seed)
    
    # Set up PSO parameters
    num_particles = 50
    inertia_weight = 1.0
    cognitive_coeff = 2.0
    social_coeff = 2.0
    velocity_limit = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0.copy())
    else:
        # Generate random initial solution if no initial is provided
        h0 = np.random.rand(n_points)
        h0 = project_feasible(h0, n_points, dx)
        particles.append(h0.copy())
    
    # Fill up particles with random solutions
    while len(particles) < num_particles:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(len(particles)):
        velocities.append(np.random.normal(0, 0.05, n_points))
    
    # Initialize personal bests
    p_best_positions = [p.copy() for p in particles]
    p_best_values = [evaluate_erdos_solution(p) for p in particles]
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_values)
    g_best_position = p_best_positions[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    # Main PSO loop
    iteration = 0
    best_c5 = g_best_value
    best_h = g_best_position.copy()
    
    # Iteration counter to control inertia
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(len(particles)):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (p_best_positions[i] - particles[i])
            social = social_coeff * r2 * (g_best_position - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Ensure feasibility after update
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate new solution
            current_c5 = evaluate_erdos_solution(particles[i])
            
            # Update personal best if better
            if current_c5 < p_best_values[i]:
                p_best_values[i] = current_c5
                p_best_positions[i] = particles[i].copy()
                
                # Update global best if better
                if current_c5 < g_best_value:
                    g_best_value = current_c5
                    g_best_position = particles[i].copy()
                    stagnation_count = 0
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = particles[i].copy()
            
            # Apply random perturbation occasionally to avoid stagnation
            if iteration % 20 == 0:
                particles[i] += np.random.normal(0, 0.01, n_points)
                particles[i] = project_feasible(particles[i], n_points, dx)
        
        # Adjust inertia weight
        if iteration % 50 == 0:
            inertia_weight = max(0.8, inertia_weight - 0.02)
            
        # Reset stagnated particles
        stagnation_count += 1
        if stagnation_count > 100:
            for i in range(len(particles)):
                if i != g_best_idx:
                    particles[i] = np.random.rand(n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    velocities[i] = np.random.normal(0, 0.05, n_points)
            stagnation_count = 0
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
