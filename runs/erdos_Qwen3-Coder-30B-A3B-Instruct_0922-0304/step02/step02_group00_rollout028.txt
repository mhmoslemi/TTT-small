import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with sawtooth pattern
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    patience = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Prevent getting stuck
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            patience = 0
        else:
            patience += 1
            if patience > 100:
                break
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Solve convex subproblem to reduce overlap at k_max
        h_var = Variable(n_points)
        
        # Construct linear objective for overlap at k_max using the correlation structure
        # For fixed k_max, overlap at shift k_max = sum_{i} h[i] * (1 - h[i + k_max])
        # If we form this as a linear program, it requires careful indexing
        # Let's build limiting constraints at k_max and solve
        # Optimizing the overlap term at shift k_max
        objective = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            objective += h_var[i] * (1 - h_var[j])
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve
        try:
            prob = Problem(Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).reshape(-1)
                h = h_new
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Revert to previous better solution
                h = best_h.copy()
        except:
            # Fallback to previous solution
            h = best_h.copy()
    
    # Ensure final result is feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
