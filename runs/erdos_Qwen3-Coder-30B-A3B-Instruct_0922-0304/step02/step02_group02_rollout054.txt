import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Use a symmetric initialization: 5-fold pattern
        h = np.ones(n_points)
        # Make it 5-fold symmetric with alternating high/low values
        segment_size = n_points // 5
        for i in range(5):
            start = i * segment_size
            end = (i + 1) * segment_size
            if i % 2 == 0:
                h[start:end] = 0.6
            else:
                h[start:end] = 0.4
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define objective function for scipy optimization
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with bounds and constraints (but within feasible region)
        try:
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10}
            )
            if result.success:
                h = result.x
                # Repair in case of slight violations:
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If optimization failed, use a smaller step
                pass
        except Exception:
            pass
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Occasionally refresh with a new symmetric pattern to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            # Perturb the symmetric initialization
            h = np.ones(n_points)
            segment_size = n_points // 5
            for i in range(5):
                start = i * segment_size
                end = (i + 1) * segment_size
                if i % 2 == 0:
                    h[start:end] = 0.6 + np.random.normal(0, 0.05, segment_size)
                else:
                    h[start:end] = 0.4 + np.random.normal(0, 0.05, segment_size)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
