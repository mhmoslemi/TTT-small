import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-4
    base_c5 = evaluate_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm
    swarm_size = 30
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize velocities
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    personal_best = [p.copy() for p in population]
    personal_best_scores = [evaluate_c5(p, dx) for p in population]
    
    # Initialize global best
    gb_idx = np.argmin(personal_best_scores)
    global_best = personal_best[gb_idx].copy()
    global_best_score = personal_best_scores[gb_idx]
    
    # PSO iterations
    max_iterations = 500
    local_refine_freq = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - population[i]) + 
                           c2 * r2 * (global_best - population[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            population[i] += velocities[i]
            population[i] = project_to_feasible(population[i], n_points)
            
            # Evaluate
            current_score = evaluate_c5(population[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = population[i].copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = population[i].copy()
                    
            # Early exit
            if global_best_score <= 0.3807:
                break
                
        # Local refinement
        if iteration % local_refine_freq == 0 and iteration > 0:
            top_k = int(swarm_size * 0.1)
            sorted_indices = np.argsort(personal_best_scores)[:top_k]
            for idx in sorted_indices:
                h_refined = personal_best[idx].copy()
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h_refined, dx)
                    h_refined -= step_size * grad
                    h_refined = project_to_feasible(h_refined, n_points)
                
                refined_score = evaluate_c5(h_refined, dx)
                if refined_score < personal_best_scores[idx]:
                    personal_best_scores[idx] = refined_score
                    personal_best[idx] = h_refined.copy()
                    
                    if refined_score < global_best_score:
                        global_best_score = refined_score
                        global_best = h_refined.copy()
                        
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_h = global_best.copy()
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
