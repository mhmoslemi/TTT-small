import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    epsilon = 0.05
    T_initial = 1.0
    beta = 0.995
    T = T_initial
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Set up a reasonable starting configuration
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Track improvement
    no_improvement_count = 0
    max_no_improvement = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
            
        # Reset temperature if no improvement
        if no_improvement_count > max_no_improvement:
            T = T_initial
            no_improvement_count = 0
            
        # Generate candidate h by perturbing h
        h_candidate = h + epsilon * T * np.random.normal(0, 1, n_points)
        
        # Apply bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_candidate, dx)
        delta_c5 = new_c5 - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            
        # Cool temperature
        if iteration % 100 == 0:
            T = beta * T
            
    # Final check for best solution
    if best_h is None:
        final_c5 = evaluate_c5(h, dx)
        return (h.tolist(), final_c5, n_points)
    else:
        return (best_h.tolist(), best_c5, n_points)
