import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the provided initial_h_values if available
    if 'initial_h_values' in globals():

        h = enforce_feasibility(h, n_points)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = enforce_feasibility(h, n_points)
    
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B with constraints
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=bounds, 
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': 1000, 'ftol': 1e-8})
    
    best_h = res.x
    best_h = enforce_feasibility(best_h, n_points)
    c5 = evaluate_overlap(best_h, dx)
    
    # Additional optimization step with different starting point
    if time.time() - start_time < budget_s - 5:
        # Try another initialization strategy
        h2 = np.zeros(n_points)
        h2[:n_points//3] = 1
        h2 = enforce_feasibility(h2, n_points)
        
        res2 = minimize(objective, h2, method='L-BFGS-B', 
                       bounds=bounds, 
                       constraints={'type': 'eq', 'fun': constraint},
                       options={'maxiter': 500, 'ftol': 1e-8})
        
        best_h2 = res2.x
        best_h2 = enforce_feasibility(best_h2, n_points)
        c5_2 = evaluate_overlap(best_h2, dx)
        
        if c5_2 < c5:
            c5 = c5_2
            best_h = best_h2
