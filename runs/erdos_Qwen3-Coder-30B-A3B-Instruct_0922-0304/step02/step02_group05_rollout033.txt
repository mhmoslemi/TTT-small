import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Convex relaxation of the optimization problem
    h_var = cp.Variable(n_points)
    # Constraint: h in [0,1] and sum = n_points/2
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
    
    # Define overlap with current h
    h_padded = np.pad(h, (0, n_points-1), mode='constant')
    h_shifted = np.concatenate([h_padded[n_points-1:], h_padded[:n_points-1]])
    # Use linear approximation for convex relaxation
    overlap_term = cp.sum(cp.multiply(h_var, 1 - h_shifted))
    
    # Regularize with entropy
    entropy_term = cp.sum(cp.multiply(h_var, cp.log(h_var + 1e-10)))
    
    objective = cp.Minimize(overlap_term - lambda_reg * entropy_term)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status == cp.OPTIMAL:
        return h_var.value
    else:
        return h.copy()

        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Gradient descent parameters
    step_size = 1e-3
    line_search_steps = [1e-3, 1e-2, 1e-1]
    relaxation_interval = 50
    lambda_reg = 0.01
    iteration = 0

        # Compute overlaps and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_idx = np.argmax(overlap)
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, n_points, dx)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (np.max(overlap_plus) - max_overlap) / eps
        
        # Line search for best step
        best_h_candidate = h.copy()
