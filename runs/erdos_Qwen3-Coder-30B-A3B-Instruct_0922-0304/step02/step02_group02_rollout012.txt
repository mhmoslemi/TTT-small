import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Homotopy approach with convex relaxation
    mu_schedule = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate convex relaxation
        h_vars = cp.Variable(n_points)
        # Regularization term: (h_i - 0.5)^2
        reg_term = cp.sum_squares(h_vars - 0.5) / n_points
        
        # Objective: mu * C5 + (1 - mu) * regularization
        # We'll compute C5 approximately since we can't directly optimize it in cvxpy
        # Instead we'll optimize a convex relaxation
        
        # Constraint for sum
        constraints = [cp.sum(h_vars) == n_points / 2,
                       h_vars >= 0,
                       h_vars <= 1]
        
        # Approximate C5 using relaxed constraints that encourage small overlaps
        # This is a simplified version that just tries to balance the h values with some convex constraints
        # The approach is to minimize a combination of overlap and regularization
        
        # Minimize a simple convex surrogate of the overlap term that includes
        # smoothing and encourages low overlap
        obj = cp.Minimize(mu * reg_term + (1 - mu) * cp.sum_squares(h_vars))
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
        except:
            # If solver fails, fall back to simple gradient descent
            pass
            
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_vars.value
        else:
            # Fallback to gradient descent if CVXPY fails
            h_new = h.copy()
            lr = 0.01
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                eps = 1e-4
                grad = np.zeros_like(h_new)
                base_c5 = evaluate_c5(h_new, dx)
                for i in range(n_points):
                    h_plus = h_new.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[i] = (plus_c5 - base_c5) / eps
                h_new = h_new - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
        h = h_new
        
        # Evaluate the current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
