import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_relaxation(h, n_points, dx, lambda_reg=0.01):
    h0 = h.copy()
    # Create CVXPY variables
    h_var = cp.Variable(n_points)
    
    # Formulate the convex relaxation
    # Overlap term at max lag
    max_lag = np.argmax(correlate(h0, 1-h0, mode='full'))
    # Simplified relaxation via direct convolution
    expr = 0
    for i in range(n_points):
        expr += h_var[i] * (1 - h_var[(i + max_lag) % n_points])
    
    # Regularization
    reg = cp.sum(cp.entr(h_var))  # negative entropy
    
    # Objective
    objective = cp.Minimize(expr - lambda_reg * reg)
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points/2,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return project_feasible(h_var.value, n_points, dx)
    except:
        pass
    return h0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Generate a random feasible initial solution
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step = 0
    lambda_reg = 0.01
    max_iter = int(budget_s / 0.1)  # rough estimate
    
    while step < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        
        step += 1
        
        # Evaluate current
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Find max over lags
        overlap = correlate(h, 1-h, mode='full')
        max_lag = np.argmax(overlap)
        
        # Numerical gradient estimation
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, n_points, dx)
            overlap_plus = correlate(h_plus, 1-h_plus, mode='full')
            c5_plus = np.max(overlap_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Line search for step size
        alpha = 1e-3
        for alpha_step in [1e-3, 5e-3, 1e-2]:
            h_candidate = h - alpha_step * grad
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < current_c5:
                alpha = alpha_step
                break

        # Update
        h = h - alpha * grad
        h = project_feasible(h, n_points, dx)
        
        # Relaxation every 50 steps
        if step % 50 == 0:
            h_relaxed = solve_relaxation(h, n_points, dx, lambda_reg)
            c5_relaxed = evaluate_c5(h_relaxed, dx)
            if c5_relaxed < current_c5:
                h = h_relaxed
                current_c5 = c5_relaxed
                if c5_relaxed < best_c5:
                    best_c5 = c5_relaxed
                    best_h = h.copy()
            else:
                lambda_reg = max(lambda_reg * 0.5, 1e-6)
    
    # Final validation
    if time.time() - start_time > budget_s - 1:
        pass
    else:
        current_c5 = evaluate_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
