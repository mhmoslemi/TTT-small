import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with 5-fold symmetry
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # 5-fold symmetric initialization
        segment_size = n_points // 5
        h = np.zeros(n_points)
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.7
            else:
                h[start_idx:end_idx] = 0.3
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        # Return negative of C5 to minimize (since scipy minimizes)
        return -evaluate_c5(h_vec, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Set up constraints
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization using L-BFGS-B
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Optimization step
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraint, options={'maxiter': 10})
        if result.success:
            h = result.x
        else:
            h = h * (n_points / 2) / np.sum(h)
            
        # Clamp to bounds and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final refinement
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraint, options={'maxiter': 5})
        if result.success:
            h = result.x
        else:
            h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    final_c5 = evaluate_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
