import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Convex relaxation parameters
    lambda_reg = 0.01
    relaxation_interval = 50
    
    iteration = 0
    max_iter = int(budget_s * 5)  # Rough estimation
    
    def compute_gradient(h, k):
        # Approximate gradient at position k
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, n_points, dx)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus) * dx
            
            # Use finite differences
            grad[i] = (c5_plus - best_c5) / eps
        return grad

    def line_search(h, grad, max_iter=3):
        # Try several step sizes
        step_sizes = [1e-4, 1e-3, 1e-2]
        for alpha in step_sizes:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                return h_candidate, c5_candidate, alpha
        return h, best_c5, 0.0

    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlaps and max overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_k = np.argmax(overlap)
        
        # Estimate gradient around the maximum
        grad = compute_gradient(h, max_k)
        
        # Perform line search
        h_new, c5_new, alpha = line_search(h, grad)
        
        # Update if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Perform convex relaxation occasionally
            if iteration % relaxation_interval == 0 and lambda_reg > 1e-8:
                try:
                    # Solve convex relaxation
                    h_var = cp.Variable(n_points)
                    obj = cp.Minimize(
                        cp.max(cp.conv((1 - h_var), h_var, method='fft')) + 
                        lambda_reg * cp.sum(cp.entr(h_var) + cp.log(2) - cp.log(2) * h_var)
                    )
                    constraints = [
                        h_var >= 0,
                        h_var <= 1,
                        cp.sum(h_var) == n_points / 2
                    ]
                    prob = cp.Problem(obj, constraints)
                    prob.solve(solver=cp.SCS, verbose=False)  # faster than ECOS
                    
                    if prob.status in ["optimal", "optimal_inaccurate"]:
                        h_relaxed = h_var.value
                        h_relaxed = np.clip(h_relaxed, 0, 1)
                        h_relaxed = project_feasible(h_relaxed, n_points, dx)
                        c5_relaxed = evaluate_c5(h_relaxed, dx)
                        
                        if c5_relaxed < best_c5:
                            best_c5 = c5_relaxed
                            best_h = h_relaxed.copy()
                            h = h_relaxed.copy()
                        else:
                            # Reduce regularization strength
                            lambda_reg *= 0.5
                    else:
                        # Reduce regularization
                        lambda_reg *= 0.5
                        
                except Exception:
                    lambda_reg *= 0.5

        # Check for target convergence
        if best_c5 <= 0.3807:
            break
            
        iteration += 1

    return (best_h.tolist(), best_c5, n_points)
