import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with a simple construction
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian parameters
    rho = 1.0
    max_iter = 500
    penalty_increase = 2.0
    
    # Initialize Lagrange multipliers
    lambda_k = np.zeros(2 * n_points - 1)
    
    # Store last few values to check convergence
    recent_c5 = []
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Update lagrange multipliers
        lambda_k += rho * (corr - np.max(corr))
        
        # Solve convex subproblem using CVXPY
        h_var = Variable(n_points)
        
        # Create the augmented Lagrangian objective
        # AL = sum(h*(1-h)) + (rho/2)*||h*(1-h) - b||^2 + lambda*(h*(1-h) - b)
        # But for our problem, we want to minimize max overlap.
        # So let's add a regularization 
        obj = cp.Minimize(
            cp_sum(cp.multiply(h_var, 1 - h_var)) + 
            rho * cp_sum(cp.square(cp.multiply(h_var, 1 - h_var) - corr))
        )
        
        # Add constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve the problem
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            pass
            
        # Update penalty if needed
        if len(recent_c5) > 2 and np.mean(recent_c5[-3:]) < current_c5:
            rho *= penalty_increase
            
        recent_c5.append(current_c5)
        if len(recent_c5) > 10:
            recent_c5.pop(0)
            
    return (best_h.tolist(), best_c5, n_points)
