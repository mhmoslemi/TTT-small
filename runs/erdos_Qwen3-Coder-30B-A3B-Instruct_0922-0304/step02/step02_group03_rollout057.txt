import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    n_particles = 30
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = project_to_feasible(h_init, n_points)
    
    # Create initial particles
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Fill remaining particles with random perturbations of block structures
    for _ in range(n_particles - 1):
        h = np.ones(n_points)
        h[n_points//3:2*n_points//3] = 0
        h += np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    decay = 1e-4
    
    # Track the best solution found
    best_global = float('inf')
    best_h_global = None
    
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = max(0.4, inertia * (1 - decay))
        
        # Evaluate each particle
        particle_fits = []
        for i, h in enumerate(particles):
            c5 = evaluate_c5(h, dx)
            particle_fits.append(c5)
            
            # Update personal best
            if c5 < best_global:
                best_global = c5
                best_h_global = h.copy()
            
            # Early stopping
            if best_global <= 0.3807:
                break
                
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = alpha * r1 * (h - particles[i])
            social = beta * r2 * (best_h_global - h)
            v = inertia * v + cognitive + social
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Apply constraints
            h_new = project_to_feasible(h_new, n_points)
            
            particles[i] = h_new
            velocities[i] = v
            
        if best_global <= 0.3807:
            break
    
    # Final evaluation of best solution
    if best_h_global is not None:
        final_c5 = evaluate_c5(best_h_global, dx)
    else:
        # Fallback if no improvement was found
        final_c5 = float('inf')
        best_h_global = particles[0] if particles else np.ones(n_points)
        best_h_global = project_to_feasible(best_h_global, n_points)
        final_c5 = evaluate_c5(best_h_global, dx)
        
    return (best_h_global.tolist(), final_c5, n_points)
