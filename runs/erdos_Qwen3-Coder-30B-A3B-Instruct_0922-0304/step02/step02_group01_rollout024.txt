import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h onto the feasible region: clip to [0,1] and rescale sum to target_sum"""
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum > 0:
        return h_clipped * target_sum / current_sum
    else:
        return np.full_like(h_clipped, target_sum / len(h_clipped))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback: simple piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add some noise for diversity
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        
    # Ensure initial sum constraint
    h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization hyperparameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Early exit if target achieved
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        
        # Find which shift corresponds to max overlap
        # In correlation result, index corresponds to shift k
        # For full mode: result size = 2*n_points - 1
        # shift k = idx - (n_points - 1)
        k = max_idx - (n_points - 1)
        
        # Compute the gradient w.r.t. h for the max overlap shift
        grad = np.zeros_like(h)
        if k >= 0:
            # h[i] * (1 - h[i + k]) contributes to overlap
            # For i = 0 to n_points - 1 - k:
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
                grad[i + k] -= h[i]
        else:
            # k < 0, so h[i] * (1 - h[i - k])  
            # For i = -k to n_points - 1:
            for i in range(-k, n_points):
                grad[i] += 1 - h[i + k]
                grad[i + k] -= h[i]
        
        # Update with momentum and step size
        new_grad = momentum * prev_grad + (1 - momentum) * grad
        h_new = h - step_size * new_grad
        
        # Project back to feasible region
        h = project_h(h_new, target_sum)
        prev_grad = new_grad
        
        # Small random perturbation occasionally to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            h += np.random.normal(0, 0.001, n_points)
            h = project_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
