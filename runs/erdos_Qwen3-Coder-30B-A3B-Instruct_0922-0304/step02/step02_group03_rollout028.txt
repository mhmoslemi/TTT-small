import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Two-stage approach: coarse then fine
    coarse_n = 35
    fine_n = 70
    
    # Coarse phase
    dx_coarse = 2.0 / coarse_n
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:coarse_n]
        h = np.clip(h, 0, 1)
        h = h * (coarse_n / 2) / np.sum(h)
    else:
        h = np.ones(coarse_n)
        h[coarse_n//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (coarse_n / 2) / np.sum(h)
    
    best_c5_coarse = float('inf')
    best_h_coarse = h.copy()
    
    # Coarse optimization
    def coarse_obj(h_coarse):
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
        return evaluate_c5(h_coarse, dx_coarse)
    
    bounds = [(0, 1) for _ in range(coarse_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - coarse_n/2}]
    
    # Use L-BFGS-B for better convergence
    options = {'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-5}
    res = minimize(coarse_obj, h, method='L-BFGS-B', bounds=bounds, 
                   constraints=constraints, options=options)
    if res.success:
        h_coarse = np.clip(res.x, 0, 1)
        h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
        c5_coarse = evaluate_c5(h_coarse, dx_coarse)
        if c5_coarse < best_c5_coarse:
            best_c5_coarse = c5_coarse
            best_h_coarse = h_coarse
    
    # Refinement phase
    dx_fine = 2.0 / fine_n
    # Upsample coarse solution
    h_fine = np.interp(np.linspace(0, 2, fine_n), 
                       np.linspace(0, 2, coarse_n), best_h_coarse)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
    
    # Add perturbations
    perturbation = 0.1 * np.random.rand(fine_n)
    h_fine = h_fine + perturbation
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
    
    best_c5_fine = float('inf')
    best_h_fine = h_fine.copy()
    
    # Optimizer for fine phase
    def fine_obj(h_fine):
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
        return evaluate_c5(h_fine, dx_fine)
    
    bounds = [(0, 1) for _ in range(fine_n)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - fine_n/2}]
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-5}
    
    res = minimize(fine_obj, h_fine, method='L-BFGS-B', bounds=bounds,
                   constraints=constraints, options=options)
    
    if res.success:
        h_fine = np.clip(res.x, 0, 1)
        h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
        c5_fine = evaluate_c5(h_fine, dx_fine)
        if c5_fine < best_c5_fine:
            best_c5_fine = c5_fine
            best_h_fine = h_fine
    
    # Final check
    final_h = best_h_fine if best_c5_fine < best_c5_coarse else best_h_coarse
    final_c5 = best_c5_fine if best_c5_fine < best_c5_coarse else best_c5_coarse
    final_n = fine_n if best_c5_fine < best_c5_coarse else coarse_n
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    if remaining > 1:
        # Try to improve with additional refinement
        h_refined = final_h.copy()
        dx = 2.0 / final_n
        def obj_refined(h_refined):
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (final_n / 2) / np.sum(h_refined)
            return evaluate_c5(h_refined, dx)
        
        bounds = [(0, 1) for _ in range(final_n)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - final_n/2}]
        options = {'maxiter': 500 if final_n == 35 else 1000, 'ftol': 1e-9, 'gtol': 1e-5}
        
        res = minimize(obj_refined, h_refined, method='L-BFGS-B', 
                       bounds=bounds, constraints=constraints, options=options)
        
        if res.success:
            h_refined = np.clip(res.x, 0, 1)
            h_refined = h_refined * (final_n / 2) / np.sum(h_refined)
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < final_c5:
                final_h = h_refined
                final_c5 = c5_refined
    
    return (final_h.tolist(), final_c5, final_n)
