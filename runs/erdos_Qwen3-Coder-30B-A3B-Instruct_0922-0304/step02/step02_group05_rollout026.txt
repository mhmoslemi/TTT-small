import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_size = 0.01
    max_line_search = 10
    relaxation_interval = 50
    regularization_lambda = 0.01
    
    step = 0
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Compute overlaps and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Approximate gradient for the maximum overlap
        # overlap_k = sum h[i] * (1 - h[i+k])
        # ∇h[i] = (1 - h[i+k]) - h[i-k] when k is positive
        # But to avoid boundary issues, we compute approximate gradient
        grad = np.zeros_like(h)
        
        # Take derivative (approximate)
        # Gradient based on structure of overlap
        # d/dh[i] sum_j h[j] * (1 - h[j + k]) = (1 - h[i+k]) when i = j
        # But non-zero only when the kernel hits i
        
        # Efficient way to compute gradient for max overlap
        # Shift h by different k values
        shifted = np.zeros_like(h)
        shifted[0] = 1
        grad = correlate(shifted, 1 - h, mode="full")[:len(h)] * dx
        
        # Update using projected gradient descent
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n_points, dx)
        
        # Line search
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()

        if step % relaxation_interval == 0:
            # Apply a regularization to escape current minimum
            regularized_h = h.copy()
            # Simple entropy regularization: add small noise and reproject
            noise = np.random.normal(0, 0.001, n_points)
            regularized_h = regularized_h + noise
            regularized_h = project_feasible(regularized_h, n_points, dx)
            c5_reg = evaluate_c5(regularized_h, dx)
            if c5_reg < best_c5:
                h = regularized_h.copy()
                best_c5 = c5_reg
                best_h = h.copy()

        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
