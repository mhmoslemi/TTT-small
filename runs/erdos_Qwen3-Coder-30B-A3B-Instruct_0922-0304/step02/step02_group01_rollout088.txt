import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def normalize_sum(h, target_sum):
    """Normalize the sum of h to target_sum"""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def clamp_h(h, min_val=0, max_val=1):
    """Clamp h to [min_val, max_val]"""
    return np.clip(h, min_val, max_val)

def mutate_individual(h, mutation_rate=0.1, noise_level=0.1):
    """Mutate an individual by adding Gaussian noise"""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, noise_level)
    return h_mut

def crossover(parent1, parent2):
    """Single point crossover between two parents"""
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population, using initial_h_values if available
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = []
    
    # Fill with random individuals
    while len(population) < 50:
        h = np.random.rand(n_points)
        h = normalize_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    no_improvement_count = 0
    max_no_improvement = 100
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate population
        fitness = []
        for h in population:
            h = clamp_h(h)
            h = normalize_sum(h, target_sum)
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness (lower C5 is better)
        fitness.sort(key=lambda x: x[0])
        current_best_c5 = fitness[0][0]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = fitness[0][1].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Selection: top 50%
        selected = [h for _, h in fitness[:len(fitness)//2]]
        
        # Crossover and mutation to generate new population
        new_population = selected.copy()
        while len(new_population) < 50:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate_individual(child1)
            child2 = mutate_individual(child2)
            
            # Normalize and clamp
            child1 = clamp_h(child1)
            child1 = normalize_sum(child1, target_sum)
            
            child2 = clamp_h(child2)
            child2 = normalize_sum(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        population = new_population[:50]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # Fallback to best from population if no best found
        fitness = [(evaluate_c5(h, dx), h) for h in population]
        fitness.sort(key=lambda x: x[0])
        final_c5, best_h = fitness[0]
    
    return best_h, final_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
