import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    epsilon = 0.05
    T_initial = 1.0
    beta = 0.995
    T = T_initial
    best_c5 = float('inf')
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    
    # Annealing loop
    iteration = 0
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 1:
        if no_improvement_count > 1000:
            T = T_initial  # Reset temperature
            no_improvement_count = 0
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Generate candidate
        h_candidate = h + epsilon * T * np.random.randn(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate overlap
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode="full")
        candidate_c5 = np.max(corr_candidate) * dx
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Update temperature
        if iteration % 100 == 0:
            T = beta * T
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
