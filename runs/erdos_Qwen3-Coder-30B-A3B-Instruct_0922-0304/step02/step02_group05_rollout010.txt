import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    # This is assumed to be pre-imported
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def project_feasible(h, n_points):
    # Project onto feasible set: h in [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple starting point
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Regularization parameter for convex relaxation
    lambda_reg = 0.01
    
    iteration = 0
    max_iterations = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute overlaps and find the maximal one
        overlaps = np.correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Approximate gradient for the maximum overlap term
        # overlap_k = sum h[i] * (1 - h[(i+k) % N])
        # ∂overlap_k/∂h[i] = 1 - h[(i+k) % N] - h[(i+k) % N] = 1 - 2*h[(i+k) % N]
        # But we need the gradient for each index. We'll move forward with a
        # simplified approach: for fixed k (max_overlap location), simulate
        # gradient update with a small correction based on the overlap computation
        
        # Gradient based on shift that produces maximum overlap 
        shift = max_overlap_idx - (len(h) - 1)  # Center the shifts properly
        h_shifted = np.roll(h, shift)
        # Gradient computation: rows from correlation
        gradient = 1 - 2 * h_shifted
        
        # Line search with step sizes {0.1, 0.5, 1.0}
        step_sizes = [0.1, 0.5, 1.0]
        best_candidate_c5 = best_c5
        best_candidate_h = h.copy()
        
        for step in step_sizes:
            h_candidate = h - step * gradient
            h_candidate = project_feasible(h_candidate, n_points)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # Update current point
        h = best_candidate_h.copy()
        
        # Update best solution
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = h.copy()
        
        # Periodically solve convex relaxation to escape local optima
        if iteration % 20 == 0:
            try:
                # Convex relaxation using CVXPY
                h_relax = cp.Variable(n_points, nonneg=True)
                # Objective: minimize overlap plus regularization
                # We approximate overlap as a linear term in the relaxed variables, 
                # and add a KL divergence penalty for smoothness
                obj = cp.Minimize(
                    cp.max(cp.conv(cp.multiply(h_relax, -1), 1 - h_relax, method="direct")) + 
                    lambda_reg * cp.sum(cp.entr(h_relax) + cp.log(n_points/2) * h_relax)
                )
                # Constraints
                constraints = [cp.sum(h_relax) == n_points/2, h_relax >= 0, h_relax <= 1]
                prob = cp.Problem(obj, constraints)
                
                # Try solving with different solvers if available
                try:
                    prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
                except:
                    try:
                        prob.solve(solver=cp.ECOS, verbose=False)
                    except:
                        # fallback 
                        pass
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_relax.value
                    h_new = project_feasible(h_new, n_points)
                    new_c5 = evaluate_erdos_solution(h_new)
                    if new_c5 < best_c5:
                        # Accept the new solution
                        best_c5 = new_c5
                        best_h = h_new.copy()
                        h = h_new.copy()
                        # Decrease regularization if improvement made
                        if lambda_reg > 1e-6:
                            lambda_reg *= 0.5
                    else:
                        # No improvement; may increase lambda to encourage simpler solutions
                        lambda_reg *= 2.0
                
            except Exception as e:
                # If convex relaxation fails, fallback to current solution
                pass
        
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
