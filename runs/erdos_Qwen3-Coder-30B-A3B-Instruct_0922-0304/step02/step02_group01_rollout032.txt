```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization approach
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve using convex optimization
        h_var = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2,
            t >= 0
        ]
        
        # Add overlap constraints
        # For all lags k from -n_points+1 to n_points-1
        # Compute overlap for k = 0 to n_points-1 first (non-negative lags)
        for k in range(n_points):
            if k == 0:
                # Special case for k=0: h(x) * (1-h(x)) 
                # This is a direct constraint on h_i terms
                continue
            else:
                # For lag k > 0, overlap = sum h_i * (1 - h_{i+k})
                # We build full overlap expression manually for constraints
                overlap_expr = 0
                for i in range(n_points - k):
                    # h_i * (1 - h_{i+k}) = h_i - h_i * h_{i+k}
                    overlap_expr += h_var[i] - h_var[i] * h_var[i + k]
                constraints.append(overlap_expr <= t)
        
        # For negative lags, we use symmetry since convolution is symmetric
        # But for full correlation, we also consider negative lags properly
        # Only add non-numeric lags or handle via indexing
        
        # Build constraint matrix for cross terms
        # We'll simplify and focus on constraints
        for k in range(1, n_points):
            overlap_expr = 0
            for i in range(n_points - k):
                overlap_expr += h_var[i] * (1 - h_var[i + k])
            constraints.append(overlap_expr <= t)
                
        # Objective
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # Try to solve with solver
        try:
            # Use GLOP for faster solving
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0, 1)
                # Normalize
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                # Check if solution improves
                current_c5 = evaluate_c5(h_opt, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                    
                if best_c5 <= 0.3807:
                    break
                # Update h for next iteration
                h = h_opt.copy()
            else:
                # If infeasible, keep trying
                pass
        except Exception:
            # Fall back to simple approach if optimization fails
            pass
            
        # Fallback to simpler approach if needed
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 <= 0.3807:
            break
            
        # Basic local search
        step = 0.001
        iterations = 50
        h_temp = h.copy()
        for _ in range(iterations):
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
