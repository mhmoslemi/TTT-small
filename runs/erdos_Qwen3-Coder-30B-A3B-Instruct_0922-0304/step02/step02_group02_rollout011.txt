import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Relaxation schedule
    mu_schedule = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convex optimization
        h_var = cp.Variable(n_points)
        # Objective: weighted combination of C5 and regularization
        # We want to minimize mu * C5 + (1 - mu) * regularization
        # Constraint: sum(h) = 35
        constraints = [
            cp.sum(h_var) == 35,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Regularization term: (h_i - 0.5)^2
        reg_term = cp.sum(cp.square(h_var - 0.5)) / n_points
        
        # For C5 approximation, we need to solve the max overlap over shifts
        # This is not straightforward to model as convex, so we solve on current h
        # as a simple convex relaxation: don't use it directly in the problem
        
        # Approximate C5 with a simpler relaxation:
        # Use a quadratic penalty on non-overlapping regions
        # but since we don't have a clean convex C5, we use the existing heuristic
        # but rephrase the optimization to stabilize convergence
        
        # Since ad-hoc convex relaxation of C5 is complicated, 
        # we keep the gradient optimization idea with better stabilization
        
        # Try using cvxpy with warm-starts and slight convex perturbation
        # of the gradient direction based on a quadratic model
        # This should allow trust region-like behavior
        objective = cp.Minimize(mu * reg_term)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            # Extract solution
            h_new = h_var.value
            # Project back to valid range
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            # Update h with trust-region-like approach
            h = h_new
        except:
            pass # Stay with the previous h if optimization fails
        
        # Local gradient-based optimization with projected gradients
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            # Update h with momentum and step size
            h = h - 0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Track best solution
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
