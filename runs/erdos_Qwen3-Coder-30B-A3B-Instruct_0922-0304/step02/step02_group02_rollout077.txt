import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while maintaining bounds"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using symmetric construction
    h = np.zeros(n_points)
    segment_size = n_points // 5  # 14 points per segment
    
    # Create a 5-fold symmetric pattern with alternating values
    for i in range(5):
        start_idx = i * segment_size
        end_idx = start_idx + segment_size
        # Alternate between 0.8 and 0.2 in each segment
        for j in range(start_idx, end_idx):
            if (j - start_idx) % 2 == 0:
                h[j] = 0.8
            else:
                h[j] = 0.2
    
    # Adjust to satisfy sum constraint 
    h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer settings
    method = 'L-BFGS-B'
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        h = h_vec.copy()
        h = project_h(h, target_sum)
        return evaluate_c5(h, dx)
    
    def callback(h_vec):
        nonlocal best_c5, best_h
        h = h_vec.copy()
        h = project_h(h, target_sum)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 <= 0.3807:
            return True
        return False
    
    # Run optimization
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 1:
        return (best_h.tolist(), best_c5, n_points)
    
    try:
        res = minimize(
            objective,
            h,
            method=method,
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-5},
            callback=callback,
            tol=1e-6
        )
        if res.success:
            h = res.x.copy()
            h = project_h(h, target_sum)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    except Exception:
        pass

    # Final refinement if near target
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 2 and best_c5 > 0.3807:
        # Simple local search
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            h = best_h.copy()
            # Perturb slightly
            idx = np.random.randint(0, n_points)
            eps = 0.01
            h[idx] = np.clip(h[idx] + np.random.normal(0, eps), 0, 1)
            h = project_h(h, target_sum)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
