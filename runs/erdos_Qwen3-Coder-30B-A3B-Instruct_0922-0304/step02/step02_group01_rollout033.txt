import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, Constraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimize using convex relaxation
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Use CVXPY for convex relaxation
        h_var = Variable(n_points)
        t = Variable()
        
        # Constraints: h in [0,1]
        constr = [h_var >= 0, h_var <= 1]
        
        # Constraint: sum(h) = n_points / 2
        constr.append(cp_sum(h_var) == n_points / 2)
        
        # Constraints: overlap <= t for all lags
        # We'll use a different approach: enumerate all possible lags
        # and add constraints one by one for better precision
        max_lag = n_points - 1
        lag_list = list(range(-max_lag, max_lag + 1))
        
        # For each lag, we create the overlap and enforce it's <= t
        # Because of convolution symmetry, we can work with positive lags only
        # and optimize smarter by focusing on the non-trivial ones
        for k in range(max_lag + 1):
            if k == 0:
                # No shift case
                overlap = cp_sum(h_var * (1 - h_var))
            else:
                # Create shifted version of h_var
                # For positive k, h[i] * (1 - h[i+k])
                if k < n_points:
                    overlap = cp_sum(h_var[:n_points-k] * (1 - h_var[k:]))
                else:
                    overlap = 0  # No support
            
            # Add constraint: overlap <= t
            constr.append(overlap <= t)
        
        # Define and solve the problem
        prob = Problem(Minimize(t), constr)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            
            h_cvx = h_var.value
            t_val = t.value
            
            if t_val is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                c5 = max(np.correlate(h_cvx, 1 - h_cvx, mode="full")) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cvx.copy()
                    last_improvement = time.time()
                    
                if c5 <= 0.3807:
                    break
                    
        except:
            pass
        
        # Fast local gradient descent to improve results
        step = 0.005
        iterations = 200
        
        h_temp = best_h.copy()
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx

            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(0, n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps

            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        best_h = h_temp.copy()
        best_c5 = max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

        if time.time() - last_improvement > 30 and best_c5 <= 0.3808:
            # Stop if no improvement in 30s and current is already good
            break
            
    return (best_h.tolist(), best_c5, n_points)
