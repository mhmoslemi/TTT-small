import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import sqrt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # Clip to [0,1] and rescale to sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def gradient_step(h, dx, num_points, max_k):
    # Approximate gradient for overlap at max_k
    # Using finite differences
    eps = 1e-4
    grad = np.zeros_like(h)
    
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    
    for i in range(num_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, num_points, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        c5_plus = np.max(corr_plus) * dx
        grad[i] = (c5_plus - max_overlap) / eps
        
    return grad

def solve_convex_relaxation(h, n_points, dx):
    # Convex relaxation of the problem to escape local minima
    # Minimize overlap_k + λ * sum(h * log(h))
    # where overlap_k is the maximum overlap
    
    # Identify the maximum overlap index
    corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(corr)
    
    # Use cvxpy for the convex relaxation:
    # Minimize e_k^T h + lambda * sum(h * log(h))
    # s.t. h >= 0, sum(h) = n_points/2
    
    h_cvx = cp.Variable(n_points)
    lambda_reg = 0.01
    
    # Formulate objective: scalar product with the gradient of the overlap
    # and entropic regularization
    # objective = cp.sum(cp.multiply(h_cvx, corr[max_k:max_k+n_points])) + lambda_reg * cp.sum(cp.entr(h_cvx))
    # Use laplacian version of regularization for better numerical stability
    
    # Instead of complex overlap gradient, simple regularization with
    # entropic term: min sum(h * log(h)) subject to h >= 0, sum(h) = n_points/2
    # This is the convex relaxation of a local minimum problem
    
    objective = cp.Minimize(lambda_reg * cp.sum(cp.entr(h_cvx)) + cp.sum(cp.multiply(h_cvx, 1-h_cvx)[:n_points]))
    constraints = [
        h_cvx >= 0,
        cp.sum(h_cvx) == n_points / 2.0
    ]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return np.array(h_cvx.value).flatten()
    except:
        pass
    return h.copy()  # fallback to original if convex relaxation fails

def run(seed=42, budget_s=600, **kwargs):
    # Set up problem
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = project_feasible(h, n_points, dx)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    step_size = 1e-3
    conv_relax_freq = 50
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current objective
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr)
        max_k = np.argmax(corr)
        
        # Compute gradient direction
        grad = gradient_step(h, dx, n_points, max_k)
        
        # Line search
        found_improvement = False
        for alpha in [1e-3, 1e-2, 1e-1]:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                best_h = h.copy()
                found_improvement = True
                break
        
        # If no improvement in this step, continue with smaller step
        if not found_improvement:
            h = h - 0.5 * step_size * grad
            h = project_feasible(h, n_points, dx)
            c5_new = evaluate_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Periodic convex relaxation
        if iteration % conv_relax_freq == 0:
            h_new = solve_convex_relaxation(h, n_points, dx)
            if h_new is not None:
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    h = h_new.copy()
                    best_c5 = c5_new
                    best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
