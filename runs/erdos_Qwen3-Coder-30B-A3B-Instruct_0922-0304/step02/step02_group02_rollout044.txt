import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy sum(h) = n_points / 2 and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, n_points)
        swarm.append(h)
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 2.0
    velocities = [np.zeros(n_points) for _ in range(P)]
    
    # Track best solutions
    global_best_c5 = float('inf')
    global_best_h = None
    
    particle_best_h = [h.copy() for h in swarm]
    particle_best_c5 = [evaluate_c5(h, dx) for h in swarm]
    
    # Local search parameters
    local_search_fraction = 0.05
    local_search_iter = 20

        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for p in range(P):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[p] = (omega * velocities[p] + 
                           c1 * r1 * (particle_best_h[p] - swarm[p]) + 
                           c2 * r2 * (global_best_h - swarm[p]))
            
            # Update position
            swarm[p] = swarm[p] + velocities[p]
            
            # Repair and evaluate
            swarm[p] = repair_h(swarm[p], n_points)

            if current_c5 < particle_best_c5[p]:

                particle_best_h[p] = swarm[p].copy()
            
            # Update global best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = swarm[p].copy()
                
        # Early exit
        if global_best_c5 <= 0.3807:
            break
            
        # Apply local search to top particles
        if iteration % 10 == 0:  # periodic local search
            # Sort by fitness
            sorted_indices = np.argsort(particle_best_c5)
            num_local = max(1, int(P * local_search_fraction))
            local_indices = sorted_indices[:num_local]
            
            for p in local_indices:
