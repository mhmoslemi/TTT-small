import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 30
    particles = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
        particles.append(h_init.copy())
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Add other particles with different structures
    for _ in range(num_particles - len(particles)):
        # Create blocky structure
        h = np.zeros(n_points)
        split1 = n_points // 3
        split2 = 2 * n_points // 3
        h[:split1] = 1
        h[split2:] = 1
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
    
    # PSO parameters
    inertia = 0.7
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia_decay = 1e-4
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    
    # Track best positions
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    global_best = personal_best[np.argmin(personal_best_scores)].copy()
    global_best_score = min(personal_best_scores)
    
    # PSO iterations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        if inertia > 0.3:
            inertia -= inertia_decay
            
        for i in range(num_particles):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            alpha * r1 * (personal_best[i] - particles[i]) +
                            beta * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    
        # Early stopping
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = global_best_score
    final_h = global_best.tolist()
    
    return (final_h, final_c5, n_points)
