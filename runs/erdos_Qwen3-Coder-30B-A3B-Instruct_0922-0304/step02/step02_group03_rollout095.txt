import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

    else:

        h[n_points//2:] = 0
    
    h = enforce_feasibility(h, n_points)
    best_c5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    
    # Constraint function for optimization
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 1000
    ftol = 1e-6
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define objective function
        def objective(h):
            return evaluate_overlap(h, dx)
        
        # Run optimization
        res = minimize(objective, h, method='L-BFGS-B', 
                       bounds=bounds, 
                       constraints={'type': 'eq', 'fun': constraint},
                       options={'maxiter': 100, 'ftol': ftol})
        
        if res.success:
            h = res.x
            h = enforce_feasibility(h, n_points)
            c5 = evaluate_overlap(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Early stopping if target reached
            if best_c5 <= 0.3807:
                break
        else:
            # If optimization fails, try a different approach
            h = np.random.rand(n_points)
            h = enforce_feasibility(h, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
