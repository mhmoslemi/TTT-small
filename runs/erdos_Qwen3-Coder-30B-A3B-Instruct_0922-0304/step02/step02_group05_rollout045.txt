import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Run particle swarm optimization for Erdős minimum overlap problem"""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    inertia_weight = 1.0
    cognitive_coeff = 2.0
    social_coeff = 2.0
    velocity_limit = 0.2
    perturbation = 0.05
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_feasible(h0, n_points, dx)
    
    particles.append(h0)
    velocities.append(np.zeros(n_points))
    
    # Add random particles to population
    for _ in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal and global bests
    p_best_positions = [p.copy() for p in particles]
    p_best_scores = [evaluate_erdos_solution(p) for p in particles]
    
    g_best_idx = np.argmin(p_best_scores)
    g_best_position = p_best_positions[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # Best solution tracking
    best_h = g_best_position.copy()
    best_c5 = g_best_score
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_i = (inertia_weight * velocities[i] + 
                   cognitive_coeff * r1 * (p_best_positions[i] - particles[i]) + 
                   social_coeff * r2 * (g_best_position - particles[i]))
            
            # Apply velocity limits
            v_i = np.clip(v_i, -velocity_limit, velocity_limit)
            
            # Add stochastic perturbation
            v_i += np.random.normal(0, perturbation, n_points)
            
            # Update position
            new_pos = particles[i] + v_i
            particles[i] = project_feasible(new_pos, n_points, dx)
            
            # Evaluate new position
            new_score = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_score < p_best_scores[i]:
                p_best_scores[i] = new_score
                p_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if new_score < g_best_score:
                    g_best_score = new_score
                    g_best_position = particles[i].copy()
                    last_improvement = iteration
                    
                    # Update best tracking
                    best_h = g_best_position.copy()
                    best_c5 = g_best_score
                    
                    # Early exit if target reached
                    if best_c5 <= 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
            
            # Update velocity
            velocities[i] = v_i
        
        # Adaptive inertia
        if iteration % 50 == 0:
            inertia_weight = max(0.8, inertia_weight - 0.02)
        
        # Reset stagnant particles
        if iteration - last_improvement > 100:
            for i in range(1, n_particles):  # Keep initial particle
                particles[i] = np.random.uniform(0, 1, n_points)
                particles[i] = project_feasible(particles[i], n_points, dx)
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                p_best_positions[i] = particles[i].copy()
                p_best_scores[i] = evaluate_erdos_solution(particles[i])
            last_improvement = iteration
        
        # Check for early exit
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
