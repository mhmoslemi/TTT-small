import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    num_particles = 30
    particles = []
    velocities = []

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, target_sum)
        particles.append(h_init.copy())
        velocities.append(np.zeros_like(h_init))
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = project_to_feasible(h_init, target_sum)
        particles.append(h_init.copy())
        velocities.append(np.zeros_like(h_init))

    # Generate other particles with blocky patterns
    for _ in range(num_particles - 1):
        h = np.ones(n_points)
        split1 = n_points // 3
        split2 = 2 * n_points // 3
        h[:split1] = 1
        h[split1:split2] = 0
        h[split2:] = 1
        # Add small Gaussian noise
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))

    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4

    # Track best solution
    global_best_idx = 0
    global_best_c5 = float('inf')
    global_best_h = None

    # Main PSO loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        particle_c5s = []
        for i, h in enumerate(particles):
            c5 = evaluate_c5(h, dx)
            particle_c5s.append(c5)
            
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                global_best_idx = i

        # Check if we've reached target
        if global_best_c5 <= 0.3807:
            break

        # Update each particle
        for i in range(len(particles)):
            h = particles[i]
            v = velocities[i]
            
            # Random weights
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            v = (inertia * v 
                 + alpha * r1 * (h - particles[i]) 
                 + beta * r2 * (global_best_h - h))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Apply constraints
            h_new = project_to_feasible(h_new, target_sum)
            
            # Store updated
            particles[i] = h_new
            velocities[i] = v

        # Update inertia
        inertia = max(0.3, inertia - inertia_decay)

    return (global_best_h.tolist(), global_best_c5, n_points)
