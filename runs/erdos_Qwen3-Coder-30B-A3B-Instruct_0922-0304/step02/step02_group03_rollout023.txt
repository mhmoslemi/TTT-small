import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)

    dx = 2.0 / n_points
    h = enforce_feasibility(h, n_points)
    
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial evaluation
    best_c5 = objective(h)
    best_h = h.copy()
    
    # Use scipy.optimize.minimize with L-BFGS-B
    res = minimize(objective, h, method='L-BFGS-B',
                   bounds=bounds,
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': 1000, 'ftol': 1e-6})
    
    if res.success:
        h_opt = res.x
        h_opt = enforce_feasibility(h_opt, n_points)
        c5 = evaluate_overlap(h_opt, dx)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
    
    # Final refinement with smaller step size if needed
    if time.time() - start_time < budget_s - 1:
        # Try a more refined optimization if we still have time
        h_refined = best_h.copy()
