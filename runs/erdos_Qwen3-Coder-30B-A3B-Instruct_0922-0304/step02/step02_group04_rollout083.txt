import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize using initial_h_values or a rectangular pulse
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Add noise and project
    h += np.random.normal(0, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Initialize parameters
    temperature = 0.1 * current_c5
    cooling_rate = 0.999
    noise_std = 0.01
    max_iterations = 10000
    gradient_steps = 5
    refinement_interval = 50
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate
        candidate_h = h + np.random.normal(0, noise_std, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = candidate_h
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            temperature *= cooling_rate
        else:
            # Accept with probability
            delta = candidate_c5 - current_c5
            if random.random() < np.exp(-delta / temperature):
                h = candidate_h
                current_c5 = candidate_c5
            temperature *= cooling_rate
            
        # Gradient descent refinement
        if iteration % refinement_interval == 0:
            step_size = 0.01
            for _ in range(gradient_steps):
                if time.time() - start_time > budget_s - 1:
                    break
                grad = compute_gradient(h, dx)
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                step_size *= 0.9
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
