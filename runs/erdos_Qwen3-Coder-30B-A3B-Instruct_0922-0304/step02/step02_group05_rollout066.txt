import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Use a random initialization
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlaps
        overlaps = correlate(h, 1 - h, mode="full") 
        max_overlap = np.max(overlaps)
        max_idx = np.argmax(overlaps)
        
        # Compute the gradient for the max overlap
        shifted = np.roll(h, max_idx - (n_points - 1))
        gradient = shifted - (1 - shifted)
        
        current_c5 = evaluate_erdos_solution(h)
        best_candidate_c5 = current_c5
        best_candidate_h = h.copy()
        
        # Line search
        for step_size in line_search_steps:
            h_candidate = h - step_size * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        h = best_candidate_h.copy()
        current_c5 = best_candidate_c5
    
        # Periodic relaxation
        if iteration % relaxation_interval == 0:
            # Convex relaxation approach
            h_relax = cp.Variable(n_points)
            obj = cp.Max(cp.sum(cp.multiply(h_relax, 1-h_relax)))
            constraints = [
                cp.sum(h_relax) == n_points/2,
                h_relax >= 0,
                h_relax <= 1
            ]
            prob = cp.Problem(cp.Minimize(obj), constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_new = h_relax.value.copy()
                    h_new = np.clip(h_new, 0, 1)
                    h_new = project_feasible(h_new, n_points, dx)
                    c5_test = evaluate_erdos_solution(h_new)
                    if c5_test < current_c5:
                        h = h_new.copy()
                        current_c5 = c5_test
                        lambda_reg *= 0.5  # Decrease regularization
            except:
                pass
                
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
