import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    # Clip to [0,1] and rescale to target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)
    
    h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    decay = 0.999
    tolerance = 1e-6
    last_improvement = 0
    patience = 100
    
    # Previous gradient for momentum
    prev_grad = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        if max_k <= 0:
            for i in range(n_points + max_k):
                grad[i] += (1 - h[i - max_k])
        else:
            for i in range(max_k, n_points):
                grad[i] += (1 - h[i - max_k])
        
        # Apply momentum
        grad = momentum * prev_grad + (1 - momentum) * grad
        prev_grad = grad.copy()
        
        # Adaptive step size
        step_size *= decay
        
        # Update h
        h -= step_size * grad
        
        # Project back to feasible region
        h = project_h(h, target_sum)
        
        # Check for convergence
        if iteration - last_improvement > patience:
            # Try to escape local minimum with random perturbation
            h += np.random.normal(0, 0.001, n_points)
            h = project_h(h, target_sum)
            last_improvement = iteration
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
