import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region [0,1] with sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    inertia_weight = 1.0
    cognitive_coeff = 2.0
    social_coeff = 2.0
    velocity_limit = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create a simple initial construction
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        particles.append(h.copy())
    
    # Add other random particles
    for _ in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial fitness and track best positions
    fitness = []
    p_best = []
    for i in range(n_particles):
        fit = evaluate_erdos_solution(particles[i])
        fitness.append(fit)
        p_best.append(particles[i].copy())
    
    # Global best
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    best_c5 = fitness[g_best_idx]
    best_h = g_best.copy()
    
    # PSO iterations
    iteration = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_coeff * r1 * (p_best[i] - particles[i]) + 
                           social_coeff * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate new fitness
            new_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_fitness < fitness[i]:
                fitness[i] = new_fitness
                p_best[i] = particles[i].copy()
                
                # Update global best
                if new_fitness < best_c5:
                    best_c5 = new_fitness
                    best_h = particles[i].copy()
                    last_improvement = iteration
                
        # Adapt inertia weight
        if iteration % 50 == 0:
            inertia_weight = max(0.8, inertia_weight - 0.02)
            
        # Early termination check
        if best_c5 <= 0.3807:
            break
            
        # Reset stagnant particles occasionally
        if iteration - last_improvement > 100 and iteration % 100 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.3:
                    particles[i] = np.random.uniform(0, 1, n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                    fitness[i] = evaluate_erdos_solution(particles[i])
                    p_best[i] = particles[i].copy()
    
    return (best_h.tolist(), best_c5, n_points)
