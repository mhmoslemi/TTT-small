import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with coarse discretization for faster exploration
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = enforce_feasibility(h, n_points)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = enforce_feasibility(h, n_points)
    
    # Optimization using L-BFGS-B with constraints
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimize with L-BFGS-B
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=[(0, 1) for _ in range(n_points)], 
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': 500, 'ftol': 1e-6})
    
    h_opt = res.x
    h_opt = enforce_feasibility(h_opt, n_points)
    c5_opt = evaluate_overlap(h_opt, dx)
    
    # Refine with finer discretization
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from previous solution
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, len(h_opt)), 
                       h_opt)
    h_fine = enforce_feasibility(h_fine, n_points)
    
    # Further optimization
    def objective_fine(h):
        return evaluate_overlap(h, dx)
    
    def constraint_fine(h):
        return np.sum(h) - n_points / 2
    
    # Optimize with L-BFGS-B on finer grid
    res = minimize(objective_fine, h_fine, method='L-BFGS-B', 
                   bounds=[(0, 1) for _ in range(n_points)], 
                   constraints={'type': 'eq', 'fun': constraint_fine},
                   options={'maxiter': 500, 'ftol': 1e-6})
    
    best_h = res.x
    best_h = enforce_feasibility(best_h, n_points)
    c5_bound = evaluate_overlap(best_h, dx)
    
    # Ensure we return the best solution found within time limit
    return (best_h.tolist(), c5_bound, n_points)
