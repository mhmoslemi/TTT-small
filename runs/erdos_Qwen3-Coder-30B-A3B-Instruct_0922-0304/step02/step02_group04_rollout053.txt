import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Add random individuals
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocity
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Initialize personal best positions and scores
    personal_best_positions = [p.copy() for p in swarm]
    personal_best_scores = [evaluate_c5(p, dx) for p in swarm]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Local refinement iteration
    local_refinement_freq = 20
    
    # Main PSO loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles and update personal bests
        for i in range(swarm_size):
            h = swarm[i]
            score = evaluate_c5(h, dx)
            
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = h.copy()
                
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = h.copy()
        
        # Early exit if target reached
        if global_best_score <= 0.3807:
            break
            
        # Update swarm
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v 
                 + c1 * r1 * (personal_best_positions[i] - h)
                 + c2 * r2 * (global_best_position - h))
            
            # Apply velocity limit
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            new_h = h + v
            new_h = project_to_feasible(new_h, n_points)
            swarm[i] = new_h
        
        # Local refinement every few iterations
        if iteration % local_refinement_freq == 0 and iteration > 0:
            # Refine top 10% particles
            num_refine = max(1, swarm_size // 10)
            sorted_indices = np.argsort(personal_best_scores)
            for i in range(num_refine):
                idx = sorted_indices[i]
                h = personal_best_positions[idx].copy()
                
                # Gradient descent refinement
                step_size = 0.01
                for _ in range(50):
                    corr = correlate(h, 1 - h, mode="full")
                    c5_val = np.max(corr) * dx
                    
                    # Approximate gradient
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - c5_val) / eps
                    
                    # Modify with gradient
                    h -= step_size * grad
                    h = project_to_feasible(h, n_points)
                
                # Update personal best if improved
                new_score = evaluate_c5(h, dx)
                if new_score < personal_best_scores[idx]:
                    personal_best_scores[idx] = new_score
                    personal_best_positions[idx] = h.copy()
                    if new_score < global_best_score:
                        global_best_score = new_score
                        global_best_position = h.copy()
    
    return (global_best_position.tolist(), global_best_score, n_points)
