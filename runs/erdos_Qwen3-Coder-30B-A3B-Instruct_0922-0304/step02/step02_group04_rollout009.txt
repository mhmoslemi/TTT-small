import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

    h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Set initial temperature
    T = 0.1 * best_c5
    cooling_rate = 0.999
    noise_std = 0.01
    
    iterations = 0
    max_iterations = 10000
    
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Generate candidate by perturbing current solution
        candidate = h + np.random.normal(0, noise_std, n_points)
        candidate = project_to_feasible(candidate, n_points)
        
        c5_candidate = evaluate_c5(candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_candidate < best_c5:
            h = candidate.copy()
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
            T *= cooling_rate
        else:
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = candidate.copy()
                T *= cooling_rate
            else:
                T *= cooling_rate
        
        # Gradient descent refinement every 50 iterations
        if iterations % 50 == 0:
            step_size = 0.01
            for _ in range(5):
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    c5_minus = evaluate_c5(h_minus, dx)
                    
                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                
                c5_new = evaluate_c5(h, dx)
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                
                step_size *= 0.9
                
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
