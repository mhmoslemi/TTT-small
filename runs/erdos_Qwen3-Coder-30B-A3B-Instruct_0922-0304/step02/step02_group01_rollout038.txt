import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    """Scale h to have sum equal to target_sum, preserving values in [0,1]"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Use binary search to find scaling factor
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    
    # Simple scaling as fallback
    scaled_h = h * target_sum / current_sum
    return np.clip(scaled_h, 0, 1)

def mutate_individual(h, mutation_rate=0.1, sigma=0.05):
    """Apply Gaussian mutation to individual"""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, sigma)
    return h_mut

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population and GA parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    elite_size = population_size // 10
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    # Fill population with random solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
        
    best_c5 = float('inf')
    best_h = None
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, h))
            
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        current_best_c5 = fitness[0][0]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = fitness[0][1].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        selected = [h for _, h in fitness[:population_size // 2]]
        
        # Create new population
        new_population = selected[:elite_size]  # Elitism
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_rate)
            child2 = mutate_individual(child2, mutation_rate)
            
            # Scale to target sum
            child1 = scale_to_target_sum(child1, target_sum)
            child2 = scale_to_target_sum(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final evaluation
    if best_h is None:
        # Fallback to last population
        fitness = [(evaluate_c5(h, dx), h) for h in population]
        fitness.sort(key=lambda x: x[0])
        best_c5 = fitness[0][0]
        best_h = fitness[0][1].copy()
    
    return (best_h.tolist(), best_c5, n_points)
