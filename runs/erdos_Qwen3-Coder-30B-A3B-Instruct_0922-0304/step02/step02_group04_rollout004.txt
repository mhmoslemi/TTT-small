import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute gradient of C5 w.r.t. h using finite differences
    eps = 1e-4
    grad = np.zeros_like(h)
    baseline_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - baseline_c5) / eps
    return grad

def pso_erdos(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        swarm.append(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        swarm.append(h0)
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        
    # Particle velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Personal best positions
    personal_best = [p.copy() for p in swarm]
    personal_best_c5 = [evaluate_c5(p, dx) for p in swarm]
    
    # Global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # Main loop
    best_solution = global_best.copy()
    best_c5 = global_best_c5
    
    for iter_num in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            cognitive = c1 * r1 * (personal_best[i] - swarm[i])
            social = c2 * r2 * (global_best - swarm[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate
            c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best[i] = swarm[i].copy()
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best = swarm[i].copy()
                    
                    # Early exit
                    if global_best_c5 <= 0.3807:
                        break
                        
            # Occasionally apply gradient refinement
            if iter_num % 20 == 0 and i < 5:  # Top 5 particles
                h_refined = swarm[i].copy()
                step_size = 0.01
                for _ in range(50):
                    grad = compute_gradient(h_refined, dx)
                    h_refined -= step_size * grad
                    h_refined = project_to_feasible(h_refined, n_points)
                    
                c5_refined = evaluate_c5(h_refined, dx)
                if c5_refined < personal_best_c5[i]:
                    personal_best_c5[i] = c5_refined
                    personal_best[i] = h_refined.copy()
                    
                    if c5_refined < global_best_c5:
                        global_best_c5 = c5_refined
                        global_best = h_refined.copy()
        
        # Update best solution found so far
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5
            best_solution = global_best.copy()
            
        if global_best_c5 <= 0.3807:
            break
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = pso_erdos(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
