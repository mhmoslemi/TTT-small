import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    rho = 1.0
    lambda_mult = np.zeros(2 * n_points - 1)
    
    max_iter = 500
    
    # ADMM with Augmented Lagrangian
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for current h
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Convex subproblem using CVXPY
        h_var = cp.Variable(n_points)
        
        # Objective: minimize overlap with AL penalty
        # augmented lagrangian: sum of objective with penalty term
        objective = cp.Minimize(
            cp.sum(cp.multiply(lambda_mult, corr)) + 
            0.5 * rho * cp.sum_squares(cp.multiply(h_var, 1 - h_var))
        )
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
        except:
            # Fallback if convex solve fails, use projected gradient step
            step = 0.01
            h_new = h.copy()
            for _ in range(100):
                corr_new = correlate(h_new, 1 - h_new, mode="full")
                c5_new = np.max(corr_new) * dx
                grad = np.zeros_like(h_new)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_new.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_new) / eps
                
                h_new -= step * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new.copy()
        
        # Update Lagrange multipliers
        corr_new = correlate(h, 1 - h, mode="full")
        lambda_mult += rho * (corr_new - np.max(corr_new))
        
        # Update penalty rho if needed
        if np.max(corr_new) > 0.99 * np.max(corr):
            rho *= 2
        
        # Maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
