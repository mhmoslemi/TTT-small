import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # L1-ball projection to maintain sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # If initial_h_values exists, use it as first particle
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        particles.append(project_feasible(h, n_points, dx))
    
    # Fill rest of particles with random
    for _ in range(num_particles - 1):
        h = np.random.rand(n_points)
        particles.append(project_feasible(h, n_points, dx))
    
    # Initialize velocities
    for _ in range(num_particles):
        velocities.append(np.random.normal(0, 0.05, n_points))
    
    # Evaluate initial fitness
    fitness = []
    for h in particles:
        fitness.append(evaluate_c5(h, dx))
    
    # Set personal bests
    p_best = particles[:]
    p_best_fitness = fitness[:]
    
    # Set global best
    best_idx = np.argmin(fitness)
    g_best = particles[best_idx].copy()
    g_best_fitness = fitness[best_idx]
    
    # PSO parameters
    w = 1.0
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    
    iter_count = 0
    last_improvement = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] + 
                c1 * r1 * (p_best[i] - particles[i]) + 
                c2 * r2 * (g_best - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            new_fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = new_fitness
            
            # Update global best
            if new_fitness < g_best_fitness:
                g_best = particles[i].copy()
                g_best_fitness = new_fitness
                last_improvement = iter_count
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Decrease inertia
        if iter_count % 10 == 0:
            w = max(0.8, w - 0.02)
        
        # Reset stagnant particles occasionally
        if iter_count % 50 == 0 and stagnation_count > 20:
            for i in range(num_particles):
                if np.random.rand() < 0.2:
                    h = np.random.rand(n_points)
                    particles[i] = project_feasible(h, n_points, dx)
                    velocities[i] = np.random.normal(0, 0.05, n_points)
        
        # Early exit
        if g_best_fitness <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_fitness, n_points)
