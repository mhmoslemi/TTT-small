import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    """Scale h to have the target sum while maintaining values in [0,1]"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    # Use binary search to find correct scaling factor
    low, high = 0.0, 2.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        scaled = h * mid
        scaled = np.clip(scaled, 0, 1)
        scaled_sum = np.sum(scaled)
        if scaled_sum > target_sum:
            high = mid
        else:
            low = mid
    scaled = h * low
    scaled = np.clip(scaled, 0, 1)
    return scaled

def crossover(parent1, parent2):
    """Single point crossover"""
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.05):
    """Gaussian mutation with scaling"""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += random.gauss(0, sigma)
    return h_mut

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * target_sum / np.sum(h_base)
        population.append(h_base)
    else:
        # Fallback initialization
        h_base = np.ones(n_points)
        h_base[n_points//2:] = 0
        h_base += np.random.normal(0, 0.05, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * target_sum / np.sum(h_base)
        population.append(h_base)
    
    # Add random individuals to population
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Stats for early stopping
    max_no_improvement = 100
    no_improvement_count = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        no_improvement_count += 1
        if no_improvement_count >= max_no_improvement:
            break
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament selection)
        selected_pop = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = random.sample(range(population_size), tournament_size)
            tournament_scores = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_scores)]
            selected_pop.append(population[winner_idx])
            
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep best individual
        elite_idx = np.argmin(fitness_scores)
        new_population.append(population[elite_idx].copy())
        elitism_count = 1
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected_pop, 2)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Enforce constraints
            child1 = scale_to_target_sum(child1, target_sum)
            child1 = np.clip(child1, 0, 1)
            
            child2 = scale_to_target_sum(child2, target_sum)
            child2 = np.clip(child2, 0, 1)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
            
        population = new_population[:population_size]
        
        if generation % 10 == 0:
            if time.time() - start_time > budget_s - 2:
                break
    
    # Final validation
    best_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
