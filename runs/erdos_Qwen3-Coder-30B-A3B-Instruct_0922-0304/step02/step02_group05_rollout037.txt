import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Algorithm parameters
    lambda_reg = 0.01
    max_iter = 1000
    relaxation_interval = 20
    step_sizes = [0.1, 0.5, 1.0]
    
    def line_search(h, grad, current_c5):
        best_h_new = h.copy()
        best_c5_new = current_c5
        for alpha in step_sizes:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate.tolist())
            if candidate_c5 < best_c5_new:
                best_h_new = h_candidate
                best_c5_new = candidate_c5
        return best_h_new, best_c5_new
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Compute gradient approximation for the maximizing shift
        # Approximate gradient: derivative of correlation w.r.t. h
        # ∇_i overlap_k = (1 - h_{i+k}) - h_{i+k}
        h_shifted_l = np.roll(h, max_k)
        h_shifted_r = np.roll(h, -max_k)
        grad = h_shifted_l - (1 - h_shifted_r)
        
        # Line search
        h_new, c5_new = line_search(h, grad, best_c5)
        
        # Update
        h = h_new
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        # Periodically solve convex relaxation
        if iteration % relaxation_interval == 0 and iteration > 0:
            # Convex relaxation: minimize overlap + lambda * sum(h * log(h))
            try:
                h_var = cp.Variable(n_points)
                h_constr = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
                # Approximate objective
                overlap_approx = 0
                # Use a simplified version: sum(h * (1 - h)) which is related to variance
                obj = cp.Minimize(cp.sum(h_var) - cp.sum(cp.multiply(h_var, 1 - h_var)) + 
                                lambda_reg * cp.sum(cp.entr(h_var)))
                prob = cp.Problem(obj, h_constr)
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax = h_var.value
                    h_relax = project_feasible(h_relax, n_points, dx)
                    if evaluate_erdos_solution(h_relax.tolist()) < best_c5:
                        best_h = h_relax.copy()
                        best_c5 = evaluate_erdos_solution(h_relax.tolist())
            except:
                pass  # fall back to previous solution if convex relaxation fails
            
            if best_c5 <= 0.3807:
                break

    return (best_h.tolist(), best_c5, n_points)
