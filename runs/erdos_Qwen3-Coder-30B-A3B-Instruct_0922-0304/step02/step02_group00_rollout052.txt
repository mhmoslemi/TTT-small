import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    patience = 0
    min_patience = 100

        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")

        k_max = np.argmax(corr)
        
        # Convex optimization to reduce this specific overlap
        h_var = Variable(n_points)
        
        # Formulate the specific overlap term as a linear function of h
        # The overlap at shift k_max is: sum h[i] * (1 - h[(i + k_max) % n_points])
        # We have sum of h[i] * (1 - h[(i + k_max) % n_points]) = sum(h[i]) - sum(h[i] * h[(i + k_max) % n_points])
        
        # This line will define a top-level variable in the function scope
        # We'll compute the expected overlap for a single k_max efficiently 
        # Create the linear term that represents the overlap at k_max
        # We'll calculate this efficiently through the correlation
        # In fact, this is one collision term, not the full formula
        # Let's treat this as a convex optimization for minimizing the overlap
        
        # Let's formulate convex optimization problem to minimize that specific overlap
        try:  
            # Create variable vector
            h_var = cp.Variable(n_points)
            
            # The constraint that sum(h) = n_points / 2 (i.e., integral of h is 1)
            constraints = [cp_sum(h_var) == n_points / 2]
            
            # Bounds
            for i in range(n_points):
                constraints.append(h_var[i] <= 1)
                constraints.append(h_var[i] >= 0)
                
            # Objective: minimize overlap at shift k_max
            # This is sum_{i=0}^{n-1} h[i] * (1 - h[(i+k_max) % n]) * dx
            # or equivalently, sum_{i=0}^{n-1} h[i] - sum_{i=0}^{n-1} h[i] * h[(i+k_max) % n]
            # linear part: sum h[i] = n_points/2 constant
            # nonlinear part: sum h[i] * h[(i+k_max) % n] 
            # We rewrite it (since sum h[i] is constant) to get:
            # minimize - sum(h[i] * h[(i+k_max) % n])
            # This is still nonconvex, but because of the very nature of the problem, 
            # and to get around the nonconvexity, we'll solve it using Frank-Wolfe style method
            # We simplify using the linearization of the objective
            
            # For a Frank-Wolfe approach, we approximate the nonlinear part:
            # value term = h[i] * h_old[(i+k_max) % n]  
            # For the linearization, here's a technique:
            # we keep the constant of the sum h[i] = n/2  
            # and we try to minimize - sum(h[i] * h[(i + k_max) % n]) with a fixed h
            # Suppose we linearize h for i: h[i] ~ h_old[i] at current point
            # Then this is a scalar product we are optimizing over
            
            # Actually a valid approach is to fix h as previous solution, and 
            # minimize a differentiable convex surrogate for the k_max term
            # But to proceed simply, let's actually use Frank-Wolfe/coordinate descent approach

            # Simplified direct approach: Use coordinate descent + projection for iterations below

            def compute_overlap_at_shift(h_vals, k_shift):

            # gradient for overlap at k_max wrt h[i] = 1 - h[(i + k_max)% n]

            for _ in range(20):  # Small iteration for gradient update

                    j = (i + k_max) % n_points
                    grad[i] = 1 - h[j]
                    
                # Update with negative gradient (descent)
                new_h = h - step_size * grad
                new_h = np.clip(new_h, 0, 1)
                # Rescale with sum constraint
                new_h = new_h * (n_points / 2) / np.sum(new_h)

        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit if we hit target or patience exceeded
        if best_c5 <= 0.3807 or patience >= min_patience:
            break

    return (best_h.tolist(), best_c5, n_points)
