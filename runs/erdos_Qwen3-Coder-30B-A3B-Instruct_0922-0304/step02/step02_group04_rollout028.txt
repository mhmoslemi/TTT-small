import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) / 2
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm with the provided starting construction if available
    if 'initial_h_values' in globals():
        swarm = [np.array(initial_h_values)]
    else:
        swarm = [np.ones(n_points) * 0.5]
        
    # Add random particles
    for _ in range(29):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        
    # Initialize personal best and global best
    personal_best = [h.copy() for h in swarm]
    personal_best_c5 = [evaluate_c5(h, dx) for h in swarm]
    gbest_idx = np.argmin(personal_best_c5)
    global_best = personal_best[gbest_idx].copy()
    global_best_c5 = personal_best_c5[gbest_idx]
    
    # PSO parameters
    w = 0.7
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Particle velocities
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(30)]
    
    # Main PSO loop
    max_iterations = 500
    local_refinement_freq = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(30):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) +
                            c2 * r2 * (global_best - swarm[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best[i] = swarm[i].copy()
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best = swarm[i].copy()
                    
        # Optional local refinement
        if iteration % local_refinement_freq == 0:
            # Refine the top 3 particles
            top_indices = np.argsort(personal_best_c5)[:3]
            for i in top_indices:
                h = personal_best[i].copy()
                step_size = 0.01
                for _ in range(50):
                    # Compute gradient using finite differences
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        
                        corr = correlate(h, 1 - h, mode="full")
                        c5 = np.max(corr) * dx
                        
                        grad[j] = (c5_plus - c5) / eps
                    
                    # Update with gradient ascent step
                    h += step_size * grad
                    h = project_to_feasible(h, n_points)
                
                # Update personal best if improved
                c5 = evaluate_c5(h, dx)
                if c5 < personal_best_c5[i]:
                    personal_best_c5[i] = c5
                    personal_best[i] = h.copy()
                    if c5 < global_best_c5:
                        global_best_c5 = c5
                        global_best = h.copy()
    
    # Final validation
    global_best = project_to_feasible(global_best, n_points)
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
