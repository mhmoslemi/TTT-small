import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if it exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Add random particles
    for _ in range(num_particles - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(num_particles):
        v = np.random.normal(0, 0.01, n_points)
        velocities.append(v.copy())
    
    # Evaluate initial positions
    fitnesses = []
    for i, p in enumerate(particles):
        c5 = evaluate_c5(p, dx)
        fitnesses.append(c5)
    
    # Initialize personal bests
    p_best_positions = [p.copy() for p in particles]
    p_best_fitnesses = fitnesses[:]
    
    # Initialize global best
    best_idx = np.argmin(fitnesses)
    g_best_position = particles[best_idx].copy()
    g_best_fitness = fitnesses[best_idx]
    
    # PSO parameters
    w = 1.0  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    max_vel = 0.2
    
    # Main PSO loop
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best_positions[i] - particles[i]) +
                           c2 * r2 * (g_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < p_best_fitnesses[i]:
                p_best_fitnesses[i] = c5
                p_best_positions[i] = particles[i].copy()
            
            # Update global best
            if c5 < g_best_fitness:
                g_best_fitness = c5
                g_best_position = particles[i].copy()
                last_improvement = iteration
                
        # Adaptive inertia
        if iteration % 50 == 0:
            w = max(0.8, w * 0.98)
        
        # Reset stagnant particles
        if iteration - last_improvement > 50:
            for i in range(num_particles):
                if np.random.rand() < 0.1:
                    particles[i] = np.random.rand(n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    velocities[i] = np.random.normal(0, 0.01, n_points)
                    
                    # Update personal best
                    c5 = evaluate_c5(particles[i], dx)
                    p_best_fitnesses[i] = c5
                    p_best_positions[i] = particles[i].copy()
            
            # Reset global best if necessary
            current_best_idx = np.argmin(p_best_fitnesses)
            if p_best_fitnesses[current_best_idx] < g_best_fitness:
                g_best_fitness = p_best_fitnesses[current_best_idx]
                g_best_position = p_best_positions[current_best_idx].copy()
            last_improvement = iteration
                
        # Early exit if target reached
        if g_best_fitness <= 0.3807:
            break
    
    return (g_best_position.tolist(), g_best_fitness, n_points)
