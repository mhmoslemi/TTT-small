import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1000.0
    alpha = 0.999
    T_min = 1.0
    max_iter = 10000
    
    # Perturbation parameters
    delta = 0.1
    perturb_fraction = 0.1
    n_perturb = max(1, int(perturb_fraction * n_points))
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create candidate solution by perturbing h
        h_candidate = h.copy()
        
        # Select random indices to perturb
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected values
        total_delta = delta * n_perturb
        for i in indices:
            h_candidate[i] += delta
        # Distribute the remaining adjustment to other indices
        remaining_sum = total_delta - np.sum(np.maximum(h_candidate[indices] - delta, 0))
        if remaining_sum > 0:
            # Distribute the remaining amount
            other_indices = np.setdiff1d(np.arange(n_points), indices)
            if len(other_indices) > 0:
                avg_reduction = remaining_sum / len(other_indices)
                for i in other_indices:
                    h_candidate[i] = max(0, h_candidate[i] - avg_reduction)
        
        # Ensure all values are in [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum to required value
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Evaluate current
        current_c5 = evaluate_c5(h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = h_candidate
            current_c5 = candidate_c5
        else:
            # Accept with probability
            delta_E = candidate_c5 - current_c5
            if np.exp(-delta_E / T) > np.random.rand():
                h = h_candidate
                current_c5 = candidate_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Cool down
        T = max(T_min, alpha * T)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Return the best solution found
    return (best_h.tolist(), final_c5, n_points)
