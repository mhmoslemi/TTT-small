import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

        h = project_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Compute overlaps for all shifts and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)  # Convert index to shift
        
        # Compute gradient (simplified gradient approximation)
        grad = np.zeros_like(h)
        eps = 1e-4
        h_plus = h.copy()
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            h_plus = project_h(h_plus, target_sum)

        h = h + velocity
        
        # Project to feasible region
        h = project_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
