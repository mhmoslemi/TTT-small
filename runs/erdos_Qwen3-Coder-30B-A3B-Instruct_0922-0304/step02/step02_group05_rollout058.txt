import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a basic initial solution
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 1e-3
    max_iterations = int(budget_s / 0.1)  # Roughly 6000 steps
    relaxation_interval = 50
    lambda_reg = 0.01
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find max
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        max_indices = np.where(overlaps == max_overlap)[0]
        
        # Use the first maximum index for gradient descent
        k = max_indices[0] - (n_points - 1)  # Convert to offset
        
        # Compute finite difference gradient
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, n_points, dx)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_plus = np.max(overlap_plus) * dx
            grad[i] = (max_plus - current_c5) / eps
        
        # Update with gradient descent
        h_new = h - learning_rate * grad
        h_new = project_feasible(h_new, n_points, dx)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Periodic convex relaxation
        if iteration % relaxation_interval == 0 and iteration > 0:
            try:
                # Convex relaxation
                h_cvx = cp.Variable(n_points)
                objective = cp.Minimize(cp.sum(cp.multiply(h_cvx, 1 - h_cvx)))
                constraints = [
                    h_cvx >= 0,
                    h_cvx <= 1,
                    cp.sum(h_cvx) == n_points/2
                ]
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_relaxed = h_cvx.value
                    h_relaxed = project_feasible(h_relaxed, n_points, dx)
                    relaxed_c5 = evaluate_c5(h_relaxed, dx)
                    if relaxed_c5 < current_c5:
                        h = h_relaxed
                        current_c5 = relaxed_c5
                        
                        if relaxed_c5 < best_c5:
                            best_c5 = relaxed_c5
                            best_h = h_relaxed.copy()
                            
            except Exception:
                pass  # Failed relaxation, continue with current solution
            
            # Adaptive regularization
            lambda_reg *= 0.95  # Reduce regularization slightly
            
    return (best_h.tolist(), best_c5, n_points)
