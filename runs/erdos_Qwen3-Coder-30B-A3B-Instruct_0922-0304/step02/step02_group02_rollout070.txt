import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 5000
    perturb_factor = 0.1
    
    T = T0
    iteration = 0
    
    while T > T_min and time.time() - start_time < budget_s - 1:
        if iteration >= max_iter:
            break
            
        # Generate candidate solution by perturbing current solution
        h_candidate = h.copy()
        
        # Randomly select indices to modify
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturbation amount
        delta = 0.1
        
        # Increase selected values
        for i in indices:
            h_candidate[i] = min(1.0, h_candidate[i] + delta)
            
        # Decrease other values to maintain sum
        total_increase = np.sum(h_candidate[indices]) - np.sum(h[indices])
        if total_increase > 0:
            remaining_indices = [i for i in range(n_points) if i not in indices]
            decrease_amount = total_increase / len(remaining_indices)
            for i in remaining_indices:
                h_candidate[i] = max(0.0, h_candidate[i] - decrease_amount)
        
        # Normalize to maintain sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute current and candidate C5 values
        current_c5 = evaluate_c5(h, dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Accept or reject the candidate
        delta_E = candidate_c5 - current_c5
        
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_candidate.copy()
            
        # Update best solution
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update temperature and iteration
        T = alpha * T
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
