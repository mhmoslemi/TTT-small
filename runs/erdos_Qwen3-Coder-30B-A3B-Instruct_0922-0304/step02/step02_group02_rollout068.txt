import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    # Ensure h is in [0,1] and sum equals n_points / 2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 500
    inertia = 0.7
    c1 = c2 = 2.0
    local_search_frac = 0.05
    
    # Initialize particles
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        # Simple initialization: half 1s, half 0s
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        
    particles = []
    for i in range(n_particles):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track best
    global_best_c5 = float('inf')
    global_best_h = None
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        particle_c5s = []
        for i, h in enumerate(particles):
            c5 = evaluate_c5(h, dx)
            particle_c5s.append(c5)
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                
        # Early exit if we hit target
        if global_best_c5 <= 0.3807:
            break
            
        # Update particles
        local_search_indices = np.argsort(particle_c5s)[:int(n_particles * local_search_frac)]
        
        for i in range(n_particles):
            if time.time() - start_time > budget_s - 1:
                break
                
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (particles[i] - particles[i]) + 
                            c2 * r2 * (global_best_h - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair
            particles[i] = repair_h(particles[i], n_points)
            
        # Local search on top particles
        for idx in local_search_indices:
            if time.time() - start_time > budget_s - 1:
                break
                
            h = particles[idx]
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
            
            # Update with gradient descent
            h = h - 0.01 * grad
            h = repair_h(h, n_points)
            particles[idx] = h
            
    # Final evaluation
    final_c5 = evaluate_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
