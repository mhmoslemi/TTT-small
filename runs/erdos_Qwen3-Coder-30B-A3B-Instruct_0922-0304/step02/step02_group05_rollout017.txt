import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with best known construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple starting configuration
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_frequency = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap and find max
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Compute gradient for the worst overlap
        h_shifted = np.roll(h, max_k)
        grad = h_shifted - (1 - h_shifted)
        
        # Line search
        best_candidate = h.copy()
        best_candidate_c5 = best_c5
        for alpha in line_search_steps:
            candidate_h = h - alpha * grad
            candidate_h = project_feasible(candidate_h, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            if candidate_c5 < best_candidate_c5:
                best_candidate = candidate_h
                best_candidate_c5 = candidate_c5
                break
        
        h = best_candidate
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = best_candidate.copy()
        
        # Periodic convex relaxation
        if iteration % relaxation_frequency == 0:
            # Convex relaxation to escape local minima
            h_relax = cp.Variable(n_points)
            obj = cp.Max(cp.multiply(overlaps, dx)) + lambda_reg * cp.sum(cp.entr(h_relax))
            prob = cp.Problem(cp.Minimize(obj), 
                              [cp.sum(h_relax) == n_points/2,
                               h_relax >= 0, 
                               h_relax <= 1])
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax_sol = h_relax.value
                    # Project to feasible set
                    h_relax_sol = project_feasible(h_relax_sol, n_points, dx)
                    relax_c5 = evaluate_erdos_solution(h_relax_sol)
                    if relax_c5 < best_c5:
                        best_c5 = relax_c5
                        best_h = h_relax_sol.copy()
                        h = h_relax_sol.copy()
                else:
                    lambda_reg *= 0.5
            except:
                lambda_reg *= 0.5
                
        # Early exit if target is reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
