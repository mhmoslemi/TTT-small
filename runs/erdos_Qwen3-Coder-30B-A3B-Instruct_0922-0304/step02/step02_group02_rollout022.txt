import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize to exact sum
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, scaled properly
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 100000
    perturbation_size = 0.1
    
    # Main SA loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current evaluation
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Create perturbed solution
        h_prime = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        delta = perturbation_size / len(indices)
        
        # Adjust values
        for i in indices:
            h_prime[i] += delta
        # Distribute excess to other entries to maintain sum
        total_excess = np.sum(h_prime) - (n_points / 2)
        if total_excess > 0:
            # Decrease other entries
            remaining = np.setdiff1d(np.arange(n_points), indices)
            decrease = total_excess / len(remaining) if len(remaining) > 0 else 0
            h_prime[remaining] = np.maximum(0, h_prime[remaining] - decrease)
        elif total_excess < 0:
            # Increase other entries
            remaining = np.setdiff1d(np.arange(n_points), indices)
            increase = -total_excess / len(remaining) if len(remaining) > 0 else 0
            h_prime[remaining] = np.minimum(1, h_prime[remaining] + increase)
            
        # Ensure bounds and normalize
        h_prime = np.clip(h_prime, 0, 1)
        h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # Evaluate perturbed solution
        candidate_c5 = evaluate_c5(h_prime, dx)
        
        # Accept or reject
        delta_E = candidate_c5 - current_c5
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.random():
            h = h_prime.copy()
            
        # Cool down
        T = max(T_min, alpha * T)
        
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
