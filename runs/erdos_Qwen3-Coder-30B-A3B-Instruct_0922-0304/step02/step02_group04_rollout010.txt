import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_relaxation(h_init, n_points, dx):
    # Create variables
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2, C >= 0]
    
    # Overlap constraints - simplified as upper bound on full correlation
    # For a more precise relaxation we'd consider each shift separately
    # But for now, we'll use a relaxed approach focusing on max overlap
    # This is an approximation
    max_overlap = cp.max(cp.convolve(h, 1 - h, mode="full"))
    constraints.append(max_overlap <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Formulate and solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == 'optimal':
            h_sol = h.value
            c5_value = prob.value
            return project_to_feasible(h_sol, n_points), c5_value
    except:
        pass
    
    return h_init, evaluate_c5(h_init, dx)

def local_search(h, dx, max_iter=50):
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for _ in range(max_iter):
        # Simple gradient estimation
        epsilon = 1e-4
        grad = np.zeros_like(h)
        c5_val = evaluate_c5(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_val) / epsilon
        
        # Update with gradient descent
        step_size = 0.001
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, len(h))
        
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        h = h_new
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize

    else:

        h[:n_points//2] = 1.0
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation step
        h_relaxed, c5_relaxed = solve_convex_relaxation(h, n_points, dx)
        
        # Local search on relaxed solution
        h_local, c5_local = local_search(h_relaxed, dx)
        
        # Update best solution
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Perturb the solution
        h = best_h + np.random.normal(0, 0.005, n_points)
        h = project_to_feasible(h, n_points)
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
