import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize population
    population_size = 30
    population = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
    
    # Create initial population
    population = [h.copy()]
    velocities = [np.zeros(n_points)]
    
    # Add perturbed versions
    for _ in range(population_size - 1):
        h_pert = h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h_pert += noise
        h_pert = project_to_feasible(h_pert, n_points)
        population.append(h_pert)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    cognitive_weight = 0.8
    social_weight = 0.8
    velocity_limit = 0.3
    inertia_weight = 0.7
    inertia_decay = 1e-4
    
    # Track personal and global bests
    personal_best = [p.copy() for p in population]
    personal_best_scores = [evaluate_c5(p, dx) for p in population]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    best_c5 = global_best_score
    best_h = global_best.copy()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia_weight = max(0.4, inertia_weight * (1 - inertia_decay))
        
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (personal_best[i] - population[i])
            social = social_weight * r2 * (global_best - population[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            population[i] += velocities[i]
            
            # Enforce constraints
            population[i] = project_to_feasible(population[i], n_points)
            
            # Evaluate
            score = evaluate_c5(population[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = population[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = population[i].copy()
                global_best_score = score
                
                if global_best_score <= 0.3807:
                    break
        
        # Update best solution found
        if global_best_score < best_c5:
            best_c5 = global_best_score
            best_h = global_best.copy()
    
    return (best_h.tolist(), best_c5, n_points)
