import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_particles = 30
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * target_sum / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros(n_points))
    
    # Add other particles with blocky patterns and noise
    for i in range(n_particles - len(particles)):
        h = np.zeros(n_points)
        # Two blocks
        mid = n_points // 2
        h[:mid] = 1
        h[mid:] = 0
        h += np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track best solutions
    best_particle = None
    best_c5 = float('inf')
    best_global = None
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update inertia
        inertia = max(0.3, inertia * (1 - inertia_decay))
        
        # Evaluate all particles
        particle_c5 = []
        for i, h in enumerate(particles):
            c5 = evaluate_c5(h, dx)
            particle_c5.append(c5)
            
            # Update individual best
            if c5 < best_c5:
                best_c5 = c5
                best_particle = h.copy()
                if best_global is None or c5 < evaluate_c5(best_global, dx):
                    best_global = h.copy()
        
        # Early termination check
        if best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            if best_global is None:
                best_global = h.copy()
            
            cognitive = alpha * r1 * (h - particles[i])
            social = beta * r2 * (best_global - h)
            
            v = inertia * v + cognitive + social
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            h_new = project_to_feasible(h_new, target_sum)
            
            # Update particle
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
    
    # Final evaluation
    if best_particle is not None:
        final_c5 = evaluate_c5(best_particle, dx)
    else:
        final_c5 = float('inf')
    
    return (best_particle.tolist(), final_c5, n_points)
