import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute gradient of C5 w.r.t h using finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def pso_erdos(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles with the initial solution and perturbations
    particles = []
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points) * 0.5
    
    particles.append(base_h)
    for _ in range(n_particles - 1):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Personal best positions and their C5 values
    personal_best = particles.copy()
    personal_best_c5 = [evaluate_c5(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[best_idx].copy()
    global_best_c5 = personal_best_c5[best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocity = (w * velocity + 
                       c1 * r1 * (personal_best[i] - particle) + 
                       c2 * r2 * (global_best - particle))
            
            # Clip velocity
            velocity = np.clip(velocity, -v_max, v_max)
            
            # Update position
            new_pos = particle + velocity
            new_pos = project_to_feasible(new_pos, n_points)
            
            # Evaluate new position
            new_c5 = evaluate_c5(new_pos, dx)
            
            # Update personal best
            if new_c5 < personal_best_c5[i]:
                personal_best[i] = new_pos.copy()
                personal_best_c5[i] = new_c5
                
                # Update global best if needed
                if new_c5 < global_best_c5:
                    global_best = new_pos.copy()
                    global_best_c5 = new_c5
            
            # Update particles and velocities
            particles[i] = new_pos
            velocities[i] = velocity
            
        # Occasionally apply local gradient-based refinement
        if iteration % 20 == 0 and iteration > 0:
            for i in range(n_particles // 5):  # Top 20% refined
                idx = random.randint(0, n_particles - 1)
                h = particles[idx].copy()
                step_size = 0.01
                
                # Gradient ascent step
                for _ in range(50):
                    grad = compute_gradient(h, dx)
                    h -= step_size * grad
                    h = project_to_feasible(h, n_points)
                
                # Update if better
                c5 = evaluate_c5(h, dx)
                if c5 < personal_best_c5[idx]:
                    personal_best[idx] = h.copy()
                    personal_best_c5[idx] = c5
                    if c5 < global_best_c5:
                        global_best = h.copy()
                        global_best_c5 = c5
    
    return global_best, global_best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = pso_erdos(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
