import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Coarse discretization for faster exploration
    n_points_coarse = 35
    dx_coarse = 2.0 / n_points_coarse
    
    # Initialize with a simple two-block construction
    h_coarse = np.ones(n_points_coarse)
    h_coarse[n_points_coarse//2:] = 0
    h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
    
    # Rescale h to sum to n_points/2
    h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
    
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    # Minimize using L-BFGS-B with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points_coarse)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_coarse / 2}
    
    for _ in range(500):  # max iterations for coarse phase
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with L-BFGS-B
        result = minimize(
            lambda x: evaluate_c5(x, dx_coarse),
            h_coarse,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'gtol': 1e-5}
        )
        
        if not result.success:
            break
            
        h_new = result.x
        new_c5 = evaluate_c5(h_new, dx_coarse)
        
        if new_c5 < best_c5_coarse:
            best_c5_coarse = new_c5
            best_h_coarse = h_new.copy()
            
        h_coarse = h_new.copy()
    
    # Stage 2: Refinement with higher resolution (n=70)
    n_points_fine = 70
    dx_fine = 2.0 / n_points_fine
    
    # Upsample from coarse to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points_fine), 
                       np.linspace(0, 2, n_points_coarse), 
                       best_h_coarse)
    
    # Add small random perturbations
    perturb = np.random.normal(0, 0.05, n_points_fine)
    h_fine = h_fine + perturb
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points_fine / 2) / np.sum(h_fine)
    
    # Keep the best in fine resolution
    best_c5_fine = float('inf')
    best_h_fine = h_fine.copy()
    
    bounds = [(0, 1) for _ in range(n_points_fine)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_fine / 2}
    
    for _ in range(1000):  # max iterations for fine refinement
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(
            lambda x: evaluate_c5(x, dx_fine),
            h_fine,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'gtol': 1e-5}
        )
        
        if not result.success:
            break
            
        h_new = result.x
        new_c5 = evaluate_c5(h_new, dx_fine)
        
        if new_c5 < best_c5_fine:
            best_c5_fine = new_c5
            best_h_fine = h_new.copy()
            
        h_fine = h_new.copy()
    
    return (best_h_fine.tolist(), best_c5_fine, n_points_fine)
