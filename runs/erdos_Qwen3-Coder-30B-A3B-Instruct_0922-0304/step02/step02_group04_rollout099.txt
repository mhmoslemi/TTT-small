import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    """Compute gradient of C5 using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def line_search(h, grad, dx, max_alpha=1.0):
    """Backtracking line search for sufficient decrease."""
    alpha = max_alpha
    c5_base = evaluate_c5(h, dx)
    while alpha > 1e-10:
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        # Armijo condition
        if c5_new <= c5_base - 0.5 * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= 0.5
    return alpha, h

    H = np.eye(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Check convergence
        if np.linalg.norm(grad) < 1e-3:
            break
            
        # Solve quadratic subproblem: min 0.5 * d^T H d + grad^T d
        # Subject to: sum(d) = 0, 0 <= h+d <= 1
        try:
            # Using scipy optimization for the quadratic subproblem
            # We'll solve this with constraint optimization
            d = - np.linalg.solve(H, grad)
            
            # Project d to satisfy bounds and sum constraint
            # The direction may not satisfy constraints. We'll project it.
            # And slightly adjust if needed to make sum zero
            current_sum = np.sum(h)
            target_sum = n_points / 2
            
            # Adjust direction to preserve sum constraint
            d_sum = np.sum(d)

            # Now perform line search
            alpha, h_new = line_search(h, d, dx)
            
            if alpha > 1e-10:
                h = h_new
                
                # Update Hessian using BFGS formula
                y = compute_gradient(h, dx) - grad  # This is an approximation
                if np.linalg.norm(y) > 1e-10 and np.linalg.norm(d) > 1e-10:
                    rho = 1.0 / np.dot(y, d)
                    I = np.eye(n_points)
                    H = (I - rho * np.outer(d, y)) @ H @ (I - rho * np.outer(y, d)) + rho * np.outer(d, d)
                
                # Update best solution

    return best_h, best_c5
