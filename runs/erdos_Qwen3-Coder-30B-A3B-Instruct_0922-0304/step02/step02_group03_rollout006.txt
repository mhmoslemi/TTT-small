import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # If initial_h_values exists, use it as one particle
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Fallback to simple block structure
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial particles with some variation
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    
    # Fill remaining particles with blocky patterns and noise
    for _ in range(population_size - 1):
        h = np.zeros(n_points)
        split1 = n_points // 3
        split2 = 2 * n_points // 3
        h[:split1] = 1.0
        h[split2:] = 0.0
        h[split1:split2] = 1.0  # Middle part
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track best particles
    best_individuals = [p.copy() for p in particles]
    best_individual_scores = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_individual_scores)
    global_best = particles[global_best_idx].copy()
    global_best_score = best_individual_scores[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            h = particles[i]
            v = velocities[i]
            best_individual = best_individuals[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia * v + 
                 alpha * r1 * (best_individual - h) + 
                 beta * r2 * (global_best - h))
            
            # Limit velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Apply constraints
            h_new = project_to_feasible(h_new, n_points)
            
            # Evaluate new position
            new_score = evaluate_c5(h_new, dx)
            
            # Update personal best
            if new_score < best_individual_scores[i]:
                best_individual_scores[i] = new_score
                best_individuals[i] = h_new.copy()
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best = h_new.copy()
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
        
        # Update inertia
        inertia = max(0.4, inertia * (1 - inertia_decay))
        
        # Early stopping
        if global_best_score <= 0.3807:
            break
    
    # Final verification
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
