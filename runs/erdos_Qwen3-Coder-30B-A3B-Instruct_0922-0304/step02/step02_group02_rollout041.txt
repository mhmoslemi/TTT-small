import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Clip values and rescale to satisfy sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2
    
    # Initialize swarm with PSO
    n_particles = 50
    max_iter = 500
    
    # Initialize particles - use initial_h_values or create symmetric pattern
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        # Create some symmetric starting configuration
        base_h = np.zeros(n_points)
        base_h[:n_points//4] = 1
        base_h[n_points//4:3*n_points//4] = 0.5
        base_h[3*n_points//4:] = 1
    
    particles = []
    velocities = []
    
    for i in range(n_particles):
        # Add small noise to base
        noise = np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = base_h + noise
        h = repair_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.7
    c1, c2 = 2.0, 2.0
    inertia = w
    
    # Track bests
    pbest = [particles[i].copy() for i in range(n_particles)]
    pbest_fitness = [float('inf')] * n_particles
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Evaluate initial particles
    for i in range(n_particles):
        fitness = evaluate_c5(particles[i], dx)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # PSO iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair feasibility
            particles[i] = repair_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = particles[i].copy()
            
            # Update global best
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest_idx = i
            
            # Early exit
            if gbest_fitness <= 0.3807:
                break
                
        # Local search on top 5%
        if iteration % 10 == 0 and iteration > 0:
            top_k = max(1, n_particles // 20)
            indices = np.argsort(pbest_fitness)[:top_k]
            for j in indices:
                if time.time() - start_time > budget_s - 2:
                    break
                h = pbest[j].copy()
                for _ in range(20):
                    if time.time() - start_time > budget_s - 2:
                        break
                    # Gradient estimation  
                    eps = 1e-4
                    grad = np.zeros_like(h)
                    base_c5 = evaluate_c5(h, dx)
                    for k in range(n_points):
                        h_plus = h.copy()
                        h_plus[k] += eps
                        h_plus = repair_h(h_plus, target_sum)
                        plus_c5 = evaluate_c5(h_plus, dx)
                        grad[k] = (plus_c5 - base_c5) / eps
                    h = h - 0.01 * grad
                    h = repair_h(h, target_sum)
                pbest[j] = h.copy()
                pbest_fitness[j] = evaluate_c5(h, dx)
                if pbest_fitness[j] < gbest_fitness:
                    gbest_fitness = pbest_fitness[j]
                    gbest_idx = j
    
    # Final evaluation
    final_h = particles[gbest_idx]
    final_c5 = evaluate_c5(final_h, dx)
    
    if final_c5 > gbest_fitness:
        # Use the best particle if it's significantly better
        final_h = pbest[gbest_idx]
        final_c5 = pbest_fitness[gbest_idx]
    
    return (final_h.tolist(), final_c5, n_points)
