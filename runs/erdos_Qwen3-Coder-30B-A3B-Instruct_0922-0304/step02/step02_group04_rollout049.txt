import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    # Compute gradient using finite differences
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def solve_quadratic_subproblem(H, g, n_points):
    # Solve unconstrained quadratic subproblem
    # min 0.5 * d^T * H * d + g^T * d
    # with sum(d) = 0 and 0 <= h+d <= 1
    
    # Use projected gradient descent for simplicity
    d = -np.linalg.solve(H, g)
    return d

def line_search(h, d, dx, max_alpha=1.0):
    # Armijo backtracking line search
    alpha = max_alpha
    c5_base = evaluate_c5(h, dx)
    c5_prev = c5_base
    eps = 1e-6
    
    # Ensure search direction is feasible
    for i in range(len(h)):
        if d[i] > 0:
            alpha = min(alpha, (1 - h[i]) / d[i])
        else:
            alpha = min(alpha, -h[i] / d[i])
    
    if alpha <= 0:
        return 0.0
    
    alpha = min(1.0, alpha)
    while alpha > eps:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < c5_base:
            return alpha
        alpha *= 0.5
    return 0.0

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    # Approximate Hessian
    H = np.eye(n_points)
    
    max_iter = 1000
    tol_grad = 1e-3
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Check convergence
        if np.linalg.norm(g) < tol_grad:
            break
            
        # Solve quadratic subproblem
        try:
            d = solve_quadratic_subproblem(H, g, n_points)
        except:
            # Fallback to gradient descent if SVD fails
            d = -g
            
        # Line search
        alpha = line_search(h, d, dx)
        
        if alpha == 0:
            # Try a different approach: simple gradient descent
            d = -g
            alpha = min(1.0, 0.1 * (1.0 / np.linalg.norm(g) + 1e-6))
            if alpha == 0:
                alpha = 1e-5
                
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Update Hessian using BFGS
        try:
            y = compute_gradient(h_new, dx) - g
            s = h_new - h
            if np.dot(s, y) > 1e-10:
                H = H + np.outer(y, y) / np.dot(s, y) - np.outer(np.dot(H, s), np.dot(s, H)) / np.dot(s, np.dot(H, s))
        except:
            pass  # Skip Hessian update if error
            
        h = h_new

    n_points = len(h)
