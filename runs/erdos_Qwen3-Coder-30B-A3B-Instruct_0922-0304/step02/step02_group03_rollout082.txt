import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_particles = 30
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, target_sum)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = project_to_feasible(h_init, target_sum)
    
    # Create initial particles
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Add other particles with variation
    for _ in range(n_particles - 1):
        h = np.ones(n_points)
        # Randomly split into two blocks
        split = np.random.randint(1, n_points-1)
        h[split:] = 0
        # Add noise
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    inertia = 0.7
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia_decay = 1e-4
    
    # Track best solutions
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (inertia * velocities[i] +
                         alpha * r1 * (personal_bests[i] - particles[i]) +
                         beta * r2 * (global_best - particles[i]))
            
            # Clip velocity
            vel_update = np.clip(vel_update, -max_velocity, max_velocity)
            velocities[i] = vel_update
            
            # Update position
            new_pos = particles[i] + velocities[i]
            
            # Project to feasible region
            new_pos = project_to_feasible(new_pos, target_sum)
            
            # Evaluate new position
            new_score = evaluate_c5(new_pos, dx)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_bests[i] = new_pos.copy()
                personal_best_scores[i] = new_score
                
                # Update global best if needed
                if new_score < global_best_score:
                    global_best = new_pos.copy()
                    global_best_score = new_score
            
            # Update particle position
            particles[i] = new_pos.copy()
        
        # Update inertia
        inertia = max(0.4, inertia * (1 - inertia_decay))
        
        # Early stopping if we've reached target
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
