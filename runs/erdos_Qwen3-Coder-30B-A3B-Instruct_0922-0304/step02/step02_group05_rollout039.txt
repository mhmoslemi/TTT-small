import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    """Solve a relaxed version of the problem using convex optimization"""
    # Convex relaxation trying to minimize overlap with regularization
    h_var = cp.Variable(n_points)
    h_target = np.array(h)
    
    # Compute correlations for reference
    corr_ref = correlate(h_target, 1 - h_target, mode="full")
    max_k = np.argmax(corr_ref)
    
    # Set up the objective
    h_shifted = np.roll(h_target, max_k)
    obj = corr_ref[max_k] * dx + lambda_reg * cp.sum(cp.entr(h_var) + cp.log(n_points/2) * h_var)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points/2
    ]
    
    # Solve convex problem
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == "optimal":
            return project_feasible(h_var.value, n_points, dx)
    except:
        pass
    return h

def line_search(h, gradient, dx):
    """Perform line search to find best step size"""
    step_sizes = [0.1, 0.5, 1.0]
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for alpha in step_sizes:
        h_candidate = project_feasible(h - alpha * gradient, len(h), dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < best_c5:
            best_h = h_candidate
            best_c5 = candidate_c5
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the best known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iterations = 0
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Compute overlap and identify where maximum occurs
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient estimate
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1.0 - h_shifted)
        
        # Iterate with line search
        h, c5 = line_search(h, gradient, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Periodic convex relaxation to escape local minima
        if iterations % 20 == 0 and iterations > 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx, lambda_reg)
            c5_relaxed = evaluate_erdos_solution(h_relaxed)
            
            if c5_relaxed < best_c5:
                best_c5 = c5_relaxed
                best_h = h_relaxed.copy()
                
            # Adjust regularization
            if c5_relaxed >= evaluate_erdos_solution(h):
                lambda_reg *= 0.5
        
        # Early stopping if target met
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
