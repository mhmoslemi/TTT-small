import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Two-stage approach: coarse then fine
    # Stage 1: Coarse discretization
    n_points_coarse = 35
    dx_coarse = 2.0 / n_points_coarse
    
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
    else:
        h_coarse = np.zeros(n_points_coarse)
        h_coarse[:n_points_coarse//2] = 1.0
        h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)

    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    # Optimization parameters
    max_iter_coarse = 500
    bounds_coarse = [(0, 1) for _ in range(n_points_coarse)]
    
    # Objective for coarse optimization
    def obj_coarse(h):
        h_rescaled = h * (n_points_coarse / 2) / np.sum(h)
        return evaluate_c5(h_rescaled, dx_coarse)
    
    # Constraints for coarse
    def constraint_coarse(h):
        return np.sum(h) - n_points_coarse / 2
    
    cons_coarse = {'type': 'eq', 'fun': constraint_coarse}
    
    for iter_coarse in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        # L-BFGS-B optimization
        result = minimize(
            obj_coarse,
            h_coarse,
            method='L-BFGS-B',
            bounds=bounds_coarse,
            constraints=cons_coarse,
            options={'maxiter': 10, 'ftol': 1e-9, 'gtol': 1e-5}
        )
        
        if result.success:
            h_new_coarse = result.x
            h_new_coarse = np.clip(h_new_coarse, 0, 1)
            h_new_coarse = h_new_coarse * (n_points_coarse / 2) / np.sum(h_new_coarse)
            c5_new_coarse = evaluate_c5(h_new_coarse, dx_coarse)
            
            if c5_new_coarse < best_c5_coarse:
                best_c5_coarse = c5_new_coarse
                best_h_coarse = h_new_coarse.copy()
                
            h_coarse = h_new_coarse
        else:
            break
    
    # Stage 2: Refine
    n_points_fine = 70
    dx_fine = 2.0 / n_points_fine
    
    # Upsample coarse solution
    h_fine = np.interp(np.linspace(0, 2, n_points_fine), 
                       np.linspace(0, 2, n_points_coarse), 
                       best_h_coarse)
    
    # Add small random perturbations 
    perturbation = 0.1 * np.random.random(n_points_fine) - 0.05
    h_fine += perturbation
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points_fine / 2) / np.sum(h_fine)
    
    best_c5_fine = float('inf')
    best_h_fine = h_fine.copy()
    
    max_iter_fine = 1000
    bounds_fine = [(0, 1) for _ in range(n_points_fine)]
    
    def obj_fine(h):
        h_rescaled = h * (n_points_fine / 2) / np.sum(h)
        return evaluate_c5(h_rescaled, dx_fine)
    
    def constraint_fine(h):
        return np.sum(h) - n_points_fine / 2
    
    cons_fine = {'type': 'eq', 'fun': constraint_fine}
    
    for iter_fine in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(
            obj_fine,
            h_fine,
            method='L-BFGS-B',
            bounds=bounds_fine,
            constraints=cons_fine,
            options={'maxiter': 10, 'ftol': 1e-9, 'gtol': 1e-5}
        )
        
        if result.success:
            h_new_fine = result.x
            h_new_fine = np.clip(h_new_fine, 0, 1)
            h_new_fine = h_new_fine * (n_points_fine / 2) / np.sum(h_new_fine)
            c5_new_fine = evaluate_c5(h_new_fine, dx_fine)
            
            if c5_new_fine < best_c5_fine:
                best_c5_fine = c5_new_fine
                best_h_fine = h_new_fine.copy()
                
            h_fine = h_new_fine
        else:
            break
    
    # Return best solution found
    return (best_h_fine.tolist(), best_c5_fine, n_points_fine)
