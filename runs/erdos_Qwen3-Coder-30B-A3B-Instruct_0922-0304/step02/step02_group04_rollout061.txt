import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import GLPK

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation_refinement(seed, budget_s):
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Load initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a rectangular pulse
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add noise
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate relaxation
        h_var = cp.Variable(n_points)
        
        # Fixed sum constraint
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Bound constraints
        constraints.extend([h_var[i] >= 0 for i in range(n_points)])
        constraints.extend([h_var[i] <= 1 for i in range(n_points)])
        
        # Convex relaxation: minimize max overlap
        # We need to add constraints for all possible overlaps
        # The overlap is a convolution with 1-h
        # For a fixed k, overlap_k = sum(h[i] * (1 - h[i+k])) for valid indices
        
        # Simplify by relaxing to bound on max overlap
        # Add a scalar variable C and constrain overlaps to be <= C
        C = cp.Variable()
        constraints.append(C >= 0)
        
        # Do approximate constraint: check several representative k values
        # This is an approximation - maximal constraint would be too slow
        num_k = min(50, n_points)
        for k in range(num_k):
            # Accumulate the overlap term
            overlap = 0
            # We limit indices to avoid boundary issues
            for i in range(n_points - k):
                overlap += h_var[i] * (1 - h_var[i+k])
            constraints.append(overlap <= C)
        
        # Minimize C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=GLPK)
            # Extract the solution
            h_relaxed = h_var.value
            if h_relaxed is not None:
                # Project to feasible region
                h_proj = project_to_feasible(h_relaxed, n_points)
                c5_proj = evaluate_c5(h_proj, dx)
                
                if c5_proj < best_c5:
                    best_c5 = c5_proj
                    best_h = h_proj.copy()
                    
                # Local search refinement
                for _ in range(10):
                    # Gradient-like update using finite differences
                    h_trial = h_proj.copy()
                    eps = 1e-4
                    grad = np.zeros(n_points)
                    base_c5 = evaluate_c5(h_trial, dx)
                    for j in range(n_points):
                        h_plus = h_trial.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        c5_plus = evaluate_c5(h_plus, dx)
                        grad[j] = (c5_plus - base_c5) / eps
                        
                    # Descend
                    step_size = 0.01
                    h_trial = h_trial - step_size * grad
                    h_trial = project_to_feasible(h_trial, n_points)
                    c5_trial = evaluate_c5(h_trial, dx)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        h_proj = h_trial.copy()
                        
        except Exception as e:
            # If optimization fails, do local search
            pass
            
        # Occasionally do full gradient-based local search
        if i % 5 == 0:
            # Simple steepest descent
            h_old = best_h.copy()
            step_size = 0.01
            for i_iter in range(20):
                base_c5 = evaluate_c5(h_old, dx)
                grad = np.zeros(n_points)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h_old.copy()
                    h_plus[j] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    grad[j] = (c5_plus - base_c5) / eps
                    
                h_new = h_old - step_size * grad
                h_new = project_to_feasible(h_new, n_points)
                
                # Check improvement
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_old = h_new.copy()
                else:
                    break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = convex_relaxation_refinement(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
