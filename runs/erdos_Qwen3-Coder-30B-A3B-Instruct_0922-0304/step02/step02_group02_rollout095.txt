import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 5000
    perturb_frac = 0.1
    
    # Start the annealing process
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current evaluation
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Create candidate solution via perturbation
        h_prime = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * perturb_frac), replace=False)
        delta = 0.1
        h_prime[indices] += delta
        h_prime[indices] = np.clip(h_prime[indices], 0, 1)
        
        # Adjust other values to maintain sum constraint
        total_change = np.sum(h_prime) - (n_points / 2)
        if abs(total_change) > 1e-8:
            # Distribute the difference to non-perturbed entries
            non_indices = np.setdiff1d(np.arange(n_points), indices)
            if len(non_indices) > 0:
                # Reduce each by a proportional amount 
                reduction_per = total_change / len(non_indices)
                h_prime[non_indices] -= reduction_per
                # Clip and redistribute to handle boundary issues
                h_prime = np.clip(h_prime, 0, 1)
                # Re-normalize to sum constraint
                h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(h_prime, dx)
        
        # Annealing decision
        delta_E = candidate_c5 - current_c5
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.rand():
            h = h_prime
            
        # Update temperature
        T = max(T_min, T * alpha)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
