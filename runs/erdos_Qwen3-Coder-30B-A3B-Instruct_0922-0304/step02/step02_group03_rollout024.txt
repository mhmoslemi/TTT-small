import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with coarse discretization for faster exploration
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = enforce_feasibility(h, n_points)
    else:
        # Fallback: simple block pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = enforce_feasibility(h, n_points)
    
    # Optimization with bounds and constraint
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Use L-BFGS-B with bounds and equality constraint
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=bounds, 
                   constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-6})
    
    # Refine at fine resolution
    if time.time() - start_time < budget_s - 5:
        n_points = 70
        dx = 2.0 / n_points
        h = enforce_feasibility(res.x, n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint}
        
        res = minimize(objective, h, method='L-BFGS-B', 
                       bounds=bounds, 
                       constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-6})
    
    best_h = enforce_feasibility(res.x, n_points)
    c5 = evaluate_overlap(best_h, dx)
    
    # Check if target C5 is achieved
    if c5 <= 0.3807:
        return (best_h.tolist(), c5, n_points)
    
    # Try local refinement if not yet improved
    if time.time() - start_time < budget_s - 5:
        # Try some local perturbations
        best_c5 = c5
        best_h_final = best_h.copy()
        
        for _ in range(10):
            if time.time() - start_time > budget_s - 2:
                break
            h_perturbed = best_h.copy() + np.random.normal(0, 0.01, n_points)
            h_perturbed = enforce_feasibility(h_perturbed, n_points)
            c5_perturbed = evaluate_overlap(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h_final = h_perturbed
        
        best_h = best_h_final
        c5 = best_c5
    
    return (best_h.tolist(), c5, n_points)
