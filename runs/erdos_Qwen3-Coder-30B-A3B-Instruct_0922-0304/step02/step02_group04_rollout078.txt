import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm
    swarm_size = 30
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Default starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random initial particles
    while len(population) < swarm_size:
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # PSO parameters
    w = 0.7
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize velocities and personal bests
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    personal_best = population.copy()
    personal_best_c5 = [evaluate_c5(h, dx) for h in population]
    
    # Global best tracking
    best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[best_idx].copy()
    global_best_c5 = personal_best_c5[best_idx]
    
    # Main PSO loop
    iterations = 500
    local_refinement_freq = 20
    
    for it in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            h = population[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = random.random(), random.random()
            v = w * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            h_new = project_to_feasible(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_c5(h_new, dx)
            
            # Update personal best
            if c5_new < personal_best_c5[i]:
                personal_best[i] = h_new.copy()
                personal_best_c5[i] = c5_new
                
                # Update global best
                if c5_new < global_best_c5:
                    global_best = h_new.copy()
                    global_best_c5 = c5_new
            
            # Update population
            population[i] = h_new
            velocities[i] = v
            
        # Periodic local refinement
        if it % local_refinement_freq == 0 and it > 0:
            # Refine top 10% particles
            top_indices = np.argsort(personal_best_c5)[:swarm_size//10]
            for idx in top_indices:
                h = personal_best[idx]
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    corr = correlate(h, 1 - h, mode="full")
                    c5_val = np.max(corr) * dx
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - c5_val) / eps
                    h = h - step_size * grad
                    h = project_to_feasible(h, n_points)
                # Update bests
                c5_refined = evaluate_c5(h, dx)
                if c5_refined < personal_best_c5[idx]:
                    personal_best[idx] = h
                    personal_best_c5[idx] = c5_refined
                    if c5_refined < global_best_c5:
                        global_best = h
                        global_best_c5 = c5_refined
        
        # Early exit
        if global_best_c5 <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_c5, n_points)
