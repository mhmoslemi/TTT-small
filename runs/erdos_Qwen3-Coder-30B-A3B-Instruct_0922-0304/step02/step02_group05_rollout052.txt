import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, n_points, dx):
    # Gradient for shift k_max: (1-h)_{shifted} - h_{shifted}
    # Since we use mode='full', we match the shift at the middle
    # k_max index is centered at n_points - 1
    # But we need to compute where the correlation actually shifts
    # Let's compute overlap with shifts explicitly
    # For shift k, overlap_k = sum_i h[i] (1-h[i+k]) = sum_i h[i] - sum_i h[i] h[i+k]
    # gradient w.r.t h[i] = 1 - (sum over k: h[i+k] if k corresponds to shift, 
    #                             0 otherwise) = 1 - h[i+k_max] if i+k_max < n_points, 
    #                             and 0 otherwise (approximate)
    # Actually, better to think as in signal processing properly:
    # overlap_k = sum_{i} h[i] * (1-h[i+k]) = sum_i h[i] - sum_i h[i] * h[i+k]
    # For k-th shift, this gradient of overlap_k w.r.t. h[j] is:
    # if i=j, it's 1 
    # if i=j+k, it's -h[j] 
    # otherwise, 0  
    # So ∇_j overlap_k = 1 - h[j+k] if j+k < n_points, 1 if j==0 and k=n_points-1, etc
    grad = np.zeros_like(h)
    # For the maximum overlap term (k_max shift), compute the full gradient
    h_shifted = np.zeros(n_points)
    # h_shifted[i] = h[i-k_max] if i >= k_max else 0
    # But since we don't want to go backwards in time, let's think:
    # For correlation h * (1-h) at k, overlap_k = sum_{i=0}^{n_points-1} h[i] * (1-h[i+k])
    # Label shift k's index as (n_points-1+k) in full mode (see correlate)
    # So k=0: index n_points-1
    # k=1: index n_points 
    # k=n_points-1: index 2*n_points - 2  
    # k=0 should be indexed as n_points-1 in full correlating
    # to get k_max, we use the index (n_points-1+k_max)
    # but we really want to compute for index of max overlap. 
    # If that's at position z (from left), and the relative k is z-n_points+1
    # though the boundary treatment is tricky. Better use a safe approach:
    # Let's simulate that accumulation on the gradient.
    # 
    # We're aiming at the gradient of max_k overlap_k
    # For overlap_k = sum_i h[i] * (1 - h[i+k])
    # ∂overlap_k / ∂h[j] = - h[j+k] if j+k < n_points 
    #                +1  if j==i
    # but if we consider full convolution, matrices are in different shapes...
    # But we are computing overlap for all shifts.
    # So let's look at a direct way: 
    # If we fixed k = k_max, and we want ∇h overlap_k:
    # expand: overlap_k = sum_h[i] - sum_h[i] * h[i+k], 
    # So d/dh[j] overlap_k = 1 if j==i, -h[i+k] if j=i+k, 0 otherwise.
    # This is a sum of deltas and a product. Makes it more accurate.
    # We can simply: 
    # Direct gradient for overlap_k w.r.t h[j]:
    # if j = i then add 1  (his sum is from h[i])
    # if j = (i+k) then add -h[i] (h[i] * h[i+k])
    # But again full arrangement requires careful toggle. 
    # Better is to compute full convolution gradient as a matrix.
    # Let's not compute full but for the specific max_k:
    # We are to compute the gradient of maximum amount of overlap.
    # Since we use convolution via correlate with full mode, the result index is from k=0 to k=2*n_points-2
    # With the result at index k + n_points-1, becomes shift k (i.e. k shifts from 0 to n_points-1)
    # This is a minor confusion in indexing.
    # Let's just compute it using naively defined finite diff for a general estimate.
    # Actually simpler just do one instantiation per k or keep the full link logic 
    # Wedo: just a derivative approximation
    # We look at overlap outputs, but if we look at the index as place of max, 
    # we can compute corresponding derivative. This works for approximation.
    # Gradient for each h[i] wrt overlap for index `k的选择`
    # We build an intermediate matrix and get it like: 
    # Let k = k_max => overlap_k_start at position (k_max + n_points - 1)  
    # The actual position of shift k is at index k + n_points - 1
    # Start over indexing
    # Full overlap is 2*n_points - 1 in length 
    # The correlation h * (1-h) has size 2*n_points - 1
    # correlation at position z = sum[i=0](min(n_points-1, z - i))   h[i] * (1 - h[z-i])
    # where i from 0 to min(n_points-1, z)
    # To find gradient of overlap for the position (k_max + n_points - 1)
    # That's the exact slicing: 
    # overlap_k[(k_max + n_points - 1)] = 
    # sum_{i=0}^{min(n_points-1, 2*n_points - 2 - (k_max + n_points - 1))} h[i] * (1 - h[i + k_max])
    # Indices in contribution:
    # pairs of h[i] with h[i+k_max] 
    # Add 1 to i-th  and 
    # subtract h[i] from (i+k_max)-th positions 
    # i = 0 to n_points-1 (but also enforced such that i+k_max < n_points)
    # So i from 0 to min(n_points-1, n_points-1-k_max)= min(n_points-1, n_points-1-k_max)
    n_pos = 2*n_points - 1 
    # If overlap_k starts from the middle (i.e. k=0 offset is at position n-1) 
    # and overlap_k[i] = the overlap estimator for shift (i + 1 - n), if i >= n-1
    # For a valid k = k_max, that corresponds to index = k_max + n_points - 1 in full array
    
    if k_max < n_points:
        # From i=0 to i = n_points-1-k_max, so i <= n_points-1-k_max  
        for i in range(n_points - k_max):
            # partial overlap_k w.r.t h[i] = 1
            # partial overlap_k w.r.t h[i + k_max] = -h[i]
            # These would in the gradient. But only one shift at a time.
            # Build partial derivative vector, for one shift.
            grad[i] += 1
            grad[i + k_max] -= h[i]
    
    return grad

def line_search(h, grad, alpha_max, dx, n_points):
    # Try several alpha in sequence to find best decrease
    # Binary search is more indirect due to its nonlinearities
    best_alpha = 0.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_sizes = np.logspace(-4, 0, 10) 
    for alpha in step_sizes:
        if alpha > alpha_max:
            continue
        h_trial = h - alpha * grad
        h_trial = project_feasible(h_trial, n_points, dx)
        c5 = evaluate_c5(h_trial, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_trial.copy()
            best_alpha = alpha
    return best_h, best_alpha, best_c5

def optimize_step(h, k_max, n_points, dx, alpha=0.01):
    grad = compute_gradient(h, k_max, n_points, dx)
    
    # find best alpha 
    h_new, alpha_used, c5_new = line_search(h, grad, alpha, dx, n_points)
    
    # Clip and scale if needed
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    return h_new, c5_new

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    # Initialize from parent solution if exists
    if 'initial_h_values' in globals():

        # Default if no parent construction
        h = np.full(n_points, 0.5)
    
    # Evaluate initial solution
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    start_time = time.time()
    step_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        step_count += 1

        # Recompute overlaps
        h_copy = h.copy()
        overlap = correlate(h_copy, 1 - h_copy, mode="full")
        k_max = np.argmax(overlap)
        # k actually corresponds to:
        # overlap index k+2-1 = k+1 in case k starts at 0, or if it uses centroid-X, Y  
        # But the shift index z = k + n-1  
        # So shift from 0 to n-1 = [0, ..., n-1]
        # Position indices from 0 to 2n-2, so shift k = z - (n-1) for z up to 2n-2
        # k_max = (k_max) - (n_points - 1)
        shift_max = k_max - (n_points - 1)
        shift_max = max(0, shift_max)
        
        # Optimize step
        h, c5_new = optimize_step(h, shift_max, n_points, dx, alpha=0.01)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Periodically perform relaxation
        if step_count % 50 == 0:
            # Add entropy regularization to escape local minima
            # Simple L2 is added as penalty to help convergence
            # But we won't change the identity of h, just add weight to soft constraint
            # Gradient would be to add more stretching to biodiversity, i.e., impulse-aware 
            # Let's recompute relaxation with different starting point
            # Now use a second level desperation approach:
            # We soften the hard constraints and introduce small variations
            # E.g., do annealed local search or scatter and replicate from current best
            
            # Small random perturbations to soften local minima
            for i in range(n_points):
                h[i] = max(0, min(1, h[i] + np.random.normal(0, 0.01)))
            
            h = project_feasible(h, n_points, dx)
            c5_trial = evaluate_c5(h, dx)
            if c5_trial < best_c5:

            if np.random.random() > 0.7:  # Occasionally random change
                h = np.random.rand(n_points)

                h = np.clip(h, 0, 1)
            
            # Periodic reset / restart prediction approach

        # Early exit if target reached
