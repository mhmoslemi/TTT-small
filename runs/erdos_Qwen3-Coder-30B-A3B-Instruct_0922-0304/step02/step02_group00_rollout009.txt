import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as starting point for some particles
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        for i in range(n_particles // 2):
            particles.append(h_init.copy())
            velocities.append(np.zeros(n_points))
        # Initialize remaining particles randomly
        for i in range(n_particles // 2, n_particles):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.rand(n_points) - 0.5)
    else:
        # Initialize all particles randomly
        for i in range(n_particles):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.rand(n_points) - 0.5)
    
    # PSO parameters
    omega = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    # Track best solutions
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_score
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = omega * v + c1 * r1 * (p - h) + c2 * r2 * (gbest - h)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.01, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            
            # Evaluate new solution
            current_c5 = evaluate_c5(h_new, dx)
            
            # Update personal best
            if current_c5 < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_score:
                    gbest = h_new.copy()
                    gbest_score = current_c5
                    
                    if gbest_score <= 0.3807:
                        best_h = gbest.copy()
                        best_c5 = gbest_score
                        return (best_h.tolist(), best_c5, n_points)
        
        # Update global best if needed
        current_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[current_gbest_idx] < gbest_score:
            gbest = pbest[current_gbest_idx].copy()
            gbest_score = pbest_scores[current_gbest_idx]
    
    # Final evaluation
    best_h = gbest.copy()
    best_c5 = gbest_score
    
    # Ensure final constraint satisfaction
    assert abs(np.sum(best_h) - n_points/2) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= 0) & (best_h <= 1)), "Bounds constraint violated"
    
    return (best_h.tolist(), best_c5, n_points)
