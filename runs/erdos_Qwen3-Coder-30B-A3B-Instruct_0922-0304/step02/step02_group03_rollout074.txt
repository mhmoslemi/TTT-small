import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    dx = 2.0 / n_points
    h = enforce_feasibility(h, n_points)
    
    def objective(h):
        return evaluate_overlap(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimize using L-BFGS-B
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=[(0, 1) for _ in range(n_points)], 
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6})
    
    best_h = res.x
    best_h = enforce_feasibility(best_h, n_points)
    c5 = evaluate_overlap(best_h, dx)
    
    # If we're still above target, try a fine-tuning approach
    if c5 > 0.3807 and time.time() - start_time < budget_s - 2:
        # Try a more careful fine-tuning
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Slight perturbation with gradient estimation
            grad = np.zeros(n_points)
            eps = 1e-5
            base_val = objective(best_h)
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = enforce_feasibility(h_plus, n_points)
                grad[i] = (objective(h_plus) - base_val) / eps
            
            # Update with small step
            step_size = 0.001
            new_h = best_h - step_size * grad
            new_h = enforce_feasibility(new_h, n_points)
            new_val = evaluate_overlap(new_h, dx)
            
            if new_val < c5:
                best_h = new_h
                c5 = new_val
    
    return (best_h.tolist(), c5, n_points)
