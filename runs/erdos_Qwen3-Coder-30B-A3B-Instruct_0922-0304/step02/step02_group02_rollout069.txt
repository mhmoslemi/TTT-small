import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # equivalent to integral constraint
    
    # Initialize using initial_h_values if available, otherwise create symmetric pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create symmetric pattern that tends to perform well
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = h * (target_sum / np.sum(h))
    
    h = repair_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    lr = 0.01
    
    # Particle swarm initialization
    particles = np.array([h.copy() for _ in range(n_particles)])
    velocities = np.zeros_like(particles)
    
    # Repair all particles to satisfy constraints
    for i in range(n_particles):
        particles[i] = repair_h(particles[i], target_sum)
    
    # Track personal bests
    personal_best_h = particles.copy()
    personal_best_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best_c5 = personal_best_c5[global_best_idx]
    global_best_h = personal_best_h[global_best_idx].copy()
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best_h[i] - particles[i]) +
                             c2 * r2 * (global_best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = repair_h(particles[i], target_sum)
            
            # Evaluate new position
            new_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_c5 < personal_best_c5[i]:
                personal_best_c5[i] = new_c5
                personal_best_h[i] = particles[i].copy()
                
                # Update global best if needed
                if new_c5 < global_best_c5:
                    global_best_c5 = new_c5
                    global_best_h = particles[i].copy()
        
        # Early exit
        if global_best_c5 <= 0.3807:
            break
    
    # Perform gradient refinement on the best solution
    h_refined = global_best_h.copy()
    best_c5_refined = global_best_c5
    
    # Refinement using SGD with small steps
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        eps = 1e-4
        grad = np.zeros_like(h_refined)
        
        base_c5 = evaluate_c5(h_refined, dx)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, target_sum)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[j] = (plus_c5 - base_c5) / eps
            
        h_refined = h_refined - lr * grad
        h_refined = repair_h(h_refined, target_sum)
        
        new_c5 = evaluate_c5(h_refined, dx)
        if new_c5 < best_c5_refined:
            best_c5_refined = new_c5
            global_best_h = h_refined.copy()
    
    # Final validation
    final_c5 = evaluate_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
