import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(corr)
        max_overlap_val = corr[max_overlap_idx] * dx
        
        # Compute gradient explicitly for the maximum overlap shift
        grad = np.zeros_like(h)
        # Maximum overlap occurs at shift k = max_overlap_idx - (n_points-1)
        k = max_overlap_idx - (n_points - 1)
        
        # Compute gradient for the current worst overlap
        for i in range(n_points):
            if 0 <= i + k < n_points and i + k >= 0:
                grad[i] += (1 - h[i + k])
            if 0 <= i - k < n_points and i - k >= 0:
                grad[i] += h[i - k]
        
        # Update h using gradient descent with adaptive step size
        step = 0.005
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
