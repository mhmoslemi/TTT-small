import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization approach
    # We solve a relaxed version with CVXPY
    
    # We create a convex relaxation by using bilinear constraints
    # Given: overlap_k = sum_i h_i * (1 - h_{i+|k|}) 
    # Relax to: h_i * (1 - h_{i+|k|}) <= t_{k,i}
    # This creates a system of linear constraints without loss of generality
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try convex optimization approach for improvement
        try:
            # Define variables for the optimization
            h_vars = cp.Variable(n_points)
            
            # Objective: minimize maximum overlap
            t = cp.Variable()
            
            # Constraints
            constraints = []
            # h in [0,1] 
            constraints.append(h_vars >= 0)
            constraints.append(h_vars <= 1)
            
            # sum h = 1
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Ensure max overlap is within t
            for k in range(1, n_points):
                # Compute overlap sum for lag k
                # Only for valid indices
                overlap_sum = 0
                for i in range(n_points - k):
                    overlap_sum += h_vars[i] * (1 - h_vars[i + k])
                constraints.append(overlap_sum <= t)
            
            # Solve
            objective = cp.Minimize(t)
            prob = cp.Problem(objective, constraints)
            
            prob.solve(solver=cp.GLOP, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
        except:
            pass
        
        # Local improvement using gradient descent (fallback)
        try:
            step = 0.01
            iterations = 100
            h_temp = h.copy()
            
            for _ in range(iterations):
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Update h
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
        except:
            pass
            
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
