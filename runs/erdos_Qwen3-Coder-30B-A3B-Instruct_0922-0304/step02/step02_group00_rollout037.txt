import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Better starting point: linear ramp with some noise
        h = np.linspace(0.05, 0.95, n_points)
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Annealing parameters
    temperature = 1.0
    cooling_rate = 0.995
    epsilon = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % 100 == 0 and iteration > 0:
            temperature *= cooling_rate
            if temperature < 1e-6:
                temperature = 1e-6
                
        # Generate candidate by perturbing current h
        h_candidate = h.copy() + np.random.normal(0, epsilon * temperature, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)

        # Evaluate both solutions
        current_c5 = evaluate_c5(h, dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)

        # Accept or reject the candidate
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            h = h_candidate
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = iteration

        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Reset temperature if no improvement in a long time
        if iteration - last_improvement > 1000:
            temperature = 1.0
            last_improvement = iteration
            
        iteration += 1

    return (best_h.tolist(), best_c5, n_points)
