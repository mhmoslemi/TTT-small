import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def initialize_particles(n_points, initial_h_values=None):
    particles = []
    velocities = []
    
    # If initial_h_values exists, use it as a particle
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    else:
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
        h += np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points, 2.0/n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Generate remaining particles with random values
    for i in range(49):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, 2.0/n_points)
        particles.append(h.copy())
        velocities.append(np.random.normal(0, 0.01, n_points))
    
    return particles, velocities

def particle_swarm_optimization(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles and velocities
    particles, velocities = initialize_particles(n_points, 
                                                initial_h_values globals().get('initial_h_values', None))
    
    # PSO parameters
    num_particles = len(particles)
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    omega = 1.0  # Inertia weight
    max_velocity = 0.2
    
    # Initialize personal best positions and fitness
    p_best = [p.copy() for p in particles]
    p_best_fitness = [evaluate_c5(p, dx) for p in particles]
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]
    
    # Best overall solution tracking
    best_h = g_best.copy()
    best_c5 = g_best_fitness
    
    iteration = 0
    stagnation_counter = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity to prevent overflow
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            current_fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best if improved
            if current_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = current_fitness
                
                # Update global best if improved
                if current_fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = current_fitness
                    stagnation_counter = 0
                    last_improvement = iteration
                    if current_fitness < best_c5:
                        best_h = particles[i].copy()
                        best_c5 = current_fitness
                        
                    # Early exit if target achieved
                    if best_c5 <= 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
        
        # Adaptive inertia
        if iteration % 50 == 0:
            omega = max(0.8, omega - 0.02)
            
        # Random perturbation for stagnation
        if iteration - last_improvement > 500:
            for i in range(num_particles):
                if np.random.rand() < 0.1:  # 10% chance per particle
                    particles[i] = project_feasible(
                        particles[i] + np.random.normal(0, 0.05, n_points), 
                        n_points, dx)
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = evaluate_c5(particles[i], dx)
            
            # Reset stagnation counter
            last_improvement = iteration
            stagnation_counter = 0
            
        stagnation_counter += 1
        
        if time.time() - start_time > budget_s - 2:
            break

    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return particle_swarm_optimization(seed, budget_s)
