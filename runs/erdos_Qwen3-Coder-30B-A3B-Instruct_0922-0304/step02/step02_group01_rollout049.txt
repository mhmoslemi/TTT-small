import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Constants for optimization
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    
    # For gradient computation
    v = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        shift = max_idx - (n_points - 1)
        
        # Compute gradient: this is based on the gradient of overlap with respect to h
        # For shift k, the overlap is sum_{i} h[i] * (1 - h[i+k])
        # So gradient w.r.t. h[i] = (1 - h[i+k]) if i+k < n_points, and 0 otherwise
        # But we're not using the full derivative per shift
        # Instead we compute the derivative of maximum overlap
            
        # Compute approximate gradient for the maximum overlap shift
        grad = np.zeros(n_points)
        if shift >= 0:
            # h[i] contributes to overlap with h[i+shift]
            for i in range(n_points - shift):
                grad[i] += (1 - h[i+shift])
        else:
            # h[i-shift] contributes to overlap with h[i]
            for i in range(-shift, n_points):
                grad[i] -= h[i+shift]
                
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Update with momentum
        v = momentum * v - step_size * grad
        h = h + v
        
        # Projection
        h = project_h(h, target_sum)
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
