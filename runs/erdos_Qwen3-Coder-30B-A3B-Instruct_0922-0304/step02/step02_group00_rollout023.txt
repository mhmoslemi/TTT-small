from scipy.signal import correlate
import time

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Construct an initial guess with a wave-like pattern
        h = np.linspace(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Use gradient descent with momentum to optimize the problematic shift
        step_size = 0.02
        momentum = 0.9
        velocity = np.zeros_like(h)
        
        # Simple local search
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current overlap
            corr_current = correlate(h, 1 - h, mode="full")
            c5_current = np.max(corr_current) * dx
            
            # Compute finite difference gradient
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update with momentum
            velocity = momentum * velocity - step_size * grad
            h = h + velocity
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
