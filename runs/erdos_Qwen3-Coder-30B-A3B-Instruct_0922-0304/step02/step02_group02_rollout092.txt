import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Ensure h is in [0,1] and sum equals target_sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) = 35
    
    # PSO parameters
    n_particles = 50
    n_iter = 500
    inertia = 0.7
    c1 = c2 = 2.0
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * target_sum / np.sum(base_h)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = base_h * target_sum / np.sum(base_h)
        
    for _ in range(n_particles):
        # Initialize particle with base + noise
        h = base_h + np.random.normal(0, 0.05 * np.max(base_h), n_points)
        h = repair_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Track best solutions
    best_global = float('inf')
    best_global_h = None
    particle_best = [float('inf')] * n_particles
    particle_best_h = [None] * n_particles
    
    # PSO iterations
    for iteration in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            c5 = evaluate_c5(h, dx)
            
            if c5 < particle_best[i]:
                particle_best[i] = c5
                particle_best_h[i] = h.copy()
                
            if c5 < best_global:
                best_global = c5
                best_global_h = h.copy()
                
        # Early exit if target achieved
        if best_global <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            if particle_best_h[i] is not None:
                p_best = particle_best_h[i]
            else:
                p_best = h
                
            if best_global_h is not None:
                g_best = best_global_h
            else:
                g_best = h
                
            v = inertia * v + c1 * r1 * (p_best - h) + c2 * r2 * (g_best - h)
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h_new = h + v
            h_new = repair_h(h_new, target_sum)
            
            particles[i] = h_new
            velocities[i] = v
            
        # Local gradient descent on top particles (20%)
        top_k = max(1, n_particles // 5)
        indices = np.argsort(particle_best)[:top_k]
        
        for idx in indices:
            h = particles[idx]
            # Compute gradient with finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, target_sum)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
            
            # Apply gradient descent
            h_new = h - 0.01 * grad
            h_new = repair_h(h_new, target_sum)
            particles[idx] = h_new
    
    # Final evaluation
    if best_global_h is not None:
        final_c5 = evaluate_c5(best_global_h, dx)
        return (best_global_h.tolist(), final_c5, n_points)
    else:
        # Fallback to base construction
        base_c5 = evaluate_c5(base_h, dx)
        return (base_h.tolist(), base_c5, n_points)
