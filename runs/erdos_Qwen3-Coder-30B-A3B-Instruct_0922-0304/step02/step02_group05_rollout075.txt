import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_shift, n_points, dx):
    # Compute gradient of overlap_k with respect to h
    # For shift k, overlap_k = sum_i h[i] * (1 - h[i+k])
    # Gradient is (1 - h[i+k]) - h[i+k] = 1 - 2*h[i+k] 
    # But we must consider that overlap_k is sum_h[i] * (1 - h[i+k])
    # So the derivative wrt h[i] is (1 - h[i+k]) 
    # This is true for all h[i] contributing to overlap_k
    grad = np.zeros_like(h)
    for i in range(n_points):
        idx = (i + k_shift) % n_points
        if idx < n_points:
            grad[i] = 1 - h[idx]
    return grad * dx

def line_search(h, grad, n_points, dx, max_alpha=1.0):
    # Simple line search to find good alpha
    best_alpha = 0.0
    best_c5 = evaluate_c5(h, dx)
    alpha = max_alpha
    for _ in range(10):
        new_h = h - alpha * grad
        new_h = project_feasible(new_h, n_points, dx)
        new_c5 = evaluate_c5(new_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_alpha = alpha
        alpha *= 0.5
    return best_alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent solution or create another
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create random feasible solution
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Compute gradient of max overlap
        grad = compute_gradient(h, max_k - n_points + 1, n_points, dx)
        
        # Do line search to pick good step size
        alpha = line_search(h, grad, n_points, dx)
        
        # Update h
        h = h - alpha * grad
        h = project_feasible(h, n_points, dx)
        
        # Evaluate new c5
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Check if we've hit target
        if best_c5 <= 0.3807:
            break
            
        step += 1
        
        # Occasionally add regularization to escape local minima
        if step % 50 == 0:
            # Apply smoothing/mild entropy to escape local minima
            h = 0.95 * h + 0.05 * np.random.rand(n_points)
            h = project_feasible(h, n_points, dx)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
