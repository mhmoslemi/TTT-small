import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    epsilon = 0.05
    T_initial = 1.0
    beta = 0.995
    T = T_initial
    best_c5 = float('inf')
    best_h = None

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Track temperature resets
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Generate candidate by perturbing with Gaussian noise scaled by T
        h_candidate = h + epsilon * T * np.random.normal(0, 1, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_candidate, dx)
        delta_c5 = new_c5 - current_c5
        
        # Accept with probability based on simulated annealing
        if new_c5 < current_c5 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Temperature cooling
        if iteration % 100 == 0:
            T = beta * T
            
        # Reset temperature if no improvement for a long time
        if iteration - last_improvement > 1000:
            T = T_initial
            last_improvement = iteration
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    if best_h is None:
        best_h = h.copy()
