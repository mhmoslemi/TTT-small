import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    """Scale h to have sum equal to target_sum using binary search."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for the scaling factor
    low, high = 0.0, 2.0
    tolerance = 1e-10
    while high - low > tolerance:
        mid = (low + high) / 2
        scaled = h * mid
        scaled_sum = np.sum(scaled)
        if scaled_sum > target_sum:
            high = mid
        else:
            low = mid
    return h * (low + high) / 2

def crossover(parent1, parent2):
    """Single-point crossover between two parents."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Gaussian mutation with clipping and rescaling."""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, sigma)
    mutated = np.clip(mutated, 0, 1)
    return mutated

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Initialize with the provided solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h.copy())
    
    # Initialize remaining individuals randomly
    for _ in range(population_size - len(population)):
        h = np.random.uniform(0, 1, n_points)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    # Evolution loop
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    generations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        generations += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort and select top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        selected = [population[i] for i in top_indices]
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep top 10%
        elite_size = population_size // 10
        for i in range(min(elite_size, len(selected))):
            new_population.append(selected[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select two parents
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Scale to target sum
            child1 = scale_to_target_sum(child1, target_sum)
            child2 = scale_to_target_sum(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Check for stagnation
        if generations % 10 == 0:
            if best_c5 < 0.38085:
                no_improvement_count += 1
                if no_improvement_count >= max_no_improvement:
                    break
            else:
                no_improvement_count = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    return (h.tolist(), c5_bound, 70)
