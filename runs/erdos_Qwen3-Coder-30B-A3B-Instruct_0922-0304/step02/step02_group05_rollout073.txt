import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # Clip to [0,1] and rescale to sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        scaling = (n_points / 2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.normal(0, 0.02, n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.normal(0, 0.02, n_points))
    
    # Fill remaining particles
    while len(particles) < n_particles:
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
        velocities.append(np.random.normal(0, 0.02, n_points))
        
    # Initialize best positions
    p_best = [p.copy() for p in particles]
    p_best_c5 = [evaluate_c5(p, dx) for p in particles]
    
    g_best_idx = np.argmin(p_best_c5)
    g_best = particles[g_best_idx].copy()
    g_best_c5 = p_best_c5[g_best_idx]
    
    # PSO parameters
    w = 1.0
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    
    best_h = g_best.copy()
    best_c5 = g_best_c5
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0 and iteration > 0:
            w = max(0.8, w - 0.02)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) +
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible set
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = c5
                
                # Update global best
                if c5 < g_best_c5:
                    g_best = particles[i].copy()
                    g_best_c5 = c5
                    last_improvement = iteration
                    
                    if g_best_c5 < best_c5:
                        best_c5 = g_best_c5
                        best_h = g_best.copy()
            
            # Reset if no improvement in a while
            if iteration - last_improvement > 100:
                particles[i] = np.random.uniform(0, 1, n_points)
                particles[i] = project_feasible(particles[i], n_points, dx)
                velocities[i] = np.random.normal(0, 0.02, n_points)
                p_best[i] = particles[i].copy()
                p_best_c5[i] = evaluate_c5(particles[i], dx)
                
                if p_best_c5[i] < g_best_c5:
                    g_best = p_best[i].copy()
                    g_best_c5 = p_best_c5[i]
                    last_improvement = iteration
        
        iteration += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
