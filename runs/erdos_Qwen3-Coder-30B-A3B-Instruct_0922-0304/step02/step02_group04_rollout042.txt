import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    F = 0.7
    CR = 0.9
    gradient_freq = 50
    max_iter_grad = 50
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = population[:population_size//2].copy()
        
        # DE mutation and crossover
        for i in range(population_size//2, population_size):
            # Select three distinct random individuals
            a, b, c = random.sample(range(population_size//2), 3)
            u = population[a]
            v = population[b]
            w = population[c]
            
            # Mutation
            mutant = u + F * (v - w)
            mutant = project_to_feasible(mutant, n_points)
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, population[i])
            
            # Selection
            trial_c5 = evaluate_c5(trial, dx)
            parent_c5 = fitness[i]
            
            if trial_c5 < parent_c5:
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = new_population
        
        # Periodic gradient refinement
        if gen % gradient_freq == 0 and gen > 0:
            # Refine best individual with gradient descent
            h_refined = best_solution.copy()
            step_size = 0.01
            eps = 1e-4
            
            for _ in range(max_iter_grad):
                if time.time() - start_time > budget_s - 1:
                    break
        
                # Compute current value
                current_c5 = evaluate_c5(h_refined, dx)
                
                # Finite difference gradient
                grad = np.zeros_like(h_refined)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Gradient step
                h_refined = h_refined - step_size * grad
                h_refined = project_to_feasible(h_refined, n_points)
                
                # Check if better
                new_c5 = evaluate_c5(h_refined, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_solution = h_refined.copy()
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = differential_evolution(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
