import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a more direct convex approach
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try convex optimization approach with CVXPY
        # Create variables
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp_sum(h_vars) == n_points / 2, t >= 0]
        
        # Convex relaxation of overlap constraints
        # For each lag k, we compute the overlap
        for k in range(-n_points+1, n_points):
            if k < 0:
                # Positive lag: h[i] * (1 - h[i-k]) for i = 0,...,n_points+k-1
                lag_expr = 0
                for i in range(n_points + k):
                    lag_expr += h_vars[i] * (1 - h_vars[i - k])
            elif k == 0:
                # Zero lag (invalid, skip)
                continue
            else:
                # Negative lag: h[i] * (1 - h[i+k]) for i = 0,...,n_points-k-1
                lag_expr = 0
                for i in range(n_points - k):
                    lag_expr += h_vars[i] * (1 - h_vars[i + k])
            
            # Convex relaxation of this product sum
            constraints.append(lag_expr <= t)
        
        # Problem definition
        prob = Problem(Minimize(t), constraints)
        
        try:
            # Solve with GLOP or PDLP
            prob.solve(solver=cp.GLPK, verbose=False)
            new_h = h_vars.value
            if new_h is not None:
                h = np.clip(new_h, 0, 1)
                # Re-normalize
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient descent if convex solve fails
            pass

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
