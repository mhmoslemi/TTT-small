import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-5):
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h, dx)
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_to_feasible(h_plus, n_points)
        h_minus = project_to_feasible(h_minus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def line_search(h, d, dx, max_alpha=1.0):
    alpha = max_alpha
    current_c5 = evaluate_c5(h, dx)
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            return alpha, h_new
        alpha *= 0.5
    return alpha, h

def solve_quadratic_subproblem(H, g, n_points, bounds=None):
    # Simple projected gradient step for quadratic subproblem
    d = -np.dot(np.linalg.inv(H), g)
    return d

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    # Initialize Hessian (identity matrix)
    H = np.eye(n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # SQP iterations
    max_iterations = 1000
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Check convergence
        if np.linalg.norm(g) < 1e-3:
            break
            
        # Solve quadratic subproblem
        d = solve_quadratic_subproblem(H, g, n_points)
        
        # Line search
        alpha, h_new = line_search(h, d, dx)
        
        # Update step
        h = h_new
        
        # Update Hessian (BFGS update)
        y = compute_gradient(h, dx) - g
        s = h - h + d * alpha  # very approximate s
        if np.linalg.norm(s) > 1e-12 and np.linalg.norm(y) > 1e-12:
            rho = 1.0 / np.dot(y, s)
            # Simplified BFGS update
            H = H - (rho * np.outer(s, y) @ H) + (rho * np.outer(y, s) @ H) - (rho * np.outer(s, s))
        
        # Evaluate new point
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
