import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h_init = np.clip(h_init, 0, 1)
        n_points = len(h_init)
        dx = 2.0 / n_points
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        n_points = 70
        dx = 2.0 / n_points
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Phase 1: Coarse optimization (n=35) for fast exploration
    n_coarse = 35
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.array([np.mean(h_init[i*2:(i+1)*2]) for i in range(n_coarse)])
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Define objective function for coarse phase
    def objective_coarse(h):
        h_rescaled = h * (n_coarse / 2) / np.sum(h)
        return evaluate_c5(h_rescaled, dx_coarse)
    
    # Define constraints for coarse phase
    def constraint_sum(h):
        return np.sum(h) - n_coarse / 2
    
    # Define bounds for coarse phase
    bounds = [(0, 1) for _ in range(n_coarse)]
    
    # L-BFGS-B optimization for coarse phase
    constraints = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(objective_coarse, h_coarse, method='L-BFGS-B', 
                      bounds=bounds, constraints=constraints, 
                      options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-5})
    
    h_opt_coarse = result.x
    if result.success:
        h_opt_coarse = h_opt_coarse * (n_coarse / 2) / np.sum(h_opt_coarse)
        c5_coarse = evaluate_c5(h_opt_coarse, dx_coarse)
        if c5_coarse < best_c5:
            best_c5 = c5_coarse
            best_h_coarse = h_opt_coarse
    
    # Phase 2: Refinement (n=70) with upsampling
    # Upsample coarse solution
    h_refined = np.zeros(n_points)
    for i in range(n_coarse):
        h_refined[2*i:2*i+2] = h_opt_coarse[i]
    
    # Add random perturbations
    perturbation = 0.1 * np.random.uniform(-1, 1, n_points)
    h_refined += perturbation
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    # Define objective function for refinement
    def objective_fine(h):
        h_rescaled = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h_rescaled, dx)
    
    # Define bounds
    bounds_fine = [(0, 1) for _ in range(n_points)]
    
    # L-BFGS-B optimization for refinement
    constraints_fine = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    result_fine = minimize(objective_fine, h_refined, method='L-BFGS-B', 
                           bounds=bounds_fine, constraints=constraints_fine,
                           options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-5})
    
    if result_fine.success:
        h_final = result_fine.x
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5_final = evaluate_c5(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Ensure we have a valid solution within budget
    end_time = time.time()
    remaining_time = budget_s - (end_time - start_time)
    
    if remaining_time < 1:
        return (best_h.tolist(), best_c5, n_points)

    for iter_i in range(50):  # A few iterations of local search
        if time.time() - start_time > budget_s - 1:

        current_c5 = evaluate_c5(best_h, dx)

        h_new = best_h + np.random.normal(0, 0.01, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            best_h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
