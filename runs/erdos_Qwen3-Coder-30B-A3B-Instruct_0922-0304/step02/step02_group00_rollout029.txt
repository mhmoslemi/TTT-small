import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    max_iter = 1000
    no_improvement_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Early exit if we hit target
        if current_c5 <= 0.3807:
            if best_h is None or current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            break
            
        # Check for convergence
        if best_h is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                break
        
        # Formulate and solve convex subproblem to reduce overlap at k_max
        # We want to minimize sum_i h[i] * (1 - h[(i + k_max) % n_points])
        # This is a separable quadratic term, but we'll linearize for convexity
        
        # CVXPY
        h_var = cp.Variable(n_points)
        
        # Construct linear approximation for the specific overlap term
        expr = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            expr += h_var[i] * (1 - h_var[j])
        
        # Minimize this linearized objective subject to constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        objective = cp.Minimize(expr)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if problem.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                h = np.clip(new_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # If optimization fails, use gradient descent approach as fallback
            step = 0.01
            iterations = 50
            h_temp = h.copy()
            for _ in range(iterations):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                    
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp

    if best_h is None:
        return (h.tolist(), best_c5, n_points)

    # Final validation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
