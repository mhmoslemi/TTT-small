def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def refine_solution(h, n_points, dx, max_iter=50):
    # Simple local refinement using gradient ascent
    # Use finite differences to estimate gradient
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Very simple bounded gradient descent step
    for _ in range(max_iter):
        current_c5 = evaluate_c5(best_h, dx)
        # Binary search for step size
        step_size = 0.1
        eps = 1e-4
        
        improved = False
        for i in range(n_points):
            # Gradient computation part
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            
            # Approximate gradient direction
            grad_i = (c5_plus - current_c5) / eps
            # Update in gradient descent direction
            h_new = best_h.copy()
            h_new[i] = best_h[i] - step_size * grad_i
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < current_c5:
                best_h = h_new
                improved = True
                
        if not improved:
            break
            
    return best_h
