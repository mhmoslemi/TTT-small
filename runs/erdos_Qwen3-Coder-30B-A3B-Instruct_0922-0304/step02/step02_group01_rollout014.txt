import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    """Scale h to have sum equal to target_sum while keeping values in [0,1]"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    scaled = h * target_sum / current_sum
    return np.clip(scaled, 0, 1)

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.1):
    """Gaussian mutation"""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += random.gauss(0, sigma)
    return np.clip(h_mut, 0, 1)

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = scale_to_target_sum(h0, target_sum)
        population.append(h0)
    else:
        # Fallback initialization
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = scale_to_target_sum(h0, target_sum)
        population.append(h0)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Find best in current generation
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament selection)
        tournament_size = 3
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
            
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep best individuals
        sorted_pop = sorted(zip(population, fitness), key=lambda x: x[1])
        elites = [x[0] for x in sorted_pop[:population_size//10]]
        new_population.extend(elites)
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            child1 = scale_to_target_sum(child1, target_sum)
            child2 = scale_to_target_sum(child2, target_sum)
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
    if best_h is None:
        # Fallback to best from initial run if we have no solution yet
        population = [scale_to_target_sum(h, target_sum) for h in population]
        fitness = [evaluate_c5(h, dx) for h in population]
        best_c5 = min(fitness)
        best_h = population[fitness.index(best_c5)]
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
