import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum, dx):
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scaling_factor = target_sum / current_sum
    h_scaled = h * scaling_factor
    return np.clip(h_scaled, 0, 1)

def crossover(parent1, parent2):
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.05):
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += random.gauss(0, sigma)
    return np.clip(h_mut, 0, 1)

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = scale_to_target_sum(h_init, target_sum, dx)
        population.append(h_init)
    
    # Add random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = scale_to_target_sum(h, target_sum, dx)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
            
        # Crossover and mutation
        new_population = []
        
        # Elitism: keep top 10%
        elite_count = population_size // 10
        elite_indices = np.argsort(fitness)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child1 = scale_to_target_sum(child1, target_sum, dx)
            
            child2 = mutate(child2)
            child2 = scale_to_target_sum(child2, target_sum, dx)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        generation += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h) if h is not None else 70
    return (h.tolist() if h is not None else [], c5_bound, n_points)
