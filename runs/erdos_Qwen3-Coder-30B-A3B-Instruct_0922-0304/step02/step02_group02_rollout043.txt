import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    # PSO parameters
    P = 50  # population size
    max_iter = 500
    omega = 0.7
    c1 = c2 = 2.0
    max_vel = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(P):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.05 * np.max(h), n_points)
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = repair_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_c5.append(evaluate_c5(particle, dx))
    
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx]
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(P):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                omega * velocities[i] +
                c1 * r1 * (pbest[i] - swarm[i]) +
                c2 * r2 * (gbest - swarm[i])
            )
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            # Repair feasibility
            swarm[i] = repair_h(swarm[i], n_points)
            
            # Evaluate and update pbest
            current_c5 = evaluate_c5(swarm[i], dx)
            if current_c5 < pbest_c5[i]:
                pbest[i] = swarm[i].copy()
                pbest_c5[i] = current_c5
                
                # Update gbest
                if current_c5 < gbest_c5:
                    gbest = swarm[i].copy()
                    gbest_c5 = current_c5
                    
            # Early exit
            if gbest_c5 <= 0.3807:
                break
                
        if gbest_c5 <= 0.3807:
            break
    
    # Gradient refinement on top particles
    top_k = max(1, P // 20)
    top_indices = np.argsort(pbest_c5)[:top_k]
    for idx in top_indices:

                    gbest_c5 = new_c5
            else:

    return (gbest.tolist(), gbest_c5, n_points)
