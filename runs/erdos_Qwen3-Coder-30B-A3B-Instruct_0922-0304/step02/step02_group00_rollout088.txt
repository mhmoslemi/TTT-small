import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing parameters
    T = 1.0
    beta = 0.995
    epsilon = 0.05
    cooldown_interval = 100
    reset_threshold = 1000
    
    iteration = 0
    no_improvement_count = 0
    
    while True:
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        
        # Perturb h
        h_candidate = h + np.random.normal(0, epsilon * T, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        new_c5 = evaluate_c5(h_candidate, dx)
        delta_c5 = new_c5 - current_c5
        
        # Accept with probability
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T):
            h = h_candidate
            current_c5 = new_c5
            
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Cooling schedule
        if iteration % cooldown_interval == 0:
            T *= beta
            
        # Reset temperature if no improvement for a while
        if no_improvement_count >= reset_threshold:
            T = 1.0
            no_improvement_count = 0
            
        iteration += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
