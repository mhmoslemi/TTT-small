import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Use a sawtooth-like pattern
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative majorization-minimization approach
    max_iter = 1000
    patience = 0
    max_patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlations
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > max_patience:
            break
            
        # Solve convex subproblem to reduce overlap at k_max
        h_var = Variable(n_points)
        
        # Build the linear overlap term for shift k_max
        # overlap = sum h[i] * (1 - h[i+k_max]) for all valid i
        # This is a linear constraint in h, so we minimize it directly
        
        # Create overlap expression for shift k_max
        obj = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            obj += h_var[i] * (1 - h_var[j])
            
        # Constraints
        constraints = [
            h_var[i] >= 0 for i in range(n_points)
        ] + [
            h_var[i] <= 1 for i in range(n_points)
        ] + [
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve the convex subproblem
        prob = Problem(Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array([float(h_var[i].value) for i in range(n_points)])
                h = h_new
            else:
                # Fallback to simple gradient step
                # Simple gradient computation using finite differences
                step = 0.01
                h_temp = h.copy()
                for _ in range(20):  # Reduce iterations for speed
                    corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_temp = np.max(corr_temp) * dx
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_temp) / eps
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback in case of solver issues
            # Simple gradient step
            step = 0.005
            h_temp = h.copy()
            for _ in range(50):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        # Ensure feasibility after update
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Fallback if sum is zero
            h = np.ones(n_points) * (n_points / 2) / n_points

    return (best_h.tolist(), best_c5, n_points)
