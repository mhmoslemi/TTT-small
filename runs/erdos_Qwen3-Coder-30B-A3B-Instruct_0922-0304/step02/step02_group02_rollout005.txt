import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using 5-fold symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create 5-fold symmetric initialization
        h = np.zeros(n_points)
        segment_size = n_points // 5
        values = [0.7, 0.3, 0.7, 0.3, 0.7]
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            h[start_idx:end_idx] = values[i]
        # Adjust to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function for optimization
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Constraint function for sum constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use L-BFGS-B optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with L-BFGS-B
        result = minimize(
            objective, 
            h, 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h = h_opt
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
