import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a piecewise constant starting solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Add some noise
    h += np.random.normal(0, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    
    # Compute initial C5
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temp = 0.1 * current_c5
    cooling_rate = 0.999
    min_temp = 1e-6
    
    max_iter = 10000
    gradient_refine_freq = 50
    
    # Main annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate
        candidate = h + np.random.normal(0, 0.01, n_points)
        candidate = project_to_feasible(candidate, n_points)
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            temp = max(temp * cooling_rate, min_temp)
        else:
            # Probabilistically accept
            delta = candidate_c5 - current_c5
            if random.random() < np.exp(-delta / temp):
                h = candidate
                current_c5 = candidate_c5
            temp = max(temp * cooling_rate, min_temp)
        
        # Occasionally refine with gradient descent
        if i % gradient_refine_freq == 0 and i > 0:
            # Simple gradient descent refinement
            step_size = 0.01
            for _ in range(5):
                eps = 1e-4
                grad = np.zeros_like(h)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    h_minus = h.copy()
                    h_minus[j] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    c5_minus = evaluate_c5(h_minus, dx)
                    
                    grad[j] = (c5_plus - c5_minus) / (2 * eps)
                
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                current_c5 = evaluate_c5(h, dx)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
                step_size *= 0.9
                
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
