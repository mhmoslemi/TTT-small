import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, k_star, dx):
    """Compute gradient for the overlap at shift k_star"""
    n = len(h)
    grad = np.zeros(n)
    
    # For shift k_star, compute gradient contribution
    if k_star >= 0:
        # h[i] * (1 - h[i + k_star])
        for i in range(n - k_star):
            grad[i] += 1 - h[i + k_star]
            if i + k_star < n:
                grad[i + k_star] -= h[i]
    else:
        # h[i - k_star] * (1 - h[i])
        for i in range(-k_star, n):
            grad[i] += 1 - h[i + k_star]
            if i + k_star >= 0:
                grad[i + k_star] -= h[i]
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available

        h = project_h(h, target_sum)

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Algorithm parameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    prev_update = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find maximum overlap shift
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        # Convert full convolution index back to shift value
        k_star = max_k - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, k_star, dx)
        
        # Apply momentum
        update = momentum * prev_update - step_size * grad
        prev_update = update
        
        # Update h
        h_new = h + update
        h = project_h(h_new, target_sum)
        
        # Add small random perturbation occasionally to escape local optima
        if iteration % 100 == 0 and iteration > 0:
            h += np.random.normal(0, 0.001, n_points)
            h = project_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
