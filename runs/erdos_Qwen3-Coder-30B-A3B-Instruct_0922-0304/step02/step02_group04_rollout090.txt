import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution_step(population, fitness, F=0.7, CR=0.9):
    new_population = []
    for i in range(len(population)):
        # Select three different individuals
        idxs = [idx for idx in range(len(population)) if idx != i]
        a, b, c = [population[j] for j in random.sample(idxs, 3)]
        
        # Mutation
        mutant = a + F * (b - c)
        
        # Crossover
        crossover_mask = np.random.random(len(mutant)) < CR
        trial = np.where(crossover_mask, mutant, population[i])
        
        new_population.append(trial)
    
    return new_population

def gradient_descent_refine(h, dx, max_iter=50, step_size=0.01, eps=1e-4):
    h = h.copy()
    for _ in range(max_iter):
        c5_val = evaluate_c5(h, dx)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_val) / eps
        
        h = h - step_size * grad
        h = project_to_feasible(h, len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    F = 0.7
    CR = 0.9
    gd_interval = 50
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # DE step
        population = differential_evolution_step(population, fitness, F, CR)
        
        # Apply feasibility projection
        for i in range(len(population)):
            population[i] = project_to_feasible(population[i], n_points)
        
        # Occasionally apply gradient descent
        if gen % gd_interval == 0 and gen > 0:
            h_refined = gradient_descent_refine(best_solution, dx)
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_solution = h_refined.copy()
        
        # Keep top 80%
        population = population[:int(0.8 * population_size)]
        
        # Add new random individuals to maintain population size
        while len(population) < population_size:
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    return (best_solution.tolist(), best_c5, n_points)
