import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_erdos_solution(h):
    # Evaluate the maximum overlap integral
    n = len(h)
    dx = 2.0 / n
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # Project h onto feasible region [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_shift, dx):
    # Compute approximate gradient for the specific shift k
    n = len(h)
    # Create shifted version of h with zero padding
    h_shifted = np.concatenate([h[k_shift:], np.zeros(k_shift)])
    grad = (1 - h) - h_shifted
    return grad * dx

def line_search(h, grad, dx, max_alpha=1.0, iterations=10):
    # Simple line search to find best step size
    best_alpha = 0.0
    best_c5 = evaluate_erdos_solution(h)
    for i in range(iterations):
        alpha = max_alpha * (i + 1) / iterations
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, len(h), dx)
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_alpha = alpha
            best_c5 = c5_new
    return best_alpha

def convex_relaxation(h, dx, lambda_reg=0.01):
    # Simple entropy regularization for exploration
    n = len(h)
    h = np.clip(h, 1e-6, 1 - 1e-6)
    entropy_grad = -np.log(h) - 1
    # Smooth manually since log can be unstable
    entropy_grad = np.where(h < 1e-6, 0, entropy_grad)
    h = h - 0.001 * (entropy_grad)  # small step
    h = project_feasible(h, n, dx)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate a simple stepped construction
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        # Slightly perturb
        h += np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent loop
    step_size = 0.01
    relaxation_interval = 50
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Total overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap_value = overlap[max_k] * dx
        if max_overlap_value < best_c5:
            best_c5 = max_overlap_value
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break

        # Compute gradient for the current maximum shift
        grad = compute_gradient(h, max_k, dx)
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n_points, dx)
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            
        iteration += 1
        if iteration % relaxation_interval == 0:
            # Apply convex relaxation periodically
            h = convex_relaxation(h, dx, lambda_reg=0.01)
    
    if time.time() - start_time > budget_s - 1:
        # Return best known solution
        pass
    
    return (best_h.tolist(), best_c5, n_points)
