import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.1):
    """Gaussian mutation"""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += random.gauss(0, sigma)
    return h

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialization
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = []
    
    # Add random initial individuals
    for _ in range(49):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness (lower C5 is better)
        fitness.sort(key=lambda x: x[0])
        
        # Update best solution
        current_best_c5, current_best_h = fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected = [h for _, h in fitness[:len(fitness)//2]]
        
        # Generate new population
        new_population = selected.copy()
        
        # Elitism: keep top 10%
        elite_count = max(1, len(selected) // 10)
        elite = [h for _, h in fitness[:elite_count]]
        new_population.extend(elite)
        
        # Crossover and mutation
        while len(new_population) < 50:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            child1 = project_to_feasible(child1, target_sum)
            child2 = project_to_feasible(child2, target_sum)
            new_population.extend([child1, child2])
        
        # Trim to correct population size
        population = new_population[:50]
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = genetic_algorithm(seed, budget_s)
    
    # Final validation
    n_points = len(best_h)
    dx = 2.0 / n_points
    # Ensure that the solution is valid
    best_h = project_to_feasible(best_h, n_points / 2)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
