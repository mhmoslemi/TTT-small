import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with 5-fold symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create 5-fold symmetric construction
        h = np.zeros(n_points)
        segment_size = n_points // 5
        values = [0.7, 0.3, 0.7, 0.3, 0.7]
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            h[start_idx:end_idx] = values[i]
        
    # Normalize to sum to 35 (which is n_points / 2)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective_and_grad(h):
        # Clip h to [0,1] to respect constraints
        h_clipped = np.clip(h, 0, 1)
        # Ensure sum constraint
        h_normalized = h_clipped * (n_points / 2) / np.sum(h_clipped)
        c5 = evaluate_c5(h_normalized, dx)
        return c5, 0  # We'll compute the gradient separately
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
        
    # Optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    max_iter = 500
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Run optimization step
        result = minimize(
            lambda x: evaluate_c5(np.clip(x, 0, 1) * (n_points / 2) / np.sum(np.clip(x, 0, 1)), dx),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8},
            callback=lambda x: None
        )
        
        if result.success:
            h = result.x
        else:
            # Use a simple gradient descent fallback
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
                
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Track best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
