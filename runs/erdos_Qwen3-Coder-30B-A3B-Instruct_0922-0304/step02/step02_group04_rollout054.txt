import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    # Compute approximation of gradient via finite differences
    eps = 1e-4
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Particle Swarm Optimization parameters
    n_particles = 30
    inertia_weight = 0.7
    cognitive_coeff = 2.0
    social_coeff = 2.0
    max_velocity = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_to_feasible(initial_h, n_points)
        swarm.append(initial_h)
    else:
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h += np.random.normal(0, 0.05, n_points)
        initial_h = project_to_feasible(initial_h, n_points)
        swarm.append(initial_h)
    
    # Add random particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize personal best positions and their fitness
    personal_best_positions = [p.copy() for p in swarm]
    personal_best_fitness = [evaluate_c5(p, dx) for p in swarm]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # Initialize velocities
    velocities = [np.random.uniform(-max_velocity, max_velocity, n_points) for _ in range(n_particles)]
    
    # Local optimization frequency
    local_opt_freq = 20
    local_opt_iter = 50
    
    # PSO iterations
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = random.random(), random.random()
            cognitive_component = cognitive_coeff * r1 * (personal_best_positions[i] - particle)
            social_component = social_coeff * r2 * (global_best_position - particle)
            new_velocity = inertia_weight * velocity + cognitive_component + social_component
            
            # Limit velocity
            new_velocity = np.clip(new_velocity, -max_velocity, max_velocity)
            
            # Update position
            new_position = particle + new_velocity
            new_position = project_to_feasible(new_position, n_points)
            
            # Evaluate new position
            new_fitness = evaluate_c5(new_position, dx)
            
            # Update personal best
            if new_fitness < personal_best_fitness[i]:
                personal_best_positions[i] = new_position.copy()
                personal_best_fitness[i] = new_fitness
                
                # Update global best if needed
                if new_fitness < global_best_fitness:
                    global_best_position = new_position.copy()
                    global_best_fitness = new_fitness
            
            # Update swarm
            swarm[i] = new_position
            velocities[i] = new_velocity
            
        # Periodic local refinement using gradient ascent
        if iteration % local_opt_freq == 0:
            top_particles = np.argsort(personal_best_fitness)[:n_particles//5]
            for p_idx in top_particles:
                h = personal_best_positions[p_idx].copy()
                for _ in range(local_opt_iter):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h, dx)
                    h -= 0.01 * grad
                    h = project_to_feasible(h, n_points)
                personal_best_positions[p_idx] = h
                personal_best_fitness[p_idx] = evaluate_c5(h, dx)
                
            # Update global best after refinement
            new_global_best_idx = np.argmin(personal_best_fitness)
            if personal_best_fitness[new_global_best_idx] < global_best_fitness:
                global_best_position = personal_best_positions[new_global_best_idx].copy()
                global_best_fitness = personal_best_fitness[new_global_best_idx]
                
        # Early termination
        if global_best_fitness <= 0.3807:
            break
            
    return (global_best_position.tolist(), global_best_fitness, n_points)
