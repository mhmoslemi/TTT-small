import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (70 / 2) / np.sum(h_init)
    else:
        h_init = np.ones(70)
        h_init[35:] = 0
        h_init += np.random.normal(0, 0.05, 70)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (70 / 2) / np.sum(h_init)

    best_c5 = float('inf')
    best_h = h_init.copy()
    dx = 2.0 / 70

    # Stage 1: Coarse optimization (n=35)
    n_points = 35
    h_coarse = np.ones(n_points)
    h_coarse[n_points//2:] = 0
    h_coarse += np.random.normal(0, 0.05, n_points)
    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse = h_coarse * (n_points / 2) / np.sum(h_coarse)
    
    # Constraint function for integral constraint
    def integral_constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds: [0,1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B
    max_iter = 500
    tolerance = 1e-6
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate overlap
        overlap = correlate(h_coarse, 1 - h_coarse, mode="full")
        max_overlap = np.max(overlap) * (2.0 / n_points)
        
        if max_overlap <= 0.3807:
            break
            
        # Optimize with L-BFGS-B
        cons = {'type': 'eq', 'fun': integral_constraint}
        result = minimize(lambda x: -np.max(correlate(x, 1 - x, mode="full")) * (2.0 / n_points),
                          h_coarse, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': tolerance, 'gtol': tolerance})
        
        if not result.success:
            break
            
        h_coarse = result.x
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_points / 2) / np.sum(h_coarse)
        
        # Evaluate
        current_c5 = -result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Interpolate to fine grid
            fine_h = np.interp(np.linspace(0, 2, 70), np.linspace(0, 2, 35), h_coarse)
            best_h = fine_h.copy()

    # Stage 2: Fine optimization (n=70)
    n_points = 70
    dx = 2.0 / n_points
    h_fine = best_h.copy()
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Add small random perturbations
    h_fine += np.random.normal(0, 0.01, n_points)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)

    # Bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

    max_iter = 1000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate overlap
        overlap = correlate(h_fine, 1 - h_fine, mode="full")
        max_overlap = np.max(overlap) * dx
        
        if max_overlap <= 0.3807:
            break
            
        # Optimization
        result = minimize(lambda x: -np.max(correlate(x, 1 - x, mode="full")) * dx,
                          h_fine, method='L-BFGS-B', bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': tolerance, 'gtol': tolerance})
        
        if not result.success:
            break
            
        h_fine = result.x
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Evaluate
        current_c5 = -result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
        
    # Final validation
    overlap = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(overlap) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h.tolist(), best_c5, 70)
