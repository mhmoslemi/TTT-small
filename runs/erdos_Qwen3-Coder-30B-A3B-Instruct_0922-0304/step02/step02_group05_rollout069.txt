import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create random initial solution
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Parameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_k = np.argmax(overlap)
        
        # Gradient computation for the maximum overlap
        # h_shifted = np.roll(h, max_k)
        # We compute the gradient directly from cross-correlation formula
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        current_c5 = evaluate_erdos_solution(h)
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        for alpha in line_search_steps:
            if time.time() - start_time > budget_s - 1:
                break
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_candidate_c5:
                best_candidate = h_candidate.copy()
                best_candidate_c5 = candidate_c5
        
        h = best_candidate
        current_c5 = best_candidate_c5
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Check for early convergence
        if best_c5 <= 0.3807:
            break
            
        # Periodically solve a convex relaxation
        if iteration % relaxation_interval == 0 and time.time() - start_time < budget_s - 1:
            # Convex relaxation via CVXPY
            h_cvx = cp.Variable(n_points)
            obj = cp.Max(cp.sum(cp.multiply(h_cvx, 1 - h_cvx)))
            constraints = [
                cp.sum(h_cvx) == n_points/2,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            prob = cp.Problem(cp.Minimize(obj), constraints)
            try:
                prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_relax = h_cvx.value
                    h_relax = project_feasible(h_relax, n_points, dx)
                    # Only update if it provides improvement
                    relax_c5 = evaluate_erdos_solution(h_relax)
                    if relax_c5 < current_c5:
                        h = h_relax
                        current_c5 = relax_c5
                        if current_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = current_c5
            except:
                pass  # Skip if optimization fails

    return (best_h.tolist(), best_c5, n_points)
