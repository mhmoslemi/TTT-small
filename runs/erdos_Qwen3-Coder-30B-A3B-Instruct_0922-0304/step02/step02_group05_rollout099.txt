import numpy as np
from scipy.signal import correlate
import time
import random


def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap


def project_feasible(h, n_points, dx):
    # Clip to [0, 1] and rescale to sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h


def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    inertia_start = 1.0
    inertia_min = 0.8
    c1 = 2.0
    c2 = 2.0
    velocity_limit = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create random initial particle
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Fill up particles
    while len(particles) < n_particles:
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(n_particles):
        v = np.random.normal(0, 0.01, n_points)
        velocities.append(v)
    
    # Evaluate initial particles
    fitness = []
    for i in range(n_particles):
        c5 = evaluate_c5(particles[i], dx)
        fitness.append(c5)
    
    # Set best positions
    p_best = [particles[i].copy() for i in range(n_particles)]
    p_best_fitness = fitness[:]
    
    # Find global best
    best_idx = np.argmin(fitness)
    g_best = particles[best_idx].copy()
    g_best_fitness = fitness[best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_fitness
    
    # PSO main loop
    inertia = inertia_start
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update inertia
        if iteration % 50 == 0:
            inertia = max(inertia_min, inertia - 0.02)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            new_velocity = (inertia * velocities[i] + 
                          c1 * r1 * (p_best[i] - particles[i]) + 
                          c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            new_velocity = np.clip(new_velocity, -velocity_limit, velocity_limit)
            
            # Update position
            new_position = particles[i] + new_velocity
            new_position = project_feasible(new_position, n_points, dx)
            
            # Evaluate new position
            new_c5 = evaluate_c5(new_position, dx)
            
            # Update particle best if improved
            if new_c5 < p_best_fitness[i]:
                p_best[i] = new_position.copy()
                p_best_fitness[i] = new_c5
                if new_c5 < g_best_fitness:
                    g_best = new_position.copy()
                    g_best_fitness = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_position.copy()
                        last_improvement = iteration
            
            # Update particle and velocity
            particles[i] = new_position
            velocities[i] = new_velocity
            
        # Occasionally reset stagnant particles
        if iteration - last_improvement > 100:
            for i in range(n_particles):
                if np.random.rand() < 0.2:
                    particles[i] = np.random.uniform(0, 1, n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    velocities[i] = np.random.normal(0, 0.01, n_points)
                    c5 = evaluate_c5(particles[i], dx)
                    fitness[i] = c5
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = c5
            
            # Re-evaluate global best
            if g_best_fitness > min(fitness):
                # Find new global best
                best_idx = np.argmin(fitness)
                g_best = particles[best_idx].copy()
                g_best_fitness = fitness[best_idx]
                
            last_improvement = iteration
            
    return (best_h.tolist(), best_c5, n_points)
