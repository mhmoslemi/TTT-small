import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Project h to sum to target_sum while keeping in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h with structured approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        # Structured initialization: symmetric, periodic pattern
        h = np.zeros(n_points)
        # Alternate high/low values in groups of 5 for 5-fold symmetry
        group_size = 14  # 70/5
        high_val = 0.7
        low_val = 0.3
        for i in range(5):
            start_idx = i * group_size
            end_idx = start_idx + group_size
            if i % 2 == 0:
                h[start_idx:end_idx] = high_val
            else:
                h[start_idx:end_idx] = low_val
        # Adjust to sum to target_sum
        h = h * target_sum / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define bounds: [0,1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        # Reshape to 1D array (scipy optimize expects flat arrays)
        h_flat = np.array(h_vec)
        # Project to feasible set
        h_proj = project_to_feasible(h_flat, target_sum)
        c5 = evaluate_c5(h_proj, dx)
        return c5
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Compute initial value
    current_c5 = objective(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    # Optimization using L-BFGS-B
    # Convert to flat array for scipy
    initial_flat = h.flatten()
    
    # Setup constraints and bounds
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization with extended time
    remaining_time = budget_s - (time.time() - start_time)
    maxiter = int(remaining_time / 2) if remaining_time > 2 else 10
    
    try:
        result = minimize(
            objective,
            initial_flat,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': min(maxiter, 1000), 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        # Extract best solution
        if result.success and result.fun < best_c5:
            h_opt = project_to_feasible(result.x, target_sum)
            current_c5 = evaluate_c5(h_opt, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
        else:
            # Fall back to simple gradient-free search if optimization fails
            for i in range(500):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Perturb the solution
                h_perturbed = best_h + 0.01 * (np.random.rand(n_points) - 0.5)
                h_perturbed = project_to_feasible(h_perturbed, target_sum)
                current_c5 = evaluate_c5(h_perturbed, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_perturbed.copy()
                
    except Exception:
        # If optimization fails, use a simple hill climbing approach
        for i in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_perturbed = best_h + 0.002 * (np.random.rand(n_points) - 0.5)
            h_perturbed = project_to_feasible(h_perturbed, target_sum)
            current_c5 = evaluate_c5(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()

    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
