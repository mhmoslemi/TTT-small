import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    P = 50  # number of particles
    particles = []
    
    # Use initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    # One particle from initial construction
    particles.append(h.copy())
    
    # Fill remaining particles with random feasible vectors
    for _ in range(P - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    velocities = [0.1 * np.random.randn(n_points) for _ in range(P)]
    
    # Evaluate initial fitness
    fitness = [evaluate_c5(p, dx) for p in particles]
    
    # Keep track of personal bests
    p_best = [p.copy() for p in particles]
    p_best_fitness = fitness.copy()
    
    # Global best
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    # PSO parameters
    omega = 1.0       # inertia weight
    c1 = 2.0          # cognitive parameter
    c2 = 2.0          # social parameter
    max_vel = 0.2     # maximum velocity
    
    # Main PSO loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(P):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            current_fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = current_fitness
                
                # Update global best
                if current_fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = current_fitness
        
        # Adapt inertia weight
        if iteration % 50 == 0:
            omega = max(0.8, omega - 0.02)
        
        # Prevent getting stuck
        if iteration % 100 == 0 and iteration > 0:
            # Reset worst particles
            worst_indices = np.argsort(fitness)[-10:]
            for idx in worst_indices:
                particles[idx] = np.random.rand(n_points)
                particles[idx] = project_feasible(particles[idx], n_points, dx)
                velocities[idx] = 0.1 * np.random.randn(n_points)
                fitness[idx] = evaluate_c5(particles[idx], dx)
                p_best[idx] = particles[idx].copy()
                p_best_fitness[idx] = fitness[idx]
        
        # Early exit if target reached
        if g_best_fitness <= 0.3807:
            break
            
        if time.time() - start_time > budget_s - 1:
            break
    
    return (g_best.tolist(), g_best_fitness, n_points)
