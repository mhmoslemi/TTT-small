import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum=h, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Compute overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    """Compute gradient via finite differences"""
    eps = 1e-4
    grad = np.zeros_like(h)
    c5_val = evaluate_c5(h, dx)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_val) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    local_refine_freq = 20
    n_local_iter = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Initialize with the provided solution or a random one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Add initial solution
    h_init = project_to_feasible(h_init, n_points)
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Add diverse random solutions
    while len(swarm) < swarm_size:
        h = h_init + np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Swarm initialization complete
    p_best = [swarm[i].copy() for i in range(swarm_size)]
    p_best_c5 = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Global best
    best_idx = np.argmin(p_best_c5)
    g_best = p_best[best_idx].copy()
    g_best_c5 = p_best_c5[best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - swarm[i]) + 
                           c2 * r2 * (g_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate
            current_c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if current_c5 < p_best_c5[i]:
                p_best[i] = swarm[i].copy()
                p_best_c5[i] = current_c5
                
                # Update global best
                if current_c5 < g_best_c5:
                    g_best = swarm[i].copy()
                    g_best_c5 = current_c5
        
        # Local refinement
        if iteration % local_refine_freq == 0 and iteration > 0:
            # Refine top 10% particles
            top_indices = np.argsort(p_best_c5)[:max(1, swarm_size // 10)]
            for idx in top_indices:
                h_refined = p_best[idx].copy()
                step_size = 0.01
                for _ in range(n_local_iter):
                    grad = compute_gradient(h_refined, dx)
                    h_refined = h_refined - step_size * grad
                    h_refined = project_to_feasible(h_refined, n_points)
                
                new_c5 = evaluate_c5(h_refined, dx)
                if new_c5 < p_best_c5[idx]:
                    p_best[idx] = h_refined
                    p_best_c5[idx] = new_c5
                    if new_c5 < g_best_c5:
                        g_best = h_refined
                        g_best_c5 = new_c5
        
        # Early termination if target reached
        if g_best_c5 <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_c5, n_points)
