import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) / 2
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        swarm.append(h_init.copy())
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        swarm.append(h_init.copy())
    
    # Fill swarm with diverse particles
    for _ in range(swarm_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        swarm.append(h.copy())
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    personal_bests = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_scores = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_bests[i] - swarm[i]) 
                           + c2 * r2 * (global_best - swarm[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            new_score = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_bests[i] = swarm[i].copy()
                # Update global best if needed
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best = swarm[i].copy()
            
            # Early exit condition
            if global_best_score <= 0.3807:
                break
                
        # Periodically refine top particles with gradient ascent
        if iteration % 20 == 0 and iteration > 0:
            top_indices = np.argsort(personal_best_scores)[:swarm_size//5]
            for idx in top_indices:
                h = personal_bests[idx].copy()
                # Simple gradient ascent
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    corr = correlate(h, 1 - h, mode="full")
                    c5_val = np.max(corr) * dx
                    
                    # Approximate gradient
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - c5_val) / eps
                    
                    h -= 0.01 * grad
                    h = project_to_feasible(h, n_points)
                
                # Update personal best if improved
                new_score = evaluate_c5(h, dx)
                if new_score < personal_best_scores[idx]:
                    personal_best_scores[idx] = new_score
                    personal_bests[idx] = h.copy()
                    if new_score < global_best_score:
                        global_best_score = new_score
                        global_best = h.copy()
            
            if global_best_score <= 0.3807:
                break
    
    # Final evaluation of global best
    best_h = global_best.copy()
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
