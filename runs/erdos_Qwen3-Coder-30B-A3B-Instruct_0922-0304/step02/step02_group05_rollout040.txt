import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimizer parameters
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_freq = 20
    lambda_reg = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        k_max = np.argmax(overlap)
        
        # Compute gradient
        shifted_h = np.roll(h, k_max)
        gradient = shifted_h - (1 - shifted_h)
        
        # Line search
        best_h_candidate = h.copy()
        best_candidate_c5 = evaluate_erdos_solution(h)
        step_found = False
        
        for step_size in line_search_steps:
            h_candidate = h - step_size * gradient
            h_candidate = project_feasible(h_candidate, n_points)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_h_candidate = h_candidate
                best_candidate_c5 = candidate_c5
                step_found = True
                
        if step_found:
            h = best_h_candidate
            if best_candidate_c5 < best_c5:
                best_c5 = best_candidate_c5
                best_h = h.copy()
        
        # Periodic relaxation
        if iteration % relaxation_freq == 0 and iteration > 0:
            # Solve convex relaxation using CVXPY
            try:
                h_relax = cp.Variable(n_points)
                obj = cp.Maximize(cp.sum(cp.multiply(h_relax, 1 - h_relax)))
                constraints = [
                    cp.sum(h_relax) == n_points / 2,
                    h_relax >= 0,
                    h_relax <= 1
                ]
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax_vals = h_relax.value
                    if h_relax_vals is not None:
                        h = project_feasible(h_relax_vals, n_points)
                        new_c5 = evaluate_erdos_solution(h)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
            except:
                pass  # Continue with current solution if convex relaxation fails
            
            # Adjust regularization parameter
            if lambda_reg > 1e-6:
                lambda_reg *= 0.95
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
