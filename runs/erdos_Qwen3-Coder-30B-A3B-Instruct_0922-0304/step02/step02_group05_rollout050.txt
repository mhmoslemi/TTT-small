import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, shift_k):
    """Compute approximate gradient for overlap at shift k"""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # Using finite differences
    for i in range(n_points):
        # Approximate derivative by shifting
        shifted = np.roll(h, shift_k)
        grad[i] = shifted[i] - h[i]
    
    return grad

def line_search(h, grad, dx, max_iter=10):
    """Simple backtracking line search"""
    alpha = 0.01
    c = 0.5
    beta = 0.5
    orig_c5 = evaluate_c5(h, dx)
    
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, len(h), dx)
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 <= orig_c5 - c * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= beta
    return h, alpha

def convex_relaxation(h, dx, lambda_reg=0.01):
    """Perform a simple convex relaxation step"""
    n_points = len(h)
    # Add regularization to avoid getting stuck
    h_relaxed = h.copy()
    h_relaxed = np.clip(h_relaxed, 1e-6, 1-1e-6)
    
    # Simple heuristic: smooth with some entropy penalty 
    entropy = -np.sum(h_relaxed * np.log(h_relaxed + 1e-10))
    # Use entropy gradient to smooth
    entropy_grad = -np.log(h_relaxed + 1e-10) - 1
    # TODO: proper gradient descent
    return project_feasible(h_relaxed - 0.001 * entropy_grad, n_points, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points

        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    step_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate overlap for all shifts to find max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Recompute gradient using approximate method
        grad = compute_gradient(h, max_k)
        
        # Line search for optimal step size
        h_new, _ = line_search(h, grad, dx)
        
        # Apply new solution
        h = h_new
        h = project_feasible(h, n_points, dx)
        
        # Update best solution
        c5_new = evaluate_c5(h, dx)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break

        if step_count % 50 == 0:

            c5_new = evaluate_c5(h, dx)
