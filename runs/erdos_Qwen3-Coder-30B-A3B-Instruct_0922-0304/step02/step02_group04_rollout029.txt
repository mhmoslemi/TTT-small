import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Add more particles
    for _ in range(swarm_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_c5 = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            new_position = swarm[i] + velocities[i]
            swarm[i] = project_to_feasible(new_position, n_points)
            
            # Evaluate new position
            new_c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if new_c5 < personal_best_c5[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_c5[i] = new_c5
            
            # Update global best
            if new_c5 < global_best_c5:
                global_best = swarm[i].copy()
                global_best_c5 = new_c5
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0 and iteration > 0:
            for i in range(swarm_size // 5):  # Refine top 20%
                ref_idx = np.argsort(personal_best_c5)[:swarm_size // 5][i]
                h_refine = personal_best[ref_idx].copy()
                
                # Simple gradient ascent
                step_size = 0.01
                for _ in range(50):
                    corr = correlate(h_refine, 1 - h_refine, mode="full")
                    c5_val = np.max(corr) * dx
                    
                    # Estimate gradient
                    grad = np.zeros_like(h_refine)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_refine.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - c5_val) / eps
                    
                    h_refine = h_refine - step_size * grad
                    h_refine = project_to_feasible(h_refine, n_points)
                
                # Update personal best if improved
                ref_c5 = evaluate_c5(h_refine, dx)
                if ref_c5 < personal_best_c5[ref_idx]:
                    personal_best[ref_idx] = h_refine.copy()
                    personal_best_c5[ref_idx] = ref_c5
                    
                # Update global best if needed
                if ref_c5 < global_best_c5:
                    global_best = h_refine.copy()
                    global_best_c5 = ref_c5
        
        # Early exit condition
        if global_best_c5 <= 0.3807:
            break
            
    return (global_best.tolist(), global_best_c5, n_points)
