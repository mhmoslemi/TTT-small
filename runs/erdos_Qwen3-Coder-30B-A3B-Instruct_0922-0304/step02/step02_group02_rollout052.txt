import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum constraint and bounds"""
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize in case clipping affected sum
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: start with symmetric construction
    h = np.ones(n_points)
    # Create 5-fold symmetric pattern
    segment_size = n_points // 5
    for i in range(5):
        start = i * segment_size
        end = start + segment_size
        if i % 2 == 0:
            h[start:end] = 0.8
        else:
            h[start:end] = 0.2
    
    # Adjust to meet constraint sum(h) = n_points/2
    h = project_to_feasible(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use scipy.optimize with L-BFGS-B
    def obj_and_grad(h_flat):
        h = h_flat.copy()
        h = project_to_feasible(h, n_points)
        c5 = evaluate_c5(h, dx)
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        base_c5 = c5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
        return c5, grad
    
    # Run optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimization
        res = minimize(
            lambda h_flat: obj_and_grad(h_flat)[0],
            h,
            method='L-BFGS-B',
            jac=lambda h_flat: obj_and_grad(h_flat)[1],
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 10}
        )
        
        if res.success:
            h = res.x
            h = project_to_feasible(h, n_points)
        else:
            h = project_to_feasible(h, n_points)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Occasionally update the h vector
        if iteration % 50 == 0:
            # Apply some perturbation to escape local minima
            h = project_to_feasible(h + 0.01 * np.random.randn(n_points), n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
