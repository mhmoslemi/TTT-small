import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        # Fallback initialization with n=70
        n_points = 70
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0

    # Initialize to feasible solution with integral=1
    h = np.clip(initial_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Stage 1: coarse grid optimization (n=35)
    n_coarse = 35
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.linspace(0, 1, n_coarse)
    h_coarse[:n_coarse//2] = 1.0
    h_coarse[n_coarse//2:] = 0.0
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Optimize coarse solution
    def objective_coarse(h_vals):
        h = np.clip(h_vals, 0, 1)
        h = h * (n_coarse / 2) / np.sum(h)
        corr = correlate(h, 1-h, mode="full")
        return np.max(corr) * dx_coarse

    # Create bounds
    bounds = [(0, 1) for _ in range(n_coarse)]
    
    # Optimization with L-BFGS-B
    try:
        result_coarse = minimize(objective_coarse, h_coarse, method='L-BFGS-B', bounds=bounds, 
                                options={'ftol': 1e-9, 'gtol': 1e-5, 'maxiter': 500})
        if result_coarse.success:
            h_coarse = np.clip(result_coarse.x, 0, 1)
            h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
            c5_coarse = objective_coarse(h_coarse)
            if c5_coarse < best_c5:
                best_c5 = c5_coarse
                # Upsample to fine grid
                h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
                h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
                best_h = h_fine.copy()
    except:
        pass
    
    # Stage 2: refinement with fine grid (n=70)
    h = best_h.copy()
    dx_fine = 2.0 / n_points
    
    def objective_fine(h_vals):
        h = np.clip(h_vals, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1-h, mode="full")
        return np.max(corr) * dx_fine

    # Create bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    iteration = 0
    max_iter = 1000
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        try:
            # Perturbation
            h_perturb = h + np.random.normal(0, 0.02, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            
            # L-BFGS-B optimization
            result = minimize(objective_fine, h_perturb, method='L-BFGS-B', bounds=bounds,
                            options={'ftol': 1e-9, 'gtol': 1e-5, 'maxiter': 100})
            
            if result.success:
                h_new = np.clip(result.x, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = objective_fine(h_new)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    # Early stopping if target reached
                    if best_c5 <= 0.3807:
                        break
                else:
                    h = h_perturb
            else:
                h = h_perturb
                
        except Exception:
            pass
            
        iteration += 1

    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = objective_fine(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
