import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        
    h = project_to_feasible(h, n_points)
    
    # Evaluate initial value
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Setup annealing parameters
    T = 0.1 * current_c5
    cooling_rate = 0.999
    min_T = 1e-6
    step_size = 0.01
    
    iterations = 0
    max_iterations = 10000
    
    while iterations < max_iterations and time.time() - start_time < budget_s - 1:
        # Generate candidate by perturbing current solution
        candidate_h = h + np.random.normal(0, 0.01, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject based on simulated annealing
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
                if best_c5 <= 0.3807:
                    break
        
        # Lower temperature
        T = max(T * cooling_rate, min_T)
        
        # Every 50 iterations, refine with gradient descent
        if iterations % 50 == 0:
            # Compute gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, n_points)
                c5_minus = evaluate_c5(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Gradient descent step
            h = h - step_size * grad
            h = project_to_feasible(h, n_points)
            current_c5 = evaluate_c5(h, dx)
            
            # Adjust step size
            step_size *= 0.99
            
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
