    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def local_optimize_step(h_orig, dx, n_points):
    # Basic optimization step using gradient descent in a simplified way
    h = h_orig.copy()
    
    # Use a local search optimization
    step_size = 0.01
    
    for iter in range(30):
        # Compute overlap with current h
        corr = correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        
        # Approximate derivatives via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_val) / eps
            
        # Gradient descent step
        h = h - step_size * grad
        h = project_to_feasible(h, n_points)
        
    return h
