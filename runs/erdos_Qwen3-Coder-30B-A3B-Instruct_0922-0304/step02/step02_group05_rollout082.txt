import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def gradient_descent_step(h, dx, n_points):
    """Compute gradient using finite differences"""
    # Find max overlap position
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    
    # Numerical gradient
    eps = 1e-5
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points, dx)
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        max_plus = np.max(overlap_plus)
        grad[i] = (max_plus - np.max(overlap)) * dx / eps
    
    return grad

def adaptive_line_search(h, grad, dx, n_points, max_iter=10):
    """Line search to find good step size"""
    step_sizes = [1e-4, 1e-3, 1e-2, 1e-1]
    current_c5 = evaluate_erdos_solution(h)
    
    for alpha in step_sizes:
        h_candidate = h - alpha * grad
        h_candidate = project_feasible(h_candidate, n_points, dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < current_c5:
            return h_candidate, candidate_c5
    
    return h, current_c5

def convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    """Use CVXPY to solve a relaxed version"""
    # Create optimization problem
    h_var = Variable(n_points)
    # Objective: max overlap + regularization

    # Use a simple constrained optimization approach
    try:
        # Simple uniformization attempt
        lambda_reg = max(lambda_reg * 0.5, 1e-6)
        h_ret = h - lambda_reg * (h - 0.5)
        h_ret = project_feasible(h_ret, n_points, dx)
        return h_ret
    except:

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_feasible(h, n_points, dx)
    
    # Initialize tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Main optimization loop
    iteration = 0
    max_iter = int(budget_s / 0.1)  # Rough estimate
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current gradient estimate
        grad = gradient_descent_step(h, dx, n_points)
        
        # Perform line search to find good step
        h_new, c5_new = adaptive_line_search(h, grad, dx, n_points)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update current solution
        h = h_new
        
        # Periodic convex relaxation (every 100 steps)
        if iteration % 100 == 0 and iteration > 0:

        # Early exit condition

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
