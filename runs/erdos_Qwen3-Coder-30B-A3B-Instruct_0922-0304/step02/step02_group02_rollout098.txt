import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing
    T = 1000.0
    alpha = 0.999
    T_min = 1.0
    
    # Perturbation parameters
    delta = 0.1
    
    # Annealing loop
    while T > T_min and time.time() - start_time < budget_s - 1:
        # Create perturbation
        h_prime = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        
        # Adjust selected indices
        for i in indices:
            h_prime[i] += delta
        # Distribute the change to maintain sum constraint
        total_change = np.sum(h_prime) - (n_points / 2)
        if total_change != 0:
            # Reduce all other elements proportionally to keep sum at 35
            scale_factor = (n_points / 2) / np.sum(h_prime)
            h_prime = h_prime * scale_factor
            
        # Ensure bounds
        h_prime = np.clip(h_prime, 0, 1)
        
        # Compute C5 for both solutions
        current_c5 = evaluate_c5(h, dx)
        candidate_c5 = evaluate_c5(h_prime, dx)
        
        # Accept or reject the candidate
        delta_E = candidate_c5 - current_c5
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.random():
            h = h_prime.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                
        # Update temperature
        T *= alpha
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
