import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iterations = 10000

        h = np.clip(h, 0, 1)

    else:
        # Create a simple initial solution
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Regularization parameter
    lambda_reg = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        # Compute gradient estimate using gradient of shifted indices with max overlap
        # We compute the gradient for the shift k that maximizes overlap
        k = max_overlap_idx - (len(h) - 1)  # Convert full correlation index to shift
        
        # Gradient of H_i * (1 - H_{i+k}) with respect to H_i is (1 - H_{i+k}) - H_{i+k} = 1 - 2*H_{i+k}
        # This is simpler to compute: just rolling the array
        h_shifted = np.roll(h, k)
        gradient = 1 - 2 * h_shifted
        
        # Line search with step sizes
        step_sizes = [0.1, 0.5, 1.0]
        best_h_candidate = h.copy()
        best_c5_candidate = best_c5

            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            
            if c5_candidate < best_c5_candidate:
                best_h_candidate = h_candidate.copy()
                best_c5_candidate = c5_candidate
                
        # Update h and best if improved
        if best_c5_candidate < best_c5:
            h = best_h_candidate.copy()
            best_c5 = best_c5_candidate
            if best_c5 < 0.3807:
                break
                
        # Periodically solve convex relaxation
        if iteration % 20 == 0 and iteration > 0:
            # Solve relaxed problem: minimize overlap + lambda_reg * sum(h*log(h))
            # We will do this approximately with a constraint.
            try:
                # Compute current gradient direction that maximizes overlap
                # Note: there is a simplification error in previous gradient modeling
                # This is a proper relaxation: use a simpler approach
                
                # Create convex problem
                h_relax = cp.Variable(n_points, bounds=(0, 1))
                h_sum = cp.sum(h_relax)
                
                # Introduce auxiliary variables to replace minimization with lhs = 1
                # This might be overly complex - let's stick to simpler strategies
                
                # Alternatively use previously computed solution as warm start even if not convex
                # To add some variation, let's try local Gaussian smoothing
                
            except:
                pass
                
            # For now, continue with simpler smooth strategy
            # Apply small random perturbations occasionally to escape local minima
            if np.random.random() < 0.1:
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = project_feasible(h, n_points, dx)
                
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
