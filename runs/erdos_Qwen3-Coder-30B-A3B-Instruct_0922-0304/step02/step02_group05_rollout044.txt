import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default problem parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Basic initialization
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Convex relaxation parameters
    lambda_reg = 0.01
    relaxation_interval = 20
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Compute gradient for the dominant overlap
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search with step sizes
        step_sizes = [0.1, 0.5, 1.0]
        best_candidate = h.copy()
        best_candidate_c5 = best_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate = h_candidate
                best_candidate_c5 = candidate_c5
                break  # Take first improving step
        
        # Update if we found improvement
        if best_candidate_c5 < best_c5:
            h = best_candidate
            best_c5 = best_candidate_c5
            
            # Update global best
            if best_c5 < evaluate_erdos_solution(best_h):
                best_h = h.copy()
                
        # Periodic convex relaxation to escape local minima
        if iteration % relaxation_interval == 0 and iteration > 0:
            # Reformulate as convex relaxation
            h_relax = cp.Variable(n_points, nonneg=True)
            constraints = [cp.sum(h_relax) == n_points/2]
            
            # KL divergence regularization
            obj = cp.Minimize(cp.max(cp.conv(cp.conv(cp.reshape(h_relax, (1, -1)), 
                                                       cp.reshape(1 - h_relax, (-1, 1)), 
                                                       mode='full')) * dx) 
                              + lambda_reg * cp.sum(cp.entr(h_relax) + cp.log(n_points/2) + 1))
            
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            new_h = np.array(h_relax.value).flatten()
            # Project to unit cube and fix sum constraint
            new_h = np.clip(new_h, 0, 1)
            sum_h = np.sum(new_h)
            if abs(sum_h - n_points/2) > 1e-10:
                new_h = new_h * (n_points/2) / sum_h
                
            # Use this as warm start
            h = new_h
            
            # Update best if needed
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
        # Minor adjustment to regularization if needed
        if iteration % 50 == 0:
            lambda_reg *= 0.9
            
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
