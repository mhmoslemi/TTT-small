import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use a construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 500
    
    # Augmented Lagrangian parameters
    rho = 1.0
    lambda_mult = np.zeros(2 * n_points - 1)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Construct convex subproblem to reduce overlap at k_max
        h_var = cp.Variable(n_points)
        
        # Overlap constraint at k_max
        overlap_constraint = cp.Variable(2 * n_points - 1)
        
        # Objective: minimize max overlap with AL penalty
        objective = cp.Minimize(
            cp.sum(cp.multiply(lambda_mult, overlap_constraint - corr)) +
            rho * cp.sum_squares(overlap_constraint - corr)
        )
        
        # Setup constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            overlap_constraint == cp.conv(cvxpy_h, 1 - cvxpy_h, mode="full")
        ]
        
        # Dummy objective for CVXPY; will be managed separately
        problem = cp.Problem(objective, constraints)
        
        try:
            # The systematic approach to optimize involves:
            # 1. Constructing proper AL subproblem
            # 2. A simplified approach with gradient descent per iteration
            
            # Instead, use a gradient based update with better convergence
            step = 0.01 * (1.0 + iteration / 100.0)
            iterations = 50
            
            # Finite difference gradient estimation
            h_new = h.copy()
            for _ in range(iterations):
                # Evaluate current overlap
                corr_current = correlate(h_new, 1 - h_new, mode="full")
                c5_current = np.max(corr_current) * dx
                
                # Estimate gradient with finite differences
                grad = np.zeros_like(h_new)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_new.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_current) / eps
                
                # Update with gradient
                h_new -= step * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            h = h_new
            
        except:
            # Fall back to basic descent if no convergence
            try:
                # Direct gradient descent step
                corr_current = correlate(h, 1 - h, mode="full")
                c5_current = np.max(corr_current) * dx
                
                # Simple gradient descent with reduction
                step = 0.01
                h_new = h - step * np.gradient(corr_current)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            except:
                pass
            
        # Enforce hard constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
