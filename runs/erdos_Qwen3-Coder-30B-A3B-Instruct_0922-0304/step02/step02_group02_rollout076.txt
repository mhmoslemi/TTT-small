import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with symmetric construction for better convergence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create symmetric pattern: 5 segments with alternating high/low
        segment_size = n_points // 5
        h = np.zeros(n_points)
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            # Alternate high and low values
            for j in range(start_idx, end_idx):
                if (j - start_idx) % 2 == 0:
                    h[j] = 0.6
                else:
                    h[j] = 0.4
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define the objective function for optimization
    def objective(h_flat):
        h = h_flat.copy()
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Clip to bounds
        h = np.clip(h, 0, 1)
        return evaluate_c5(h, dx)
    
    # Define constraint for sum
    def constraint_sum(h_flat):
        h = h_flat.copy()
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B optimization
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization loop with multiple restarts
    max_restarts = 10
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Run optimization
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            h_opt = result.x.copy()
            # Enforce feasibility
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            current_c5 = evaluate_c5(h_opt, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                
                if best_c5 <= 0.3807:
                    break
                    
        # Random restart with slight perturbation
        h = best_h.copy()
        # Add small noise
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        restart_count += 1

    # Final validation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
