import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    # Compute gradient using finite differences
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def line_search(h, d, dx, max_alpha=1.0, c1=1e-4):
    # Armijo backtracking line search
    alpha = max_alpha
    c5_base = evaluate_c5(h, dx)
    grad = compute_gradient(h, dx)
    gTd = np.dot(grad, d)
    
    while alpha > 1e-10:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new <= c5_base + c1 * alpha * gTd:
            return alpha
        alpha *= 0.5
    
    return alpha

def solve_quadratic_subproblem(h, grad, H, dx):
    # Solve quadratic subproblem for step direction
    # Minimize: 0.5 * d^T * H * d + grad^T * d
    # Subject to: sum(d) = 0, 0 <= h + d <= 1
    
    n = len(h)
    
    # Try simple projected gradient descent
    d = -grad
    
    # Project step to feasible region
    # Ensure h + d in [0,1] and sum(d) = 0
    d_proj = d.copy()
    
    # Check bounds
    for i in range(n):
        if h[i] + d_proj[i] < 0:
            d_proj[i] = -h[i]
        elif h[i] + d_proj[i] > 1:
            d_proj[i] = 1 - h[i]
    
    # Project to zero-sum
    d_proj = d_proj - np.mean(d_proj)
    
    return d_proj

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h

        h = project_to_feasible(h, n_points)
    else:
        # Initialize with a reasonable starting point
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    # Initialize Hessian approximation as identity
    H = np.eye(n_points)
    
    # Early exit if target achieved
    c5_prev = evaluate_c5(h, dx)
    if c5_prev <= 0.3807:
        return h, c5_prev
    
    # Main SQP loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Solve quadratic subproblem
        d = solve_quadratic_subproblem(h, grad, H, dx)
        
        # Line search
        alpha = line_search(h, d, dx)
        
        # Update h
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Update Hessian using BFGS
        y = grad.copy()
        s = alpha * d
        
        # In practice, this would be more involved, but we just replace the Hessian
        # with identity to simplify (to avoid precision issues with large matrices)  
        H = np.eye(n_points)
        
        # Project to feasible domain
        h = project_to_feasible(h_new, n_points)
        
        # Check convergence
        c5_new = evaluate_c5(h, dx)
        if abs(c5_new - c5_prev) < 1e-5:
            break
        c5_prev = c5_new
        
        # Early exit if target reached
        if c5_new <= 0.3807:
            break

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
