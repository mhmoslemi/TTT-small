import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    
    # Evaluate original
    c5_orig = evaluate_c5(h, dx)
    
    # Finite differences for gradient
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_orig) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # PSO parameters
    swarm_size = 30
    iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm with initial solution if available
    swarm = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
        swarm.append(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = project_to_feasible(h_init, n_points)
        swarm.append(h_init)
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Personal best positions and fitness
    personal_best_positions = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_fitness = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Global best
    best_idx = np.argmin(personal_best_fitness)
    global_best_position = personal_best_positions[best_idx].copy()
    global_best_fitness = personal_best_fitness[best_idx]
    
    # PSO Main Loop
    for iter_num in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best_positions[i] - swarm[i]) +
                            c2 * r2 * (global_best_position - swarm[i]))
            
            # Apply max velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            fitness = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best_positions[i] = swarm[i].copy()
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best_position = swarm[i].copy()
        
        # Occasionally refine top particles with gradient ascent
        if iter_num % 20 == 0 and iter_num > 0:
            # Sort by fitness and refine top 10%
            sorted_indices = np.argsort(personal_best_fitness)
            top_count = max(1, swarm_size // 10)
            for i in range(top_count):
                idx = sorted_indices[i]
                h_refine = personal_best_positions[idx].copy()
                
                # Gradient ascent
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h_refine, dx)
                    h_refine = h_refine - step_size * grad
                    h_refine = project_to_feasible(h_refine, n_points)
                
                # Update personal best if new solution is better
                new_fitness = evaluate_c5(h_refine, dx)
                if new_fitness < personal_best_fitness[idx]:
                    personal_best_fitness[idx] = new_fitness
                    personal_best_positions[idx] = h_refine.copy()
                    if new_fitness < global_best_fitness:
                        global_best_fitness = new_fitness
                        global_best_position = h_refine.copy()
                        
        # Early exit condition
        if global_best_fitness <= 0.3807:
            break
    
    # Final evaluation of global best
    final_c5 = evaluate_c5(global_best_position, dx)
    
    return (global_best_position.tolist(), final_c5, n_points)
