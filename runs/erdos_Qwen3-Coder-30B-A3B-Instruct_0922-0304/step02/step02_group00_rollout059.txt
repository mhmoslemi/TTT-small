import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    if 'initial_h_values' in globals():
        # Seed particles with initial solution and random variations
        for i in range(n_particles):
            if i < n_particles // 2:
                h = np.array(initial_h_values)
            else:
                h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
            pbest.append(h.copy())
            pbest_values.append(float('inf'))
    else:
        # Random initialization
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
            pbest.append(h.copy())
            pbest_values.append(float('inf'))
    
    # Global best
    gbest = None
    gbest_value = float('inf')
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            value = evaluate_erdos_solution(h, dx)
            
            # Update personal best
            if value < pbest_values[i]:
                pbest_values[i] = value
                pbest[i] = h.copy()
                
            # Update global best
            if value < gbest_value:
                gbest_value = value
                gbest = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            pb = pbest[i]
            gb = gbest
            
            # Velocity update
            v = w * v + c1 * r1 * (pb - h) + c2 * r2 * (gb - h)
            velocities[i] = v
            
            # Position update
            h = h + v
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h

        # Adjust inertia weight
        if iteration > 100:
            w = 0.5

        # Early termination if target reached
        if gbest_value <= 0.3807:
            break

    return (gbest.tolist(), gbest_value, n_points)
