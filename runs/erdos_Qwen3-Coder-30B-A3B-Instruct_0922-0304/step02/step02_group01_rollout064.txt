import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    h_scaled = h * target_sum / current_sum
    return h_scaled

def crossover(parent1, parent2, crossover_point):
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate, sigma=0.1):
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, sigma)
    return np.clip(h_mut, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population parameters
    pop_size = 50
    mutation_rate = 0.1
    elite_size = pop_size // 10
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    # Add random individuals to population
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Create next generation
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = random.randint(0, pop_size//2)
            parent2_idx = random.randint(0, pop_size//2)
            parent1 = sorted_population[parent1_idx]
            parent2 = sorted_population[parent2_idx]
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child1, child2 = crossover(parent1, parent2, crossover_point)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Ensure sum constraint
            child1 = scale_to_target_sum(child1, target_sum)
            child2 = scale_to_target_sum(child2, target_sum)
            
            # Add to new population
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
