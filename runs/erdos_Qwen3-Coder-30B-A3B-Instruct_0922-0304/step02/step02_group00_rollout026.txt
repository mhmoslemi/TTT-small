import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    patience = 0
    patience_limit = 100
    
    # Frank-Wolfe style majorization-minimization approach
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Convex optimization to reduce overlap at k_max
        h_var = Variable(n_points)
        
        # Create linear objective for overlap at shift k_max
        # We want to minimize sum_{i} h[i] * (1 - h[(i + k_max) % n_points])
        objective = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            objective += h_var[i] * (1 - h_var[j])
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve LP subproblem
        try:
            prob = Problem(Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_var.value).flatten()
                h = h_new
            else:
                # Fallback if solve fails
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback to simple projection if solver fails
            h = h * (n_points / 2) / np.sum(h)
        
        # Maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new result
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > patience_limit:
            break
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
