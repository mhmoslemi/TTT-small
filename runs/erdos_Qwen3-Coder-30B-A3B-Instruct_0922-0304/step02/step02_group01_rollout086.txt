import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def initialize_population(n_points, pop_size):
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    for _ in range(pop_size - len(population)):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1):
    n_points = len(h)
    for i in range(n_points):
        if random.random() < mutation_rate:
            h[i] += random.gauss(0, 0.05)
    return h

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    generations = 1000
    
    population = initialize_population(n_points, pop_size)
    
    best_c5 = float('inf')
    best_h = None
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Find best in current generation
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Selection - tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_size = 3
            tournament_indices = random.sample(range(pop_size), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[tournament_fitness.index(min(tournament_fitness))]
            selected.append(population[winner_index].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            child1 = repair(child1, n_points)
            child2 = repair(child2, n_points)
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
        # Elitism: keep best solution so far
        if best_h is not None:
            population[0] = best_h.copy()
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # Fallback to the best from population
        fitness = [evaluate_c5(h, dx) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        final_c5 = fitness[best_idx]
    
    return (best_h.tolist(), final_c5, n_points)
