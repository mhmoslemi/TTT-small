import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, norm

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def local_refinement(h_init, n_points, dx, budget_remaining):
    # Use simpler hill-climbing approach for refinement
    h = h_init.copy()
    original_c5 = evaluate_c5(h, dx)
    
    best_h = h.copy()
    best_c5 = original_c5
    
    step_size = 0.01
    iterations = min(1000, max(100, int(budget_remaining * 200)))
    
    for i in range(iterations):
        # Compute gradient approx via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        baseline_c5 = evaluate_c5(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[j] = (c5_plus - baseline_c5) / eps
        
        # Gradient descent step
        h = h - step_size * grad
        h = project_to_feasible(h, n_points)
        
        # Update best
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
    return best_h, best_c5
