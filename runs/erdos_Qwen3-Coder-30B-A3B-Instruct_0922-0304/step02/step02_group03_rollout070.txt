import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        
    n_points = len(h)
    dx = 2.0 / n_points
    h = enforce_feasibility(h, n_points)
    
    # Define objective function
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Define constraint for integral = 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    maxiter = 1000
    ftol = 1e-6
    
    # Run optimization with L-BFGS-B
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=bounds,
                   constraints={'type': 'eq', 'fun': constraint},
                   options={'maxiter': maxiter, 'ftol': ftol})
    
    # Get best solution
    best_h = res.x
    best_h = enforce_feasibility(best_h, n_points)
    c5 = evaluate_overlap(best_h, dx)
    
    # Try to improve with local search if we have time
    if time.time() - start_time < budget_s - 1:
        best_c5 = c5
        best_h_final = best_h.copy()
        
        # Try a few variations to avoid local minima
        for _ in range(10):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Small random perturbation
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = enforce_feasibility(h_perturbed, n_points)
            c5_perturbed = evaluate_overlap(h_perturbed, dx)
