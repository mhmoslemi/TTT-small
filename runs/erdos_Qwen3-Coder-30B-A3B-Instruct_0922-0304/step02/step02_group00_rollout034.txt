import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as base for some particles
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
        for i in range(n_particles // 2):
            h = base_h + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
        
        # Fill the rest with random particles
        for i in range(n_particles // 2, n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
    else:
        # Initialize completely random particles
        for _ in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.zeros(n_points))
    
    # Track best positions
    pbest = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * n_particles
    gbest = None
    gbest_fitness = float('inf')
    
    # PSO parameters
    omega = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            fitness = evaluate_c5(h, dx)
            
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = h.copy()
                
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest = h.copy()
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.uniform(0, 1, n_points)
            r2 = np.random.uniform(0, 1, n_points)
            
            cognitive = c1 * r1 * (pbest[i] - h)
            social = c2 * r2 * (gbest - h)
            
            v = omega * v + cognitive + social
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Apply mutation with small probability
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.03, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
        # Early exit
        if gbest_fitness <= 0.3807:
            break
            
        # Adjust inertia weight over time
        if iteration > 100:
            omega = 0.6
    
    return (gbest.tolist(), gbest_fitness, n_points)
