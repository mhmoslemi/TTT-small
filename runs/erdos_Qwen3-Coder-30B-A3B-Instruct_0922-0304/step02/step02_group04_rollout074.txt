import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_to_feasible(h_plus, n_points)
        h_minus = project_to_feasible(h_minus, n_points)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def solve_quadratic_subproblem(H, g, n_points, dx):

    # More realistic: use quadratic programming or active set
    # For this case we'll project to box and sum constraints
    d = -np.dot(np.linalg.inv(H), g)
    
    # Project to feasible bounds
    d = np.clip(d, -h, 1 - h)
    
    # Ensure sum constraint
    d_sum = np.sum(d)
    if abs(d_sum) > 1e-6:
        d -= d_sum / n_points
    
    return d

def line_search(h, d, dx, max_alpha=1.0):
    alpha = max_alpha
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        c5_old = evaluate_c5(h, dx)
        
        if c5_new < c5_old:
            return alpha
        alpha *= 0.5
    return alpha

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    tol = 1e-3
    hessian_tol = 1e-6
    
    # Initialize starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    # Initialize Hessian approximation (identity)
    H = np.eye(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_val = evaluate_c5(h, dx)
        
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Check termination
        if np.linalg.norm(grad) < tol:
            break
            
        # Solve quadratic subproblem (approximate)
        d = -np.dot(np.linalg.inv(H), grad)
        
        # Ensure sum constraint
        d_sum = np.sum(d)
        d -= d_sum / n_points
        
        # Satisfy bounds
        for i in range(n_points):
            if d[i] > 0 and h[i] + d[i] > 1:
                d[i] = 1 - h[i]
            elif d[i] < 0 and h[i] + d[i] < 0:
                d[i] = -h[i]
        
        # Line search
        alpha = line_search(h, d, dx)
        h = h + alpha * d
        h = project_to_feasible(h, n_points)
        
        # Update Hessian approximation using BFGS
        if iteration > 0 and iteration < max_iter - 1:
            y = compute_gradient(h, dx) - grad
            if np.linalg.norm(y) > hessian_tol:
                rho = 1.0 / np.dot(y, d)
                A = np.eye(n_points) - rho * np.outer(d, y)
                B = np.eye(n_points) - rho * np.outer(y, d)
                H = np.dot(np.dot(A, H), B) + rho * np.outer(d, d)
    
    return h, evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
