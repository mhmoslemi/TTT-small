import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Symmetric initialization: 5-fold pattern
        h = np.ones(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = (i+1) * segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.6
            else:
                h[start_idx:end_idx] = 0.4
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define objective function and constraints
    def objective(h_flat):
        # Reshape h_flat to proper shape and ensure it sums to n_points/2
        h = h_flat.copy()
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint: sum of h should be n_points/2
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints dictionary for scipy.optimize.minimize
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use L-BFGS-B with bounds and constraints
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        if iter_count > 0:
            # Randomly perturb h to escape local minima
            perturbation = np.random.normal(0, 0.02, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Minimize using L-BFGS-B with bounds and constraints
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options=options,
            callback=lambda x: None
        )
        
        if result.success:
            h = result.x * (n_points / 2) / np.sum(result.x)
            h = np.clip(h, 0, 1)
        else:
            # If L-BFGS-B failed, use a simpler approach
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
