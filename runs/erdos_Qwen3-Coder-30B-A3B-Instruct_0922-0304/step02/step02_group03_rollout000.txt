import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Coarse grid for fast exploration
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with piecewise constant approximation
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    iteration = 0
    max_iter_stage1 = 500
    
    # Stage 1: Optimize coarse grid
    for iteration in range(max_iter_stage1):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Objective function for optimization
        def obj(h_flat):
            h = h_flat.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_c5(h, dx)
        
        # Gradient calculation using finite differences
        def gradient(h_flat):
            eps = 1e-5
            grad = np.zeros_like(h_flat)
            base_c5 = obj(h_flat)
            for i in range(len(h_flat)):
                h_plus = h_flat.copy()
                h_plus[i] += eps
                grad[i] = (obj(h_plus) - base_c5) / eps
            return grad
        
        # Perform optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, 
                         jac=gradient, options={'maxiter': 1, 'ftol': 1e-8})
        
        if result.success:
            h = result.x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

            pert = np.random.normal(0, 0.01, n_points)
            h = h + pert

            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break

    n_points = 70
    dx = 2.0 / n_points
    
    h = best_h.copy()

                           np.linspace(0, 2, len(best_h)), 
                           best_h)
    h = h_upsampled.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Add small random perturbations
    perturb = np.random.normal(0, 0.01, n_points)
    h = h + perturb
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Refinement parameters
    max_iter = 1000
    iteration = 0

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        def obj(h_flat):
            h = h_flat.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_c5(h, dx)
        
        # Gradient calculation using finite differences
        def gradient(h_flat):
            eps = 1e-5

            for i in range(len(h_flat)):

                h_plus[i] += eps
                grad[i] = (obj(h_plus) - base_c5) / eps
            return grad

        bounds = [(0, 1) for _ in range(n_points)]

            h = np.clip(h, 0, 1)
