import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Sawtooth initialization
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    no_improvement_count = 0
    last_best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap to determine k_max
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # For the convex subproblem, we need to look at shift k_max
        # The overlap at shift k_max is sum_{i=0}^{n-1} h[i] * (1 - h[(i+k_max)%n])
        # Which linearizes to sum_{i=0}^{n-1} h[i] - sum_{i=0}^{n-1} h[i] * h[(i+k_max)%n]
        
        # Construct the quadratic term in an efficient way
        # The overlap at shift k_max is sum_i h[i] * (1 - h[(i+k_max)%n])
        # i.e., sum_i h[i] - sum_i h[i]*h[(i+k_max)%n]
        
        # Linearize objective for convex subproblem: minimize sum_i h[i] - sum_i h[i]*h[(i+k_max)%n]
        # We want to minimize the overlap at shift k_max
        # We use variable h_var for the new h
        
        # Set up the convex subproblem: min h^T * (1 - H_k*h) s.t. sum h = n/2, h in [0,1]
        # This is actually non-convex in variable h itself, but we can treat it as a linearization
        # by fixing the current h values for the cross terms, and doing iterative update
        
        # So we actually solve:
        # minimize sum_i h[i] - sum_i h[i] * h_prev[(i+k_max)%n] 
        # where h_prev is current h
        
        # The first term is just sum(h) which is constant if we maintain sum(h) = n/2 
        # Fixed term: sum_i h_prev[(i+k_max)%n] * h[i] 
        # In quadratic form: h[i] * h_prev[(i+k_max)%n] - we keep h_prev fixed and optimize h
        
        # Construct the linearized subproblem 
        # This is a *convex* subproblem where we minimize a linear function
        # The cross-term is constant and thus linear in h
        # We construct a linear objective: -sum_i h[i] * h_prev[(i+k_max)%n]
        # So minimize: sum_i (1 - h_prev[(i+k_max)%n]) * h[i] 
        # Subject to sum(h) = n/2, h[i] in [0,1]
        
        # Objective
        # We minimize sum_i (1 - h_prev[(i+k_max)%n]) * h[i] 
        # Which gives us: sum_i (1 - h_prev[(i+k_max)%n]) * h[i] 
        # linear in h
        
        # Convex optimization subproblem
        h_var = Variable(n_points)
        
        # Compute the linear coefficient (1 - h_prev[(i+k_max)%n])
        linear_coeff = 1 - h[np.roll(np.arange(n_points), k_max)]
        # Efficient formulation using matrix operation (may be impractical for large n;
        # We directly handle it in constraint form)
        
        # Vertically stack constraints:
        # linear_coeff @ h_var <= C, sum(h_var) = n/2, h_var[i] in [0,1]
        # Minimize linear_coeff @ h_var, which gives min sum_i (1 - h_prev[(i+k_max)%n]) * h[i]
        # This is convex in h_var anywhere
        
        # Reformulation: minimize -h_var^T @ (h_prev[(i+k_max)%n] - 1)
        # But we already know the linear part coeff; just compute:
        
        # Let row[key] = 1 - h_prev[(i+k_max)%n] for each i
        row = 1 - h[np.roll(np.arange(n_points), k_max)]
        
        # Define the linear objective, i.e. minimize row^T * h_var
        obj = cp_sum(cp.multiply(row, h_var))
        
        # Constraints
        constraints = [
            cp_sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve subproblem
        try:
            prob = Problem(Minimize(obj), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status == 'optimal':
                h_new = h_var.value
                h = np.array(h_new)  
                # Clip to [0,1] just in case
                h = np.clip(h, 0, 1)
                # Rescale to maintain sum(n/2)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to a simple update if CVXPY fails
                h = np.clip(h + 0.01 * (np.random.random(n_points) - 0.5), 0, 1)  
                h = h * (n_points / 2) / np.sum(h)
        except Exception as e:
            # If problem solving fails, just do a small random step
            h = np.clip(h + 0.01 * (np.random.random(n_points) - 0.5), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Re-evaluate the current solution for possible improvement
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Keep track of the best solution so far
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                break  # early stop if no improvement  

        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
