import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_erdos_solution(h_values):
    # Computes C5 for given h_values
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # Projects h onto feasible region
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def line_search(h, grad, current_c5, n_points, dx, alpha_max=1e-1):
    # Performs line search to find best step size
    alphas = [1e-3, 2e-3, 5e-3, 1e-2, 2e-2, 5e-2, 1e-1]
    for alpha in alphas:
        h_candidate = h - alpha * grad
        h_candidate = project_feasible(h_candidate, n_points, dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < current_c5:
            return h_candidate, candidate_c5, alpha
    return h, current_c5, 0.0

def constrained_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Solve relaxed convex problem using CVXPY
    h_cvxpy = cp.Variable(n_points)
    h_init = np.array(h)
    
    # Objective: minimize max overlap plus regularization
    # We'll minimize just the overlap term at current shift
    # This is a lazy simplification, but focuses on structure
    
    # Simply try to make h uniform via entropy regularization
    objective = cp.Minimize(
        cp.sum(cp.multiply(h_cvxpy, cp.log(h_cvxpy + 1e-12))) + 
        lambda_reg * cp.sum_squares(h_cvxpy - 0.5)
    )
    
    constraints = [
        cp.sum(h_cvxpy) == n_points / 2,
        h_cvxpy >= 0,
        h_cvxpy <= 1
    ]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            sol = h_cvxpy.value
            return project_feasible(sol, n_points, dx)
    except:
        pass
    return h
