import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate a simple initial solution
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Convex relaxation parameter
    lambda_reg = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Find the shift that gives maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient approximation
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search with step sizes
        step_sizes = [0.1, 0.5, 1.0]
        best_step = 0.0
        best_h_candidate = h.copy()
        
        for alpha in step_sizes:
            h_candidate = project_feasible(h - alpha * gradient, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < current_c5:
                best_h_candidate = h_candidate
                best_step = alpha
                break
                
        if best_step > 0:
            h = best_h_candidate
        else:
            # If no step improved it, try small random perturbations
            perturbation = np.random.normal(0, 0.01, n_points)
            h_candidate = project_feasible(h + perturbation, n_points, dx)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < current_c5:
                h = h_candidate
        
        # Periodically solve a convex relaxation to escape local minima
        if iteration % 20 == 0:
            try:
                # Solve convex relaxation using CVXPY
                h_relax = cp.Variable(n_points)
                # Regularized objective: overlap term + entropy regularization
                # We approximate the overlap maximization with a linear term
                # and add L1 penalty to encourage sparsity
                
                # Approximate objective (simplified version)
                obj = cp.Minimize(
                    cp.sum(cp.multiply(h_relax, 1 - h_relax)) + 
                    lambda_reg * cp.sum(cp.entr(h_relax) + cp.entr(1 - h_relax))
                )
                
                # Constraints
                constraints = [
                    cp.sum(h_relax) == n_points / 2,
                    h_relax >= 0,
                    h_relax <= 1
                ]
                
                # Solve problem
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_new = np.array(h_relax.value)
                    h_new = project_feasible(h_new, n_points, dx)
                    c5_new = evaluate_c5(h_new, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                    else:
                        # Reduce regularization if no improvement
                        lambda_reg *= 0.5
                else:
                    lambda_reg *= 0.5
                    
            except Exception:
                lambda_reg *= 0.5
    
    return (best_h.tolist(), best_c5, n_points)
