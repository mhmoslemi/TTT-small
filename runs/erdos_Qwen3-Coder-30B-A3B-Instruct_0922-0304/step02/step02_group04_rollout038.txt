import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, norm, sum as cp_sum

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_relaxation(h_initial, n_points, dx):

    h = Variable(n_points)

    # Objective: minimize C
    objective = Minimize(C)
    
    # Constraints
    constraints = [0 <= h, h <= 1, cp_sum(h) == n_points / 2, C >= 0]
    
    # Add overlap constraints for each k
    for k in range(2 * n_points - 1):

        # constraints would be linearized differently
        pass
    
    # Relaxation - directly use initial solution as a heuristic
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT,(verbose=False)
    except:
        # Fallback to a simpler heuristic
        pass
    
    h_solution = np.array(h.value).flatten() if h.value is not None else h_initial
    h_solution = project_to_feasible(h_solution, n_points)
    
    return h_solution

def local_search(h_start, n_points, dx, max_iter=50):
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current, dx)
    
    step_size = 0.01
    for _ in range(max_iter):
        corr = correlate(h_current, 1 - h_current, mode="full")
        c5_val = np.max(corr) * dx
        
        # Simple gradient ascent approach
        grad = np.zeros_like(h_current)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_val) / eps
        
        # Update with gradient
        h_current = h_current - step_size * grad
        h_current = project_to_feasible(h_current, n_points)
        
        c5_new = evaluate_c5(h_current, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_current.copy()
    
    return best_h, best_c5
