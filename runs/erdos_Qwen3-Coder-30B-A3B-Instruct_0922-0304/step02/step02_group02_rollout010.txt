import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using convex relaxation with homotopy
    # Schedule for relaxation parameter mu from 0.5 to 1.0 in 50 steps
    mu_values = np.logspace(np.log10(0.5), 0, 50, endpoint=True)
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define CVXPY problem
        h_var = cp.Variable(n_points)
        
        # Constraint: sum = 35
        constraints = [cp.sum(h_var) == 35]
        
        # Constraint: bounds
        constraints += [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Objective function
        # We approximate C5 by evaluating at some sampled shifts
        # But since C5 is the max over all shifts, we'll compute it exactly
        # after solving to get a good estimate for the final check
        
        # For safety, we optimize the weighted sum of smooth terms
        # plus small smoothing in the mean
        objective = cp.Minimize(
            mu * cp.sum(cp.multiply(h_var, 1 - h_var)) + 
            (1 - mu) * cp.sum(cp.power(h_var - 0.5, 2))
        )
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Ensure the sum is correct and bounded
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                current_c5 = evaluate_c5(h_new, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
                    
                h = h_new
            else:
                # If optimization fails, we fall back to previous best
                pass
        except Exception:
            # If optimization fails, fall back to previous best
            pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
