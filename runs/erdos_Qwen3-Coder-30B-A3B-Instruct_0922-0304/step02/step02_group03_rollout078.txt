import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Phase 1: Coarse optimization (n=35) for fast exploration
    if n_points >= 35:
        n_coarse = 35
        dx_coarse = 2.0 / n_coarse
        h_coarse = np.zeros(n_coarse)
        
        # Initial guess: simple two-block construction
        half_coarse = n_coarse // 2
        h_coarse[:half_coarse] = 1.0
        
        # Rescale to meet integral constraint
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        
        # Optimization with L-BFGS-B for coarse solution
        def objective_coarse(h_coarse):
            h_full = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_coarse), h_coarse)
            h_full = np.clip(h_full, 0, 1)
            return evaluate_c5(h_full, dx_coarse)
        
        def constraint_coarse(h_coarse):
            return np.sum(h_coarse) - n_coarse / 2
        
        bounds_coarse = [(0, 1) for _ in range(n_coarse)]
        constraints_coarse = {'type': 'eq', 'fun': constraint_coarse}
        
        try:
            result_coarse = minimize(objective_coarse, h_coarse, method='L-BFGS-B', 
                                   bounds=bounds_coarse, constraints=constraints_coarse,
                                   options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-5})
            if result_coarse.success:
                h_opt_coarse = np.clip(result_coarse.x, 0, 1)
                h_opt_coarse = h_opt_coarse * (n_coarse / 2) / np.sum(h_opt_coarse)
                # Upsample to full resolution
                h_full = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_opt_coarse)
                h = np.clip(h_full, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
    
    # Phase 2: Fine optimization
    dx = 2.0 / n_points
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    # Add random perturbations to escape local minima
    for i in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_perturbed = h.copy()
        perturb = np.random.normal(0, 0.02, n_points)
        h_perturbed += perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            result = minimize(objective, h_perturbed, method='L-BFGS-B', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-5})
            if result.success:
                h_test = np.clip(result.x, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                c5_test = evaluate_c5(h_test, dx)
                if c5_test < best_c5:
                    best_c5 = c5_test  
                    best_h = h_test.copy()
        except:
            continue
    
    return (best_h.tolist(), best_c5, n_points)
