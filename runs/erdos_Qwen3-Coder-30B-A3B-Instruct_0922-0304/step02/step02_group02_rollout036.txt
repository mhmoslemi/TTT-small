```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Homotopy continuation approach
    mu_values = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define convex optimization problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(
            mu * cp.max(cp.conv(weights=h_var, kernel=1-h_var, mode='full')) * dx + 
            (1 - mu) * cp.sum_squares(h_var - 0.5) / n_points
        )
        
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(h_var.value).flatten()
                # Project back to feasible region
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            # If optimization fails, fallback to previous
            pass
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
