import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate the C5 value using correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles for PSO
    n_particles = 30
    particles = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_seed = np.array(initial_h_values)
        h_seed = np.clip(h_seed, 0, 1)
        h_seed = h_seed * (n_points / 2) / np.sum(h_seed)
    else:
        # Default initialization
        h_seed = np.ones(n_points)
        h_seed[n_points//2:] = 0
        h_seed += np.random.normal(0, 0.05, n_points)
        h_seed = np.clip(h_seed, 0, 1)
        h_seed = h_seed * (n_points / 2) / np.sum(h_seed)
    
    # Initialize particles with blocky structures + noise
    for i in range(n_particles):
        if i == 0:
            # First particle is initialized with the seed
            h = h_seed.copy()
        else:
            # Other particles with simple block patterns
            h = np.zeros(n_points)
            # Two blocks: 1's in first half, 0's in second half, then random noise
            h[:n_points//2] = 1.0
            noise = np.random.normal(0, 0.1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        particles.append(h)
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    inertia = 0.7
    max_velocity = 0.3
    inertia_decay = 1e-4
    
    # Track best positions
    best_positions = [p.copy() for p in particles]
    best_fitness = [evaluate_c5(p, dx) for p in particles]
    
    global_best_idx = np.argmin(best_fitness)
    global_best = best_positions[global_best_idx].copy()
    global_best_fitness = best_fitness[global_best_idx]
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    iteration = 0
    max_iter = 1000
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update inertia
        inertia = max(0.1, inertia * (1 - inertia_decay))
        
        for i in range(n_particles):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Update particle velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             alpha * r1 * (best_positions[i] - particles[i]) +
                             beta * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            new_pos = particles[i] + velocities[i]
            
            # Project to feasible set
            new_pos = project_to_feasible(new_pos, n_points)
            
            # Evaluate new solution
            fitness = evaluate_c5(new_pos, dx)
            
            # Update personal best
            if fitness < best_fitness[i]:
                best_fitness[i] = fitness
                best_positions[i] = new_pos.copy()
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = new_pos.copy()
                    global_best_fitness = fitness
            
            # Update particle position
            particles[i] = new_pos.copy()
            
            # Early stopping
            if global_best_fitness <= 0.3807:
                break
    
    return (global_best.tolist(), global_best_fitness, n_points)
