import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum=h, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    """Compute gradient using finite differences"""
    n_points = len(h)
    grad = np.zeros(n_points)
    h_val = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_to_feasible(h_plus, n_points)
        h_minus = project_to_feasible(h_minus, n_points)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def line_search(h, d, dx, max_alpha=1.0, c1=1e-4):
    """Armijo backtracking line search"""
    alpha = max_alpha
    h_val = evaluate_c5(h, dx)
    
    while alpha > 1e-8:
        h_new = project_to_feasible(h + alpha * d, len(h))
        h_new_val = evaluate_c5(h_new, dx)
        
        if h_new_val <= h_val - c1 * alpha * np.dot(d, compute_gradient(h, dx)) or alpha < 1e-8:
            return alpha
        alpha *= 0.5
    
    return alpha

def solve_qp_subproblem(H, g, n_points, bounds):
    """Solve quadratic subproblem with bounds and constraint"""
    from scipy.optimize import minimize
    
    def obj(d):
        return 0.5 * np.dot(d, H @ d) + np.dot(g, d)
    
    constraints = [{'type': 'eq', 'fun': lambda d: np.sum(d)}]
    bounds = [(max(-bounds[i], -1.0), min(1.0 - bounds[i], 1.0)) 
              for i in range(n_points)]
    
    result = minimize(obj, np.zeros(n_points), method='L-BFGS-B', 
                     bounds=bounds, constraints=constraints, tol=1e-8)
    
    return result.x

def sequential_quadratic_programming(seed, budget_s):
    """Main SQP algorithm for minimizing C5"""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    H = np.eye(n_points)  # Initial Hessian approximation (Identity)
    max_iterations = 1000
    tol = 1e-3
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Compute bounds for line search
        bounds = h
        
        # Solve quadratic subproblem
        d = solve_qp_subproblem(H, g, n_points, bounds)
        
        # Line search for step size
        alpha = line_search(h, d, dx)
        
        # Update h
        h_new = project_to_feasible(h + alpha * d, n_points)
        
        # Update Hessian using BFGS
        y = compute_gradient(h_new, dx) - g
        s = alpha * d
        if np.dot(s, y) > 1e-10:
            rho = 1.0 / np.dot(s, y)
            I = np.eye(n_points)
            H = (I - rho * np.outer(s, y)) @ H @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
        
        h = h_new
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
