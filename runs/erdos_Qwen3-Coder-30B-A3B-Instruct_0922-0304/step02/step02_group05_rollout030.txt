import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # Solve min (overlap_k + lambda * sum(h * log(h)))
    # using CVXPY with constraints h in [0,1], sum(h) = n_points/2
    
    h_var = cp.Variable(n_points)
    
    # Build the overlap term for the current h
    # We take the overlap at the position of maximum overlap
    old_overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(old_overlap)
    
    # Calculate overlap_k = sum h[i] * (1 - h[(i+k) % n_points]) 
    # We'll need to build this term carefully in CVXPY
    
    # For convex relaxation, we use a simple formulation:
    # Minimize max overlap plus regularization
    
    # General approach: estimate the maximizing lag and 
    # do an unconstrained convex relaxation in auxiliary variables
    
    objective = cp.Minimize(cp.sum(h_var * (1 - h_var)) 
                            + lambda_reg * cp.sum(cp.entr(h_var)))
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points/2
    ]
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        new_h = h_var.value
        return new_h
    except:
        return h  # fallback to old if convex relaxation fails

def evaluate_gradient(h, dx, k_shift):
    # This is a simple finite difference approximation to the gradient
    eps = 1e-5
    grad = np.zeros_like(h)
    h0 = h.copy()
    h0 = project_feasible(h0, len(h0), dx)
    
    for i in range(len(h0)):
        h_plus = h0.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, len(h_plus), dx)
        
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        c5_plus = np.max(overlap_plus) * dx
        
        overlap_0 = correlate(h0, 1 - h0, mode="full")
        c5_0 = np.max(overlap_0) * dx
        
        grad[i] = (c5_plus - c5_0)/eps
    
    return grad

def line_search(h, grad, dx, max_iter=3):
    # If we want to take a step, do a simple line search
    # Testing alpha in [1e-3, 1e-2, 1e-1]
    alphas = [1e-3, 1e-2, 1e-1]
    h0 = h.copy()
    c5_0 = evaluate_c5(h0, dx)
    for alpha in alphas:
        h_test = h - alpha * grad
        h_test = project_feasible(h_test, len(h_test), dx)
        c5_test = evaluate_c5(h_test, dx)
        if c5_test < c5_0:
            return h_test, c5_test
    return h0, c5_0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values[:n_points], dtype=float)
        # Rescale to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    iter_count = 0
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute overlaps
        overlap_s = correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap_s) * dx
        max_k = np.argmax(overlap_s)
        
        # Early exit if target reached
        if c5 <= 0.3807:
            break
            
        # Compute approximate gradient
        grad = evaluate_gradient(h, dx, max_k)
        
        # Perform line search
        h_new, new_c5 = line_search(h, grad, dx)
        
        # Update if improvement
        if new_c5 < c5:
            h = h_new
        else:
            h = project_feasible(h, n_points, dx)  # ensure feasibility
            
        # Periodic convex relaxation
        if iter_count % 50 == 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx, lambda_reg)
            c5_relaxed = evaluate_c5(h_relaxed, dx)
            if c5_relaxed < c5:
                h = h_relaxed
                lambda_reg *= 0.5
            else:
                lambda_reg *= 0.5
                
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Ensure final feasibility
    best_h = project_feasible(best_h, n_points, dx)
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
