import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create 5-fold symmetric initialization
        h = np.zeros(n_points)
        segment_length = n_points // 5
        values = [0.7, 0.3, 0.7, 0.3, 0.7]
        for i in range(5):
            start_idx = i * segment_length
            end_idx = start_idx + segment_length
            h[start_idx:end_idx] = values[i]
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 1000
    tol = 1e-6
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def feasibility_repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Use scipy.optimize.minimize with L-BFGS-B
    constraints = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using scipy
        result = minimize(objective, h.copy(), method='L-BFGS-B', bounds=bounds, 
                         constraints=constraints, tol=tol, options={'maxiter': 10})
        
        if result.success:
            h = result.x
            h = feasibility_repair(h)
        else:
            # If optimization fails, use the last h and perform local search
            pass
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
