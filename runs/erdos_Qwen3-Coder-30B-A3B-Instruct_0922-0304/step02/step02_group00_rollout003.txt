import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative majorization-minimization approach
    max_iter = 1000
    patience = 0
    last_improvement = 0
    
    # Track how many iterations we did
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            last_improvement = iteration
        else:
            patience += 1
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # If no improvement for too long, exit
        if patience > 100:
            break
            
        # Formulate convex subproblem for this k_max
        # Objective: minimize correlation at k_max
        h_var = Variable(n_points)
        
        # Linearize the constraint: correlation at k_max
        # This is sum over i of h[i] * (1 - h[(i+k_max) % n_points])
        # Linear in h, so we can minimize it directly
        # Yi = h[i], Zi = h[(i+k_max) % n_points]
        # We want to minimize sum_i (Yi * (1 - Zi)) = sum_i (Yi - Yi*Zi)
        # But since h_var are decision variables, not Zi, we compute:
        # For each i, we know h[i] but we want to optimize them.
        # Let's compute the contribution to the correlation at k_max directly.
        
        # Construct the linear term using k_max as the shift
        # Correlation at shift k_max = sum_i h[i] * (1 - h[(i + k_max) % n_points])
        # Let's define the constraints:
        # h_var[i] in [0,1], sum(h_var) == n_points / 2 (but let it be 1.0 and scale later) 
        
        # Formulate linear objective to minimize at k_max
        # We create the objective directly
        objective_terms = []
        for i in range(n_points):
            j = (i + k_max) % n_points
            objective_terms.append(h_var[i] * (1 - h_var[j]))
        
        # Minimize the sum of the correlation terms at k_max
        objective = cp.sum(objective_terms) * dx
        
        # Constraints
        constraints = [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        constraints.append(cp_sum(h_var) == n_points / 2)
        
        # Solve subproblem
        prob = Problem(Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                # Ensure feasibility via projection and rescaling
                h = np.clip(new_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Revert to previous solution if solver fails
                h = best_h.copy()
        except:
            # In case of solver error, skip update
            pass

        iteration += 1

    return (best_h.tolist(), best_c5, n_points)
