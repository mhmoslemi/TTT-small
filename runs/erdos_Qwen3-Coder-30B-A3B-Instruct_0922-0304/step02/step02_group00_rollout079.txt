import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe inspired approach
    max_iter = 1000
    no_improvement_iter = 0
    patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_iter = 0
        else:
            no_improvement_iter += 1
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_iter >= patience:
            break
            
        # Formulate and solve convex subproblem to minimize the k_max overlap
        h_var = Variable(n_points)
        
        # Build the linear objective for the k_max overlap term
        # Sum h[i] * (1 - h[i + k_max]) for valid indices
        obj = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            obj += h_var[i] * (1 - h_var[j])
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Create problem
        prob = Problem(Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                h = h_new
            else:
                # fallback to gradient-based approach
                step = 0.01
                iterations = 50
                h_temp = h.copy()
                for _ in range(iterations):
                    corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_temp = np.max(corr_temp) * dx
                    grad = np.zeros_like(h_temp)
                    eps = 1e-5
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_temp) / eps
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # fallback to gradient-based approach
            step = 0.01
            iterations = 50
            h_temp = h.copy()
            for _ in range(iterations):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                grad = np.zeros_like(h_temp)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
            
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    return (best_h.tolist(), best_c5, n_points)
