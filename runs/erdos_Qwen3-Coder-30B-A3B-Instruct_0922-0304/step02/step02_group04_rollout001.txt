import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) / 2
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    eps = 1e-4
    grad = np.zeros_like(h)
    h_base = h.copy()
    c5_base = evaluate_c5(h_base, dx)
    
    for i in range(len(h)):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Personal best positions and fitness
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_fitness = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Global best
    best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[best_idx].copy()
    global_best_fitness = personal_best_fitness[best_idx]
    
    # PSO iteration
    for iteration in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocities
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_best[i] - swarm[i]) +
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = swarm[i].copy()
                    global_best_fitness = fitness
        
        # Occasionally perform local optimization on top particles
        if iteration % 20 == 0 and iteration > 0:
            top_indices = np.argsort(personal_best_fitness)[:swarm_size//5]
            for i in top_indices:
                h = personal_best[i].copy()
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h, dx)
                    h -= step_size * grad
                    h = project_to_feasible(h, n_points)
                personal_best[i] = h
                personal_best_fitness[i] = evaluate_c5(h, dx)
                
                # Update global best if we found a better one
                if personal_best_fitness[i] < global_best_fitness:
                    global_best = h.copy()
                    global_best_fitness = personal_best_fitness[i]
        
        # Early exit if target reached
        if global_best_fitness <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_fitness, n_points)
