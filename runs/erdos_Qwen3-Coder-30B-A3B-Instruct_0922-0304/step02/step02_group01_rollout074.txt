import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    momentum = 0.9
    step_size = 0.01
    last_improvement = 0
    iter_no_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Stop if no improvement for a while
        if iter_no_improvement > 100:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        max_shift = max_idx - (n_points - 1)  # Convert index to shift
        
        # Compute gradient based on max shift
        grad = np.zeros_like(h)
        
        # Compute gradient for the most constraining shift
        if max_shift >= 0:
            # h[i] * (1 - h[i + max_shift]) for i = 0..n-max_shift-1
            for i in range(n_points - max_shift):
                grad[i] += (1 - h[i + max_shift])
            # -h[i] for i = max_shift..n-1 (for the second term of the product)
            for i in range(max_shift, n_points):
                grad[i] -= h[i - max_shift]
        else:
            # max_shift < 0, so similar but shifted
            abs_shift = -max_shift
            for i in range(n_points - abs_shift):
                grad[i] -= h[i + abs_shift]
            for i in range(abs_shift, n_points):
                grad[i] += (1 - h[i - abs_shift])
        
        # Update using momentum and adaptive step size
        if iteration == 0:
            velocity = -grad
        else:
            velocity = momentum * velocity - step_size * grad
            
        h = h + velocity
        h = project_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
