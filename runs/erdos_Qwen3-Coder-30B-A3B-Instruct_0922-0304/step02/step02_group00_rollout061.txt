import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    epsilon = 0.05
    T_initial = 1.0
    beta = 0.995
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    T = T_initial
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        
        # Generate candidate
        h_candidate = h + epsilon * T * np.random.normal(0, 1, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new overlap
        new_c5 = evaluate_c5(h_candidate, dx)
        delta_c5 = new_c5 - current_c5
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Cooling
        if iteration % 100 == 0:
            T = beta * T
            
        # Reset temperature if no improvement
        if last_improvement > 1000:
            T = T_initial
            last_improvement = 0
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
