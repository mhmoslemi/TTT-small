import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Sawtooth-like initialization
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    no_improve_count = 0
    min_improve = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlations
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        if no_improve_count > 100:
            break
            
        # Formulate convex subproblem
        h_var = Variable(n_points)
        
        # Objective: minimize overlap at k_max
        # The overlap at shift k_max is sum h[i] * (1 - h[i + k_max])
        # But we need to consider the circular nature of correlation
        objective = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            objective += h_var[i] * (1 - h_var[j])
        
        objective = objective * dx
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve convex subproblem
        prob = Problem(Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_var.value).reshape(-1)
                # Ensure feasibility
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If LPO fails, try gradient descent to escape local minima
                step = 0.01
                h_temp = h.copy()
                for _ in range(100):
                    corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_temp = np.max(corr_temp) * dx
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_temp) / eps
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback to naive gradient descent
            step = 0.01
            h_temp = h.copy()
            for _ in range(100):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp

    return (best_h.tolist(), best_c5, n_points)
