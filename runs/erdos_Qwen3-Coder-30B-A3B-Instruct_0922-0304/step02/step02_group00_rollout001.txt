import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Track last improvement
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Formulate convex subproblem
        h_var = cp.Variable(n_points)
        
        # Objective: minimize overlap at k_max using linear approximation
        # The overlap at lag k_max is: \sum_i h[i] * (1 - h[i + k_max]) 
        # Approximate this linearly in terms of h_var
        obj_terms = []
        for i in range(n_points):
            j = (i + k_max) % n_points
            # Approximate h_var[i] * (1 - h_var[j]) with linear term
            # This is a simplified linearization, we're minimizing the correlation directly
            obj_terms.append(h_var[i] * (1 - h[j]))
            
        objective = cp.Minimize(cp.sum(obj_terms))
            
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Simple gradient step if solver fails
                step = 0.01
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    corr_current = correlate(h, 1 - h, mode="full")
                    c5_current = np.max(corr_current) * dx
                    grad[i] = (c5_plus - c5_current) / eps
                h -= step * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback on simple update
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
        iteration += 1
        
        # Stop if no improvement for long
        if iteration - last_improvement > 100:
            break
            
    # Final evaluation
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode="full")
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Return the last computed values
        corr = correlate(h, 1 - h, mode="full") 
        current_c5 = np.max(corr) * dx
        return (h.tolist(), current_c5, n_points)
