import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Two-stage strategy: coarse then fine
    n_points_list = [35, 70]
    best_c5 = float('inf')
    best_h = None
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if 'initial_h_values' in globals() and n_points == 70:
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            # Coarse initialization
            h = np.ones(n_points)
            # Split into two parts with different values
            split_point = n_points // 2
            h[:split_point] = 1.0
            h[split_point:] = 0.0
            
        # Ensure correct integral
        h = h * (n_points / 2) / np.sum(h)
        
        # Bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the objective function for minimization
        def objective(h_vals):
            h_vals = np.array(h_vals)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            return evaluate_c5(h_vals, dx)
        
        # Callback to save best solution
        def callback(h_vals):
            nonlocal best_c5, best_h
            h_vals = np.array(h_vals)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            c5 = evaluate_c5(h_vals, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals.copy()
        
        # Optimize with L-BFGS-B
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            callback=callback,
            options={'maxiter': 500 if n_points == 35 else 1000, 'ftol': 1e-9, 'gtol': 1e-5}
        )
        
        # Final check
        h_final = np.array(result.x)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        final_c5 = evaluate_c5(h_final, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final.copy()
    
    # If no improvement, fallback to best found
    if best_h is None:
        best_h = np.ones(n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, len(best_h))
