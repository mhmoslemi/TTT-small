import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    # Initial feasibility check
    h = enforce_feasibility(h, n_points)
    
    # Objective function for optimization
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Constraint function
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    res = minimize(
        objective, 
        h, 
        method='L-BFGS-B', 
        bounds=bounds,

    )

    best_h = enforce_feasibility(res.x, n_points)

    time_remaining = budget_s - (time.time() - start_time)

        # Initialize a new heuristic 
        h_alt = np.zeros(n_points)
