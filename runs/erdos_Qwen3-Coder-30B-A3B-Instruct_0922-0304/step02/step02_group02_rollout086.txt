import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Relaxation parameters
    mu_schedule = np.logspace(np.log10(0.5), 0.0, 50)  # 50 steps from 0.5 to 1.0
    
    for mu in mu_schedule:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Create optimization variables
        h_vars = cp.Variable(n_points)
        
        # Objective: weighted combination of C5 and regularization
        # Regularizer: (h_i - 0.5)^2 to encourage balanced values
        regularizer = cp.sum(cp.square(h_vars - 0.5)) / n_points
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Approximate C5 using a conservative upper bound in the problem
        # We focus on convex relaxation instead of exact computation
        # This uses a smooth approximation to the max operation (sum of all overlaps weighted)
        # This will be solved approximately but gives us a descent direction
        objective = cp.Minimize(mu * regularizer + (1 - mu) * cp.sum(h_vars * (1 - h_vars)))
        
        # Solve the convex problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                h_new = np.clip(h_new, 0, 1)
                # Rescale to maintain integral constraint (sum = n_points / 2)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If optimization fails, keep the old h
                pass
        except:
            pass  # In case of solver failure, continue with current h
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
