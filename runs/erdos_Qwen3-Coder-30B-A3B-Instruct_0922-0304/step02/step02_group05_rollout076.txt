import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_shift, dx):
    # Compute gradient of overlap_k with respect to h
    # For simplicity, we will compute approximate gradient 
    # by finite difference around the current maximum overlap
    n = len(h)
    grad = np.zeros(n)
    # Explicitly compute: d/dh_i overlap_k = (1 - h[i+k]) - h[i+k]
    # But only for the shift that gave max overlap...
    # Let's use a more precise approach: we compute the derivative at the 
    # position that maximizes overlap
    # reconstruct the overlap_k part: 
    # overlap_k = sum_i h[i] * (1 - h[i+k]) 
    # d/dh_i overlap_k = 1 - h[i+k] if i is in valid range
    # This is actually the kernel: (1 - h) convolved with h (but flipped for indexing)
    # boundary handling in correlate matters!
    # The key form: specific overlap_k = sum_i h[i] * (1 - h[i+k]) where k is shift.
    # We look at position k_max where this is maximized
    # Gradient w.r.t. h[j] is: 1 - h[j+k_max] and -h[j+k_max] when i==j 
    # no wait i want to compute gradient of overlap_k w.r.t h_i for a fixed k
    # so for h_i we get gradient as d/dh_i sum_j h[j]*(1-h[j+k]) = 1-h[i+k] 
    # if we are shifting h by k positions we shift the kernel 1-h backward
    # so the derivative of overlap_k = sum_h[i] * (1 - h[i+k]) is:
    # grad[i] = 1 - h[i+k] or -1 * h[i+k] if we want derivative of index i in h
    # But this is not quite right because when computing derivative of sum, we only 
    # want contributions of h[i] to the total overlap
    # Better:
    # overlap_k = sum_i h[i] * (1 - h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]
    # d/dh_j (overlap_k) = δ_{j,i} - (δ_{i,j} * h[i+k] + h[j] * δ_{i+k,j})
    # = 1 δ_{j,i} - h[i+k] δ_{j,i} (the second term means when it contributes to the two-ary product)
    # = (1-h[i+k]) δ_{j,i}  when i = j-k (but need to account for k shift)
    # if i = j+k, then then gradient of overlap_k w.r.t h[j] is just (1-h[i]) = (1-h[j+k])
    # Let's sort out if i is from fixed k or from some position j:
    # OVERLAP_k = sum_i h[i] * (1 - h[i+k]) 
    # dq/dh_j = sum_i (d/dh_j)[h[i] * (1 - h[i+k])] = sum_i δ{j,i} * (1 -h[i+k]) = 1-h[j+k]
    # If i = j+k, this should be 1-h[j+k]. So one contribution of h[i] to overlap_k is 
    # from the pair h[j] * (1 - h[j+k]). The contribution is 1 - h[j+k]. 
    # special case when j+k >= N, or when doing valid shift 
    # We do zero padding, so results would be based on this padding, but let's just 
    # say what happens at shift k:
    # grad[j] = 1 - h[j+k] but careful about plotting
    # Including the full array is the correlate (1-h) * h compute that we know... 
    # but shifting by k, so we project the inside-out representation
    
    # the kernel is (1 - h), we flip it and do convolution cross: so for h + v, 
    # d/dv overlap_k = (1-h)_{flip} * (delta + h) = (1-h) * 1 = (1-h) 
    # but h's contribution is fixed
    # Let's be more careful:
    # overlap_k = sum_i h[i] (1 - h[i+k])
    # d/dh_j = sum_i d/dh_j [h[i] * (1 - h[i+k])]
    # if i != j: additive term is zero
    # if i = j: we get d/dh_j [h[j] * (1 - h[j+k])] = 1 - h[j+k]
    # But here we had i = j+k and get contribution from i = j+k
    # So just compute:
    # let s = j+k  
    # if j+k < len(h):
    #   grad[j] = 1 - h[j+k]
    # if j+k >= len(h), we just have zero coverage. 
    # However, correlation does zero-padding and so is not relevant here. 
    # Make gradient_effect of width (2*len(h)-1) 
    # [1, 1, 1, 1] with shift 2
    # h_l = [1,1,1,1], (1-h) = [0,0,0,0]
    # overlap_2 = h[0]*(1-h[2]) + h[1]*(1-h[3]) 
    # d/dh_0 = 1-h[2], d/dh_1 = 1-h[3], d/dh_2 = 0, d/dh_3 = 0  
    # The easiest is to compute the gradient of a specific overlap_k = scalar:
    # This gradient is just (1 - h_{i+k}) over i, for i = 0, ..., n-1

    # Instead, use what we already know how to compute with vecs. 
    # conv (1-h) vs h
    grad = np.zeros_like(h)
    # we don't want to condense this yet into a proper convolution,
    # but we can do necessary indexing manually 
    
    # use the full correlation again, initialize h_smooth = h with padding, 
    # kernel = 1-h, and make 1-h_f = 1 - h_mod to do 
    # g_{i} = 1 - h[i+k]
    # So set up:
    h_f = np.zeros(2*len(h)-1)
    h_f[:len(h)] = h
    h_shift = np.flip(1-h)
    # -> correlate result of len(h) * 2 - 1  
    # We know from correlate or fft how one shift works.
    # Let's not do overcomplicated version:
    
    if k_shift < len(h):
        grad = np.zeros_like(h)
        grad[:len(h) - k_shift] = 1 - h[k_shift:]
    else:
        grad = np.zeros_like(h)
    return grad

def line_search(h0, grad, max_step=0.05, dx=0.02857142857142857):
    # most basic line search. Given h0, step direction grad, find best step 
    # Find α s.t. next_h = h0 - α * grad is feasible and C5 minimized
    
    def test_alpha(alpha):
        h_test = h0 - alpha * grad
        h_test = project_feasible(h_test, len(h0), dx)
        return evaluate_c5(h_test, dx)
    
    best_c5 = evaluate_c5(h0, dx)
    best_step = 0.0
    for alpha in np.linspace(0.0, max_step, 20):
        c5 = test_alpha(alpha)
        if c5 < best_c5:
            best_c5 = c5
            best_step = alpha
    return best_step

def optimize_step(h, dx, k_max):
    # compute gradient of overlap_k_max w.r.t. h 
    # d/dh_i overlap_k_max = 1 - h[i+k_max] 
    # why not compute via correlate directly?
    # Convolve with [1,1,1,1] with default h is giving wrong indexing
    # Standard approach is correlate(h, 1-h) by shift position k, we want
    # gradient of exactly overlap[k_max]
    # Given that overlap_k = sum_i h[i] * (1 - h[i+k])
    
    # Actually:
    # If we do overlap_k = corr(h, 1-h)[k]  where k = 0,1,... 
    # d/dh_i overlap_k = d/dh_i sum_j h[j] * (1 - h[j+k]) -> (1-h[i+k]) when i = j+k? (if we need to interpret this)
    # Sum over j: d/dh_i h[j]* (1-h[j+k]) = δ_{i,j} * (1-h[j+k])  = (1-h[i+k]) (only when j=i in the sum)
    # We had (I - shift_k(h)) * h = full overlap vector under full convolution
    # see: https://math.stackexchange.com/q/1520782
    # There is a way to describe this by shifted version:
    # If C_k = sum_{i} h[i] * (1 - h[i+k])
    # then we define the len(h) to len(h) shift:
    # If j = i+k, or k = j-i, we can rearrange:
    # C_k = sum_{i, j=i+k} h[i] * (1-h[j])
    # = sum_{j} (1-h[j]) * sum_{i, i+k=j} h[i]  
    # = sum_{j} (1-h[j]) * (h shifted by -k)  
    # Another interpretation: 
    # Given: 
    # C[k] = sum_{i} h[i] * (1 - h[i+k]) 
    # The contribution of h[j] to C[k] :
    # Only h[j] appears when i = j-k, i + k = j, so that term will be h[j]* (1-h[j])
    # No wait we want:
    # sum_{i=0}^N h[i] * (1 - h[i+k]) 
    # which means, treatment of indices i=k, i=k+1, ..., i=N+k-1
    # for i = i, term is h[i] * (1 - h[i+k])
    # for i = j-k -> i+k = j, we find h[j-k] * (1- h[j]) 
    # What gradient_does try to compute for a specific overlap_k?
    # Overlap_k = sum_i h[i] * (1-h[i+k])  
    # d/dh_j Overlap_k = sum_i d/dh_j [h[i] * (1 - h[i+k])] = (if neighbor index j) 1-h[j+k]
    # try: 
    grad = np.zeros_like(h)
    if k_max < len(h):
        for i in range(len(h)-k_max):
            grad[i] = 1 - h[i+k_max]
    else:
        # Beyond valid range - already zero
        pass
    
    # line search
    a = line_search(h, grad, 0.05, dx)
    h_next = h - a * grad
    h_next = project_feasible(h_next, len(h), dx)

    return h_next

def main_optimize(h0, n_points, dx, budget_s):
    # Simple but more directed optimization
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, dx)
    start_time = time.time()
    steps = 0
    while time.time() - start_time < budget_s - 1:
        steps += 1
        # compute overlaps to find max
        overlaps = correlate(best_h, 1 - best_h, mode="full")
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        # optimize the corresponding shift of overlap
        h_new = optimize_step(best_h, dx, k_max)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # every 50 steps, add layer of stochasticity
        if steps % 50 == 0:
            # add small noise if not at optimal yet:
            if c5_new > 0.381:
                noise = np.random.normal(0, 0.005, n_points)
                h_new = best_h + noise
                h_new = project_feasible(h_new, n_points, dx)
                c5_new2 = evaluate_c5(h_new, dx)
                if c5_new2 < best_c5:
                    best_h = h_new
                    best_c5 = c5_new2

        if best_c5 <= 0.3807:
            break

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Initialize with parent solution
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure h sums to n_points / 2
        h = h * (n_points / 2.0) / np.sum(h)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    best_h, best_c5 = main_optimize(h, n_points, dx, budget_s)
    
    return (best_h.tolist(), best_c5, n_points)
