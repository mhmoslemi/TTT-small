import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum=h, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Compute overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def finite_diff_gradient(h, dx, eps=1e-4):
    """Compute gradient using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, len(h))
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

    start_time = time.time()

    noise = np.random.normal(0, 0.05, n_points)
    h = h + noise
    h = project_to_feasible(h, n_points)
    
    # Current and best solutions
    current_h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Annealing parameters
    temp = 0.1 * best_c5
    cooling_rate = 0.999
    min_temp = 1e-6
    
    iter_count = 0
    max_iter = 10000
    
    # Main annealing loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Generate candidate by perturbing current solution
        candidate_h = current_h + np.random.normal(0, 0.01, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject according to Metropolis criterion
        if candidate_c5 < best_c5:
            current_h = candidate_h.copy()
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            temp *= cooling_rate
        else:
            delta = candidate_c5 - best_c5
            if random.random() < np.exp(-delta / temp):
                current_h = candidate_h.copy()
                temp *= cooling_rate
            else:
                temp *= cooling_rate
                
        # Occasionally perform gradient descent refinement
        if iter_count % 50 == 0:
            grad = finite_diff_gradient(current_h, dx)
            step_size = 0.01
            current_h = current_h - step_size * grad
            current_h = project_to_feasible(current_h, n_points)
            
            # Update best if improved
            c5_now = evaluate_c5(current_h, dx)
            if c5_now < best_c5:

                best_c5 = c5_now
                
        # Early exit if target reached

    grad = finite_diff_gradient(best_h, dx)
    step_size = 0.01
    for _ in range(5):
        best_h = best_h - step_size * grad

    final_c5 = evaluate_c5(best_h, dx)
