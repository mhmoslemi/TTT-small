import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population with PSO
    n_particles = 30
    particles = []
    velocities = []
    
    # Seed particle
    if 'initial_h_values' in globals():
        h_seed = np.array(initial_h_values)
        h_seed = np.clip(h_seed, 0, 1)
    else:
        h_seed = np.ones(n_points)
        h_seed[n_points//2:] = 0
        h_seed += np.random.normal(0, 0.05, n_points)
        h_seed = np.clip(h_seed, 0, 1)
    h_seed = project_to_feasible(h_seed, n_points)
    
    # Initialize other particles with blocky patterns + noise
    for i in range(n_particles):
        if i == 0:
            h = h_seed.copy()
        else:
            h = np.ones(n_points)
            split = n_points // 2
            h[:split] = 1
            h[split:] = 0
            # Add some noise
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track best position
    best_p = [p.copy() for p in particles]
    best_c5 = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_c5)
    global_best = particles[global_best_idx].copy()
    global_c5 = best_c5[global_best_idx]
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = max(0.3, inertia * (1 - inertia_decay))
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity_update = (
                alpha * r1 * (best_p[i] - particles[i]) +
                beta * r2 * (global_best - particles[i])
            )
            
            velocities[i] = np.clip(velocity_update, -max_velocity, max_velocity)
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < best_c5[i]:
                best_c5[i] = current_c5
                best_p[i] = particles[i].copy()
            
            # Update global best
            if current_c5 < global_c5:
                global_c5 = current_c5
                global_best = particles[i].copy()
                
                # Early termination
                if global_c5 <= 0.3807:
                    break
        
        # Periodic time check
        if iteration % 50 == 0 and time.time() - start_time > budget_s - 1:
            break
    
    return (global_best.tolist(), global_c5, n_points)
