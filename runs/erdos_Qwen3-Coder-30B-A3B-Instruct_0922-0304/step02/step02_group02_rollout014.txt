import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Relaxation schedule
    mu_values = np.logspace(np.log10(0.5), 0, 50, endpoint=True)
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create the relaxed optimization problem
        h_var = cp.Variable(n_points)
        h_sum = cp.sum(h_var)
        constraint_sum = cp.Equality(h_sum, n_points / 2)
        constraint_bounds = [h_var >= 0, h_var <= 1]
        
        # Compute C5 using CVXPY
        # We approximate this with a smooth formulation
        C5_approx = 0
        for k in range(-n_points + 1, n_points):
            if k < 0:
                # h[i] * (1 - h[i-k]) for i = 0..n_points+k-1
                expr = 0
                for i in range(n_points + k):
                    expr += h_var[i] * (1 - h_var[i-k])
                C5_approx = cp.maximum(C5_approx, expr * dx)
            elif k == 0:
                # h[i] * (1 - h[i]) for i = 0..n_points-1
                expr = 0
                for i in range(n_points):
                    expr += h_var[i] * (1 - h_var[i])
                C5_approx = cp.maximum(C5_approx, expr * dx)
            else:
                # h[i] * (1 - h[i+k]) for i = 0..n_points-k-1
                expr = 0
                for i in range(n_points - k):
                    expr += h_var[i] * (1 - h_var[i+k])
                C5_approx = cp.maximum(C5_approx, expr * dx)
        
        # Objective function
        obj = mu * C5_approx + (1 - mu) * cp.sum(cp.power(h_var - 0.5, 2))
        
        # Create problem
        prob = cp.Problem(cp.Minimize(obj), [constraint_sum] + constraint_bounds)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                # Early exit if we hit target
                if best_c5 <= 0.3807:
                    break
        except:
            # Fallback to previous solution
            pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
