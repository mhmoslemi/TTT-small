import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum, dx):
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale_factor = target_sum / current_sum
    h_scaled = h * scale_factor
    return h_scaled

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.05):
    h_mutated = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mutated[i] += random.gauss(0, sigma)
    return h_mutated

def genetic_algorithm(seed=42, budget_s=600, n_points=70, population_size=50, generations=500):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum, dx)
        population.append(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum, dx)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = scale_to_target_sum(h, target_sum, dx)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(c5)
            
        # Find best in current generation
        min_fitness = min(fitness_scores)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness_scores.index(min_fitness)].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores.sort()
        
        # Keep top 50% and regenerate the rest
        next_generation = population[:population_size//2]
        
        # Add mutated offspring
        while len(next_generation) < population_size:
            # Select two parents
            parent1 = population[random.randint(0, population_size//2 - 1)]
            parent2 = population[random.randint(0, population_size//2 - 1)]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Enforce constraints
            child1 = np.clip(child1, 0, 1)
            child1 = scale_to_target_sum(child1, target_sum, dx)
            child2 = np.clip(child2, 0, 1)
            child2 = scale_to_target_sum(child2, target_sum, dx)
            
            next_generation.append(child1)
            if len(next_generation) < population_size:
                next_generation.append(child2)
                
        population = next_generation
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # GA parameters
    n_points = 70
    population_size = 50
    generations = 500
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(seed, budget_s, n_points, population_size, generations)
    
    # Final evaluation
    dx = 2.0 / n_points
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
