import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple starting point
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Set up convex relaxation parameters
    lambda_reg = 0.01
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Approximate gradient using the max overlap index
        # Let g[i] = (1 - h[(i+k_max) % n_points]) - h[(i+k_max) % n_points]
        h_shifted = np.roll(h, k_max)
        grad = h_shifted - (1 - h_shifted)
        
        # Line search over a few step sizes
        best_candidate_c5 = best_c5
        best_candidate_h = h.copy()
        
        step_sizes = [0.1, 0.5, 1.0]
        for alpha in step_sizes:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_c5(h_candidate, dx)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # If we found an improvement, update
        if best_candidate_c5 < best_c5:
            h = best_candidate_h
            best_c5 = best_candidate_c5
        
        # Periodic convex relaxation
        if iteration % relaxation_interval == 0:
            try:
                # Convex relaxation using CVXPY
                h_relax = cp.Variable(n_points)
                objective = cp.Minimize(
                    overlaps[k_max] + lambda_reg * cp.sum(cp.entr(h_relax) + cp.log(n_points/2) * h_relax)
                )
                constraints = [
                    h_relax >= 0,
                    h_relax <= 1,
                    cp.sum(h_relax) == n_points/2
                ]
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_relax_sol = h_relax.value
                    # Project back to feasible set
                    h_relax_sol = project_feasible(h_relax_sol, n_points, dx)
                    # Use this as warm start for next iteration
                    candidate_c5 = evaluate_c5(h_relax_sol, dx)
                    if candidate_c5 < best_c5:
                        h = h_relax_sol
                        best_c5 = candidate_c5
                        lambda_reg *= 0.5  # Reduce regularization if successful
            except:
                # If convex solve fails, continue with previous step
                pass
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Prevent exact early exit
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
