import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum == n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Default construction
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = project_to_feasible(h_init, n_points)
    
    # Generate initial population
    for i in range(n_particles):
        if i == 0:
            h = h_init.copy()
        else:
            # Create perturbed variant
            h = h_init.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track personal bests
    personal_bests = [evaluate_c5(p, dx) for p in particles]
    personal_best_positions = [p.copy() for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_bests)
    global_best = personal_bests[global_best_idx].copy()
    global_best_position = particles[global_best_idx].copy()
    
    # PSO iterations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (alpha * r1 * (personal_best_positions[i] - particles[i]) + 
                          beta * r2 * (global_best_position - particles[i]))
            velocities[i] = inertia * velocities[i] + vel_update
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < personal_bests[i]:
                personal_bests[i] = c5
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if c5 < global_best:
                    global_best = c5
                    global_best_position = particles[i].copy()
        
        # Update inertia
        inertia = max(0.4, inertia * (1 - inertia_decay))
        
        # Early stopping - target hit
        if global_best <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(global_best_position, dx)
    
    return (global_best_position.tolist(), final_c5, n_points)
