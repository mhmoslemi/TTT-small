import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    # Compute overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from existing solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Hyperparameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % relaxation_interval == 0:
            # Solve convex relaxation
            h_relax = cp.Variable(n_points)
            constraints = [0 <= h_relax, h_relax <= 1, cp.sum(h_relax) == n_points/2]
            # Approximate objective (simplified form)
            try:
                # Compute max overlap in current solution
                overlap_vals = correlate(h, 1 - h, mode="full")
                max_idx = np.argmax(np.abs(overlap_vals))
                # Simplified: just add regularization
                objective = cp.Minimize(cp.sum(cp.multiply(h_relax, cp.log(h_relax + 1e-12))) + lambda_reg * cp.sum(h_relax))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_new = np.maximum(0, np.minimum(1, h_relax.value))
                    h = project_feasible(h_new, n_points, dx)
                    current_c5 = evaluate_erdos_solution(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
            except:
                pass

        # Compute overlap and find max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Approximate gradient
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        best_candidate_c5 = best_c5
        best_candidate_h = h.copy()
        for alpha in line_search_steps:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate.copy()
        
        # Update if better
        if best_candidate_c5 < best_c5:
            h = best_candidate_h
            best_c5 = best_candidate_c5
            best_h = h.copy()
        
        # Early exit for target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
