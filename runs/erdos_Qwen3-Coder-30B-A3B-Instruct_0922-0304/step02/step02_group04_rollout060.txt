import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_relaxation(h_initial, n_points, dx):
    # Variables
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Compute correlation constraints
    # For each lag k, we want overlap_k <= C
    for k in range(2 * n_points - 1):
        # We manually calculate correlation constraint
        corr_val = 0.0
        if k < n_points:
            # First part of full correlation
            for i in range(n_points - k):
                corr_val += h[i] * (1 - h[i + k])
        else:
            # Second part of full correlation
            for i in range(k - n_points + 1, n_points):
                corr_val += h[i] * (1 - h[i + k - (n_points - 1)])
        constraints.append(corr_val <= C * dx)
        
    # Objective
    objective = Minimize(C)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        h_optimal = h.value
    except:
        h_optimal = h_initial.copy()
    
    return h_optimal

def local_search_step(h_start, n_points, dx, iterations=50):
    h_current = h_start.copy()
    c5_best = evaluate_c5(h_current, dx)
    
    for _ in range(iterations):
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_best) / eps
        
        # Gradient descent with small step
        h_current = h_current - 0.01 * grad
        h_current = project_to_feasible(h_current, n_points)
        
        c5_trial = evaluate_c5(h_current, dx)
        if c5_trial < c5_best:
            c5_best = c5_trial
    
    return h_current, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values[:n_points])
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
    
    h = project_to_feasible(h, n_points)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation step
        h_relaxed = solve_convex_relaxation(h, n_points, dx)
        h_projected = project_to_feasible(h_relaxed, n_points)
        
        # Local search on solution
        h_local, c5_local = local_search_step(h_projected, n_points, dx)
        
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Continue with projected solution for next iteration
        h = h_local.copy()
        
    return (best_h.tolist(), best_c5, n_points)
