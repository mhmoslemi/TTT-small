import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with 5-fold symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create symmetric pattern: 5 segments of 14 points each
        h = np.zeros(n_points)
        segment_size = n_points // 5
        values = [0.7, 0.3, 0.7, 0.3, 0.7]
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            h[start_idx:end_idx] = values[i]
    
    # Adjust to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 1000
    iterations_without_improvement = 0
    patience = 20
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Use L-BFGS-B with bounds
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with L-BFGS-B
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                         options={'maxiter': 10, 'ftol': 1e-8})
        
        if result.success:
            h = result.x.copy()
        else:
            # If optimization fails, use a gradient step
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            
            learning_rate = 0.1
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Repair feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        if iterations_without_improvement > patience:
            # Start over with a slightly different initialization
            h = best_h.copy()
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            iterations_without_improvement = 0

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
