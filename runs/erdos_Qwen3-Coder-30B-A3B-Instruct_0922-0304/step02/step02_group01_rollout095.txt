import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    # Clip to [0,1] and rescale to target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    patience = 100
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if patience_counter > patience:
            # Reset if no improvement
            h = best_h.copy()
            patience_counter = 0
            
        # Compute overlaps for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        # Find the index of maximum overlap, considering the shift
        max_idx = np.argmax(overlap_full)
        max_overlap = overlap_full[max_idx] * dx
        
        # Compute gradient for the max overlap shift
        # The gradient w.r.t. h for shift k can be derived
        # For overlap at shift k, the contribution to gradient of h[i] is:
        # (1 - h[i + k]) if k >= 0 and both i and i+k are valid
        # or (1 - h[i - k]) if k < 0 and both i and i-k are valid
        
        # For computational efficiency, we'll compute it analytically for max shift
        k_shift = max_idx - (n_points - 1)  # Convert index to actual shift
        grad = np.zeros_like(h)
        
        # Gradient computation based on shift k_shift
        if k_shift >= 0:
            # h[i] contributes (1 - h[i + k_shift]) to overlap
            for i in range(n_points - k_shift):
                grad[i] += -(1 - h[i + k_shift])
        else:
            # h[i] contributes (1 - h[i - k_shift]) to overlap
            k_shift_abs = -k_shift
            for i in range(k_shift_abs, n_points):
                grad[i] += -(1 - h[i - k_shift_abs])
        
        # Apply momentum and update
        velocity = momentum * velocity - step_size * grad
        h = h + velocity
        
        # Project to feasible set
        h = project_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
