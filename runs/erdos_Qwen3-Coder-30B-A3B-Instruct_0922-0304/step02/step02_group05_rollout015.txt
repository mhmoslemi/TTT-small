import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, k):
    """Compute gradient for shift k"""
    h_shifted = np.roll(h, k)
    return h_shifted - (1 - h_shifted)

def line_search(h, gradient, n_points, dx):
    """Perform line search with step sizes [0.1, 0.5, 1.0]"""
    step_sizes = [0.1, 0.5, 1.0]
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for alpha in step_sizes:
        h_candidate = project_feasible(h - alpha * gradient, n_points, dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < best_c5:
            best_h = h_candidate
            best_c5 = candidate_c5
            break  # Take the first improving step
    return best_h, best_c5

def convex_relaxation(h, n_points, dx, lam=0.01):
    """Solve convex relaxation of the problem using CVXPY"""
    # Variables with log barrier to avoid zero
    h_relax = cp.Variable(n_points)
    
    # Constraint
    constraints = [h_relax >= 0, h_relax <= 1, cp.sum(h_relax) == n_points/2]
    
    # Objective: approximate squared overlap term + regularization  
    # We interpret this as minimizing overlap + regularization
    # the convex relaxation approximates the original objective with a smooth function
    obj = cp.Minimize(cp.sum_squares(h_relax) + lam * cp.sum(cp.entr(h_relax) + cp.log(n_points/2)))
    
    # Solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        # fallback if solver fails
        return h
    
    if prob.status == 'optimal':
        new_h = np.array(h_relax.value).flatten()
        return project_feasible(new_h, n_points, dx)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initializing
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iteration = 0
    relaxation_interval = 20
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        k = np.argmax(overlap)
        
        # Compute gradient for max overlap 
        gradient = compute_gradient(h, k)
        
        # Line search
        h, c5 = line_search(h, gradient, n_points, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Periodic convex relaxation
        if iteration % relaxation_interval == 0:
            h_relaxed = convex_relaxation(h, n_points, dx, lambda_reg)
            c5_relaxed = evaluate_erdos_solution(h_relaxed)
            
            if c5_relaxed < c5:
                h = h_relaxed.copy()
                c5 = c5_relaxed
                
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                    
            # Adjust regularization
            lambda_reg *= 0.5
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Check if time is running out
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
