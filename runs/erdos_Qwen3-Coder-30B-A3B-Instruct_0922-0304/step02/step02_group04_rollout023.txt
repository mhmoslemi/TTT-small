import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    # Compute gradient with finite differences
    grad = np.zeros_like(h)
    c5_base = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

        c5_trial = evaluate_c5(h_trial, dx)
        if c5_trial < evaluate_c5(h, dx) - 1e-6:
            return alpha * d
        alpha *= 0.5
    return d

def trust_region_sqp(h, dx, max_iter=1000, tol=1e-3):
    n_points = len(h)
    H = np.eye(n_points)  # Initial Hessian approximation
    h = project_to_feasible(h, n_points)

        if c5_val < 0.3807:
            break
            
        grad = compute_gradient(h, dx)
        if np.linalg.norm(grad) < tol:
            break
        
        # Compute step
        try:
            step = solve_quadratic_subproblem(H, grad, n_points, [0, 1])
        except:
            # Fallback to simple gradient descent if Hessian is problematic
            step = -grad
        
        # Line search
        alpha = 1.0
        while alpha > 1e-10:
            h_trial = h + alpha * step
            h_trial = project_to_feasible(h_trial, n_points)
            c5_trial = evaluate_c5(h_trial, dx)
            if c5_trial < c5_val - 1e-6 * alpha * np.dot(grad, step):
                break
            alpha *= 0.5
            
        if alpha <= 1e-10:
            break
            
        h = h_trial
        
        # Update Hessian using BFGS
        y = compute_gradient(h, dx) - grad
        if np.linalg.norm(y) > 1e-8 and np.linalg.norm(step) > 1e-8:
            rho = 1.0 / np.dot(y, step)
            I = np.eye(n_points)
            H = (I - rho * np.outer(step, y)) @ H @ (I - rho * np.outer(y, step)) + rho * np.outer(step, step)
        
    return h
