import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_constraint(h, target_sum, dx):
    """Scale h to meet the integral constraint sum(h)*dx = 1"""
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) < 1e-10:
        return h
    scale_factor = target_sum / current_sum
    return h * scale_factor

def clip_and_normalize(h, target_sum, dx):
    """Clip h to [0,1] and normalize to meet integral constraint"""
    h = np.clip(h, 0, 1)
    h = scale_to_constraint(h, target_sum, dx)
    return h

def crossover(parent1, parent2, crossover_rate=0.5):
    """Perform single-point crossover on two parents"""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, noise_std=0.05):
    """Apply Gaussian mutation to h"""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, noise_std)
    return mutated

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_normalize(h, target_sum, dx)
        population.append(h)
    
    # Add random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = clip_and_normalize(h, target_sum, dx)
        population.append(h)
    
    # GA parameters
    max_generations = 1000
    elite_size = population_size // 10  # Top 10%
    mutation_rate = 0.1
    crossover_rate = 0.5
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection (tournament)
            idx1 = np.random.randint(0, population_size // 2)
            idx2 = np.random.randint(0, population_size // 2)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Ensure constraints
            child1 = clip_and_normalize(child1, target_sum, dx)
            child2 = clip_and_normalize(child2, target_sum, dx)
            
            new_population.extend([child1, child2][:population_size - len(new_population)])
            
        population = new_population[:population_size]
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5 = genetic_algorithm(seed, budget_s)
    return (h.tolist(), c5, len(h))
