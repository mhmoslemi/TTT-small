import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing parameters
    T = 1.0
    beta = 0.995
    epsilon = 0.05
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cooling schedule
    iterations_per_cooling = 100
    no_improvement_reset = 1000
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        no_improvement_count += 1
        
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Generate candidate h
        h_candidate = h + epsilon * T * np.random.randn(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new overlap
        corr_candidate = correlate(h_candidate, 1 - h_candidate, mode="full")
        candidate_c5 = np.max(corr_candidate) * dx
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            current_c5 = candidate_c5
            no_improvement_count = 0
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Cooling and reset
        if iter_count % iterations_per_cooling == 0:
            T *= beta
            
        if no_improvement_count >= no_improvement_reset:
            T = 1.0
            no_improvement_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
