import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def solve_quadratic_subproblem(H, g, n_points, bounds):
    # Simplified quadratic subproblem solver using projected gradient method
    # We'll solve with simple projected gradient descent for simplicity
    d = np.zeros(n_points)
    step_size = 0.01
    max_iter = 50
    
    for _ in range(max_iter):
        # Compute gradient
        grad = H @ d + g
        
        # Descent direction
        d_new = d - step_size * grad
        
        # Project to bounds and sum constraint
        d_new = np.clip(d_new, bounds[0], bounds[1])
        d_new = project_to_feasible(d_new, n_points)
        
        # Check for convergence
        if np.linalg.norm(d_new - d) < 1e-6:
            break
            
        d = d_new
    
    return d
