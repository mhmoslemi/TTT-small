import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Structured initialization: 5-fold symmetric pattern
        h = np.zeros(n_points)
        segment_size = n_points // 5  # 14
        for i in range(5):
            start = i * segment_size
            end = start + segment_size
            # Alternate high/low values within each segment
            for j in range(start, end):
                if (j - start) % 2 == 0:
                    h[j] = 0.7
                else:
                    h[j] = 0.3
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Use scipy.optimize with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Define inner optimization loop
    def optimize_step(h_in, bounds, constraints, max_iter=50):
        res = minimize(objective, h_in, method='L-BFGS-B', bounds=bounds, 
                       constraints=constraints, options={'maxiter': max_iter})
        if res.success:
            h_out = res.x
        else:
            h_out = h_in.copy()
            
        # Repair to ensure bounds and sum constraint
        h_out = np.clip(h_out, 0, 1)
        h_out = h_out * (n_points / 2) / np.sum(h_out)
        return h_out
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Optimization step
        h = optimize_step(h, bounds, constraints, max_iter=30)
        
        # Add small random perturbation occasionally to escape local minima
        if iteration % 20 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
