import numpy as np
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Annealing parameters
    T = 1.0
    beta = 0.995
    epsilon = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    no_improve_count = 0
    reset_threshold = 1000
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improve_count > reset_threshold:
            T = 1.0
            no_improve_count = 0
            
        # Generate candidate
        h_candidate = h + epsilon * T * np.random.normal(0, 1, n_points)
        
        # Enforce bounds and sum constraint
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            
        # Cooling schedule
        if no_improve_count % 100 == 0:
            T = beta * T
            
    return (best_h.tolist(), best_c5, n_points)
