import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create random initial solution
        h = np.random.random(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Fill up with random particles
    for _ in range(num_particles - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(num_particles):
        velocities.append(np.random.normal(0, 0.05, n_points))
    
    # Evaluate initial fitness
    fitness_scores = []
    for i, h in enumerate(particles):
        c5 = evaluate_c5(h, dx)
        fitness_scores.append((c5, i))
    
    # Track personal bests
    p_best_indices = [i for _, i in fitness_scores]
    p_best_scores = [c5 for c5, _ in fitness_scores]
    
    # Find global best
    best_idx = np.argmin([c5 for c5, _ in fitness_scores])
    g_best_idx = best_idx
    best_c5 = p_best_scores[best_idx]
    best_h = particles[best_idx].copy()
    
    # PSO parameters
    omega = 1.0
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            v = omega * velocities[i]
            v += c1 * r1 * (particles[p_best_indices[i]] - particles[i])
            v += c2 * r2 * (particles[g_best_idx] - particles[i])
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            particles[i] += v
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < p_best_scores[i]:
                p_best_scores[i] = c5
                p_best_indices[i] = i
                
                # Update global best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particles[i].copy()
                    g_best_idx = i
                    last_improvement = iteration
            
            velocities[i] = v
        
        # Adaptive inertia
        if iteration % 50 == 0:
            omega = max(0.8, omega - 0.02)
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Reset stagnant particles occasionally
        if iteration - last_improvement > 100:
            for i in range(num_particles):
                if np.random.random() < 0.1:
                    h = np.random.random(n_points)
                    h = project_feasible(h, n_points, dx)
                    particles[i] = h
                    velocities[i] = np.random.normal(0, 0.05, n_points)
            last_improvement = iteration
        
        # Check time limit
        if time.time() - start_time > budget_s - 1:
            break

    return (best_h.tolist(), best_c5, n_points)
