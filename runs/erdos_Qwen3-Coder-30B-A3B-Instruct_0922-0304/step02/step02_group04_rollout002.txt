import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    # Compute gradient of C5 with respect to h
    # Using finite differences approach for gradient
    eps = 1e-4
    grad = np.zeros_like(h)
    c5_base = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iterations = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    local_refine_freq = 20
    
    # Initialize swarm with projective PSO
    swarm = []
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = project_to_feasible(h_base, n_points)
        swarm.append(h_base.copy())
    else:
        h_base = np.ones(n_points) * 0.5
        h_base = project_to_feasible(h_base, n_points)
        swarm.append(h_base.copy())
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        swarm.append(project_to_feasible(h, n_points))
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Personal best positions and values
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_values = [evaluate_c5(swarm[i], dx) for i in range(swarm_size)]
    
    # Global best
    best_idx = np.argmin(personal_best_values)
    global_best = personal_best[best_idx].copy()
    global_best_value = personal_best_values[best_idx]
    
    # PSO main loop
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            c5_val = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if c5_val < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = c5_val
                
                # Update global best if needed
                if c5_val < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = c5_val
                    
        # Periodic local refinement
        if iter % local_refine_freq == 0 and iter > 0:
            # Refine top 10% particles
            top_k = max(1, swarm_size // 10)
            sorted_indices = np.argsort(personal_best_values)[:top_k]
            for idx in sorted_indices:
                h_refined = personal_best[idx].copy()
                # Gradient ascent with step size control
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h_refined, dx)
                    h_refined -= step_size * grad
                    h_refined = project_to_feasible(h_refined, n_points)
                    
                # Update personal best if improvement
                new_c5 = evaluate_c5(h_refined, dx)
                if new_c5 < personal_best_values[idx]:
                    personal_best[idx] = h_refined
                    personal_best_values[idx] = new_c5
                    if new_c5 < global_best_value:
                        global_best = h_refined
                        global_best_value = new_c5
                        
        # Early exit if target reached
        if global_best_value <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_value, n_points)
