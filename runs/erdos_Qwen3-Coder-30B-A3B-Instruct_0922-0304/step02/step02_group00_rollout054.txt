import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum, norm
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Sawtooth pattern with slight noise for diversity
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frame-based iteration
    max_iter = 1000
    no_improvement_count = 0
    convergence_threshold = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > convergence_threshold:
            break

        # Optimization step: minimize overlap at k_max
        # We formulate the subproblem to reduce the overlap at shift k_max
        h_var = Variable(n_points)
        
        # The overlap at k_max can be written as:
        # sum_{i} h[i] * (1 - h[(i + k_max) % n_points])
        # which is a linear function in h

        # Create the constraint:
        # sum h = n/2
        # h[i] in [0, 1]
        # We minimize overlap at k_max
        
        # Formulate linear objective corresponding to overlap index k_max
        # The overlap is h[i] * (1 - h[(i+k_max)%n]) for i = 0 to n-1
        # Hence we want to minimize sum h[i]*(1 - h[(i+k_max)%n])
        # which is sum h[i] - sum h[i]*h[(i+k_max)%n]
        # The first term is constant, so minimize sum h[i]*h[(i+k_max)%n]  
        obj = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            obj += h_var[i] * h_var[j]
        
        # Minimize this objective subject to constraints
        constraints = [
            cp_sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve the convex problem
        problem = Problem(Minimize(obj), constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If solver failed, fall back to some local change
                h = h * 0.99 + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # If any exception occurs revert to previous h and back off
            h = h * 0.99 + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
