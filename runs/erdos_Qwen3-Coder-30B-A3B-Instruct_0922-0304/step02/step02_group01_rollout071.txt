import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    momentum = 0.9
    step_size = 0.01
    prev_update = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        
        # Determine shift k from correlation index
        k = max_idx - (n_points - 1)
        
        # Compute gradient of overlap with respect to h
        grad = np.zeros_like(h)
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i in range(n_points - k)
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k])
                if i + k < n_points:
                    grad[i + k] -= h[i]
        else:
            # h[i] * (1 - h[i + k]) for i in range(n_points + k)  
            for i in range(n_points + k):
                grad[i] += (1 - h[i - k])
                if i - k < n_points:
                    grad[i - k] -= h[i]
        
        # Apply momentum and update
        update = momentum * prev_update - step_size * grad
        h_new = h + update
        prev_update = update
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Simple line search for better step size
        h_test = h_new.copy()
        test_c5 = evaluate_c5(h_test, dx)
        if test_c5 < current_c5:
            h = h_test
        else:
            # Try smaller step if still in time
            step_size *= 0.5
            if step_size < 1e-6:
                step_size = 0.01
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
