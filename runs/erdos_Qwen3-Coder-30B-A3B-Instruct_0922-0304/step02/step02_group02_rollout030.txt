import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # 5-fold symmetric initialization
        h = np.zeros(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            if i % 2 == 0:
                h[start_idx:end_idx] = 0.7
            else:
                h[start_idx:end_idx] = 0.3
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraint for sum(h) = 1
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Get current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Use L-BFGS optimization
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Run optimization
        result = minimize(
            lambda x: evaluate_c5(x, dx),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if result.success:
            h = result.x
            # Ensure constraints are satisfied
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Feasibility correction
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Perturb slightly and continue
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        iter_count += 1

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
