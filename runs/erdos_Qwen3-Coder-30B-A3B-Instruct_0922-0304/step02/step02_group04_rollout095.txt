import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_to_feasible(h_plus, n_points)
        h_minus = project_to_feasible(h_minus, n_points)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def line_search(h, d, dx, alpha_max=1.0, c1=1e-4):
    alpha = alpha_max
    c5_base = evaluate_c5(h, dx)
    
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new <= c5_base + c1 * alpha * np.dot(d, compute_gradient(h, dx))
            return alpha
        alpha *= 0.5
    
    return alpha

def solve_quadratic_subproblem(H, g, n_points):
    # Quadratic subproblem min 0.5 * d^T * H * d + g^T * d 
    # subject to 0 <= d_i <= 1, sum(d) = 0
    
    # Simple gradient projection approach
    d = -np.dot(np.linalg.inv(H), g)
    d = d - np.mean(d)  # ensure sum(d) = 0
    return d

def generate_initial_h(n_points):
    # Generate initial step function from initial_h_values if available
    try:
        return np.array(initial_h_values)
    except NameError:
        # Create a simple alternating pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        return h

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    tolerance = 1e-3
    
    # Initialize
    h = generate_initial_h(n_points)
    h = project_to_feasible(h, n_points)
    
    # Hessian approximation (initial identity)
    H = np.eye(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return best_h, best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Check for convergence
        if np.linalg.norm(g) < tolerance:
            break
            
        # Solve quadratic subproblem
        try:
            d = solve_quadratic_subproblem(H, g, n_points)
        except:
            # Fall back to gradient descent if matrix inversion fails
            d = -g
            d = d - np.mean(d)
            
        # Line search for step size
        alpha = line_search(h, d, dx, alpha_max=1.0)
        
        # Update h
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Update Hessian using BFGS
        y = compute_gradient(h_new, dx) - g
        s = h_new - h
        
        # Avoid division by zero
        if np.dot(s, y) > 1e-10:
            rho = 1.0 / np.dot(s, y)
            I = np.eye(n_points)
            H = (I - rho * np.outer(s, y)) @ H @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
        
        h = h_new
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
