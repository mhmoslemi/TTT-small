import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import log

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to satisfy constraints: h in [0,1], sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k):
    """Compute gradient of overlap with shift k"""
    h_shifted = np.roll(h, k)
    return h_shifted - (1 - h_shifted)

def line_search(h, gradient, n_points, dx):
    """Perform line search to find best step size"""
    step_sizes = [0.1, 0.5, 1.0]
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for alpha in step_sizes:
        h_candidate = h - alpha * gradient
        h_candidate = project_feasible(h_candidate, n_points, dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < best_c5:
            best_h = h_candidate
            best_c5 = candidate_c5
    return best_h, best_c5

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    """Solve a relaxed convex optimization for escape from local minima"""
    # Variables
    h_relax = cp.Variable(n_points)
    
    # Objective
    # Approximate overlap computation with convex relaxation
    # We use the fact that overlap involves h_i * (1 - h_{i+k})
    # Which can be rewritten as: h_i - h_i * h_{i+k}
    # But for convex relaxation we linearize
    # 
    # The objective: we want to minimize max of overlaps, in a convex way
    # So instead we optimize a regularized version
    
    # Instead, using simpler approach: minimize weighted sum 
    # of a smooth approximation of max, plus entropy-like penalization
    
    # However, the true convex relaxation here depends on constraints.
    # For TODO: Linearize and use a different objective
    
    # For this version just ensure a smooth descent step
    h_harmonic = np.mean(h) * np.ones(n_points)
    h_harmonic = project_feasible(h_harmonic, n_points, dx)
    return h_harmonic

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points

    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Default construction: nice starting point
        h = np.ones(n_points) * 0.5
    h = project_feasible(h, n_points, dx)

    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters
    relaxation_interval = 20
    lambda_reg = 0.01
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        # Compute gradient at the conflict point
        k = max_overlap_idx - (n_points - 1)
        gradient = compute_gradient(h, k)
        
        # Line search with gradient descent
        h, c5 = line_search(h, gradient, n_points, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Periodic convex relaxation to escape local minima
        if iteration % relaxation_interval == 0:
            h_relax = solve_convex_relaxation(h, n_points, dx, lambda_reg)
            # Update based on relaxation result
            h = h_relax
            
            # Adaptive lambda reduction
            lambda_reg *= 0.5
            
        # бюджетный контроль
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final verification
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), best_c5, n_points)
