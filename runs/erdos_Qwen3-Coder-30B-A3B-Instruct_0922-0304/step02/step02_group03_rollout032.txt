import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible region: h[i] in [0,1], sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def initialize_particles(n_points, n_particles, initial_h=None):
    """Initialize particle swarm with diverse starting positions"""
    particles = []
    if initial_h is not None:
        particles.append(initial_h.copy())
    else:
        # Default initialization
        particles.append(np.ones(n_points) * 0.5)
    
    # Fill remaining particles with blocky patterns and noise
    while len(particles) < n_particles:
        h = np.zeros(n_points)
        # Randomly choose split point and block sizes
        split1 = np.random.randint(1, n_points-1)
        split2 = np.random.randint(split1+1, n_points)
        h[:split1] = 1
        h[split2:] = 0
        # Add some noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = project_to_feasible(h, n_points / 2)
        particles.append(h)
    
    return particles

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_particles = 30
    
    # Initialize particles
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_to_feasible(initial_h, target_sum)
    
    particles = initialize_particles(n_points, n_particles, initial_h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Initialize best positions
    particle_best = [p.copy() for p in particles]
    particle_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    global_best_idx = np.argmin(particle_best_scores)
    global_best = particle_best[global_best_idx].copy()
    global_best_score = particle_best_scores[global_best_idx]
    
    # PSO parameters
    alpha = 0.8  # Cognitive weight
    beta = 0.8   # Social weight
    inertia = 0.7
    inertia_decay = 1e-4
    max_velocity = 0.3
    
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia * velocities[i] +
                alpha * r1 * (particle_best[i] - particles[i]) +
                beta * r2 * (global_best - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best = particles[i].copy()
        
        # Decay inertia
        inertia = max(0.1, inertia - inertia_decay)
        
        # Early stopping
        if global_best_score <= 0.3807:
            break

    return (global_best.tolist(), global_best_score, n_points)
