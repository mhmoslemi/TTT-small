import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent if available, otherwise use a simple pulse
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
    
    # Initial C5 value
    current_c5 = evaluate_c5(h, dx)
    best_solution = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    initial_temp = 0.1 * current_c5
    temperature = initial_temp
    cooling_rate = 0.999
    noise_std = 0.01
    max_iter = 10000
    gradient_refinement_freq = 50
    gradient_steps = 5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate by adding noise
        candidate = h + np.random.normal(0, noise_std, n_points)
        candidate = project_to_feasible(candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = h.copy()
        else:
            # Accept with probability
            if random.random() < np.exp(-(candidate_c5 - current_c5) / (temperature + 1e-10)):
                h = candidate
                current_c5 = candidate_c5
        
        # Lower temperature
        temperature *= cooling_rate
        
        # Gradient refinement
        if iteration % gradient_refinement_freq == 0:
            # Compute gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, n_points)
                c5_minus = evaluate_c5(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient descent
            step_size = 0.01
            for _ in range(gradient_steps):
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                # TODO: might be more stable to recompute gradient and restart
                
            current_c5 = evaluate_c5(h, dx)
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_solution.tolist(), best_c5, n_points)
