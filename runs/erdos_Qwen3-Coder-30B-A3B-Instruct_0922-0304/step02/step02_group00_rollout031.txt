import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial construction and perturb around it
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
        for i in range(n_particles // 2):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
        # Add random particles
        for i in range(n_particles // 2):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    else:
        # Initialize entirely random
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    
    # Track best solutions
    pbest = [p.copy() for p in particles]
    pbest_scores = [float('inf')] * n_particles
    gbest = None
    gbest_score = float('inf')
    
    # PSO parameters
    omega = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i, h in enumerate(particles):
            score = evaluate_c5(h, dx)
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = h.copy()
            if score < gbest_score:
                gbest_score = score
                gbest = h.copy()
                
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            g = gbest
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = omega * v + c1 * r1 * (p - h) + c2 * r2 * (g - h)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
            
        if gbest_score <= 0.3807:
            break

    return (gbest.tolist(), gbest_score, n_points)
