import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum, dx):
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    scaled = h * (target_sum / current_sum)
    return np.clip(scaled, 0, 1)

def crossover(parent1, parent2):
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.1):
    mutated = h.copy()
    for i in range(len(mutated)):
        if random.random() < mutation_rate:
            mutated[i] += random.gauss(0, sigma)
    return mutated

def genetic_algorithm(seed, budget_s):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = scale_to_target_sum(h0, target_sum, dx)
        population.append(h0)
    
    # Fill the rest with random individuals
    while len(population) < population_size:
        h = np.random.uniform(0, 1, n_points)
        h = scale_to_target_sum(h, target_sum, dx)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness (lower C5 is better)
        fitness.sort(key=lambda x: x[0])
        current_best_c5, current_best_h = fitness[0]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if best_c5 <= 0.3807:
            break
            
        # Selection - keep top 50%
        elite_size = population_size // 2
        elite = [h for _, h in fitness[:elite_size]]
        
        # Generate new population
        new_population = elite.copy()
        
        # Create offspring through crossover and mutation
        while len(new_population) < population_size:
            parent1 = random.choice(elite)
            parent2 = random.choice(elite)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = scale_to_target_sum(child1, target_sum, dx)
            child2 = scale_to_target_sum(child2, target_sum, dx)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    if best_h is None:
        # Use final population if no better solution found
        last_fitness = [(evaluate_c5(h, dx), h) for h in population]
        last_fitness.sort(key=lambda x: x[0])
        best_c5, best_h = last_fitness[0]
    
    return best_h, best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5, n_points = genetic_algorithm(seed, budget_s)
    return (best_h.tolist(), best_c5, n_points)
