import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Since integral should be 1
    
    # Initialize swarm
    P = 50  # Population size
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    # Create initial swarm
    for _ in range(P):
        h = base_h + np.random.normal(0, 0.05 * np.max(base_h), n_points)
        h = repair_h(h, target_sum, dx)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = c2 = 2.0  # acceleration coefficients
    
    # Track global best
    global_best_c5 = float('inf')
    global_best_h = None
    
    # Particle bests
    particle_best_h = [h.copy() for h in swarm]
    particle_best_c5 = [evaluate_c5(h, dx) for h in swarm]
    
    # PSO iterations
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(P):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (particle_best_h[i] - swarm[i]) + c2 * r2 * (global_best_h - swarm[i])
            
            # Update position
            h_new = swarm[i] + v
            h_new = repair_h(h_new, target_sum, dx)
            
            # Evaluate new position
            new_c5 = evaluate_c5(h_new, dx)
            
            # Update particle best
            if new_c5 < particle_best_c5[i]:
                particle_best_c5[i] = new_c5
                particle_best_h[i] = h_new.copy()
                
            # Update global best
            if new_c5 < global_best_c5:
                global_best_c5 = new_c5
                global_best_h = h_new.copy()
                
            # Update swarm and velocity
            swarm[i] = h_new.copy()
            velocities[i] = v.copy()
            
            # Early exit
            if global_best_c5 <= 0.3807:
                break
                
        # Local search on top performing particles
        if iteration % 20 == 0 and iteration > 0:
            # Get indices of top 5% particles
            sorted_indices = np.argsort(particle_best_c5)[:max(1, P // 20)]
            
            # Gradient descent on top particles
            for i in sorted_indices:
                h = particle_best_h[i].copy()
                learning_rate = 0.01
                
                # Compute gradient
                eps = 1e-4
                grad = np.zeros(n_points)
                base_c5 = particle_best_c5[i]
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, target_sum, dx)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[j] = (plus_c5 - base_c5) / eps
                    
                # Update h
                h = h - learning_rate * grad
                h = repair_h(h, target_sum, dx)
                
                # Evaluate and update particle best
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < particle_best_c5[i]:
                    particle_best_c5[i] = new_c5
                    particle_best_h[i] = h.copy()
                    
                    # Update global best if necessary
                    if new_c5 < global_best_c5:
                        global_best_c5 = new_c5
                        global_best_h = h.copy()
                        
                if global_best_c5 <= 0.3807:
                    break
        
    # Final evaluation
    if global_best_h is None:
        final_h = base_h
        final_c5 = evaluate_c5(final_h, dx)
    else:
        final_h = global_best_h
        final_c5 = global_best_c5
    
    return (final_h.tolist(), final_c5, n_points)
