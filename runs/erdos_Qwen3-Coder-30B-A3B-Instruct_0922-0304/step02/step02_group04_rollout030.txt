import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    """Project h to feasible region ensuring sum=h and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Compute overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx, eps=1e-4):
    """Compute gradient via finite differences"""
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_to_feasible(h_plus, n)
        h_minus = project_to_feasible(h_minus, n)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70

    h = project_to_feasible(h, n_points)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Annealing parameters
    temp = 0.1 * best_c5  # Initial temperature
    cooling_rate = 0.999
    min_temp = 1e-6
    
    # Gradient descent parameters
    grad_step = 0.01
    grad_decay = 0.9
    grad_steps = 5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0:
            # Apply gradient descent refinement
            for _ in range(grad_steps):
                grad = compute_gradient(h, dx)
                h = h - grad_step * grad
                h = project_to_feasible(h, n_points)
                grad_step *= grad_decay
        
        # Generate candidate
        candidate_h = h + np.random.normal(0, 0.01, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            h = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
        else:
            # Accept with probability
            delta = candidate_c5 - best_c5
            if np.random.random() < np.exp(-delta / max(temp, min_temp)):
                h = candidate_h.copy()
        
        # Update temperature
        temp *= cooling_rate
        
        # Early exit if target reached
