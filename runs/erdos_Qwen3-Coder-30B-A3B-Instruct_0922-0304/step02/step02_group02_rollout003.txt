import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric 5-fold pattern for better distribution
        h = np.zeros(n_points)
        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            # Alternate high and low values in each segment
            for j in range(start_idx, end_idx):
                h[j] = 0.5 + 0.3 * ((j - start_idx) % 2) - 0.15
            
    # Adjust to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    def objective(h_flat):
        # Reshape and normalize
        h = h_flat.reshape(-1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return evaluate_c5(h, dx)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - n_points / 2
    
    # Setup optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization parameters
    max_iter = 300
    best_c5 = float('inf')
    best_h = h.copy()
    
    # L-BFGS-B optimization with bounds
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using L-BFGS-B
        result = minimize(objective, h.flatten(), method='L-BFGS-B', 
                         bounds=bounds, constraints=cons, options={'maxiter': 20})
        
        if result.success:
            h_opt = result.x.reshape(-1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            current_c5 = evaluate_c5(h_opt, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                
            # Early exit if target reached
            if best_c5 <= 0.3807:
                break
            
            h = h_opt.copy()
        else:
            # Fallback to gradient descent if optimization fails
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            
            # Update with small step
            h = h - 0.001 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
