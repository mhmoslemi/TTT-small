import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k_max, dx):
    # Compute gradient for shift k_max
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # Approximate gradient using finite differences
        # We look at the derivative of h[i] (1 - h[i+k_max])
        # Through the overlap function
        if i + k_max < n:
            grad[i] = 1 - h[i + k_max]
        else:
            grad[i] = 0
        # Also consider the contribution from the h[i] term
        # When computing partial derivative of the overlap with respect to h[i]
        # Only the i-k_max term from h contributes to change in overlap when h[i] changes
        # But this is approximated by the correlation at shift k_max
    return grad

def line_search(h, grad, step_size, n_points, dx):
    # Perform a simple line search to find good step size
    alpha = step_size
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for _ in range(10):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points, dx)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        alpha *= 0.5

    return best_h, best_c5

def optimize_with_gradient_descent(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_size = 0.01
    step_count = 0
    relaxation_interval = 50
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_c5 = overlap[k_max] * dx

        # Compute gradient
        # We consider full correlation; for simplicity, we compute an estimate
        # Use derivative w.r.t h[i] that contributes to max_c5
        # Approximate with a single shift gradient
        grad = np.zeros(n_points)
        # For shift k_max, derivative wrt h[i] is (1 - h[i + k_max]) if k_max is positive
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = 1 - h[i + k_max]
            else:
                grad[i] = 0

        # Line search to find good step size
        h_new, c5_new = line_search(h, grad, step_size, n_points, dx)
        
        # Update if improvement
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()

        # Early exit
        if best_c5 <= 0.3807:
            break

        h = h_new
        step_count += 1

        # Periodic convex relaxation (simplified)
        if step_count % relaxation_interval == 0:
            # Add a small regularization term to encourage diversity
            # This is a rough approximation
            h += 0.0001 * np.random.normal(0, 1, n_points)
            h = project_feasible(h, n_points, dx)
            new_c5 = evaluate_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Prevent too many updates from pushing us into numerical issues
        if step_count > 1000:
            break

    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return optimize_with_gradient_descent(seed, budget_s)
