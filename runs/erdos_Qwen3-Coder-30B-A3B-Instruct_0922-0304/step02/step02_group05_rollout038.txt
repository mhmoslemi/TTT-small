import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available, otherwise use a random construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    lambda_reg = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % relaxation_interval == 0:
            # Try to escape with a convex relaxation
            try:
                h_relax = cp.Variable(n_points)
                h_constraint = cp.sum(h_relax) == n_points/2
                h_bounds = [0 <= h_relax, h_relax <= 1]
                obj = cp.Minimize(cp.sum(cp.multiply(h_relax, cp.log(h_relax))) + lambda_reg * cp.sum(h_relax))
                prob = cp.Problem(obj, [h_constraint] + h_bounds)
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_new = np.array(h_relax.value).flatten()
                    h_new = project_feasible(h_new, n_points, dx)
                    candidate_c5 = evaluate_c5(h_new, dx)
                    if candidate_c5 < best_c5:
                        h = h_new.copy()
                        best_c5 = candidate_c5
                        best_h = h.copy()
            except Exception:
                pass
        
        # Compute current overlap and gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Gradient estimate for maximum overlap shift
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        current_c5 = evaluate_c5(h, dx)
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        for alpha in line_search_steps:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_c5(h_candidate, dx)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate = h_candidate.copy()
                best_candidate_c5 = candidate_c5
        
        h = best_candidate.copy()
        current_c5 = best_candidate_c5
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adjust regularization if no improvement over several iterations
        if iteration % (relaxation_interval * 2) == 0 and iteration > 0:
            if lambda_reg > 1e-6:
                lambda_reg *= 0.5
        
        iteration += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
