import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s (clipped to [0,1])
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start with medium relaxation to escape local minima
    relaxation_schedule = np.logspace(np.log10(0.5), 0.0, 50)  # From 0.5 to 1.0
    
    for mu in relaxation_schedule:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Define variables
        h_var = cp.Variable(n_points)
        
        # Objective: relaxed version minimizing both C5 and deviation from 0.5
        c5_obj = cp.Constant(0)
        
        # Get current c5 for reference - need to compute it in this context
        current_c5 = evaluate_c5(h, dx)
        
        # The relaxed objective: weighted combination of c5 and deviation from 0.5
        objective = cp.Minimize(
            mu * current_c5 + (1 - mu) * cp.sum_squares(h_var - 0.5)
        )
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Create and solve problem using CVXPY with ECOS solver
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Enforce the constraint exactly
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Compute the new c5
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    
                if best_c5 <= 0.3807:
                    break
        except Exception as e:
            # If CVX fails, fall back to previous h
            pass
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
