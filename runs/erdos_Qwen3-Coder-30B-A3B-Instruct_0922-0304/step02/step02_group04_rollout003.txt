import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) / 2
    return h

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    # Simplified gradient computation using finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    c5_base = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Algorithm parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iter = 500
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, n_points)
        particles.append(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = project_to_feasible(h_init, n_points)
        particles.append(h_init)
    
    # Generate additional particles
    for _ in range(swarm_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
    
    # Initialize velocities and personal bests
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new position
            current_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_bests[i] = particles[i].copy()
                personal_best_scores[i] = current_score
                
                # Update global best
                if current_score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = current_score
        
        # Optional local refinement every few iterations
        if iteration % 20 == 0 and iteration > 0:
            # Refine top 10% of particles
            top_indices = np.argsort(personal_best_scores)[:swarm_size//10]
            for idx in top_indices:
                h_refined = personal_bests[idx].copy()
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h_refined, dx)
                    h_refined = h_refined - step_size * grad
                    h_refined = project_to_feasible(h_refined, n_points)
                personal_bests[idx] = h_refined
                personal_best_scores[idx] = evaluate_c5(h_refined, dx)
                
                if personal_best_scores[idx] < global_best_score:
                    global_best = h_refined.copy()
                    global_best_score = personal_best_scores[idx]
        
        # Early exit if target is reached
        if global_best_score <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_score, n_points)
