import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    if 'initial_h_values' in globals():
        # Use initial solution as base for half particles
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        for i in range(n_particles // 2):
            h = h_base + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
        
        # Generate random particles for the rest
        for i in range(n_particles // 2):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
    else:
        # All random initialization
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
    
    # Initialize velocities, personal bests, and global best
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = particles.copy()
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w = 0.8  # inertia weight
    c1 = 1.2  # cognitive weight
    c2 = 1.2  # social weight
    mutation_rate = 0.05
    
    # Iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            v = w * v + c1 * r1 * (p - h) + c2 * r2 * (gbest - h)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new position
            current_score = evaluate_c5(h_new, dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = current_score
                
                # Update global best if applicable
                if current_score < gbest_score:
                    gbest = h_new.copy()
                    gbest_score = current_score
            
            # Store updated values
            particles[i] = h_new
            velocities[i] = v
        
        # Early exit if target reached
        if gbest_score <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
