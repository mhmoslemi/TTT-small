import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def enforce_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a coarse grid for faster initial search
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize from the parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]  # Trim if necessary
    else:
        # Create a simple initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        
    # Enforce initial feasibility
    h = enforce_feasibility(h, n_points)
    
    # Objective function
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Constraint for the integral
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with scipy
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=bounds, 
                   constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-6})
    
    h = res.x
    h = enforce_feasibility(h, n_points)
    
    # Refine at higher resolution
    n_points = 70
    dx = 2.0 / n_points
    
    # Interpolate initial solution to higher resolution
    h_coarse = h.copy()
    h = np.interp(np.linspace(0, 2, n_points), 
                  np.linspace(0, 2, len(h_coarse)), 
                  h_coarse)
    
    h = enforce_feasibility(h, n_points)
    
    # Second optimization pass
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    res = minimize(objective, h, method='L-BFGS-B', 
                   bounds=bounds, 
                   constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-6})
    
    h = res.x
    h = enforce_feasibility(h, n_points)
    
    c5 = evaluate_overlap(h, dx)
    
    # Final evaluation to verify we're close to target
    while time.time() - start_time < budget_s - 1:
        # Try a few gradient steps to fine tune
        grad = np.zeros(n_points)
        eps = 1e-4
        base_c5 = c5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = enforce_feasibility(h_plus, n_points)
            c5_plus = evaluate_overlap(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update with small step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = enforce_feasibility(h_new, n_points)
        c5_new = evaluate_overlap(h_new, dx)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            step_size *= 0.8
            if step_size < 1e-6:
                break
        if c5 <= 0.3807:
            break
    
    return (h.tolist(), c5, n_points)
