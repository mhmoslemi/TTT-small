import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_population(n_points, population_size):
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create a basic starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add more diversified individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    return population

def differential_evolution_step(population, fitness, F=0.7, CR=0.9):
    n_points = len(population[0])
    new_population = []
    
    for i in range(len(population)):
        # Select three random individuals different from current
        candidates = list(range(len(population)))
        candidates.remove(i)
        a, b, c = random.sample(candidates, 3)
        
        # Mutation
        v = population[a] + F * (population[b] - population[c])
        
        # Crossover
        crossover_mask = np.random.rand(n_points) < CR
        u = np.where(crossover_mask, v, population[i])
        
        # Ensure feasibility
        u = project_to_feasible(u, n_points)
        new_population.append(u)
    
    return new_population

def gradient_descent_refinement(h, dx, iterations=50, step_size=0.01):
    for _ in range(iterations):
        c5_val = evaluate_c5(h, dx)
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_val) / eps
        
        # Update
        h = h - step_size * grad
        h = project_to_feasible(h, len(h))
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    F = 0.7
    CR = 0.9
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    
    # Evaluate fitness
    fitness = [evaluate_c5(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_solution = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main DE loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Apply differential evolution
        new_population = differential_evolution_step(population, fitness, F, CR)
        
        # Apply gradient descent refinement to top individual every 50 gens
        if gen % 50 == 0 and gen > 0:
            refined = gradient_descent_refinement(population[0].copy(), dx)
            refined_c5 = evaluate_c5(refined, dx)
            if refined_c5 < fitness[0]:
                new_population[0] = refined
        
        # Keep population size
        population = new_population
        fitness = [evaluate_c5(h, dx) for h in population]
    
    return (best_solution.tolist(), best_c5, n_points)
