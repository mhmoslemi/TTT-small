import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as seed for some particles
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        for i in range(n_particles // 2):
            particles.append(h_init.copy())
            velocities.append(np.zeros(n_points))
        
        # Random particles
        for i in range(n_particles // 2, n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # All random
        for i in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Keep track of best positions
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    # Iterate
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best if needed
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
                    
        # Apply mutation occasionally
        if np.random.rand() < mutation_rate:
            idx = np.random.randint(n_particles)
            noise = np.random.normal(0, 0.05, n_points)
            particles[idx] += noise
            particles[idx] = np.clip(particles[idx], 0, 1)
            particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
            score = evaluate_c5(particles[idx], dx)
            if score < pbest_scores[idx]:
                pbest_scores[idx] = score
                pbest[idx] = particles[idx].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[idx].copy()
        
        # Early exit if target achieved
        if gbest_score <= 0.3807:
            break
    
    # Final evaluation
    final_score = evaluate_c5(gbest, dx)
    
    return (gbest.tolist(), final_score, n_points)
