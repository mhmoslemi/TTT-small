import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        n_points = len(h)
        dx = 2.0 / n_points
        h = h * (n_points / 2) / np.sum(h)
    else:
        n_points = 70
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5  # Simple initialization
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization parameters
    max_iterations = 2000
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps and find max
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(np.abs(overlaps))
        max_overlap = overlaps[max_overlap_idx]
        
        # Compute gradient approximation
        # For shift k where overlap is maximized
        k = max_overlap_idx - (len(h) - 1)
        h_shifted = np.roll(h, k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search for step size
        best_candidate_h = h.copy()
        best_candidate_c5 = best_c5
        
        for step_size in line_search_steps:
            candidate_h = h - step_size * gradient
            candidate_h = project_feasible(candidate_h, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_h = candidate_h
                best_candidate_c5 = candidate_c5
        
        h = best_candidate_h
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = h.copy()
            
        # Periodic relaxation to escape local optima
        if iteration % 20 == 0 and iteration > 0:
            # Convex relaxation approach
            try:
                h_relax = cp.Variable(n_points)
                objective = cp.Minimize(
                    cp.sum(cp.multiply(h_relax, 1 - h_relax)) + 
                    lambda_reg * cp.sum(cp.entr(h_relax))  # KL regularization
                )
                
                constraints = [
                    cp.sum(h_relax) == n_points/2,
                    h_relax >= 0,
                    h_relax <= 1
                ]
                
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_relax.value
                    h_new = project_feasible(h_new, n_points, dx)
                    c5_new = evaluate_erdos_solution(h_new)
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                
            except Exception as e:
                pass
            
            # Adjust regularization if no improvement
            if best_c5 >= 0.3807:
                lambda_reg *= 0.8
            else:
                lambda_reg *= 0.95
                
        if best_c5 <= 0.3807:
            break
    
    # Final re-evaluate
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
