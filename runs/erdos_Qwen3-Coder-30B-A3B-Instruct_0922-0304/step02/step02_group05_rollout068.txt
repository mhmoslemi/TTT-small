import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h."""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        n_points = len(h)
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent with line search
    max_iterations = 10000
    relaxation_interval = 20
    lambda_reg = 0.01
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Check if we should do convex relaxation
        if iteration % relaxation_interval == 0 and iteration > 0:
            # Try convex relaxation to escape local minima
            try:
                h_relaxed = solve_convex_relaxation(h, n_points, lambda_reg)
                h_relaxed = project_feasible(h_relaxed, n_points, dx)
                c5_relaxed = evaluate_erdos_solution(h_relaxed)
                if c5_relaxed < best_c5:
                    best_h = h_relaxed.copy()
                    best_c5 = c5_relaxed
                h = h_relaxed
            except Exception:
                pass
        
        # Compute overlaps and find the maximum
        overlaps = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient for this shift
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Try different step sizes using line search
        step_sizes = [0.1, 0.5, 1.0]
        best_step = 0
        best_h_candidate = h.copy()
        best_c5_candidate = best_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < best_c5_candidate:
                best_c5_candidate = c5_candidate
                best_h_candidate = h_candidate.copy()
                best_step = alpha
                
        if best_c5_candidate < best_c5:
            h = best_h_candidate.copy()
            best_c5 = best_c5_candidate
            
            # Update best solution
            if best_c5_candidate < evaluate_erdos_solution(best_h):
                best_h = h.copy()
        
        # Adjust lambda if no improvement
        if iteration % relaxation_interval == relaxation_interval - 1:
            if best_c5 > evaluate_erdos_solution(h) * 1.01:
                lambda_reg *= 0.5
            else:
                lambda_reg = min(lambda_reg * 1.1, 1.0)
    
    return (best_h.tolist(), best_c5, n_points)

def solve_convex_relaxation(h, n_points, lambda_reg):
    """Solve convex relaxation of the optimization problem."""
    # Variable
    h_var = cp.Variable(n_points)
    
    # Objective: max overlap + lambda * sum(h * log(h))
    # Since log is concave, this is a convex problem if constraints are satisfied
    
    # Approximate overlap term using the maximum in the current solution
    overlap_vec = correlate(h, 1 - h, mode="full")    
    max_k_idx = np.argmax(overlap_vec)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Objective: minimize -overlap_k where overlap_k is from current solution
    H_shifted = np.roll(h, max_k_idx)
    target_overlap = np.dot(h, 1 - H_shifted) 
    
    # Minimize penalty term: -max_overlap + lambda * sum(h * log(h))
    # Note: log(h) is handled using cp.entrl for numerical stability
    objective = cp.Minimize(-target_overlap + lambda_reg * cp.sum(cp.entr(h_var)))
    
    # Solve problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    
    # Return solution if successful
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h_var.value
    else:
        return h  # Return original if not solving well
