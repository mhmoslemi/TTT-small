import numpy as np
from scipy.signal import correlate
import time
import random

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    # Compute gradient via finite differences
    eps = 1e-4
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 30
    max_iter = 500
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Add random individuals
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Initialize personal best positions
    personal_best = [p.copy() for p in swarm]
    personal_best_c5 = [evaluate_c5(p, dx) for p in swarm]
    
    # Initialize global best
    best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[best_idx].copy()
    global_best_c5 = personal_best_c5[best_idx]
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best[i] = swarm[i].copy()
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best = swarm[i].copy()
            
            # Early exit
            if global_best_c5 <= 0.3807:
                break
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0 and iteration > 0:
            # Refine top 10% particles
            top_indices = np.argsort(personal_best_c5)[:swarm_size // 10]
            for i in top_indices:
                h_current = personal_best[i].copy()
                step_size = 0.01
                for _ in range(50):
                    if time.time() - start_time > budget_s - 1:
                        break
                    grad = compute_gradient(h_current, dx)
                    h_current -= step_size * grad
                    h_current = project_to_feasible(h_current, n_points)
                    new_c5 = evaluate_c5(h_current, dx)
                    if new_c5 < personal_best_c5[i]:
                        personal_best_c5[i] = new_c5
                        personal_best[i] = h_current.copy()
                        if new_c5 < global_best_c5:
                            global_best_c5 = new_c5
                            global_best = h_current.copy()
                    else:
                        step_size *= 0.95  # Reduce step size if no improvement
    
    return (global_best.tolist(), global_best_c5, n_points)
