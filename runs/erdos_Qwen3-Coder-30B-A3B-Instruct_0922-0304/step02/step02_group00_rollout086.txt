import numpy as np
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    epsilon = 0.05
    T_initial = 1.0
    beta = 0.995
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Linear initialization with bounds
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing variables
    T = T_initial
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Simulated annealing loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Perturb the function
        h_candidate = h + np.random.normal(0, epsilon * T, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Rescale to satisfy constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept with probability or always if better
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        # Cooling
        if iteration % 100 == 0:
            T = beta * T
            
        # Reset temperature if no improvement for too long
        if last_improvement > 1000:
            T = T_initial
            last_improvement = 0
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
