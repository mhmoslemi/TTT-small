import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    best_c5 = float('inf')

    rho = 1.0

    tau = np.zeros(2 * n_points - 1)
    
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5

        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Solve convex subproblem using CVXPY with AL terms
        h_var = cp.Variable(n_points)
        
        # Objective: minimize max overlap term plus AL terms
        overlap_expr = cp.Variable(2 * n_points - 1)
        
        # Create constraints to compute correlation via inequalities
        # This is a simplification - we directly use correlation from numpy for this step
        # and focus AL on solving for h
        
        # For a tight constraint, we start by solving with penalty term
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # For the AL part we use the structure
        # To minimize AL objective: sum_i (overlap_i + (rho/2 ||overlap - tau||^2))

        # Convex subproblem: solve for h to reduce largest overlap        
        # We'll solve the convex relaxation directly with warm start
        obj = cp.Minimize(cp.sum_squares(h_var - h))
        
        # We solve a simpler proximal step for now
        # See if we can make progress by improving the specific overlap axis
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # fallback to previous solution 
            h_new = h.copy()
        else:
            h_new = h_var.value
            
        # Update h with small step from previous and ensure feasibility
        h = h_new
        
        # Maintain sum and bounds

        # Update lambda multipliers
        corr_new = correlate(h, 1 - h, mode="full")
        overlap_vec = corr_new * dx
        
        # Update lagrange multipliers
        lambda_mult += rho * (overlap_vec - tau)
        
        # Update penalty if needed
        if np.any(np.abs(overlap_vec - tau) > 1e-3):
            rho *= 1.1
        
        # Update constraint bounds
        tau = overlap_vec.copy()

    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
