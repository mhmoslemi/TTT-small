import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with uniform base and slight randomization
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # AL parameters
    rho = 1.0
    max_iter = 500
    penalty_increase = 2.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

        # Convex subproblem using AL
        h_var = Variable(n_points)
        objective = cp.Minimize(cp_sum(h_var) * dx)  # This is just to make a convex problem, we'll use constraint-based approach

        # Create constraints for AL subproblem
        # Note: this is a simplified LA approach that works well in practice
        # More precisely, we construct the AL term: this is a heuristic approximation

        # Solve the convex subproblem with constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # Solve the problem
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
        
        # Update with AL-like penalty
        h = h_new
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final check
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
