import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 100000
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Perturbation parameters
    delta = 0.1
    perturb_fraction = 0.1
    
    iter_count = 0
    
    while T > T_min and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        # Create candidate solution by perturbing h
        h_new = h.copy()
        num_perturb = max(1, int(n_points * perturb_fraction))
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Increase selected values
        for i in perturb_indices:
            h_new[i] += delta
        # Decrease others to keep sum = 35
        total_increase = np.sum(h_new[perturb_indices]) - np.sum(h[perturb_indices])
        decrease_amount = total_increase / (n_points - num_perturb)
        for i in range(n_points):
            if i not in perturb_indices:
                h_new[i] -= decrease_amount
                
        # Ensure h_new is within bounds
        h_new = np.clip(h_new, 0, 1)
        # Normalize to sum = 35
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            # Accept with probability based on temperature
            delta_E = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta_E / T):
                h = h_new
        
        # Cool down
        T *= alpha
        
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
