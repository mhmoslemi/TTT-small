import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    # Ensure sum constraint and bounds
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with structured approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
    else:
        # Create a symmetric 5-fold pattern
        segment_size = n_points // 5
        h = np.zeros(n_points)
        for i in range(5):
            start = i * segment_size
            end = (i + 1) * segment_size
            if i % 2 == 0:
                h[start:end] = 0.5 + 0.3
            else:
                h[start:end] = 0.5 - 0.3
        h = repair_h(h, n_points)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer function
    def obj_and_grad(h_vec):
        h = h_vec.copy()
        h = repair_h(h, n_points)
        c5 = evaluate_c5(h, dx)
        return c5, np.zeros_like(h_vec)  # dummy gradient for now
        
    # Optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    options = {'maxiter': 1000, 'disp': False, 'gtol': 1e-6}
    
    # Initial optimization step
    res = minimize(obj_and_grad, h, method='L-BFGS-B', bounds=bounds, 
                   options=options, jac=False)
    h = res.x
    h = repair_h(h, n_points)
    
    # Local refinement
    max_iter = 1000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try local search by small perturbations
        h_perturbed = h.copy()
        # Slightly adjust values
        indices = np.random.choice(n_points, size=min(10, n_points//4), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.05, 0.05)
            h_perturbed[i] = h[i] + delta
        h_perturbed = repair_h(h_perturbed, n_points)
        
        c5_pert = evaluate_c5(h_perturbed, dx)
        if c5_pert < current_c5:
            h = h_perturbed.copy()
            h = repair_h(h, n_points)
    
    # Final check
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
