import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 50000
    
    # Perturbation parameters
    delta = 0.1
    perturb_frac = 0.1
    num_perturb = max(1, int(n_points * perturb_frac))
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new candidate solution
        h_new = h.copy()
        
        # Select random indices to perturb
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Increase selected indices
        for i in indices:
            h_new[i] += delta
        # Decrease the rest to maintain sum
        decrease_amount = delta * num_perturb
        decrease_indices = np.setdiff1d(np.arange(n_points), indices)
        if len(decrease_indices) > 0:
            decrease_per_index = decrease_amount / len(decrease_indices)
            for i in decrease_indices:
                h_new[i] -= decrease_per_index
        
        # Clamp to [0,1] and rescale
        h_new = np.clip(h_new, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale_factor
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        c5_old = evaluate_c5(h, dx)
        delta_e = c5_new - c5_old
        
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Cool down
        T = max(T_min, alpha * T)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
