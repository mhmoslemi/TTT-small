import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, k_star, dx):
    n = len(h)
    grad = np.zeros(n)
    
    # Compute gradient for the specific shift k_star
    if k_star >= 0:
        # h[i] * (1 - h[i + k_star]) for i = 0 to n - k_star - 1
        for i in range(n - k_star):
            grad[i] += 1 - h[i + k_star]
            if i + k_star < n:
                grad[i + k_star] -= h[i]
    else:
        # k_star is negative, so we're looking at h[i + |k_star|] * (1 - h[i])
        k_abs = -k_star
        for i in range(n - k_abs):
            grad[i] += -(1 - h[i + k_abs])
            if i + k_abs < n:
                grad[i + k_abs] -= h[i]
    
    return grad * dx

def project_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, target_sum)
    else:
        # Fallback initialization: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_feasible(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 5000
    step_size = 0.01
    momentum = 0.9
    prev_update = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        overlap_values = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_values)
        # Convert index back to shift k
        k_star = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, k_star, dx)
        
        # Update with momentum and step size
        update = momentum * prev_update - step_size * grad
        h = h + update
        
        # Project back to feasible region
        h = project_feasible(h, target_sum)
        
        prev_update = update
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
