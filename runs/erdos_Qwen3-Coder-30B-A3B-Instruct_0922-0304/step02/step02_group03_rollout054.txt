import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Two-stage approach: coarse then fine
    coarse_n = 35
    fine_n = 70
    dx_coarse = 2.0 / coarse_n
    dx_fine = 2.0 / fine_n
    
    best_c5 = float('inf')
    best_h = None
    
    # Stage 1: Coarse optimization with n=35
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
    else:
        h_coarse = np.ones(coarse_n)
        h_coarse[coarse_n//2:] = 0
        h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
    
    # Define constraint function for sum = n/2
    def constraint_sum(h):
        return np.sum(h) - coarse_n / 2
    
    # Optimization with L-BFGS-B
    bounds = [(0, 1) for _ in range(coarse_n)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    max_iter_coarse = 500
    step_size = 0.1
    
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h_coarse, dx_coarse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_coarse.copy()
            
        result = minimize(
            lambda h: evaluate_c5(h, dx_coarse),
            h_coarse,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-5}
        )
        
        if result.success:
            h_coarse = result.x
        else:
            # Fallback: simple projected gradient
            h_coarse = h_coarse - step_size * np.random.normal(0, 0.01, coarse_n)
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
    
    # Stage 2: Refinement with n=70
    h_fine = np.interp(np.linspace(0, 2, fine_n), np.linspace(0, 2, coarse_n), h_coarse)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
    
    # Add random perturbations at 10% level
    perturb = np.random.uniform(-0.1, 0.1, fine_n)
    h_fine += perturb
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
    
    bounds = [(0, 1) for _ in range(fine_n)]
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - fine_n / 2}
    
    max_iter_fine = 1000
    step_size = 0.1
    
    for iteration in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h_fine, dx_fine)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            
        result = minimize(
            lambda h: evaluate_c5(h, dx_fine),
            h_fine,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-5}
        )
        
        if result.success:
            h_fine = result.x
        else:
            # Fallback: simple projected gradient
            h_fine = h_fine - step_size * np.random.normal(0, 0.01, fine_n)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (fine_n / 2) / np.sum(h_fine)
    
    # Ensure final evaluation is accurate
    final_c5 = evaluate_c5(best_h, dx_fine)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, fine_n)
