import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, target_sum)
    else:
        # Create a basic block pattern
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        h0 = project_to_feasible(h0, target_sum)
    
    # Add one particle with initial solution
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    
    # Add other particles with blocky patterns + noise
    for _ in range(n_particles - 1):
        h = np.zeros(n_points)
        # Two blocks
        mid = n_points // 2
        h[:mid] = 1.0
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))

    # PSO parameters
    alpha, beta = 0.8, 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track best positions
    best_positions = [p.copy() for p in particles]
    best_values = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_values)
    global_best_value = best_values[global_best_idx]
    global_best_position = best_positions[global_best_idx].copy()
    
    # PSO iterations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        inertia *= (1 - inertia_decay)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v = (inertia * velocities[i] + 
                 alpha * r1 * (best_positions[i] - particles[i]) +
                 beta * r2 * (global_best_position - particles[i]))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[i] = v
            
            # Update position
            new_pos = particles[i] + v
            new_pos = project_to_feasible(new_pos, target_sum)
            
            # Evaluate
            new_c5 = evaluate_c5(new_pos, dx)
            
            # Update personal best
            if new_c5 < best_values[i]:
                best_values[i] = new_c5
                best_positions[i] = new_pos.copy()
                
                # Update global best
                if new_c5 < global_best_value:
                    global_best_value = new_c5
                    global_best_position = new_pos.copy()
            
            particles[i] = new_pos
    
    # Final evaluation
    final_c5 = evaluate_c5(global_best_position, dx)
    
    return (global_best_position.tolist(), final_c5, n_points)
