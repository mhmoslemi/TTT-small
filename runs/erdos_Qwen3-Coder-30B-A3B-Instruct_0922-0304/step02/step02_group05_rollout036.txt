import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a simple construction  
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    max_iterations = int(budget_s * 10)  # Roughly estimate iterations
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find index of max
        overlaps = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(np.abs(overlaps))
        max_overlap = overlaps[max_k]
        
        # Compute gradient approximation for shift k
        # h_shifted[i] = h[(i + k) % n_points]
        h_shifted = np.roll(h, max_k)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        current_c5 = evaluate_erdos_solution(h)
        best_candidate_h = h.copy()
        best_candidate_c5 = current_c5
        
        for alpha in line_search_steps:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_h = h_candidate
                best_candidate_c5 = candidate_c5
        
        # Update
        h = best_candidate_h
        current_c5 = best_candidate_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Periodically solve convex relaxation to escape local minima
        if iteration % relaxation_interval == 0 and iteration > 0:
            # Convex relaxation
            h_relax = cp.Variable(n_points, pos=True)
            objective = cp.Minimize(
                cp.sum(cp.multiply(overlaps, cp.Variable(len(overlaps), nonneg=True))) + 
                lambda_reg * cp.sum(cp.entr(h_relax))  # KL-divergence regularization
            )
            constraints = [
                cp.sum(h_relax) == n_points / 2,
                h_relax >= 0,
                h_relax <= 1
            ]
            
            try:
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False)
                
                # Check relaxation was successful
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_relax_sol = h_relax.value
                    h_relax_sol = np.clip(h_relax_sol, 0, 1)
                    h_relax_sol = h_relax_sol * (n_points / 2) / np.sum(h_relax_sol)
                    
                    # Evaluate this new point
                    relax_c5 = evaluate_erdos_solution(h_relax_sol)
                    if relax_c5 < current_c5:
                        h = h_relax_sol
                        current_c5 = relax_c5
                        
                # Reduce lambda if no improvement from relaxation
                if current_c5 >= best_c5:
                    lambda_reg *= 0.5
                else:
                    lambda_reg = max(0.0001, lambda_reg)  # Prevent too small
                    
            except Exception:
                # If convex relaxation fails, continue with original approach
                lambda_reg *= 0.5
                
        # Occasionally refresh h with noise to help exploration
        if iteration % 100 == 0 and iteration > 0:
            h = project_feasible(h + np.random.normal(0, 0.01, n_points), n_points, dx)
    
    return (best_h.tolist(), best_c5, n_points)
