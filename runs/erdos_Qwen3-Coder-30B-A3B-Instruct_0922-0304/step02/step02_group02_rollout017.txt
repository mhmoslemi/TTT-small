import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm
    n_particles = 50
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    else:
        # Simple initialization
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    # Create initial swarm
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.1 * np.max(base_h), n_points)
        h = repair_h(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    inertia = 0.7
    c1 = c2 = 2.0
    max_iter = 500
    
    global_best = float('inf')
    global_best_h = base_h.copy()
    
    # Local search parameters
    local_search_fraction = 0.05
    local_max_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate particles
        particle_best_c5 = np.full(n_particles, float('inf'))
        particle_best_h = [None] * n_particles
        
        for p in range(n_particles):
            h = swarm[p]
            c5 = evaluate_c5(h, dx)
            if c5 < particle_best_c5[p]:
                particle_best_c5[p] = c5
                particle_best_h[p] = h.copy()
            
            if c5 < global_best:
                global_best = c5
                global_best_h = h.copy()
                
        # Early exit
        if global_best <= 0.3807:
            break
            
        # Update particles
        for p in range(n_particles):
            r1, r2 = np.random.random(2)
            v = inertia * velocities[p] + c1 * r1 * (particle_best_h[p] - swarm[p]) + c2 * r2 * (global_best_h - swarm[p])
            velocities[p] = v
            new_h = swarm[p] + v
            
            # Repair and update
            new_h = repair_h(new_h, n_points)
            swarm[p] = new_h
            
        # Local search on top particles
        if iteration >= 100:  # Start local search after some iterations
            top_k = max(1, int(n_particles * local_search_fraction))
            particle_indices = np.argsort(particle_best_c5)[:top_k]
            
            for p_idx in particle_indices:
                h = swarm[p_idx]
                h_best = h.copy()
                best_c5 = particle_best_c5[p_idx]
                
                # Gradient descent local search
                learning_rate = 0.01
                for _ in range(local_max_iter):
                    if time.time() - start_time > budget_s - 1:
                        break
                        
                    eps = 1e-4
                    grad = np.zeros_like(h)
                    base_c5 = evaluate_c5(h, dx)
                    
                    for i in range(n_points):
                        h_plus = h.copy()
                        h_plus[i] += eps
                        h_plus = repair_h(h_plus, n_points)
                        plus_c5 = evaluate_c5(h_plus, dx)
                        grad[i] = (plus_c5 - base_c5) / eps
                    
                    h = h - learning_rate * grad
                    h = repair_h(h, n_points)
                    
                    new_c5 = evaluate_c5(h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        h_best = h.copy()
                        
                # Update particle if better found
                if best_c5 < particle_best_c5[p_idx]:
                    swarm[p_idx] = h_best.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
