import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def scale_to_target_sum(h, target_sum):
    """Scale h to have the target sum while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        return h
    scaled = h * target_sum / current_sum
    return np.clip(scaled, 0, 1)

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover with probability crossover_rate."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.05):
    """Gaussian mutation with probability mutation_rate."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, sigma)
    return mutated

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = []
    
    # Add diverse initial solutions
    for _ in range(population_size - len(population)):
        h = np.random.uniform(0, 1, n_points)
        h = scale_to_target_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = population[0].copy()
    
    generations_no_improvement = 0
    max_generations_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = evaluate_c5(ind, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = ind.copy()
                generations_no_improvement = 0
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament selection)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = np.random.choice(len(population), tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        for i in range(0, population_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child1 = scale_to_target_sum(child1, target_sum)
            
            child2 = mutate(child2)
            child2 = scale_to_target_sum(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        generations_no_improvement += 1
        if generations_no_improvement >= max_generations_no_improvement:
            break
    
    return (best_individual.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
