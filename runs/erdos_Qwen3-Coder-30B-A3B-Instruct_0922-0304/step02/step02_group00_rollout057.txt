import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Use initial solution for 25 particles, random for others
        for i in range(n_particles):
            if i < 25:
                particles.append(h_init.copy())
            else:
                h_rand = np.random.rand(n_points)
                h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                particles.append(h_rand)
    else:
        # All random initialization
        for _ in range(n_particles):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            particles.append(h_rand)
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.zeros(n_points))
    
    # Initialize personal bests and global best
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    inertia_decay = 0.99
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        if iteration > 100:
            w *= inertia_decay
            
        for i in range(n_particles):
            # Get current particle
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (pbest[i] - h)
            social = c2 * r2 * (gbest - h)
            v = w * v + cognitive + social
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            current_c5 = evaluate_c5(h_new, dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = h_new.copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest = h_new.copy()
                    gbest_value = current_c5
                    
            # Store updated values
            particles[i] = h_new
            velocities[i] = v
            
        # Early exit if target achieved
        if gbest_value <= 0.3807:
            break
    
    # Final verification
    final_c5 = evaluate_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
