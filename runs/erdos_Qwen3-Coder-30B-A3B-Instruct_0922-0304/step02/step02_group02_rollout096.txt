import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters for SA
    T0 = 1000.0
    alpha = 0.999
    Tmin = 1.0
    T = T0
    max_iter = 10000
    
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Placeholder for perturbation size (will get updated)
    delta = 0.1
    
    iter_count = 0
    while T > Tmin and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        # Create candidate solution by perturbing h
        h_candidate = h.copy()
        
        # Select indices to modify (10% of points)
        num_modify = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_modify, replace=False)
        
        # Ensure we don't go out of bounds
        delta_allowed = np.minimum(h_candidate[indices], delta)
        h_candidate[indices] += delta_allowed
        
        # Reduce other points to keep sum constant
        total_increase = np.sum(delta_allowed)
        other_indices = np.setdiff1d(np.arange(n_points), indices)
        if len(other_indices) > 0:
            decrease_amount = total_increase / len(other_indices)
            h_candidate[other_indices] -= decrease_amount
            # Ensure non-negativity
            h_candidate = np.clip(h_candidate, 0, 1)
        
        # Re-normalize to maintain constraint sum(h) = n_points / 2
        scaling_factor = (n_points / 2) / np.sum(h_candidate)
        h_candidate *= scaling_factor
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Accept or reject
        delta_E = candidate_c5 - current_c5
        
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_candidate
            current_c5 = candidate_c5
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Cool down
        T = alpha * T

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
