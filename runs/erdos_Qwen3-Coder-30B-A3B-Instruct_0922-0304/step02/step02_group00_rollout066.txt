import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    rho = 1.0
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Convex optimization step
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # AL approach: minimize sum of overlap + penalty term
        overlap = correlate(h_var.value, 1 - h_var.value, mode="full")
        obj = cp.Minimize(cp.max(overlap) + 0.5 * rho * cp.sum_squares(h_var - h))
        
        problem = cp.Problem(obj, constraints)
        
        # Try different solvers
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                problem.solve(solver=cp.CVXOPT, verbose=False)
            except:
                problem.solve(solver=cp.SCS, verbose=False)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_var.value
            h = np.clip(h_new, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Fallback: gradient descent if optimization fails
            lr = 0.01
            for _ in range(20):
                corr_h = correlate(h, 1 - h, mode="full")
                max_idx = np.argmax(corr_h)
                
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    grad[i] = (corr_plus[max_idx] - corr_h[max_idx]) / eps
                
                h -= lr * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Update penalty
        try:
            corr = correlate(h, 1 - h, mode="full")
            current_c5 = np.max(corr) * dx
            if current_c5 > best_c5:
                rho *= 2.0
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
