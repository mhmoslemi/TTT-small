import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    # Compute the overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem setup
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 3000  # approximate max iterations based on budget
    
    # Initialize particles (PSO parameters)
    n_particles = 50
    w = 1.0  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    v_max = 0.2  # max velocity
    
    # Initialize particles
    particles = []
    velocities = []
    
    # If initial_h_values exists, use it
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0)
    else:
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        particles.append(h0)
    
    # Fill rest of particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h)
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.normal(0, 0.01, n_points))
    
    # Evaluate initial positions
    fitness = []
    for i in range(n_particles):
        fit = evaluate_erdos_solution(particles[i])
        fitness.append(fit)
    
    # Set personal bests and global best
    pbest_positions = particles.copy()
    pbest_fitness = fitness.copy()
    gbest_idx = np.argmin(pbest_fitness)
    gbest_fitness = pbest_fitness[gbest_idx]
    gbest_position = particles[gbest_idx].copy()
    
    # Track best solution
    best_c5 = gbest_fitness
    best_h = gbest_position.copy()
    
    # PSO iterations
    iter_count = 0
    stagnation_count = 0
    prev_best = best_c5
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Maintain feasibility
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            fit = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if fit < pbest_fitness[i]:
                pbest_fitness[i] = fit
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if fit < gbest_fitness:
                    gbest_fitness = fit
                    gbest_position = particles[i].copy()
                    
                    # Update best solution
                    best_c5 = fit
                    best_h = particles[i].copy()
                    stagnation_count = 0
            else:
                stagnation_count += 1
                
            # Reduce inertia if stagnating
            if iter_count % 50 == 0 and stagnation_count > 20:
                w = max(0.8, w * 0.95)
                stagnation_count = 0
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Reset stagnant particles occasionally
        if iter_count % 100 == 0:
            for i in range(n_particles):
                if pbest_fitness[i] == float('inf') or abs(pbest_fitness[i] - gbest_fitness) > 1e-5:
                    particles[i] = np.random.rand(n_points)
                    particles[i] = project_feasible(particles[i], n_points, dx)
                    pbest_positions[i] = particles[i].copy()
                    pbest_fitness[i] = evaluate_erdos_solution(particles[i])
    
    # Final evaluation of best solution
    best_c5_final = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5_final, n_points)
