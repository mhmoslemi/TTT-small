import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to maintain feasibility constraints"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles (PSO swarm)
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create a random initial solution
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize remaining particles
    for _ in range(num_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(num_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial fitness
    fitness = []
    for h in particles:
        fit = evaluate_erdos_solution(h)
        fitness.append(fit)
    
    # Keep track of personal bests
    p_best = particles.copy()
    p_best_fitness = fitness.copy()
    
    # Global best
    best_idx = np.argmin(fitness)
    g_best = particles[best_idx].copy()
    g_best_fitness = fitness[best_idx]
    
    # PSO parameters
    inertia_weight = 1.0
    c1 = 2.0   # Cognitive coefficient
    c2 = 2.0   # Social coefficient
    velocity_max = 0.2
    velocity_min = -0.2
    
    # PSO iterations
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        iteration += 1
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], velocity_min, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible set
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate fitness
            current_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = current_fitness
            
            # Update global best
            if current_fitness < g_best_fitness:
                g_best = particles[i].copy()
                g_best_fitness = current_fitness
                last_improvement = iteration
        
        # Adaptive inertia
        if iteration % 50 == 0:
            inertia_weight = max(0.8, inertia_weight - 0.02)
        
        # Occasionally reset stagnated particles
        if iteration - last_improvement > 100:
            for i in range(1, num_particles):  # Skip the g_best
                if np.random.random() < 0.2:
                    h = np.random.uniform(0, 1, n_points)
                    h = project_feasible(h, n_points, dx)
                    particles[i] = h
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                    fitness[i] = evaluate_erdos_solution(particles[i])
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = fitness[i]
        
        # Early exit if target found
        if g_best_fitness <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_fitness, n_points)
