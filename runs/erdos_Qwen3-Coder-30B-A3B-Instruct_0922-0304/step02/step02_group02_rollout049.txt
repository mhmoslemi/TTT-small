import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 10000
    
    # Perturbation parameters
    perturbation_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Create perturbed solution
        h_prime = h.copy()
        
        # Select random indices to perturb (10% of points)
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb selected indices
        delta = perturbation_size / n_perturb
        for i in indices:
            h_prime[i] += delta
            if h_prime[i] > 1:
                h_prime[i] = 1
                
        # Adjust other elements to maintain sum constraint
        total_diff = np.sum(h_prime) - (n_points / 2)
        if abs(total_diff) > 1e-10:
            # Distribute the difference
            indices_to_reduce = np.setdiff1d(np.arange(n_points), indices)
            # Reduce all other values proportionally
            if len(indices_to_reduce) > 0:
                reduction = total_diff / len(indices_to_reduce)
                for i in indices_to_reduce:
                    h_prime[i] -= reduction
                    if h_prime[i] < 0:
                        h_prime[i] = 0
                        
        # Ensure final constraints
        h_prime = np.clip(h_prime, 0, 1)
        h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # New C5 value
        new_c5 = evaluate_c5(h_prime, dx)
        
        # Accept or reject based on simulated annealing criterion
        delta_E = new_c5 - current_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_prime.copy()
            
        # Cool down
        T = max(T_min, alpha * T)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
