import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(h) - 1)  # Convert index to shift value
        
        # Compute gradient for the maximum overlap shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Compute partial derivative with respect to h[i]
            if 0 <= i + max_k < len(h):
                grad[i] += (1 - h[i + max_k]) * dx
            if 0 <= i - max_k < len(h) and max_k != 0:
                grad[i] += h[i - max_k] * dx
                
        # Update h using projected gradient descent
        step = 0.01
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Adaptive step size adjustment based on improvement
        h_temp = h_new.copy()
        temp_c5 = evaluate_c5(h_temp, dx)
        
        if temp_c5 < current_c5:
            h = h_temp
        else:
            # Try smaller step size
            step *= 0.5
            h_new = h - step * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            temp_c5 = evaluate_c5(h_new, dx)
            if temp_c5 < current_c5:
                h = h_new
                
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
