import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    if 'initial_h_values' in globals():
        # Use initial construction as base
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
        # Create 25 particles around the base
        for _ in range(25):
            h = base_h + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
            pbest.append(h.copy())
            pbest_scores.append(float('inf'))
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h += np.random.normal(0, 0.05, n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
        
    # Initialize remaining particles randomly
    for _ in range(n_particles - len(particles)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_scores.append(float('inf'))
    
    # Swarm global best
    gbest = None
    gbest_score = float('inf')
    
    # PSO parameters
    omega = 0.8
    c1 = c2 = 1.2
    mutation_rate = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i, h in enumerate(particles):
            # Compute overlap
            corr = correlate(h, 1 - h, mode="full")
            score = np.max(corr) * dx
            
            # Update particle best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = h.copy()
                
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Velocity update
            v = omega * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            
            # Position update
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                particles[i] = h_new
            
        # Early stopping
        if gbest_score <= 0.3807:
            break
            
    # Return best solution found
    return (gbest.tolist(), gbest_score, n_points)
