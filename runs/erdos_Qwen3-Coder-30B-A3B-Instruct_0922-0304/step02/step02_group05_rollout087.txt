import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available, otherwise create from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a better initial solution
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters
    max_iterations = 10000
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find the maximum
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        max_idx = np.argmax(overlaps)
        
        # Compute gradient based on the position of max overlap
        # For correlation, gradient w.r.t. h[i] is the sum of shifted terms
        # Approximate gradient: h_shifted - (1 - h_shifted) for shift that maximizes
        h_shifted = np.roll(h, max_idx - (len(h) - 1))
        gradient = h_shifted - (1 - h_shifted)
        
        # Try line search
        current_c5 = evaluate_c5(h, dx)
        best_candidate_c5 = current_c5
        best_candidate_h = h
        
        for step_size in line_search_steps:
            h_candidate = project_feasible(h - step_size * gradient, n_points, dx)
            candidate_c5 = evaluate_c5(h_candidate, dx)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_candidate_h = h_candidate
        
        # Update if we found a better point
        if best_candidate_c5 < current_c5:
            h = best_candidate_h
            current_c5 = best_candidate_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Periodically solve convex relaxation (every 20 iterations)
        if iteration % 20 == 0:
            # CVXPY optimization
            h_var = cp.Variable(n_points)
            # Soft constraint on sum
            objective = cp.Minimize(cp.max(cp.convolve(h_var, 1 - h_var, mode='full')) + 
                                   lambda_reg * cp.sum(cp.entr(h_var) + cp.log(n_points/2)))
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-4)
                if prob.status == cp.OPTIMAL:
                    h_relaxed = h_var.value
                    h_relaxed = project_feasible(h_relaxed, n_points, dx)
                    relaxed_c5 = evaluate_c5(h_relaxed, dx)
                    if relaxed_c5 < best_c5:
                        best_c5 = relaxed_c5
                        best_h = h_relaxed.copy()
                        h = h_relaxed.copy()
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
