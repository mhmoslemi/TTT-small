```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import log

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters
    lambda_reg = 0.01
    line_search_steps = [0.1, 0.5, 1.0]
    relaxation_interval = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps and find max
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient estimate
        # Using shifted version to approximate gradient
        h_shifted = np.roll(h, k_max)
        gradient = h_shifted - (1 - h_shifted)
        
        # Line search
        current_c5 = evaluate_c5(h, dx)
        best_step_size = 0.0
        for step_size in line_search_steps:
            h_candidate = h - step_size * gradient
            h_candidate = project_feasible(h_candidate, n_points, dx)
            candidate_c5 = evaluate_c5(h_candidate, dx)
            if candidate_c5 < current_c5:
                h = h_candidate
                current_c5 = candidate_c5
                best_step_size = step_size
                break
        
        # Periodically solve convex relaxation
        if iteration % relaxation_interval == 0:
            try:
                # Create convex problem
                h_relax = cp.Variable(n_points)
                # Approximate entropy penalty
                entropy = cp.sum(cp.multiply(h_relax, cp.log(h_relax + 1e-12)))
                # Overlap approximation
                overlap_approx = overlaps[k_max]
                
                constraints = [
                    cp.sum(h_relax) == n_points/2,
                    h_relax >= 0,
                    h_relax <= 1
                ]
                
                objective = cp.Minimize(overlap_approx - lambda_reg * entropy)
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_relax.value
                    h_new = project_feasible(h_new, n_points, dx)
                    new_c5 = evaluate_c5(h_new, dx)
                    if new_c5 < current_c5:
                        h = h_new
                        current_c5 = new_c5
                else:
                    lambda_reg *= 0.5  # Reduce regularization if infeasible
            except:
                lambda_reg *= 0.5
                
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we're close to target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
```
