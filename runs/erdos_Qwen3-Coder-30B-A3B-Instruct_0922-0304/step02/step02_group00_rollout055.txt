import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        # Use initial solution as seed for half particles
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles = [h_init.copy() for _ in range(n_particles//2)]
        # Add random particles
        for _ in range(n_particles//2):
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            particles.append(h_rand)
    else:
        particles = [np.random.uniform(0, 1, n_points) for _ in range(n_particles)]
        for p in particles:
            p = p * (n_points / 2) / np.sum(p)
    
    # Initialize velocities, personal bests, and global best
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_c5(p, dx) for p in particles]
    
    # Global best tracking
    best_idx = np.argmin(pbest_c5)
    gbest = pbest[best_idx].copy()
    gbest_c5 = pbest_c5[best_idx]
    
    # PSO parameters
    w = 0.8
    c1, c2 = 1.2, 1.2
    mutation_rate = 0.05
    
    # Iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) +
                           c2 * r2 * (gbest - particles[i]))
            
            # Position update
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update pbest
            if current_c5 < pbest_c5[i]:
                pbest_c5[i] = current_c5
                pbest[i] = particles[i].copy()
                
                # Update gbest
                if current_c5 < gbest_c5:
                    gbest_c5 = current_c5
                    gbest = particles[i].copy()
        
        # Adaptive inertia (lower later)
        if iteration > 100:
            w = 0.7
        
        # Early exit
        if gbest_c5 <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_c5, n_points)
