import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Alternative initialization: sawtooth pattern
        h = np.linspace(0, 1, n_points)
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style optimization: iteratively improve worst shift
    max_iter = 1000
    patience = 0
    patience_limit = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > patience_limit:
            break
            
        # Formulate convex subproblem to reduce overlap at k_max
        h_var = Variable(n_points)
        
        # Objective: minimize overlap at shift k_max
        # Overlap at shift k_max is sum_i h[i] * (1 - h[i + k_max])
        # Handle circular shift correctly
        
        # The overlap term at shift k_max summation is:
        # sum_{i=0}^{n-1} h[i] * (1 - h[(i + k_max) % n])
        # This is linear in h, so we can create a convex problem
        
        # Create lagged terms for circular shifting
        # h[i] * (1 - h[(i + k_max) % n])
        # Since this is not a bilinear term, proceed with a relaxation approach
        
        # Compute the linearized (in first order) objective around current h
        # Instead use Lagrangian formulation to avoid bilinear term directly
        
        # As an alternative approach, use a convex approximation to the constraint
        # but with actual cost function in linear form
        
        # Let's define subproblem
        # min h_var^T * (1 - h_{shifted}) * dx where shifted index depends on k_max
        # But we want to minimize a bilinear term, which makes this not directly linear
        
        # So we look for a proximal or majorization scheme
        # We'll look at gradient of the overlap at shifted positions
        
        # New approach: minimize overlap as a linear least-squares like problem
        
        # Let's solve a convex problem to reduce overlap at k_max
        # Build constraints around h with the sum constraint
        constraints = [h_var >= 0, h_var <= 1, cp_sum(h_var) == n_points / 2]
        
        # Naive approach: linearize around h
        # For the overlap term, it's sum_i h[i] * (1 - h[i + k_max])
        # We know h[i + k_max] = h[i + k_max] in current array
        shifted = np.roll(h, k_max)
        objective = cp_sum(h_var * (1 - shifted)) * dx
        
        # Problem formulation
        prob = Problem(Minimize(objective), constraints)
        
        try:
            # Use ECOS solver for speed
            prob.solve(solver=cp.ECOS, verbose=False)
            new_h = h_var.value
            if new_h is not None:
                h = np.clip(new_h, 0, 1)
        except:
            # Fallback to last known good h if solver fails
            pass
        
        # Ensure sum constraint is maintained
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check of the solution
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
