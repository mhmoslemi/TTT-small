import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate the C5 bound using correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # PSO parameters
    n_particles = 30
    inertia_start = 0.7
    inertia_decay = 1e-4
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Initial seeding
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create particle swarm
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            # Create new particle with block structure and noise
            h = np.ones(n_points)
            # Randomly place two blocks
            split1 = np.random.randint(1, n_points//3)
            split2 = np.random.randint(2*n_points//3, n_points-1)
            h[split1:split2] = 0
            # Add noise
            h += np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Track global best
    global_best_c5 = float('inf')
    global_best_h = None
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        inertia = inertia_start * np.exp(-inertia_decay * iteration)
        
        for i in range(n_particles):
            h = particles[i]
            
            # Evaluate current particle
            current_c5 = evaluate_c5(h, dx)
            
            # Update personal best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = h.copy()
                
            # PSO update
            r1, r2 = np.random.rand(2)
            for j in range(n_points):
                velocities[i][j] = (
                    inertia * velocities[i][j] 
                    + alpha * r1 * (h[j] - h[j]) 
                    + beta * r2 * (global_best_h[j] - h[j])
                )
                
                # Clip velocity
                velocities[i][j] = np.clip(velocities[i][j], -max_velocity, max_velocity)
                
                # Update position
                h[j] += velocities[i][j]
            
            # Ensure constraints
            particles[i] = project_to_feasible(h, n_points)

    return (global_best_h.tolist(), global_best_c5, n_points)
