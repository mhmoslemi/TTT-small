import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    # Ensure sum constraint and bounds
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using 5-fold symmetric construction for better starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Symmetric pattern: 5 blocks of 14 points each
        segment_length = n_points // 5
        h = np.zeros(n_points)
        for i in range(5):
            start_idx = i * segment_length
            end_idx = (i + 1) * segment_length
            # Alternate between high and low values in segment
            for j in range(start_idx, end_idx):
                if (j - start_idx) % 2 == 0:
                    h[j] = 0.6
                else:
                    h[j] = 0.4
        h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Method to compute gradient using finite differences
    def compute_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        base_c5 = evaluate_c5(h_vals, dx)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
        return grad
    
    # Optimization loop
    iter_count = 0
    max_iter = 5000
    learning_rate = 0.01
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Gradient computation
        grad = compute_gradient(h)
        
        # Update step
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Accept step if it improves the objective
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            h = h_new
        else:
            # Reduce step size and try again
            learning_rate *= 0.9
            if learning_rate < 1e-6:
                break
                
        iter_count += 1

    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
