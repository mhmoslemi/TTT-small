def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def local_search_refinement(h, n_points, dx, max_iter=50):
    # Simple gradient descent using finite differences on the maximum overlap
    def compute_max_overlap(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
        
    step_size = 0.005
    for i in range(max_iter):
        old_value = compute_max_overlap(h)
        grad_h = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            new_val = compute_max_overlap(h_plus)
            grad_h[j] = (new_val - old_value) / eps
        
        # Apply gradient step
        h_new = h - step_size * grad_h
        h = project_to_feasible(h_new, n_points)
        
        # Check if improvement
        new_value = compute_max_overlap(h)
        if new_value >= old_value:
            # No improvement, reduce step size
            step_size *= 0.95
            if step_size < 1e-6:
                break
                
    return h
