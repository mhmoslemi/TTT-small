import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution_step(population, fitness, F, CR, n_points, dx):
    # DE mutation and crossover
    new_population = []
    for i in range(len(population)):
        # Select three different candidates
        candidates = list(range(len(population)))
        candidates.remove(i)
        a, b, c = random.sample(candidates, 3)
        
        # Mutation
        mutant = population[a] + F * (population[b] - population[c])
        
        # Crossover
        crossover_mask = np.random.random(n_points) < CR
        trial = np.where(crossover_mask, mutant, population[i])
        
        # Ensure feasibility
        trial = project_to_feasible(trial, n_points)
        new_population.append(trial)
    
    return new_population

def gradient_descent_step(h, dx, iterations=50):
    # Simple gradient descent with finite differences
    h = h.copy()
    step_size = 0.01
    eps = 1e-4
    
    for _ in range(iterations):
        c5_val = evaluate_c5(h, dx)
        grad = np.zeros_like(h)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, len(h))
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient descent
        h = h - step_size * grad
        h = project_to_feasible(h, len(h))
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    F = 0.7
    CR = 0.9
    generations = 1000
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main DE loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Perform DE step
        new_population = differential_evolution_step(population, fitness, F, CR, n_points, dx)
        
        # Occasionally apply gradient descent
        if gen % 50 == 0 and gen > 0:
            best_individual = population[0]
            refined = gradient_descent_step(best_individual, dx)
            refined = project_to_feasible(refined, n_points)
            new_population[-1] = refined
            
        population = new_population
    
    return (best_solution.tolist(), best_c5, n_points)
