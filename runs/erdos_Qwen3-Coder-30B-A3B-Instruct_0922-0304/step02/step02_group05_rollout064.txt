import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import log

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    """Project h to feasible region with sum = n_points/2 and h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, k, n_points):
    """Compute approximate gradient for shift k"""
    h_shifted = np.roll(h, k)
    # ∇h_i overlap_k = (1 - h_{i+k}) - h_{i+k} = 1 - 2*h_{i+k}
    gradient = 1 - 2 * h_shifted
    return gradient

def line_search(h, gradient, n_points, dx, current_c5):
    """Perform line search to find best step size"""
    step_sizes = [0.1, 0.5, 1.0]
    best_h = h.copy()
    best_c5 = current_c5
    for alpha in step_sizes:
        h_candidate = project_feasible(h - alpha * gradient, n_points, dx)
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        if candidate_c5 < best_c5:
            best_h = h_candidate
            best_c5 = candidate_c5
    return best_h, best_c5

def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    """Solve a convex relaxation to escape local minima"""
    try:
        h_var = cp.Variable(n_points)
        h_target = np.array(h)
        
        # KL divergence regularization
        kl_reg = cp.sum(cp.multiply(h_var, cp.log(h_var + 1e-12) - cp.log(h_target + 1e-12)))
        
        # Overlap approximation (simplified)
        h1 = h_var
        h2 = cp.flip(h_var)
        overlap_approx = cp.convolution(h1, h2, mode='full')
        max_overlap = cp.max(cp.abs(overlap_approx))
        
        objective = max_overlap + lambda_reg * kl_reg
        
        constraints = [cp.sum(h_var) == n_points/2, 
                       h_var >= 0, h_var <= 1]
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            return project_feasible(h_var.value, n_points, dx)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    lambda_reg = 0.01
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration > 10000:
            break
            
        # Compute overlaps and find k giving maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(np.abs(overlap))
        max_overlap = overlap[max_overlap_idx]
        
        # Compute gradient
        k = max_overlap_idx - (n_points - 1)  # convert to shift
        gradient = compute_gradient(h, k, n_points)
        
        # Line search
        h, c5 = line_search(h, gradient, n_points, dx, evaluate_erdos_solution(h))
        
        # Update best
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Periodic convex relaxation
        if iteration % 20 == 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx, lambda_reg)
            c5_relaxed = evaluate_erdos_solution(h_relaxed)
            if c5_relaxed < best_c5:
                best_h = h_relaxed.copy()
                best_c5 = c5_relaxed
                h = h_relaxed
                lambda_reg = min(1.0, lambda_reg * 1.1)
                
        # Check early termination
        if best_c5 <= 0.3807:
            break
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
