import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize swarm
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution for 25 particles
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for _ in range(25):
            particles.append(h0.copy())
            velocities.append(np.zeros(n_points))
    else:
        # Start with simple step function
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for _ in range(25):
            particles.append(h0.copy())
            velocities.append(np.zeros(n_points))
    
    # Initialize other particles randomly
    for _ in range(n_particles - 25):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    omega = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    mutation_std = 0.05
    
    # Track best solutions
    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_c5
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new particle
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
                    
                    if gbest_c5 < best_c5:
                        best_c5 = gbest_c5
                        best_h = gbest.copy()
            
            # Mutation step
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_std, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                
                # Re-evaluate mutated particle
                current_c5 = evaluate_c5(particles[i], dx)
                if current_c5 < pbest_c5[i]:
                    pbest[i] = particles[i].copy()
                    pbest_c5[i] = current_c5
                    
                    if current_c5 < gbest_c5:
                        gbest = particles[i].copy()
                        gbest_c5 = current_c5
                        
                        if gbest_c5 < best_c5:
                            best_c5 = gbest_c5
                            best_h = gbest.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adjust inertia weight over time
        if iteration > 100:
            omega = 0.5
    
    return (best_h.tolist(), best_c5, n_points)
