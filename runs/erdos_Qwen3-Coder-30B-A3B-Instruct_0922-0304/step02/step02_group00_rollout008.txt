import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Swarm initialization
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for i in range(n_particles):
        if i < n_particles // 2:
            h = initial_h.copy()
            h += np.random.normal(0, 0.02, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
        
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h.copy())
        pbest_values.append(evaluate_c5(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    omega = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) +
                            c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = particles[i].copy()
        
        # Mutation
        if np.random.rand() < mutation_rate:
            idx = np.random.randint(0, n_particles)
            noise = np.random.normal(0, 0.05, n_points)
            particles[idx] += noise
            particles[idx] = np.clip(particles[idx], 0, 1)
            particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
            current_c5 = evaluate_c5(particles[idx], dx)
            if current_c5 < pbest_values[idx]:
                pbest_values[idx] = current_c5
                pbest_positions[idx] = particles[idx].copy()
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = particles[idx].copy()

        # Early exit
        if gbest_value <= 0.3807:
            break
    
    return (gbest_position.tolist(), gbest_value, n_points)
