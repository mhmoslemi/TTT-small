import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate the C5 bound."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Use existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
    
    # Add initial solution as one particle
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add other particles with different patterns
    for i in range(n_particles - 1):
        # Alternate blocky patterns
        h_new = np.zeros(n_points)
        if i % 3 == 0:
            # Two uniform blocks
            mid = n_points // 2
            h_new[:mid] = 1
        elif i % 3 == 1:
            # Three uniform blocks
            third = n_points // 3
            h_new[:third] = 1
            h_new[2*third:] = 1
        else:
            # Gaussian-like
            x = np.linspace(0, 1, n_points)
            h_new = np.exp(-((x - 0.5) / 0.2) ** 2)
        
        # Add small perturbations
        h_new += np.random.normal(0, 0.05, n_points) * 0.1
        h_new = project_to_feasible(h_new, target_sum)
        particles.append(h_new.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track best
    best_global_c5 = float('inf')
    best_global_h = None
    
    for iter_count in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate particles
        particle_c5s = []
        for i, h in enumerate(particles):
            c5 = evaluate_c5(h, dx)
            particle_c5s.append(c5)
            
            if c5 < best_global_c5:
                best_global_c5 = c5
                best_global_h = h.copy()
                
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            c5_current = particle_c5s[i]
            
            # Find personal best and global best
            pbest_h = h  # Use particle's own previous best for simplicity
            pbest_c5 = c5_current
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            new_v = (inertia * v + 
                     alpha * r1 * (pbest_h - h) + 
                     beta * r2 * (best_global_h - h))
            
            # Clip velocity
            new_v = np.clip(new_v, -max_velocity, max_velocity)
            
            # Update position
            new_h = h + new_v
            
            # Project to feasible region
            new_h = project_to_feasible(new_h, target_sum)
            
            # Store updated values
            particles[i] = new_h.copy()
            velocities[i] = new_v.copy()
            
        # Update inertia
        inertia = max(0.4, inertia * (1 - inertia_decay))
        
        # Early termination if target met
        if best_global_c5 <= 0.3807:
            break
    
    return (best_global_h.tolist(), best_global_c5, n_particles)
