import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, epsilon=1e-4):
    # Compute gradient using centered finite differences
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += epsilon
        h_minus[i] -= epsilon
        
        h_plus = project_to_feasible(h_plus, n)
        h_minus = project_to_feasible(h_minus, n)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    h = project_to_feasible(h, n_points)
    
    # Annealing parameters
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    temperature = 0.1 * current_c5
    cooling_rate = 0.999
    noise_std = 0.01
    refinement_frequency = 50
    max_iterations = 10000
    
    step_size = 0.01
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate
        candidate_h = h + np.random.normal(0, noise_std, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = candidate_h
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            temperature *= cooling_rate
        else:
            # Accept with probability
            if random.random() < np.exp(-(candidate_c5 - current_c5) / max(temperature, 1e-10)):
                h = candidate_h
                current_c5 = candidate_c5
            temperature *= cooling_rate
            
        # Gradient descent refinement
        if iteration % refinement_frequency == 0:
            for _ in range(5):
                grad = compute_gradient(h, dx)
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                step_size *= 0.9
                
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
