import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def solve_convex_relaxation(h, max_k, n_points, dx, lambda_reg=0.01):
    h_var = cp.Variable(n_points)
    h_target = h.copy()
    
    # Objective: minimize overlap at max_k plus regularization
    # Overlap at k is h * (1 - h_{shifted})
    overlap_term = 0
    for i in range(n_points):
        j = (i + max_k) % n_points
        overlap_term += h_var[i] * (1 - h_var[j])
    
    # Regularization part: entropy-like term
    reg_term = lambda_reg * cp.sum(cp.entr(h_var) + cp.log(n_points/2))
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points/2
    ]
    
    # Minimize
    prob = cp.Problem(cp.Minimize(overlap_term + reg_term), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value
    except:
        pass
    return h_target

def compute_gradient(h, max_k, dx):
    # Gradient estimation via finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, len(h), dx)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

def line_search_step(h, grad, dx, max_k, alpha_max=1e-1):
    # Try decreasing step sizes
    alpha_candidates = [alpha_max * 0.1, alpha_max * 0.5, alpha_max]
    
    for alpha in alpha_candidates:
        h_candidate = h - alpha * grad
        h_candidate = project_feasible(h_candidate, len(h), dx)
        candidate_c5 = evaluate_c5(h_candidate, dx)
        if candidate_c5 < evaluate_c5(h, dx):
            return h_candidate, candidate_c5, alpha
    
    return h, evaluate_c5(h, dx), 0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iterations = max(1000, int(budget_s * 5))
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points, dx)
    else:
        # Random initial solution
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    iteration = 0
    last_relaxation = 0
    lambda_reg = 0.01
	
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute overlaps and find max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        current_c5 = overlap[max_k] * dx
        
        # If we found a better solution, update
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        # Gradient step
        grad = compute_gradient(h, max_k, dx)
        new_h, new_c5, arclen = line_search_step(h, grad, dx, max_k)
        
        if new_c5 < current_c5:
            h = new_h
        else:
            # Try smaller steps even if not improving
            small_grad = grad * 0.01
            h_try = h - small_grad
            h_try = project_feasible(h_try, len(h), dx)
            try_c5 = evaluate_c5(h_try, dx)
            if try_c5 < current_c5:
                h = h_try
        
        # Periodic convex relaxation
        if iteration % 50 == 0 and iteration - last_relaxation > 10:
            try:
                relaxed_h = solve_convex_relaxation(h, max_k, n_points, dx, lambda_reg)
                if relaxed_h is not None:
                    relaxed_c5 = evaluate_c5(relaxed_h, dx)
                    if relaxed_c5 < current_c5:
                        h = relaxed_h
                        lambda_reg *= 0.5  # Reduce regularization
                    else:
                        lambda_reg *= 0.9  # Slightly decrease regularization
                last_relaxation = iteration
            except:
                lambda_reg *= 0.5  # Reduce reg if solving fails

    return (best_h.tolist(), best_c5, n_points)
