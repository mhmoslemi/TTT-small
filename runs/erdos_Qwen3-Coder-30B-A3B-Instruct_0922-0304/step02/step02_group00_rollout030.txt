import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as base
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
        # 25 particles around initial
        for _ in range(25):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        # 25 random particles
        for _ in range(25):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # All random particles
        for _ in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    # Track best solution
    best_c5 = float('inf')
    best_h = None
    
    # PSO parameters
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    mutation_rate = 0.05
    
    # Track personal bests
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    if gbest_score < best_c5:
        best_c5 = gbest_score
        best_h = gbest.copy()

    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        if iteration > 100:
            w = 0.5
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best if needed
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    
                    if gbest_score < best_c5:
                        best_c5 = gbest_score
                        best_h = gbest.copy()
            
            # Mutation with small probability
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    if best_h is None:
        # Fallback: return last good solution
        best_h = particles[gbest_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
