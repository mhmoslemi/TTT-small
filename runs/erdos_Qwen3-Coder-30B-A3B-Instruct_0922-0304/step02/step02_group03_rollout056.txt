```python
import numpy as np
from scipy.signal import correlate
import time

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Initialize with a seed if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
    
    # One particle with initial solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Generate other particles
    for _ in range(n_particles - 1):
        # Start with blocky structure
        h = np.ones(n_points)
        h[n_points//3:] = 0
        # Add noise
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    alpha = 0.8
    beta = 0.8
    max_velocity = 0.3
    inertia = 0.7
    inertia_decay = 1e-4
    
    # Track best positions
    best_positions = [p.copy() for p in particles]
    best_c5s = [evaluate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_c5s)
    global_best_c5 = best_c5s[global_best_idx]
    global_best_h = best_positions[global_best_idx].copy()
    
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = max(0.4, inertia * (1 - inertia_decay))
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_vec = (inertia * velocities[i] + 
                       alpha * r1 * (best_positions[i] - particles[i]) +
                       beta * r2 * (global_best_h - particles[i]))
            
            # Clip velocity
            vel_vec = np.clip(vel_vec, -max_velocity, max_velocity)
            velocities[i] = vel_vec
            
            # Update position
            new_pos = particles[i] + vel_vec
            
            # Project to feasible set
            new_pos = project_to_feasible(new_pos, n_points)
            
            # Evaluate and update
            new_c5 = evaluate_c5(new_pos, dx)
            
            particles[i] = new_pos
            
            if new_c5 < best_c5s[i]:
                best_c5s[i] = new_c5
                best_positions[i] = new_pos.copy()
                
                if new_c5 < global_best_c5:
                    global_best_c5 = new_c5
                    global_best_h = new_pos.copy()
        
        # Early termination if we've found what we want
        if global_best_c5 <= 0.3807:
            break
    
    # Ensure final solution is feasible
    final_h = project_to_feasible(global_best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
