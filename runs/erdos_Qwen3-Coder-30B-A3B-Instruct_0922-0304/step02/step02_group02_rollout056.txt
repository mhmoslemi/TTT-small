import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using symmetric pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a symmetric 5-fold pattern
        h = np.zeros(n_points)
        segment_size = n_points // 5
        values = [0.7, 0.3, 0.7, 0.3, 0.7]
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size
            h[start_idx:end_idx] = values[i]
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 1000
    min_iter = 50
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization initialization
    last_improvement = 0
    patience = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current C5 value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Perform optimization step with L-BFGS-B
        def objective(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_scaled = h_clipped * (n_points / 2) / np.sum(h_clipped)
            return evaluate_c5(h_scaled, dx)
            
        def gradient(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_scaled = h_clipped * (n_points / 2) / np.sum(h_clipped)
            eps = 1e-4
            grad = np.zeros_like(h_vec)
            base_c5 = evaluate_c5(h_scaled, dx)
            
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus_clipped = np.clip(h_plus, 0, 1)
                h_plus_scaled = h_plus_clipped * (n_points / 2) / np.sum(h_plus_clipped)
                plus_c5 = evaluate_c5(h_plus_scaled, dx)
                grad[i] = (plus_c5 - base_c5) / eps
            return grad
            
        try:
            # Use minimize with L-BFGS-B
            res = minimize(
                objective, 
                h, 
                method='L-BFGS-B',
                bounds=bounds,
                jac=gradient,
                options={'maxiter': 5},
                callback=lambda x: None,
            )
            
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to simple gradient descent if optimization fails
                grad = gradient(h)
                step_size = 0.001
                h = h - step_size * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        except Exception:
            # Fallback if optimization fails
            grad = gradient(h)
            step_size = 0.001
            h = h - step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check for stagnation and restore best
        if iteration - last_improvement > patience and iteration > min_iter:
            h = best_h.copy()
            last_improvement = iteration

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
