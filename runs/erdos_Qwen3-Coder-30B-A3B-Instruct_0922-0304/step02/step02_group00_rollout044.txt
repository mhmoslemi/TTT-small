import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from parent construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize AL parameters
    rho = 1.0
    lambda_mult = np.zeros(2 * n_points - 1)
    tau = np.zeros(2 * n_points - 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Convex subproblem using CVXPY
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
        
        # Add constraint for maximum overlap
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Augmented Lagrangian term
        al_term = 0
        for k in range(2 * n_points - 1):
            al_term += lambda_mult[k] * (corr[k] * dx - tau[k])
        
        # Construct convexified overlap approximation
        approx_overlap = correlate(h_var.value, 1 - h_var.value, mode="full") if h_var.value is not None else np.zeros(2 * n_points - 1)
        
        # Solve using CVXPY with different solver
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # Fall back to simple gradient descent step
                h = h + 0.001 * (np.random.rand(n_points) - 0.5)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fall back to gradient descent
            h = h + 0.001 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Update Lagrange multipliers and tau (simplified)
        lambda_mult = lambda_mult + rho * (corr * dx - tau)
        rho *= 1.1  # Increase penalty
    
    return (best_h.tolist(), best_c5, n_points)
