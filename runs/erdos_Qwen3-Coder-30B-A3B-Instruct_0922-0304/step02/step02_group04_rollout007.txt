import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Efficiently compute the maximum overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, n_points):
    # Project h to feasible region satisfying sum(h) = n_points/2 and 0 <= h[i] <= 1
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 1e-10:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    # Compute approximate gradient via finite differences
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n_points)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

    h = project_to_feasible(h, n_points)
    
    # Current state
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T = 0.1 * current_c5  # Initial temperature

    gradient_freq = 50

    step_size = 0.01

    # Main optimization loop
    for iteration in range(max_iterations):

        # Simulated annealing step
        h_candidate = current_h + np.random.normal(0, 0.01, n_points)
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Accept or reject the candidate
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / max(T, 1e-10)):

            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()

        if iteration % gradient_freq == 0 and iteration > 0:

        T *= cooling_rate

        if best_c5 <= 0.3807:
