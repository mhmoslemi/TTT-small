import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, epsilon=1e-4):
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_c5(h, dx)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project_to_feasible(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iterations = 10000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Rectangular pulse: 1 for first half, 0 for second half
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    
    # Current best
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    initial_temp = 0.1 * current_c5
    temperature = initial_temp
    cooling_rate = 0.999
    min_temp = 1e-6
    
    iteration = 0
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Generate candidate by adding noise
        candidate_h = current_h + np.random.normal(0, 0.01, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
            temperature *= cooling_rate
        else:
            # Accept with probability
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / max(temperature, min_temp)):
                current_h = candidate_h
                current_c5 = candidate_c5
            temperature *= cooling_rate
            
        # Gradient descent refinement every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            # Compute gradient
            grad = compute_gradient(current_h, dx)
            
            # Update with gradient descent
            step_size = 0.01
            new_h = current_h - step_size * grad
            new_h = project_to_feasible(new_h, n_points)
            
            # Evaluate
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
                    
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
