import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # AL parameters
    rho = 1.0
    lambda_multipliers = np.zeros(2 * n_points - 1)
    tau = np.zeros(2 * n_points - 1)
    
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
        
        # Convex subproblem using CVXPY
        h_var = cp.Variable(n_points)
        
        # AL objective
        overlap_obj = cp.sum(cp.multiply(lambda_multipliers, corr)) + \
                      0.5 * rho * cp.sum_squares(cp.multiply(h_var, 1 - h_var) - tau)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(overlap_obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h = h_new
                # Enforce constraints
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to gradient descent
                step = 0.005
                iterations = 50
                h_temp = h.copy()
                for _ in range(iterations):
                    corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_temp = np.max(corr_temp) * dx
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_temp) / eps
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback
            step = 0.005
            iterations = 50
            h_temp = h.copy()
            for _ in range(iterations):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        # Update Lagrange multipliers
        corr = correlate(h, 1 - h, mode="full")
        lambda_multipliers += rho * (corr - tau)
        
        # Increase penalty if needed
        violation = np.max(np.abs(corr - tau))
        if violation > 1e-3:
            rho *= 2

    corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
