import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    # Clip to [0,1] and rescale to match sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Equivalent to integral = 1
    
    # Initialize swarm
    P = 50  # Population size
    swarm = []
    for _ in range(P):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            noise_scale = 0.1 * np.max(h) if np.max(h) > 0 else 0.1
            h += np.random.normal(0, noise_scale, n_points)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
        h = repair_h(h, target_sum)
        swarm.append(h)
    
    # Initialize personal bests
    pbest = [swarm[i].copy() for i in range(P)]
    pbest_c5 = [evaluate_c5(swarm[i], dx) for i in range(P)]
    
    # Global best
    global_best_idx = np.argmin(pbest_c5)
    gbest = pbest[global_best_idx].copy()
    gbest_c5 = pbest_c5[global_best_idx]
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 2.0
    velocities = [np.zeros(n_points) for _ in range(P)]
    
    # PSO iterations
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(P):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Repair and evaluate
            swarm[i] = repair_h(swarm[i], target_sum)
            curr_c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if curr_c5 < pbest_c5[i]:
                pbest[i] = swarm[i].copy()
                pbest_c5[i] = curr_c5
                
                # Update global best
                if curr_c5 < gbest_c5:
                    gbest = swarm[i].copy()
                    gbest_c5 = curr_c5
                    
        # Early exit if we hit target
        if gbest_c5 <= 0.3807:
            break
    
    # Apply local search to top 5% particles
    top_k = max(1, P // 20)
    top_indices = np.argsort(pbest_c5)[:top_k]
    
    for i in top_indices:
        h = pbest[i].copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate_c5(h, dx)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, target_sum)
                plus_c5 = evaluate_c5(h_plus, dx)
                grad[j] = (plus_c5 - base_c5) / eps
                
            h = h - 0.01 * grad
            h = repair_h(h, target_sum)
            
            new_c5 = evaluate_c5(h, dx)
            if new_c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = new_c5
                if new_c5 < gbest_c5:
                    gbest = h.copy()
                    gbest_c5 = new_c5
    
    # Final evaluation
    final_c5 = evaluate_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
