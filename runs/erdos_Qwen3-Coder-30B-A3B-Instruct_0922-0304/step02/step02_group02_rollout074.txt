import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1000.0
    T_min = 1.0
    alpha = 0.999
    max_iter = 100000
    
    # Perturbation parameters
    perturb_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate by perturbing 10% of indices
        candidate_h = h.copy()
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Move mass from selected indices
        total_move = perturb_size
        move_amount = total_move / n_perturb
        
        for i in indices:
            if candidate_h[i] > 0:
                candidate_h[i] = max(0, candidate_h[i] - move_amount)
        
        # Distribute the moved mass
        remaining_indices = [i for i in range(n_points) if i not in indices]
        if remaining_indices:
            excess = total_move
            for i in remaining_indices:
                if excess > 0:
                    can_add = min(excess, 1 - candidate_h[i])
                    candidate_h[i] = min(1, candidate_h[i] + can_add)
                    excess -= can_add
                if excess <= 0:
                    break
            # Adjust to maintain sum constraint
            current_sum = np.sum(candidate_h)
            if abs(current_sum - n_points / 2) > 1e-6:
                scale = (n_points / 2) / current_sum
                candidate_h *= scale
        
        # Evaluate
        candidate_c5 = evaluate_c5(candidate_h, dx)
        current_c5 = evaluate_c5(h, dx)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = candidate_h.copy()
        
        # Update best
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Cool down
        T = max(T_min, alpha * T)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
