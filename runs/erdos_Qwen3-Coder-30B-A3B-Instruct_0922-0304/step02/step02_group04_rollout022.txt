import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def line_search(h, d, dx, max_alpha=1.0, c1=1e-4):
    alpha = max_alpha
    c5_base = evaluate_c5(h, dx)
    while alpha > 1e-8:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        if c5_new <= c5_base + c1 * alpha * np.dot(d, compute_gradient(h, dx)):
            return alpha
        alpha *= 0.5
    return alpha

def solve_quadratic_subproblem(H, g, n_points, h):
    # Simplified quadratic solver: project onto constraints
    # Solve: min 0.5*d^T H d + g^T d subject to 0 <= h+d <= 1, sum(d) = 0
    # Projected gradient descent approach
    d = -np.dot(np.linalg.inv(H), g)
    d = project_to_feasible(h + d, n_points) - h
    return d

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.uniform(0.3, 0.7, n_points)
    
    h = project_to_feasible(h, n_points)
    
    # Initialize Hessian as identity
    H = np.eye(n_points)
    
    # SQP iteration
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Compute search direction (simplified)
        try:
            d = -np.dot(np.linalg.inv(H), g)
        except:
            # Fallback to gradient direction
            d = -g
            
        # Line search to find step size
        alpha = line_search(h, d, dx)
        
        # Update h
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Update Hessian with BFGS
        y = h_new - h
        s = alpha * d
        if np.linalg.norm(s) > 1e-10 and np.linalg.norm(y) > 1e-10:
            rho = 1. / np.dot(y, s)
            I = np.eye(n_points)
            H = (I - rho * np.outer(s, y)) @ H @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
        
        h = h_new
        
        # Early exit
        if evaluate_c5(h, dx) <= 0.3807:
            break
    
    return h, evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
