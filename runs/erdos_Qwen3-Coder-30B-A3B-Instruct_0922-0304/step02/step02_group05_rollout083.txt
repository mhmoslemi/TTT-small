import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import linprog

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def compute_gradient(h, n_points, dx):
    # compute gradient of overlap at maxk
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    
    # compute gradient through finite differences
    grad = np.zeros_like(h)
    eps = 1e-5
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points, dx)
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (overlap_plus[max_k] - overlap[max_k]) / eps
    return grad

def line_search(h, grad, n_points, dx, max_iter=3):
    alpha_vals = [1e-3, 1e-2, 1e-1]
    current_c5 = evaluate_c5(h, dx)
    
    for alpha in alpha_vals:
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points, dx)
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            return h_new, new_c5
    return h, current_c5

def convex_relaxation(h, n_points, dx, lambda_reg=0.01):
    # solve relaxed version using cvxpy
    h_cvx = cp.Variable(n_points)
    # compute max overlap index to use in the relaxation
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    
    # the expression we want to minimize
    # this is a convex relaxation of the max overlap term
    # plus entropy regularization
    obj = cp.Minimize(
        cp.sum(cp.multiply(h_cvx, 1 - h_cvx)) + 
        lambda_reg * cp.sum(cp.entr(h_cvx))  # entropy term
    )
    
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2
    ]
    
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_new = h_cvx.value
            return project_feasible(h_new, n_points, dx)
    except:
        pass
    return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step = 0
    lambda_reg = 0.01
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlap) * dx
        
        # check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # early exit
        if best_c5 <= 0.3807:
            break
            
        # compute gradient
        grad = compute_gradient(h, n_points, dx)
        
        # gradient step
        h_new, new_c5 = line_search(h, grad, n_points, dx)
        
        # update solution
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            # fallback if no improvement
            h = project_feasible(h, n_points, dx)
            
        # convex relaxation every 50 steps
        if step % 50 == 0:
            h = convex_relaxation(h, n_points, dx, lambda_reg)
            # reduce regularization if it doesn't help
            current_c5 = evaluate_c5(h, dx)
            if current_c5 >= best_c5:
                lambda_reg *= 0.5
            
        # check time limit
        if time.time() - start_time > budget_s - 1:
            break
    
    # final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
