import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution_step(population, fitness, F, CR, n_points, dx):
    new_population = []
    for i in range(len(population)):
        # Select three different individuals
        idxs = [idx for idx in range(len(population)) if idx != i]
        a, b, c = [population[j] for j in random.sample(idxs, 3)]
        
        # Mutation
        mutant = a + F * (b - c)
        
        # Crossover
        crossover_mask = np.random.rand(n_points) < CR
        trial = np.where(crossover_mask, mutant, population[i])
        
        # Ensure it's feasible
        trial = project_to_feasible(trial, n_points)
        
        # Selection
        trial_fitness = evaluate_c5(trial, dx)
        if trial_fitness < fitness[i]:
            new_population.append(trial)
        else:
            new_population.append(population[i])
    
    return new_population

def gradient_descent_refinement(h, dx, max_iter=100, step_size=0.01, epsilon=1e-6):
    h = h.copy()
    for _ in range(max_iter):
        # Compute current value
        current_c5 = evaluate_c5(h, dx)
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, len(h))
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update
        h = h - step_size * grad
        h = project_to_feasible(h, len(h))
        
        # Check for convergence
        if np.linalg.norm(grad) < 1e-8:
            break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    F = 0.7
    CR = 0.9
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add some random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_solution = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main DE loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # DE step
        population = differential_evolution_step(population, fitness, F, CR, n_points, dx)
        
        # Re-evaluate
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Update best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_c5 = fitness[new_best_idx]
            best_solution = population[new_best_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Occasionally refine with gradient descent
        if gen % 50 == 0:
            refined = gradient_descent_refinement(best_solution, dx)
            refined_c5 = evaluate_c5(refined, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_solution = refined.copy()
    
    return (best_solution.tolist(), best_c5, n_points)
