import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Coarse optimization (n=35)
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize with two uniform blocks
    h = np.ones(n_points)
    h[n_points//2:] = 0
    h = h * (n_points / 2) / np.sum(h)
    
    # Objective function for optimization
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint function: integral constraint
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization using L-BFGS-B
    max_iter_coarse = 500
    step_size_coarse = 0.1
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop for coarse stage
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use L-BFGS-B for optimization
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'ftol': 1e-6, 'gtol': 1e-5})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            h = h_new
        else:
            break
            
        # Early termination if improved enough
        if best_c5 <= 0.3807:
            break
            
    # Stage 2: Refinement (n=70)
    n_points = 70
    dx = 2.0 / n_points
    
    # Upsample coarse solution
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 35), best_h)
    h = h * (n_points / 2) / np.sum(h)
    
    # Add small random perturbations
    h += np.random.normal(0, 0.02, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Refinement with L-BFGS-B
    max_iter_fine = 1000
    step_size_fine = 0.1
    
    for iteration in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'ftol': 1e-6, 'gtol': 1e-5})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            h = h_new
        else:
            break
            
        # Early termination if target met
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
