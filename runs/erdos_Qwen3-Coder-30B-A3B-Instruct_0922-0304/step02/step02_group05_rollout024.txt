import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    # Evaluate C5 using the full correlation
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Clip to [0,1] and rescale to sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 50
    inertia_weight = 1.0
    cognitive_coeff = 2.0
    social_coeff = 2.0
    velocity_limit = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Create a random initial vector
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    # Fill up particles
    while len(particles) < n_particles:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points, dx)
        particles.append(h.copy())
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track personal bests and global best
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p) for p in particles]
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best_score = personal_best_scores[global_best_idx]
    global_best_position = personal_best_positions[global_best_idx].copy()
    
    # PSO loop
    iteration = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update inertia weight
        if iteration % 50 == 0:
            inertia_weight = max(0.8, inertia_weight - 0.02)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = random.random(), random.random()
            cognitive = cognitive_coeff * r1 * (personal_best_positions[i] - particles[i])
            social = social_coeff * r2 * (global_best_position - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Apply velocity limit
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], n_points, dx)
            
            # Evaluate new solution
            score = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
                    last_improvement = iteration
            
            # Check for early termination
            if global_best_score <= 0.3807:
                break
                
        # Periodically reset stagnant particles
        if iteration - last_improvement > 100:
            for i in range(n_particles):
                # Reset the particle to random position around current best
                particles[i] = np.random.rand(n_points) * 0.1 + global_best_position * 0.9
                particles[i] = project_feasible(particles[i], n_points, dx)
                velocities[i] = np.random.rand(n_points) * 0.01 - 0.005
            last_improvement = iteration
        
        # Time check
        if time.time() - start_time > budget_s - 1:
            break
    
    return (global_best_position.tolist(), global_best_score, n_points)
