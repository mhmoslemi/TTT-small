import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def solve_quadratic_subproblem(H, g, n_points):
    # Simplified: solve H*d = -g with bounds
    # Using quadratic programming: min 0.5*d'*H*d + g'*d
    # subject to 0 <= d_i <= 1, sum(d) = 0
    
    # Simple approach: take gradient step, project to feasible set
    d = -np.dot(H, g)
    return d

def line_search(h, d, dx, alpha_max=1.0, beta=0.5):
    alpha = alpha_max
    while alpha > 1e-10:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        c5_old = evaluate_c5(h, dx)
        # Armijo condition (simplified)
        if c5_new <= c5_old:
            return alpha
        alpha *= beta
    return alpha_max

def sequential_quadratic_programming(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    tol = 1e-3
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    # Initialize Hessian approximation (Identity)
    H = np.eye(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = compute_gradient(h, dx)
        
        # Compute search direction
        d = solve_quadratic_subproblem(H, g, n_points)
        
        # Do line search
        alpha = line_search(h, d, dx)
        if alpha <= 1e-10:
            break
            
        # Update h
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, n_points)
        
        # Update Hessian using BFGS
        y = compute_gradient(h_new, dx) - g
        s = h_new - h
        
        if np.dot(s, y) > 1e-10:
            rho = 1.0 / np.dot(s, y)
            I = np.eye(n_points)
            H = (I - rho * np.outer(s, y)) @ H @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
        
        h = h_new
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = sequential_quadratic_programming(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
