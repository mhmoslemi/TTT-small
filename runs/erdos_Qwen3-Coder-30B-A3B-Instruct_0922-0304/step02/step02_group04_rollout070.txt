import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def finite_diff_gradient(h, dx, eps=1e-4):
    # Compute gradient using finite differences
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h_plus))
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, len(h_minus))
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def backtracking_line_search(h, d, grad, dx, alpha_max=1.0, beta=0.5, c=1e-4):
    # Backtracking line search for sufficient decrease
    alpha = alpha_max
    c5_base = evaluate_c5(h, dx)
    while alpha > 1e-10:
        h_new = h + alpha * d
        h_new = project_to_feasible(h_new, len(h_new))
        c5_new = evaluate_c5(h_new, dx)
        if c5_new <= c5_base + c * alpha * np.dot(grad, d):
            return alpha
        alpha *= beta
    return alpha

def solve_quadratic_subproblem(H, g, n_points):
    # Solve quadratic subproblem with bounds and sum constraint
    # Simple gradient method with step size control
    
    # For simplicity, we'll do projected gradient descent with fixed steps
    # A more sophisticated approach would solve the QP exactly
    d = -g
    # Project onto feasible set
    d = np.clip(d, -np.minimum(np.ones(n_points), h), 
               np.maximum(np.zeros(n_points), 1-h))
    return d
