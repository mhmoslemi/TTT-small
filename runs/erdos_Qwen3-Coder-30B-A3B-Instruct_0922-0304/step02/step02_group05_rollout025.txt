import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # L1-ball projection to maintain sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def gradient_step(h, k_shift, dx, step_size=0.01):
    # Compute approximate gradient
    n_points = len(h)
    grad = np.zeros_like(h)
    
    # For shift k, compute gradient of h[i] 
    # gradient w.r.t. h[i] is (1-h[i+k]) if k >= 0, or (1-h[i-k]) if k < 0
    # In zero-padding version, pad with 0 on the right side for negative shifts
    
    # Build shifted versions for finite difference approximation
    h_shifted = np.zeros(n_points + abs(k_shift))
    if k_shift >= 0:
        h_shifted[k_shift:] = h
        grad = 1 - h_shifted[:-k_shift] if k_shift > 0 else 1 - h
    else:
        h_shifted[:k_shift] = h
        # We do the gradient evaluation manually
        for i in range(n_points):
            if i - k_shift < n_points:
                grad[i] = 1 - h[i - k_shift]
            else:
                grad[i] = 1  # This is the zero padded value, assumed 0
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Rescale to correct sum
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            h = h * (n_points / 2) / current_sum
    else:
        # Default initialization (will not be used if parent exists)
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
        h = project_feasible(h, n_points, dx)

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    step_size = 0.01
    max_iter = int(budget_s * 2)  # Rough estimate for iterations
    iteration = 0

    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current overlaps and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx

        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()

        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

        # Compute gradient w.r.t. shift k that produces max overlap
        k_shift = max_k - (len(h) - 1)
        grad = gradient_step(h, k_shift, dx, step_size)

        # Perform gradient descent step
        h = h - step_size * grad
        h = project_feasible(h, n_points, dx)

        # Periodically perform a relaxation step with L2 regularization to escape local minima
        if iteration % 50 == 0:
            # Add small noise to encourage exploration
            h = h + np.random.normal(0, 1e-4, n_points)
            h = project_feasible(h, n_points, dx)

        iteration += 1

    # Final validation
    best_c5 = evaluate_c5(best_h, dx)
    # Ensure final h is feasible
    best_h = project_feasible(best_h, n_points, dx)
    
    return (best_h.tolist(), best_c5, n_points)
