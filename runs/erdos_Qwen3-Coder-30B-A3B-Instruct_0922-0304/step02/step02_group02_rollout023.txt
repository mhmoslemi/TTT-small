import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1000.0
    alpha = 0.999
    T_min = 1.0
    max_iter = 10000
    perturb_size = 0.1
    
    # Temperature schedule based on number of iterations
    max_iters = int(budget_s / 0.01) if budget_s > 0 else 100000
    current_iter = 0
    
    while T > T_min and current_iter < max_iters:
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Make a perturbation
        h_new = h.copy()
        
        # Select random indices to perturb (10% of the array)
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        
        # Add small random changes
        delta = np.random.normal(0, perturb_size, len(indices))
        h_new[indices] += delta
        
        # Ensure values are within [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to correct sum
        total = np.sum(h_new)
        h_new = h_new * (n_points / 2) / total
        
        # Calculate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject the new solution
        delta_e = new_c5 - current_c5
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h = h_new
        
        # Update temperature
        T *= alpha
        current_iter += 1

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure the final solution satisfies constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
