import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing parameters
    T0 = 1000.0
    alpha = 0.999
    T_min = 1.0
    max_iter = 50000
    
    T = T0
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    # Loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        delta = 0.1
        # Increase selected indices
        for i in indices:
            h_new[i] += delta
        # Decrease others to keep sum constant
        total_change = np.sum(h_new[indices]) - np.sum(h[indices])
        decrease_amount = total_change / (n_points - len(indices))
        for i in range(n_points):
            if i not in indices:
                h_new[i] -= decrease_amount
                
        # Clip and rescale to [0,1] and sum = 35
        h_new = np.clip(h_new, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale_factor
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject
        delta_E = new_c5 - current_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cool down
        T = max(T_min, alpha * T)
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    # Final recomputation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
