import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    inertia_weight = 0.7
    cognitive_param = 2.0
    social_param = 2.0
    learning_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    personal_bests = []
    personal_best_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_temp = np.array(initial_h_values)
            noise_scale = 0.1 * np.max(h_temp)
            h_temp = np.clip(h_temp + np.random.normal(0, noise_scale, n_points), 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        else:
            h_temp = h.copy()
        
        particles.append(h_temp.copy())
        velocities.append(np.zeros(n_points))
        personal_bests.append(h_temp.copy())
        personal_best_c5.append(float('inf'))
    
    global_best = h.copy()
    global_best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            c5_val = evaluate_c5(particle, dx)
            
            if c5_val < personal_best_c5[i]:
                personal_best_c5[i] = c5_val
                personal_bests[i] = particle.copy()
                
            if c5_val < global_best_c5:
                global_best_c5 = c5_val
                global_best = particle.copy()
                
        # Early exit if target achieved
        if global_best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            cognitive = cognitive_param * r1 * (personal_bests[i] - particle)
            social = social_param * r2 * (global_best - particle)
            velocity = inertia_weight * velocity + cognitive + social
            
            # Update position
            particle = particle + velocity
            
            # Repair particle
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle.copy()
            velocities[i] = velocity.copy()
            
        # Apply local search to top performers
        top_indices = np.argsort(personal_best_c5)[:n_particles // 20]  # Top 5%
        for idx in top_indices:
            local_h = personal_bests[idx].copy()
            local_c5 = personal_best_c5[idx]
            
            for _ in range(20):  # At most 20 gradient steps
                if time.time() - start_time > budget_s - 1:
                    break
                
                eps = 1e-4
                grad = np.zeros_like(local_h)
                base_c5 = evaluate_c5(local_h, dx)
                
                for j in range(n_points):
                    h_plus = local_h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    plus_c5 = evaluate_c5(h_plus, dx)
                    grad[j] = (plus_c5 - base_c5) / eps
                    
                local_h = local_h - learning_rate * grad
                local_h = np.clip(local_h, 0, 1)
                local_h = local_h * (n_points / 2) / np.sum(local_h)
                
                new_c5 = evaluate_c5(local_h, dx)
                if new_c5 < local_c5:
                    local_c5 = new_c5
                else:
                    break
                    
            # Update if better
            if local_c5 < personal_best_c5[idx]:
                personal_best_c5[idx] = local_c5
                personal_bests[idx] = local_h.copy()
                
            if local_c5 < global_best_c5:
                global_best_c5 = local_c5
                global_best = local_h.copy()

    # Final evaluation
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
