import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian parameters
    rho = 1.0
    max_iter = 500
    
    # Initialize Lagrange multipliers
    lambda_k = np.zeros(2 * n_points - 1)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Formulate convex subproblem
        h_var = cp.Variable(n_points)
        
        # Objective: minimize overlap plus augmented Lagrangian penalty
        objective = cp.Minimize(
            cp.max(cp.multiply(corr, h_var)) + 
            rho * cp.sum_squares(h_var - h) / 2 +
            cp.sum(cp.multiply(lambda_k, corr - np.max(corr)))
        )
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve convex subproblem
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
        except:
            # Fallback to gradient-based update if optimization fails
            step = 0.01
            h_temp = h.copy()
            for _ in range(50):
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_temp) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h_new = h_temp
        
        h = h_new
        
        # Maintain sum and bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update Lagrange multipliers
        new_corr = correlate(h, 1 - h, mode="full")
        lambda_k = lambda_k + rho * (new_corr - np.max(new_corr))
        
        # Update penalty
        rho = min(rho * 1.1, 1000)
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
