import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    F = 0.7
    CR = 0.9
    gd_frequency = 50
    gd_iterations = 50
    step_size = 0.01
    eps = 1e-4
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create start solution as a rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add diverse random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Track best solution
    best_solution = None
    best_c5 = float('inf')
    
    # Main DE loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Create new population through DE
        new_population = population[:population_size//2]  # Elitism
        
        while len(new_population) < population_size:
            # Select three distinct individuals
            a, b, c = random.sample(range(population_size//2), 3)
            if gen % 2 == 0:
                # Standard DE mutation
                mutant = population[a] + F * (population[b] - population[c])
            else:
                # Alternate mutation
                mutant = population[b] + F * (population[c] - population[a])
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, population[a])
            
            # Ensure feasibility
            trial = project_to_feasible(trial, n_points)
            
            new_population.append(trial)
        
        population = new_population[:population_size]
        
        # Periodically refine best individual with gradient descent
        if gen % gd_frequency == 0 and gen > 0:
            h_refined = best_solution.copy()
            for _ in range(gd_iterations):
                if time.time() - start_time > budget_s - 1:
                    break
                c5_val = evaluate_c5(h_refined, dx)
                grad = np.zeros_like(h_refined)
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    h_minus = h_refined.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    c5_minus = evaluate_c5(h_minus, dx)
                    grad[i] = (c5_plus - c5_minus) / (2*eps)
                
                h_refined = h_refined - step_size * grad
                h_refined = project_to_feasible(h_refined, n_points)
                
            # Update best if refined version is better
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_solution = h_refined.copy()
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = differential_evolution(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
