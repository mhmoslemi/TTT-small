import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    # Ensure sum constraint and bounds
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate symmetric initial guess: 5-fold symmetry
        h = np.zeros(n_points)
        segment_size = n_points // 5
        for i in range(5):
            start = i * segment_size
            end = start + segment_size
            if i % 2 == 0:
                h[start:end] = 0.6
            else:
                h[start:end] = 0.4
        h = repair_h(h, n_points)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h_vec):
        # Reshape back to array
        h = h_vec.copy()
        h = repair_h(h, n_points)
        return evaluate_c5(h, dx)
    
    def callback(h_vec):
        nonlocal best_c5, best_h
        if time.time() - start_time > budget_s - 1:
            return True
        h = repair_h(h_vec.copy(), n_points)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        return False

    # Use L-BFGS-B optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        callback=callback,
        options={'maxiter': 1000},
        tol=1e-6
    )
    
    # Final repair and evaluation
    final_h = repair_h(result.x, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Ensure we return the best found so far
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
