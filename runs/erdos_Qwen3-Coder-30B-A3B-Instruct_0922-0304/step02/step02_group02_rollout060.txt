import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Relaxation parameter schedule
    mu_values = np.logspace(np.log10(0.5), 0, 50)  # 50 steps from 0.5 to 1.0
    
    # Warm start the first solution
    current_h = h.copy()
    
    for mu in mu_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        
        # Objective: convex relaxation
        # Minimize mu*C5 + (1-mu)*regularization
        # For regularization, use squared deviation from midpoint
        regularization = cp.sum_squares(h_var - 0.5) / n_points
        
        # Here we don't directly solve C5 = max_k sum(h_i * (1 - h_{i+k}))
        # Instead, use a relaxation that approximates the maximum
        
        # The optimization problem is hard to formulate directly in CVXPY
        # since the maximum over shifts is not convex.
        # So instead we:
        # 1. Pick a batch of shifts to sample
        # 2. Add constraints for those shifts to make the problem convex
        # 3. It's a greedy approach that might miss optimum
        
        # Let's use this approach: project the solution from previous iteration
        # potentially with convex relaxation
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # This empirical method does better than a pure gradient descent
        # Find gradients for finite differencing 
        eps = 1e-4
        grad = np.zeros_like(current_h)
        base_c5 = evaluate_c5(current_h, dx)
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
            
        # Stochastic gradient descent with proper updates
        step_size = 0.1 * (1 - mu)  # Reduce step as we approach final problem
        current_h = current_h - step_size * grad
        
        # Snap to feasible region
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Compute current C5
        current_c5 = evaluate_c5(current_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
