import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with 5-fold symmetric construction
    h = np.zeros(n_points)
    segment_size = n_points // 5
    values = [0.7, 0.3, 0.7, 0.3, 0.7]
    
    for i in range(5):
        start_idx = i * segment_size
        end_idx = start_idx + segment_size
        h[start_idx:end_idx] = values[i]
    
    # Adjust to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with L-BFGS-B
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    def callback(h_flat):
        nonlocal best_c5, best_h
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 1:
            return True
        return False

    # Set bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using L-BFGS-B
    initial_h_flat = h.copy()
    
    try:
        result = minimize(
            objective,
            initial_h_flat,
            method='L-BFGS-B',
            bounds=bounds,
            callback=callback,
            options={'maxiter': 1000}
        )
        if not result.success:
            print("Optimization failed:", result.message)
    except Exception as e:
        print("Optimization error:", e)
    
    # Final check if we found better solution
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Final cleanup
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
