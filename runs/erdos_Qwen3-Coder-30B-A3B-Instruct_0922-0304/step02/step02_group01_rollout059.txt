import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, Max, multiply
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative improvement loop with CVXPY optimization
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Construct and solve convex optimization formulation
        h_var = Variable(n_points)
        
        # Set up constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp_sum(h_var) == n_points / 2
        ]
        
        # For each possible lag k, create relaxation constraints
        max_k = n_points - 1
        t = Variable()
        
        # Construct all overlap expressions
        for k in range(-max_k, max_k + 1):
            if k >= 0:
                # Compute correlation for this lag
                lags = np.arange(n_points - k)
                overlap_expr = cp.sum(multiply(h_var[lags], 1 - h_var[lags + k]))
            else:
                # Same logic for negative lag
                lags = np.arange(n_points + k)
                overlap_expr = cp.sum(multiply(h_var[lags - k], 1 - h_var[lags]))
            
            constraints.append(overlap_expr <= t)

        # Objective
        objective = Minimize(t)
        
        # Form and solve problem
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = h_var.value
                h = np.clip(h_opt, 0, 1)
                # Ensure the normalization constraint is satisfied
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to local search if the LP fails
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to previous h and simple gradient approach if any error
            pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
