import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # L1-ball projection to maintain sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        np.random.shuffle(h)
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    step_size = 0.01
    max_iterations = int(budget_s / 0.5)
    relaxation_interval = 50
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Compute approximate gradient
        # For shift k, gradient is (1 - h)_{shifted} - h_{shifted}
        # But remember we use zero-padding for non-periodic boundary handling
        grad = np.zeros_like(h)
        
        # Compute difference between actual shifts
        shifted_h = np.zeros_like(h)
        shifted_h[max_k:] = h[:-max_k] if max_k > 0 else h[:]
        grad += (1 - shifted_h) - shifted_h
        
        # Update h
        h -= step_size * grad
        h = project_feasible(h, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Periodic relaxation
        if iteration % relaxation_interval == 0 and iteration > 0:
            # Add a small entropy regularization to escape local minima
            # Using projected gradient descent with entropy:
            # min H(h) + λ * overlap (where λ small)
            lambda_reg = 0.001
            h_new = h.copy()
            
            # Add small random dip to mix up solution
            noise_strength = 0.005
            noise = np.random.normal(0, noise_strength, n_points)
            h_new += noise
            h_new = project_feasible(h_new, n_points, dx)
            c5_new_relax = evaluate_c5(h_new, dx)
            if c5_new_relax < best_c5:
                best_c5 = c5_new_relax
                best_h = h_new.copy()
            h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
