import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: two uniform blocks
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    max_iter = 1000
    step_size = 0.1
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap_val = overlap[max_overlap_idx] * dx
        
        # Early termination if target met
        if max_overlap_val <= 0.3807:
            break
            
        # Use a more intelligent gradient approach with smoothing
        h_new = h.copy()
        
        # Try several small perturbations
        best_h_pert = h.copy()
        best_c5_pert = max_overlap_val
        
        for _ in range(10):  # Small number of sample perturbations
            pert = np.random.normal(0, 0.01, n_points)
            h_pert = h + pert
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            c5_pert = max_overlap_val
            # Computing C5 from scratch
            overlap_pert = correlate(h_pert, 1 - h_pert, mode="full")
            c5_pert = np.max(overlap_pert) * dx
            
            if c5_pert < best_c5_pert:
                best_c5_pert = c5_pert
                best_h_pert = h_pert.copy()
        
        # Apply the best perturbation
        if best_c5_pert < max_overlap_val:
            h = best_h_pert
        else:
            # Reduce step size occasionally to prevent oscillation
            step_size *= 0.995
            
        if best_c5_pert < best_c5:
            best_c5 = best_c5_pert
            best_h = best_h_pert.copy()
    
    # Final validation
    overlap_final = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(overlap_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
