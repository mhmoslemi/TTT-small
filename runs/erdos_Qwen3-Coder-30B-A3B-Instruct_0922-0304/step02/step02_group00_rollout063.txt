import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    epsilon = 0.05
    T_initial = 1.0
    beta = 0.995
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.linspace(0.05, 0.95, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing variables
    T = T_initial
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    def evaluate_c5(h_vec):
        # Compute overlap integral C5
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
        
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 1000 == 0:
            T = T_initial  # Reset temperature
        
        # Generate candidate
        h_candidate = h + np.random.normal(0, epsilon * T, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute overlaps
        c5_current = evaluate_c5(h)
        c5_candidate = evaluate_c5(h_candidate)
        delta_c5 = c5_candidate - c5_current
        
        # Accept or reject
        if c5_candidate < c5_current or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                last_improvement = iteration
            
        # Cool down
        if iteration % 100 == 0:
            T *= beta
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
