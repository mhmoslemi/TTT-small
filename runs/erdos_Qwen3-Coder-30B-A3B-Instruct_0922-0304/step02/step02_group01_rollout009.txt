import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Convex optimization approach
        # Create variables
        h_vars = Variable(n_points)
        
        # Objective: minimize t
        t = Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp_sum(h_vars) == n_points / 2, t >= 0]
        
        # Add overlap constraints using convex hull relaxation
        # For each lag k, we want to bound correlation sum
        max_lag = n_points - 1
        
        # For efficient computation, we'll deal with different lags
        h_full = np.concatenate([h_vars, np.zeros(max_lag)])
        
        # Add constraints for all possible lags
        for k in range(-max_lag, max_lag + 1):
            if k == 0:
                continue # No need for overlap with itself in this problem
            if k > 0:
                # Right lag case h_i * (1 - h_{i+k})
                corr_k = cp_sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            elif k < 0:
                # Left lag case h_{i-k} * (1 - h_i)
                corr_k = cp_sum(cp.multiply(h_vars[-k:], 1 - h_vars[:-k]))
                
            constraints.append(corr_k <= t)
        
        # Formulate and solve the problem
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # Solve with solver
        try:
            prob.solve(solver=cp.GLOP, verbose=False, max_iters=500)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_vars.value).flatten()
                if np.isfinite(np.sum(h_new)) and np.all(h_new >= 0) and np.all(h_new <= 1):
                    h = h_new.copy()
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    # Revert to previous good solution if invalid
                    h = best_h.copy()
            else:
                # If optimization failed, use gradient step
                h = best_h.copy()
        except:
            # If optimization fails, fallback to previous good point
            h = best_h.copy()
    
    # Final evaluation with best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
