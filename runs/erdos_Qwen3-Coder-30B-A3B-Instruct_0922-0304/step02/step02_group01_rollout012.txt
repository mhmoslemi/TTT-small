import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def initialize_population(n_points, pop_size):
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def crossover(parent1, parent2):
    n = len(parent1)
    point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.1):
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += np.random.normal(0, sigma)
    return np.clip(h, 0, 1)

def rescale(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    generations = 500
    mutation_rate = 0.1
    
    population = initialize_population(n_points, pop_size)
    
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            h = rescale(h, target_sum)
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = fitness.index(min_fitness)
            best_h = population[best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
            
        # Crossover and mutation
        next_generation = []
        for i in range(0, pop_size, 2):
            if i + 1 < pop_size:
                p1, p2 = selected[i], selected[i+1]
                c1, c2 = crossover(p1, p2)
            else:
                c1 = selected[i].copy()
                c2 = selected[i].copy()
                
            c1 = mutate(c1, mutation_rate)
            c2 = mutate(c2, mutation_rate)
            
            next_generation.extend([c1, c2])
            
        # Ensure population size
        next_generation = next_generation[:pop_size]
        
        # Repair and rescale
        for i in range(len(next_generation)):
            next_generation[i] = rescale(next_generation[i], target_sum)
            
        population = next_generation
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # Fallback
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points)
            h = h * (n_points / 2) / np.sum(h)
        final_c5 = evaluate_c5(h, dx)
        best_h = h
        
    return (best_h.tolist(), final_c5, n_points)
