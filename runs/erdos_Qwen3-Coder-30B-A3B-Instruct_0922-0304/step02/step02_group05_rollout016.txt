```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    # Computes C5 = max_k int h(x)(1-h(x+k)) dx
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap

def project_feasible(h, n_points, dx):
    # Project h to feasible region: [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with the known good construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        h = project_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    regularization_lambda = 0.01
    relaxation_interval = 20
    line_search_steps = [0.1, 0.5, 1.0]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap at all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        k_max = np.argmax(overlap)
        
        # Gradient w.r.t. h for the worst-case shift:
        # For shift k: grad[i] = h[(i+k) % n] - (1 - h[i])
        h_shifted = np.roll(h, k_max)
        gradient = h_shifted - (1 - h)
        
        # Perform line search
        current_c5 = best_c5
        h_candidate = h.copy()
        for alpha in line_search_steps:
            h_test = project_feasible(h - alpha * gradient, n_points, dx)
            test_c5 = evaluate_erdos_solution(h_test)
            if test_c5 < current_c5:
                h_candidate = h_test
                current_c5 = test_c5
                break
        
        h = h_candidate
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Periodically solve convex relaxation
        if iteration % relaxation_interval == 0:
            # Convex relaxation: solve min max_overlap + λ * sum(h * log(h+1e-12))
            # This is a heuristic to escape local minima
            h_cvx = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.maximum(cp.multiply(h_cvx, 1 - h_cvx)) + 
                regularization_lambda * cp.sum(cp.multiply(h_cvx, cp.log(h_cvx + 1e-12)))
            )
            constraints = [
                cp.sum(h_cvx) == n_points/2,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                h_relax = h_cvx.value
                if h_relax is not None:
                    h_relax = project_feasible(h_relax, n_points, dx)
                    c5_relax = evaluate_erdos_solution(h_relax)
                    if c5_relax < best_c5:
                        best_c5 = c5_relax
                        best_h = h_relax.copy()
                    h = h_relax
            except:
                pass  # If convex optimization fails, continue

        # Early stopping if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
```
