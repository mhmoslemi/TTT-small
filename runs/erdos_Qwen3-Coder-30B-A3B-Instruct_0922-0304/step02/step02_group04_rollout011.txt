import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def refine_solution(h, n_points, dx, max_iter=50):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    for _ in range(max_iter):
        current_c5 = evaluate_c5(best_h, dx)
        corr = correlate(best_h, 1 - best_h, mode="full")
        
        # Simple gradient estimation by finite differences
        eps = 1e-4
        grad = np.zeros_like(best_h)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps

        # Gradient step
        step_size = 0.01
        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)

        new_c5 = evaluate_c5(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            step_size *= 0.5
            if step_size < 1e-6:
                break
    return best_h, best_c5

def solve_convex_relaxation(h_initial, n_points, dx):
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Objective: minimize C
    objective = Minimize(C)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]
    
    # Compute overlap constraints for full correlation
    # Since we don't have a direct CVXPY equivalent for convolution,
    # we'll use an iterative approach with numerical estimates of gradients
    
    # Instead of directly representing correlation constraints, 
    # we'll use a local refinement method
    
    # Solve with the initial (projected) solution
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == 'optimal':
            h_solution = h.value
            return h_solution
    except:
        pass
    return h_initial

    np.random.seed(seed)

    h = project_to_feasible(h, n_points)
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    max_iter = 500
    iteration = 0
    
    # Main optimization loop
    while iteration < max_iter and (time.time() - start_time) < budget_s - 2:
        iteration += 1
        
        # First, try a local refinement step
        improved_h, improved_c5 = refine_solution(h, n_points, dx, max_iter=20)
        
        if improved_c5 < best_c5:
            h = improved_h
            best_c5 = improved_c5
            best_h = h.copy()
            if best_c5 <= 0.3807:
                break
                
        # Occasionally perturb h for escape from local minima
        if iteration % 20 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_to_feasible(h, n_points)
            
        # Occasionally try convex relaxation
        if iteration % 30 == 0:
            relaxed_h = solve_convex_relaxation(h, n_points, dx)
            if relaxed_h is not None:
                relaxed_h = project_to_feasible(relaxed_h, n_points)
                relaxed_c5 = evaluate_c5(relaxed_h, dx)
                if relaxed_c5 < best_c5:
                    best_c5 = relaxed_c5
                    best_h = relaxed_h.copy()
                    
    return (best_h.tolist(), best_c5, n_points)
