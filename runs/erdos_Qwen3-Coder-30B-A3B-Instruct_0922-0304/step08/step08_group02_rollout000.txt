import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2):
    """Blend two parents to create offspring."""
    alpha = np.random.random()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.05, max_mutation=0.1):
    """Mutate a solution."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            delta = np.random.uniform(-max_mutation, max_mutation)
            h[i] = h[i] + delta
    return h

def get_initial_population(n_points, n_pop, target_sum):
    """Generate initial population."""
    population = []
    
    # Add initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    
    # Create random individuals
    while len(population) < n_pop:
        h = np.random.random(n_points)
        # Normalize to sum to target_sum
        h = h * (target_sum / np.sum(h))
        h = repair_h(h, target_sum)
        population.append(h)
    
    return population

def local_refinement(h, dx, max_iter=50):
    """Perform local refinement on a solution."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        if np.random.random() < 0.5:
            # Gradient-like step
            idx = np.random.randint(len(h))
            delta = np.random.normal(0, 0.01)
            h[idx] = np.clip(h[idx] + delta, 0, 1)
        else:
            # Slight random exploration
            h = np.random.normal(h, 0.01)
            h = np.clip(h, 0, 1)
            
        # Renormalize
        h = h * (1.0 / np.sum(h) * 2.0 / 2.0)  # sum should remain 1.0
        # Reset to scale properly for dx=2/n
        h = h * (2.0 / 2.0 * n_points / 2.0 / np.sum(h)) * np.sum(h) / (n_points / 2.0)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_pop = 20
    mutation_rate = 0.05
    crossover_prob = 0.8
    
    # Initialize population
    population = get_initial_population(n_points, n_pop, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    # Genetic algorithm loop
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate population
        fitnesses = [compute_c5(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Select parents (tournament selection)
        parents = []
        for _ in range(n_pop//2):
            i1, i2 = np.random.choice(n_pop, 2, replace=False)
            if fitnesses[i1] < fitnesses[i2]:
                parents.append(population[i1])
            else:
                parents.append(population[i2])
                
        # Create new population
        new_population = deepcopy(parents)
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            if np.random.random() < crossover_prob and len(parents) >= 2:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = crossover(p1, p2)
                child = mutate(child, mutation_rate)
                child = repair_h(child, target_sum)
                new_population.append(child)
            else:
                # Just mutate one parent
                parent = np.random.choice(parents)
                child = mutate(parent, mutation_rate)
                child = repair_h(child, target_sum)
                new_population.append(child)
                
        # Trim population to exact size
        population = new_population[:n_pop]
        
        generation += 1
        
        # Occasionally refine top solutions
        if generation % 10 == 0:
            top_indices = np.argsort(fitnesses)[:3]
            for i in top_indices:
                refined_h, refined_c5 = local_refinement(population[i], dx)
                if refined_c5 < fitnesses[i]:
                    population[i] = refined_h
                    if refined_c5 < best_c5:
                        best_c5 = refined_c5
                        best_h = refined_h.copy()
                        
    # Final refinement of best solution
    if best_h is not None:
        refined_h, refined_c5 = local_refinement(best_h, dx, 100)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Ensure best_h is correct size
    if best_h is None:
        # Fallback if no good solution found
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (target_sum / np.sum(best_h))
    
    return (best_h.tolist(), best_c5, n_points)
