import numpy as np
from scipy.signal import correlate
import time
from math import exp
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a solution."""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] = max(0, min(1, h_mut[i] + random.gauss(0, 0.05)))
    return h_mut

def local_refinement(h, target_sum, dx, steps=50):
    """Refine solution using gradient-like steps."""
    h_ref = h.copy()
    for _ in range(steps):
        if random.random() < 0.5:  # Random perturbation
            idx = random.randint(0, len(h_ref) - 1)
            delta = random.gauss(0, 0.02)
            h_ref[idx] = max(0, min(1, h_ref[idx] + delta))
        else:  # More systematic search
            # Simple gradient step in direction of decreased overlap
            overlap = correlate(h_ref, 1 - h_ref, mode="full")
            max_idx = np.argmax(overlap)
            # Adjust values around the max overlap region
            for i in np.clip([max_idx-2, max_idx-1, max_idx, max_idx+1, max_idx+2], 0, len(h_ref)-1):
                h_ref[i] = max(0, min(1, h_ref[i] - 0.01 * (h_ref[i] - 0.5)))
        h_ref = repair_h(h_ref, target_sum)
    return h_ref

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = []
    
    # Include initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        population.append(h_init)
    
    # Fill rest of population with random valid solutions
    while len(population) < n_pop:
        h = np.random.random(n_points)
        # Make it somewhat balanced
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        population.append(h)

    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0

    generation = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best = population[0]
        if fitness[0] < best_c5:
            best_h = current_best.copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Local refinement of top solutions
        for i in range(min(3, len(population))):
            refined = local_refinement(population[i], target_sum, dx)
            c5_refined = compute_c5(refined, dx)
            if c5_refined < fitness[i]:
                population[i] = refined
                fitness[i] = c5_refined

        # Create new generation
        new_population = [population[0].copy()]  # Elitism - keep best
            
        # Tournament selection
        while len(new_population) < n_pop:
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # Another parent
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent2 = population[winner_idx]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Repair
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < n_pop:
                new_population.append(child2)
        
        population = new_population
        generation += 1
        
        if best_c5 <= 0.3807:
            break

    # Final check if we're still improving
    if time.time() - start_time < budget_s - 1:
        final_h = local_refinement(best_h, target_sum, dx, 100)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
