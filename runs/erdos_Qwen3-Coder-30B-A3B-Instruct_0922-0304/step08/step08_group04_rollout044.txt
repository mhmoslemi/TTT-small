import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def local_search_segmented(h, dx, n_points, n_segments, max_iter=1000):
    """Perform local search with segmented approach."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Initialize segments
    segment_width = 2.0 / n_segments
    segment_vals = np.random.rand(n_segments).astype(np.float64)
    segment_vals = normalize_h(segment_vals, n_points / 2 / n_segments)
    
    for iter_count in range(max_iter):
        # Generate candidate h from segments
        h_candidate = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(i * n_points / n_segments)
            end_idx = int((i + 1) * n_points / n_segments)
            h_candidate[start_idx:end_idx] = segment_vals[i]
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = normalize_h(h_candidate, n_points / 2)
        
        # Evaluate
        c5 = evaluate_c5(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
        
        # Small random perturbations
        for i in range(n_segments):
            perturb = np.random.uniform(-0.05, 0.05)
            new_val = segment_vals[i] + perturb
            new_val = np.clip(new_val, 0, 1)
            
            # Test new value
            old_val = segment_vals[i]
            segment_vals[i] = new_val
            
            # Construct new h
            h_test = np.zeros(n_points)
            for j in range(n_segments):
                start_idx = int(j * n_points / n_segments)
                end_idx = int((j + 1) * n_points / n_segments)
                h_test[start_idx:end_idx] = segment_vals[j]
            
            h_test = np.clip(h_test, 0, 1)
            h_test = normalize_h(h_test, n_points / 2)
            
            test_c5 = evaluate_c5(h_test, dx)
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_h = h_test.copy()
            else:
                segment_vals[i] = old_val
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Try different segment counts
    segment_counts = [3, 4, 5, 6, 7, 8, 9, 10]
    
    for n_segments in segment_counts:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        h_seg, c5_seg = local_search_segmented(h, dx, n_points, n_segments, max_iter=500)
        
        if c5_seg < best_c5:
            best_c5 = c5_seg
            best_h = h_seg.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
