```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to feasible region: 0 <= h <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def sample_shifts(n, sample_size):
    """Sample shifts for QP constraints"""
    shifts = np.random.choice(range(1, 2*n - 1), size=sample_size, replace=False)
    return shifts

def compute_c5(h, dx):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, n_points)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, n_points)

    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)

    # Initialize QP variables
    h_vars = cp.Variable(n_points)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    target_overlap = 0.38085
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Sample a subset of shifts for QP
        sample_shifts_indices = sample_shifts(n_points, min(50, n_points))
        
        # Compute overlap for all shifts
        overlap_full = correlate(h_current, 1 - h_current, mode="full")
        
        # Build the problem
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == target_sum
        ]
        
        # Add constraints for sampled shifts
        for shift_idx in sample_shifts_indices:
            # The overlap for shift k is h * (1-h_{k}) 
            # In full convolution, h[i] * (1 - h[i+k]) 
            # This is a sparse constraint
            shift = shift_idx - n_points + 1  # Convert index to actual shift
            if shift < 0:
                # h[i] * (1 - h[i-shift]) for i=0 to n_points+shift-1
                # This becomes h[i] * (1 - h[i-shift]) where i = 0..n_points+shift-1
                # So i goes from 0 to n_points+shift-1  
                # And i-shift goes from shift to n_points-1 
                # This needs careful indexing
                overlap_val = 0
                for i in range(max(0, -shift), min(n_points, n_points-shift)):
                    overlap_val += h_vars[i] * (1 - h_vars[i-shift])
            elif shift >= 0:
                # h[i] * (1 - h[i+shift]) for i=0 to n_points-shift-1
                overlap_val = 0
                for i in range(n_points-shift):
                    overlap_val += h_vars[i] * (1 - h_vars[i+shift])
            
            constraints.append(overlap_val <= target_overlap)
        
        # Minimize max overlap (use target as upper bound)
        objective = cp.Minimize(target_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try to solve
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_vars.value
                if h_new is not None:
                    h_new = project_h(h_new, n_points)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        no_improvement_count = 0
                        # Reduce target since we have found a better solution
                        target_overlap = min(target_overlap, best_c5 * 0.9999)
                    else:
                        no_improvement_count += 1
                    h_current = h_new.copy()
        except:
            pass  # Fall back to simple local search if QP fails
            
        iteration += 1
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with some additional local search
    h_refined = best_h.copy()
    for i in range(50):  # Small number of refinement steps
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
```
