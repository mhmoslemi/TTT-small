import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_population(n_points, target_sum, num_individuals):
    """Create diverse initial population."""
    population = []
    
    # Add the parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Generate random individuals
    for _ in range(num_individuals - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def crossover(parent1, parent2):
    """Create offspring via average crossover."""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Mutate individual with Gaussian noise."""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual = individual + noise
        individual = np.clip(individual, 0, 1)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 1000
    elite_count = 5
    mutation_rate = 0.1
    sigma = 0.1
    
    # Initialize population
    population = create_initial_population(n_points, target_sum, population_size)
    
    # Evaluate initial population
    fitnesses = [compute_c5_exact(h, dx) for h in population]
    
    best_c5 = min(fitnesses)
    best_h = population[np.argmin(fitnesses)].copy()
    
    # Tracking
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep the elite
        new_population = sorted_population[:elite_count]
        new_fitnesses = sorted_fitnesses[:elite_count]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutate(offspring, mutation_rate, sigma)
            
            # Maintain constraint
            offspring = normalize_h(offspring, target_sum)
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
            new_fitnesses.append(compute_c5_exact(offspring, dx))
        
        # Update population
        population = new_population
        fitnesses = new_fitnesses
        
        # Update best
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitnesses)].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        generation += 1
        
        # Log progress periodically
        if generation % 100 == 0:
            print(f"Generation {generation}: Best C5 = {best_c5:.6f}")
    
    # Final computation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
