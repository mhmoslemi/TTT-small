import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def clip_and_normalize(h, target_sum):
    """Clip h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def tabu_search_erdos(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with slight perturbations
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = normalize_h(h, target_sum)
    
    # Ensure initial solution is feasible
    h = clip_and_normalize(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Track search progress
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Temperature for simulated annealing aspect
    temperature = 1.0
    cooling_rate = 0.999
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Generate candidates (10 candidates)
        candidates = []
        for _ in range(10):
            candidate = h.copy()
            
            # Perform neighborhood moves (mix of flips and swaps)
            if np.random.rand() < 0.3:
                # Flip operation
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                # Perturb
                candidate[idx1] *= 0.97 + np.random.normal(0, 0.01)
                candidate[idx2] *= 0.97 + np.random.normal(0, 0.01)
            else:
                # Swap operation
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                candidate[idx1], candidate[idx2] = candidate[idx2], candidate[idx1]
            
            # Repair constraints
            candidate = clip_and_normalize(candidate, target_sum)
            c5 = compute_c5_exact(candidate, dx)
            candidates.append((candidate, c5))
        
        # Evaluate candidates
        best_candidate = None
        best_candidate_c5 = float('inf')
        
        for candidate, c5 in candidates:
            # Check tabu list
            tabu_key = tuple(sorted(candidate.round(5)))
            if tabu_key in tabu_list:
                # Allow aspiration if it's much better
                if c5 < best_c5:
                    best_candidate = candidate
                    best_candidate_c5 = c5
            else:
                if c5 < best_candidate_c5:
                    best_candidate = candidate
                    best_candidate_c5 = c5
        
        # If we found a candidate, update solution
        if best_candidate is not None:
            h = best_candidate
            current_c5 = best_candidate_c5
            
            # Update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Add to tabu list
            tabu_key = tuple(sorted(h.round(5)))
            tabu_list.append(tabu_key)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        else:
            no_improvement_count += 1
        
        # Cooling
        temperature *= cooling_rate
        
        # Early exit check
        if best_c5 <= 0.3807:
            break
    
    return best_h.tolist(), best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = tabu_search_erdos(seed, budget_s)
    return (h_values, c5_bound, n_points)
