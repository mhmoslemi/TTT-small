import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize and clip
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    spent_time = time.time() - start_time
    remaining_time = budget_s - spent_time - 1  # Leave 1s for cleanup
    
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    
    # Alternate between relaxation and rounding heuristic
    while spent_time < remaining_time and iter_count < max_iter:
        # 1. Relaxation Stage (Solve continuous relaxation)
        # This is a heuristic use of convex relaxation without full LP
        # Instead we use gradient-based modified approach
        
        # 2. Rounding heuristic with adjustment
        # Since we can't do full relaxation easily within budget,
        # we refine our current solution by gradient-like updates
        
        # Generate candidate updates using cross-correlation insights
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_pos = np.argmax(overlap)
        
        # Perturb gradient-like: look around max overlap position
        delta = np.zeros_like(h)
        # Try to reduce overlap by adjusting nearby points
        # Use a simplified gradient ascent-like method
        for i in range(n_points):
            # Try to reduce the value of overlap at position max_pos
            # by nudging h[i] in a direction that might reduce it
            # If we can change h[i] to reduce the overlap, do so
            
            # We estimate direction by simulating small changes
            # and measuring effect on overlap
            current_val = h[i]
            
            # Slight perturbations to see effect
            candidates = []
            for delta_move in [-0.01, 0.01]:
                h_test = h.copy()
                h_test[i] = np.clip(current_val + delta_move, 0, 1)
                test_overlap = np.max(correlate(h_test, 1 - h_test, mode="full")) * dx
                candidates.append((test_overlap, delta_move, i))
            
            # Pick best improvement direction
            candidates.sort(key=lambda x: x[0])
            new_c5, delta_move, idx = candidates[0]
            if new_c5 < best_c5:
                delta[idx] = delta_move
                h[idx] = np.clip(current_val + delta_move, 0, 1)
        
        # Renormalize
        h = h * (target_sum / np.sum(h))
        
        # Check if we improved
        current_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
        
        iter_count += 1
        spent_time = time.time() - start_time
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
