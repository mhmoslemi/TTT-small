import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a piecewise linear construction
        h = np.zeros(n_points)
        # Create a ramp function
        for i in range(n_points):
            x = i * dx
            if x < 1:
                h[i] = x
            else:
                h[i] = 2 - x
        # Normalize
        h = normalize_h(h, target_sum)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Use a hybrid approach with simulated annealing for global search
    # Start with fine-grained local search, but occasionally perform random perturbations
    
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 2000
    temp = 1.0
    cooling_rate = 0.999
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Sometimes perform global search - random perturbation
        if np.random.rand() < 0.05:
            h_trial = h.copy()
            # Random change to a few points
            n_changes = max(1, n_points // 20)
            indices = np.random.choice(n_points, n_changes, replace=False)
            for i in indices:
                # Perturb randomly
                perturb = np.random.normal(0, 0.05)
                h_trial[i] = np.clip(h_trial[i] + perturb, 0, 1)
            # Renormalize
            h_trial = normalize_h(h_trial, target_sum)
            c5_new = compute_c5(h_trial, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_trial.copy()
                h = h_trial.copy()
                no_improvement_count = 0
        else:
            # Local optimization via gradient approximation
            h_trial = h.copy()
            # Gradient approximation via finite differences
            for i in range(n_points):
                h_plus = h_trial.copy()
                h_plus[i] = min(1.0, h_trial[i] + 0.01)
                h_plus = normalize_h(h_plus, target_sum)
                h_plus = np.clip(h_plus, 0, 1)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                max_plus = np.max(overlap_plus)
                
                h_minus = h_trial.copy()
                h_minus[i] = max(0.0, h_trial[i] - 0.01)
                h_minus = normalize_h(h_minus, target_sum)
                h_minus = np.clip(h_minus, 0, 1)
                
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
                max_minus = np.max(overlap_minus)
                
                grad = (max_plus - max_minus) / 0.02
                
                # Update with gradient
                step_size = 0.05
                new_val = h_trial[i] - step_size * grad
                h_trial[i] = np.clip(new_val, 0, 1)
            
            # Renormalize
            h_trial = normalize_h(h_trial, target_sum)
            h_trial = np.clip(h_trial, 0, 1)
            
            c5_new = compute_c5(h_trial, dx)
            
            # Accept or reject
            if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h = h_trial.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                
        temp *= cooling_rate
        
        if no_improvement_count > max_no_improvement:
            # Reset with new randomization
            h = np.random.rand(n_points) * 0.5
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            no_improvement_count = 0
            
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
