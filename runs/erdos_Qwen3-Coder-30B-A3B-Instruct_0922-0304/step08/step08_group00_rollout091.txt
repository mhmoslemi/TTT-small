import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Gradient-based optimization with Nesterov momentum
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    
    # Initialize momentum variables
    v = np.zeros_like(h)
    h_prev = h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Precompute all overlap values to speed up gradient computation
    def compute_overlap_indices(h):
        # compute all overlaps for all shifts
        # For each k, the overlap is sum_{i=0}^{n-1} h_i * (1 - h_{(i+k)%n})
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps[k] = overlap * dx
        return overlaps
    
    # Compute initial overlap and maximum
    all_overlaps = compute_overlap_indices(h)
    max_idx = np.argmax(all_overlaps)
    max_overlap = all_overlaps[max_idx]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient of max overlap
        # Gradient of overlap_k w.r.t. h_i = 1 - h_{i-k}
        # Gradient of max overlap is the gradient of that specific overlap with max_idx
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - max_idx) % n_points
            grad[i] = 1 - h[j]
        
        # Update with Nesterov momentum
        v = momentum * v - step_size * grad
        h_new = h_prev + v
        
        # Projection onto [0,1] 
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Update for next iteration
        h_prev = h.copy()
        h = h_new.copy()
        
        # Recompute all overlaps
        all_overlaps = compute_overlap_indices(h)
        max_idx = np.argmax(all_overlaps)
        max_overlap = all_overlaps[max_idx]
        
        # Update best solution
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
        
        # Adaptive step size adjustment
        if iteration % 10 == 0 and iteration > 0:
            step_size = max(min_step_size, step_size * 0.99)
        
        # Early stopping if improvement is negligible
        if iteration > 10 and abs(max_overlap - best_c5) < tol:
            break
    
    # Final refinement using scipy minimize with constraints
    from scipy.optimize import minimize
    
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    sum_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    result = minimize(objective, best_h, method='SLSQP',
                      bounds=bounds, constraints=sum_constraint,
                      options={'maxiter': 500, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Ensure final solution is valid
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
