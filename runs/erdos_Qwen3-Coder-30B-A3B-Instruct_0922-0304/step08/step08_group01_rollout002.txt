import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().astype(np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the constraint: sum(h) == target_sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function for scipy.optimize.minimize
    def objective(h):
        return compute_c5(h, dx)
    
    # Run optimization
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # SLSQP optimizer options
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options=options, 
                         callback=lambda x: None)
        
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    except Exception as e:
        pass  # Fallback to original if optimization fails
    
    # Final validation and refinement
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
