import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed):
    """Round relaxed solution to binary with sum constraint."""
    n_points = len(h_relaxed)
    target_sum = n_points / 2
    
    # Compute threshold as median
    threshold = np.median(h_relaxed)
    
    # Start with binary vector
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h_binary)
    diff = current_sum - target_sum
    
    if diff > 0:
        # Too many 1s, flip some to 0s (flip smallest values)
        flip_indices = np.argsort(h_relaxed)[:int(diff)]
        h_binary[flip_indices] = 0
    elif diff < 0:
        # Too many 0s, flip some to 1s (flip largest values that are 0)
        flip_indices = np.argsort(h_relaxed)[::-1][:int(-diff)]
        h_binary[flip_indices] = 1
    
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # LP relaxation approach
    max_iter = 10
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Formulate LP relaxation
        h_r = cp.Variable(n_points, name="h_r", lb=0, ub=1)
        z = cp.Variable(name="z")
        
        # Objective
        objective = cp.Minimize(z)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_r) == target_sum)
        
        # Overlap constraints for all shifts
        for k in range(-n_points+1, n_points):
            # Compute indices for valid x and x+k
            if k >= 0:
                # x = 0 to n-1-k
                expr = cp.sum(cp.multiply(h_r[:n_points-k], 1 - h_r[k:]))
            else:
                # x = -k to n-1
                expr = cp.sum(cp.multiply(h_r[-k:], 1 - h_r[:n_points+k]))
            
            constraints.append(expr * dx <= z)
        
        # Create and solve LP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_relaxed = h_r.value
                h_binary = round_to_binary(h_relaxed)
                
                # Validate and update best
                c5 = compute_c5(h_binary, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_binary.copy()
                    
                # Early stopping if target achieved
                if best_c5 <= 0.3807:
                    break
        except:
            pass
        
        iter_count += 1
    
    # Final refinement using Basin Hopping or simple local search
    # Take best solution and do some local search
    h = best_h.copy()
    
    # Local search refinement
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Pick random index to perturb
        i = np.random.randint(n_points)
        
        # Small perturbation
        old = h[i]
        step = np.random.normal(0, 0.01)
        h[i] = np.clip(old + step, 0, 1)
        
        # Renormalize
        h = normalize_h(h, target_sum)
        
        # Check improvement
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
