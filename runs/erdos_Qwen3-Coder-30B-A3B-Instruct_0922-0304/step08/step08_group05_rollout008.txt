import numpy as np
from scipy.signal import correlate
import time
from collections import deque

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_constraints(h, target_sum):
    """Repair constraints by clipping and normalizing."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def generate_neighbor(h, target_sum, flip_prob=0.3):
    """Generate a neighbor solution by flipping or swapping elements."""
    h_new = h.copy()
    n = len(h)
    
    # Decide on operation (flip or swap) based on probability
    if np.random.rand() < flip_prob:
        # Flip operation: modify two random indices
        idx1, idx2 = np.random.choice(n, 2, replace=False)
        h_new[idx1] = max(0, min(1, h_new[idx1] * 0.97 + np.random.normal(0, 0.01)))
        h_new[idx2] = max(0, min(1, h_new[idx2] * 0.97 + np.random.normal(0, 0.01)))
    else:
        # Swap operation: swap two random indices
        idx1, idx2 = np.random.choice(n, 2, replace=False)
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
    
    # Repair constraints
    h_new = repair_constraints(h_new, target_sum)
    return h_new

def tabu_search(seed=42, budget_s=600):
    """Tabu search for the Erdős overlap problem."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

    # Ensure feasible bounds
    h = repair_constraints(h, target_sum)
    
    # Tabu list (keeps track of recently visited solutions)
    tabu_list = deque(maxlen=100)
    
    # Parameters
    max_no_improvement = 1000
    tabu_tenure = 50
    temperature = 1.0
    cooling_rate = 0.999
    
    # Tracking
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < max_no_improvement:
        # Generate neighbors
        neighbors = []
        num_neighbors = 10
        
        for _ in range(num_neighbors):
            neighbor = generate_neighbor(h, target_sum)
            # Check if neighbor is tabu (simplified)
            if tuple(neighbor.round(6)) in tabu_list:
                # Allow aspiration if better than current best
                if compute_c5_exact(neighbor, dx) < best_c5:
                    neighbors.append(neighbor)
            else:
                neighbors.append(neighbor)
        
        # Evaluate neighbors
        scores = [compute_c5_exact(neighbor, dx) for neighbor in neighbors]
        
        # Find best neighbor
        best_neighbor_idx = np.argmin(scores)
        best_neighbor = neighbors[best_neighbor_idx]
        neighbor_score = scores[best_neighbor_idx]
        
        # Update if improvement
        if neighbor_score < best_c5:
            best_c5 = neighbor_score
            best_h = best_neighbor.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Update temperature

        # Add current h to tabu list (if not too long)

    h_values, c5_bound, n_points = tabu_search(seed, budget_s)
    return (h_values.tolist(), c5_bound, n_points)
