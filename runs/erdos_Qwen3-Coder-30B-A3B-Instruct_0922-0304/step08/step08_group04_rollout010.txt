import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def solve_convex_relaxation(n_points, dx):
    """Solve convex relaxation to get a good initial point."""
    # Variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [0 <= h, h <= 1]
    
    # For each shift, add overlap constraint
    for k in range(2 * n_points - 1):
        shift_constraint = 0
        for x in range(n_points):
            x_k = x + k - (n_points - 1)
            if 0 <= x_k < n_points:
                shift_constraint += h[x] * (1 - h[x_k])
        constraints.append(shift_constraint <= t)
    
    constraints.append(sum(h) == n_points / 2)
    
    # Objective
    objective = Minimize(t)
    
    # Formulate and solve
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return np.array(h.value).reshape(-1)
    except:
        pass
    
    return None

def compute_gradient(h, dx, n_points):
    """Compute a stable approximation of the gradient."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    max_val = overlap[max_idx]
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Gradients for overlapping terms
    for i in range(n_points):
        # Find all terms where h[i] appears in the correlation sum
        for k in range(2 * n_points - 1):
            x = i
            pos2 = i + k - (n_points - 1)
            if 0 <= x < n_points and 0 <= pos2 < n_points:
                # Gradient contribution from term h[x] * (1 - h[pos2])
                grad[i] += (1 - 2 * h[pos2]) * dx
        
        # Take maximum gradient contribution over all shifts
        grad[i] = grad[i] if abs(grad[i]) > 1e-10 else 1e-10
    
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999, tolerance=1e-8):
    """Optimize with gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Check termination conditions
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 1000:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialization using convex relaxation and fallback to initial_h_values
    h = None
    
    # Try convex relaxation first
    relaxed_solution = solve_convex_relaxation(n_points, dx)
    
    if relaxed_solution is not None:
        h = relaxed_solution
    elif 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initializations with constraint satisfaction
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimize with gradient descent if we haven't timed out
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 1:
        best_h, best_c5 = optimize_with_gradient_descent(
            h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999)
        return (best_h.tolist(), best_c5, n_points)
    
    # Otherwise return the initial solution
    return (h.tolist(), evaluate_c5(h, dx), n_points)
