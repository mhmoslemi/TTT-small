import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength=0.1):
    """Apply Gaussian mutation to an individual."""
    h_new = h + np.random.normal(0, mutation_strength, len(h))
    h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover_parents(h1, h2):
    """Average two parents to create offspring."""
    offspring = (h1 + h2) / 2.0
    return offspring

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select parent via tournament selection."""
    selected_indices = np.random.choice(len(population), size=tournament_size)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitnesses)]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    mutation_strength = 0.1
    
    # Initialize population
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    population.append(h.copy())
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    
    # Evolutionary algorithm
    generation = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate fitness
        fitnesses = [compute_c5_exact(ind, dx) for ind in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_population = [sorted_population[0].copy()]  # Keep best
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(sorted_population, sorted_fitnesses)
            parent2 = tournament_selection(sorted_population, sorted_fitnesses)
            
            # Crossover
            offspring = crossover_parents(parent1, parent2)
            
            # Mutation
            offspring = mutate_individual(offspring, mutation_strength)
            
            # Normalize
            offspring = normalize_h(offspring, target_sum)
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
        
        # Replace old population
        population = new_population[:population_size]
        
        generation += 1
    
    # Final evaluation
    if best_h is not None:
        final_c5 = compute_c5_exact(best_h, dx)
    else:
        # Fallback to last generation
        fitnesses = [compute_c5_exact(ind, dx) for ind in population]
        min_fitness_idx = np.argmin(fitnesses)
        final_c5 = fitnesses[min_fitness_idx]
        best_h = population[min_fitness_idx].copy()
    
    return (best_h.tolist(), final_c5, n_points)
