import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute gradient of overlap with respect to h."""
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
    
    # Seed population with the current best
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = repair_h(h_random, target_sum)
        population.append(h_random)
    
    # Parameters
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_generations = 2000
    no_improvement_limit = 100
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    no_improvement_count = 0
    generation = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        # Evaluate all individuals in population
        fitness_scores = []
        for h_individual in population:
            c5 = evaluate_overlap(h_individual, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        selected = population[:population_size // 2]
        
        # Create new population through crossover and mutation
        new_population = [best_h.copy()]  # Elitism: keep best
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = selected[np.random.randint(0, len(selected))], \
                                  selected[np.random.randint(0, len(selected))]
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = selected[np.random.randint(0, len(selected))].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child[mutate_idx] = np.random.rand()
            
            # Repair and refine
            child = repair_h(child, target_sum)
            # Gradient descent refinement
            gradient = compute_gradient(child, n_points)
            step_size = 0.02
            child -= step_size * gradient
            child = repair_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
        
        # Adapt parameters
        if len(fitness_scores) > 1 and fitness_scores[0] >= fitness_scores[1]:
            no_improvement_count += 1
            if no_improvement_count > no_improvement_limit:
                mutation_rate = min(0.5, mutation_rate * 1.1)
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            if mutation_rate > 0.05:
                mutation_rate *= 0.95
        
        generation += 1
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
