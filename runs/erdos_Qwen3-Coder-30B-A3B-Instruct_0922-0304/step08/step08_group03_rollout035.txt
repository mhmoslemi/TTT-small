import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n_points):
    """Create a triangular ramp function for initialization."""
    x = np.linspace(0, 2, n_points)
    h = np.clip(1 - np.abs(x - 1), 0, 1)
    return h

def project_h(h, target_sum):
    """Project h to ensure sum equals target_sum and values are in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_h(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Multi-resolution grid search
    resolution_levels = [1, 2, 4]  # Coarse, medium, fine
    
    for level in resolution_levels:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Refine grid
        grid_size = n_points * level
        x = np.linspace(0, 2, grid_size)
        h_fine = np.interp(x, np.linspace(0, 2, n_points), h)
        h_fine = project_h(h_fine, grid_size / 2)
        
        # Gradient-based optimization on refined grid
        max_iter = 2000 // level
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradients
            overlap_full = correlate(h_fine, 1 - h_fine, mode="full")
            max_idx = np.argmax(overlap_full)
            # Approximate gradient direction
            grad = np.zeros_like(h_fine)
            
            # Perturb each point to estimate gradient
            epsilon = 0.001
            for i in range(len(h_fine)):
                h_plus = h_fine.copy()
                h_plus[i] = min(1.0, h_fine[i] + epsilon)
                h_plus = project_h(h_plus, grid_size / 2)
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                max_plus = np.max(overlap_plus)
                
                h_minus = h_fine.copy()
                h_minus[i] = max(0.0, h_fine[i] - epsilon)
                h_minus = project_h(h_minus, grid_size / 2)
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                max_minus = np.max(overlap_minus)
                
                grad[i] = (max_plus - max_minus) / (2 * epsilon)
            
            # Update step
            lr = 0.001
            h_fine = h_fine - lr * grad
            h_fine = project_h(h_fine, grid_size / 2)
            
            if iter_count % 50 == 0:
                # Downsample to original resolution for evaluation
                h_coarse = np.array([np.mean(h_fine[i:i+level]) for i in range(0, grid_size, level)])
                if len(h_coarse) != n_points:
                    h_coarse = np.resize(h_coarse, n_points)
                h_coarse = project_h(h_coarse, target_sum)
                c5 = compute_c5(h_coarse, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_coarse.copy()
        
        if best_c5 <= 0.3807:
            return (best_h.tolist(), best_c5, n_points)
    
    # Final refinement on original grid
    max_iter = 5000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap value
        overlap_full = correlate(h, 1 - h, mode="full")
        current_max = np.max(overlap_full)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Gradient approximation
        grad = np.zeros_like(h)
        epsilon = 0.001
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + epsilon)
            h_plus = project_h(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - epsilon)
            h_minus = project_h(h_minus, target_sum)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / (2 * epsilon)
        
        # Update step
        lr = 0.005
        h = h - lr * grad
        h = project_h(h, target_sum)
        
        # Add small random perturbation occasionally to escape local minima
        if iter_count % 100 == 0:
            noise = np.random.normal(0, 0.001, len(h))
            h = h + noise
            h = project_h(h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
