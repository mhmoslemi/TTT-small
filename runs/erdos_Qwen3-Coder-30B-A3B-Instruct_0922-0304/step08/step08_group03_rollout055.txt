import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    # Create a triangular ramp from 0 to 1 and back to 0
    half = n // 2
    up = np.linspace(0, 1, half)
    down = np.linspace(1, 0, n - half)
    return np.concatenate([up, down])

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Phase 1: Gradient descent with momentum
    momentum = 0.9
    learning_rate = 0.001
    decay_rate = 0.99
    velocity = np.zeros(n_points)
    
    # Phase 2: Coordinate-wise updates
    coord_sampling_rate = 0.2
    coord_update_count = max(1, int(n_points * coord_sampling_rate))
    
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 500
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Phase 1: Momentum-based gradient descent
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap_pos = np.argmax(overlap)
        
        # Approximate gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Simple finite difference
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.001)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.001)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            gradient[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / 0.002
        
        # Momentum update
        velocity = momentum * velocity + (1 - momentum) * gradient
        h_candidate = h - learning_rate * velocity
        
        # Project to feasibility
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        if candidate_c5 < best_c5:
            h = h_candidate
            best_c5 = candidate_c5
            stagnation_count = 0
            learning_rate *= decay_rate
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Switch to coordinate-wise updates
                indices = np.random.choice(n_points, coord_update_count, replace=False)
                for i in indices:
                    # Try small upward and downward step
                    backward = h.copy()
                    backward[i] = max(0.0, h[i] - 0.005)
                    backward = clip_and_renormalize(backward, target_sum)
                    backward_c5 = compute_c5(backward, dx)
                    
                    forward = h.copy()
                    forward[i] = min(1.0, h[i] + 0.005)
                    forward = clip_and_renormalize(forward, target_sum)
                    forward_c5 = compute_c5(forward, dx)
                    
                    if backward_c5 < forward_c5 and backward_c5 < best_c5:
                        h = backward
                        best_c5 = backward_c5
                    elif forward_c5 < best_c5:
                        h = forward
                        best_c5 = forward_c5
                        
                stagnation_count = 0
                # Reset learning rate
                learning_rate = 0.001
        
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
