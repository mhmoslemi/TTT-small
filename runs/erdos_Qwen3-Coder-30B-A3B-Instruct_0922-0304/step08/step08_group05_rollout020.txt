import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_solution(h, n_points, dx):
    """Perturb solution using simulated annealing approach."""
    new_h = h.copy()
    
    # Select 10% of indices to modify
    indices_to_modify = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
    
    for i in indices_to_modify:
        # Randomly choose either a small flip or swap
        if np.random.rand() < 0.5:
            # Flip: multiply by small factor
            new_h[i] *= np.random.uniform(0.9, 1.1)
        else:
            # Swap: exchange with random element
            j = np.random.randint(0, n_points)
            new_h[i], new_h[j] = new_h[j], new_h[i]
    
    # Normalize after perturbation
    new_h = np.clip(new_h, 0, 1)
    target_sum = n_points / 2
    new_h = normalize_h(new_h, target_sum)
    
    return new_h

def cutting_plane_step(h, dx, n_points, current_c5):
    """Improve solution using cutting-plane approximation."""
    # Compute the overlap
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Focus on the worst lag (max overlap)
    # Make small adjustments in the direction that reduces overlap
    new_h = h.copy()
    
    # Identify where to reduce overlap (approximately)
    k = max_idx - (n_points - 1)  # lag index
    # Reduce values around the peak of overlap to decrease that specific correlation
    for i in range(n_points):
        if 0 <= i + k < n_points and i + k != i:
            # Adjust values to reduce the specific overlap
            adjustment = 0.01 * (1 - 2 * h[i]) * (1 - 2 * h[i + k])
            new_h[i] = np.clip(new_h[i] - adjustment * 0.5, 0, 1)
            new_h[i + k] = np.clip(new_h[i + k] - adjustment * 0.5, 0, 1)
    
    # Normalize and adjust
    target_sum = n_points / 2
    new_h = normalize_h(new_h, target_sum)
    return new_h

        # Add some noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Annealing parameters
    T = 1.0
    decay_rate = 0.999
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 2000:
        # Simulated Annealing Phase
        new_h = perturb_solution(h, n_points, dx)
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = new_h
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                last_improvement = 0

                best_h = h.copy()

        # Early termination if sufficiently improved
