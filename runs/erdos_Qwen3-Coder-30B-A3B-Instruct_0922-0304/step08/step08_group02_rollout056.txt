import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient of correlation at given shift"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        idx = i + shift
        if 0 <= idx < n_points:
            grad[i] = 1 - h[idx]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Set up parameters
    step_size = 0.1
    no_improve_count = 0
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute the correlation
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Compute gradient direction from the largest correlation
        shift = max_overlap_idx - (n_points - 1)  # Get the actual shift
        gradient = compute_gradient(h_current, shift, n_points)
        
        # Take projected gradient step
        h_new = h_current - step_size * gradient
        h_new = project_h(h_new, target_sum)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Update current solution
        h_current = h_new.copy()
        
        # Adaptive step size
        if no_improve_count % 100 == 0 and no_improve_count > 0:
            step_size *= 0.5
            
        # Perturbation to escape local extrema
        if no_improve_count > 100 and np.random.random() < 0.05:
            perturb_mag = 0.1 * np.mean(h_current)
            noise = np.random.normal(0, perturb_mag, n_points)
            h_current = project_h(h_current + noise, target_sum)
            no_improve_count = 0
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with gradient descent
    h_refined = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        overlap = correlate(h_refined, 1 - h_refined, mode="full")
        max_overlap_idx = np.argmax(overlap)
        shift = max_overlap_idx - (n_points - 1)
        gradient = compute_gradient(h_refined, shift, n_points)
        h_temp = h_refined - 0.05 * gradient
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
