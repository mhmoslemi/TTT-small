```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try alternating relaxation and rounding stages
    max_iter = 50
    iter_count = 0
    stagnation_count = 0
    last_best_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Relaxation Stage: Formulate and solve LP to approximate optimal h
        h_relaxed = np.copy(h)
        try:
            # Create a simple convex relaxation
            # We solve a convex upper bound on C5 by using a sum of pairwise constraints
            
            # For simplicity, simplified model with fewer constraints
            h_cvx = cp.Variable(n_points, nonneg=True)
            constraints = [cp.sum(h_cvx) == target_sum]
            
            # Constraint to bound overlap - upper bound
            # We approximate upper bound on max overlap
            z = cp.Variable()
            # Collect constraints for a few key lag values
            # To save time, limit number of constraints to small set
            lag_values = list(range(0, max(n_points//4, 1)))
            for k in lag_values:
                # For k=0: we don't include the self correlation term
                # But since h[i](1-h[i+k]) on the full domain, 
                # we can bound not all but partial overlaps.
                expr = 0
                for i in range(n_points):
                    if i + k < n_points:
                        expr += h_cvx[i] * (1 - h_cvx[i+k])
                constraints.append(expr <= z * dx)  # This overapproximates since we are not summing over entire range
            
            prob = cp.Problem(cp.Minimize(z), constraints)
            
            # Setting solver to be faster and deterministic if solver is available
            try:
                prob.solve(solver=cp.GLPK, verbose=False, time_limit=5)
            except:
                try:
                    prob.solve(solver=cp.CVXOPT, verbose=False)
                except:
                    # Fallback to softer constraints using random restart
                    h_relaxed = np.clip([np.random.rand() for _ in range(n_points)], 0, 1)
                    h_relaxed = normalize_h(h_relaxed, target_sum)
                    # Continue with random h_relaxed
                    pass
            else:
                if h_cvx.value is not None:
                    h_relaxed = np.clip(h_cvx.value, 0, 1)
                    h_relaxed = normalize_h(h_relaxed, target_sum)
        except:
            # Keep previous h_relaxed if we couldn't solve
            pass
            
        # Rounding heuristic to create better binary-like h
        h_rounded = h_relaxed.copy()
        median_val = np.median(h_rounded)
        h_rounded = np.where(h_rounded > median_val, 1.0, 0.0)
        
        # Adjust sum to target
        current_sum = np.sum(h_rounded)
        if abs(current_sum - target_sum) > 1e-6:
            diff = current_sum - target_sum
            if diff > 0:
                # Need to flip some 1s to 0s
                to_flip = int(diff)
                indices = np.argsort(h_rounded[:])[:to_flip]
                for i in indices:
                    h_rounded[i] = 0.0
            else:
                to_flip = int(-diff)
                indices = np.argsort(h_rounded[:])[::-1][:to_flip]
                for i in indices:
                    h_rounded[i] = 1.0
        
        # Final validation and update best
        h_final = h_rounded
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            last_best_c5 = best_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update h for next iteration
        h = h_final.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        if stagnation_count >= 10:
            # Try restart with randomization to escape local minima
            h = np.random.rand(n_points).astype(np.float64)
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            stagnation_count = 0
            
        iter_count += 1

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
