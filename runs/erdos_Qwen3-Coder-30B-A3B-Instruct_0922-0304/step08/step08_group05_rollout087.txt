import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the maximum overlap term."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each active index, accumulate gradient contributions
    for k_idx in max_indices:
        # Index k transition from -n+1 to n-1, but we're looking at linear
        # correlation indices, so k ranges from 0 to 2*n-2.
        # When k_idx = 0, this is overlap at lag -(n-1) = -69
        # When k_idx = n-1, this is overlap at lag 0
        # When k_idx = 2*n-2, this is overlap at lag n-1 = 69
        k = k_idx - (n - 1)
        
        # Add contributions from each position i
        # h[i] contributes to overlap at lag k as h[i] * (1 - h[i+k])
        # So grad[i] += 2 * (1 - 2 * h[i+k]) for term h[i] * (1 - h[i+k]) 
        # and similar for h[i+k] contribution to term h[i] * (1 - h[i+k])
        for i in range(n):
            # For i in h[i], it contributes to overlap_k if i+k is within [0, n-1]
            j = i + k
            if 0 <= j < n:
                grad[i] += 2 * (1 - 2 * h[j]) * dx
                grad[j] += -2 * h[i] * dx
                
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available, otherwise random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum_factor = 0.9
    velocity = np.zeros(n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination check
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Apply momentum
        velocity = momentum_factor * velocity + grad
        h_new = h - learning_rate * velocity
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Update h
        h = h_new
        
        # Adjust learning rate adaptively
        if last_improvement % 100 == 0 and last_improvement > 0:
            learning_rate *= 0.99
            
    # Final check
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
