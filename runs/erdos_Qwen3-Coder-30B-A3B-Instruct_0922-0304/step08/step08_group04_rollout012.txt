def compute_gradient(h, dx, n_points):
    """Compute approximate gradient of the C5 objective."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient based on the maximum overlap
    for i in range(n_points):
        total_grad = 0.0
        for k in range(2 * n_points - 1):
            # Calculate actual indices considering wraparound
            # shift k, the difference is (i + k) % n_points 
            pos1 = i
            pos2 = (i + k - (n_points - 1)) % n_points   # k index in flipped form
            
            # Only consider valid indices
            if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                # i+k shifted by (n_points-1) (padding)
                q = k - (n_points - 1)
                index_in_original = i + q
                if index_in_original >= 0 and index_in_original < n_points:
                    total_grad += 2 * (overlap[k] / dx) * (1 - 2 * h[pos1])
        
        grad[i] = total_grad
    
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    return grad

def optimize_with_adam(h, dx, n_points, max_iter=100000, lr=0.01, decay=0.999, tolerance=1e-8):
    """Optimize h using Adam-like gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Compute loss and gradient
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Check early stopping
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 1000:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5
