import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Use a triangular distribution as initial guess
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i]
            else:
                h[i] = 2 - x[i]
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use simulated annealing with geometric cooling schedule
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    # Initial temperature and cooling rate
    temp = 0.1
    cooling_rate = 0.999
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Choose two random indices to modify
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        
        # Create a candidate solution by slightly perturbing two points
        h_new = h.copy()
        delta = np.random.normal(0, 0.05)
        
        # Perturb both points
        h_new[i1] = np.clip(h[i1] + delta, 0, 1)
        h_new[i2] = np.clip(h[i2] - delta, 0, 1)
        
        # Renormalize
        h_new = normalize_h(h_new, target_sum)
        
        # Calculate new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):
                h = h_new.copy()
            no_improvement_count += 1
            
        # Temperature update
        temp *= cooling_rate
        
        if no_improvement_count > max_no_improvement:
            # Reset temperature if stuck
            temp = min(temp * 1.1, 1.0)
            no_improvement_count = 0
            
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
