import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Adaptive step size
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Work with the worst lag
        k = max_idx - (n_points - 1)  # Convert from full correlation index to lag
        
        # Find the locations that contribute most to the maximum overlap
        # We'll modify h in a way that reduces the overlap at this lag
        h_new = h.copy()
        
        # Simple strategy: adjust for all points
        # Increase h where it's low and decrease where it's high, towards reducing max overlap
        # The idea comes from reducing sensitivities where overlap is maximized
        # This approach focuses on adjusting in ways that avoid increasing maximum overlap
        
        # Generate adjustments for current points to reduce overlap
        step = epsilon
        best_h_new = h_new.copy()
        best_c5_new = current_c5
        
        # Try several possible adjustments
        for i in range(n_points):
            # Only modify adjacent points to influence correlation at lag k  
            if 0 <= i + k < n_points and 0 <= i < n_points:
                h_test = h_new.copy()
                delta = step
                h_test[i] = min(1.0, h_test[i] + delta)  # Slightly increase
                h_test[i + k] = max(0.0, h_test[i + k] - delta)  # Slightly decrease
                h_test = normalize_h(h_test, target_sum)
                h_test = np.clip(h_test, 0, 1)
                
                # Check if this new configuration improves the overlap metric
                new_c5 = compute_c5_exact(h_test, dx)
                if new_c5 < best_c5_new:
                    best_c5_new = new_c5
                    best_h_new = h_test.copy()
        
        # Apply best change
        h = best_h_new.copy()
        
        # Normalize and clip to stay in bounds
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        if iteration % 100 == 0:
            epsilon *= 0.995  # Adaptive learning rate
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
