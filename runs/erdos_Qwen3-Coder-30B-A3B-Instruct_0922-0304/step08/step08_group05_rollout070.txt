import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_solution(h, n_points, dx):
    """Generate a perturbed solution."""
    new_h = h.copy()
    
    # Randomly choose 10% of points to perturb
    indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
    
    # For each selected index, apply either a flip or swap
    for i in indices:
        if np.random.random() < 0.5:
            # Flip operation: scale by 0.97 (reduce value)
            new_h[i] *= 0.97
        else:
            # Swap operation: exchange with a random index
            j = np.random.randint(0, n_points)
            new_h[i], new_h[j] = new_h[j], new_h[i]
            
    # Clip values to be within bounds
    new_h = np.clip(new_h, 0, 1)
    
    # Re-normalize
    target_sum = n_points / 2
    new_h = normalize_h(new_h, target_sum)
    
    return new_h

def cutting_plane_refinement(h, n_points, dx):
    """Perform cutting-plane refinement on the worst overlap."""
    # Get all correlation values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the worst lag
    max_lag_idx = np.argmax(overlap)
    max_overlap = overlap[max_lag_idx]
    
    # Try to reduce the worst overlap by adjusting h values
    # This approximation modifies two points that contribute to max_lag_idx overlap
    
    # Approximate where the worse points are by considering the peak bandwidth
    # For simplicity, we offset by 1/4 of the total length to avoid edge effects
    shift = n_points // 4
    if max_lag_idx >= shift and max_lag_idx < 3 * n_points // 4:
        # Focus on points around the peak area
        start_idx = max(0, max_lag_idx - shift)
        end_idx = min(n_points, max_lag_idx + shift)
        
        # Adjust points that could contribute
        for i in range(start_idx, end_idx):
            # Simple gradient descent step to reduce the overlap
            # This is a very simplified gradient descent
            if i < n_points - 1 and i + max_lag_idx < 2 * n_points - 1:
                # Reduce by small amount, and compensate by increasing neighbor
                step = 0.005
                # Prevent going out of bounds
                if h[i] > 0 and h[i + max_lag_idx] < 1:
                    h[i] = max(0, h[i] - step)
                    h[i + max_lag_idx] = min(1, h[i + max_lag_idx] + step)
    
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points / 2)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a simple pattern: step function
        h = np.zeros(n_points)
        h[n_points//4:3*n_points//4] = 1.0
        # Add a bit of noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    no_improvement_count = 0
    max_no_improvement = 1000
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Perturb solution
        new_h = perturb_solution(h, n_points, dx)
        
        # Compute C5 for new solution
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept with probability based on temperature
        delta = new_c5 - evaluate_c5(h, dx)
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            h = new_h.copy()
            
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Refinement with cutting-plane approach every 50 steps
        if iteration % 50 == 0:
            h = cutting_plane_refinement(h, n_points, dx)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Cool down temperature
        temp = max(min_temp, temp * cooling_rate)
        
        iteration += 1
        
    # Final normalization and computation
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
