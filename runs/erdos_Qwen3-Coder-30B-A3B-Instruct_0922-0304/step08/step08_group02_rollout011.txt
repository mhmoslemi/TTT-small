import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create initial guess
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1.0
        # Add some randomness
        for i in range(n_points//10):
            idx = np.random.randint(0, n_points)
            h_current[idx] = np.random.rand()
        h_current = h_current * (target_sum / np.sum(h_current))
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    n_steps = 100
    no_improvement_count = 0
    max_no_improvement = 500
    
    temp = T0
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        
        # Generate perturbation
        delta = np.random.normal(0, sigma)
        h_new[idx] += delta
        
        # Clamp to [0,1]
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Scale to maintain sum constraint
        current_sum = np.sum(h_new)
        h_new = h_new * (target_sum / current_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Compute cost difference
        c5_old = compute_c5(h_current, dx)
        delta_c5 = c5_new - c5_old
        
        # Metropolis criterion
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Adaptive cooling
        if no_improvement_count >= max_no_improvement:
            temp = np.maximum(T_min, T0 * alpha**(iteration / 1000))
            no_improvement_count = 0
            # Reduce perturbation
            sigma = np.maximum(0.001, sigma * 0.95)
            
        iteration += 1
        
        # Early stopping if we've improved significantly
        if best_c5 <= 0.3807:
            break

    # Final refinement with local search
    h_refined = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_temp = h_refined.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        
        # Rescale
        current_sum = np.sum(h_temp)
        h_temp = h_temp * (target_sum / current_sum)
        
        c5_temp = compute_c5(h_temp, dx)
        c5_refined = compute_c5(h_refined, dx)
        
        if c5_temp < c5_refined:
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    
    return (h_refined.tolist(), final_c5, n_points)
