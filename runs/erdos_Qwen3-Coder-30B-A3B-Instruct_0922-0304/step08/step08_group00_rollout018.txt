import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Gradient descent parameters
    lr = 0.1
    momentum = 0.9
    min_lr = 1e-5
    tol = 1e-6
    max_iter = int(1e5)
    
    # For momentum
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Precompute all overlap values
    def compute_gradient(h_vec):
        # Compute overlap for all shifts
        overlap = correlate(h_vec, 1 - h_vec, mode="full") * dx
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Gradient computation
        grad = np.zeros_like(h_vec)
        # The maximum overlap is at shift k = max_idx - (n_points - 1)
        k = max_idx - (n_points - 1)
        for i in range(n_points):
            j = (i - k) % n_points
            grad[i] = 1 - h_vec[j]
        return grad, max_val
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        grad, current_c5 = compute_gradient(h)
        
        # Nesterov momentum update
        v = momentum * v - lr * grad
        h_new = h + v
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Adaptive learning rate
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
            lr *= 1.1
            lr = min(lr, 1.0)
        else:
            lr *= 0.5
            lr = max(lr, min_lr)
            h = h_new
        
        if abs(current_c5 - new_c5) < tol:
            break
    
    # Final local refinement using scipy
    from scipy.optimize import minimize
    
    def objective(h_vec):
        # Ensure it sums to target_sum
        h_vec = h_vec * (target_sum / np.sum(h_vec))
        return evaluate_overlap(h_vec, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Allow very small tolerance since time is limited
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 50, 'ftol': 1e-6})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
