import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to [0,1] and scale to match target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient for the given shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():

        h_current = np.clip(h_current, 0, 1)

    else:

    # Initialize tracking variables
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute the current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Compute gradient for the current maximum overlap
        shift = max_overlap_idx - (n_points - 1)
        grad = compute_gradient(h_current, shift, n_points)
        
        # Update h using projected gradient descent
        h_new = h_current - step_size * grad
        
        # Project h back to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Compute new C5 value
        c5_new = compute_c5(h_new, dx)
        
        # Check if this improved the solution
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check if we should perturb the solution
        if no_improvement_count > 10 and iteration % 20 == 0:
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h_current + noise
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)

                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
                
        # Use the best one for next iteration
        h_current = best_h.copy()
        
        # Adaptive step size
        if no_improvement_count > 50:
            step_size *= 0.9
            no_improvement_count = 0
            
        iteration += 1
        
        # Early stopping if we've achieved our target
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    h_refined = best_h.copy()
    for i in range(20):

        h_temp = h_refined.copy()

            h_refined = h_temp.copy()
