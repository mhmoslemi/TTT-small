import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_population(n_points, target_sum, num_individuals):
    """Create an initial population of valid h vectors."""
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    else:
        # Generate a random individual
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Fill remaining individuals
    for _ in range(num_individuals - 1):
        # Generate a diverse individual
        h = np.random.rand(n_points)
        # Add some structure to prevent being too random
        h = 0.5 * h + 0.25 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = np.random.randint(len(population))
    for i in range(tournament_size - 1):
        candidate_idx = np.random.randint(len(population))
        if fitnesses[candidate_idx] < fitnesses[selected_idx]:
            selected_idx = candidate_idx
    return population[selected_idx].copy()

def crossover(parent1, parent2, target_sum):
    """Perform average-based crossover."""
    child = (parent1 + parent2) / 2.0
    child = normalize_h(child, target_sum)
    child = np.clip(child, 0, 1)
    child = normalize_h(child, target_sum)
    return child

def mutate(individual, mutation_rate, target_sum):
    """Apply Gaussian mutation."""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, 0.1, len(individual))
        individual = individual + noise
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    mutation_rate = 0.1
    max_generations = 1000
    elite_size = 5
    
    # Create initial population
    population = create_initial_population(n_points, target_sum, population_size)
    
    # Evaluate initial fitness
    fitnesses = [compute_c5_exact(ind, dx) for ind in population]
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_c5 = fitnesses[best_idx]
    best_h = population[best_idx].copy()
    
    # Evolution loop
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 1000:
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitnesses[0] < best_c5:
            best_c5 = sorted_fitnesses[0]
            best_h = sorted_population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population with elitism
        new_population = sorted_population[:elite_size]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child = crossover(parent1, parent2, target_sum)
            
            # Mutation
            child = mutate(child, mutation_rate, target_sum)
            
            new_population.append(child)
        
        # Update population
        population = new_population
        
        # Re-evaluate fitness
        fitnesses = [compute_c5_exact(ind, dx) for ind in population]
        
        generation += 1
        
        # Slow down mutation as generations increase
        if generation > 100:
            mutation_rate *= 0.995
    
    # Final computation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
