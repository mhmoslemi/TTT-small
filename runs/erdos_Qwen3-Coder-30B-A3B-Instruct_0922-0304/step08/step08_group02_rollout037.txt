import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create a reasonable starting point
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1.0
        h_current = h_current * (target_sum / np.sum(h_current))
        # Add some random perturbation
        perturb = np.random.normal(0, 0.02, n_points)
        h_current += perturb
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # MCMC parameters
    temperature = 0.1
    cooling_rate = 0.999
    min_temperature = 1e-6
    perturbation_scale = 0.05
    max_iter = int(budget_s * 10)  # Rough estimate
    
    # Iteration tracking
    no_improvement_count = 0
    max_no_improvement = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Propose new state
        h_proposed = h_current.copy()
        idx = np.random.randint(n_points)
        
        # Perturb one element
        delta = np.random.normal(0, perturbation_scale)
        h_proposed[idx] += delta
        h_proposed[idx] = np.clip(h_proposed[idx], 0, 1)
        
        # Renormalize to maintain sum constraint
        h_proposed = h_proposed * (target_sum / np.sum(h_proposed))
        
        # Compute objective values
        current_c5 = compute_c5(h_current, dx)
        proposed_c5 = compute_c5(h_proposed, dx)
        
        # Metropolis acceptance criterion
        delta_c5 = proposed_c5 - current_c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / temperature):
            h_current = h_proposed
            if proposed_c5 < best_c5:
                best_h = h_proposed.copy()
                best_c5 = proposed_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Adaptive cooling
        temperature = max(min_temperature, temperature * cooling_rate)
        
        # Adaptive perturbation
        if no_improvement_count > 100:
            perturbation_scale *= 0.95
            no_improvement_count = 0
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    refined_h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp = refined_h.copy()
        h_temp[idx] += delta
        h_temp[idx] = np.clip(h_temp[idx], 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(refined_h, dx):
            refined_h = h_temp.copy()
    
    final_c5 = compute_c5(refined_h, dx)
    return (refined_h.tolist(), final_c5, n_points)
