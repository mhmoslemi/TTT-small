```python
import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective from frequency domain"""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, surrogate, learning_rate=0.01):
    """Update H in frequency domain to reduce surrogate"""
    # Find high contribution bins
    idxs = np.argsort(surrogate)[::-1][:len(H)//4]
    # Adjust these bins
    for i in idxs:
        # Reduce magnitude of high surrogate values
        H[i] -= learning_rate * (H[i] - 0.5) * surrogate[i]
    return H

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.rand(n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Fourier approach: transform h, work in frequency domain
        H = fft(h_current)
        
        # Compute surrogate objective in frequency domain
        surrogate = compute_surrogate(H)
        
        # Update in frequency domain
        H_new = H.copy()
        H_new = spectral_update(H_new, surrogate, learning_rate)
        
        # Reconstruct h
        h_new = np.real(ifft(H_new))
        
        # Apply feasibility projection
        h_new = project_h(h_new, target_sum)
        
        # Compute C5 for new h
        c5_new = compute_c5(h_new, dx)
        
        # Update best solution if improved
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Use the best one for next iteration
        h_current = h_new.copy()
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.5
        
        iteration += 1
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with local search
    h_refined = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Gradient-like local search
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
```
