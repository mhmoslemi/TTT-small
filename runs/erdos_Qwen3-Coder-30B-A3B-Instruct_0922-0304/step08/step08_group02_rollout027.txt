import numpy as np
from scipy.signal import correlate
import time
from math import inf

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to ensure it sums to target_sum and stays in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05, max_mutation=0.1):
    """Mutate h with given rate and max mutation."""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            delta = np.random.uniform(-max_mutation, max_mutation)
            h_mut[i] = np.clip(h_mut[i] + delta, 0, 1)
    return h_mut

def local_refinement(h, target_sum, dx, steps=50):
    """Apply local gradient-like refinement."""
    h_ref = h.copy()
    for _ in range(steps):
        # Find where max overlap occurs
        overlap = correlate(h_ref, 1 - h_ref, mode="full")
        max_idx = np.argmax(overlap)
        
        # Create small perturbations to reduce max overlap
        h_trial = h_ref.copy()
        i = np.random.randint(len(h_ref))
        delta = np.random.normal(0, 0.01)
        h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        
        # Renormalize
        h_trial = repair_h(h_trial, target_sum)
        
        # If better, accept
        if compute_c5(h_trial, dx) < compute_c5(h_ref, dx):
            h_ref = h_trial
    return h_ref

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = []
    
    # Add the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
        population.append(h_init)
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Fill the rest with random constructions
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = inf
    no_improvement = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = fitness[0]
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Select top 20% as parents
        n_parents = max(2, n_pop // 5)
        parents = sorted_population[:n_parents]
        
        # Generate new population
        new_population = parents.copy()  # Elitism
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Repair all individuals
        for i in range(len(new_population)):
            new_population[i] = repair_h(new_population[i], target_sum)
            
        # Keep only n_pop individuals
        population = new_population[:n_pop]
        
        # Local refinement on best individuals
        for i in range(min(3, len(population))):
            refined = local_refinement(population[i], target_sum, dx)
            population[i] = refined
            
        generation += 1
    
    # Final refinement
    if best_h is not None:
        final_h = local_refinement(best_h, target_sum, dx, steps=100)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    if best_h is None:
        # If no solution found, return the best from last population
        final_fitness = [compute_c5(h, dx) for h in population]
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        best_c5 = final_fitness[best_idx]
        
    return (best_h.tolist(), best_c5, n_points)
