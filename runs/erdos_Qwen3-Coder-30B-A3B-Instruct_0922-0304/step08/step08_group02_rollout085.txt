import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    tau = 1000
    n_steps_no_improvement = 0
    max_no_improvement = 500
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    temperature = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current C5 value
        c5_current = compute_c5(h_current, dx)
        
        # Propose new state
        h_new = h_current.copy()
        i = np.random.randint(n_points)
        delta = np.random.normal(0, sigma)
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        
        # Adjust other elements to maintain sum constraint
        current_sum = np.sum(h_new)
        scale_factor = target_sum / current_sum
        h_new = h_new * scale_factor
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Compute acceptance probability
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0:  # Better solution
            accept = True
        else:
            if temperature > T_min:
                accept = np.random.rand() < exp(-delta_c5 / temperature)
            else:
                accept = False
        
        # Accept or reject the new state
        if accept:
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
                n_steps_no_improvement = 0
        else:
            n_steps_no_improvement += 1
        
        # Update temperature
        temperature = max(T_min, T0 * (alpha ** (iteration / tau)))
        
        # Adaptive adjustment
        if n_steps_no_improvement > max_no_improvement:
            sigma *= 0.9
            n_steps_no_improvement = 0
            if temperature < T_min:
                temperature = T0 * 0.1
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Fine-tune the best solution
    h_refined = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp = h_refined.copy()
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
