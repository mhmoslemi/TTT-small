import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    scale_factor = (n_points / 2) / np.sum(h)
    return h * scale_factor

def compute_gradient(h, shift, n_points):
    """Compute gradient for given shift k* (index in full correlation)"""
    grad = np.zeros(n_points)
    # The gradient for h[i] is (1 - h[i + shift]) if valid
    # Full correlation has length 2*n_points - 1
    # shift index goes from 0 to 2*n_points - 2
    # For correlation index shift, we're looking at h[i] and (1 - h[i + shift - (n_points-1)])
    #   Actually for cross-correlation with 1-h, the gradient is 1 - h (x+shift)
    if shift < n_points - 1:
        # We're considering h[i] against (1-h[i+shift])
        for i in range(n_points - shift):
            grad[i] += 1 - h[i + shift]
    else:
        # shift >= n_points-1, so we're considering h[i+shift-(n_points-1)] against (1-h[i])
        for i in range(shift - (n_points - 1), n_points):
            grad[i] += 1 - h[i + shift - (n_points - 1)]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Compute gradient
        gradient = compute_gradient(h_current, max_overlap_idx, n_points)
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Project to feasible set
        h_new = project_h(h_new, n_points)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update h for next iteration
        h_current = h_new.copy()
        
        # Adjust step size
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.5
            
        # Perturbation occasionally to escape local minima
        if no_improvement_count > 100 and np.random.rand() < 0.05:
            perturbation_magnitude = 0.05 * np.abs(h_current)
            h_current += np.random.normal(0, perturbation_magnitude)
            h_current = project_h(h_current, n_points)
            no_improvement_count = 0
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
