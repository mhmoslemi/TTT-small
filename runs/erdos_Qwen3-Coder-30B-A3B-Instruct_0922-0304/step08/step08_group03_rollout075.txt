import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    mid = n // 2
    left = np.linspace(0, 1, mid)
    right = np.linspace(1, 0, n - mid)
    return np.append(left, right)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        # Validate and normalize
        if np.sum(h) > 0:
            h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    # Optimization parameters
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iter = 20000
    
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_idx = np.argmax(overlap)
        
        # Compute exact gradient (this is approximate but more direct)
        # Based on the idea that for the maximum overlap, we want to minimize it
        # and its gradient w.r.t. h is approximately:
        # sign(overlap_max) * (1 - 2*h) / n_points
        h_gradient = (1 - 2 * h) * np.sign(overlap_max) / n_points
        
        # Momentum update
        velocity = momentum * velocity + learning_rate * h_gradient
        h_candidate = h - velocity  # Note: we are minimizing, so we subtract
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
        else:
            # Failure to improve: shrink step size and retry once
            learning_rate *= shrink_factor
            velocity = momentum * velocity + learning_rate * h_gradient
            h_candidate = h - velocity
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            new_c5 = compute_c5(h_candidate, dx)
            if new_c5 < best_c5:
                h = h_candidate.copy()
                best_c5 = new_c5
                best_h = h.copy()
            else:
                # No improvement even after reducing learning rate, reset
                learning_rate = 0.001  # Reset
                velocity = np.zeros(n_points)
    
    return (best_h.tolist(), best_c5, n_points)
