import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Blend two parents to create offspring."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h by slightly adjusting values."""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += random.uniform(-0.1, 0.1)
    return h

def local_refinement(h, target_sum, dx, steps=50):
    """Apply gradient-like local refinement."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for _ in range(steps):
        if random.random() < 0.5:
            # Small random perturbation
            idx = random.randint(0, len(h) - 1)
            delta = random.uniform(-0.02, 0.02)
            h[idx] = np.clip(h[idx] + delta, 0, 1)
        else:
            # Adjust toward reducing max overlap
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            # Try to reduce the maximum overlap by modifying a few points
            for i in range(min(5, len(h))):
                idx = random.randint(0, len(h) - 1)
                delta = random.uniform(-0.01, 0.01)
                h[idx] = np.clip(h[idx] + delta, 0, 1)
        
        # Renormalize
        h = h * (target_sum / np.sum(h))
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = []
    
    # Add the initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
        population.append(h_current)
    
    # Fill the rest with random valid solutions
    for _ in range(n_pop - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep top 20%
        elite_count = max(1, n_pop // 5)
        parents = population[:elite_count]
        
        # Create new population
        new_population = parents.copy()
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, 0.05 * (1 - generation/100))
            child2 = mutate(child2, 0.05 * (1 - generation/100))
            
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < n_pop:
                new_population.append(child2)
        
        # Local refinement on top solutions
        for i in range(min(3, len(new_population))):
            refined_h, refined_c5 = local_refinement(new_population[i], target_sum, dx)
            if refined_c5 < fitness[i]:
                new_population[i] = refined_h
        
        population = new_population
        generation += 1
        
        if generation % 10 == 0:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    
    # Final local refinement on the best solution
    if best_h is not None:
        best_h, final_c5 = local_refinement(best_h, target_sum, dx, 30)
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
