import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use convex optimization to solve the problem exactly
    # Variables
    h_var = cp.Variable(n_points)
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Objective: minimize the maximum overlap
    # We formulate this as a second-order cone program
    
    # Compute the full correlation matrix in a structured way
    # For maximum overlap, we want to minimize max_k int h(x)(1-h(x+k)) dx
    # This is the same as minimizing the maximum of all cross-correlations
    # We solve a convex relaxation with a symmetric approach
    
    # Instead of directly solving the nonlinear problem, 
    # we iteratively refine with local search
    
    # Start with a better initial construction leveraging known structure
    
    # Use SLSQP for final refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints_slsqp = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Try a few restarts with different warm starts
    restart_count = 3
    for restart in range(restart_count):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create a refined starting point
        if restart == 0:
            start_h = h.copy()
        else:
            # Add more structure from previous solutions
            start_h = np.random.uniform(0, 1, n_points)
            start_h = np.clip(start_h, 0, 1)
            current_sum = np.sum(start_h)
            if current_sum > 0:
                start_h = start_h * (target_sum / current_sum)
        
        # Local optimization
        result = minimize(objective, start_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints_slsqp, 
                         options={'maxiter': 1000, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
        else:
            # Try also direct CVXPY approach
            h_solver = cp.Variable(n_points)
            obj = cp.Minimize(cp.max(cp.Convolution(h_solver, 1 - h_solver)))
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status == cp.OPTIMAL:
                    sol = h_solver.value
                    sol = np.clip(sol, 0, 1)
                    sol_sum = np.sum(sol)
                    if sol_sum > 0:
                        sol = sol * (target_sum / sol_sum)
                    c5_test = evaluate_overlap(sol, dx)
                    if c5_test < best_c5:
                        best_h = sol
                        best_c5 = c5_test
            except:
                pass
        # Early exit if time suffices
        if time.time() - start_time > budget_s - 10:
            break

    # Final validation and cleanup
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)  
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
