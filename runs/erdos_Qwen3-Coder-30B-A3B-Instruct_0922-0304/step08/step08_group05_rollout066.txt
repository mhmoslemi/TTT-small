import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Parameters
    learning_rate = 0.01
    max_iter = 10000
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5_exact(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps and find the maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_lag = max_idx - (n_points - 1)
        
        # Adjust h to reduce the maximum overlap at this lag
        # Simple gradient-based adjustment (doesn't compute exact gradient)
        step_size = learning_rate * (1.0 / (1.0 + iteration / 1000.0))
        
        # Perturb h to reduce overlap at max_lag
        h_new = h.copy()
        for i in range(n_points):
            j = (i + max_lag) % n_points
            if 0 <= j < n_points:
                # Adjust to decrease overlap contribution at this lag
                h_new[i] = np.clip(h_new[i] - step_size * (1 - 2 * h[i]), 0, 1)
                h_new[j] = np.clip(h_new[j] + step_size * (1 - 2 * h[j]), 0, 1)
        
        # Re-normalize
        h_new = h_new * (target_sum / np.sum(h_new))
        h = h_new
        
        # Occasionally adjust learning rate
        if iteration % 1000 == 0:
            learning_rate *= 0.995
    
    # Final refinement
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
