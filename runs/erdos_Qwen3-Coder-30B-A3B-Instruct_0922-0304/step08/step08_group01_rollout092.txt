import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed, target_sum):
    """Round relaxed solution to binary with sum constraint."""
    # Use median as threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h_binary)
    diff = current_sum - target_sum
    
    if diff > 0:
        # Need to flip some 1s to 0s
        # Find indices where h_binary is 1 but h_relaxed is below threshold
        # Flip least significant 1s
        indices = np.where(h_binary == 1)[0]
        # Sort by how close they are to threshold
        diffs = np.abs(h_relaxed[indices] - threshold)
        to_flip = np.argsort(diffs)[:int(diff)]
        for i in to_flip:
            h_binary[indices[i]] = 0
    elif diff < 0:
        # Need to flip some 0s to 1s
        # Find indices where h_binary is 0 but h_relaxed is above threshold
        indices = np.where(h_binary == 0)[0]
        # Sort by how close they are to threshold
        diffs = np.abs(h_relaxed[indices] - threshold)
        to_flip = np.argsort(diffs)[::-1][:int(-diff)]
        for i in to_flip:
            h_binary[indices[i]] = 1
            
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try LP relaxation approach
    try:
        # Define relaxed LP problem
        h_r = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = []
        # Sum constraint
        constraints.append(cp.sum(h_r) == target_sum)
        
        # Bound constraints
        constraints.append(h_r >= 0)
        constraints.append(h_r <= 1)
        
        # Overlap constraints
        for k in range(1 - n_points, n_points):
            start_idx = max(0, k)
            end_idx = min(n_points, n_points + k)
            if start_idx < end_idx:
                # Sum of h_r[i] * (1 - h_r[i + k]) for valid i
                if k >= 0:
                    expr = cp.sum(cp.multiply(h_r[start_idx:end_idx], 1 - h_r[start_idx+k:end_idx+k]))
                else:
                    expr = cp.sum(cp.multiply(h_r[start_idx:end_idx], 1 - h_r[start_idx:end_idx-k]))
                constraints.append(expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        problem = cp.Problem(objective, constraints)
        
        # Try solving with different solvers
        try:
            problem.solve(solver=cp.MOSEK, verbose=False, time_limit=10)
        except:
            try:
                problem.solve(solver=cp.GLPK, verbose=False, time_limit=10)
            except:
                try:
                    problem.solve(solver=cp.CVXOPT, verbose=False, time_limit=10)
                except:
                    pass
        
        # If successful, extract solution and get binary version
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] and \
           problem.value is not None and not np.isnan(problem.value):
            h_relaxed = h_r.value
            if h_relaxed is not None:
                h_binary = round_to_binary(h_relaxed, target_sum)
                # Check if this gives improved result
                c5_binary = compute_c5(h_binary, dx)
                if c5_binary < best_c5:
                    best_c5 = c5_binary
                    best_h = h_binary.copy()
                    
            # Do local search around the binary solution
            h_local = h_binary.copy()
            for _ in range(1000):
                if time.time() - start_time >= budget_s - 1:
                    break
                    
                # Update one element at a time
                i = np.random.randint(0, n_points)
                old_val = h_local[i]
                new_val = old_val + np.random.normal(0, 0.02)
                new_val = np.clip(new_val, 0, 1)
                h_local[i] = new_val
                
                # Normalize
                h_local = normalize_h(h_local, target_sum)
                
                # Check improvement
                c5_local = compute_c5(h_local, dx)
                if c5_local < best_c5:
                    best_c5 = c5_local
                    best_h = h_local.copy()
                    
    except Exception as e:
        # If any error occurs, just continue with standard optimization
        pass
    
    # If we're still below target or haven't improved much, do gradient-based optimization
    if best_c5 > 0.3807:
        h = normalize_h(best_h, target_sum)
        max_iter = 30000
        iter_count = 0
        
        while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
            # Simple gradient descent with adjustments
            overlap = correlate(h, 1 - h, mode="full") * dx
            
            # Find max overlap and index
            current_max = np.max(overlap)
            max_idx = np.argmax(overlap)
            
            # Create simple update weights based on spatial distribution
            weights = np.zeros(n_points)
            for i in range(n_points):
                # Gradient-like update (simplified)
                for k in range(-n_points + 1, n_points):
                    if i + k >= 0 and i + k < n_points:
                        old_weight = 0
                        if i + k == max_idx:
                            old_weight = 1
                        weights[i] += (1 - 2 * h[i]) * (1 - h[i + k])
            
            # Update
            step_size = 0.01
            for i in range(n_points):
                h[i] = h[i] + step_size * weights[i]
                h[i] = np.clip(h[i], 0, 1)
            
            # Normalize
            h = normalize_h(h, target_sum)
            
            # Evaluate
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
            # Early stopping
            if best_c5 <= 0.3807:
                break
                
            iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
