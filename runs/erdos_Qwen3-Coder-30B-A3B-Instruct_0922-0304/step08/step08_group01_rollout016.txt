import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed(h_init, n_points, dx):
    """Solve the relaxed LP problem."""
    # Add slack variable
    z = cp.Variable()
    
    # Decision variables
    h = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
    ]
    
    # Add overlap constraints
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
        else:
            expr = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
        constraints.append(expr <= z)
    
    # Objective
    objective = cp.Minimize(z)
    
    # Solve with MOSEK
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except:
        # Fallback to a basic gradient-based approach if failing
        return h_init
    
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        return h.value
    else:
        return h_init

def round_to_binary(h_relaxed):
    """Round the relaxed solution to binary values."""
    # Use median threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h_binary)
    target_sum = len(h_binary) / 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = int(current_sum - target_sum)
        # Find indices where value was >= threshold
        idxs = np.where(h_binary == 1)[0]
        values = h_relaxed[idxs]
        # Flip those closest to threshold
        order = np.argsort(np.abs(values - threshold))
        to_flip = idxs[order[:diff]]
        h_binary[to_flip] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        diff = int(target_sum - current_sum)
        # Find indices where value was < threshold
        idxs = np.where(h_binary == 0)[0]
        values = h_relaxed[idxs]
        # Flip those closest to threshold
        order = np.argsort(np.abs(values - threshold))
        to_flip = idxs[order[:diff]]
        h_binary[to_flip] = 1
        
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Start with a uniform random
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try several iterations of relaxation + rounding
    max_iter = 10
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve relaxed LP
        h_relaxed = solve_relaxed(h, n_points, dx)
        
        # Convert to binary
        h_binary = round_to_binary(h_relaxed)
        
        # Enforce exact constraint
        h_binary = normalize_h(h_binary, target_sum)
        
        # Evaluate
        c5 = compute_c5(h_binary, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_binary.copy()
            
        # Small perturbation to h if we have time
        if time.time() - start_time < budget_s - 1 and _ < max_iter - 1:
            # Use binary values as a new starting point
            h = h_binary
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
