```python
import numpy as np
from scipy.signal import correlate
import time
from scipy.fft import fft, ifft
import math

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to sum=target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_surrogate(H):
    """Compute the surrogate objective in frequency domain"""
    conj = np.conj(1 - H)
    product = H * conj
    return np.abs(product)

def spectral_update(H, freq_idx, learning_rate):
    """Update spectral components to reduce objective"""
    H_new = H.copy()
    # Simple gradient descent step on the frequency domain
    H_new[freq_idx] -= learning_rate * np.real(H_new[freq_idx])
    return H_new

def compute_c5(h, dx):
    """Compute C5 value"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    max_iter = 10000
    
    # Frequency domain approach
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Transform to frequency domain
        H = fft(h)
        
        # Compute surrogate
        surrogate = compute_surrogate(H)
        
        # Find indices with highest surrogate values
        freq_indices = np.argsort(surrogate)[-10:]  # Top 10
        
        # Update these components
        learning_rate = 0.01 * np.exp(-iter_count / 500)
        H_new = H.copy()
        for idx in freq_indices:
            H_new = spectral_update(H_new, idx, learning_rate)
        
        # Reconstruct in time domain
        h_new = np.real(ifft(H_new))
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new.copy()
        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    h_refined = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp = h_refined.copy()
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
```
