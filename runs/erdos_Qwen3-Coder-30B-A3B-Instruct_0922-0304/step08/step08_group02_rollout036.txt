import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create a feasible starting point
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1.0
        h_current = h_current * (target_sum / np.sum(h_current))
        
    # Ensure sum constraint is satisfied
    h_current = h_current * (target_sum / np.sum(h_current))
    
    # MCMC with simulated annealing parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    sigma = 0.05
    tau = 1000
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Annealing temperature
    T = T0
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        c5_current = compute_c5(h_current, dx)
        
        # Propose new state
        h_new = h_current.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.normal(0, sigma)
        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Metropolis-Hastings acceptance
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.random() < exp(-delta_c5 / T):
            h_current = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (iteration / tau)))
        
        # Adaptive perturbation adjustment
        if no_improvement_count > max_no_improvement:
            sigma *= 0.9
            no_improvement_count = 0
            
        iteration += 1
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
