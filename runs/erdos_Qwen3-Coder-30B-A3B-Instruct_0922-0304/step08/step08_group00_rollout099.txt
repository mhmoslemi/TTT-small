import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function with some structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iterations = 2000
    w = 0.8
    c1 = 2.0
    c2 = 2.0
    v_max = 0.05
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(n_particles - 1):
        perturbation = np.random.normal(0, 0.05, n_points)
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        particles.append(new_h)
    
    # Evaluate initial fitness
    fitness = [evaluate_overlap(p, dx) for p in particles]
    
    # Initialize best positions
    p_best = particles.copy()
    p_best_fitness = fitness.copy()
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    # Track stagnation
    stagnation_count = 0
    previous_best = g_best_fitness
    
    # PSO iteration
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * (particles[i] - particles[i]) + \
                c1 * r1 * (p_best[i] - particles[i]) + \
                c2 * r2 * (g_best - particles[i])
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_h = particles[i] + v
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize
            current_sum = np.sum(new_h)
            if current_sum > 0:
                new_h = new_h * (target_sum / current_sum)
            
            # Evaluate new fitness
            new_fitness = evaluate_overlap(new_h, dx)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_h.copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < g_best_fitness:
                    g_best = new_h.copy()
                    g_best_fitness = new_fitness
                    stagnation_count = 0
                    if g_best_fitness < best_c5:
                        best_h = g_best.copy()
                        best_c5 = g_best_fitness
            else:
                particles[i] = new_h.copy()
        
        # Check for stagnation
        if g_best_fitness >= previous_best:
            stagnation_count += 1
        else:
            stagnation_count = 0
        previous_best = g_best_fitness
        
        # Apply shaking if stagnating
        if stagnation_count > 50:
            for i in range(n_particles):
                if np.random.rand() < 0.3:  # 30% chance to shake
                    shake_magnitude = 0.03
                    perturbation = np.random.normal(0, shake_magnitude, n_points)
                    particles[i] = particles[i] + perturbation
                    particles[i] = np.clip(particles[i], 0, 1)
                    current_sum = np.sum(particles[i])
                    if current_sum > 0:
                        particles[i] = particles[i] * (target_sum / current_sum)
            stagnation_count = 0
    
    # Local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, g_best, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
