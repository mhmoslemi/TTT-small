import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_step_function(h, dx, n_points):
    """Slightly perturb a step function."""
    h_new = h.copy()
    # Choose a random segment to modify
    idx = np.random.randint(0, n_points)
    # Add small random perturbation
    pert = np.random.normal(0, 0.05)
    h_new[idx] += pert
    # Clip and renormalize
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, n_points / 2)
    return h_new

def pattern_search(initial_h, dx, n_points, max_iter=5000):
    """Perform pattern search on step function."""
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    no_improvement_count = 0
    
    for i in range(max_iter):
        # Perturb the function
        new_h = perturb_step_function(best_h, dx, n_points)
        new_c5 = evaluate_c5(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 1000:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 1000:
        # Perform pattern search
        h_new, c5_new = pattern_search(best_h, dx, n_points, max_iter=1000)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        # Occasionally try a fresh random initialization
        if iter_count % 10 == 0 and iter_count > 0:
            h = np.random.rand(n_points).astype(np.float64)
            h = normalize_h(h, n_points / 2)
            h_new, c5_new = pattern_search(h, dx, n_points, max_iter=500)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
        iter_count += 1
        
        # Early stop if we've reached our target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
