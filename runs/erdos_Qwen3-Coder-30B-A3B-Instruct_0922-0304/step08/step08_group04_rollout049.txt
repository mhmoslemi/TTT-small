import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the C5 objective with respect to h."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient based on maximum overlap contribution
    # For each position i, we compute the contribution to the gradient
    # from the correlation at max_idx
    # The contribution to -h[i] is the derivative of sum over k of h[i]*h[i+k] in 1-h[i+k]
    # which is -h[i+k] where k = max_idx - i (mod n_points)
    for i in range(n_points):
        # We look at the shift that causes the maximum correlation
        # For lag k=max_idx - i (mod n_points), we have contributions
        # But we're taking derivative of sum of h[i]*(1-h[i+k])
        # So d/dh[i] = 1-h[i+k], d/dh[i+k] = -h[i]
        # So overall contribution for position i is derivative of max overlap at lag k_max
        k = max_idx - i
        if k >= 0:
            j = (i + k) % n_points
        else:
            j = (i - (-k)) % n_points
        
        # The contribution to gradient from this overlapping element
        # corresponds to d/dh[i] of the overlap term at k_max
        # which is -h[j] if j is the element contributing to this shift (but sign is flipped)
        # The derivative of the overlap with respect to h[i] is multiplied by -1 
        # to get the overall effect on the objective (C5 is max overlap, we want to minimize)
        
        # Simplified version:
        # If the maximum correlation occurs at k, then we consider the derivative
        # with respect to h[i] of h[i]*h[i+k], which gives 1-h[i+k] when added to gradient
        
        grad[i] = -overlap[max_idx] # just approximate; this gradient is used for Frank-Wolfe
        
    return grad

def solve_linear_subproblem(grad, n_points):
    """Solve the linear subproblem: minimize grad^T v subject to constraint sum(v) = n_points/2, 0 <= v <= 1."""
    # This is a simple LP: minimize c^T x subject to A x = b, lb <= x <= ub
    # Here: c = grad, A = [1,1,...,1] (1-row matrix), b = n_points/2, lb=0, ub=1
    
    # More efficient approach:
    # Want to minimize sum_i grad[i]*v[i] s.t. sum_i v[i] = n_points/2, 0 <= v[i] <= 1
    
    # Sort gradient by value (indices)
    sorted_indices = np.argsort(grad)
    
    # Initialize v
    v = np.zeros(n_points)
    
    # Strategy: take the best values (smallest grad) and set them to 1.0 
    # until we reach the sum constraint
    remaining_sum = n_points / 2.0
    idx = 0
    
    while remaining_sum > 0 and idx < n_points:
        # Take the index with smallest gradient and set it to 1 if possible
        i = sorted_indices[idx]
        if remaining_sum >= 1.0:
            v[i] = 1.0
            remaining_sum -= 1.0
        else:
            v[i] = remaining_sum
            remaining_sum = 0
        idx += 1
    
    # Even if remaining_sum is not exactly zero, set the rest to 0
    # This might happen due to roundoff errors
    for i in sorted_indices[idx:]:
        if remaining_sum > 0:
            v[i] = min(remaining_sum, 1.0)
            remaining_sum -= v[i]
        else:
            v[i] = 0
    
    # Ensure that sum(v) = n_points/2 exactly 
    if np.abs(np.sum(v) - n_points/2) > 1e-8:
        # Adjust last element
        diff = n_points/2 - np.sum(v)
        if v[-1] + diff >= 0 and v[-1] + diff <= 1:
            v[-1] += diff
        else:
            # Just correct sum via uniform adjustment
            v += diff / n_points
            
    return v

def frank_wolfe_step(h, dx, n_points, max_iter=1000, tolerance=1e-8):
    """Frank-Wolfe optimization for C5 minimization."""
    # We will maximize the overlap C5 = max_k int h(x)(1 - h(x+k)) dx
    # to minimize C5, we can consider the worst-case shift
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        # Compute current gradient (approximate)
        grad = compute_gradient(best_h, dx, n_points)
        
        # Solve linear subproblem
        v = solve_linear_subproblem(grad, n_points)
        
        # Compute step size via line search (simplified)
        # Try to choose step size η such that we minimize the objective
        # We do this by trying a few values of η between 0.1 and 0.9
        best_eta = 0.1
        best_new_c5 = best_c5
        
        for eta in np.linspace(0.1, 0.9, 10):
            new_h = (1 - eta) * best_h + eta * v
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, n_points / 2)
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_eta = eta
        
        # Update
        best_h = (1 - best_eta) * best_h + best_eta * v
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, n_points / 2)
        
        if best_new_c5 < best_c5:
            best_c5 = best_new_c5
        else:
            # Small improvement check
            if abs(best_c5 - best_new_c5) < tolerance:
                break
                
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n_points = 70 for consistency 
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        # Start with uniform distribution
        h = np.full(n_points, 0.5)
        h = normalize_h(h, n_points / 2)
    
    # Optimize with Frank-Wolfe method
    best_h, best_c5 = frank_wolfe_step(h, dx, n_points, max_iter=10000, tolerance=1e-8)
    
    # Final refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
