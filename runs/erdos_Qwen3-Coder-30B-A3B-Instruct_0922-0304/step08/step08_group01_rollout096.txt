import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    dx = 2.0 / n_points

    else:

    # Normalize to target sum
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simulated Annealing parameters
    current_temp = 0.1
    cooling_rate = 0.999
    max_iterations = 100000
    
    iter_count = 0
    last_improvement = 0
    stagnation_threshold = 1000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        # Make a copy for perturbation
        h_new = h.copy()
        
        # Select a random perturbation
        choice = np.random.choice(['flip', 'adjust', 'shuffle'], 
                                p=[0.5, 0.3, 0.2])
        
        if choice == 'flip':
            i = np.random.randint(n_points)
            h_new[i] = 1 - h_new[i]
        elif choice == 'adjust':
            i = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        else:  # shuffle
            length = np.random.randint(1, min(6, n_points))
            i = np.random.randint(n_points - length + 1)
            segment = h_new[i:i+length]
            np.random.shuffle(segment)
            h_new[i:i+length] = segment
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the move
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iter_count
        else:
            delta_e = new_c5 - best_c5
            if np.exp(-delta_e / current_temp) > np.random.rand():
                h = h_new.copy()
                best_c5 = new_c5
                last_improvement = iter_count
            else:
                # Revert to best if no improvement
                h = best_h.copy()
        
        # Temperature cooling
        current_temp *= cooling_rate
        
        # Stagnation detection and reheating
        if iter_count - last_improvement > stagnation_threshold:
            current_temp = max(0.01, current_temp * 2)
            last_improvement = iter_count
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
