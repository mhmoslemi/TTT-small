import numpy as np
from scipy.fftpack import fft, ifft
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1], scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate frequency-domain objective."""
    conj = np.conj(1 - H)
    return np.abs(H * conj)

def spectral_update(H, learning_rate=0.01):
    """Update H to reduce surrogate objective."""
    S = compute_surrogate(H)
    # Target frequencies with highest surrogate values
    high_freq_idx = np.argsort(S)[-5:]  # Top 5 frequencies
    # Update those bins
    for i in high_freq_idx:
        # Reduce magnitude along the frequency dimension
        H[i] -= learning_rate * np.sign(H[i]) * np.abs(H[i])
    return H

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.rand(n_points)
        h_current = project_h(h_current, target_sum)
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Periodically reset to best h if we're not improving
        if iteration % 50 == 0 and iteration > 0:
            h_current = best_h.copy()
            
        # Forward FFT
        H = fft(h_current)
        
        # Update in frequency domain
        H_new = spectral_update(H.copy(), learning_rate)
        
        # Inverse FFT to get updated h
        h_new = np.real(ifft(H_new))
        h_new = project_h(h_new, target_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h_current = h_new.copy()
        iteration += 1
        
        # Decay learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.5
            
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with local perturbations
    h_refined = best_h.copy() 
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
