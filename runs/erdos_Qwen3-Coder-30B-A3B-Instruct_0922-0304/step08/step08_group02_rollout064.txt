import numpy as np
from scipy.signal import correlate
import time
from math import exp

def evaluate_solution(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Start with a basic construction
        h_current = np.ones(n_points) * 0.5
        h_current[10:20] = 1.0
        h_current[50:60] = 0.0
        h_current = h_current * (target_sum / np.sum(h_current))
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    tau = 1000  # cooling timescale
    N_steps_no_improvement = 1000
    restart_factor = 0.5
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = evaluate_solution(h_current, dx)
    
    T = T0
    t = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Propose new state
        h_proposed = h_current.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.normal(0, sigma)
        h_proposed[i] = np.clip(h_proposed[i] + delta, 0, 1)
        
        # Project to feasible region maintaining sum constraint
        current_sum = np.sum(h_proposed)
        scale_factor = target_sum / current_sum
        h_proposed = h_proposed * scale_factor
        
        # Compute C5 values
        c5_current = evaluate_solution(h_current, dx)
        c5_proposed = evaluate_solution(h_proposed, dx)
        
        # Accept with Metropolis-Hastings probability
        delta_c5 = c5_proposed - c5_current
        
        if delta_c5 < 0 or np.random.random() < exp(-delta_c5 / T):
            h_current = h_proposed.copy()
            if c5_proposed < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_proposed
                no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (t / tau)))
        
        # Adaptive adjustment
        if no_improvement_count > N_steps_no_improvement:
            sigma *= 0.9
            T = T0 * restart_factor
            no_improvement_count = 0
            
        t += 1
        
        # Early stopping if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with local search
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_test = h_final.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.005)
        h_test[i] = np.clip(h_test[i] + delta, 0, 1)
        h_test = h_test * (target_sum / np.sum(h_test))
        c5_test = evaluate_solution(h_test, dx)
        if c5_test < evaluate_solution(h_final, dx):
            h_final = h_test.copy()
    
    final_c5 = evaluate_solution(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
