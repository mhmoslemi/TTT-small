import numpy as np
from scipy.signal import correlate
import time

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_gradient(h, dx):
    """Compute approximate gradient of C5 with respect to h."""
    # This is simplified -- in practice, you'd compute it more precisely
    # Here we return a rough approximation based on change in overlap
    n = len(h)
    return (1 - 2 * h) / n

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_generations = 2000
    no_improvement_patience = 100
    patience_counter = 0
    
    # Create initial population
    population = []
    for _ in range(population_size):
        ph = h + np.random.normal(0, 0.01, n_points)
        ph = repair_h(ph, target_sum)
        population.append(ph)
    
    # Add the best solution to the population
    population[0] = h.copy()
    
    generation = 0
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        current_best_c5 = fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep top 50%
        selected = population[:population_size//2]
        
        # Crossover and mutation
        new_population = selected[:]
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = selected[np.random.randint(len(selected))], selected[np.random.randint(len(selected))]
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = selected[np.random.randint(len(selected))].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for idx in mutate_indices:
                    child[idx] += np.random.normal(0, 0.02)
            
            # Repair and add to population
            child = repair_h(child, target_sum)
            
            # Gradient descent refinement for better local search
            gradient = compute_gradient(child, dx)
            step_size = 0.05 / (1 + np.linalg.norm(gradient))
            child = child - step_size * gradient
            
            # Repair again after gradient step
            child = repair_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
        
        # Parameter adaptation
        if generation % 10 == 0:
            if len(fitness) > 1 and fitness[0] >= fitness[1]:
                patience_counter += 1
                if patience_counter > no_improvement_patience:
                    mutation_rate = min(0.5, mutation_rate * 1.2)
                    patience_counter = 0
            else:
                patience_counter = 0
        
        generation += 1
    
    # Final check with original evaluator
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
