import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Projected gradient descent with Nesterov momentum
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tolerance = 1e-6
    
    # Initialize variables
    y = h.copy()
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    prev_c5 = best_c5 + 1.0
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Nesterov momentum update
        h = y + momentum * (y - v)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        # Compute overlap for all lags to find the maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        # Compute gradient
        # Gradient for h_i is the sum of (1-h_{i-k}) for all k
        # but we only consider the maximum overlap contribution
        grad = np.zeros(n_points)
        lag = max_overlap_idx - (len(h) - 1)
        for i in range(len(h)):
            j = (i - lag) % len(h)
            grad[i] = 1 - h[j]
        
        # Projected gradient step
        y = h - step_size * grad
        y = np.clip(y, 0, 1)
        
        # Re-scale to satisfy sum constraint
        current_sum = np.sum(y)
        if current_sum > 0:
            y = y * (target_sum / current_sum)
        
        # Store previous values for momentum
        v = y.copy()
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(y, dx)
        
        if new_c5 < best_c5:
            best_h = y.copy()
            best_c5 = new_c5
            
        # Adaptive step size
        if abs(prev_c5 - new_c5) < tolerance:
            step_size = max(min_step_size, step_size * 0.5)
        prev_c5 = new_c5
        
        # Periodic warm restarts
        if iteration % 1000 == 0 and iteration > 0:
            step_size = 0.1
            momentum = min(0.95, momentum + 0.01)
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
