import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5_value(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy the sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def frank_wolfe_step(h, dx, n_points, max_iter=100):
    """Frank-Wolfe step with linear subproblem solved via CVXPY."""
    # Solve linear subproblem: minimize grad^T * v subject to constraints
    # using CVXPY (more accurate than simple approach)
    
    # Compute current gradient
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # The gradient contribution
    # For each i, grad[i] corresponds to the derivative w.r.t. h[i]
    # The gradient is approximately: 
    # grad[i] = (1 - 2*h[i]) * sum_{j: overlap_k_max} 1 - h[j]
    # But let's compute the actual gradient for the maximum overlap
    
    # Just compute gradient as more detailed way:
    # grad[i] = constant might be wrong here as actually it's not just scalar
    # We compute gradient to subproblem with respect to overlap term
    
    # simulate linearization:
    # For small changes, we linearize over the maximum term
    # Assume maximum overlap is at position max_idx:
    # overlap_k = sum_i h[i] * (1 - h[(i+max_idx) % n_points])
    # Gradient of overlap w.r.t. h[i] is 1 - h[(i+max_idx) % n_points] 
    # or something similar...
    # But let's simplify and compute the linearized gradient correctly:
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + max_idx) % n_points
        grad[i] = (1 - h[j]) 
    
    # Ensure we don't make gradient inconsistent 
    # We expect minimize grad.T @ v
    # where v is the next iterate
    # Constraint: sum(v) = target_sum, and 0 <= v[i] <= 1
    
    # This is a linear program:
    # minimize sum_i grad[i] * v[i]
    # subject to sum_i v[i] = target_sum
    # 0 <= v[i] <= 1

    v = Variable(n_points)
    target_sum = n_points / 2.0
    
    # Create problem
    objective = Minimize(grad @ v)
    constraints = [sum(v) == target_sum,
                   v >= 0,
                   v <= 1]
    
    try:
        prob = Problem(objective, constraints)
        prob.solve(solver=ECOS, verbose=False)
        v_opt = np.array(v.value).flatten()
    except:
        # fallback
        # just find the simplest solution
        v_opt = np.ones(n_points)
        v_opt = v_opt * (target_sum / n_points)
        
    return v_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.full(n_points, 0.5)  # uniform initialization works well
    
    best_h = h.copy()
    best_c5 = compute_c5_value(h, dx)
    
    # Frank-Wolfe iterations
    max_iter = 10000
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Step 1: Compute gradient approximation
        # We use a linearized view: 
        # maximize sum_i h[i] * (1 - h[(i+k) % n]) over all k
        # gradient terms approximately as above in detail
        # take it simple approach: we can make a clear linearization of 
        # what we mean by gradient to find next point
        v = frank_wolfe_step(h, dx, n_points)
        
        # Line search parameter
        alpha = 0.5
        
        # Update h
        h = (1 - alpha) * h + alpha * v
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
        
        # Evaluate and store best
        current_c5 = compute_c5_value(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stop if we are very close
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
