import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighborhood(h, n_candidates=10):
    """Generate neighborhood by flipping or swapping elements."""
    n = len(h)
    candidates = []
    
    for _ in range(n_candidates):
        h_new = h.copy()
        
        # Randomly choose between flip (30%) and swap (70%)
        if np.random.rand() < 0.3:
            # Flip two random elements
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i] = np.clip(h_new[i] * 0.97 + np.random.normal(0, 0.01), 0, 1)
            h_new[j] = np.clip(h_new[j] * 0.97 + np.random.normal(0, 0.01), 0, 1)
        else:
            # Swap two random elements
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Repair constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n / 2)
        
        candidates.append(h_new)
    
    return candidates

    n_points = 70

    # Initial constraints check
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    last_improvement = 0
    temperature = 1.0
    cooling_rate = 0.999
    
    # Tabu list to store recent moves
    visited_states = set()
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate neighborhood
        candidates = generate_neighborhood(h, 10)
        
        # Evaluate all candidates
        candidate_scores = []
        for candidate in candidates:
            c5 = compute_c5_exact(candidate, dx)
            # Ensure state is not tabu (simplified check)
            state_key = tuple(np.round(candidate, decimals=6))
            if state_key not in visited_states or c5 < best_c5 - 1e-10:
                candidate_scores.append((c5, candidate))
        
        if not candidate_scores:
            # If all are tabu, just take the best
            candidate_scores = [(compute_c5_exact(c, dx), c) for c in candidates]
        
        # Find best improvement
        candidate_scores.sort()
        if not candidate_scores:
            break
            
        current_c5 = candidate_scores[0][0]
        current_h = candidate_scores[0][1]
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Update tabu list
        state_key = tuple(np.round(h, decimals=6))
        if len(tabu_list) >= tabu_tenure:
            tabu_list.pop(0)
        tabu_list.append(state_key)
        
        # Add to visited states
        visited_states.add(state_key)
        
        # Accept with some probability if not better (simulated annealing)
        if current_c5 > best_c5 and np.random.rand() > np.exp(-(current_c5 - best_c5) / temperature):
            # Keep current
            pass
        else:
            h = current_h.copy()
            
        # Update temperature
        temperature *= cooling_rate
        
        # Occasionally jitter the solution to avoid plateaus
        if np.random.rand() < 0.05:
            idx = np.random.randint(n_points)
            h[idx] = np.clip(h[idx] + np.random.normal(0, 0.01), 0, 1)
            h = normalize_h(h, target_sum)
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
