import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_coefficients(h, n_basis):
    """Compute Fourier coefficients for given h."""
    x = np.linspace(0, 2, len(h))
    coeffs = np.zeros(n_basis)
    
    for a in range(1, n_basis + 1):
        basis = np.sin(a * np.pi * x / 2) * (1 - np.abs(x - 1))
        coeffs[a-1] = np.dot(h, basis)
    
    return coeffs

def backtransform(coeffs, n_basis):
    """Backtransform from coefficients to h."""
    x = np.linspace(0, 2, len(coeffs) * 4)  # Higher resolution for smoother function
    h = np.zeros_like(x)
    
    for a in range(1, n_basis + 1):
        basis = np.sin(a * np.pi * x / 2) * (1 - np.abs(x - 1))
        h += coeffs[a-1] * basis
        
    return h[:len(coeffs)]

def project(h, target_sum):
    """Project h onto feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != target_sum:
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Validate and fix sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * (target_sum / current_sum)
            h = np.clip(h, 0, 1)
    else:
        h = triangular_ramp(n_points)
        
    h = project(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    n_basis = max(1, n_points // 4)
    coeffs = compute_coefficients(h, n_basis)
    
    # Optimization loop
    max_iter = 20000
    iter_count = 0
    lr = 0.001
    momentum = 0.9
    velocity = np.zeros_like(coeffs)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute gradients
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap_idx = np.argmax(overlaps)
        max_val = np.max(overlaps)
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute approximate gradient for coefficients using finite differences
        grad = np.zeros_like(coeffs)
        eps = 1e-4
        
        for i in range(n_basis):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = backtransform(coeffs_plus, n_basis)
            h_plus = project(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = backtransform(coeffs_minus, n_basis)
            h_minus = project(h_minus, target_sum)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / (2 * eps)
        
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        coeffs += velocity
        
        # Reconstruct h and project
        h = backtransform(coeffs, n_basis)
        h = project(h, target_sum)
        
        iter_count += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
