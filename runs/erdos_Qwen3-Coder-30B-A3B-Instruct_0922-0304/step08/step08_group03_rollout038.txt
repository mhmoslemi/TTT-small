import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    return np.minimum(x, 2 - x)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def fourier_basis(x, n_basis):
    """Create windowed sine basis functions."""
    basis = []
    for i in range(1, n_basis + 1):
        # Windowed sine
        window = 1 - np.abs(x - 1)
        window = np.clip(window, 0, 1)
        basis.append(window * np.sin(i * np.pi * x / 2))
    return np.array(basis)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)

    # Fourier representation
    x = np.linspace(0, 2, n_points)
    n_basis = min(15, n_points // 2)
    
    # Precompute basis functions
    basis = fourier_basis(x, n_basis)
    
    # Transform initial h to coefficients
    def h_to_coefficients(h_vec):
        # Normalize to fit into basis
        h_vec = project_to_feasible(h_vec, target_sum)
        coefficients = np.linalg.lstsq(basis.T, h_vec, rcond=None)[0]
        return coefficients
    
    def coefficients_to_h(coefficients):
        h = basis.T @ coefficients
        return h
    
    # Optimization using projected gradient descent
    max_iter = 20000
    lr = 0.0005
    momentum = 0.9
    velocity = np.zeros(n_basis)
    
    c5_history = []
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and derivatives
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Approximate gradient with finite differences
        grad = np.zeros(n_basis)
        eps = 1e-4
        baseline = h_to_coefficients(h)
        
        for i in range(n_basis):
            delta = np.zeros(n_basis)
            delta[i] = eps
            h_plus = coefficients_to_h(baseline + delta)
            h_minus = coefficients_to_h(baseline - delta)
            
            c5_plus = compute_c5(project_to_feasible(h_plus, target_sum), dx)
            c5_minus = compute_c5(project_to_feasible(h_minus, target_sum), dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        # Update velocity and coefficients
        velocity = momentum * velocity - lr * grad
        new_coefficients = baseline + velocity
        
        # Transform back
        new_h = coefficients_to_h(new_coefficients)
        h = project_to_feasible(new_h, target_sum)
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
