import numpy as np
from scipy.fft import fft, ifft
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to sum=target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain"""
    H_conj = np.conj(H)
    # Compute H * (1 - H_conj) elementwise
    S = np.abs(H * (1 - H_conj))
    return S

def spectral_update(H, update_mask, learning_rate=0.01):
    """Update H in frequency domain to reduce surrogate"""
    H_new = H.copy()
    # Apply gradient descent step on the real and imaginary parts
    # We modify the spectrum to reduce high-magnitude regions
    H_new[update_mask] -= learning_rate * np.sign(H_new[update_mask])
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        # Random initialization
        h_current = np.random.rand(n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    # Safe exit with sufficient time for final computation
    while time.time() - start_time < budget_s - 1:
        # Compute FFT of h
        H = fft(h_current)
        
        # Compute surrogate for frequency domain
        S = compute_surrogate(H)
        
        # Select top frequency bins to focus update
        # Focus on bins contributing most to maximum in frequency domain
        # Use relative importance for numerical stability
        S_sorted = np.sort(S)
        # Select top 20% of bins by magnitude
        threshold_idx = int(0.8 * len(S_sorted))
        threshold = S_sorted[threshold_idx]
        update_mask = S > threshold
        
        # Update in frequency domain
        H_new = spectral_update(H, update_mask, learning_rate)
        
        # Reconstruct h and project
        h_new = np.real(ifft(H_new))
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Reset learning rate on improvement
            learning_rate = 0.01
        
        h_current = h_new.copy()
        iteration += 1

        # Adaptive learning rate decay
        if iteration % 100 == 0:
            learning_rate *= 0.9
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
