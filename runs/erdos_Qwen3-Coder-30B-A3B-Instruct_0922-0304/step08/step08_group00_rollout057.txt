import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using correlate with 'full' mode to compute all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    # The correlation is computed for all lags from -n+1 to n-1
    # We only care about the positive lags from 0 to n-1
    # The maximum overlap corresponds to the highest value among positive lags
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        
    # Local refinement using sequential quadratic programming
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Add a small noise to avoid local minima
    h_perturbed = h + np.random.normal(0, 1e-5, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    current_sum = np.sum(h_perturbed)
    if current_sum > 0:
        h_perturbed = h_perturbed * (target_sum / current_sum)
    
    # Use SLSQP solver
    result = minimize(objective, h_perturbed, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 1000, 'ftol': 1e-10})
    
    final_h = np.clip(result.x, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    # Validation and return
    return (final_h.tolist(), final_c5, n_points)
