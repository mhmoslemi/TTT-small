import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate h with Gaussian noise."""
    if random.random() > mutation_rate:
        return h.copy()
    
    noise = np.random.normal(0, noise_magnitude, len(h))
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    return h_new

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select parent using tournament selection."""
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_index)
    return [population[i] for i in selected_indices]

def evaluate_population(population, dx):
    """Evaluate all individuals in population in parallel."""
    results = Parallel(n_jobs=1)(delayed(evaluate_c5)(h, dx) for h in population)
    return results

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.3
    crossover_rate = 0.8
    tournament_size = 2
    noise_magnitude = 0.05
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        parent_h = np.clip(parent_h, 0, 1)
        parent_h = normalize_h(parent_h, n_points / 2)
    else:
        parent_h = np.random.rand(n_points).astype(np.float64)
        parent_h = normalize_h(parent_h, n_points / 2)
    
    population = [parent_h.copy()]
    for _ in range(population_size - 1):
        h = mutate(parent_h, mutation_rate, noise_magnitude)
        h = normalize_h(h, n_points / 2)
        population.append(h)

    best_h = parent_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # GA iterations
    for generation in range(max_generations):
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = evaluate_population(population, dx)
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected_population = tournament_selection(population, fitnesses, tournament_size)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected_population[i]
            parent2 = selected_population[i+1] if i+1 < population_size else selected_population[0]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate, noise_magnitude)
            child2 = mutate(child2, mutation_rate, noise_magnitude)
            
            child1 = normalize_h(child1, n_points / 2)
            child2 = normalize_h(child2, n_points / 2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Decrease noise magnitude
        noise_magnitude *= 0.99
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
