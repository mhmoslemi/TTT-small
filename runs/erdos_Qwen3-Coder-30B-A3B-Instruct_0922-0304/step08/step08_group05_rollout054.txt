import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_individual(n_points, target_sum):
    """Create a valid individual with sum constraint."""
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def crossover(parent1, parent2):
    """Average-based crossover."""
    child = (parent1 + parent2) / 2.0
    return child

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Gaussian mutation with constraint repair."""
    n_points = len(individual)
    mask = np.random.random(n_points) < mutation_rate
    if np.any(mask):
        individual += np.random.normal(0, sigma, n_points) * mask
        individual = np.clip(individual, 0, 1)
        # Repair constraint
        target_sum = n_points / 2.0
        individual = normalize_h(individual, target_sum)
    return individual

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

        # Evaluate fitness
        fitnesses = []
        for individual in population:

            fitnesses.append(c5)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)

        new_population = []
        
        # Keep elite
        elite_count = population_size // 5
        elite_indices = np.argsort(fitnesses)[:elite_count]
