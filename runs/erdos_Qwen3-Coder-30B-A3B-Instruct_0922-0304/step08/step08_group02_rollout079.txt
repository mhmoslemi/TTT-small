import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def initialize_population(n_points, n_pop, target_sum, initial_h=None):
    """Initialize population for genetic algorithm."""
    population = []
    
    if initial_h is not None:
        population.append(initial_h.copy())
    
    while len(population) < n_pop:
        h = np.random.random(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    """Perform single-point crossover."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h with given rate."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
    return h

def local_search(h, target_sum, dx, steps=50):
    """Perform local gradient-like search on h."""
    h_best = h.copy()
    c5_best = compute_c5(h_best, dx)
    
    for _ in range(steps):
        if np.random.random() < 0.5:  # 50% chance to random perturb
            h_temp = h_best.copy()
            idx = np.random.randint(len(h_best))
            delta = np.random.uniform(-0.1, 0.1)
            h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = repair_h(h_temp, target_sum)
        else:  # Or simply perturb some elements
            h_temp = h_best.copy()
            num_pert = max(1, int(len(h_best) * 0.1))
            for _ in range(num_pert):
                idx = np.random.randint(len(h_best))
                delta = np.random.uniform(-0.05, 0.05)
                h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = repair_h(h_temp, target_sum)
        
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < c5_best:
            h_best = h_temp
            c5_best = c5_temp
    
    return h_best, c5_best

    best_h = None
    best_c5 = float('inf')
    
    gen = 0
    no_improvement_count = 0
    
    # Main GA loop
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness of population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Print progress
        if gen % 10 == 0:
            print(f"Generation {gen}, Best C5: {best_c5:.6f}")
            
        # Elitism: keep top 20%
        elite_count = max(1, n_pop // 5)
        elite = population[:elite_count]
        
        # Create new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            # Selection: tournament selection
            parents = []
            for _ in range(2):
                tournament_size = 3
                tournament_indices = np.random.choice(len(population), tournament_size)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(population[winner_idx])
            
            # Crossover
            if np.random.random() < 0.8:  # Crossover probability
                child1, child2 = crossover(parents[0], parents[1])
            else:
                child1, child2 = parents[0].copy(), parents[1].copy()
            
            # Mutation
            child1 = mutate(child1, 0.05)
            child2 = mutate(child2, 0.05)
            
            # Repair
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < n_pop:
                new_population.append(child2)
        
        population = new_population
        
        gen += 1
    
    # Final local refinement of best solution
    if best_h is not None:
        best_h_refined, refined_c5 = local_search(best_h, target_sum, dx, 50)
        if refined_c5 < best_c5:
            best_h = best_h_refined
            best_c5 = refined_c5
    
    return (best_h.tolist(), best_c5, n_points)
