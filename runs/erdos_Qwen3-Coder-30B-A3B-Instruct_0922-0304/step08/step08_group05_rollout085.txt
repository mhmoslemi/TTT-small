import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Find all lags that achieve maximum overlap
    max_val = overlap[max_idx]
    active_lags = []
    for k in range(len(overlap)):
        if abs(overlap[k] - max_val) < 1e-10:
            active_lags.append(k)
    
    subgrad = np.zeros(n)
    for lag in active_lags:
        # For the case where we have h[i] * (1 - h[i+lag]) or similar
        # the subgradient is effectively (1 - h[i+lag]) - h[i+lag] = 1 - 2*h[i+lag]
        for i in range(n):
            j = i + lag - (n-1)  # index in original h vector
            if 0 <= j < n:
                # This is a simplified approximation of the subgradient
                subgrad[i] += 1 - 2 * h[j]
    
    # Normalize subgradient to unit norm
    norm = np.linalg.norm(subgrad)
    if norm > 0:
        subgrad = subgrad / norm
    
    # Project out the component orthogonal to the constraint
    # Gradient of sum constraint is all 1's
    grad_sum = np.ones(n) / target_sum
    proj = np.dot(subgrad, grad_sum)
    subgrad = subgrad - proj * grad_sum
    
    return subgrad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h * (target_sum / np.sum(h))  # Normalize to target sum
    h = np.clip(h, 0, 1)
    
    # Adaptive learning rate and momentum parameters
    learning_rate = 0.01
    beta = 0.9
    momentum = np.zeros(n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement_count = 0
    iteration = 0
    
    while (time.time() - start_time) < budget_s - 1 and no_improvement_count < 500:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        subgrad = compute_subgradient(h, dx, target_sum)
        
        # Update momentum and parameters
        momentum = beta * momentum + (1 - beta) * subgrad
        h = h - learning_rate * momentum
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))  # Ensure sum constraint
        
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.995  # Slowly decrease learning rate
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
