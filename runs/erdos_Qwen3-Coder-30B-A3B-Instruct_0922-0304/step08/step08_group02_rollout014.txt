import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Initialize with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    n_steps = 1000
    no_improvement_count = 0
    no_improvement_limit = 500
    cooling_timescale = 1000
    
    # Initialize
    current_c5 = compute_c5(h_current, dx)
    best_h = h_current.copy()
    best_c5 = current_c5
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Proposal step
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, sigma)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Compute acceptance probability
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0:
            accept_prob = 1.0
        else:
            accept_prob = np.exp(-delta_c5 / T) if T > 0 else 0.0
            
        # Accept or reject
        if np.random.random() < accept_prob:
            h_current = h_new
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update temperature and perturbation
        iteration += 1
        T = max(T_min, T0 * (alpha ** (iteration / cooling_timescale)))
        
        # Adaptive perturbation scaling
        if no_improvement_count > no_improvement_limit:
            sigma *= 0.9
            no_improvement_count = 0
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
