import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def constraint_repair(h, target_sum):
    """Repair constraints: clip and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def generate_neighborhood(h, target_sum, n_candidates=10):
    """Generate neighborhood by flipping or swapping elements."""
    candidates = []
    n = len(h)
    for _ in range(n_candidates):
        h_new = h.copy()
        if random.random() < 0.3:
            # Flip two random elements
            i, j = random.sample(range(n), 2)
            h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
            h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
        else:
            # Swap two random elements
            i, j = random.sample(range(n), 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        h_new = constraint_repair(h_new, target_sum)
        candidates.append(h_new)
    
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = constraint_repair(h, target_sum)
    h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    last_improvement = 0
    temperature = 1.0
    cooling_rate = 0.999
    
    # Main search loop
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate neighborhood
        candidates = generate_neighborhood(h, target_sum)
        
        # Evaluate all candidates
        candidate_c5s = [compute_c5_exact(c, dx) for c in candidates]
        
        # Find best non-tabu candidate
        best_candidate_idx = None
        for i, c5 in enumerate(candidate_c5s):
            candidate_h = candidates[i]
            candidate_tuple = tuple(candidate_h)
            if candidate_tuple not in tabu_list or c5 < best_c5:
                if best_candidate_idx is None or c5 < candidate_c5s[best_candidate_idx]:
                    best_candidate_idx = i
        
        if best_candidate_idx is not None:
            new_h = candidates[best_candidate_idx]
            new_c5 = candidate_c5s[best_candidate_idx]
            
            # Update tabu list
            tabu_list.append(tuple(h))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            h = new_h
            
            # Update best solution
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
            # Accept worse solutions with probability based on temperature
            if new_c5 > best_c5:
                delta = new_c5 - best_c5
                if np.random.random() < np.exp(-delta / temperature):
                    h = new_h
                else:
                    last_improvement += 1
                    
        else:
            # No valid candidate found, may happen if all tabu
            last_improvement += 1
            
        # Update temperature
        temperature *= cooling_rate
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
