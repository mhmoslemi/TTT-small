import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    
    # Find all lags that achieve maximum overlap
    active_lags = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each active lag k, compute contributions to gradient
    for k in active_lags:
        # For positions i and j where 0 <= i < n and 0 <= j < n:
        # overlap[k] = sum_{i+j = k} h[i] * (1 - h[j])
        # Gradient w.r.t. h[i] = -sum_{j: i+j=k} (1 - h[j])  if k >= 0, otherwise
        # Actually, let's recomputing correctly:
        # overlap[k] = sum_{i} h[i] * (1 - h[i+k]) where i+k is within bounds
        
        if k < n:  # Case 1: left-side overlapping
            for i in range(n):
                j = i + k
                if 0 <= j < n:
                    grad[i] += -(1 - h[j])
        else:  # Case 2: right-side overlapping
            for i in range(n):
                j = i + k - n
                if 0 <= j < n:
                    grad[i] += -(1 - h[j])
    
    # Normalize gradient for numerical stability 
    if np.linalg.norm(grad) > 0:
        grad = grad / np.linalg.norm(grad) * 0.01
    
    # Project onto feasible direction to respect sum constraint
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
        
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum_factor = 0.9
    velocity = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum and learning rate
        velocity = momentum_factor * velocity + grad * learning_rate
        h = h - velocity
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 50 == 0 and iteration > 0:
            learning_rate *= 0.995
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
