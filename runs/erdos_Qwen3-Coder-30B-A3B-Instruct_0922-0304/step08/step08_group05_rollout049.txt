import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a piecewise constant construction
        h = np.zeros(n_points)
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        h[mid1:mid2] = 1.0
        # Add some noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = compute_c5_exact(h, dx)
    best_h = h.copy()
    last_improvement = 0
    
    # Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    perturbation_strength = 0.05
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Simulated Annealing Step
        # Create a new solution by perturbing current
        candidate = h.copy()
        
        # Randomly perturb 10% of the values
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            # With 50% chance flip, 50% chance change value
            if np.random.rand() < 0.5:
                candidate[i] = 1 - candidate[i]  # Flip
            else:
                # Small random change
                candidate[i] += np.random.normal(0, perturbation_strength)
        
        # Ensure bounds and normalization
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5_exact(candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        if candidate_c5 < best_c5 or np.random.rand() < np.exp(-(candidate_c5 - best_c5) / temp):
            h = candidate
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        # Reduce temperature
        temp = max(min_temp, temp * cooling_rate)
        
        # Cutting plane refinement every 100 iterations
        if iteration % 100 == 0 and iteration > 0:
            # Find the lag that maximizes overlap
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            
            # Apply small adjustments to reduce overlap at max lag
            for i in range(n_points):
                if 0 <= i + max_idx < len(overlap) and overlap[max_idx] > 0.001:
                    # Very crude adjustment
                    adjustment = min(0.02, overlap[max_idx] * 0.1)
                    if i + max_idx < n_points:
                        # Try reducing h[i] and increasing h[i + max_idx] if valid
                        if h[i] >= adjustment:
                            h[i] -= adjustment
                            h[i + max_idx] = min(1.0, h[i + max_idx] + adjustment)
            
            # Normalize again
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Update best if improved
            current_c5 = compute_c5_exact(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final check
    if compute_c5_exact(best_h, dx) < best_c5:
        best_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
