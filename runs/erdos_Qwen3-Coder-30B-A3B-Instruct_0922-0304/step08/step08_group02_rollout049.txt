```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of shifts for QP formulation"""
    # Sample shifts from -n+1 to n-1 (full correlation support)
    shifts = np.random.randint(-n+1, n, sample_size)
    return np.unique(shifts)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = np.max(correlate(h_current, 1 - h_current, mode="full")) * dx
    
    # Iteration parameters
    max_iterations = 1200
    iteration = 0
    stagnation_count = 0
    stagnation_threshold = 200
    target_overlap = 0.38085  # Start with known bound
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Sample shifts for QP
        shifts = sample_shifts(n_points, min(50, n_points))
        
        # Create variables for optimization
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize max overlap (surrogate)
        max_overlap = cp.Variable()
        
        # Constraints: for each sampled shift, overlap <= max_overlap
        constraints = []
        for k in shifts:
            # Compute the overlap constraint coefficients
            overlap_coeff = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    overlap_coeff[i] = 1.0
            
            # Skip if all zeros in overlap (edge case)
            if np.sum(np.abs(overlap_coeff)) == 0:
                continue
                
            # Form constraint: sum of coefficient * h_vars <= max_overlap
            # And we also consider the 1-h terms
            # Correlation calculation: int h(x) (1-h(x+k)) dx
            # Discrete version: sum h[i] * (1 - h[i+k]) 
            # This is a quadratic constraint
            # Let's treat it as a hard constraint limitation
            
            # We'll compute the minimum required overlap for this shift
            # and add it as a constraint on max_overlap
            
            # Simple approach: make inequalities that help minimize the peaks
            # For simplification, we'll solve using dual gradient approach but with 
            # a more refined feasibility checking
            
        # Instead of quadratic optimization, we'll do approximate gradient approach
        # sample many more shifts but solve a relaxed problem with linearization
        
        # Instead, build a simpler efficient approach:
        
        # Simple perturbation-based local search for better results
        h_new = h_current.copy()
        # Perturb solution
        for i in range(n_points // 5):
            idx = np.random.randint(n_points)
            delta = np.random.uniform(-0.05, 0.05)
            h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Renormalize
        h_new = project_h(h_new, n_points)
        
        # Compute new C5
        overlap = correlate(h_new, 1 - h_new, mode="full")
        new_c5 = np.max(overlap) * dx
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        h_current = h_new.copy()
        iteration += 1
        
        # Early termination if improvement is good enough
        if best_c5 <= 0.3807:
            break
            
        # Adaptive target adjustment based on performance
        if iteration % 50 == 0 and stagnation_count > 10:
            target_overlap *= 0.99
            
        # Occasionally reinitialize with a randomly perturbed version of best
        if np.random.rand() < 0.05 and iteration > 300:
            h_temp = best_h.copy()
            for i in range(n_points // 10):
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.03, 0.03)
                h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = project_h(h_temp, n_points)
            temp_overlap = correlate(h_temp, 1 - h_temp, mode="full") * dx
            temp_c5 = np.max(temp_overlap)
            if temp_c5 < best_c5:
                best_h = h_temp.copy()
                best_c5 = temp_c5
                
    # Final refinement with gradient descent type approach
    h_final = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_final.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = np.max(correlate(h_temp, 1 - h_temp, mode="full")) * dx
        if c5_temp < np.max(correlate(h_final, 1 - h_final, mode="full")) * dx:
            h_final = h_temp.copy()
    
    final_c5 = np.max(correlate(h_final, 1 - h_final, mode="full")) * dx
    return (h_final.tolist(), final_c5, n_points)
```
