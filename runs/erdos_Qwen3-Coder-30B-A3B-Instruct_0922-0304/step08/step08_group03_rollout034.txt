import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    x = np.arange(n)
    return np.minimum(x + 1, 2 * n - x - 1) / n

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Phase 1: Gradient descent with momentum
    velocity = np.zeros(n_points)
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    
    max_iter_phase1 = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iter_count < max_iter_phase1 and time.time() - start_time < budget_s - 1:
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Find where the max overlap occurs
        max_idx = np.argmax(overlap)
        # The overlap involves shifting h by k positions
        k = max_idx - (n_points - 1)  # k ranges from -(n_points-1) to (n_points-1)
        
        # Compute approximate gradient for each position
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.001)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.001)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / 0.002
        
        # Apply momentum update
        velocity = beta * velocity + (1 - beta) * grad
        h_new = h - alpha * velocity
        h_new = clip_and_renormalize(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < overlap_max:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            alpha *= gamma  # decay learning rate
            no_improvement_count = 0
        else:
            h = clip_and_renormalize(h, target_sum)
            alpha *= gamma  # decay learning rate
            no_improvement_count += 1
            
        iter_count += 1
        
        if no_improvement_count > max_no_improvement:
            break
    
    # Phase 2: Coordinate-wise search
    if best_c5 > 0.3807 and time.time() - start_time < budget_s - 1:
        max_iter_phase2 = 10000
        iter_count = 0
        no_improvement_count = 0
        max_no_improvement = 2000
        update_percent = 20
        
        while iter_count < max_iter_phase2 and time.time() - start_time < budget_s - 1:
            overlap = correlate(h, 1 - h, mode="full") * dx
            overlap_max = np.max(overlap)
            
            if overlap_max < best_c5:
                best_c5 = overlap_max
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Select random coordinates to modify
            update_count = max(1, int(n_points * update_percent / 100))
            update_indices = np.random.choice(n_points, size=update_count, replace=False)
            
            improved = False
            for i in update_indices:
                current_val = h[i]
                # Try different step sizes
                steps = [0.001, 0.005, 0.01, 0.02, 0.05]
                best_step = 0.0
                best_new_c5 = overlap_max
                
                for step in steps:
                    # Try increasing
                    h_try = h.copy()
                    h_try[i] = min(1.0, current_val + step)
                    h_try = clip_and_renormalize(h_try, target_sum)
                    new_c5 = compute_c5(h_try, dx)
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_step = step
                    
                    # Try decreasing
                    h_try = h.copy()
                    h_try[i] = max(0.0, current_val - step)
                    h_try = clip_and_renormalize(h_try, target_sum)
                    new_c5 = compute_c5(h_try, dx)
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_step = -step
                
                # Apply the best change
                if best_step != 0.0:
                    h[i] = np.clip(current_val + best_step, 0, 1)
                    h = clip_and_renormalize(h, target_sum)
                    improved = True
            
            new_c5 = compute_c5(h, dx)
            
            if improved and new_c5 < overlap_max:
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            iter_count += 1
            
            if no_improvement_count > max_no_improvement:
                break
    
    return (best_h.tolist(), best_c5, n_points)
