import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)

        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try convex optimization approach
    try:
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        # Objective: minimize max overlap
        # We will use a trick: introduce a variable t such that all overlaps <= t
        # We'll use a more direct approach with iterative refinement
        
        # Initialize with a good starting point
        current_h = h.copy()
        current_c5 = best_c5
        
        # Try local optimization with SLSQP
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try multiple local optimization attempts with different initializations
        max_local_attempts = 10
        for attempt in range(max_local_attempts):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Random perturbation of current solution
            h_perturbed = current_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum > 0:
                h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)
            
            # Local optimization
            result = minimize(objective, h_perturbed, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-8})

                refined_h = np.clip(result.x, 0, 1)

                refined_c5 = evaluate_overlap(refined_h, dx)
                if refined_c5 < current_c5:
                    current_h = refined_h
                    current_c5 = refined_c5
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        # Early termination if it's a good enough solution
                        if best_c5 < 0.3807:
                            break

                # Generate a random function satisfying constraints
                h_random = np.random.uniform(0, 1, n_points)

                    h_random = h_random * (target_sum / h_random_sum)

        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
