import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # This computes the full convolution of h with (1-h) and returns the maximum
    n = len(h)
    # For each shift k, compute sum_{i} h[i] * (1 - h[i+k])
    # where indices wrap around (circular convolution)
    # We use the fact that h[i+k] is just a shifted version of h
    # So we compute convolution with 1-h and take max
    h1 = h.copy()
    h2 = 1 - h.copy()
    crosscorr = np.correlate(h1, h2, mode='full')
    # The result has length 2*n-1; the maximum corresponds to the non-negative shifts
    # since each shift k corresponds to index k of the full convolution
    max_shift = len(crosscorr) - n
    max_val = np.max(crosscorr[n-1:n-1+max_shift])
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization parameters
    # Perturbation strength
    delta = 1e-3
    decay_factor = 0.98
    improvement_threshold = 1e-5
    max_iterations = 100
    perturbation_frequency = 5
    
    # Track iterations and time
    iteration = 0
    improved = True
    
    # Store current h for solving subproblems
    current_h = h.copy()
    
    while time.time() - start_time < budget_s - 10 and improved and iteration < max_iterations:
        # Partition into even and odd indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        even_vars = current_h[even_indices]
        odd_vars = current_h[odd_indices]
        
        def solve_subproblem(fixed_vars, fixed_indices, free_vars, free_indices, is_even):
            # Objective function: max correlation for current fixed solution
            def obj(free_vals):
                # Reconstruct full vector
                temp_h = current_h.copy()
                temp_h[free_indices] = free_vals
                
                # Ensure full vector satisfies norm constraint
                sum_fixed = np.sum(temp_h[fixed_indices])
                required_free_sum = target_sum - sum_fixed
                
                if np.sum(free_vals) != required_free_sum:
                    # Adjust free variables to meet constraint
                    adjust = (required_free_sum - np.sum(free_vals)) / len(free_vals)
                    temp_h[free_indices] += adjust
                    
                return evaluate_overlap(temp_h, dx)
            
            # Constraints: all free variables in [0,1]
            bounds = [(0, 1) for _ in range(len(free_vars))]
            
            # Initial guess for free variables
            x0 = free_vars.copy()
            
            try:
                # Solve using SLSQP
                result = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                                options={'ftol': 1e-10, 'maxiter': 100})
                if result.success:
                    new_free_vars = np.clip(result.x, 0, 1)
                    # Make sure the sum is correct
                    sum_fixed = np.sum(current_h[fixed_indices])  
                    required_free_sum = target_sum - sum_fixed
                    free_to_add = required_free_sum - np.sum(new_free_vars)
                    if abs(free_to_add) > 1e-8:
                        adjust = free_to_add / len(new_free_vars)
                        new_free_vars += adjust
                        
                    return new_free_vars
            except:
                pass
            
            return free_vars
            
        # Solve for even vars while fixing odd vars
        new_even = solve_subproblem(odd_vars, odd_indices, even_vars, even_indices, True)
        
        # Construct new current_h
        temp_h = current_h.copy()
        temp_h[even_indices] = new_even
        temp_h[odd_indices] = odd_vars
        
        # Solve for odd vars while fixing new even vars
        new_odd = solve_subproblem(new_even, even_indices, odd_vars, odd_indices, False)
        
        # Construct new current_h again
        temp_h = current_h.copy()
        temp_h[even_indices] = new_even
        temp_h[odd_indices] = new_odd
        
        # Enforce sum constraint again
        total_sum = np.sum(temp_h)
        if total_sum > 0:
            temp_h = temp_h * (target_sum / total_sum)
        temp_h = np.clip(temp_h, 0, 1)
        
        # Update current_h
        current_h = temp_h.copy()
        
        # Evaluate new C5
        new_c5 = evaluate_overlap(current_h, dx)
        
        # Check for improvement
        improved = new_c5 < best_c5 - improvement_threshold
        if improved:
            best_c5 = new_c5
            best_h = current_h.copy()
        
        # Occasionally perturb to escape local minima
        if iteration % perturbation_frequency == 0:
            perturbed_h = current_h + np.random.uniform(-delta, delta, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            sum_pert = np.sum(perturbed_h)
            if sum_pert > 0:
                perturbed_h = perturbed_h * (target_sum / sum_pert)
            delta *= decay_factor # decay the perturbation
            current_h = perturbed_h.copy()
        
        iteration += 1
        
    # Final refinement with local optimizer
    def final_objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Constraint: sum equals target sum
    def sum_constraint(h_vector):
        return np.sum(h_vector) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Use current best as initial point for refinement
    result = minimize(final_objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Validate final solution
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
