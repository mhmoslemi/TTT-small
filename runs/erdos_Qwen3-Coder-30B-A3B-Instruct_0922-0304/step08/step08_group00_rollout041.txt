import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient descent with Nesterov momentum
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    patience = 500
    
    # For tracking improvement
    prev_c5 = best_c5 + 1.0
    no_improvement = 0
    
    # Nesterov momentum variables
    v = np.zeros(n_points)
    y = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Update h with Nesterov momentum
        h = y - step_size * v
        
        # Project h to feasible region [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        # Compute overlap and gradient
        current_c5 = evaluate_overlap(h, dx)
        
        # Check for improvement
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Adapt step size based on improvement
        if abs(current_c5 - prev_c5) < tol:
            step_size *= 0.9
        else:
            step_size = min(step_size * 1.1, 1.0)
            
        step_size = max(step_size, min_step_size)
        
        # Check for convergence
        if no_improvement > patience:
            break
            
        prev_c5 = current_c5
        
        # Compute gradient (simplified version)
        # We'll approximate dC_k/dh_i by computing the difference in overlap
        # when we perturb h_i slightly. This is a rough approximation but will
        # suffice for the momentum update.
        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += 1e-8
            h_pert = np.clip(h_pert, 0, 1)
            current_sum_pert = np.sum(h_pert)
            if current_sum_pert > 0:
                h_pert = h_pert * (target_sum / current_sum_pert)
            c5_pert = evaluate_overlap(h_pert, dx)
            gradient[i] = (c5_pert - current_c5) / 1e-8
        
        # Update momentum
        v = momentum * v + gradient
        y = h - step_size * v
        
        # Ensure projection still satisfies constraints
        y = np.clip(y, 0, 1)
        current_sum_y = np.sum(y)
        if current_sum_y > 0:
            y = y * (target_sum / current_sum_y)
    
    # Final local refinement using scipy
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
