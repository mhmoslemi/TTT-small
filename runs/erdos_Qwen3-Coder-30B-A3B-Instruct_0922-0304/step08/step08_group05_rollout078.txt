import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2):
    """Create offspring by averaging parents."""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Apply Gaussian mutation to individual."""
    noise = np.random.normal(0, sigma, len(individual))
    mutated = individual + noise * mutation_rate
    # Clip to feasible range
    mutated = np.clip(mutated, 0, 1)
    return mutated

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    population.append(h.copy())
    
    # Fill population with random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Evolution parameters
    num_generations = 1000
    elite_size = 5
    mutation_rate = 0.1
    sigma = 0.1
    
    best_c5 = float('inf')
    best_h = None
    no_improvement_count = 0
    
    # Evolution loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5_exact(ind, dx) for ind in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count >= 1000:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutate(offspring, mutation_rate, sigma)
            
            # Normalize to meet constraint
            offspring = normalize_h(offspring, target_sum)
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
            
        population = new_population[:population_size]
        
        # Adjust mutation rate over time
        if generation % 100 == 0 and generation > 0:
            mutation_rate *= 0.95
    
    # Final evaluation
    if best_h is not None:
        final_c5 = compute_c5_exact(best_h, dx)
    else:
        # Fallback if nothing was found
        final_h = population[0] if population else np.zeros(n_points)
        final_c5 = compute_c5_exact(final_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
