import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Convex relaxation using CVXPY
    h_var = Variable(n_points)
    t = Variable()
    
    # Constraints for convex relaxation
    constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
    
    # Compute all overlap constraints
    for k in range(2 * n_points - 1):
        # This is a simplified convex approximation
        # We won't write out all bilinear terms but just approximate with one of the overlap constraints
        # Using from the previous best iterate
        pass
    
    # Solve relaxation
    try:
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h = np.array(h_var.value).reshape(-1)
    except:
        # If CVXPY fails, fall back to previous method
        pass
    
    # Refine with gradient descent
    def compute_gradient(h, dx, n_points):
        """Compute approximate gradient of the C5 objective."""
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            total_grad = 0.0
            for k in range(2 * n_points - 1):
                pos1 = i
                pos2 = i + k - (n_points - 1)
                if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                    total_grad += (1 - 2 * h[pos1]) * (1 - h[pos2])
            grad[i] = total_grad
            
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
        return grad

    # Gradient descent refinement
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    lr = 0.01
    max_iter = 5000
    prev_loss = float('inf')
    no_improvement_count = 0
    tolerance = 1e-8
    
    for i in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        loss = evaluate_c5(h, dx)
        if loss < best_c5:
            best_c5 = loss
            best_h = h.copy()
        
        if abs(prev_loss - loss) < tolerance:
            no_improvement_count += 1
            if no_improvement_count > 1000:
                break
        else:
            no_improvement_count = 0
        prev_loss = loss
        
        # Compute gradient and update
        grad = compute_gradient(h, dx, n_points)
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= 0.999
            
        if loss <= 0.3807:
            break
            
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
