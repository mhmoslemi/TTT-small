import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_r, target_sum):
    """Round relaxed solution to binary with target sum."""
    # Compute median threshold
    tau = np.median(h_r)
    h_b = (h_r > tau).astype(float)
    
    # Adjust to match exact sum
    current_sum = np.sum(h_b)
    diff = int(current_sum - target_sum)
    
    if diff > 0:
        # Need to flip some 1s to 0s
        idxs = np.argsort(h_r)[::-1]  # Sort by descending value
        for i in range(diff):
            h_b[idxs[i]] = 0
    elif diff < 0:
        # Need to flip some 0s to 1s
        idxs = np.argsort(h_r)  # Sort by ascending value
        for i in range(-diff):
            h_b[idxs[i]] = 1
    
    return h_b

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

    h = normalize_h(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    improvement_count = 0
    max_no_improvement = 10
    
    # Relaxation-based approach
    for iteration in range(50):
        if time.time() - start_time > budget_s - 1:
            break

        # Relaxation stage - solve a relaxed version where we find a feasible h
        # that minimizes an upper bound of the overlap
        h_r = h.copy()
        
        # Use a simple gradient-like idea to improve the relaxation
        # Perform few iterations of local change
        for _ in range(20):
            current_c5 = compute_c5(h_r, dx)
            gradients = np.zeros(n_points)
            for i in range(n_points):
                grad = 0
                for k in range(n_points):
                    if i + k < n_points:
                        grad += (1 - 2 * h_r[i]) * (1 - h_r[i + k])
                    if i - k >= 0:
                        grad += (1 - 2 * h_r[i]) * (1 - h_r[i - k])
                gradients[i] = grad
            
            # Use simple update step based on gradient
            step_size = 0.02
            h_r_new = h_r + step_size * (1 - 2 * h_r) * gradients
            h_r_new = np.clip(h_r_new, 0, 1)
            h_r_new = normalize_h(h_r_new, target_sum)
            h_r = h_r_new
            
        # Rounding step
        h_b = round_to_binary(h_r, target_sum)
        h_b = normalize_h(h_b, target_sum)
        
        # Evaluate the new rounded solution
        new_c5 = compute_c5(h_b, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_b.copy()
            improvement_count = 0
        else:
            improvement_count += 1
        
        # Early stopping
        if improvement_count >= max_no_improvement:
            break
        
        # Update h for next iteration
        h = h_b.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
