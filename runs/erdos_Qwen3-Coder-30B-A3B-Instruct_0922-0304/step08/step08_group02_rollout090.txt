import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute frequency-domain surrogate objective."""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, learning_rate=0.01):
    """Update H in frequency domain to reduce surrogate objective."""
    S = compute_surrogate(H)
    # Identify high contribution frequencies
    # Simple greedy: reduce magnitude of largest contributions
    # Apply gradient descent in frequency domain
    grad = H * (1 - 2 * np.real(H))  # Approximate gradient
    H_new = H - learning_rate * grad
    return H_new

    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_c5 = compute_c5(h_current, dx)

    while time.time() - start_time < budget_s - 1:
        # Time-based learning rate decay
        if iteration > 0 and iteration % 100 == 0:
            learning_rate *= 0.5
            
        # Fourier transform of h
        H = fft(h_current)
        
        # Update in frequency domain
        H_new = spectral_update(H, learning_rate)
        
        # Inverse transform & project
        h_new = np.real(ifft(H_new))
        h_new = project_h(h_new, target_sum)
        
        # Constrain to same sum
        h_new = h_new * (target_sum / np.sum(h_new))
