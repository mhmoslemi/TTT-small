import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # If no initial construction, start with a random one
    if best_h is None:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Local search with different perturbations
    max_iter = 10000
    iter_count = 0
    
    # Define search space
    search_space = np.linspace(0.05, 0.95, 19)  # 19 values between 0.05 and 0.95
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        # Create a new candidate through small perturbations
        h_new = best_h.copy()
        
        # Select random number of indices to modify
        n_changes = np.random.randint(1, min(10, n_points//2))
        indices = np.random.choice(n_points, n_changes, replace=False)
        
        # Modify values in those indices
        for i in indices:
            # Randomly shift value
            shift = np.random.uniform(-0.1, 0.1)
            h_new[i] = h_new[i] + shift
            
        # Ensure values stay in [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy integral constraint
        h_new = normalize_h(h_new, n_points / 2)
        
        # Evaluate new candidate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept new solution if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
            # Early stopping if we reach target
            if best_c5 <= 0.3807:
                break
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback to a simple construction
        best_h = np.ones(n_points) * 0.5
        best_h[0] = 0.7
        best_h[1] = 0.3
        best_h = normalize_h(best_h, n_points / 2)
        best_c5 = evaluate_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
