import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple symmetric ramp
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, h[::-1]])
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Early termination check
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Optimizer parameters
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    step_size = 0.01
    
    # Gradient computation helper
    def compute_gradients(h, dx):
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Perturb h[i]
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            
            overlap_plus = evaluate_overlap(h_plus, dx)
            overlap_minus = evaluate_overlap(h_minus, dx)
            
            gradients[i] = (overlap_plus - overlap_minus) / 0.02
        return gradients
    
    # Optimization loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute gradients
        gradients = compute_gradients(h, dx)
        
        # Update h in direction of negative gradient
        h_new = h - step_size * gradients
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Update current solution
        h = h_new.copy()
        
        iter_count += 1
        
        # Early stopping
        if new_c5 >= best_c5:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
        else:
            no_improvement_count = 0
    
    # Final check
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
