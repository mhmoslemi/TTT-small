import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization parameters
    max_iterations = 100
    improvement_threshold = 1e-5
    perturbation_delta = 1e-3
    perturbation_decay = 0.98
    
    # Main alternating optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Periodically perturb the solution to escape local minima
        if iteration % 10 == 0 and iteration > 0:
            perturb = np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
            h = np.clip(h + perturb, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 0:
                h = h * (target_sum / current_sum)
        
        # Partition variables into even and odd indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Objective function for optimizing even indices with fixed odd
        def objective_even(even_vals):
            h_copy = h.copy()
            h_copy[even_indices] = even_vals
            # Ensure sum constraint
            current_sum = np.sum(h_copy)
            if current_sum > 0:
                h_copy = h_copy * (target_sum / current_sum)
            return evaluate_overlap(h_copy, dx)
            
        # Objective function for optimizing odd indices with fixed even
        def objective_odd(odd_vals):
            h_copy = h.copy()
            h_copy[odd_indices] = odd_vals
            # Ensure sum constraint
            current_sum = np.sum(h_copy)
            if current_sum > 0:
                h_copy = h_copy * (target_sum / current_sum)
            return evaluate_overlap(h_copy, dx)
        
        # Constraint for even indices
        def constraint_even():
            h_copy = h.copy()
            h_copy[even_indices] = h[even_indices]
            return np.sum(h_copy) - target_sum
            
        # Constraint for odd indices
        def constraint_odd():
            h_copy = h.copy()
            h_copy[odd_indices] = h[odd_indices]
            return np.sum(h_copy) - target_sum
            
        # Optimize even indices with fixed odd
        bounds_even = [(0, 1) for _ in even_indices]
        initial_even = h[even_indices].copy()
        
        try:
            result_even = minimize(objective_even, initial_even, method='SLSQP', 
                                 bounds=bounds_even, options={'maxiter': 50, 'ftol': 1e-8})
            if result_even.success:
                h[even_indices] = np.clip(result_even.x, 0, 1)
        except:
            pass
            
        # Optimize odd indices with fixed even
        bounds_odd = [(0, 1) for _ in odd_indices]
        initial_odd = h[odd_indices].copy()
        
        try:
            result_odd = minimize(objective_odd, initial_odd, method='SLSQP', 
                                bounds=bounds_odd, options={'maxiter': 50, 'ftol': 1e-8})
            if result_odd.success:
                h[odd_indices] = np.clip(result_odd.x, 0, 1)
        except:
            pass
            
        # Re-normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Evaluate new solution
        new_c5 = evaluate_overlap(h, dx)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Check improvement
        if abs(new_c5 - best_c5) < improvement_threshold:
            break

        # Decrease perturbation level
        perturbation_delta *= perturbation_decay
    
    # Final local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-10})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
