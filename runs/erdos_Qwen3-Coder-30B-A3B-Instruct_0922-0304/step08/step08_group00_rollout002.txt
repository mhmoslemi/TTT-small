from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)

    h_var = cp.Variable(n_points)

    # Minimize the maximum overlap by using a slightly different strategy
    
    # Start with the current best solution
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Direct local optimization with bounds
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # We use SLSQP for refinement with multiple starting points
    best_solution = best_h.copy()

            if result.success:

                if refined_sum > 0:

                    best_solution = refined_h.copy()
