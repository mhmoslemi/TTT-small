import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create a simple initial configuration
        h_current = np.ones(n_points) * 0.5
        
    # MCMC settings
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    tau = 1000
    n_steps_no_improve = 0
    max_no_improve = 1000
    eta = 0.9
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        current_c5 = compute_c5(h_current, dx)
        
        # Propose a new state
        h_proposed = h_current.copy()
        i = np.random.randint(n_points)
        delta = np.random.normal(0, sigma)
        h_proposed[i] = np.clip(h_proposed[i] + delta, 0, 1)
        
        # Maintain sum constraint
        scale_factor = target_sum / np.sum(h_proposed)
        h_proposed = h_proposed * scale_factor
        
        # Compute new overlap
        proposed_c5 = compute_c5(h_proposed, dx)
        
        # Compute change in objective
        delta_c5 = proposed_c5 - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.random() < exp(-delta_c5 / T):
            h_current = h_proposed
            current_c5 = proposed_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            n_steps_no_improve = 0
        else:
            n_steps_no_improve += 1
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (iteration / tau)))
        
        # Adapt perturbation scale
        if n_steps_no_improve > max_no_improve:
            sigma *= eta
            n_steps_no_improve = 0
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break

    # Final refinements
    final_h = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = final_h.copy()
        i = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[i] = np.clip(h_temp[i] + delta, 0, 1)
        scale_factor = target_sum / np.sum(h_temp)
        h_temp = h_temp * scale_factor
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(final_h, dx):
            final_h = h_temp.copy()
            
    final_c5 = compute_c5(final_h, dx)
    return (final_h.tolist(), final_c5, n_points)
