import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def gradient_descent_refinement(h, target_sum, dx, max_iter=10):
    """Perform gradient descent refinement on h."""
    for _ in range(max_iter):
        # Compute the overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_val = np.max(overlap)
        
        # Compute gradient for each point
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            # Perturb h[i] slightly
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_plus = repair_h(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            h_minus = repair_h(h_minus, target_sum)
            
            # Compute approximate gradient
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            gradient[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / 0.02
            
        # Update h in direction that reduces overlap
        step_size = 0.01
        h_new = h - step_size * gradient
        h_new = repair_h(h_new, target_sum)
        
        # Accept if it improves
        new_overlap = correlate(h_new, 1 - h_new, mode="full") * dx
        if np.max(new_overlap) < max_val:
            h = h_new
        else:
            break
    return h

    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:

    # Population-based optimization with GA-style operations
    population_size = 50
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_new = np.random.rand(n_points)
        h_new = repair_h(h_new, target_sum)
        population.append(h_new)
    
    mutation_rate = 0.1
    crossover_prob = 0.7
    no_improvement_count = 0
    max_no_improvement = 200
    
    generation = 0
    max_generations = 2000
    
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        # Evaluate the whole population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        
        # Select best individuals for reproduction
        idxs = np.argsort(fitness)
        selected = [population[i] for i in idxs[:population_size//2]]
        
        # Crossover and mutation to create new population
        next_generation = selected.copy()
        
        while len(next_generation) < population_size:

                parent1, parent2 = np.random.choice(selected, 2, replace=False)
