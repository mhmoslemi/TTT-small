def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def adjust_to_binary(h_r, n_points, target_sum):
    """Convert relaxed solution to binary-like with sum constraint."""
    # Median threshold rounding
    threshold = np.median(h_r)
    h_b = (h_r > threshold).astype(float)
    
    # Adjust to exact sum
    current_sum = np.sum(h_b)
    diff = current_sum - target_sum
    if diff > 0:
        # Need to turn some 1s to 0s
        diff = int(diff)
        # Find indices where h_r is highest among 1s and flip them to 0
        ones_idx = np.where(h_b == 1)[0]
        # Sort by original value descending to keep lowest values as 1
        sort_idx = np.argsort(h_r[ones_idx])[::-1]
        to_flip = ones_idx[sort_idx[:diff]]
        h_b[to_flip] = 0
    elif diff < 0:
        # Need to turn some 0s to 1s
        diff = int(-diff)
        zeros_idx = np.where(h_b == 0)[0]
        # Sort by original value ascending to turn highest zeros into ones
        sort_idx = np.argsort(h_r[zeros_idx])
        to_flip = zeros_idx[sort_idx[:diff]]
        h_b[to_flip] = 1
        
    return h_b
