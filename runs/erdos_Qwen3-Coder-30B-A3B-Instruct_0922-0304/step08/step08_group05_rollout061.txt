import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and [0,1] bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find maximum overlap and corresponding indices
    max_val = np.max(overlap)
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # Gradient is based on the overlapping terms that achieve max
    for k in max_indices:
        # k is the lag, we need to consider all positions
        for i in range(n):
            # Position in h
            if 0 <= i < n:
                # Position in (1-h)
                j = i + k - (n-1)  # adjust index for correlation
                if 0 <= j < n:
                    # contribution to gradient of h[i]
                    grad[i] += -1.0 * dx
                elif 0 <= j < n:
                    # the effect of h[i] through (1-h[j])
                    grad[i] += 1.0 * dx
                else:
                    # This handling is outside the domain
                    pass
    
    # Apply constraint gradient for sum = target_sum
    grad_sum = np.ones(n) / target_sum
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality 
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available

    else:
        h = np.random.rand(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
    
    h = project_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = project_h(h, target_sum)
    
    # Parameters for subgradient descent
    learning_rate = 0.01
    momentum = 0.9
    
    # Tracking for early stopping
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    v = np.zeros(n_points)  # velocity for momentum
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 500:
        # Compute C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update using momentum
        v = momentum * v - learning_rate * grad
        h = h + v
        
        # Apply projection
        h = project_h(h, target_sum)
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            if last_improvement > 300:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
