import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each position, compute subgradient contribution
    for i in range(n):
        for k in max_indices:
            # k is the lag index in the full correlation
            # i + k corresponds to the left position in h, 
            # and k corresponds to right position in (1-h)
            if 0 <= i < n and 0 <= k < n:
                # This contributes to the overlap at position k
                grad[i] += (1 - h[k]) * dx  # gradient contribution is 1-h[k]
            # The case for k-i (when h[i] connects with 1-h[k]) in the correlation
            if 0 <= k < n and 0 <= i-k < n:
                grad[i] += h[i-k] * dx  # gradient contribution is h[i-k]
                
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    
    # Normalize to sum = target_sum, but respect bounds
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Parameters for subgradient descent
    learning_rate = 0.01
    momentum = 0.9
    vel = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum
        vel = momentum * vel - learning_rate * grad
        h = h + vel
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Update learning rate adaptively
        if iteration % 10 == 0 and iteration > 0:
            learning_rate *= 0.995
        
        iteration += 1
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
