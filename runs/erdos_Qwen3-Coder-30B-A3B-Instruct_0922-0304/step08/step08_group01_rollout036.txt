import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, norm, sum as cp_sum, multiply
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_r, n_points):
    """Round relaxed h to binary values"""
    # Simple thresholding at median
    median_val = np.median(h_r)
    h_b = (h_r > median_val).astype(float)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h_b)
    target_sum = n_points / 2
    
    if current_sum > target_sum:
        # Too many 1s, flip some to 0
        diff = int(current_sum - target_sum)
        # Flip largest values to 0
        flip_indices = np.argsort(h_r)[-diff:]
        h_b[flip_indices] = 0
    elif current_sum < target_sum:
        # Too many 0s, flip some to 1
        diff = int(target_sum - current_sum)
        # Flip smallest values to 1
        flip_indices = np.argsort(h_r)[:diff]
        h_b[flip_indices] = 1
        
    return h_b
