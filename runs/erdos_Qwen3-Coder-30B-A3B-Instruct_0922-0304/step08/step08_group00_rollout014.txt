import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function with linear ramp
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization with perturbation
    max_iter = 100
    perturbation_scale = 1e-3
    improvement_threshold = 1e-5
    iter_count = 0
    
    while time.time() - start_time < budget_s - 10 and iter_count < max_iter:
        # Store previous best
        prev_best = best_c5
        
        # Partition into even and odd indexed elements
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Optimize even elements fixing odd elements
        def optimize_even():
            def objective(even_vals):
                h_temp = best_h.copy()
                h_temp[even_indices] = even_vals
                return evaluate_overlap(h_temp, dx)
            
            bounds = [(0, 1) for _ in range(len(even_indices))]
            # Initial guess from current solution
            start_guess = best_h[even_indices]
            
            # Constraint: sum should be target_sum
            def constraint_sum(even_vals):
                h_temp = best_h.copy()
                h_temp[even_indices] = even_vals
                return np.sum(h_temp) - target_sum
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, start_guess, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'ftol': 1e-10, 'gtol': 1e-10})
            
            if result.success:
                new_even = np.clip(result.x, 0, 1)
                h_temp = best_h.copy()
                h_temp[even_indices] = new_even
                return h_temp
            return best_h.copy()
        
        # Optimize odd elements fixing even elements
        def optimize_odd():
            def objective(odd_vals):
                h_temp = best_h.copy()
                h_temp[odd_indices] = odd_vals
                return evaluate_overlap(h_temp, dx)
            
            bounds = [(0, 1) for _ in range(len(odd_indices))]
            # Initial guess from current solution
            start_guess = best_h[odd_indices]
            
            # Constraint: sum should be target_sum
            def constraint_sum(odd_vals):
                h_temp = best_h.copy()
                h_temp[odd_indices] = odd_vals
                return np.sum(h_temp) - target_sum
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, start_guess, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'ftol': 1e-10, 'gtol': 1e-10})
            
            if result.success:
                new_odd = np.clip(result.x, 0, 1)
                h_temp = best_h.copy()
                h_temp[odd_indices] = new_odd
                return h_temp
            return best_h.copy()
        
        # Alternating optimization step
        h_new = optimize_even()
        h_new = optimize_odd()
        
        # Test improved solution
        new_c5 = evaluate_overlap(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        # Perturb solution to escape local minima
        if iter_count % 5 == 0:
            perturb = np.random.uniform(-perturbation_scale, perturbation_scale, n_points)
            h_perturbed = np.clip(best_h + perturb, 0, 1)
            current_sum = np.sum(h_perturbed)
            if current_sum > 0:
                h_perturbed = h_perturbed * (target_sum / current_sum)
            temp_c5 = evaluate_overlap(h_perturbed, dx)
            if temp_c5 < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = temp_c5
        
        # Decrease perturbation scale
        perturbation_scale *= 0.98
        
        iter_count += 1
        
        # Early termination if no improvement
        if abs(prev_best - best_c5) < improvement_threshold:
            break
    
    # Local refinement on best solution
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
