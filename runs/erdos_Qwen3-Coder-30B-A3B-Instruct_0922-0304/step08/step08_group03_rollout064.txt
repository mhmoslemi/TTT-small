import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function over [0,2]."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to the feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_h(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Multi-resolution grid search
    coarse_grid = np.linspace(0, 2, 35)
    medium_grid = np.linspace(0, 2, 70)
    fine_grid = np.linspace(0, 2, 140)
    
    # Refinement levels (Coarse, Medium, Fine)
    grids = [coarse_grid, medium_grid, fine_grid]
    
    def evaluate_at_grid(grid_points, h_input):
        h_eval = np.interp(grid_points, np.linspace(0, 2, n_points), h_input)
        return h_eval
    
    # Gradient estimation
    def approximate_gradient(h, dx, eps=1e-4):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = project_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            h_minus = project_h(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Optimization iterations
    max_iter = 10000
    it = 0
    improvement_threshold = 1e-6
    no_improvement_count = 0
    max_no_improvement = 2000
    
    while it < max_iter and time.time() - start_time < budget_s - 1:
        # Periodically check for better solutions in grids
        if it % 50 == 0:
            for grid in grids:
                if len(grid) <= n_points:
                    h_interp = np.interp(grid, np.linspace(0, 2, n_points), h)
                    h_interp = project_h(h_interp, target_sum)
                    c5_test = compute_c5(h_interp, dx)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h_interp.copy()
                        
        # Apply gradient decent step
        grad = approximate_gradient(h, dx)
        
        # Update with momentum
        if it == 0:
            momentum = np.zeros_like(h)
        else:
            momentum = 0.9 * momentum - 0.001 * grad
        
        h_new = h + momentum
        h_new = project_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        if new_c5 < best_c5 - improvement_threshold:
            h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
                
        it += 1

    # Ensure best solution meets constraints
    final_h = project_h(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
