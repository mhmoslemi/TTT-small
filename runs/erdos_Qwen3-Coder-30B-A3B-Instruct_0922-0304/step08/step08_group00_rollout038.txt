import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # For a step function, compute correlation of h with (1-h)
    # Using full convolution mode to get all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The result has length 2*n-1 (shifts from -n+1 to n-1)
    # The actual overlap values are from shift 0 to 2*n-2
    # But we want max of all overlap values as defined in problem
    return np.max(correlation) * dx

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure h is valid before we start
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization strategy
    # Partition variables into even and odd indexed elements
    # Alternate fixing one group and optimizing the other
    
    max_iter = 200
    improvement_threshold = 1e-6
    perturb_mag = 1e-3
    decay_factor = 0.98
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Perturb the solution slightly and re-partition
        # This helps escape local optima and ensures diversity
        h_perturbed = h + np.random.uniform(-perturb_mag, perturb_mag, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum > 0:
            h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)
            
        # Partition into even and odd indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)

        h_even = h_perturbed[even_indices]

        def opt_even_objective(even_vals):
            h_new = h_perturbed.copy()
            h_new[even_indices] = even_vals

            h_new = h_perturbed.copy()
            h_new[even_indices] = even_vals
            return np.sum(h_new) - target_sum
            
        # Optimize even variables
        bounds_even = [(0, 1) for _ in range(len(even_indices))]
        cons_even = {'type': 'eq', 'fun': opt_even_constraint}
        
        result_even = minimize(opt_even_objective, h_even, method='SLSQP',
                               bounds=bounds_even, constraints=cons_even,
                               options={'ftol': 1e-8, 'maxiter': 100})
        
        if result_even.success:
            h_new = h_perturbed.copy()
            h_new[even_indices] = result_even.x
            current_sum = np.sum(h_new)
            if current_sum > 0:

            c5_even = evaluate_overlap(h_new, dx)

                best_h = refined_h
                best_c5 = final_c5
    except:
        pass  # If refinement fails, keep previous best
        
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
