import numpy as np
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to sum=target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate in frequency domain"""
    conj = np.conj(1 - H)
    return np.abs(H * conj)

def spectral_update(H, learning_rate):
    """Modify H to reduce magnitude of surrogate"""
    S = compute_surrogate(H)
    # Identify high contribution bins
    idxs = np.argsort(S)[::-1][:max(1, len(S)//10)]
    # Update selected bins
    for i in idxs:
        H[i] -= learning_rate * np.real(H[i]) * (S[i] - np.mean(S))
    return H

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.rand(n_points)
        h_current = project_h(h_current, target_sum)
        
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Main optimization loop
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # FFT transform
        H_current = fft(h_current)
        
        # Update in frequency domain
        H_new = H_current.copy()
        H_new = spectral_update(H_new, lr)
        
        # Reconstruct time domain signal
        h_new = np.real(ifft(H_new))
        
        # Project to feasible domain
        h_new = project_h(h_new, target_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current solution
        h_current = h_new.copy()
        
        # Adapt learning rate
        if iteration % 100 == 0 and iteration > 0:
            lr *= 0.5
            
        iteration += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    h_refined = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Make small random perturbation
        h_temp = h_refined.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] += delta
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
