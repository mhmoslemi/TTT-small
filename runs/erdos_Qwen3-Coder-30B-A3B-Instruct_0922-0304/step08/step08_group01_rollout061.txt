import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed(h_r, n_points, dx):
    """Solve relaxation: minimize z such that z >= sum h_r[x] * (1 - h_r[x+k]) * dx for all k."""
    # Add slack variable
    z = cp.Variable()
    
    # Get all possible lags (k values)
    # In correlate with mode="full", k ranges from -(n-1) to (n-1)
    constraints = []
    # We compute it using the full correlation and sample the needed values
    for k in range(-n_points + 1, n_points):
        k_abs = abs(k)
        # We construct the sums appropriately
        if k >= 0:
            # h_r[x] * (1 - h_r[x+k]) for x in range(0, n_points-k)
            expr = sum(h_r[i] * (1 - h_r[i + k]) for i in range(n_points - k))
        else:
            # h_r[x] * (1 - h_r[x+k]) for x in range(-k, n_points)
            expr = sum(h_r[i] * (1 - h_r[i + k]) for i in range(-k, n_points))
        constraints.append(z >= expr * dx)
    
    objective = cp.Minimize(z)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False, max_iters=5000)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return h_r.value, prob.value
    except:
        pass
    return h_r, np.inf

def threshold_round(h_r, target_sum):
    """Rounding heuristic with threshold."""
    # Use median threshold
    threshold = np.median(h_r)
    h_b = (h_r > threshold).astype(float)
    
    # Adjust to ensure sum matches target
    current_sum = np.sum(h_b)
    deficit = int(current_sum - target_sum)
    
    if deficit > 0:
        # Need to flip deficit 1s to 0s
        # Find indices of 1s, sort by closeness to threshold
        one_indices = np.where(h_b == 1)[0]
        diffs = np.abs(h_r[one_indices] - threshold)
        # Sort by differences ascending (0s closest to threshold)
        sort_indices = np.argsort(diffs)
        to_flip = one_indices[sort_indices[:deficit]]
        h_b[to_flip] = 0
    elif deficit < 0:
        # Need to flip |deficit| 0s to 1s
        zero_indices = np.where(h_b == 0)[0]
        diffs = np.abs(h_r[zero_indices] - threshold)
        # Sort by differences ascending (1s closest to threshold)
        sort_indices = np.argsort(diffs)
        to_flip = zero_indices[sort_indices[:abs(deficit)]]
        h_b[to_flip] = 1
    
    return h_b

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Alternate between relaxation and rounding
    max_iter = 200  # Decrease it for faster testing
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        h_r = h.copy()
        # Solve relaxed version
        try:
            h_r, z_relaxed = solve_relaxed(h_r, n_points, dx)
            if z_relaxed < np.inf and z_relaxed < best_c5 * 1.1:
                # Rounding heuristic
                h_b = threshold_round(h_r, target_sum)
                # Validate and update
                c5_b = compute_c5(h_b, dx)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                # Set h to h_b for next iteration
                h = h_b
            else:
                # Switch to simple gradient approach if relaxation fails
                # This is a fallback strategy to avoid getting stuck
                h = np.clip(h + 0.01 * (1 - 2 * h), 0, 1)
                h = normalize_h(h, target_sum)
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        except:
            # Fallback gradient descent
            h = np.clip(h + 0.01 * (1 - 2 * h), 0, 1)
            h = normalize_h(h, target_sum)
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        iter_count += 1
        if no_improvement_count > 10:
            # Occasionally attempt perturbation
            h = best_h * 0.9 + np.random.rand(n_points) * 0.1
            h = normalize_h(h, target_sum)
            no_improvement_count = 0
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
