import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Solve convex relaxation using CVXPY
        h_relaxed = np.array(h)
        
        # Create the relaxation problem
        z = cp.Variable()
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Overlap constraints - use outer product approximation
        for k in range(2 * n_points - 1):
            if k < n_points:
                # Compute overlap for shift k
                overlap_expr = 0
                for i in range(n_points):
                    j = i + k
                    if j < n_points:
                        overlap_expr += h_vars[i] * (1 - h_vars[j])
            else:
                # Handle negative shifts differently
                overlap_expr = 0
                for i in range(n_points):
                    j = i + k - n_points
                    if j >= 0:
                        overlap_expr += h_vars[i] * (1 - h_vars[j])
            
            constraints.append(overlap_expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, eps=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed = h_vars.value
        except:
            pass
        
        # Round the relaxed solution
        h_new = np.zeros(n_points)
        for i in range(n_points):
            if h_relaxed[i] > 0.5:
                h_new[i] = 1.0
            else:
                h_new[i] = 0.0
        
        # Scale to match target sum
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        h_new = np.clip(h_new, 0, 1)
        
        # Greedy adjustment for better result
        current_c5 = compute_c5(h_new, dx)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
        
        # Apply gradient like adjustments for some iterations
        if iter_count % 100 == 0:
            for i in range(n_points):
                # Approximate gradient
                grad = 0
                for k in range(n_points):
                    if i + k < n_points:
                        grad += (1 - 2 * h_new[i]) * (1 - h_new[i + k])
                    if i - k >= 0:
                        grad += (1 - 2 * h_new[i]) * (1 - h_new[i - k])
                
                # Update with small step size
                step_size = 0.002 * np.sign(grad)
                h_new[i] = h_new[i] + step_size * (1 - 2 * h_new[i])
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            test_c5 = compute_c5(h_new, dx)
            if test_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = test_c5
                
        h = h_new.copy()
        iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
