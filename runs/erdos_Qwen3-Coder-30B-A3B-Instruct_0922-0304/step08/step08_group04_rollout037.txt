import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def solve_convex_relaxation(h_init, n_points, dx):
    """Solve the convex relaxation of the problem."""
    # Variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    
    # Add constraint for the sum
    constraints.append(sum(h) == n_points / 2)
    
    # Add overlap constraints
    for k in range(2 * n_points - 1):
        # For each shift k, we create the constraint sum(h[i] * (1 - h[i+k])) <= t
        # We use auxiliary variables to linearize this
        if k < n_points:
            # Index conversion for circular boundary
            shifted_indices = list(range(k, min(k + n_points, 2 * n_points - 1)))
            # Only take valid indices in h
            valid_indices = [i for i in range(n_points) if i < len(shifted_indices)]
            # Adjust shifted indices to actual h indices
            actual_shifted = [(i + k) % n_points for i in valid_indices]
            # Create the sum for this shift
            overlap_sum = sum(h[i] * (1 - h[j]) for i, j in zip(valid_indices, actual_shifted))
            constraints.append(overlap_sum <= t)
        else:
            # The remaining shifts are handled with properly shifted indices
            start_idx = k - (n_points - 1)
            # Only valid for positions where we can form a valid overlap
            overlap_sum = sum(h[i] * (1 - h[(i + k) % n_points]) for i in range(n_points) if i + k < 2 * n_points - 1)
            constraints.append(overlap_sum <= t)
    
    # Objective
    objective = Minimize(t)
    
    # Problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        return h.value
    except:
        return h_init

def compute_gradient(h, dx, n_points):
    """Compute the gradient of C5 at h."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient based on the maximum overlap
    for i in range(n_points):
        # Calculate gradients with respect to h[i]
        grad[i] = 0.0
        for k in range(2 * n_points - 1):
            pos1 = i
            pos2 = i + k - (n_points - 1)
            if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                # There's overlap at some shift k
                # We're computing the derivative of sum h[i] * (1 - h[i+k]) with respect to h[i]
                grad[i] += 1.0 - 2 * h[pos1] - 2 * h[pos2]
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=5000, lr=0.01, decay=0.995, tolerance=1e-8):
    """Optimize h using gradient descent with a max norm constraint."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    prev_loss = float('inf')
    no_improvement_count = 0
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Compute loss and gradient
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
                
            # Check termination conditions
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 2000:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient (using short-circuit approach)
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Use step size limiting for stability
        update = np.clip(update, -0.1, 0.1)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Try convex relaxation for better initial solution
    try:
        relaxed_h = solve_convex_relaxation(h, n_points, dx)
        h = relaxed_h
    except:
        pass  # Silently proceed with random initialization
    
    # Optimize with gradient descent
    best_h, best_c5 = optimize_with_gradient_descent(h, dx, n_points, max_iter=5000, lr=0.01, decay=0.995)
    
    # Make sure we have the exact best solution within the recorded time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        # Do one final check with exact computation
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
