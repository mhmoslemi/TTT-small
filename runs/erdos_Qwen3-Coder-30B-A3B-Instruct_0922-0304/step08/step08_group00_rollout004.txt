import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Convex optimization approach
    # Formulate and solve the problem using CVXPY
    # Variables
    h_var = cp.Variable(n_points)
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Objective: minimize the maximum correlation (through a relaxation)
    # We can introduce auxiliary variables to bound the maximum overlap.
    
    # Introduce the maximum overlap as a variable
    max_overlap = cp.Variable()
    
    # Create a list of constraints to bound the overlap at each shift
    # For this problem, we can't directly form all constraints without explicit computation
    # So we just use a more manageable approach:
    # Minimize max_k integral h(x)(1-h(x+k)) dx
    
    # Substitute the most effective convex relaxation
    # Let's denote A as the matrix such that each row computes h_k = sum h_i (1-h_{i+k})
    # This will be hard directly, so we'll try an alternative:
    
    # In practice, we solve directly with a slightly modified approach:
    
    # Run the solver with a deterministic convex approach first
    # We'll solve with a fixed iterative scheme to locally refine
    
    t = time.time()
    max_time = budget_s - 10  # for safety
    
    # Start with a simple refinement
    refined_h = best_h.copy()
    previous_c5 = best_c5 + 1.0  # force at least one iteration 
    patience = 0
    max_patience = 20
    
    while time.time() - start_time < max_time:
        if abs(previous_c5 - best_c5) < 1e-7:
            patience += 1
            if patience > max_patience:
                break
        else:
            patience = 0
        previous_c5 = best_c5
        
        # Local SLSQP optimization
        from scipy.optimize import minimize
        
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        result = minimize(objective, refined_h, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            current_sum = np.sum(refined_h)
            if current_sum > 0:
                refined_h = refined_h * (target_sum / current_sum)
            new_c5 = evaluate_overlap(refined_h, dx)
            if new_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = new_c5
        else:
            # fallback to ad-hoc local update
            # try small perturbations to improve solution
            h = refined_h.copy()
            step_size = 0.005

                    Improved = True

    # Final evaluation

    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
