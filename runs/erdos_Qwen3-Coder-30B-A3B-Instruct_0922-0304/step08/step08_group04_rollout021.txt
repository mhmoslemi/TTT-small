import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def frank_wolfe_step(h, dx, n_points):
    """Compute the Frank-Wolfe step for the current h."""
    # Compute overlap for all lags
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_k = np.argmax(overlap)
    
    # Compute gradient at current h
    # The gradient of overlap_k w.r.t. h[i] is (1 - 2*h[i]) * sum of 1 - h[j] where j is such that
    # the correlation index that contributes is i + k (mod n_points) = j, or equivalently 
    # j - k (mod n_points) = i
    
    # We can get gradient as a vector
    grad = np.zeros_like(h)
    for i in range(n_points):
        # For k-th lag, we get overlap contribution from h[i] * (1 - h[i+k])
        # But since we're looking at max overlap_k, we compute the gradient
        # as a sum over all positions where this overlap is largest
        for j in range(n_points):
            if (j - max_k) % n_points == i:
                grad[i] += 1 - 2 * h[i]
    
    # Solve linear subproblem: min_v <grad, v> s.t. sum(v) = n_points/2, 0 <= v <= 1
    # This is a simple LP that can be solved using linprog with bounds
    c = grad  # Coefficients for linear objective
    A_eq = np.array([np.ones(n_points)])  # Equality constraint: sum v = n_points / 2
    b_eq = np.array([n_points / 2])  # RHS of equality constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Solve
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        v = result.x
    else:
        # Fallback to projection approach if optimization fails
        v = np.ones(n_points) * (n_points / 2) / n_points
        v = np.clip(v, 0, 1)
        v = v * (n_points / 2) / np.sum(v)
    
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    tolerance = 1e-6
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize h to have the exact sum
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    for i in range(max_iter):
        # Check if enough time remains
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Compute Frank-Wolfe step
        v = frank_wolfe_step(h, dx, n_points)
        
        # Line search to find optimal step size
        # Try a few candidate step sizes
        eta_candidates = np.linspace(0.01, 1.0, 20)
        best_eta = 0.01
        best_new_c5 = best_c5
        
        for eta in eta_candidates:
            h_new = (1 - eta) * h + eta * v
            # Project onto constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_eta = eta
        
        # Update h
        h = (1 - best_eta) * h + best_eta * v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if best_new_c5 < best_c5:
            best_c5 = best_new_c5
            best_h = h.copy()
        
        # Early stopping based on small improvement
        if abs(best_c5 - evaluate_c5(h, dx)) < tolerance:
            break
            
        if best_c5 < 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
