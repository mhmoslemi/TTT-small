import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a structured construction
        h = np.zeros(n_points)
        # Set a central region to 1, others 0
        start_idx = n_points // 4
        end_idx = 3 * n_points // 4
        h[start_idx:end_idx] = 1.0
        # Add small noise
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-6
    last_improvement = 0
    max_idle = 1000
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < max_idle:
        # Simulated Annealing Step
        candidate_h = h.copy()
        
        # Perturb 10% of the values
        perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in perturb_indices:
            # Flip or swap value
            if np.random.random() < 0.5:
                # Scale down by small factor
                candidate_h[i] *= 0.97
            else:
                # Find another random index to swap with
                j = np.random.randint(0, n_points)
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Ensure values in valid range
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = normalize_h(candidate_h, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        # Acceptance criteria
        current_c5 = compute_c5_exact(h, dx)
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            h = candidate_h
            current_c5 = candidate_c5
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Cool down
        T = max(min_T, T * cooling_rate)
        
        # Periodic refinement with cutting plane method
        if iteration % 50 == 0 and iteration > 0:
            # Refine using gradient-like heuristic on worst overlap
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            
            # Try small adjustments to reduce the maximum overlap
            k = max_idx - (n_points - 1)  # lag corresponding to max overlap
            # Modify around the indices that contribute to this overlap
            for i in range(max(0, k), min(n_points, n_points - k)):
                # Try local small adjustments
                test_h = h.copy()
                test_h[i] = max(0, test_h[i] - 0.01)
                test_h[i + k] = min(1, test_h[i + k] + 0.01)
                # Normalize
                test_h = normalize_h(test_h, target_sum)
                test_c5 = compute_c5_exact(test_h, dx)
                if test_c5 < current_c5:
                    h = test_h
                    current_c5 = test_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        last_improvement = 0
                        
        iteration += 1
    
    # Final check
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
