import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient with respect to a given shift k*"""
    grad = np.zeros(n_points)
    # For each position, the gradient is (1 - h[x + shift]) if valid
    for i in range(n_points):
        idx = i + shift
        if 0 <= idx < n_points:
            grad[i] = 1 - h[idx]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Project to satisfy constraint
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    patience = 0
    max_patience = 100
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        # The shift corresponds to the full correlation length at max overlap
        # Correlation is computed from index 0 to n_points-1 in the full result
        # For a full correlation, an offset n_points-1 means shift of -n_points+1
        # Using the edges of overlap vector, we need to map index back to shift
        # overlap index 0 means shift -n_points+1 (right shift of 0, but left in correlation)
        # overlap index n_points-1 means shift 0 (no shift)
        # overlap index 2*n_points-2 means shift n_points-1 (left shift of 0, but right)
        k_star = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        gradient = compute_gradient(h_current, k_star, n_points)
        
        # Update h using gradient descent
        h_new = h_current - step_size * gradient
        
        # Project h to feasible region
        h_new = project_h(h_new, n_points)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
            step_size = min(step_size * 1.05, 0.5)  # Adaptive step size up
        else:
            patience += 1
            if patience > max_patience:
                # Reset with some noise to escape local minima
                h_new = h_current + np.random.normal(0, 0.02, n_points)
                h_new = project_h(h_new, n_points)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                patience = 0
                step_size = max(step_size * 0.9, 0.01)  # Reduce step size
        
        h_current = h_new
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
