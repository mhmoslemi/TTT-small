import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, noise_level=0.05):
    """Mutate an individual with noise."""
    noise = np.random.normal(0, noise_level, len(h))
    h = h + noise
    h = np.clip(h, 0, 1)
    return h

def tournament_selection(population, fitnesses, k=2):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), size=k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def genetic_algorithm(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, target_sum)
        population.append(parent.copy())
        for _ in range(population_size - 1):
            individual = mutate(parent.copy(), 0.1)
            individual = normalize_h(individual, target_sum)
            population.append(individual)
    else:
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, target_sum)
            population.append(individual)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    generation = 0
    
    # GA loop
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitnesses = Parallel(n_jobs=1)(
            delayed(evaluate_c5)(ind, dx) for ind in population
        )
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection, crossover, mutation
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, 0.05 * 0.99 ** (generation // 10))
            child2 = mutate(child2, 0.05 * 0.99 ** (generation // 10))
            
            # Repair feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population
    
    if best_h is None:
        # Fallback to the best individual from the final population
        final_fitnesses = Parallel(n_jobs=1)(
            delayed(evaluate_c5)(ind, dx) for ind in population
        )
        min_fitness_idx = np.argmin(final_fitnesses)
        best_h = population[min_fitness_idx].copy()
        best_c5 = final_fitnesses[min_fitness_idx]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = genetic_algorithm(seed, budget_s)
    return (best_h.tolist(), best_c5, 70)
