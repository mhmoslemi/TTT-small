import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the penalty approximation."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute overlap for all lags
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # Compute subgradient contributions from active indices
    for k in max_indices:
        # For each lag, add contribution to gradient
        if k < n:
            # h[i] contributes to overlap[k] when h[i] * (1-h[i+k]) is in max
            for i in range(n):
                if i+k < n:
                    grad[i] += 1 - 2 * h[i+k]
        else:
            # Handle symmetric part k >= n
            offset = k - n + 1
            for i in range(n):
                if i-offset >= 0:
                    grad[i] += 1 - 2 * h[i-offset]
    
    # Normalize gradient
    if np.linalg.norm(grad) > 0:
        grad = grad / np.linalg.norm(grad)
    
    # Apply constraint gradient (sum = target_sum)
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.99
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
