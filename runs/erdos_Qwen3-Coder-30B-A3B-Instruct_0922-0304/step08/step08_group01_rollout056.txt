import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try to improve with convex relaxation approach
    max_iter = 5000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Convex relaxation
        h_relaxed = h.copy()
        h_relaxed = normalize_h(h_relaxed, target_sum)
        
        # Binary rounding heuristic
        # Use a threshold of 0.5
        h_rounded = np.zeros_like(h_relaxed)
        for i in range(len(h_relaxed)):
            if h_relaxed[i] < 0.5:
                h_rounded[i] = 0.0
            else:
                h_rounded[i] = 1.0
        
        # Adjust to meet sum constraint
        current_sum = np.sum(h_rounded)
        diff = target_sum - current_sum
        
        if diff != 0:
            # Try to adjust few points to meet constraint
            # Find indices where we should change
            sorted_indices = np.argsort(np.abs(h_relaxed - 0.5))[::-1]
            adjust_count = int(abs(diff))
            if diff > 0:  # Need to add 1s
                for i in range(min(adjust_count, len(sorted_indices))):
                    idx = sorted_indices[i]
                    if h_rounded[idx] == 0.0:
                        h_rounded[idx] = 1.0
            else:  # Need to subtract 1s
                for i in range(min(adjust_count, len(sorted_indices))):
                    idx = sorted_indices[i]
                    if h_rounded[idx] == 1.0:
                        h_rounded[idx] = 0.0
        
        # Ensure bounds
        h_rounded = np.clip(h_rounded, 0, 1)
        h_rounded = normalize_h(h_rounded, target_sum)
        
        # Evaluate the new solution
        new_c5 = compute_c5(h_rounded, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_rounded.copy()
        
        # Sometimes make small random changes for exploration
        if iter_count % 100 == 0 or np.random.rand() < 0.1:
            h_trial = h.copy()
            # Make a small random change
            idx = np.random.randint(0, len(h_trial))
            step_size = 0.05
            h_trial[idx] += step_size * (np.random.rand() - 0.5)
            # Clip and re-normalize
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = normalize_h(h_trial, target_sum)
            trial_c5 = compute_c5(h_trial, dx)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = h_trial.copy()
        
        # Early stopping condition
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
