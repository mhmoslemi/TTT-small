import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
    
    # Initialize parameters for trust-region-like updates
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Coordinate-wise descent with momentum
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    momentum = 0.9
    step_size = 0.01
    
    # Velocity storage
    velocity = np.zeros(n_points)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Gradient estimation for each coordinate
        gradients = np.zeros(n_points)
        eps = 0.001
        
        for i in range(n_points):
            # Forward difference
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            gradients[i] = (max_plus - current_max) / eps
        
        # Use momentum to update velocity
        velocity = momentum * velocity - step_size * gradients
        
        # Apply update
        h_new = h + velocity
        h_new = normalize_h(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate the new function value
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < current_max:
            h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            # Reduce step size or change strategy
            step_size *= 0.9
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
