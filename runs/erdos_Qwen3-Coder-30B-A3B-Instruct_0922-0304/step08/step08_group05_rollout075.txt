import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_individual(n_points, target_sum, use_initial=False):
    """Create a valid individual (h vector)"""
    if use_initial and 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        return h
    
    # Generate a random individual
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.1):
    """Mutate an individual with Gaussian noise"""
    # Apply Gaussian noise
    noise = np.random.normal(0, sigma, len(h))
    h = h + noise
    
    # Clip to valid range
    h = np.clip(h, 0, 1)
    
    # Re-normalize
    h = normalize_h(h, np.sum(h))
    return h

def crossover_parents(p1, p2):
    """Create offspring from two parents by averaging"""
    # Average the parents
    offspring = (p1 + p2) / 2.0
    
    # Clip and normalize
    offspring = np.clip(offspring, 0, 1)
    return normalize_h(offspring, np.sum(offspring))

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    tournament_indices = random.sample(range(len(population)), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.1
    sigma = 0.1
    
    # Initialize population
    population = []
    for i in range(population_size):
        use_initial = (i == 0)  # Use initial if first
        individual = create_individual(n_points, target_sum, use_initial)
        population.append(individual)
    
    # Evaluate fitness for all individuals
    fitnesses = []
    for individual in population:
        fitness = compute_c5_exact(individual, dx)
        fitnesses.append(fitness)
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolution loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Selection
        new_population = []
        elite_count = population_size // 5
        top_indices = np.argsort(fitnesses)[:elite_count]
        for idx in top_indices:
            new_population.append(population[idx].copy())
        
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover_parents(parent1, parent2)
            
            # Mutation
            if random.random() < mutation_rate:
                offspring = mutate_individual(offspring, sigma=sigma)
            
            new_population.append(offspring)
        
        # Update population
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            fitness = compute_c5_exact(individual, dx)
            fitnesses.append(fitness)
        
        # Check for new best
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        generation += 1
    
    # Final verification
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
