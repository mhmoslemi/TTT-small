import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_coefficients(h, n_basis=15):
    """Compute Fourier coefficients for h using windowed sine basis."""
    x = np.linspace(0, 2, len(h))
    coeffs = np.zeros(n_basis)
    for i in range(n_basis):
        basis = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        coeffs[i] = np.dot(h, basis) / np.sum(basis**2)
    return coeffs

def backtransform(coeffs, n_points=70):
    """Transform Fourier coefficients back to function values."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i, c in enumerate(coeffs):
        basis = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        h += c * basis
    return h

def project(h, target_sum=35.0):
    """Project h to feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    
    h = project(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Fourier-based optimization with momentum
    n_basis = min(15, n_points // 4)
    coeffs = compute_coefficients(h, n_basis)
    
    # Optimizer settings
    learning_rate = 0.001
    momentum = 0.9
    velocity = np.zeros_like(coeffs)
    
    iterations = 0
    max_iterations = 50000
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iterations < max_iterations and time.time() - start_time < budget_s - 1:
        # Transform coefficients to function space
        h_trial = backtransform(coeffs, n_points)
        h_trial = project(h_trial, target_sum)
        
        # Analyze overlap
        overlap = correlate(h_trial, 1 - h_trial, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h_trial.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient indirectly: approximate change in overlap
        # by finite differencing with small perturbations
        grad = np.zeros_like(coeffs)
        epsilon = 1e-5
        for i in range(n_basis):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += epsilon
            h_plus = backtransform(coeffs_plus, n_points)
            h_plus = project(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= epsilon
            h_minus = backtransform(coeffs_minus, n_points)
            h_minus = project(h_minus, target_sum)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / (2 * epsilon)
        
        # Update using momentum
        velocity = momentum * velocity - learning_rate * grad
        coeffs += velocity
        
        iterations += 1
        no_improvement_count += 1
        
        if current_max < best_c5:
            no_improvement_count = 0
    
    # Final validation and refinement
    final_h = backtransform(coeffs, n_points)
    final_h = project(final_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
