import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a symmetric triangular ramp
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
    
    # Normalize and ensure bounds
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use a hybrid approach: coordinate descent with bounds checking
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 2000
    
    # Define a window for local optimization
    window_size = 5
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient estimates for all points
        gradients = np.zeros(n_points)
        
        for i in range(n_points):
            # Estimate gradient for coordinate i
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.005)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.005)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            gradients[i] = (max_plus - max_minus) / 0.01
        
        # Modify top gradient points
        update_indices = np.argsort(np.abs(gradients))[-window_size:][::-1]
        improved = False
        
        for i in update_indices:
            if not improved:
                current_val = h[i]
                step = 0.005
                if gradients[i] < 0:
                    # If increasing degrades, decrease
                    new_val = max(0.0, current_val - step)
                else:
                    new_val = min(1.0, current_val + step)
                
                h_new = h.copy()
                h_new[i] = new_val
                h_new = normalize_h(h_new, target_sum)
                h_new = np.clip(h_new, 0, 1)
                
                new_c5 = compute_c5(h_new, dx)
                
                if new_c5 < current_c5:
                    h = h_new.copy()
                    improved = True
                else:
                    # Try a smaller step
                    step = step / 2
                    if gradients[i] < 0:
                        new_val = max(0.0, current_val - step)
                    else:
                        new_val = min(1.0, current_val + step)
                        
                    h_new = h.copy()
                    h_new[i] = new_val
                    h_new = normalize_h(h_new, target_sum)
                    h_new = np.clip(h_new, 0, 1)
                    
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < current_c5:
                        h = h_new.copy()
                        improved = True
        
        # If we didn't improve, try random small adjustments
        if not improved:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Random restart
                if 'initial_h_values' in globals():
                    h = np.array(initial_h_values)
                    h = normalize_h(h, target_sum)
                    h = np.clip(h, 0, 1)
                else:
                    h = np.zeros(n_points)
                    mid = n_points // 2
                    for j in range(mid):
                        h[j] = j / mid
                        h[n_points - 1 - j] = j / mid
                    h[mid] = 1.0
                    h = normalize_h(h, target_sum)
                    h = np.clip(h, 0, 1)
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
