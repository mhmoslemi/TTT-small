import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function from 0 to 1 and back to 0 over [0,2]."""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    return h

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Gradient-based optimization using Nesterov momentum
    max_iter = 80000
    lr = 0.001
    momentum = 0.9
    
    v = np.zeros_like(h)
    x_prev = h.copy()
    
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        x = h.copy()
        # Gradient of overlap with respect to h
        # Use central difference approximation for robustness
        eps = 1e-6
        grad = np.zeros_like(h)
        
        # For each point, compute finite difference gradient
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            
            h_plus = project_h(h_plus, target_sum)
            h_minus = project_h(h_minus, target_sum)
            
            val_plus = compute_c5(h_plus, dx)
            val_minus = compute_c5(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
            
        # Gradient descent with momentum
        v = momentum * v - lr * grad
        x_new = x + v
        h = project_h(x_new, target_sum)
        
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
