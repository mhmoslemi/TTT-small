import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function for initialization"""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_coefficients(h, n_basis):
    """Compute Fourier coefficients for a given h"""
    x = np.linspace(0, 2, len(h))
    coeffs = np.zeros(n_basis)
    for i in range(n_basis):
        basis = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        coeffs[i] = np.dot(h, basis) / np.sum(basis**2)
    return coeffs

def backtransform(coeffs, n_points, n_basis):
    """Transform Fourier coefficients back to h space"""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_basis):
        basis = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        h += coeffs[i] * basis
    return h

def project(h, target_sum):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = project(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Fourier basis optimization
    n_basis = min(n_points // 4, 16)  # Number of basis functions
    coeffs = compute_coefficients(h, n_basis)
    
    # Adaptive gradient descent with momentum
    momentum = 0.9
    learning_rate = 0.001
    velocity = np.zeros_like(coeffs)
    max_iter = 50000
    patience = 1000
    no_improvement_count = 0
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and find max position
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Approximate gradient computation
        grad = np.zeros(n_basis)
        eps = 1e-4
        for i in range(n_basis):
            # Perturb coefficient
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            
            h_plus = backtransform(coeffs_plus, n_points, n_basis)
            h_plus = project(h_plus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            
            h_minus = backtransform(coeffs_minus, n_points, n_basis)
            h_minus = project(h_minus, target_sum)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / (2 * eps)
            
        # Update coefficients with momentum
        velocity = momentum * velocity - learning_rate * grad
        coeffs += velocity
        
        # Transform back to h space
        h = backtransform(coeffs, n_points, n_basis)
        h = project(h, target_sum)
        
        if no_improvement_count > patience:
            break

    # Final computation of best values
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
