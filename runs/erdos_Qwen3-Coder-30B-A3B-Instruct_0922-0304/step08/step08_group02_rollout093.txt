import numpy as np
from scipy.fft import fft, ifft
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1], scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain."""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, surrogate, learning_rate=0.01):
    """Update frequency domain representation to reduce surrogate."""
    # Apply gradient descent step
    grad = H * (1 - 2 * H)  # derivative of |H(1-H)|^2 with respect to H
    H_new = H - learning_rate * grad / (1e-10 + np.abs(H)**2)
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.rand(n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Compute Fourier transform
        H = fft(h_current)
        
        # Compute surrogate objective
        surrogate = compute_surrogate(H)
        
        # Pass frequency domain update to reduce the objective
        H_new = spectral_update(H, surrogate, learning_rate)
        
        # Inverse FFT to reconstruct h
        h_new = np.real(ifft(H_new))
        
        # Apply feasibility constraints
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Use current solution for next iteration
        h_current = h_new.copy()
        
        # Adjust learning rate
        if iteration > 0 and iteration % 100 == 0:
            learning_rate *= 0.5
            
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
