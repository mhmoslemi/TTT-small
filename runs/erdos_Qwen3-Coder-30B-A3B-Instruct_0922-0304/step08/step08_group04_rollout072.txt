import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe algorithm parameters
    max_iter = 10000
    tolerance = 1e-8
    
    for iteration in range(max_iter):
        # Time check
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (approximate by focusing on max overlap)
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Compute the gradient of the maximum overlap term with respect to h
        grad = np.zeros_like(h)
        # The gradient of h[i] in h[i] * (1 - h[i + k]) is (1 - 2*h[i])
        for i in range(n_points):
            j = (i + max_idx - (n_points - 1)) % n_points
            grad[i] = 1 - 2 * h[i]
        
        # Solve linear subproblem: minimize grad^T * v subject to constraints
        # v >= 0 and sum(v) = n_points / 2
        # We can solve this analytically since it's a simple LP
        # v = argmin v^T * grad, subject to v_i >= 0, sum(v) = n_points / 2
        # The optimal solution sets v to be as negative as possible in the directions of grad
        # For h_i, we want to increase toward 1 if grad[i] < 0
        # But we're constrained by bounds and the sum constraint
        # Method: Set v = max(0, (n_points/2 - sum(max(0, h)))*e_i) where e_i=1 if grad[i]<0
        
        # Initialize v
        v = np.zeros_like(h)
        
        # Optimize by making v flip the sign of grad
        # The optimal is v_i = 1 where grad_i is negative and similar for constraint
        # We project the gradient - positive if grad[i] is negative, weights towards 0
        # For simplicity, let's just make v that pushes h in direction of the gradient
        
        v = np.maximum(0, -grad)
        v_sum = np.sum(v)
        if v_sum > 0:
            v = v * (n_points / 2) / v_sum
        else:
            v = np.full_like(v, 1.0 / n_points)
        
        # Another much more stable and robust method: let's use a descent direction
        # proposed by Frank-Wolfe: some feasible point v minimizing gradient^T * v
        # Here we can use the fact that we want to minimize sum gradient_i * v_i
        # subject to sum v_i = n/2 and 0 <= v_i <= 1. 
        # If grad[i] > 0, set v[i]=0, otherwise set v[i]=1. 
        # If needed, scale to meet sum constraint.
        
        v = np.zeros_like(h)
        v[grad < 0] = 1
        
        current_sum = np.sum(v)
        if current_sum > 0:
            v = v * (n_points / 2) / current_sum
        else:
            v = np.full_like(h, 1.0 / n_points)
            
        # Line search parameter
        eta = 0.1 # Start with a conservative step
        # We try to find a suitable value for eta by trying different values,
        # especially small ones for simple convergence
        
        # Frank-Wolfe update
        new_h = (1 - eta) * h + eta * v
        
        # Project into convex hull: ensure bounds [0,1] and sum = n_points/2
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new value
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        # Stopping criterion
        if np.abs(best_c5 - evaluate_c5(h, dx)) < tolerance:
            break
            
        h = new_h
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
