import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, dx, target_sum):
    """Ensure h is in [0,1] and has correct integral."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h: use the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Construct initial function - a smooth triangular ramp
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = 1.0 - (i - mid) / (n_points - mid)
        h = normalize_h(h, target_sum)
        
    h = project_h(h, dx, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Simpler gradient-based optimization with momentum
    max_iter = 20000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    # Momentum term
    velocity = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.01
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current gradients
        gradients = np.zeros(n_points)
        current_overlap = evaluate_overlap(h, dx)
        
        if current_overlap < best_c5:
            best_c5 = current_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Approximate gradient: finite difference approach
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = project_h(h_plus, dx, target_sum)
            overlap_plus = evaluate_overlap(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            h_minus = project_h(h_minus, dx, target_sum)
            overlap_minus = evaluate_overlap(h_minus, dx)
            
            gradients[i] = (overlap_plus - overlap_minus) / (2 * eps)

        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradients
        h = h + velocity
        h = project_h(h, dx, target_sum)
        
        # Check for improvement
        new_c5 = evaluate_overlap(h, dx)
        if new_c5 < current_overlap:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                no_improvement_count = 0
                learning_rate *= 0.99  # Reduce learning rate
                
        iter_count += 1

    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
