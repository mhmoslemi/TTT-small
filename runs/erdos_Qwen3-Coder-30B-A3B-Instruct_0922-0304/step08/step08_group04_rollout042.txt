import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Early stopping condition
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Pattern search with different strategies
    strategies = [
        # Uniform grid search
        lambda h, dx, n_points: grid_search(h, dx, n_points, 0.05),
        # Adaptive search with different steps
        lambda h, dx, n_points: adaptive_search(h, dx, n_points),
        # Random perturbation with normalization
        lambda h, dx, n_points: random_perturbation(h, dx, n_points)
    ]
    
    search_iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Try different strategies
        strategy_idx = search_iterations % len(strategies)
        h_new = strategies[strategy_idx](h, dx, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        search_iterations += 1
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
            if best_c5 <= 0.3807:
                break
        else:
            no_improvement_count += 1
            
        # Update solution for next iteration
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)

def grid_search(h, dx, n_points, step_size):
    """Perform grid search over h values with specified step size."""
    h_new = h.copy()
    
    # Try small perturbations
    for i in range(n_points):
        original = h_new[i]
        best_val = original
        best_c5 = evaluate_c5(h_new, dx)
        
        for pert in np.arange(max(0, original - step_size), min(1, original + step_size + step_size/2), step_size):
            h_new[i] = pert
            c5 = evaluate_c5(h_new, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_val = pert
        h_new[i] = best_val
        
    # Normalize after perturbations
    h_new = normalize_h(h_new, n_points / 2)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def adaptive_search(h, dx, n_points):
    """Adaptive search with different step sizes."""
    h_new = h.copy()
    
    # Try to improve by modifying approximately 10% of values at a time
    random_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
    step_size = 0.02
    
    for i in random_indices:
        if np.random.random() < 0.7:  # 70% chance to modify
            original = h_new[i]
            best_val = original
            best_c5 = evaluate_c5(h_new, dx)
            
            # Try a wider range initially
            start_val = max(0, original - step_size*2)
            end_val = min(1, original + step_size*2)
            
            for pert in np.arange(start_val, end_val + step_size/2, step_size):
                h_new[i] = pert
                c5 = evaluate_c5(h_new, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_val = pert
            h_new[i] = best_val
    
    # Normalize after perturbations
    h_new = normalize_h(h_new, n_points / 2)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def random_perturbation(h, dx, n_points):
    """Apply small random perturbations to the function."""
    h_new = h.copy()
    
    # Randomly modify 15% of values
    num_modify = max(1, n_points // 7)
    indices = np.random.choice(n_points, size=num_modify, replace=False)
    
    for i in indices:
        # Add small random noise and clip
        noise = np.random.normal(0, 0.03)
        h_new[i] = np.clip(h_new[i] + noise, 0, 1)
    
    # Normalize to satisfy integral constraint
    h_new = normalize_h(h_new, n_points / 2)
    return h_new
