```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum

    # Formulate and solve QP problem using CVXPY
    h_var = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints for all shifts k
    # We compute overlap for each shift k in the full correlation
    # from -n_points+1 to n_points-1
    for k in range(-(n_points - 1), n_points):
        # Build the constraint based on correlation
        expr = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h_var[i] * (1 - h_var[j])
        constraints.append(expr <= t)
    
    # Create and solve the QP problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_bound = t.value
        else:
            # Fallback to original solution
            h_opt = h
            c5_bound = evaluate_c5(h_opt, dx)
    except Exception:
        # Fallback to original solution on any solver error
        h_opt = h
        c5_bound = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), c5_bound, n_points)
```
