import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().astype(np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function for optimization
    def objective(x):
        return compute_c5(x, dx)
    
    # Optimization with SLSQP
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time <= 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Run optimization
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6},
            callback=lambda x: None
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = compute_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
