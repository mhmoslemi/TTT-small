import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (target_sum / np.sum(h))
    
    # Formulate as QP
    # Variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == target_sum,
                   h_var >= 0,
                   h_var <= 1,
                   t >= 0]
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        # Compute the valid indices for shift k
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i = 0 to n_points - k - 1
            expr = 0
            for i in range(n_points - k):
                expr += h_var[i] * (1 - h_var[i + k])
        else:
            # h[i] * (1 - h[i + k]) for i = -k to n_points - 1
            expr = 0
            for i in range(-k, n_points):
                expr += h_var[i] * (1 - h_var[i + k])
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_final = h_var.value
            c5_bound = t.value
        else:
            # Fallback to gradient descent if QP fails
            h_final = h.copy()
            c5_bound = evaluate_c5(h_final, dx)
    except Exception:
        # Fallback to gradient descent if QP fails
        h_final = h.copy()
        c5_bound = evaluate_c5(h_final, dx)
    
    # Final validation
    if np.abs(np.sum(h_final) - target_sum) > 1e-5:
        h_final = h_final * (target_sum / np.sum(h_final))
    
    # Ensure bounds
    h_final = np.clip(h_final, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), float(final_c5), n_points)
