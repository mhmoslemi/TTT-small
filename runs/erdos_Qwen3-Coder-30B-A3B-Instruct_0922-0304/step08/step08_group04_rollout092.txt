import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Local search with random perturbations
    max_iter = 100000
    for i in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create perturbed version
        h_pert = h.copy()
        # Randomly change a few values
        n_changes = min(5, n_points // 10)
        change_indices = np.random.choice(n_points, n_changes, replace=False)
        for idx in change_indices:
            # Perturb by small amount
            pert = np.random.normal(0, 0.05)
            h_pert[idx] += pert
        # Ensure constraints
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = normalize_h(h_pert, n_points / 2)
        
        # Evaluate
        c5 = evaluate_c5(h_pert, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        # Occasionally restart with current best
        if i % 1000 == 0 and i > 0:
            h = best_h.copy()
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback to initial h if no improvement found
        best_h = h
        best_c5 = evaluate_c5(h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
