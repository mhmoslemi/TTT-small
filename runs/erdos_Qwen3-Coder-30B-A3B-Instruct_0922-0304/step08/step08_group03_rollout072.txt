import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, n_points):
    """Ensure h is feasible: in [0,1] and sums to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute gradient of overlap maximum w.r.t. h."""
    # This is an approximation based on the fact that C5 is dominated by some 
    # specific shifted correlations, but we compute a simplified gradient
    # for numerical reasons that should still guide descent.
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Optimization parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_gen = 2000
    no_improve_limit = 100
    no_improve_count = 0
    
    # For simplicity and robustness, we use a simpler genetic approach with 
    # limited diversity but more controlled search.
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            cand = h.copy()
        else:
            cand = np.random.rand(n_points)
        cand = repair_h(cand, n_points)
        population.append(cand)
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (smaller C5 is better)
        fitness = []
        for ind in population:
            c5 = evaluate_h(ind, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep top 50%
        selected = population[:pop_size//2]
        
        # Create new population
        new_population = selected[:]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two parents
            idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
            parent1, parent2 = selected[idx1], selected[idx2]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = repair_h(child1, n_points)
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.02)
                    child1[i] = np.clip(child1[i], 0, 1)
            child1 = repair_h(child1, n_points)
            
            child2 = repair_h(child2, n_points)
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.02)
                    child2[i] = np.clip(child2[i], 0, 1)
            child2 = repair_h(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population
        
        # Adaptive parameters
        if gen > 0 and gen % 100 == 0:
            if fitness[0] >= best_c5 and no_improve_count < no_improve_limit:
                no_improve_count += 1
                mutation_rate *= 1.1
            else:
                no_improve_count = 0
        else:
            no_improve_count = 0
            
        # Refinement step on best individual using simple gradient-based step
        # This might be helpful for staying in the local region around good solutions
        if gen % 10 == 0:
            refine_h = population[0].copy()
            loss = evaluate_h(refine_h, dx)
            grad = compute_gradient(refine_h, n_points)
            step_size = 0.01
            refine_h -= step_size * grad
            refine_h = repair_h(refine_h, n_points)
            new_loss = evaluate_h(refine_h, dx)
            if new_loss < loss:
                population[0] = refine_h
    
    # Final check
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
