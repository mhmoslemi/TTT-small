import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def convex_relaxation_initialization(h, n_points, dx):
    """Use convex relaxation to get a good starting point."""
    # Variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Bound constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints
    for k in range(2 * n_points - 1):
        # Compute the correlation constraint for shift k
        expr = 0
        for x in range(n_points):
            xk = x + k - (n_points - 1)
            if 0 <= xk < n_points:
                expr += h_var[x] * (1 - h_var[xk])
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return np.maximum(0, np.minimum(1, h_var.value))
    except:
        pass
    
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the C5 objective."""
    # Compute overlap values and find maximum index
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    grad = np.zeros_like(h)
    for i in range(n_points):
        total_grad = 0
        for k in range(2 * n_points - 1):
            # Indices for convolution
            pos1 = i
            pos2 = i + k - (n_points - 1)
            # Ensure they are within bounds
            if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                # Gradient of h_i * (1 - h_{i+k}) with respect to h_i
                # This is just (1 - h_{i+k}) but considering max_idx
                # Let's compute it differently for each term in the summation
                
                # Actually, we want to compute the gradient of max(overlap)
                # i.e., d/dr max_k \sum_{x} h_x (1 - h_{x+k})
                # We know that d/dr h_x (1 - h_{x+k}) = 
                # 1 if x is involved and we're differentiating w.r.t h_x, 
                # -1 if x+k is involved and we're diff w.r.t h_{x+k}
                # but this isn't directly like a sum over k that can be 
                # independently. Let's do it directly for the maximum overlap term.
                w = 0.0
                if pos1 == i:
                    # h_x = h[i]
                    w += (1 - h[pos2])
                if pos2 == i:
                    # h_{x+k} = h[i]  
                    w -= h[pos1]
                # But actually just considering the term nth of full convolution
                # Let's look at cite:
                # If overlap[idx] = h * (1-h) at shift k where k = idx - (npts-1):
                # then over all inputs i, for magnitude of that contribution to overlap[idx]  
                # to gradient of overlap w.r.t. h[i], we need to sum up over all 
                # valid k for which i is in their position
                # so for each shift k, the h[i]*h[i+k] term is part of the convolution.
                # But we are maximizing max_k over all i 
                # this is invalid - anyway back to the point:
                # what we need is derivative of h[i]*h[i+k] w.r.t. h[i] = h[i+k],  
                # and w.r.t. h[i+k] = h[i]. That'll be most of simulation.
                # Just look at each contribution of h[i] in one of overlapping terms.
                # pairwise for j where dot product h[j]=h[i] occurs, and (1-h[i+k]) for j+k
                # Direct computation not terribly wide.
                # Just compute stacked gradients and average. 
                # But this is very specific to overlapping, let's proceed more passively
                # or pseudo-gradient!
                
        # Simplified: for max overlap index, compute the influence of each h[i]
        # The influence of h[i] on the overlap term at index max_idx is:
        # (1 - h[i + shift]) + (1 - h[i - shift] if pos2 is in range)
        shift = max_idx - (n_points - 1)
        # Derivative terms: Diff of h[i]*(1-h[i+k]) w.r.t h[i] = (1-h[i+k])
        #        w.r.t h[i+k] = (-h[i])
        # Since we're computing for each i, we look at where its contribution 
        # happens. The approximate derivatives:
        # For computing criteria max of convolutions:
        # grad_i = sum over k:  (d/dh_i) (h_i (1-h_{i+k})) for all valid (i,k) s.t. it's part of that term.
        # = sum_{k; i at pos1 seq} (1 - h_{i+k}) + (sum over k weak where i+k at pos2) (-h_i)
        # = (1 - h[i+shift]) + (1 - h[i-shift]) or so. 
        # This estimate not very tight. Let's follow the original implementation for now.
                
        # For now, approximate the gradient by checking difference with a small perturbation
        # Skipped for now, stick with better one below:
        pass
    
    # Derivative of overlap over all k full convolution wrt h[i]
    grad = np.zeros_like(h)
    for i in range(n_points):
        for k in range(2 * n_points - 1):
            x = i
            xk = x + k - (n_points - 1)
            if 0 <= xk < n_points:
                # In the full convolution h*(1-h), the term 
                # h[i]*(1-h[i+k]) contributes to index k in convolution
                # Thus visualize what the derivative of that term wrt h[i] is:
                grad[i] += 1 - h[xk]
                # derivative of h[i]*h[i+k] wrt h[i] is h[i+k] 
                if xk != x:
                    grad[xk] -= h[x]
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=5000, lr=0.01, decay=0.999):
    """Optimize h using gradient descent."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        loss = evaluate_c5(h, dx)
        if loss < best_c5:
            best_c5 = loss
            best_h = h.copy()
            
        if loss <= 0.3807:
            break
            
        # Compute gradient using a better approach
        grad = np.zeros_like(h)
        h_conv = correlate(h, 1-h, mode='full')
        max_idx = np.argmax(h_conv)
        shift = max_idx - (n_points - 1)
        # Compute simpler gradient estimation:
        # For a given shift k = idx - (n_points-1)  
        # contribution in overlap = sum_i h[i] (1-h[i+k])
        # Gradient wrt h[i] = 1 - h[i+k], and wrt h[i+k] = -h[i]  
        # So we'm grade the contribution dependent on shift, and where:
        conv = np.zeros_like(h)
        for k in range(-n_points+1, n_points):
            for i in range(n_points):
                if 0 <= i+k < n_points:
                    conv[i] += (1 - h[i+k])  # derivative of h[i]*(1-h[i+k])
                    if k != 0:
                        conv[i+k] -= h[i]   # derivative of h[i]*(1-h[i+k]) w.r.t h[i+k]
        grad = conv
        
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Convex relaxation to get good starting point
    h = convex_relaxation_initialization(h, n_points, dx)
    
    # Gradient descent refinement
    best_h, best_c5 = optimize_with_gradient_descent(h, dx, n_points, 
                                                     max_iter=5000, lr=0.01, decay=0.999)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
