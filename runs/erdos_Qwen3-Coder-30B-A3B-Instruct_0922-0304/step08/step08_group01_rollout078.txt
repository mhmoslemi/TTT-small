import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function for minimization
    def objective(x):
        return compute_c5(x, dx)
    
    # Use SLSQP optimizer
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9})
    
    # If that fails, use a simpler approach
    if not result.success:
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) / 2
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
    
    best_h = result.x if result.success else h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Refine using gradient-based method if time allows
    if time.time() - start_time < budget_s - 1:
        # Further local refinement with a simple heuristic
        current_h = best_h.copy()
        last_c5 = best_c5
        improvement_threshold = 1e-6
        
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient approximation
            gradients = np.zeros(n_points)
            for i in range(n_points):
                grad = 0
                for k in range(n_points):
                    if i + k < n_points:
                        grad += (1 - 2 * current_h[i]) * (1 - current_h[i + k])
                    if i - k >= 0:
                        grad += (1 - 2 * current_h[i]) * (1 - current_h[i - k])
                gradients[i] = grad
            
            # Update using gradient
            step_size = 0.01
            new_h = current_h + step_size * gradients
            
            # Apply bounds
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize
            new_h = new_h * (target_sum / np.sum(new_h))
            
            # Check if this improves C5
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < last_c5:
                current_h = new_h
                last_c5 = new_c5
                best_h = current_h
                best_c5 = last_c5
            else:
                # If no improvement, try with reduced step size
                step_size *= 0.5
                current_h = current_h + step_size * gradients
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (target_sum / np.sum(current_h))
                temp_c5 = compute_c5(current_h, dx)
                if temp_c5 < last_c5:
                    last_c5 = temp_c5
                    best_h = current_h.copy()
                    best_c5 = last_c5
        
        # Final validation
        best_c5 = compute_c5(best_h, dx)
        if best_c5 > last_c5:
            best_c5 = last_c5
    
    # Ensure final result is valid
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if final_sum > 0:
        best_h = best_h * (target_sum / final_sum)
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
