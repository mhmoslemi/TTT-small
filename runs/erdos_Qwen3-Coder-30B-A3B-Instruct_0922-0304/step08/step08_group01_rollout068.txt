import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 50000  # Reduced iteration count to fit within time budget
    iter_count = 0
    
    # Precompute shift indices for efficiency
    shift_indices = []
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            shift_indices.append((k, np.arange(n_points - k)))
        else:
            shift_indices.append((k, np.arange(-k, n_points)))
    
    # Helper for computing overlap efficiently
    def compute_overlap(h):
        overlap = np.zeros(2 * n_points - 1)
        for k, idxs in shift_indices:
            if k >= 0:
                overlap[k:k + len(idxs)] += h[idxs] * (1 - h[idxs + k])
            else:
                overlap[k + len(idxs):k + len(idxs) + len(idxs)] += h[idxs + k] * (1 - h[idxs])
        return overlap
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current overlap
        overlap = compute_overlap(h)
        current_max = np.max(overlap)
        
        # Identify maximum indices
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        
        # Gradient approximation using finite differences
        gradients = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            
            if h_plus[i] > h[i]:
                overlap_plus = compute_overlap(h_plus)
                grad_plus = np.max(overlap_plus)
            else:
                grad_plus = current_max
            
            if h_minus[i] < h[i]:
                overlap_minus = compute_overlap(h_minus)
                grad_minus = np.max(overlap_minus)
            else:
                grad_minus = current_max
            
            gradients[i] = (grad_plus - grad_minus) / (2 * eps)
        
        # Update by selecting top gradient indices
        idxs = np.argsort(np.abs(gradients))[::-1]
        update_count = max(1, n_points // 10)
        selected_idxs = idxs[:update_count]
        
        # Perform updates with dynamic step size
        for i in selected_idxs:
            old_h_i = h[i]
            grad = gradients[i]
            
            # Use a more adaptive step size
            if abs(grad) > 1e-6:
                step_size = 0.1 * (1.0 / (abs(grad) + 1e-6))
            else:
                step_size = 0.01
            step_size = min(0.1, step_size)
            
            h[i] = h[i] + step_size * (1 - 2 * h[i]) * grad
            h[i] = np.clip(h[i], 0, 1)
            
            # Project to maintain sum constraint
            h = normalize_h(h, target_sum)
        
        # Check improvement
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
