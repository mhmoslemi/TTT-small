import numpy as np
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    # Compute overlap integral using convolution
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_h(h, n_points):
    """Generate a perturbed version of h."""
    # Select 10% of indices to perturb
    n_perturb = max(1, n_points // 10)
    indices = np.random.choice(n_points, n_perturb, replace=False)
    new_h = h.copy()
    
    # Apply combined perturbation: flip or swap
    for i in indices:
        if np.random.random() < 0.5:
            # Flip with small perturbation
            new_h[i] *= 0.97
        else:
            # Swap with random index
            j = np.random.randint(0, n_points)
            new_h[i], new_h[j] = new_h[j], new_h[i]
    
    # Clip and normalize
    new_h = np.clip(new_h, 0, 1)
    new_h = normalize_h(new_h, n_points / 2)
    
    return new_h

    h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = compute_c5_exact(h, dx)
    best_h = h.copy()

    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-6
    stagnant_steps = 0
    max_stagnant = 1000
    
    iteration = 0

        new_h = perturb_h(h, n_points)
        new_c5 = compute_c5_exact(new_h, dx)
        
        # Accept or reject based on simulated annealing
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / temperature):
            h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnant_steps = 0
            else:
                stagnant_steps += 1
        
        # Cool down
        temperature = max(min_temperature, temperature * cooling_rate)
        
        # Update iteration counter
        iteration += 1
