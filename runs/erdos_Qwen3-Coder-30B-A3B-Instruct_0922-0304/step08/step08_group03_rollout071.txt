import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute approximate gradient of C5."""
    # This is an approximation based on how each component affects overlap
    # For a rough gradient, we use: dC5/dh_i = (1 - 2*h_i) / n_points
    return (1 - 2 * h) / n_points

def gradient_descent_step(h, dx, step_size=0.01):
    """Perform gradient descent step on h to reduce C5."""
    overlap = correlate(h, 1 - h, mode="full") * dx
    current_max = np.max(overlap)
    
    grad = compute_gradient(h, len(h))
    
    # Try small steps in gradient direction
    h_new = h - step_size * grad
    h_new = repair_h(h_new, len(h) / 2)
    
    new_overlap = correlate(h_new, 1 - h_new, mode="full") * dx
    new_max = np.max(new_overlap)
    
    if new_max < current_max:
        return h_new
    else:
        # Accept small improvements even if not dramatic
        return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Parameters for the hybrid GA + local search
    population_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_generations = 2000
    
    # Population initialization
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
        else:
            ind = np.random.rand(n_points)
        ind = repair_h(ind, target_sum)
        population.append(ind)
    
    # Track improvement
    last_improvement_gen = 0
    no_improvement_count = 0
    
    # Main optimization loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
            
        # Sort individuals by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep top 50%
        top_count = population_size // 2
        selected = population[:top_count]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                child = np.zeros_like(parent1)
                mask = np.random.rand(len(parent1)) < 0.5
                child[mask] = parent1[mask]
                child[~mask] = parent2[~mask]
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                for i in mut_indices:
                    child[i] += np.random.normal(0, 0.02)
                child = repair_h(child, target_sum)

            # Add to new population
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # Adaptive parameters

            mutation_rate = min(0.3, mutation_rate * 1.05)
            last_improvement_gen = gen
        else:
            mutation_rate = max(0.02, mutation_rate * 0.995)
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
