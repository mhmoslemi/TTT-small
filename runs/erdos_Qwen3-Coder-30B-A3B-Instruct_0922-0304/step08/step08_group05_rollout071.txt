import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a simple structured function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid - n_points//4:mid + n_points//4] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    best_c5 = compute_c5_exact(h, dx)
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Simulated Annealing parameters
    T = 1.0        # Initial temperature
    cooling_rate = 0.995
    min_T = 1e-6   # Minimum temperature
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate candidate via perturbation
        candidate_h = h.copy()
        
        # Perturb 10% of points
        indices_to_change = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        
        for i in indices_to_change:
            # Flip or swap (random choice)
            if np.random.rand() < 0.5:
                # Flip: make it 0.97 of current value
                candidate_h[i] *= 0.97
            else:
                # Swap with a random element (small effect)
                j = np.random.randint(0, n_points)
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Ensure values are within bounds and re-normalize
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = normalize_h(candidate_h, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        # Accept or reject the candidate
        delta = candidate_c5 - best_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, min_T)):
            h = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
        
        # Cooling schedule
        T *= cooling_rate
        
        # Occasionally refine with local method
        if iteration % 50 == 0:  # Perform local refinement less frequently
            # Find maximum overlap lag
            overlap = correlate(h, 1 - h, mode="full")
            max_k = np.argmax(overlap)
            
            # Adjust locally
            # Try to reduce the max overlap by modifying nearby points
            for i in range(max(0, max_k - 5), min(n_points, max_k + 5)):
                if h[i] > 0.01:  # Skip near-zero values to avoid numerical issues
                    h[i] -= 0.001
                    h = normalize_h(h, target_sum)  # Re-normalize after change
                    new_c5 = compute_c5_exact(h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        last_improvement = 0
                    else:
                        h[i] += 0.001  # Revert if didn't improve
        
        iteration += 1
        if iteration % 1000 == 0:
            # Occasionally increase temperature if stuck
            T = max(T * 0.999, 1e-4)
    
    return (best_h.tolist(), best_c5, n_points)
