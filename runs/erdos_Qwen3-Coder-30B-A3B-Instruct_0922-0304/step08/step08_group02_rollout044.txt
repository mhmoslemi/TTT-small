import numpy as np
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1], scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h using correlation."""
    # Using correlate with mode="full" to compute cross-correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain."""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, surrogate, learning_rate=0.01):
    """Update the frequency domain representation to reduce the surrogate."""
    # Compute gradient-like update to reduce the surrogate
    # Simplified: just reduce contribution of high-frequency components
    # Note: This is a handcrafted heuristic for optimization
    grad = H * np.conj(1 - H)
    # Boost small magnitude components to reduce the impact
    # of high-frequency peaks
    H_new = H - learning_rate * grad * np.sign(H)
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h

        h_current = project_h(h_current, target_sum)
    else:

        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Training parameters
    learning_rate = 0.01
    decay = 0.99
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Fourier transform
        H = fft(h_current)
        
        # Compute surrogate in frequency domain
        surrogate = compute_surrogate(H)
        max_idx = np.argmax(surrogate)
        
        # Apply our custom spectral update
        H_new = spectral_update(H, surrogate, learning_rate)
        
        # Inverse FFT to get h back
        h_temp = np.real(ifft(H_new))
        
        # Project h_temp to feasible region
        h_temp = project_h(h_temp, target_sum)
        
        # Compute C5 for new h

        h_current = h_temp.copy()

            learning_rate *= decay
