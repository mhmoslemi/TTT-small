import numpy as np

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy the sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the C5 objective with respect to h."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Compute gradient based on the maximum overlap
    grad = np.zeros_like(h)
    
    # Loop through each position to compute partial derivative
    for i in range(n_points):
        # For position i, the contribution to max overlap comes from terms where 
        # h[i] * (1 - h[i + k]) contributes to the maximum overlap
        # We compute the partial derivative of max overlap with respect to h[i]
        grad[i] = 0.0
        # This is simplified: the actual derivative involves complex interactions
        # For now, use a simple heuristic approach
        for k in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - 2 * h[i]) * (1 - h[j])
    
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    return grad

def solve_linear_subproblem(grad, n_points, target_sum):
    """Solve the linear subproblem: minimize_grad^T v, subject to sum(v) = target_sum, 0 <= v <= 1."""
    # This is a simple LP: minimize sum(grad[i] * v[i]) subject to sum(v) = target_sum, 0 <= v[i] <= 1
    # We can solve it efficiently as follows:
    # For minimizing sum(grad_i * v_i) subject to sum(v) = target_sum and 0 <= v_i <= 1:
    # Sort gradient by values to identify variables to fix
    
    # Use CVXPY or a simpler method
    # Simple heuristic: set v = 0 for variables with negative gradient, 1 for positive
    v = np.zeros(n_points)
    for i in range(n_points):
        v[i] = 1 if grad[i] < 0 else 0
    
    # Adjust to meet the sum constraint by assigning remaining to elements with low gradient
    current_sum = np.sum(v)
    if current_sum < target_sum:
        # Need to increase sum
        diff = target_sum - current_sum
        # Increase v[i] where grad[i] is largest (most negative) and still 0
        indices = np.argsort(grad)
        for i in range(len(indices)):
            idx = indices[i]
            if v[idx] < 1:
                increase = min(diff, 1 - v[idx])
                v[idx] += increase
                diff -= increase
                if diff == 0:
                    break
    elif current_sum > target_sum:
        # Need to decrease sum
        diff = current_sum - target_sum
        # Decrease v[i] where grad[i] is smallest (least negative) and still 1
        indices = np.argsort(grad)[::-1]
        for i in range(len(indices)):
            idx = indices[i]
            if v[idx] > 0:
                decrease = min(diff, v[idx])
                v[idx] -= decrease
                diff -= decrease
                if diff == 0:
                    break
        
    return v

def line_search(h, v, grad, dt, n_points, dx):
    """Find a good step size using backtracking line search."""
    alpha = 1.0
    c1 = 1e-4
    rho = 0.5
    
    # Compute the current objective value
    current_c5 = evaluate_c5(h, dx)
    
    # Compute directional derivative
    dir_deriv = np.dot(grad, v - h)
    
    while alpha > 1e-10:
        # Try this alpha
        h_new = h + alpha * (v - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_h(h_new, n_points / 2)
        
        # Compute objective at new point
        new_c5 = evaluate_c5(h_new, dx)
        
        # Check Armijo condition
        if new_c5 <= current_c5 + c1 * alpha * dir_deriv:
            return alpha
        else:
            alpha *= rho
            
    return alpha

def frank_wolfe(h, dx, n_points, max_iter=10000, tolerance=1e-8):
    """Frank-Wolfe algorithm for optimizing h."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Self-tuning Frank-Wolfe
    for i in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Solve linear subproblem
        v = solve_linear_subproblem(grad, n_points, n_points / 2)
        
        # Line search
        alpha = line_search(h, v, grad, dx, n_points, dx)
        
        # Update
        h = (1 - alpha) * h + alpha * v
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
        
        # Check solution quality
        new_c5 = evaluate_c5(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        if abs(new_c5 - best_c5) < tolerance:
            break
        
        # Minor feedback
        if i % 100 == 0:

                break
    
    return best_h, best_c5
