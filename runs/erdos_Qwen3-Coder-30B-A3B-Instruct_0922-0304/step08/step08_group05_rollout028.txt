import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2):
    """Average-based crossover with normalization."""
    child = (parent1 + parent2) / 2.0
    return child

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Apply Gaussian mutation with constraint repair."""
    mutated = individual + np.random.normal(0, sigma, len(individual)) * mutation_rate
    mutated = np.clip(mutated, 0, 1)
    return mutated

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    idx = np.random.choice(len(population), size=k, replace=False)
    best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    sigma = 0.1
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    population.append(h.copy())
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Main evolution loop
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5_exact(h, dx)
            fitnesses.append(c5)
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection and reproduction
        new_population = []
        
        # Keep best individuals
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size//4]
        for idx in top_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses, k=5)
            parent2 = tournament_selection(population, fitnesses, k=5)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child, mutation_rate, sigma)
            
            # Constraint repair
            child = np.clip(child, 0, 1)
            child = normalize_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Return final result
    if best_h is None:
        # Fallback to last population if no improvement
        best_h = population[0]
        best_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
