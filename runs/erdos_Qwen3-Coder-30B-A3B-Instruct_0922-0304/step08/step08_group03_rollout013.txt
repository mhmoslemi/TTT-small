import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    h = np.minimum(x, 2 - x)
    return h / np.sum(h) * (n / 2)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: sum = target_sum, values in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_overlap_matrix(h):
    """Compute the full overlap matrix for faster gradient computation."""
    n = len(h)
    # Precompute all shifts
    shifts = []
    for k in range(2 * n - 1):
        if k < n:
            shift = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            shift = np.concatenate([h[k-n+1:], np.zeros(n-k+n-1)])
        shifts.append(shift)
    
    # Compute overlaps for all shifts
    overlaps = []
    for shift in shifts:
        overlap = np.sum(h * (1 - shift))
        overlaps.append(overlap)
    
    return np.array(overlaps)

def compute_gradients(h, dx):
    """Compute gradient of max overlap with respect to h."""
    n = len(h)
    # Compute overlap for each shift
    overlaps = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        shift = np.zeros(n)
        if k < n:
            shift[k:] = h[:n-k]
        else:
            shift[:n-(k-n+1)] = h[k-n+1:]
        overlaps[k] = np.sum(h * (1 - shift))
    
    # Find maximum overlap index
    max_idx = np.argmax(overlaps)
    max_val = overlaps[max_idx]
    
    # Approximate gradient at maximum overlap point
    grad = np.zeros(n)
    
    # Finite difference to compute approximate gradient
    epsilon = 1e-5
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = min(1, h[i] + epsilon)
        h_plus = project_h(h_plus, np.sum(h))
        
        h_minus = h.copy()
        h_minus[i] = max(0, h[i] - epsilon)
        h_minus = project_h(h_minus, np.sum(h))
        
        # Recompute overlaps with new h_plus and h_minus
        overlap_plus = np.zeros(2*n - 1)
        overlap_minus = np.zeros(2*n - 1)
        
        for k in range(2*n - 1):
            shift_plus = np.zeros(n)
            shift_minus = np.zeros(n)
            if k < n:
                shift_plus[k:] = h_plus[:n-k]
                shift_minus[k:] = h_minus[:n-k]
            else:
                shift_plus[:n-(k-n+1)] = h_plus[k-n+1:]
                shift_minus[:n-(k-n+1)] = h_minus[k-n+1:]
            overlap_plus[k] = np.sum(h_plus * (1 - shift_plus))
            overlap_minus[k] = np.sum(h_minus * (1 - shift_minus))
            
        grad[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * epsilon)
        
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():

        if not (np.isclose(np.sum(h), target_sum) and np.all((h >= 0) & (h <= 1))):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Early exit if already good enough
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Optimization loop
    n_iter = 0
    max_iter = 100000
    alpha = 0.001  # Learning rate
    momentum = 0.9
    v = np.zeros_like(h)
    
    while n_iter < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current C5 and gradients
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
        
        # Compute gradient using finite differences for smoothing
        grad = compute_gradients(h, dx)
        
        # Update momentum and step
        v = momentum * v + alpha * grad
        h = h - v
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Occasionally adjust learning rate
        if n_iter % 1000 == 0 and alpha > 0.0001:
            alpha *= 0.99
        
        n_iter += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
