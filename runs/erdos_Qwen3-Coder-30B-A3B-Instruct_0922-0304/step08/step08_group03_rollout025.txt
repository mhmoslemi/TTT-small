```python
import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    mid = n // 2
    left = np.linspace(0, 1, mid)
    right = np.linspace(1, 0, n - mid)
    return np.append(left, right)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if np.sum(h) < target_sum * 0.99 or np.sum(h) > target_sum * 1.01:
            h = triangular_ramp(n_points)
        else:
            h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    # Momentum and parameters
    velocity = np.zeros(n_points)
    alpha = 0.9  # momentum
    beta = 0.001  # learning rate
    gamma = 0.5  # shrink factor
    max_iter = 20000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        overlap_max_idx = np.argmax(overlap)
        
        # Compute exact gradient
        # ∇C₅ = sign(overlap_max) * (1 - 2h) / n_points
        grad = (1 - 2 * h) * np.sign(overlap_max) / n_points
        
        # Momentum update
        velocity = alpha * velocity + beta * grad
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Shrink step size and retry
            beta *= gamma
            if no_improvement_count > 100:
                # Reset with a new random perturbation if no progress
                h = clip_and_renormalize(h + np.random.normal(0, 0.001, n_points), target_sum)
                no_improvement_count = 0
                beta = 0.001  # Reset learning rate
        
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
