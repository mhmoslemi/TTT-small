import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient descent parameters
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    max_iter = int(1e5)
    
    # For Nesterov momentum
    z = h.copy()
    v = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute overlap values for all shifts
        overlaps = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k] * dx
        
        # Compute gradient
        # For each h[i], gradient is sum of (1-h[j]) where j = (i-k)%n and k is the maximizing shift
        grad = np.zeros_like(h)
        # Since we have a full correlation, we can work with the fact that
        # C_k = sum_i h_i * (1 - h_{i+k})
        # dC_k/dh_i = 1 - h_{i+k}
        # We focus on the maximum C_k, so find its gradient
        k_max = max_k - (n_points - 1)  # Map full correlation index to actual lag

        # Compute actual gradient with respect to the maximum overlap
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = 1 - h[j]
        
        # Nesterov momentum update
        v = momentum * v + grad
        z = h - step_size * v
        
        # Apply projection to [0,1]
        z = np.clip(z, 0, 1)
        
        # Rescale to satisfy the constraint
        current_sum = np.sum(z)
        if current_sum > 0:
            z = z * (target_sum / current_sum)
        
        # Update h
        h = z
        
        # Evaluate
        new_c5 = evaluate_overlap(h, dx)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            step_size = min(0.5, step_size * 1.1)  # Increase step size if improving
            if step_size > 0.5:
                step_size = 0.5
        else:
            step_size = max(min_step_size, step_size * 0.9)  # Decrease step size if not improving
            
        # Check for convergence
        if abs(new_c5 - best_c5) < tol:
            break
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
