import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def feasibility_repair(h, target_sum):
    """Clip h to [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def crossover(parent1, parent2, rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, noise_magnitude):
    """Mutate an individual."""
    noise = np.random.normal(0, noise_magnitude, len(h))
    h = h + noise
    return h

def tournament_selection(population, fitness, k=2):
    """Select an individual using tournament selection."""
    indices = random.sample(range(len(population)), k)
    best_idx = min(indices, key=lambda i: fitness[i])
    return population[best_idx].copy()

def genetic_algorithm(initial_h, n_points, dx, max_generations=1000, 
                      population_size=50, crossover_rate=0.8, mutation_rate=0.3,
                      target_sum=35.0, noise_magnitude=0.05):
    """Run genetic algorithm to optimize the Erdős minimum overlap problem."""
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        individual = mutate(initial_h.copy(), noise_magnitude)
        individual = feasibility_repair(individual, target_sum)
        population.append(individual)
    
    best_individual = initial_h.copy()
    best_c5 = evaluate_c5(best_individual, dx)
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitness = Parallel(n_jobs=1)(delayed(evaluate_c5)(ind, dx) for ind in population)
        
        # Track best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_individual = population[best_idx].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(best_individual.copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            if random.random() < mutation_rate:
                child1 = mutate(child1, noise_magnitude)
            if random.random() < mutation_rate:
                child2 = mutate(child2, noise_magnitude)
                
            child1 = feasibility_repair(child1, target_sum)
            child2 = feasibility_repair(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Reduce noise magnitude
        noise_magnitude *= 0.99
        if noise_magnitude < 0.001:
            noise_magnitude = 0.001
    
    return best_individual, best_c5

def run(seed=42, budget_s=600, **kwargs):

    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(h, n_points, dx, max_generations=1000,
                                       population_size=50, crossover_rate=0.8,
                                       mutation_rate=0.3, target_sum=target_sum,
                                       noise_magnitude=0.05)
    
    # Ensure we have exact evaluation of the best candidate
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
