from scipy.signal import correlate
import time

    h = h * (target_sum / np.sum(h))

    # Normalize again to be exactly correct
    h = h * (target_sum / np.sum(h))
    
    # Optimization parameters
    max_iter = 100000
    epsilon = 0.01
    decay_rate = 0.99995
    current_epsilon = epsilon
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h, dx):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
        
    def adaptive_adjustment(h, dx, epsilon, target_sum):
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Get the maximum overlap value
        max_val = overlap[max_idx]
        
        # Find the lag that produces maximum overlap
        lag = max_idx - (len(h) - 1)
        
        # Create a small adjustment
        h_new = h.copy()
        
        # We'll adjust h to reduce the maximum overlap at this lag
        # Try adjusting at multiple points to reduce overlap
        n = len(h)
        adjustment = current_epsilon
        
        # Determine indices that contribute to this lag
        # The overlap at lag k = sum_i h[i] * (1 - h[i+k])
        # This is effectively adjusting h[i] and h[i+k] to minimize the contribution
        # We choose to locally adjust where the maximum overlap occurs
        
        # Pick a few positions to adjust
        indices_to_adjust = []
        for i in range(n):
            if 0 <= i + lag < n:
                indices_to_adjust.append(i)

        for i in indices_to_adjust[:3]:

            h_new[i] = np.clip(h_new[i] - delta, 0, 1)

        # Re-normalize
        h_new = h_new * (target_sum / np.sum(h_new))
        
        return h_new
    
    iteration = 0

        current_c5 = compute_c5(h, dx)

        iteration += 1
    
    # Finalize with actual C5 calculation
    final_c5 = compute_c5(best_h, dx)
