import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_individual(n_points, target_sum):
    """Create a valid individual for the population."""
    # Start with random values
    h = np.random.rand(n_points)
    # Apply slight gaussian perturbation to get smoother values
    h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
    h = normalize_h(h, target_sum)
    return h

def crossover(parent1, parent2, target_sum):
    """Create offspring from two parents."""
    # Simple average crossover
    offspring = (parent1 + parent2) / 2.0
    # Ensure it's valid
    offspring = np.clip(offspring, 0, 1)
    offspring = normalize_h(offspring, target_sum)
    return offspring

def mutate(individual, mutation_rate, target_sum):
    """Mutate an individual."""
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_rate, len(individual))
    mutated = individual + noise
    # Clip and renormalize
    mutated = np.clip(mutated, 0, 1)
    mutated = normalize_h(mutated, target_sum)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.05
    elite_size = population_size // 5
    
    # Initialize population
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = create_individual(n_points, target_sum)
    population.append(h)
    
    # Fill rest of population
    for _ in range(population_size - 1):
        h = create_individual(n_points, target_sum)
        population.append(h)
    
    # Evolution loop
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for ind in population:
            c5 = compute_c5_exact(ind, dx)
            fitnesses.append(c5)
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(len(sorted_population), tournament_size, replace=False)
            tournament_fitnesses = [sorted_fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            
            # Select second parent (not the same)
            second_idx = np.random.choice(len(sorted_population))
            while second_idx == winner_idx:
                second_idx = np.random.choice(len(sorted_population))
            
            # Crossover
            offspring = crossover(sorted_population[winner_idx], sorted_population[second_idx], target_sum)
            
            # Mutation
            if np.random.random() < 0.8:  # 80% mutation rate
                offspring = mutate(offspring, mutation_rate, target_sum)
            
            new_population.append(offspring)
        
        population = new_population[:population_size]
        
        # Adaptive mutation rate
        if gen % 100 == 0 and gen > 0:
            mutation_rate *= 0.95
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5_exact(best_h, dx)
    else:
        final_c5 = float('inf')
        best_h = population[0]
        
    return (best_h.tolist(), final_c5, n_points)
