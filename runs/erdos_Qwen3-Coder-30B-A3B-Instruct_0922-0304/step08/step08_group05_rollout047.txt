import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_solution(h, n_points, epsilon=0.03):
    """Generate a perturbed solution."""
    h_new = h.copy()
    
    # Randomly decide to flip or swap
    if np.random.random() < 0.5:
        # Flip some elements
        flip_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        for i in flip_indices:
            h_new[i] = 1 - h_new[i]
    else:
        # Swap two elements
        i, j = np.random.choice(n_points, size=2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
    
    # Add small noise
    noise = np.random.normal(0, epsilon, n_points)
    h_new += noise
    
    # Clip and normalize
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, n_points / 2)
    
    return h_new

def cutting_plane_step(h, dx, target_sum):
    """Perform a cutting plane step to reduce overlap."""
    n_points = len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Approximate the gradient by looking at the influence of each point
    # on the maximum overlap
    h_new = h.copy()
    max_lag = max_idx - (n_points - 1)  # Convert full index to lag
    
    # Reduce overlap by modifying the relevant elements
    # Modify one element with the strongest impact
    impact = np.zeros(n_points)
    
    # Estimate the gradient based on the derivative of the convolution
    # This is a rough approximation, but sufficient for our purposes
    for i in range(n_points):
        # Find the closest indices that contribute to overlap at max_lag
        for j in range(n_points):
            if abs((j - i) - max_lag) <= 1:  # Approximate impact
                # For simplicity: reduce by affecting the contribution
                impact[i] += (1 - 2 * h[i]) * (1 - 2 * h[j])
    
    # Make improvements based on gradient sign
    step_size = 0.02
    positive_impact = np.where(impact > 0)[0]
    negative_impact = np.where(impact < 0)[0]
    
    if len(positive_impact) > 0:
        i = positive_impact[0]
        h_new[i] = np.clip(h_new[i] - step_size, 0, 1)
    if len(negative_impact) > 0:
        i = negative_impact[0] 
        h_new[i] = np.clip(h_new[i] + step_size, 0, 1)
    
    h_new = normalize_h(h_new, target_sum)
    return h_new

        h += noise

    h = normalize_h(h, target_sum)
    
    # Tracking variables
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    current_h = h.copy()
    
    iteration = 0
    T = 1.0  # Initial temperature
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Perform a simulated annealing step
        candidate_h = perturb_solution(current_h, n_points)
        current_c5 = compute_c5_exact(current_h, dx)
