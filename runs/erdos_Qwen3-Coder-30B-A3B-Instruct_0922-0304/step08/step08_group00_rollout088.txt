import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The peak value is the maximum overlap
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 100
    improvement_threshold = 1e-6
    perturbation_scale = 1e-3
    decay_factor = 0.98
    slsqp_tolerance = 1e-10
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple uniform distribution with some variation
        h = np.ones(n_points) * 0.5
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Perturb and partition into even and odd indexed elements
    def perturb_and_partition(h, delta):
        perturbed = h + np.random.uniform(-delta, delta, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        h_sum = np.sum(perturbed)
        if h_sum > 0:
            perturbed = perturbed * (target_sum / h_sum)
        even_idx = np.arange(0, n_points, 2)
        odd_idx = np.arange(1, n_points, 2)
        return perturbed, even_idx, odd_idx
    
    # Alternate optimization between even and odd indices
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb current solution to escape local minima
        h, even_idx, odd_idx = perturb_and_partition(h, perturbation_scale)
        
        # Optimize even variables while fixing odd
        def optimize_even():
            def objective(even_vals):
                h_temp = h.copy()
                h_temp[even_idx] = even_vals
                return evaluate_overlap(h_temp, dx)
            
            # Initial guess from previous even values, or even indices of h
            init_guess = h[even_idx].copy()
            bounds = [(0, 1) for _ in init_guess]
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - np.sum(h[even_idx])}]
            
            result = minimize(objective, init_guess, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'ftol': slsqp_tolerance, 'maxiter': 100})
            
            if result.success:
                new_even = np.clip(result.x, 0, 1)
                h[even_idx] = new_even
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (target_sum / h_sum)
            return h
            
        h = optimize_even()
        current_c5 = evaluate_overlap(h, dx)
        
        # Optimize odd variables while fixing even
        def optimize_odd():
            def objective(odd_vals):
                h_temp = h.copy()
                h_temp[odd_idx] = odd_vals
                return evaluate_overlap(h_temp, dx)
                
            init_guess = h[odd_idx].copy()
            bounds = [(0, 1) for _ in init_guess]
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - np.sum(h[odd_idx])}]
            
            result = minimize(objective, init_guess, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'ftol': slsqp_tolerance, 'maxiter': 100})
            
            if result.success:
                new_odd = np.clip(result.x, 0, 1)
                h[odd_idx] = new_odd
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (target_sum / h_sum)
            return h
            
        h = optimize_odd()
        current_c5 = evaluate_overlap(h, dx)
        
        # Update best solution if improvement
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            perturbation_scale = max(1e-6, perturbation_scale * decay_factor)
        
        # If improvement less than threshold, decrease perturbation
        if iteration > 0 and abs(best_c5 - current_c5) < improvement_threshold:
            perturbation_scale = max(1e-6, perturbation_scale * 0.95)
            
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
