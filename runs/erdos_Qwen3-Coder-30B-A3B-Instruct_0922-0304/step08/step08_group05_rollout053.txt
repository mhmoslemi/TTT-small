import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population parameters
    population_size = 50
    generations = 1000
    tournament_size = max(2, population_size // 10)
    
    # Initialize population
    population = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Fill rest of population with random valid individuals
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Keep track of best solution
    best_h = None
    best_c5 = float('inf')
    last_improvement = 0
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of all individuals
        fitness = []
        for h in population:
            c5 = compute_c5_exact(h, dx)
            fitness.append(c5)
            
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Selection: tournament selection
        new_population = []
        
        # Keep best individual (elitism)
        new_population.append(population[min_fitness_idx])
        
        # Tournament selection for rest
        while len(new_population) < population_size:
            # Tournament
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
            
        # Crossover: average top 30% of population
        top_indices = np.argsort(fitness)[:population_size // 3]
        if len(top_indices) >= 2:
            num_offspring = population_size - len(new_population)
            for _ in range(num_offspring):
                parent1, parent2 = population[top_indices[np.random.randint(len(top_indices))]], \
                                   population[top_indices[np.random.randint(len(top_indices))]]
                offspring = (parent1 + parent2) / 2
                offspring = normalize_h(offspring, target_sum)
                offspring = np.clip(offspring, 0, 1)
                offspring = normalize_h(offspring, target_sum)
                new_population.append(offspring)
        
        # Mutation: add small Gaussian noise
        mutation_rate = 0.1
        mutation_strength = 0.05
        
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                new_population[i] = new_population[i] + noise
                new_population[i] = np.clip(new_population[i], 0, 1)
                new_population[i] = normalize_h(new_population[i], target_sum)
        
        population = new_population
        
        if last_improvement > 1000:
            break
    
    # Final evaluation of best solution
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
