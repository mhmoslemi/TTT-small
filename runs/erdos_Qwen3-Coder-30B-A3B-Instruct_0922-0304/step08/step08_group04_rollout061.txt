import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def solve_convex_relaxation(n_points, dx):
    """Solve convex relaxation to get initial guess."""
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [0 <= h, h <= 1]
    
    # Add the overlap constraints for all shifts
    for k in range(2 * n_points - 1):
        # Calculate the indices
        if k < n_points:
            # Forward shifts
            indices1 = list(range(n_points - k))
            indices2 = list(range(k, n_points))
        else:
            # Backward shifts
            indices1 = list(range(k - n_points + 1, n_points))
            indices2 = list(range(n_points - (k - n_points + 1)))
        
        # Constraint: h_x * (1 - h_{x+k}) <= t * dx for all valid pairs
        if len(indices1) > 0:
            for i in range(len(indices1)):
                if indices2[i] < n_points:
                    constraints.append(h[indices1[i]] * (1 - h[indices2[i]]) <= t * dx)
    
    constraints.append(sum(h) == n_points / 2)
    
    # Objective
    objective = Minimize(t)
    
    # Problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return np.array(h.value).flatten()
    except:
        pass
    
    return None

def project_onto_feasible(h, target_sum):
    """Project h onto feasible set."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the maximum overlap."""
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    
    grad = np.zeros(n_points)
    
    # Simplified gradient computation based on worst-case shift k
    for i in range(n_points):
        # Using direct method for gradient of the projection of the overlap onto the worst case
        # This is not exact but a reasonable approximation
        # The analytical derivative would involve complex indexing
        # Here we approximate by finite differences
        trunc_factor = 1e-8
        
        h_pert = h.copy()
        h_pert[i] = min(1.0, h[i] + trunc_factor)
        h_pert = project_onto_feasible(h_pert, n_points / 2)
        
        h_neg = h.copy()
        h_neg[i] = max(0.0, h[i] - trunc_factor)
        h_neg = project_onto_feasible(h_neg, n_points / 2)
        
        val_pert = evaluate_c5(h_pert, dx)
        val_neg = evaluate_c5(h_neg, dx)
        
        grad[i] = (val_pert - val_neg) / (2 * trunc_factor)
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999, tolerance=1e-8):
    """Optimize using gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        loss = evaluate_c5(h, dx)
        if loss < best_c5:
            best_c5 = loss
            best_h = h.copy()
        
        # Check termination conditions
        if loss <= 0.3807:
            break
            
        if abs(prev_loss - loss) < tolerance:
            no_improvement_count += 1
            if no_improvement_count > 1000:
                break
        else:
            no_improvement_count = 0
        prev_loss = loss
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = project_onto_feasible(h, n_points / 2)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_onto_feasible(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = project_onto_feasible(h, n_points / 2)
    
    # Try to do convex relaxation first if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 100:
        relaxed_h = solve_convex_relaxation(n_points, dx)
        if relaxed_h is not None:
            h = relaxed_h
    
    # Optimize the function using gradient descent
    best_h, best_c5 = optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999)
    
    return (best_h.tolist(), best_c5, n_points)
