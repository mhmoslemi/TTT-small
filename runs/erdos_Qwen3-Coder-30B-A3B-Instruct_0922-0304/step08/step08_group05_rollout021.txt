import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_solution(h, n_points):
    """Generate a neighbor solution by slightly perturbing h."""
    new_h = h.copy()
    # Choose 10% of indices to modify
    indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
    for i in indices:
        # 50% chance to flip (multiply by 0.97) or 50% to swap with another position
        if np.random.random() < 0.5:
            new_h[i] *= 0.97
        else:
            # Swap with a random position
            j = np.random.randint(0, n_points)
            new_h[i], new_h[j] = new_h[j], new_h[i]
    
    # Clip to [0,1]
    new_h = np.clip(new_h, 0, 1)
    # Normalize to sum = n_points / 2
    target_sum = n_points / 2
    new_h = normalize_h(new_h, target_sum)
    
    return new_h

def cutting_plane_refinement(h, dx, target_sum, max_iter=100):
    """Refine solution using cutting plane method focused on worst overlap."""
    h = h.copy()
    for _ in range(max_iter):
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k = max_idx - (len(h) - 1)  # Convert index to lag
        
        # Determine significant positions in the worst overlap
        if k >= 0:
            # h[i] * (1 - h[i+k]) contributes significantly
            for i in range(len(h) - k):
                if h[i] > 0 and h[i+k] < 1:
                    # Slightly reduce h[i] and increase h[i+k] if possible
                    epsilon = min(h[i] * 0.01, (1 - h[i+k]) * 0.01)
                    if epsilon > 1e-8:
                        h[i] -= epsilon
                        h[i+k] += epsilon
        else:
            # h[i+k] * (1 - h[i]) contributes significantly
            for i in range(-k, len(h)):
                if h[i] > 0 and h[i+k] < 1:
                    epsilon = min(h[i] * 0.01, (1 - h[i+k]) * 0.01)
                    if epsilon > 1e-8:
                        h[i] -= epsilon
                        h[i+k] += epsilon
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    return h

    n_points = 70

    else:

        h[n_points//4:3*n_points//4] = 1.0
        # Add a bit of noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize temperature
    T = 1.0
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Simulated Annealing Phase
        new_h = perturb_solution(h, n_points)
        current_c5 = compute_c5_exact(h, dx)
        new_c5 = compute_c5_exact(new_h, dx)
        
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
            h = new_h
            if new_c5 < current_c5:
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Cutting Plane Refinement
        if iteration % 100 == 0 and iteration > 0:
            h = cutting_plane_refinement(h, dx, target_sum)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Cooling
        T *= 0.999
        
        iteration += 1
    
    # Final refinement
    h = cutting_plane_refinement(best_h, dx, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    final_c5 = compute_c5_exact(h, dx)
    
    return (h.tolist(), final_c5, n_points)
