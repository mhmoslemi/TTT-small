import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, dx):
    """Compute approximate gradient for C5."""
    # We estimate the gradient for each point with finite differences
    n_points = len(h)
    grad = np.zeros(n_points)
    eps = 1e-4
    base_c5 = compute_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + eps)
        h_plus = repair_h(h_plus, n_points / 2)
        c5_plus = compute_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - eps)
        h_minus = repair_h(h_minus, n_points / 2)
        c5_minus = compute_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def gradient_descent_step(h, grad, dx, step_size=0.1):
    """Perform one gradient descent step."""
    h_new = h - step_size * grad
    h_new = repair_h(h_new, len(h) / 2)
    return h_new

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # If we already have a good value, return it

        return (best_h.tolist(), best_c5, n_points)

    population_size = 50
    generations = 2000
    crossover_prob = 0.7
    mutation_rate = 0.1
    stagnation_limit = 100

    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
            ind = repair_h(ind, target_sum)
        else:
            ind = np.random.rand(n_points)
            ind = repair_h(ind, target_sum)
        population.append(ind)

    best_in_generation = []
    no_improvement_count = 0
    current_best = best_c5
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for i, individual in enumerate(population):
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
