import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with square wave pattern and add noise
        h = np.zeros(n_points)
        # Create a square wave pattern
        mid = n_points // 2
        h[mid//2:3*mid//2] = 1.0
        # Add some noise and normalize
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    temp = 1.0
    cooling_rate = 0.9995
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 2000:
        # Simulated Annealing Phase
        # Create candidate by perturbing current solution
        candidate = h.copy()
        
        # Perturb about 10% of indices
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Randomly scale and clip
            candidate[i] *= np.random.uniform(0.9, 1.1)
            # Occasionally swap with neighbor
            if np.random.random() < 0.2 and i + 1 < n_points:
                candidate[i], candidate[i+1] = candidate[i+1], candidate[i]
        
        # Ensure bounds
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        
        # Compute current and candidate C5
        current_c5 = compute_c5_exact(h, dx)
        candidate_c5 = compute_c5_exact(candidate, dx)
        
        # Accept with probability based on simulated annealing
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-abs(candidate_c5 - current_c5) / temp):
            h = candidate.copy()
            current_c5 = candidate_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Cooling schedule
        temp *= cooling_rate
        
        # Occasionally refine using cutting-plane style approach
        if iteration % 100 == 0 and iteration > 0:
            # Refine best solution using local optimization
            overlap = correlate(best_h, 1 - best_h, mode="full")
            max_idx = np.argmax(overlap)
            # Try to adjust around the worst overlap
            for i in range(max(0, max_idx - 10), min(len(overlap), max_idx + 10)):
                if i < n_points:
                    # Apply small adjustment
                    h_adj = best_h.copy()
                    h_adj[i] = min(1.0, max(0.0, h_adj[i] + 0.01))
                    h_adj = normalize_h(h_adj, target_sum)
                    c5_adj = compute_c5_exact(h_adj, dx)
                    if c5_adj < best_c5:
                        best_h = h_adj.copy()
                        best_c5 = c5_adj
                        
        iteration += 1
    
    # Final recomputation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
