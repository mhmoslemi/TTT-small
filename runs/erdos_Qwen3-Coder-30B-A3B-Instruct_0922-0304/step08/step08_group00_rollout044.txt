import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:
        # Create a base step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add small random perturbation
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Current best solution
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient-based optimization with Nesterov momentum
    max_iter = 100000
    lr = 0.1
    momentum = 0.9
    min_lr = 1e-5
    tol = 1e-6
    patience = 1000
    
    # Nesterov momentum variables
    v = np.zeros_like(h)
    y = h.copy()
    
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all shifts to find maximum
        overlap = correlate(y, 1 - y, mode="full")
        max_shift = np.argmax(overlap)
        max_overlap = overlap[max_shift] * dx
        
        # Compute gradients at the point of maximum overlap
        # For shift k that maximizes overlap, the gradient of C_k with respect to h_i is:
        # dC_k/dh_i = 1 - h_{i-k} (when we consider cyclic indexing)
        # We need sum over all i of h_i * (1 - h_{i-k}) = (1 - h_{i-k}) * h_i
        # Hence gradient at h_i = sum over all such terms (for the max k)
        
        # Use the shift k that maximizes the correlation from current y
        grad = np.zeros_like(y)
        # Add gradient contribution from each term in maximum overlap
        # From correlation calculation, for shift k: d/dh_i of h_i*(1-h_{i+k}) = (1-h_{i+k})
        # We compute this in effect by convolving with -1 on the other side
        # Actually direct gradient computation
        # For each i, we look at the term in maximum overlap,
        # it's a sum including h_i(1-h_{i+k}) for some k
        # Differentiating wrt h_i we get (1 - h_{i+k})
        # Let's compute influence from maximum overlap index
        # Simply using the k from correlation computation
        shifted_y = np.roll(y, -max_shift)
        grad = 1 - shifted_y
        
        # Apply gradient with momentum
        v = momentum * v - lr * grad
        y = y + v
        
        # Project to feasible set: [0,1] and sum = target_sum
        y = np.clip(y, 0, 1)
        current_sum = np.sum(y)
        if current_sum > 0:
            y = y * (target_sum / current_sum)
        
        # Evaluate objective
        new_c5 = evaluate_overlap(y, dx)
        
        if new_c5 < best_c5:
            best_h = y.copy()
            best_c5 = new_c5
            last_improvement = iteration
        elif iteration - last_improvement > patience:
            # Reduce learning rate if no improvement for some time
            lr = max(lr * 0.5, min_lr)
            last_improvement = iteration
            
            # Restart from best state with new learning rate
            y = best_h.copy()
    
    # Final refinement with scipy minimize to ensure feasibility
    from scipy.optimize import minimize
    
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 100, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    else:
        # Fallback to best solution if optimization failed
        pass
    
    return (best_h.tolist(), best_c5, n_points)
