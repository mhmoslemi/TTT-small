import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * (target_sum / np.sum(h))
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient vector for given shift."""
    gradient = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            gradient[i] = 1 - h[j]
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2 / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap to get the shift causing maximum overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        # The shift from the full correlation output that gives max value
        shift = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        gradient = compute_gradient(h_current, shift, n_points)
        
        # Update h using projected gradient descent
        h_new = h_current - step_size * gradient
        
        # Project h to feasible region
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current solution
        h_current = h_new.copy()
        
        # Adjust step size if needed
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.5
            
        # Periodically perturb to escape local minima
        if no_improvement_count > 50 and iteration % 10 == 0:
            perturb_mag = 0.02
            for i in range(5):  # Perturb 5 random indices
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-perturb_mag, perturb_mag)
                h_current[idx] = np.clip(h_current[idx] + delta, 0, 1)
            h_current = project_h(h_current, n_points)
        
        iteration += 1
        
        # Early stop if we've hit target
        if best_c5 <= 0.3807:
            break
    
    # Final optimization: try one more local search
    h_final = best_h.copy()
    for _ in range(30):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_final.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_final, dx):
            h_final = h_temp.copy()
    
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
