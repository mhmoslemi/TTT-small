import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum of these correlations scaled by dx
    return np.max(correlation) * dx

    if current_sum > 0:

    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)

    bounds = [(0, 1) for _ in range(n_points)]

    # Run the optimization
    result = minimize(objective, h, method='SLSQP',
