import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Use CVXPY to solve the convex optimization problem
    # Variables
    h_vars = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == target_sum
    ]
    
    # Maximize the overlap by minimizing negative of it
    # The issue with direct convex modeling of max is complexity
    # So, we approximate using a large number of shifted versions
    
    # For a more direct numerical approach, use scipy optimization with better 
    # gradient information, but we can also slightly refine the initial solution
    
    # Local refinement using SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints_slsqp = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Allow more iterations in local search
    result = minimize(objective, h, method='SLSQP', 
                     bounds=bounds, constraints=constraints_slsqp, 
                     options={'maxiter': 500, 'ftol': 1e-10})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx) 
    else:
        refined_h = h.copy()
        final_c5 = evaluate_overlap(h, dx)
    
    # Second-stage local optimization using L-BFGS-B for better refinement
    result2 = minimize(objective, refined_h, method='L-BFGS-B',
                      bounds=[(0, 1) for _ in range(n_points)],
                      options={'maxiter': 300, 'ftol': 1e-12})
    
    if result2.success:
        final_h = np.clip(result2.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
    else:
        final_h = refined_h
    
    # One last check for feasibility
    final_h = np.clip(final_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
