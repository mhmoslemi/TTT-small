import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose n_points to balance size and performance
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from 'initial_h_values' if available, else use triangular ramp
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Validate and adjust if necessary
        if np.abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    # Ensure h satisfies constraints
    h = clip_and_renormalize(h, target_sum)
    
    # Initialize optimization variables
    velocity = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iters = 20000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Early termination
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Main optimization loop
    for iteration in range(max_iters):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        # Track the best solution found so far
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        # This is the direction that reduces overlap (approximately)
        # Gradient approximation based on change in max overlap
        gradient = np.zeros(n_points)
        
        # Use finite differences to approximate gradient direction
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            gradient[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps)
        
        # Apply momentum update
        velocity = momentum * velocity + learning_rate * gradient
        h_candidate = h - velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute candidate C5
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate.copy()
        else:
            # Shrink step size and try again
            velocity *= shrink_factor
            
            # Try with reduced velocity
            h_temp = h - velocity
            h_temp = clip_and_renormalize(h_temp, target_sum)
            temp_c5 = compute_c5(h_temp, dx)
            
            if temp_c5 < overlap_max:
                h = h_temp.copy()
                
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
