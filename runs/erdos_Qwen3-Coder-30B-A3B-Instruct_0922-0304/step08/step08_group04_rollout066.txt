import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx)
    else:
        # Start with a simple construction
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, n_points / 2)
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx)
    
    # Use a multi-start approach with local search
    np.random.seed(seed)
    n_starts = 50
    for start in range(n_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start with a perturbed version of the best solution
        if start == 0:
            h = best_h.copy()
        else:
            # Random perturbation
            h = best_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, n_points / 2)
        
        # Local optimization using a simple gradient-like approach
        for iter in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            old_c5 = evaluate_c5(h, dx)
            
            # Simple gradient estimation
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Numerical derivative of overlap wrt h[i]
                eps = 1e-4
                h_plus = h.copy()
                h_plus[i] = min(1.0, h[i] + eps)
                h_minus = h.copy()
                h_minus[i] = max(0.0, h[i] - eps)
                
                c5_plus = evaluate_c5(h_plus, dx)
                c5_minus = evaluate_c5(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            step_size = 0.01
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, n_points / 2)
            
            # Accept if better
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < old_c5:
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            
            # Early stopping
            if new_c5 < 0.3807:
                break
    
    # Final verification
    if best_h is None:
        # Fallback to simple uniform solution
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, n_points / 2)
        best_c5 = evaluate_c5(h, dx)
        best_h = h
    
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
