import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, n_points):
    # Single point crossover
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    h = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(n_points, target_sum, n_pop=20, budget_s=600):
    np.random.seed(42)
    start_time = time.time()
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (target_sum / np.sum(h))
        population.append(h)
    
    # Fill the rest of the population randomly
    for _ in range(n_pop - 1):
        h = np.random.random(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, 2.0/n_points)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Elitism - keep top 20%
        new_population = population[:int(0.2 * n_pop)]
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            # Selection with tournament
            parent1_idx = np.random.randint(0, int(0.5 * n_pop))
            parent2_idx = np.random.randint(0, int(0.5 * n_pop))
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.random() < 0.8:  # crossover probability
                child1, child2 = crossover(parent1, parent2, n_points)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, 0.05)
            child2 = mutate(child2, 0.05)
            
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < n_pop:
                new_population.append(child2)
        
        population = new_population
        generation += 1
        
        # Check for improvement
        if generation > 1 and fitness[0] >= best_c5:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
    
    # Local refinement of top solutions
    refined_population = []
    for h in population[:3]:
        h_refined = h.copy()
        current_c5 = compute_c5(h_refined, 2.0/n_points)
        # Gradient-like descent
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            h_temp = h_refined.copy()
            idx = np.random.randint(n_points)
            delta = np.random.normal(0, 0.02)
            h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = repair_h(h_temp, target_sum)
            c5_temp = compute_c5(h_temp, 2.0/n_points)
            if c5_temp < current_c5:
                h_refined = h_temp
                current_c5 = c5_temp
        refined_population.append(h_refined)
    
    # Find best among refined
    for h in refined_population:
        c5 = compute_c5(h, 2.0/n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Start timing
    start_time = time.time()
    
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(n_points, target_sum, budget_s=budget_s)
    
    return (best_h.tolist(), best_c5, n_points)
