import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute approximate gradient of C5."""
    # The derivative of the overlap wrt h[i] - simplified estimation
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h - use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 2000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_size = 10
    max_no_improvement = 200
    
    # Create initial population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = h.copy()
            individual += np.random.normal(0, 0.05, n_points)
        else:
            individual = np.random.rand(n_points)
        individual = repair_h(individual, target_sum)
        population.append(individual)
    
    best_of_gen = []
    no_improvement_count = 0
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Selection - top 50%
        elites = population[:elite_size]
        
        # Create new population
        new_population = elites.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, population_size//2)]
            parent2 = population[np.random.randint(0, population_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.clip(child1[i] + np.random.normal(0, 0.05), 0, 1)
                if np.random.rand() < mutation_rate:
                    child2[i] = np.clip(child2[i] + np.random.normal(0, 0.05), 0, 1)
            
            # Repair
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            # Local refinement with gradient descent
            if np.random.rand() < 0.5:
                step_size = 0.01
                grad1 = compute_gradient(child1, n_points)
                candidate1 = child1 - step_size * grad1
                candidate1 = repair_h(candidate1, target_sum)
                if evaluate_c5(candidate1, dx) < evaluate_c5(child1, dx):
                    child1 = candidate1
                    
            if np.random.rand() < 0.5:
                step_size = 0.01
                grad2 = compute_gradient(child2, n_points)
                candidate2 = child2 - step_size * grad2
                candidate2 = repair_h(candidate2, target_sum)
                if evaluate_c5(candidate2, dx) < evaluate_c5(child2, dx):
                    child2 = candidate2
                    
            new_population.extend([child1, child2])
        
        # Trim to population_size
        population = new_population[:population_size]
        
        # Adapt mutation rate
        if no_improvement_count > 50:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        elif no_improvement_count < 10:
            mutation_rate = max(0.01, mutation_rate * 0.95)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
