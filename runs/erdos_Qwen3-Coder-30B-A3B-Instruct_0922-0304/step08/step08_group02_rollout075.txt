```python
import numpy as np
from scipy.signal import correlate
import time
from math import floor

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Ensure h is in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def crossover(parent1, parent2, n_points):
    """Blend two parents at a random point."""
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Randomly adjust some elements of h."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h

def init_population(n_pop, n_points, target_sum, initial_h_values):
    """Initialize a population of valid solutions."""
    population = []
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        # Add the provided initial solution
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
        population.append(h)
    
    # Fill the rest with random solutions
    while len(population) < n_pop:
        # Generate random values
        h = np.random.random(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
        
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = init_population(n_pop, n_points, target_sum, initial_h_values)
    
    # GA parameters
    elite_size = max(1, n_pop // 5)
    mutation_rate = 0.05
    crossover_prob = 0.8
    
    best_c5 = float('inf')
    best_h = None
    
    gen = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = sorted_population[:elite_size]
        
        # Generate new population through crossover and mutation
        while len(new_population) < n_pop:
            if np.random.random() < crossover_prob:
                # Select two parents with probability proportional to fitness
                parents = sorted(np.random.choice(n_pop, size=2, replace=False, p=np.array(sorted_fitness)**(-2)))
                parent1, parent2 = sorted_population[parents[0]], sorted_population[parents[1]]
                child1, child2 = crossover(parent1, parent2, n_points)
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                child1 = repair_h(child1, target_sum)
                child2 = repair_h(child2, target_sum)
                new_population.extend([child1, child2])
            else:
                # Directly mutate a random individual
                idx = np.random.randint(0, n_pop)
                child = mutate(sorted_population[idx], mutation_rate)
                child = repair_h(child, target_sum)
                new_population.append(child)
        
        population = new_population[:n_pop]
        
        # Check for stagnation
        if gen > 0 and sorted_fitness[0] >= best_c5:
            no_improvement += 1
        else:
            no_improvement = 0
            
        gen += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Local refinement on the best solution
    if best_h is not None:
        h_refined = best_h.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            h_temp = h_refined.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            h_temp[idx] += delta
            h_temp = repair_h(h_temp, target_sum)
            c5_temp = compute_c5(h_temp, dx)
            if c5_temp < compute_c5(h_refined, dx):
                h_refined = h_temp.copy()
        
        final_c5 = compute_c5(h_refined, dx)
        best_c5 = min(best_c5, final_c5)
        best_h = h_refined
    
    # Final check
    if best_h is None:
        # Fallback to initial if anything went wrong
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values)
            best_h = repair_h(best_h, target_sum)
        else:
            best_h = np.ones(n_points) * 0.5
            best_h = repair_h(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
```
