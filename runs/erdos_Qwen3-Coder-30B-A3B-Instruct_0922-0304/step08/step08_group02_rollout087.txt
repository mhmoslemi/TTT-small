import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create an initial solution from scratch
        h_current = np.zeros(n_points)
        # Place 1s in roughly half the positions
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_current[ones_indices] = 1.0
        # Add noise to make it a proper solution
        h_current += np.random.normal(0, 0.05, n_points)
        h_current = np.clip(h_current, 0, 1)
        # Scale to sum to target_sum
        h_current = h_current * (target_sum / np.sum(h_current))
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    max_steps_no_improvement = 500
    step_count = 0
    current_T = T0
    no_improvement_count = 0
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    while time.time() - start_time < budget_s - 1:
        step_count += 1
        
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        
        # Perturb one element
        delta = np.random.normal(0, sigma)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Maintain sum constraint by scaling the other elements
        current_sum = np.sum(h_new)
        if current_sum > 0:
            scale_factor = target_sum / current_sum
            h_new = np.where(np.arange(n_points) != idx, h_new * scale_factor, h_new)
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Compute C5
        c5_new = compute_c5(h_new, dx)
        c5_current = compute_c5(h_current, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.random() < exp(-delta_c5 / max(current_T, T_min)):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
                no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Cooling schedule
        current_T = max(T_min, T0 * (alpha ** (step_count / 1000.0)))
        
        # Adapting perturbation scale if needed
        if no_improvement_count > max_steps_no_improvement:
            sigma *= 0.9
            no_improvement_count = 0
            
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    refinement_steps = 200
    for step in range(refinement_steps):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            scale_factor = target_sum / current_sum
            for i in range(n_points):
                if i != idx:
                    h_new[i] *= scale_factor
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
    return (best_h.tolist(), best_c5, n_points)
