import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 1000
    iter_count = 0
    
    # Precompute shift indices
    shift_indices = []
    for k in range(n_points):
        shift_indices.append((k, slice(None, n_points-k) if k < n_points else slice(None, 0)))
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Convex relaxation step
        # Create the optimization problem
        h_var = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
            z >= 0
        ]
        
        # Overlap constraints
        for shift_idx in range(n_points):
            # Overlap for shift k where k = shift_idx 
            if shift_idx < n_points:
                overlap_expr = cp.sum(cp.multiply(h_var[:n_points-shift_idx], 1 - h_var[shift_idx:]))
                constraints.append(overlap_expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                relaxed_h = h_var.value
                # Rounding heuristic
                rounded_h = np.zeros(n_points)
                # Sort relaxed values and apply thresholding
                sorted_indices = np.argsort(relaxed_h)[::-1]
                remaining_sum = target_sum
                for idx in sorted_indices:
                    if remaining_sum > 0:
                        rounded_h[idx] = 1
                        remaining_sum -= 1
                    else:
                        rounded_h[idx] = 0
                # Adjust to exactly target_sum
                current_sum = np.sum(rounded_h)
                diff = target_sum - current_sum
                if diff > 0:
                    # Find zeros and turn them to ones
                    zero_indices = np.where(rounded_h == 0)[0]
                    for i in range(int(diff)):
                        if i < len(zero_indices):
                            rounded_h[zero_indices[i]] = 1
                elif diff < 0:
                    # Find ones and turn them to zeros
                    one_indices = np.where(rounded_h == 1)[0]
                    for i in range(int(-diff)):
                        if i < len(one_indices):
                            rounded_h[one_indices[i]] = 0
                
                rounded_h = np.clip(rounded_h, 0, 1)
                new_c5 = compute_c5(rounded_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = rounded_h.copy()
        except:
            pass
        
        # Local search on the rounded result
        if 'rounded_h' in locals():
            local_h = rounded_h.copy()
            for _ in range(50):
                # Randomly perturb
                idx = np.random.randint(0, n_points)
                old_val = local_h[idx]
                local_h[idx] = np.clip(old_val + (np.random.rand() - 0.5) * 0.1, 0, 1)
                
                # Renormalize
                local_h = local_h * (target_sum / np.sum(local_h))
                local_h = np.clip(local_h, 0, 1)
                
                new_c5 = compute_c5(local_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = local_h.copy()
        
        iter_count += 1
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
