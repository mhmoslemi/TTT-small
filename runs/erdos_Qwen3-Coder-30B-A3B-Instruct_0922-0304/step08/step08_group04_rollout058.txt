import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, n_points):
    """Repair infeasible h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points / 2)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate individual with noise."""
    if random.random() > mutation_rate:
        return h.copy()
    
    n = len(h)
    noise = np.random.normal(0, noise_magnitude, n)
    h_mut = h + noise
    return h_mut

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual via tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def evaluate_population(population, dx, n_points):
    """Evaluate all individuals in the population."""
    fitnesses = Parallel(n_jobs=1)(delayed(evaluate_c5)(ind, dx) for ind in population)
    return fitnesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            individual = parent + noise
            individual = repair_feasibility(individual, n_points)
            population.append(individual)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, n_points / 2)
            population.append(h)
    
    best_individual = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break
            
        # Evaluate fitness
        fitnesses = evaluate_population(population, dx, n_points)
        
        # Update best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Select parents and create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Repair feasibility
            child1 = repair_feasibility(child1, n_points)
            child2 = repair_feasibility(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final evaluation of best individual
    if best_individual is not None:
        final_c5 = evaluate_c5(best_individual, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    else:
        # Fallback to best from initial population
        fitnesses = evaluate_population(population, dx, n_points)
        min_fitness_idx = np.argmin(fitnesses)
        best_c5 = fitnesses[min_fitness_idx]
        best_individual = population[min_fitness_idx].copy()
        
    return (best_individual.tolist(), best_c5, n_points)
