import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    ramp = np.linspace(0, 1, n // 2)
    return np.concatenate([ramp, ramp[::-1]])

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Optimizer parameters
    momentum = 0.9
    learning_rate = 0.001
    decay_rate = 0.99
    coord_sample_rate = 0.2
    max_iter = 10000
    no_improvement_limit = 1000
    
    velocity = np.zeros(n_points)
    no_improvement_count = 0
    
    # Main optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        # Update best if improved
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient approximation
        gradients = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            # Finite difference gradient estimate
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            gradients[i] = (max_plus - max_minus) / (2 * eps)
            
        # Gradient descent step with momentum
        velocity = momentum * velocity + (1 - momentum) * gradients
        h_candidate = h - learning_rate * velocity
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate candidate
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            learning_rate *= decay_rate
            no_improvement_count = 0
        else:
            h = clip_and_renormalize(h, target_sum)
            learning_rate *= decay_rate
            no_improvement_count += 1
            
        if no_improvement_count > no_improvement_limit:
            # Coordinate-wise optimization
            coord_sample_size = max(1, int(n_points * coord_sample_rate))
            update_indices = np.random.choice(n_points, coord_sample_size, replace=False)
            
            improved = False
            for idx in update_indices:
                # Try increasing/decreasing this coordinate
                current_val = h[idx]
                best_step_size = 0.0
                best_new_c5 = overlap_max
                
                # Try several small steps
                for step_strength in [0.001, 0.005, 0.01, 0.02]:
                    for direction in [-1, 1]:
                        h_new = h.copy()
                        new_val = current_val + direction * step_strength
                        h_new[idx] = max(0.0, min(1.0, new_val))
                        h_new = clip_and_renormalize(h_new, target_sum)
                        
                        new_c5 = compute_c5(h_new, dx)
                        if new_c5 < best_new_c5:
                            best_new_c5 = new_c5
                            best_step_size = direction * step_strength
                
                if best_new_c5 < overlap_max:
                    h[idx] = max(0.0, min(1.0, current_val + best_step_size))
                    h = clip_and_renormalize(h, target_sum)
                    improved = True
                    
            if improved:
                no_improvement_count = 0
            else:
                no_improvement_count += 1

    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
