import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Construct initial h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (target_sum / np.sum(h_init))
    
    # Formulate and solve QP
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints for all shifts k from -n+1 to n-1
    for k in range(-n_points+1, n_points):
        # Construct the overlap sum for shift k
        if k >= 0:
            # h[i] * (1 - h[i+k]) for valid indices
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i+k])
        else:
            # h[i] * (1 - h[i+k]) for valid indices
            expr = 0
            for i in range(-k, n_points):
                expr += h[i] * (1 - h[i+k])
        
        constraints.append(expr <= t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solver_list = [cp.ECOS, cp.CVXOPT, cp.GLPK]
    best_h = h_init
    best_c5 = float('inf')
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=1000, reltol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h.value
                if h_sol is not None:
                    # Evaluate the solution
                    c5_val = np.max(np.correlate(h_sol, 1 - h_sol, mode="full")) * dx
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_sol.copy()
            break
        except:
            continue
    
    # If no solver worked, fallback to initial solution
    if best_c5 == float('inf'):
        best_h = h_init
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    # Verify feasibility
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-5
    
    # Final evaluation
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
