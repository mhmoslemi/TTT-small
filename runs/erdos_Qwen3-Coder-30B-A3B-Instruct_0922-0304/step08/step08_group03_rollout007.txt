import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    """Create a triangular ramp as initial guess."""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Parameters
    learning_rate = 0.001
    momentum = 0.9
    decay = 0.99
    coord_sample_rate = 0.2
    max_iter = 10000
    start_time = time.time()
    
    # Initialize momentum
    velocity = np.zeros(n_points)
    
    no_improvement_count = 0
    max_no_improvement = 2000
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Return early if we're already there
        if overlap_max <= 0.3807:
            break
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
        
        # Gradient: This is an approximation; in reality computing 
        # exact gradients for non-smooth functions is challenging.
        # We are using a small finite difference approach to estimate
        # the sensible direction to move h.
        # Below we apply a numerically approximated gradient update strategy.
        
        # Phase 1: Gradient descent with momentum
        # This follows a simple gradient descent approach with momentum
        h_plus = h.copy()
        h_plus[max_pos % n_points] = np.clip(h[max_pos % n_points] + 0.01, 0, 1)
        h_plus = clip_and_renormalize(h_plus, target_sum)
        
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
        max_plus = np.max(overlap_plus)
        
        h_minus = h.copy()
        h_minus[max_pos % n_points] = np.clip(h[max_pos % n_points] - 0.01, 0, 1)
        h_minus = clip_and_renormalize(h_minus, target_sum)
        
        overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
        max_minus = np.max(overlap_minus)
        
        estimated_grad = (max_plus - max_minus) / 0.02
        
        # Simple momentum-based update
        velocity = momentum * velocity + (1 - momentum) * estimated_grad
        h_candidate = h - learning_rate * velocity
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 <= overlap_max:
            h = h_candidate.copy()
            learning_rate *= decay
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            learning_rate *= decay
            if no_improvement_count > max_no_improvement:
                break
        
        iter_count += 1
    
    # Final refinement step with coordinate-wise optimization
    no_improvement_count = 0
    coord_sample_size = max(1, int(coord_sample_rate * n_points))
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Sample some coordinates to update
        coord_indices = np.random.choice(n_points, size=coord_sample_size, replace=False)
        
        improved = False
        for i in coord_indices:
            # Test small steps in both directions
            orig = h[i]
            
            # Try positive and negative steps
            step_sizes = [0.001, 0.005, 0.01]
            
            for step in step_sizes:
                for sign in [1, -1]: 
                    h_new = h.copy()
                    h_new[i] = np.clip(orig + sign * step, 0, 1)
                    h_new = clip_and_renormalize(h_new, target_sum)
                    
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < best_c5:
                        h = h_new.copy()
                        best_c5 = new_c5
                        improved = True
                        break
                else:
                    continue
                break
            if improved:
                break

        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                break
                
        iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
