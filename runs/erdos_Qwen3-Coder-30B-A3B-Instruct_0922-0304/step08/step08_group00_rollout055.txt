import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a baseline
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial guess: step function
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    
    # Use sequential quadratic programming (SQP)
    def objective(x):
        return compute_overlap(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    # Max iterations for the optimizer
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        # Solve the optimization problem
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if result.success:
            # Ensure the solution is feasible
            new_h = np.clip(result.x, 0, 1)
            current_sum = np.sum(new_h)
            if current_sum > 0:
                new_h = new_h * (target_sum / current_sum)
            new_c5 = compute_overlap(new_h, dx)
            
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                iter_count = 0  # Reset iteration count
            else:
                iter_count += 1
        else:
            # If optimization fails, try perturbing the solution a bit and try again
            perturbation = np.random.normal(0, 1e-4, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            if current_sum > 0:
                h_perturbed = h_perturbed * (target_sum / current_sum)
            perturbed_c5 = compute_overlap(h_perturbed, dx)
            if perturbed_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = perturbed_c5
                iter_count = 0
            else:
                iter_count += 1
                
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = compute_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)

def compute_overlap(h, dx):
    """Calculate the maximum overlap integral"""
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx
