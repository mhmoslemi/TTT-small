import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Convex optimization approach
    # Create variables
    h_var = cp.Variable(n_points)
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Objective: minimize max overlap
    # This is more complex to model directly; we'll use a proxy approach
    
    # Instead, use a more direct approach: consider discretizing the overlap integral
    def solve_convex_optimization(h_start):
        # h_var is the variable for optimization
        # For now, use simple heuristic of global optimization
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        result = minimize(objective, h_start, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 500, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            return refined_h
        else:
            return h_start
    
    # GPU-friendly or fast optimization
    # Direct solve using CVXPY (though not very tractable for this specific case)
    # Instead, apply gradient-based refinement
    current_h = h.copy()
    
    # Allow for multiple local refinement strategies
    max_refinements = 200
    last_improvement = 0
    improvement_count = 0
    
    for i in range(max_refinements):
        # Local refinement
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        result = minimize(objective, current_h, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            refined_c5 = evaluate_overlap(refined_h, dx)
            
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
                improvement_count += 1
                last_improvement = 0

        else:

            refined_h = current_h.copy()
            
        if time.time() - start_time > budget_s - 5:

        # Early stopping for stagnation

            break

    # Final validation and possible final local search
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
