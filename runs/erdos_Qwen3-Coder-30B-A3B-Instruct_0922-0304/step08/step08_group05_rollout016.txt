import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Ensure the initial sum is correct
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    max_iter = 100000
    epsilon = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Find the worst overlap lag
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Adjust for the worst overlap
        # Try to reduce that specific maximum overlap
        lag = max_idx - (n_points - 1)  # convert full index to lag
        if lag < 0:
            # Positive lag: look at h[i] and h[i+lag]
            for i in range(n_points - abs(lag)):
                if h[i] > 0 and h[i + abs(lag)] < 1:
                    # Reduce h[i] and increase h[i+lag] to reduce overlap
                    step = min(epsilon, h[i], 1 - h[i + abs(lag)])
                    h[i] -= step
                    h[i + abs(lag)] += step
        else:
            # Negative lag: look at h[i] and h[i+lag]
            for i in range(abs(lag), n_points):
                if h[i] > 0 and h[i - abs(lag)] < 1:
                    # Reduce h[i] and increase h[i-lag] to reduce overlap
                    step = min(epsilon, h[i], 1 - h[i - abs(lag)])
                    h[i] -= step
                    h[i - abs(lag)] += step
        
        # Normalize and reclip
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Adjust epsilon
        if iteration % 1000 == 0:
            epsilon *= 0.999
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
