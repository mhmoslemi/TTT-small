import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available

    else:
        h = np.random.rand(n_points)

    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Get the worst lag
        worst_lag = max_idx - (n_points - 1)
        
        # Make an adjustment to reduce the maximum overlap
        # Focus on nearby points to reduce this specific overlap
        # Change h at positions that contribute most to this overlap
        
        # Find indices contributing to max overlap
        h_new = h.copy()
        
        # Adjust a few key points around where the maximum overlap occurs
        # This is a greedy strategy to attack the max overlap immediately
        adjustments = np.zeros(n_points)
        
        # We're reducing the overlap at the specific lag 
        # by adjusting points in a targeted fashion
        for i in range(n_points):
            # Determine how much to modify h[i] to minimize overlap at lag `worst_lag`
            # We use a simple heuristic: if h[i] contributes to overlap_at_lag, 
            # then adjust in direction which reduces overlap
            if 0 <= i + worst_lag < n_points:
                # Estimate effect on overlap due to changing h[i]  
                # (Here we do a simplified gradient-like update)
                # If partially increasing h[i] reduces overlap, we make small adjusts
                diff = h[i] - 0.5
                # Add small adjustments to direction that reduces overlap
                adjustments[i] -= epsilon * diff
                
                # Also adjust for h[i + worst_lag] 
                if i + worst_lag >= 0 and i + worst_lag < n_points:
                    adjustments[i + worst_lag] += epsilon * diff
                
        # Apply adjustments with small step size
        h_new = h + adjustments
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        h = h_new
        
        iteration += 1
        if iteration % 100 == 0:
            epsilon *= 0.999
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
