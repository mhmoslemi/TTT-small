import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller sample size for faster optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function  
        h = np.zeros(n_points)
        # Create a piecewise constant function with some randomness
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iterations = 1000
    w = 0.8      # inertia weight
    c1 = 2.0     # cognitive parameter  
    c2 = 2.0     # social parameter
    
    # Initialize particles
    particles = []
    velocities = []
    
    for i in range(n_particles):
        if i == 0:
            # Start with best solution so far
            particle = h.copy()
        else:
            # Create variation around initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
            
        # Renormalize
        current_sum = np.sum(particle)
        if current_sum > 0:
            particle = particle * (target_sum / current_sum)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Evaluate initial fitness
    fitness = [evaluate_overlap(p, dx) for p in particles]
    
    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_fitness = fitness[:]
    
    # Global best
    best_idx = np.argmin(fitness)
    g_best = particles[best_idx].copy()
    g_best_fitness = fitness[best_idx]
    
    # PSO main loop
    no_improvement_iter = 0
    stagnation_threshold = 50
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -0.05, 0.05)
            
            # Update particle position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize
            current_sum = np.sum(particles[i])
            if current_sum > 0:
                particles[i] = particles[i] * (target_sum / current_sum)
            
            # Evaluate new fitness
            new_fitness = evaluate_overlap(particles[i], dx)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best if needed
                if new_fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = new_fitness
                    no_improvement_iter = 0
            else:
                no_improvement_iter += 1
        
        # Apply shaking to prevent stagnation
        if no_improvement_iter > stagnation_threshold:
            for i in range(n_particles):
                shake_magnitude = 0.05 * (1.0 - float(no_improvement_iter) / max_iterations)
                particles[i] += np.random.normal(0, shake_magnitude, n_points)
                particles[i] = np.clip(particles[i], 0, 1)
                # Renormalize
                current_sum = np.sum(particles[i])
                if current_sum > 0:
                    particles[i] = particles[i] * (target_sum / current_sum)
            no_improvement_iter = 0
        
        # Periodic refinement
        if iteration % 50 == 0:
            # Refine global best with local optimization
            def objective(h_vector):
                return evaluate_overlap(h_vector, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            
            result = minimize(objective, g_best, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 50, 'ftol': 1e-8})
            
            if result.success:
                refined_g_best = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_g_best)
                if refined_sum > 0:
                    refined_g_best = refined_g_best * (target_sum / refined_sum)
                
                new_g_best_fitness = evaluate_overlap(refined_g_best, dx)
                if new_g_best_fitness < g_best_fitness:
                    g_best = refined_g_best.copy()
                    g_best_fitness = new_g_best_fitness

    # Final refinement on the best solution found
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, g_best, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < g_best_fitness:
            g_best = final_h
            g_best_fitness = final_c5
    
    return (g_best.tolist(), g_best_fitness, n_points)
