```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Try to solve with convex relaxation first
    try:
        h_cvx = cp.Variable(n_points)
        t = cp.Variable()
        
        # Add constraints
        constraints = []
        constraints.append(cp.sum(h_cvx) == n_points / 2)
        constraints.append(h_cvx >= 0)
        constraints.append(h_cvx <= 1)
        
        # Add overlap constraints
        for k in range(2 * n_points - 1):
            if k < n_points:
                # Sum over x where both h_x and h_{x+k} exist
                expr = 0
                for x in range(n_points - k):
                    expr += h_cvx[x] * (1 - h_cvx[x + k])
                constraints.append(expr <= t)
            else:
                # Handle the wrap-around case
                expr = 0
                for x in range(k - n_points + 1, n_points):
                    expr += h_cvx[x] * (1 - h_cvx[x + k - n_points])
                constraints.append(expr <= t)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=ECOS, verbose=False)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h = np.array(h_cvx.value).reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass  # Fall back to regular optimization if convex fails
    
    # Refine with gradient descent
    max_iter = 10000
    lr = 0.01
    decay = 0.999
    tolerance = 1e-8
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute loss and gradient
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Early stopping
            if loss <= 0.3807:
                break
        
        # Compute approximate gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        grad = np.zeros_like(h)
        
        # Simplified gradient based on max overlap
        for k in range(n_points):
            # Compute contributions for each possible shift k
            temp_grad = 0.0
            for x in range(n_points):
                if x + k < n_points:
                    temp_grad += ((1 - 2 * h[x]) * (1 - h[x+k]))
                else:
                    # Wrap around
                    x2 = x + k - n_points
                    temp_grad += ((1 - 2 * h[x]) * (1 - h[x2]))
            grad = grad + temp_grad * np.ones_like(h)
        
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
```
