import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    max_overlap = overlap[max_idx]
    
    # Find the corresponding lag k
    k = max_idx - (n_points - 1)
    
    # Compute gradient for the max overlap
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Find all indices in the correlation
        j = (i + k) % n_points
        grad[i] = (1 - 2 * h[i])
    
    return grad

def solve_linear_subproblem(grad, n_points, target_sum):
    # Solve: min <grad, v> s.t. sum(v) = target_sum, 0 <= v <= 1
    # This is a linear program
    c = grad
    A_eq = np.array([1] * n_points)
    b_eq = target_sum
    
    bounds = [(0, 1)] * n_points
    
    # Solve using scipy
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback to a simple heuristic if optimization fails
        v = np.zeros(n_points)
        # Try to set v to be proportional to -grad (but within bounds)
        unnormalized = np.maximum(-grad, 0)
        sum_unnormalized = np.sum(unnormalized)
        if sum_unnormalized > 0:
            v = unnormalized * target_sum / sum_unnormalized
        # Clip to bounds
        v = np.clip(v, 0, 1)
        # Normalize
        v = v * target_sum / np.sum(v)
        return v

def frank_wolfe_optimize(h, dx, n_points, max_iter=10000, tolerance=1e-8):
    target_sum = n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Solve linear subproblem: min <grad, v> s.t. sum(v) = target_sum, 0 <= v <= 1
        v = solve_linear_subproblem(grad, n_points, target_sum)
        
        # Compute step size η using Armijo line search
        eta_min, eta_max = 0.01, 1.0
        eta = (eta_min + eta_max) / 2
        
        # Simple backtracking line search
        alpha = 0.5
        beta = 0.5
        while eta > 1e-6:
            h_new = (1 - eta) * h + eta * v
            h_new = project_h(h_new, target_sum)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            if c5_new <= best_c5 - alpha * eta * np.dot(grad, v - h):
                break
            eta *= beta
        
        # Update h
        h = (1 - eta) * h + eta * v
        h = project_h(h, target_sum)
        
        # Early stopping if improvement is small
        if i % 100 == 0:
            c5 = evaluate_c5(h, dx)
            if c5 <= 0.3807:
                break
        
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points

    # Optimize the function
    best_h, best_c5 = frank_wolfe_optimize(h, dx, n_points, max_iter=10000)
    
    # Make sure we have the exact best solution within the recorded time
