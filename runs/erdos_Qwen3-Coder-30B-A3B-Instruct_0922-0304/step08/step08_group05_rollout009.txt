import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighbor(h, n_points):
    """Generate a neighbor by flipping or swapping elements."""
    h_new = h.copy()
    # Pick type of move: 0 for flip, 1 for swap
    move_type = np.random.randint(0, 2)
    
    if move_type == 0:  # Flip
        i = np.random.randint(0, n_points)
        # Perturb the value
        h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
    else:  # Swap
        i, j = np.random.choice(n_points, size=2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Normalize and clip to bounds
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Initialize tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    current_c5 = best_c5
    temperature = 1.0
    cooling_rate = 0.999
    stagnation_count = 0
    max_stagnation = 1000
    
    # Neighborhood size
    neighborhood_size = 10
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < max_stagnation:
        # Generate neighborhood
        candidates = []
        for _ in range(neighborhood_size):
            h_new = generate_neighbor(h, n_points)
            # Apply bounds
            h_new = np.clip(h_new, 0, 1)
            # Normalize
            h_new = normalize_h(h_new, target_sum)
            c5_new = compute_c5_exact(h_new, dx)
            candidates.append((h_new, c5_new))
        
        # Find best non-tabu candidate
        best_candidate = None
        best_candidate_c5 = float('inf')
        
        for h_cand, c5_cand in candidates:
            # Check if we're tabu
            hashable_cand = tuple(h_cand)
            if hashable_cand in tabu_list:
                # Check aspiration criteria
                if c5_cand < best_c5:
                    best_candidate = (h_cand, c5_cand)
                    best_candidate_c5 = c5_cand
            else:
                if c5_cand < best_candidate_c5:
                    best_candidate = (h_cand, c5_cand)
                    best_candidate_c5 = c5_cand
        
        # Update current solution if we found a better one
        if best_candidate is not None:
            h, current_c5 = best_candidate
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            # Add current state to tabu list
            tabu_list.append(tuple(h))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Update temperature
            temperature *= cooling_rate
            
            # Early termination
            if best_c5 <= 0.3807:
                break
                
        iteration += 1
        
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
