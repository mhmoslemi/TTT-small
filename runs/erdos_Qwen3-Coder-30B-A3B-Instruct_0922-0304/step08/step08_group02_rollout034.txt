import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient for given shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    patience = 0
    max_patience = 100
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # The shift that gives maximum overlap
        shift = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h_current, shift, n_points)
        
        # Update h using projected gradient descent
        h_new = h_current - step_size * grad
        
        # Project to feasible region
        h_new = project_h(h_new, n_points)
        
        # Evaluate the new solution
        c5_new = compute_c5(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
            # Reduce step size after improvement
            step_size *= 0.95
        else:
            patience += 1
        
        # Periodic perturbation to escape local minima
        if patience > max_patience and np.random.rand() < 0.1:
            # Add small noise
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h_current + noise
            h_new = project_h(h_new, n_points)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
                step_size *= 0.95
        
        # Restore current solution for next iteration
        h_current = h_new.copy()
        
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
