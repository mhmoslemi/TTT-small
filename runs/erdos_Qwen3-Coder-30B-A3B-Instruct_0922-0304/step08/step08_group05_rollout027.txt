import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, sigma=0.1):
    """Apply Gaussian mutation to an individual."""
    h_new = h + np.random.normal(0, sigma, len(h))
    h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover_parents(p1, p2):
    """Create offspring from two parents by averaging."""
    offspring = (p1 + p2) / 2.0
    return offspring

def initialize_population(n_points, n_pop, target_sum):
    """Initialize a population of individuals."""
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill the rest with random individuals
    for _ in range(n_pop - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Hyperparameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_pop = 50
    n_gen = 1000
    mutation_rate = 0.1
    elite_ratio = 0.2
    tournament_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    population = initialize_population(n_points, n_pop, target_sum)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    
    # Evolution process
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5_exact(ind, dx)
            fitness.append(c5)
            
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = []
        elite_count = int(elite_ratio * n_pop)
        sorted_indices = np.argsort(fitness)
        
        # Elitism
        for i in range(elite_count):
            selected.append(population[sorted_indices[i]])
            
        # Tournament selection
        remaining_count = n_pop - elite_count
        for _ in range(remaining_count):
            # Tournament
            tournament = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
            
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        idx = elite_count
        while idx < n_pop:
            if np.random.rand() < crossover_rate and idx + 1 < n_pop:
                parent1 = selected[idx]
                parent2 = selected[idx + 1]
                offspring = crossover_parents(parent1, parent2)
                offspring = normalize_h(offspring, target_sum)
                if np.random.rand() < mutation_rate:
                    offspring = mutate_individual(offspring)
                    offspring = np.clip(offspring, 0, 1)
                    offspring = normalize_h(offspring, target_sum)
                new_population.append(offspring)
                idx += 2
            else:
                # Direct mutation
                if idx < n_pop:
                    mutant = mutate_individual(selected[idx], 0.1)
                    mutant = np.clip(mutant, 0, 1)
                    mutant = normalize_h(mutant, target_sum)
                    new_population.append(mutant)
                    idx += 1
                    
        # Replace population
        population = new_population
        
        generation += 1
        
        # Adaptive parameters
        if generation % 100 == 0:
            mutation_rate = max(0.05, mutation_rate * 0.99)
    
    # Final evaluation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
