import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cvx_sum, norm
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h onto the feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def frank_wolfe_step(h, dx, n_points, max_iter=1000):
    """Perform Frank-Wolfe step to minimize C5."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        # Compute overlap values
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Gradient computation for the maximum overlap term
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = max_idx - (n_points - 1)  # Adjust k for the full correlation
            # Compute gradient with respect to h[i]
            for j in range(n_points):
                pos = (i + j) % n_points
                if pos == (i + k) % n_points:
                    grad[i] += 1 - 2 * h[i]
        
        # Solve linear subproblem using CVXPY
        v = Variable(n_points)
        objective = Minimize(grad.T @ v)
        constraints = [
            cvx_sum(v) == n_points / 2,
            v >= 0,
            v <= 1
        ]
        prob = Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status != cp.OPTIMAL:
            break
            
        v_sol = np.array(v.value).flatten()
        
        # Line search for step size
        eta_candidates = np.linspace(0.01, 0.99, 20)
        best_eta = 0.1
        min_c5 = best_c5
        
        for eta in eta_candidates:
            h_new = (1 - eta) * h + eta * v_sol
            h_new = project_h(h_new, n_points / 2)
            c5 = evaluate_c5(h_new, dx)
            if c5 < min_c5:
                min_c5 = c5
                best_eta = eta
        
        # Update h
        h = (1 - best_eta) * h + best_eta * v_sol
        h = project_h(h, n_points / 2)
        
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = h.copy()
            
        if iteration % 100 == 0:
            if best_c5 <= 0.3807:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

    # Optimize using Frank-Wolfe
    best_h, best_c5 = frank_wolfe_step(h, dx, n_points, max_iter=10000)
    
    # Final verification
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
