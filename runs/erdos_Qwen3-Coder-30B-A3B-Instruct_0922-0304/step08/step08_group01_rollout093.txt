import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed):
    """Convert relaxed solution to binary vector."""
    # Use median as threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet the sum constraint
    current_sum = np.sum(h_binary)
    target_sum = len(h_binary) / 2
    
    if current_sum > target_sum:
        # Need to reduce by flipping some 1s to 0s
        # Find indices where h_binary is 1 but h_relaxed is below threshold,
        # or flip from 1 to 0 in order of increasing h_relaxed
        diff = int(current_sum - target_sum)
        to_zero = np.argsort(h_relaxed)[:diff]
        h_binary[to_zero] = 0
    elif current_sum < target_sum:
        # Need to increase by flipping some 0s to 1s
        diff = int(target_sum - current_sum)
        to_one = np.argsort(h_relaxed)[-diff:]
        h_binary[to_one] = 1
    
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try using convex relaxation scheme as a heuristic
    max_iterations = 10
    iteration_count = 0
    
    while time.time() - start_time < budget_s - 5 and iteration_count < max_iterations:
        # Try a relaxation approach: solve relaxed optimization
        try:
            # Set up the problem with CVXPY
            h_relaxed = cp.Variable(n_points, name="h_relaxed")
            z = cp.Variable(1, name="z")
            
            # Constraints
            constraints = [
                cp.sum(h_relaxed) == target_sum,
                h_relaxed >= 0,
                h_relaxed <= 1
            ]
            
            # Create overlap constraint matrix - for each k in [-n+1,n-1]
            # Sum over all valid pairs for shift k
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    # Sum over x from 0 to n-k-1
                    expr = 0
                    for x in range(n_points - k):
                        expr += h_relaxed[x] * (1 - h_relaxed[x + k])
                else:
                    # Sum over x from 0 to n+|k|-1
                    expr = 0
                    for x in range(n_points + k):
                        expr += h_relaxed[x] * (1 - h_relaxed[x - k])
                constraints += [expr * dx <= z]
            
            # Objective
            objective = cp.Minimize(z)
            problem = cp.Problem(objective, constraints)
            
            # Solve
            problem.solve(solver=cp.MOSEK, verbose=False, time_limit=10)
            
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed_val = np.array(h_relaxed.value).flatten()
                h_binary = round_to_binary(h_relaxed_val)
                
                # Ensure constraint
                h_binary = normalize_h(h_binary, target_sum)
                
                new_c5 = compute_c5(h_binary, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_binary.copy()
                    
                # Update the initial guess for next iteration
                h = h_binary.copy()
            else:
                # Fall back to simple iteration
                pass
                
        except Exception:
            pass  # Fall back to the original method
        
        # Apply traditional gradient-based refinement
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Identify the maximum conflict indices
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        
        # Select top coordinates to update (based on gradient magnitude)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Compute partial derivative of overlap with respect to h[i]
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            gradients[i] = grad
        
        # Rank coordinates by absolute gradient
        idxs = np.argsort(np.abs(gradients))[::-1]
        update_count = max(1, n_points // 10)  # Update top 10%
        selected_idxs = idxs[:update_count]
        
        # Update selected coordinates
        for i in selected_idxs:
            # Gradient-based update
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            
            # Simple step size heuristic
            step_size = 0.005
            if grad != 0:
                step_size = min(0.05, max(0.001, -grad * 0.002))
            
            # Apply update
            h[i] = h[i] + step_size * (1 - 2 * h[i])
            h[i] = np.clip(h[i], 0, 1)
            
            # Renormalize
            h = normalize_h(h, target_sum)
            
            # Check improvement
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            
            # Early stopping if target achieved
            if best_c5 <= 0.3807:
                break
        
        iteration_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
