import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def ensure_constraints(h, target_sum):
    """Ensure h stays within constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def generate_neighborhood(h, target_sum, size=10):
    """Generate neighborhood solutions."""
    neighbors = []
    n = len(h)
    for _ in range(size):
        h_new = h.copy()
        
        # 30% chance of flipping two points
        if random.random() < 0.3:
            # Flip two points
            i, j = random.sample(range(n), 2)
            h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
            h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
        else:
            # Swap two points
            i, j = random.sample(range(n), 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        h_new = ensure_constraints(h_new, target_sum)
        neighbors.append(h_new)
    
    return neighbors

def tabu_search(seed=42, budget_s=600):
    """Tabu search implementation for minimizing C5."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Initialize tabu list
    tabu_list = []
    tabu_tenure = 50
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    iteration = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 1000:
        # Generate neighborhood
        neighbors = generate_neighborhood(h, target_sum, size=10)
        
        # Evaluate neighbors and find best non-tabu
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor in neighbors:
            neighbor_c5 = compute_c5_exact(neighbor, dx)
            
            # Check tabu status
            tabu_status = False
            for prev_h in tabu_list:
                if np.allclose(neighbor, prev_h):
                    tabu_status = True
                    break
            
            # Check aspiration criteria
            if not tabu_status or neighbor_c5 < best_c5:
                if neighbor_c5 < best_neighbor_c5:
                    best_neighbor = neighbor
                    best_neighbor_c5 = neighbor_c5
        
        # Update tabu list
        tabu_list.append(h.copy())
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Update solution
        if best_neighbor is not None:
            h = best_neighbor
            if best_neighbor_c5 < best_c5:
                best_c5 = best_neighbor_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1
            
        iteration += 1
                
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = tabu_search(seed, budget_s)
    return (h_values.tolist(), c5_bound, n_points)
