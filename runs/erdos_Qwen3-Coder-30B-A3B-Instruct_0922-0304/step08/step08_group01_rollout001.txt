import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:

    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    # Define objective function
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    # Use SLSQP optimization
    result = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraint,
        options={'maxiter': 10000, 'ftol': 1e-9},
        callback=lambda x: None
    )
    
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h, dx)
    else:
        # Fallback to initial solution
        best_h = h
        best_c5 = compute_c5(h, dx)
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
