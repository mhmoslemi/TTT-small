import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n_points):
    """Create a triangular ramp function for initialization."""
    x = np.linspace(0, 2, n_points)
    h = np.minimum(x, 2 - x)
    h = h / np.sum(h) * (n_points / 2)
    return h

def project_to_feasible(h, target_sum):
    """Project h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Multi-resolution grid search
    resolutions = [35, 50, 70]
    max_iter_per_res = 2000
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
        h_res = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h)
        h_res = project_to_feasible(h_res, target_sum)
        
        # Optimizer: projected gradient descent
        learning_rate = 0.001
        momentum = 0.9
        velocity = np.zeros_like(h_res)
        
        for _ in range(max_iter_per_res):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradients
            grad = np.zeros_like(h_res)
            for i in range(len(h_res)):
                # Finite difference approximation of gradient
                eps = 1e-4
                h_plus = h_res.copy()
                h_plus[i] = min(1.0, h_res[i] + eps)
                h_plus = project_to_feasible(h_plus, target_sum)
                
                h_minus = h_res.copy()
                h_minus[i] = max(0.0, h_res[i] - eps)
                h_minus = project_to_feasible(h_minus, target_sum)
                
                c_plus = compute_c5(h_plus, dx)
                c_minus = compute_c5(h_minus, dx)
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Apply gradient descent with momentum
            velocity = momentum * velocity - learning_rate * grad
            h_new = h_res + velocity
            h_new = project_to_feasible(h_new, target_sum)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                
            h_res = h_new
            
            if best_c5 <= 0.3807:
                break
                
    return (best_h.tolist(), best_c5, n_points)
