```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    # Ensure h is within [0,1] and sums to n_points/2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    # Sample shifts from -n+1 to n-1
    shifts = np.random.randint(-n+1, n, sample_size)
    return shifts

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 200
    target_overlap = 0.38085
    sample_size = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        
        # Sample shifts for this iteration
        shifts = sample_shifts(n_points, sample_size)
        
        # Create a quadratic program to minimize the maximum overlap
        h_vars = cp.Variable(n_points)
        # Constraint for h to be between 0 and 1
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == target_sum
        ]
        
        # For each sampled shift, add a constraint that the overlap is bounded
        # We'll try to achieve target_overlap for max overlap
        max_overlap = cp.Variable()
        for k in shifts:
            # Compute correlation for index range in overlap
            if k >= 0:
                corr = cp.sum(h_vars[k:] * (1 - h_vars[:n_points-k]))
            else:
                corr = cp.sum(h_vars[:n_points+k] * (1 - h_vars[-k:]))
            constraints.append(corr <= max_overlap)
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
            else:
                # Fall back to previous point if optimization fails
                h_new = h_current.copy()
        except:
            h_new = h_current.copy()
        
        h_new = project_h(h_new, n_points)
        
        # Check C5 for the new solution
        c5_current = compute_c5(h_new, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_new.copy()
            no_improvement_count = 0
            # Occasionally adjust target overlap based on recent progress
            target_overlap = min(target_overlap, c5_current * 0.95)
        else:
            no_improvement_count += 1
            
        # Use new solution for next iteration
        h_current = h_new.copy()
        
        # Early stopping if we've found a good enough solution
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with small local search
    if best_c5 > 0.3807:
        h_refined = best_h.copy()
        for i in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            h_temp = h_refined.copy()
            idx = np.random.randint(n_points)
            delta = np.random.normal(0, 0.01)
            h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = project_h(h_temp, n_points)
            c5_temp = compute_c5(h_temp, dx)
            if c5_temp < compute_c5(h_refined, dx):
                h_refined = h_temp.copy()
        final_c5 = compute_c5(h_refined, dx)
        best_h = h_refined.copy()
        best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
```
