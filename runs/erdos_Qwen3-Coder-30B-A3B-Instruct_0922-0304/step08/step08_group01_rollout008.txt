import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex relaxation
        h_relaxed = h.copy()
        
        # Create the problem
        h_var = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # Overlap constraints with linear relaxation
        # For each shift k, we want sum h[i] * (1 - h[i+k]) <= z
        # Replace h[i] * h[i+k] with its convex envelope: 
        # h[i] * h[i+k] <= (h[i] + h[i+k]) / 2 - (h[i] - h[i+k])^2 / 4
        # But in this case, we directly use the bilinear term relaxation
        # By first computing the full overlap with linear constraints
        
        # Let's solve a simpler LP relaxation by using a different approach:
        # We'll use binary rounding heuristic + gradient descent refinement
        
        # Binary rounding approach combined with gradient search
        def round_to_binary(h, target_sum):
            # Sort and apply threshold
            indices = np.argsort(h)[::-1]
            h_binary = h.copy()
            h_binary[:] = 0
            
            current_sum = 0
            for i in indices:
                if current_sum < target_sum:
                    h_binary[i] = 1
                    current_sum += 1
                else:
                    break
            # Adjust if needed, ensure exact summation
            if current_sum < target_sum:
                # Add missing ones
                for i in range(n_points):
                    if h_binary[i] == 0 and current_sum < target_sum:
                        h_binary[i] = 1
                        current_sum += 1
            elif current_sum > target_sum:
                # Remove excess ones
                for i in range(n_points-1, -1, -1):
                    if h_binary[i] == 1 and current_sum > target_sum:
                        h_binary[i] = 0
                        current_sum -= 1
            return h_binary

        # Simple gradient-based improvement
        def improve_h(h_current):
            improvements_made = 0
            for i in range(n_points):
                if time.time() - start_time > budget_s - 1:
                    break
                original = h_current[i]
                # Try small perturbations
                for step in [-0.02, 0.02]:
                    h_try = h_current.copy()
                    h_try[i] = original + step
                    h_try[i] = np.clip(h_try[i], 0, 1)
                    # Renormalize
                    h_try = h_try * (target_sum / np.sum(h_try))
                    new_c5 = compute_c5(h_try, dx)
                    if new_c5 < best_c5:
                        h_current = h_try
                        improvements_made += 1
            return h_current

        # First, round to binary
        h_binary = round_to_binary(h_relaxed, target_sum)
        # Now refine using gradient descent with small steps
        h_improved = improve_h(h_binary)
        
        # Ensure normalization
        h_improved = h_improved * (target_sum / np.sum(h_improved))
        h_improved = np.clip(h_improved, 0, 1)
        
        # Check improvement
        new_c5 = compute_c5(h_improved, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_improved.copy()
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Save current state
        h = h_improved.copy()
        
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
