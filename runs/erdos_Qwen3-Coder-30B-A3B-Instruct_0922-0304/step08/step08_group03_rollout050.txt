import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    mid = n // 2
    left = np.linspace(0, 1, mid)
    right = np.linspace(1, 0, n - mid)
    return np.append(left, right)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if np.sum(h) > 0:
            h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
        
    best_h = h.copy()
    current_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    best_c5 = current_c5
    
    # Optimization parameters
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iterations = 20000
    velocity = np.zeros(n_points)
    
    iter_count = 0
    while iter_count < max_iterations and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        # Find where the maximum occurs
        max_idx = np.argmax(overlap)
        sign_max = 1.0 if overlap[max_idx] >= 0 else -1.0
        
        # Compute gradient
        h_gradient = sign_max * (1 - 2 * h) / n_points
        
        # Momentum update
        velocity = momentum * velocity + learning_rate * h_gradient
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new overlap
        new_overlap = correlate(h_candidate, 1 - h_candidate, mode="full") * dx
        new_c5 = np.max(new_overlap)
        
        # Accept improvement
        if new_c5 < current_c5:
            h = h_candidate.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Reduce step size and retry
            learning_rate *= shrink_factor
            if learning_rate < 1e-8:
                break
                
        iter_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
