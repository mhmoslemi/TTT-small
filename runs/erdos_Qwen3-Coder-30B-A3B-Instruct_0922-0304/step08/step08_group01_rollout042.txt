import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try a more principled approach using CVXPY for relaxation
        try:
            # LP relaxation approach
            h_r = cp.Variable(n_points, name='h_r')
            z = cp.Variable(1, name='z')
            
            # Constraints
            constraints = [0 <= h_r, h_r <= 1, cp.sum(h_r) == target_sum, z >= 0]
            
            # Generate overlap constraints
            # For all valid shifts k = 1 to n_points-1, and k = -1 to -(n_points-1)
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    expr = sum(h_r[i] * (1 - h_r[i + k]) for i in range(n_points - k))
                else:
                    expr = sum(h_r[i] * (1 - h_r[i - k]) for i in range(-k, n_points))
                constraints.append(expr * dx <= z)
                
            # Objective
            objective = cp.Minimize(z)
            
            # Create and solve the problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_r.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                
                # Binary rounding
                threshold = np.median(h_relaxed)
                h_binary = (h_relaxed >= threshold).astype(float)
                
                # Adjust to match sum constraint
                current_sum_binary = np.sum(h_binary)
                diff = current_sum_binary - target_sum
                
                if diff > 0:
                    # Need to flip diff entries from 1 to 0
                    flip_indices = np.argsort(h_binary)[:int(diff)]
                    h_binary[flip_indices] = 0
                elif diff < 0:
                    # Need to flip |diff| entries from 0 to 1
                    flip_indices = np.argsort(1 - h_binary)[:int(-diff)]
                    h_binary[flip_indices] = 1
                
                # Compute the actual overlap for binary version
                binary_c5 = compute_c5(h_binary, dx)
                if binary_c5 < best_c5:
                    best_c5 = binary_c5
                    best_h = h_binary.copy()
        
        except Exception as e:
            # If LP fails, fallback to gradient method
            pass
            
        # Always fallback to gradient method for more fine tuning
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Identify the maximum conflict indices
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        
        # Select top coordinates to update (based on gradient magnitude)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Compute partial derivative of overlap with respect to h[i]
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            gradients[i] = grad
        
        # Rank coordinates by absolute gradient
        idxs = np.argsort(np.abs(gradients))[::-1]
        update_count = max(1, n_points // 20)  # Update top 5%
        selected_idxs = idxs[:update_count]
        
        # Update selected coordinates
        for i in selected_idxs:
            old_h_i = h[i]
            # Use a simple gradient descent step with small step size
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            
            # Add a clip on the step size
            step_size = 0.02
            if grad != 0:
                step = -step_size * grad
            else:
                step = 0
            h[i] = h[i] + step
            h[i] = np.clip(h[i], 0, 1)
            
            # Renormalize
            h = normalize_h(h, target_sum)
            
            # Check improvement
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        iter_count += 1

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
