import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed_lp(h_initial, n_points, dx, max_iter=10):
    """Solve relaxed LP to get a continuous solution."""
    # Variables
    h_r = cp.Variable(n_points)
    z = cp.Variable()
    
    # Constraints
    constraints = [h_r >= 0, h_r <= 1]
    constraints.append(cp.sum(h_r) == n_points / 2)
    
    # Add overlap constraints
    for k in range(-n_points+1, n_points):
        # For valid shifts
        if k >= 0:
            # Sum over x from 0 to n-k-1
            expr = 0
            for x in range(n_points - k):
                expr += h_r[x] * (1 - h_r[x+k])
        else:
            # Sum over x from 0 to n+ k -1
            expr = 0
            for x in range(n_points + k):
                expr += h_r[x] * (1 - h_r[x+k])
        constraints.append(expr * dx <= z)
    
    # Objective
    objective = cp.Minimize(z)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, max_iters=max_iter)
        return h_r.value
    except:
        return h_initial

def round_to_binary(h_relaxed, n_points):
    """Round relaxed solution to binary like solution."""
    # Use median threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to match sum constraint
    current_sum = np.sum(h_binary)
    target_sum = n_points / 2
    
    if current_sum < target_sum:
        # Need to flip zeros to ones
        diff = int(target_sum - current_sum)
        # Find zeros with the smallest values
        zero_indices = np.where(h_binary == 0)[0]
        # Sort by h_relaxed values to identify best candidates to flip
        values = np.array([h_relaxed[i] for i in zero_indices])
        flip_indices = zero_indices[np.argsort(values)[:diff]]
        for idx in flip_indices:
            h_binary[idx] = 1
    elif current_sum > target_sum:
        # Need to flip ones to zeros
        diff = int(current_sum - target_sum)
        # Find ones with the largest values
        one_indices = np.where(h_binary == 1)[0]
        # Sort by h_relaxed values to identify candidates to flip
        values = np.array([h_relaxed[i] for i in one_indices])
        flip_indices = one_indices[np.argsort(values)[-diff:]]
        for idx in flip_indices:
            h_binary[idx] = 0
    
    return h_binary

    np.random.seed(seed)

    n_points = 70

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Zero-out some extreme values to guide optimization
    h = np.clip(h, 0, 1)
    
    # Try LP relaxation for improved start
    h_relaxed = solve_relaxed_lp(h, n_points, dx, max_iter=5)
    if h_relaxed is not None:
        h_rounded = round_to_binary(h_relaxed, n_points)
        if compute_c5(h_rounded, dx) < best_c5:
            best_h = h_rounded.copy()
            best_c5 = compute_c5(best_h, dx)
    
    # Local search around best solution
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Identify the maximum conflict indices
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        
        # Select top coordinates to update
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Compute approximate gradient
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            gradients[i] = grad
        
        idxs = np.argsort(np.abs(gradients))[::-1]
        update_count = max(1, n_points // 10)
        selected_idxs = idxs[:update_count]
        
        # Update selected points with small step
        for i in selected_idxs:
            old_h_i = h[i]
            grad = gradients[i]
            step_size = 0.005 if grad != 0 else 0.001
            # Adaptive step size
            if grad == 0:
                h[i] = h[i] + np.random.uniform(-0.01, 0.01)
            else:
                h[i] = h[i] + step_size * (1 - 2 * h[i])
            h[i] = np.clip(h[i], 0, 1)
            
        # Renormalize
        h = normalize_h(h, target_sum)
        
        # Check improvement
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Occasionally do relaxation pass
        if iter_count % 200 == 0 and time.time() - start_time < budget_s - 2:
            h_t = solve_relaxed_lp(h, n_points, dx, max_iter=3)
            if h_t is not None:
                h_t = round_to_binary(h_t, n_points)
                if compute_c5(h_t, dx) < best_c5:
                    best_c5 = compute_c5(h_t, dx)
                    best_h = h_t.copy()
                    
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
