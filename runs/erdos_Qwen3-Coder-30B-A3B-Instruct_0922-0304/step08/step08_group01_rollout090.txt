import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_r, target_sum):
    """Round relaxed solution to binary while maintaining sum."""
    # Compute median as threshold
    threshold = np.median(h_r)
    h_b = (h_r >= threshold).astype(float)
    
    # Adjust to meet exact sum constraint
    current_sum = np.sum(h_b)
    deficit = int(current_sum - target_sum)
    
    if deficit > 0:  # Too many 1s, flip some 1s to 0s
        # Find indices of 1s and sort by how close to threshold they are
        ones_idx = np.where(h_b == 1)[0]
        distances = np.abs(h_r[ones_idx] - threshold)
        # Flip the ones with smallest distances to threshold (closest to 0.5)
        flip_indices = np.argsort(distances)[:deficit]
        for idx in flip_indices:
            h_b[ones_idx[idx]] = 0
    elif deficit < 0:  # Too few 1s, flip some 0s to 1s
        # Find indices of 0s and sort by how close to threshold they are
        zeros_idx = np.where(h_b == 0)[0]
        distances = np.abs(h_r[zeros_idx] - threshold)
        # Flip the zeros with smallest distances to threshold (closest to 0.5)
        flip_indices = np.argsort(distances)[:abs(deficit)]
        for idx in flip_indices:
            h_b[zeros_idx[idx]] = 1
            
    return h_b

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # LP relaxation approach with alternating refinement
    # Use CVXPY with MOSEK solver for the quadratic program
    max_iterations = 20
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        # Solve LP relaxation using CVXPY
        h_r = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_r) == target_sum)
        for i in range(n_points):
            constraints.append(0 <= h_r[i])
            constraints.append(h_r[i] <= 1)
        
        # Overlap constraints for all shifts
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                # Overlap for positive shift
                overlap_sum = 0
                for i in range(n_points - k):
                    overlap_sum += h_r[i] * (1 - h_r[i + k])
                constraints.append(overlap_sum * dx <= z)
            else:
                # Overlap for negative shift
                overlap_sum = 0
                for i in range(n_points + k):
                    overlap_sum += h_r[i] * (1 - h_r[i - k])
                constraints.append(overlap_sum * dx <= z)
        
        # Objective
        prob = cp.Problem(cp.Minimize(z), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=20)
            if prob.status == cp.OPTIMAL:
                h_r_val = h_r.value
                # Convert to binary
                h_b = round_to_binary(h_r_val, target_sum)
                c5_b = compute_c5(h_b, dx)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    
            # Early termination
            if best_c5 <= 0.3807:
                break
        except:
            # Fallback to local optimization if LP fails
            pass
        
        iter_count += 1
    
    # Final refinement with local optimization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    # Use simpler local search approach to refine solution
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 2000:
        # Optimize using random search with gradient estimation
        # Generate random neighbor
        h_new = h.copy()
        
        # Perturb some points
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            perturb = np.random.normal(0, 0.05)
            h_new[idx] = np.clip(h_new[idx] + perturb, 0, 1)
        
        # Normalize
        h_new = normalize_h(h_new, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
