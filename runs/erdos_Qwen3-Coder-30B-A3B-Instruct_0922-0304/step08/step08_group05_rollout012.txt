import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum, max_lags):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute overlap for all lags
    overlap = correlate(h, 1 - h, mode="full")
    
    # For each active lag, compute contribution to gradient
    # For the max overlap, contributions come from positions that align
    for k in max_lags:
        # k is an offset in the full correlation
        # Convert offset to actual position in the correlation array
        # correlation has length 2*n-1, centered at n-1
        offset_in_corr = k + n - 1
        # For fixed lag k, compute gradient contribution
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                # Gradient contribution from h[i] * (1 - h[j]) term
                # Partial derivative of overlap_k w.r.t. h[i] is (1 - h[j]), 
                # and since overlap is sum over all such terms, 
                # we accumulate contributions from valid pairs
                # The gradient for h[i] depends on which h[j] terms appear in max overlap
                # For simplicity, based on where the maximum occurs, we approximate
                grad[i] += (1 - h[j]) * dx
                
    # Normalize
    if np.linalg.norm(grad) > 0:
        grad = grad / np.linalg.norm(grad)
    
    # Subtract projection onto the sum constraint gradient
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

    # Subgradient descent optimizer parameters
    learning_rate = 0.01
    max_iterations = 100000
    patience = 1000
    last_improvement = 0
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    iteration = 0
    
    # A very simple periodic adjustment to help escape local minima
    adjustment_period = 1000
    
    while time.time() - start_time < budget_s - 1 and last_improvement < patience:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap and identify maximum lag(s)
        overlap = correlate(h, 1 - h, mode="full")
        max_val = np.max(overlap)
        max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
        # Convert back to lag offsets
        max_lags = max_indices - n_points + 1
        
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum, max_lags)
        
        # Update parameters using subgradient
        h = h - learning_rate * grad
        
        # Apply constraints
