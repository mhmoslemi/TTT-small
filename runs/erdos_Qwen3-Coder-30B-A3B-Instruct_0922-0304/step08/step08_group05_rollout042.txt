import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    max_iter = int(1e5)
    epsilon = 0.01
    decay_rate = 0.999
    patience = 0
    max_patience = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        if patience > max_patience:
            # Reset if no improvement
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
            else:
                h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            patience = 0
            
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Get top overlapping lags
        sorted_indices = np.argsort(overlap)[::-1][:5]
        
        # For largest overlaps, adjust h to reduce those
        for idx in sorted_indices:
            if idx >= len(overlap) // 2:  # Positive lag
                k = idx - len(overlap) // 2
                if k == 0:
                    continue
            else:  # Negative lag
                k = len(overlap) // 2 - idx
                if k == 0:
                    continue
                
            # Adjust h to reduce this lag
            # Add small changes to reduce overlap
            for i in range(n_points):
                if i + k < n_points:
                    # Reduce the contribution to this overlap
                    h[i] = max(0, h[i] - epsilon)
                    h[i + k] = min(1, h[i + k] + epsilon)
            
            # Normalize
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            
        # Apply decay to epsilon
        epsilon *= decay_rate
        epsilon = max(epsilon, 1e-6)
    
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
