import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using circular correlation with reflected function
    n = len(h)
    # Sliding sum over all shifts
    max_overlap = 0.0
    for k in range(n):
        overlap = 0.0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1.0 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx

    start_time = time.time()

    n_points = 70

    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Perturb to create initial partition
    perturbation = 1e-3
    h_perturbed = h + np.random.uniform(-perturbation, perturbation, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    current_sum = np.sum(h_perturbed)
    if current_sum > 0:
        h_perturbed = h_perturbed * (target_sum / current_sum)
    
    # Partition into even and odd indices
    E = h_perturbed[::2]
    O = h_perturbed[1::2]
    
    improvement_threshold = 1e-5
    max_iterations = 100
    perturb_factor = 0.98
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
            
        # Alternate optimization ofEven and Odd subsets
        # Optimize E, fix O
        def objective_E(h_E):
            # Reconstruct full h
            h_full = h_perturbed.copy()
            h_full[::2] = h_E
            return evaluate_overlap(h_full, dx)
            
        bounds_E = [(0, 1) for _ in range(len(E))]
        constraints_E = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum + np.sum(O) - np.sum(h_perturbed[1::2])}]
        
        try:
            result_E = minimize(objective_E, E, method='SLSQP', 
                               bounds=bounds_E, constraints=constraints_E,
                               options={'ftol': 1e-10, 'maxiter': 100})
            if result_E.success:
                E_new = np.clip(result_E.x, 0, 1)
                h_new = h_perturbed.copy()
                h_new[::2] = E_new
                current_sum = np.sum(h_new)
                if current_sum > 0:
                    h_new = h_new * (target_sum / current_sum)
                c5_new = evaluate_overlap(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
        
        # Optimize O, fix E
        def objective_O(h_O):
            h_full = h_perturbed.copy()
            h_full[1::2] = h_O
            return evaluate_overlap(h_full, dx)
            
        bounds_O = [(0, 1) for _ in range(len(O))]

            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:

            final_c5 = evaluate_overlap(refined_h, dx)

                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
