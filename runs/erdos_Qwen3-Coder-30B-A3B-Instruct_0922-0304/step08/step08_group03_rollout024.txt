import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def gradient_descent_step(h, dx, max_iter=10):
    """Perform gradient descent to refine h."""
    target_sum = len(h) / 2
    for _ in range(max_iter):
        # Approximate gradient direction
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(overlap)
        # Simple gradient update (not exact derivative but works for direction)
        grad = 1 - 2 * h
        # Move along direction that reduces max overlap
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = repair_h(h_new, target_sum)
        if compute_c5(h_new, dx) < compute_c5(h, dx):
            h = h_new
        else:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2

    else:

    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic Algorithm with Gradient Descent refinement
    population_size = 50
    generations = 2000
    
    # Initialize population
    population = [repair_h(np.random.rand(n_points), target_sum) for _ in range(population_size)]
    population_c5 = [compute_c5(ind, dx) for ind in population]
    
    # Track best
    min_c5_idx = np.argmin(population_c5)
    if population_c5[min_c5_idx] < best_c5:
        best_c5 = population_c5[min_c5_idx]
        best_h = population[min_c5_idx].copy()
    
    # GA parameters
    mutation_rate = 0.1
    crossover_prob = 0.7
    no_improvement_count = 0
    max_no_improvement = 100
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Selection (tournament selection)
        tournament_size = 3
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([population_c5[i] for i in tournament])]
            selected.append(population[winner])
        
        # Crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, population_size-1)]
            
            if np.random.rand() < crossover_prob:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    if np.random.rand() < mutation_rate:
                        child2[j] += np.random.normal(0, 0.05)
            
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            # Gradient descent refinement
            child1 = gradient_descent_step(child1, dx)
            child2 = gradient_descent_step(child2, dx)
            
            offspring.append(child1)
            if len(offspring) < population_size:
                offspring.append(child2)
        
        # Evaluate offspring
        offspring_c5 = [compute_c5(ind, dx) for ind in offspring]
        
        # Replace worst with best offspring
        worst_idx = np.argmax(population_c5)
        if offspring_c5[0] < population_c5[worst_idx]:
            population[worst_idx] = offspring[0]
            population_c5[worst_idx] = offspring_c5[0]
        
        # Update best
        min_c5_idx = np.argmin(population_c5)
        if population_c5[min_c5_idx] < best_c5:
            best_c5 = population_c5[min_c5_idx]
            best_h = population[min_c5_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Increase mutation rate slightly due to stagnation
            mutation_rate = min(0.5, mutation_rate * 1.2)
            
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
