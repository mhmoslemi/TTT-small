```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample shifts from [-n+1, n-1]"""
    return np.random.randint(-n+1, n, sample_size)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Tune hyperparameters
    n_samples = 50
    
    iterations = 0
    last_improvement = 0
    target_overlap = best_c5 * 1.1
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        if iterations - last_improvement > 200:
            break
            
        # Sample shifts
        shifts = sample_shifts(n_points, n_samples)
        
        # Formulate and solve QP
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_vars - h_current))
        
        constraints = []
        constraints.append(cp.sum(h_vars) == target_sum)
        for i in range(n_points):
            constraints.append(0 <= h_vars[i])
            constraints.append(h_vars[i] <= 1)
        
        # Add overlap constraints
        for k in shifts:
            if k >= 0:
                overlap = cp.sum(cp.multiply(h_vars[k:], 1 - h_current[:n_points-k]))
            else:
                overlap = cp.sum(cp.multiply(h_vars[:n_points+k], 1 - h_current[-k:]))
            constraints.append(overlap <= target_overlap * dx)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_new = h_vars.value
                h_new = project_h(h_new, n_points)
                
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iterations
                    target_overlap = best_c5 * 1.05  # Make target slightly tighter
        except:
            # If QP fails, do a simple local search
            h_new = h_current.copy()
            idx = np.random.randint(n_points)
            delta = np.random.normal(0, 0.03)
            h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
            h_new = project_h(h_new, n_points)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                last_improvement = iterations
                
        h_current = best_h.copy()
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
