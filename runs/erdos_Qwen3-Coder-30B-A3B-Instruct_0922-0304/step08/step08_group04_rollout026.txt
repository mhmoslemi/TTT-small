import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # Build constraints matrix for QP
    # For shift k, we need sum h_i * (1 - h_{i+k}) <= t
    # This is equivalent to sum h_i - sum h_i * h_{i+k} <= t
    # Which means sum h_i * h_{i+k} >= sum h_i - t
    
    # Use CVXPY for the convex QP
    h_var = cp.Variable(n_points)
    t_var = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Boundary constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints for all shifts
    # We'll approximate by a subset of shifts for performance
    shifts = range(-n_points + 1, n_points)
    # Just sample a reasonable number of shifts to keep problem tractable
    step = max(1, len(shifts) // 50)
    relevant_shifts = shifts[::step]
    
    for k in relevant_shifts:
        # Compute overlap for shift k: sum h_i * (1 - h_{i+k})
        expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h_var[i] * (1 - h_var[j])
        constraints.append(expr <= t_var)
    
    # Objective
    objective = cp.Minimize(t_var)
    
    # Formulate and solve QP
    prob = cp.Problem(objective, constraints)
    
    # Solve with ECOS or default solver
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Extract solution
            h_opt = h_var.value
            c5_bound = t_var.value
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_bound = evaluate_c5(h_opt, dx)
        else:
            # Fallback to initial solution
            h_opt = h
            c5_bound = evaluate_c5(h_opt, dx)
    except Exception as e:
        # If CVXPY fails, return initial guess
        h_opt = h
        c5_bound = evaluate_c5(h_opt, dx)
    
    # Final check
    final_c5 = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
