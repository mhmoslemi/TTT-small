import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighbor(h, n_points):
    """Generate a neighbor solution by swapping two elements."""
    h_new = h.copy()
    i, j = np.random.choice(n_points, 2, replace=False)
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def apply_tabu_search(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Initialize tabu list
    tabu_list = []
    tabu_tenure = 100
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Parameters for tabu search
    max_no_improvement = 1000
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Generate neighbors - 10 candidates
        neighbors = []
        for _ in range(10):
            h_neighbor = generate_neighbor(h, n_points)
            h_neighbor = np.clip(h_neighbor, 0, 1)
            h_neighbor = normalize_h(h_neighbor, target_sum)
            c5_neighbor = compute_c5_exact(h_neighbor, dx)
            neighbors.append((h_neighbor, c5_neighbor))
        
        # Find best non-tabu neighbor
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        for h_n, c5_n in neighbors:
            # Check if neighbor is tabu
            if tuple(h_n) in tabu_list:
                continue
            if c5_n < best_neighbor_c5:
                best_neighbor = h_n
                best_neighbor_c5 = c5_n
        
        # If no non-tabu neighbor found, take the best candidate
        if best_neighbor is None:
            best_neighbor = min(neighbors, key=lambda x: x[1])[0]
            best_neighbor_c5 = min(neighbors, key=lambda x: x[1])[1]
        
        # Check aspiration criterion
        if best_neighbor_c5 < best_c5:
            best_h = best_neighbor.copy()
            best_c5 = best_neighbor_c5
            no_improvement_count = 0
            
        if best_c5 <= 0.3807:
            break
            
        # Update tabu list
        tabu_list.append(tuple(h))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Move to best neighbor
        h = best_neighbor
        iteration += 1
        no_improvement_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = apply_tabu_search(seed, budget_s)
    return (h_values.tolist(), c5_bound, n_points)
