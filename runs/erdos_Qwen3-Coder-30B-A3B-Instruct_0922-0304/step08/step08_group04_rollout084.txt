```python
import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def feasible_h(h, n_points):
    """Ensure h is feasible: within [0,1] and sum equals n_points/2."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points / 2)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, noise_magnitude):
    """Mutate h with noise."""
    noise = np.random.normal(0, noise_magnitude, len(h))
    h = h + noise
    return feasible_h(h, len(h))

def select_parents(population, fitnesses, tournament_size=2):
    """Select parents using tournament selection."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = random.sample(range(len(population)), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_index].copy())
    return selected

def evaluate_population(population, dx):
    """Evaluate the fitness of each individual in the population."""
    fitnesses = Parallel(n_jobs=-1)(delayed(evaluate_c5)(ind, dx) for ind in population)
    return fitnesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population.append(parent)
        # Generate diverse individuals based on parent
        for _ in range(pop_size - 1):
            noise_magnitude = 0.05
            individual = mutate(parent.copy(), noise_magnitude)
            population.append(individual)
    else:
        # Random initialization with sum constraint
        for _ in range(pop_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    best_individual = None
    best_c5 = float('inf')
    
    # GA parameters
    crossover_rate = 0.8
    mutation_rate = 0.3
    noise_magnitude = 0.05
    
    for generation in range(max_generations):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = evaluate_population(population, dx)
        
        # Track best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Generate new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Crossover and mutation
        for i in range(0, pop_size - 1, 2):
            parent1 = parents[i]
            parent2 = parents[i+1]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            if random.random() < mutation_rate:
                child1 = mutate(child1, noise_magnitude)
            if random.random() < mutation_rate:
                child2 = mutate(child2, noise_magnitude)
                
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
        # Decrease noise magnitude over time
        noise_magnitude *= 0.99
    
    # Final evaluation
    final_c5 = evaluate_c5(best_individual, dx)
    
    # Ensure exact final check
    best_individual = feasible_h(best_individual, n_points)
    final_c5 = evaluate_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
```
