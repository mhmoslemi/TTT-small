import numpy as np
from scipy.signal import correlate
import time
import math

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_solution(h, n_points, perturb_fraction=0.1):
    """Generate a perturbed solution by swapping or flipping elements."""
    h_new = h.copy()
    n_perturb = max(1, int(n_points * perturb_fraction))
    
    # Randomly select indices to perturb
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    
    # Apply perturbations
    for i in indices:
        if np.random.random() < 0.5:
            # Flip operation
            h_new[i] = 1 - h_new[i]
        else:
            # Swap with a random index
            j = np.random.randint(0, n_points)
            h_new[i], h_new[j] = h_new[j], h_new[i]
    
    return h_new

def cooling_schedule(t, iteration, max_iter):
    """Adaptive cooling schedule."""
    return 1.0 * (0.999 ** iteration)

    temperature = 1.0

    while time.time() - start_time < budget_s - 1 and last_improvement < 1000 and iteration < max_iter:
        # Generate candidate solution
        h_candidate = perturb_solution(h, n_points)

        h_candidate = np.clip(h_candidate, 0, 1)

        current_c5 = compute_c5_exact(h, dx)

        # Accept or reject the candidate
        if candidate_c5 < current_c5:
            h = h_candidate
            current_c5 = candidate_c5
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.random.random() < math.exp(-delta / temperature):

        if current_c5 < best_c5:
