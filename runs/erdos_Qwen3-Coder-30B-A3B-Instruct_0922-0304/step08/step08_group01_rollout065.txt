import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # LP-based refinement approach
    # Set up the problem with CVXPY
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try LP relaxation approach if it hasn't failed yet
        try:
            # Create CVXPY variables and constraints
            h_relaxed = cp.Variable(n_points)
            z = cp.Variable()
            
            # Objective
            objective = cp.Minimize(z)
            
            # Constraints:
            # Sum constraint
            constraints = [cp.sum(h_relaxed) == target_sum,
                           h_relaxed >= 0,
                           h_relaxed <= 1,
                           z >= 0]
            
            # Overlap constraints:
            # Generate all valid shift indices
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    # Shift k >= 0
                    expr = 0
                    for i in range(n_points - k):
                        expr += h_relaxed[i] * (1 - h_relaxed[i + k])
                    constraints.append(expr * dx <= z)
                else:
                    # Shift k < 0
                    expr = 0
                    for i in range(-k, n_points):
                        expr += h_relaxed[i] * (1 - h_relaxed[i + k])
                    constraints.append(expr * dx <= z)
            
            # Create and solve problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=30)
            
            # Get relaxed solution
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed_values = h_relaxed.value
                if h_relaxed_values is not None:
                    # Rounding heuristic
                    median_val = np.median(h_relaxed_values)
                    h_binary = (h_relaxed_values >= median_val).astype(float)
                    h_binary = normalize_h(h_binary, target_sum)
                    current_c5 = compute_c5(h_binary, dx)
                    
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_binary.copy()
                    
                    # If target achieved, early stop
                    if best_c5 <= 0.3807:
                        break
                        
        except Exception as e:
            # Fall back to gradient method if needed
            pass
        
        # Continue with gradient method
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Identify the maximum conflict indices
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            gradients[i] = grad
        
        # Use top 10% of coordinates for updates
        idxs = np.argsort(np.abs(gradients))[::-1]
        update_count = max(1, n_points // 10)
        selected_idxs = idxs[:update_count]
        
        # Update selected coordinates
        for i in selected_idxs:
            old_h_i = h[i]
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            
            step_size = 0.01
            if grad != 0:
                step_size = min(1.0, max(0.0, -grad * 0.01))
            
            h[i] = h[i] + step_size * (1 - 2 * h[i])
            h[i] = np.clip(h[i], 0, 1)
            
            # Renormalize
            h = normalize_h(h, target_sum)
            
            # Check improvement
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
                
        iter_count += 1
        
        # Occasionally recompute for precision
        if iter_count % 1000 == 0:
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
