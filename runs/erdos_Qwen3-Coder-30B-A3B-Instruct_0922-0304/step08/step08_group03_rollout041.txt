import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Use a triangular construction for initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = 2 * i / (mid - 1)
            else:
                h[i] = 2 * (n_points - 1 - i) / (mid - 1)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    # Used for finding good improvement directions
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use a simpler, more robust gradient-based approach
    max_iter = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Use a fixed step size for updates
    step_size = 0.005
    
    # Approach:
    # 1. Compute gradients for all coordinates.
    # 2. Update in the direction that decreases the maximum overlap.
    # 3. Project back to feasible region.
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient estimates for each coordinate
        gradients = np.zeros(n_points)
        
        for i in range(n_points):
            # Perturb h[i] slightly
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            # Estimate gradient
            gradients[i] = (max_plus - max_minus) / 0.02
            
        # Update in the direction that minimizes overlap
        # Move in direction -gradient, but do not violate bounds
        h_new = h.copy()
        for i in range(n_points):
            # Adjustment based on gradient
            new_val = h[i] - step_size * gradients[i]
            h_new[i] = np.clip(new_val, 0, 1)
            
        # Renormalize to maintain integral constraint
        h_new = normalize_h(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        
        # Check if this improves the solution
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5:
            h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            # Perturb lightly to escape local minimum
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
