import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 50  # Reduced iterations for time constraint
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Relaxation stage - solving a simplified convex relaxation
        # We'll use a gradient-based approach with line search
        h_old = h.copy()
        
        # Generate gradient estimates via finite differences to explore
        # Update each component with a descent approach using simple gradient approximation
        for i in range(n_points):
            # Estimate gradient using finite differences
            eps = 1e-4
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Renormalize
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            val_plus = compute_c5(h_plus, dx)
            val_minus = compute_c5(h_minus, dx)
            
            grad = (val_plus - val_minus) / (2 * eps)
            
            # Step size selection
            alpha = 0.1 / (1.0 + iter_count)
            
            # Descent update
            h[i] = h[i] - alpha * grad
            
            # Clip and renormalize
            h[i] = np.clip(h[i], 0, 1)
            h = h * (target_sum / np.sum(h))
        
        # Check improvement
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
