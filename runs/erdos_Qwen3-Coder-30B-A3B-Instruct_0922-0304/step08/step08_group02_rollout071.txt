```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure it's in [0,1] and sums to n_points/2."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of valid shifts for the correlation."""
    # Sample shifts in [-1, 1] corresponding to indices in [0, 2*n-2]
    shifts = np.random.randint(0, 2*n-1, sample_size)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Convert to CVXPY variables
    h = Variable(n_points)
    constraints = [h >= 0, h <= 1, sum(h) == target_sum]
    
    # Initialize target overlap to ~0.38085
    target_overlap = 0.38085
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 1:
        # Sample shifts for QP formulation
        shifts = sample_shifts(n_points, 50)
        
        # Solve QP to reduce overlap
        # Create constraints for sampled shifts
        # We minimize a variable that upper bounds all relevant overlaps
        max_overlap = Variable()
        
        # Generate the constraints for overlaps
        cons = [max_overlap >= 0]  # ensure we're minimizing non-negative
        for k in shifts:
            if k < n_points:
                # Left overlap
                overlap_k = sum(h[i] * (1 - h[i + k]) for i in range(n_points - k))
            else:
                # Right overlap with padding
                overlap_k = sum(h[i] * (1 - h[i + k - n_points]) for i in range(n_points - (k - n_points + 1)))
            
            cons.append(overlap_k <= max_overlap)
        
        # Add constraints to our list
        prob = cp.Problem(cp.Minimize(max_overlap), constraints + cons)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            # Fallback to simple projection if QP fails
            h_new = h_current.copy()
            h_new = project_h(h_new, n_points)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h_current = h_new.copy()
            iteration += 1
            continue
        
        # Check if solution is feasible
        if prob.status != cp.OPTIMAL:
            # If QP fails, use simple projection
            h_new = h_current.copy()
            h_new = project_h(h_new, n_points)
        else:
            h_new = np.array(h.value).flatten()
        
        # Renormalize to target sum
        h_new = project_h(h_new, n_points)
        
        # Evaluate the new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adaptively adjust the target overlap based on progress
        if no_improvement_count > 50:
            target_overlap *= 0.99
            no_improvement_count = 0
            
        # Update current solution
        h_current = h_new.copy()
        iteration += 1
        
        # Stop condition
        if best_c5 <= 0.3807:
            break
            
        # Break if no improvement in while loop
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final validation and return
    final_h = project_h(best_h, n_points)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
