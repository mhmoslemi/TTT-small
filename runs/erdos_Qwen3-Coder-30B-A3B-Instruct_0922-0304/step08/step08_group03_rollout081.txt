import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available, else triangular ramp
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
        
    # Ensure initial h is feasible
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Gradient descent parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    momentum = np.zeros(n_points)
    
    # Coordinate-wise search parameters
    coord_sample_rate = 0.2
    coord_update_count = max(1, int(n_points * coord_sample_rate))
    
    iter_count = 0
    max_iter = 20000
    stall_count = 0
    max_stall = 100
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Phase 1: Gradient descent with momentum
        if stall_count < 50:
            # Find position of maximum overlap
            k_max = np.argmax(overlap)
            
            # Compute gradient estimate
            h_shifted = np.zeros_like(h)
            h_shifted[k_max:] = h[:-k_max] if k_max > 0 else h[:]
            gradient = (1 - 2 * h) * (1 - h_shifted)
            
            # Apply momentum update
            momentum = beta * momentum + (1 - beta) * gradient
            h_candidate = h - alpha * momentum
            
            # Project to feasible region
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            new_c5 = compute_c5(h_candidate, dx)
            
            if new_c5 < current_max:
                h = h_candidate.copy()
                alpha *= gamma
                stall_count = 0
            else:
                alpha *= gamma
                stall_count += 1
        else:
            # Phase 2: Coordinate-wise search
            # Sample coordinates
            indices = np.random.choice(n_points, size=coord_update_count, replace=False)
            
            for i in indices:
                old_val = h[i]
                
                # Try to decrease overlap by small step in both directions
                best_h_candidate = h.copy()
                best_new_c5 = current_max
                
                for step in [0.01, -0.01]:
                    h_candidate = h.copy()
                    h_candidate[i] = np.clip(old_val + step, 0, 1)
                    
                    # Renormalize
                    h_candidate = clip_and_renormalize(h_candidate, target_sum)
                    new_c5 = compute_c5(h_candidate, dx)
                    
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_h_candidate = h_candidate.copy()
                
                # Use the best candidate if it improves
                if best_new_c5 < current_max:
                    h = best_h_candidate.copy()
                    break
                    
        iter_count += 1
    
    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
