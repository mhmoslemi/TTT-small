import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 500:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Get the lag of maximum overlap
        lag = max_idx - (n_points - 1)
        
        # Adjust to reduce this overlap
        # Find positions that contribute to this overlap
        # The effect of changing h[i] on overlap at lag k is approximately:
        # delta_overlap ≈ -2*dx*h[i] (when i+k is in correct range)
        
        # Simplified adjustment: reduce h at positions that increase overlap at this lag
        adjustment = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at lag k, we consider contribution of h[i] to 1-h[i+k]
            if 0 <= i + lag < n_points:
                adjustment[i] -= 1  # Reducing h[i] reduces overlap at lag
            if 0 <= i - lag < n_points:
                adjustment[i] -= 1  # Reducing h[i] reduces overlap at lag
            if 0 <= i + lag < n_points:
                adjustment[i + lag] += 1  # Increasing h[i+lag] reduces overlap
        
        # Normalize adjustment direction
        norm = np.linalg.norm(adjustment)
        if norm > 0:
            adjustment = adjustment / norm
        
        # Gradient step
        step_size = epsilon
        h_new = h - step_size * adjustment
        
        # Ensure h stays in [0,1] and sum == target_sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        h = h_new
        
        # Update epsilon
        if iteration % 50 == 0 and epsilon > 0.0001:
            epsilon *= 0.95
            
        iteration += 1
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
