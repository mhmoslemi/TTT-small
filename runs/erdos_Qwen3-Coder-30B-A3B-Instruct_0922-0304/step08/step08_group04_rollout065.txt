```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Perform pattern search with partitioning
    # For each segment count from 3 to 10
    for n_segments in range(3, 11):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 5:
            break
            
        # Create breakpoints
        breakpoints = np.linspace(0, 2, n_segments + 1)
        
        # Grid search for each segment
        step_size = 0.05
        grid_points = np.arange(0, 1.001, step_size)
        
        if n_segments <= 5:
            search_granularity = 0.05
        else:
            search_granularity = 0.02
            
        grid_points = np.arange(0, 1.001, search_granularity)
        
        # Try all combinations of segment values (limited for computational efficiency)
        trial_combinations = []
        n_combinations = min(1000, len(grid_points) ** n_segments)
        
        # Instead of full cartesian product, sample randomly
        for _ in range(min(n_combinations, 500)):
            segment_values = np.random.choice(grid_points, size=n_segments)
            trial_combinations.append(segment_values)
            
        # Also include some structured attempts
        for _ in range(100):
            # Try corners: all 0s, all 1s, some 0s 1s
            # Uniform with some zeros
            segment_values = np.random.choice([0.0, 0.5, 1.0], size=n_segments)
            trial_combinations.append(segment_values)
            
        # Evaluate combinations
        for segment_values in trial_combinations:
            # Reconstruct h from segments
            segment_h = np.zeros(n_points)
            for i in range(n_segments):
                start_idx = int(breakpoints[i] / dx)
                end_idx = int(breakpoints[i+1] / dx)
                if end_idx > start_idx:
                    segment_h[start_idx:end_idx] = segment_values[i]
            
            # Make sure we fill the whole array
            if start_idx < n_points:
                segment_h[start_idx:] = segment_values[-1]
                
            # Normalize
            segment_h = normalize_h(segment_h, n_points / 2)
            segment_h = np.clip(segment_h, 0, 1)
            
            # Evaluate
            c5 = evaluate_c5(segment_h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = segment_h.copy()
                
            # Early termination if target reached
            if best_c5 <= 0.3807:
                break
                
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    actual_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), actual_c5, n_points)
```
