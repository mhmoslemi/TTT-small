import numpy as np
import time
from scipy.signal import correlate

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
    else:
        # Initialize with a structured pattern + noise
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid//2:3*mid//2] = 1.0
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-6
    patience = 0
    max_patience = 500
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Simulated Annealing step
        candidate_h = h.copy()
        
        # Randomly perturb 10% of entries
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            if np.random.random() < 0.5:
                # Flip with small perturbation
                candidate_h[i] *= 0.97
            else:
                # Swap with random position
                j = np.random.randint(0, n_points)
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Ensure constraints
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = normalize_h(candidate_h, target_sum)
        
        # Evaluate candidate
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5 or np.random.random() < np.exp(-(candidate_c5 - best_c5) / max(T, 1e-10)):
            h = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Cut-plane refinement (once in a while)
        if iteration % 50 == 0:
            # Analyze the worst overlap lag
            overlap = correlate(h, 1 - h, mode="full")
            k_max = np.argmax(overlap)
            
            # Modify h to reduce worst overlap
            # We can make small adjustments near worst lag
            adjustment = 0.005
            if k_max < len(h):
                # Focus on entries that contribute to worst overlap
                for i in range(max(0, k_max - 2), min(len(h), k_max + 3)):
                    if i < len(h):
                        h[i] -= adjustment
                        h[(i + k_max) % len(h)] += adjustment
                        
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Recompute and update best if needed
            current_c5 = compute_c5_exact(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Annealing cooling
        if T > min_T:
            T *= cooling_rate
            
        # Early termination if good enough
        if best_c5 <= 0.3807:
            break
            
        # If no improvement for too long, terminate
        if patience > max_patience:
            break
    
    # Final check
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
