import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, noise_magnitude):
    """Mutate an individual by adding random noise."""
    noise = np.random.normal(0, noise_magnitude, len(h))
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, len(h) / 2)
    return h_new

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    crossover_point = random.randint(1, len(p1) - 1)
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    
    return child1, child2

def select_tournament(population, fitnesses, tournament_size=2):
    """Select individuals using tournament selection."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = random.sample(range(len(population)), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_index].copy())
    return selected

def evaluate_population(population, dx):
    """Evaluate all individuals in the population."""
    fitnesses = Parallel(n_jobs=1)(
        delayed(evaluate_c5)(ind, dx) for ind in population
    )
    return fitnesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.3
    noise_magnitude = 0.05
    tournament_size = 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            individual = mutate_individual(parent, noise_magnitude)
            population.append(individual)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, n_points / 2)
            population.append(h)
    
    best_fitness = float('inf')
    best_individual = None
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = evaluate_population(population, dx)
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
            
        if best_fitness <= 0.3807:
            break
            
        # Selection
        selected = select_tournament(population, fitnesses, tournament_size)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            p1, p2 = selected[i], selected[i+1] if i+1 < population_size else selected[0]
            c1, c2 = crossover_parents(p1, p2, crossover_rate)
            
            if random.random() < mutation_rate:
                c1 = mutate_individual(c1, noise_magnitude)
            if random.random() < mutation_rate:
                c2 = mutate_individual(c2, noise_magnitude)
                
            new_population.extend([c1, c2])
            
        population = new_population[:population_size]
        
        # Update noise magnitude (decay)
        noise_magnitude *= 0.99
        if noise_magnitude < 1e-6:
            noise_magnitude = 1e-6
    
    # Final evaluation of best individual
    if best_individual is not None:
        final_c5 = evaluate_c5(best_individual, dx)
        if final_c5 < best_fitness:
            best_fitness = final_c5
    
    return (best_individual.tolist(), best_fitness, n_points)
