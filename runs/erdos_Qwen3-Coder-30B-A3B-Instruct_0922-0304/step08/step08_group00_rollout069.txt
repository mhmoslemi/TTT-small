import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Gradient descent with Nesterov accelerated updates
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    max_iter = 100000
    improvement_tolerance = 1e-6
    
    # For Nesterov momentum
    v = np.zeros(n_points)
    h_prev = h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute overlap for all lags to find max
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k] * dx
        
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
            
        # Compute gradient: d/dh_i of max overlap
        # This is simplified estimate of gradient
        # For each i, the contribution to max overlap is roughly (1-h_j) 
        # where j = (i + k) % n and k is the lag that achieves max overlap.
        # But for computation, we compute approximate gradient:
        # dC/dh_i = sum over all k of dC_k/dh_i
        # where dC_k/dh_i = (1-h_{i+k}) in original domains
        
        # Compute derivative using the full overlap array logic
        grad = np.zeros(n_points)
        # This is an approximate gradient using only max overlap lag
        max_lag = max_k - (n_points - 1)  # convert full convolution index to actual lag
        # For each i, dC/dh_i = 1 - h_{i+lag} (if lag is from max overlap)
        lag_indices = np.arange(n_points) + max_lag
        lag_indices = lag_indices % n_points
        grad = 1 - h[lag_indices]  # approximate gradient
        
        # Apply Nesterov momentum
        v = momentum * v - step_size * grad
        h_new = h_prev + v  # new h estimate
        
        # Project onto feasible set [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Update for next iteration
        h_prev = h.copy()
        h = h_new.copy()
        
        # Update step size if improvement is small
        if iteration % 10 == 0:
            # Check if we're stuck or making progress
            current_c5 = evaluate_overlap(h, dx)
            if abs(current_c5 - best_c5) < improvement_tolerance:
                step_size = max(step_size * 0.9, min_step_size)
            elif current_c5 < best_c5:
                step_size = min(step_size * 1.1, 0.5)  # increase step size
            else:
                step_size = max(step_size * 0.9, min_step_size)
                
        # Occasionally check if improving
        if iteration % 100 == 0:
            current_c5 = evaluate_overlap(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    # Final validate, recompute overlap for best solution
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
