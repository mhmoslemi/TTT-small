import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    h = np.minimum(x, 2 - x)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Initialize momentum
    velocity = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.001
    decay_rate = 0.99
    threshold = 1e-6
    
    # Phase 1: Gradient descent with momentum
    max_iter_phase1 = 10000
    iter_count = 0
    early_stop = 0
    max_early_stop = 500
    
    while iter_count < max_iter_phase1 and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Find the index where maximum overlap occurs
        max_idx = np.argmax(overlap)
        # Since mode="full", overlap has length 2*n_points-1
        # The maximum will be at some center shift
        shift = max_idx - (n_points - 1)
        
        # Compute the gradient roughly by finite differences
        # We approximate gradient with a shifted version of h
        if shift >= 0:
            shifted_h = np.roll(h, -shift)
        else:
            shifted_h = np.roll(h, -shift)
            
        # For the exact gradient, we would use the derivative of the correlation
        # But here we use the approximate: gradient = (1 - 2h) * (1 - shift_k(h))
        gradient = (1 - 2 * h) * (1 - shifted_h)
        
        # Apply momentum
        velocity = momentum * velocity + (1 - momentum) * gradient
        
        # Take step
        h_candidate = h - learning_rate * velocity
        
        # Project to feasible set
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new overlap
        new_overlap = correlate(h_candidate, 1 - h_candidate, mode="full") * dx
        new_max = np.max(new_overlap)
        
        if new_max < current_max:
            h = h_candidate.copy()
            learning_rate *= decay_rate
            early_stop = 0
        else:
            early_stop += 1
            learning_rate *= decay_rate
            if learning_rate < threshold:
                break
                
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        iter_count += 1
        
        if early_stop > max_early_stop:
            break
            
    # Phase 2: Coordinate-wise updates with adaptive sampling
    if time.time() - start_time < budget_s - 1:
        max_iter_phase2 = 10000
        iter_count = 0
        keep_improving = True
        improvement_tolerance = 1e-6
        
        while iter_count < max_iter_phase2 and time.time() - start_time < budget_s - 1:
            # Sample coordinates to update
            num_update = max(1, n_points // 5)  # Update about 20% of coordinates
            indices = np.random.choice(n_points, num_update, replace=False)
            updated = False
            
            for i in indices:
                # Save current h
                current_h = h[i]
                
                # Compute approximate gradient for this coordinate
                # We do this by finite differences
                h_plus = h.copy()
                h_plus[i] = min(1.0, current_h + 0.01)
                h_plus = clip_and_renormalize(h_plus, target_sum)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                max_plus = np.max(overlap_plus)
                
                h_minus = h.copy()
                h_minus[i] = max(0.0, current_h - 0.01)
                h_minus = clip_and_renormalize(h_minus, target_sum)
                
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
                max_minus = np.max(overlap_minus)
                
                # Estimate gradient
                grad = (max_plus - max_minus) / 0.02
                
                # Take a small step
                step_size = 0.01
                if grad < 0:
                    step = min(step_size, current_h)
                    new_h = current_h - step
                else:
                    step = min(step_size, 1 - current_h)
                    new_h = current_h + step
                
                h_new = h.copy()
                h_new[i] = new_h
                h_new = clip_and_renormalize(h_new, target_sum)
                
                new_c5 = compute_c5(h_new, dx)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    best_h = h.copy()
                    updated = True
                    if best_c5 <= 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
                elif new_c5 < current_max:
                    h = h_new.copy()
                    updated = True
                    
            if not updated:
                break
                
            iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
