import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    
    # Find all lags that achieve maximum overlap
    active_lags = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each active lag, compute contribution to gradient
    for lag_idx in active_lags:
        # Lag index translates to actual shift k in [0, n-1]
        # But full convolution mode gives indices in [0, 2n-2]
        k = lag_idx - (n - 1)
        
        # Compute contribution for each h[i]
        for i in range(n):
            # Position j = i + k in the second function
            j = i + k
            if 0 <= j < n:
                # Contribution to position i: (1-h[j]) * sign(1-h[j]) = 1-h[j]
                grad[i] += (1 - h[j]) * dx
                
    # Normalize by number of active lags and wide to avoid division by zero
    if len(active_lags) > 0:
        grad = grad / len(active_lags)
    
    # Apply constraint gradient (sum = target_sum)
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum = 0.9
    v = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 500:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum
        v = momentum * v + grad
        h = h - learning_rate * v
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.99
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
