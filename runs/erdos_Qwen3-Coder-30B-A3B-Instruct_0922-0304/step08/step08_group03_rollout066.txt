import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a simple symmetric piecewise constant function
        h = np.ones(n_points) * 0.5
        # Set a sharp peak in the middle to reduce overlap
        mid_idx = n_points // 2
        h[mid_idx] = 1.0
        h[mid_idx-1] = 0.5
        h[mid_idx+1] = 0.5
    
    # Normalize to desired integral
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    # Use the best known solution as default
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Early return if already good enough
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Try several random restarts
    n_restarts = 5
    restart_interval = (budget_s - 5) / n_restarts
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute time slice for this restart
        restart_time = time.time()
        
        # Initialize a new random starting point
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        
        # Local optimization loop
        max_iter = 10000
        iter_count = 0
        decay_factor = 0.99
        adaptivity = 1.0
        no_improvement = 0
        max_no_improvement = 200
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 1:
            if time.time() - restart_time > restart_interval:
                break
                
            current_c5 = compute_c5(h, dx)
            
            # Keep the best solution so far
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
            
            # Compute approximate gradient
            grad = np.zeros(n_points)
            eps = 0.001
            
            for i in range(n_points):
                # Perturb h[i] up and down
                h_up = h.copy()
                h_up[i] = min(1.0, h[i] + eps)
                h_up = normalize_h(h_up, target_sum)
                h_up = np.clip(h_up, 0, 1)
                
                h_down = h.copy()
                h_down[i] = max(0.0, h[i] - eps)
                h_down = normalize_h(h_down, target_sum)
                h_down = np.clip(h_down, 0, 1)
                
                up_c5 = compute_c5(h_up, dx)
                down_c5 = compute_c5(h_down, dx)
                
                grad[i] = (up_c5 - down_c5) / (2 * eps)
            
            # Update using projected gradient descent
            step_size = 0.01 * adaptivity
            
            # Apply gradient update
            h_new = h - step_size * grad
            
            # Project to [0, 1] and renormalize
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            # Check if we improved
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                adaptivity *= decay_factor
                
            if no_improvement > max_no_improvement:
                break
                
            iter_count += 1
            
            # Occasionally make a more aggressive change
            if iter_count % 100 == 0 and np.random.rand() < 0.3:
                h = np.random.rand(n_points)
                h = normalize_h(h, target_sum)
                h = np.clip(h, 0, 1)
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
