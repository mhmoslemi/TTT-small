import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_r, target_sum):
    """Round relaxed solution to binary with target sum."""
    # Round to 0 or 1 based on threshold
    threshold = np.median(h_r)
    h_b = (h_r >= threshold).astype(float)
    
    # Adjust to match target sum
    current_sum = np.sum(h_b)
    diff = current_sum - target_sum
    
    if diff > 0:
        # Need to flip some 1s to 0s
        # Find the indices with smallest values (closest to threshold)
        idxs = np.argsort(np.abs(h_r - threshold))
        for i in idxs:
            if diff <= 0:
                break
            if h_b[i] == 1:
                h_b[i] = 0
                diff -= 1
    elif diff < 0:
        # Need to flip some 0s to 1s
        idxs = np.argsort(np.abs(h_r - threshold))[::-1]
        for i in idxs:
            if diff >= 0:
                break
            if h_b[i] == 0:
                h_b[i] = 1
                diff += 1
    
    return h_b

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Use iterative LP relaxation approach for better explorations
    max_iter = 50
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Solve relaxed LP problem
        h_r = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [0 <= h_r[i] <= 1 for i in range(n_points)]
        constraints.append(cp.sum(h_r) == target_sum)
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                expr = cp.sum(cp.multiply(h_r[:n_points-k], 1 - h_r[k:]))
            else:
                expr = cp.sum(cp.multiply(h_r[-k:], 1 - h_r[:n_points+k]))
            constraints.append(dx * expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_r_sol = h_r.value
                h_b = round_to_binary(h_r_sol, target_sum)
                
                # Compute overlap for binary version
                new_c5 = compute_c5(h_b, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_b.copy()
        except:
            pass
        
        # Simple gradient-based update to escape local minima
        h = best_h.copy()
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            gradients[i] = grad
        
        # Gradient descent step
        step_size = 0.005
        h_new = h + step_size * gradients
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        iter_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
