import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Track the best solution
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient descent with Nesterov momentum
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    
    # For momentum
    v = np.zeros(n_points)
    y = h.copy()
    
    # Compute initial overlap and gradient index
    def compute_gradient_and_max_overlap(h):
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx] * dx
        
        # Compute gradient of the maximum overlap
        # In the full convolution, the index max_idx represents the shift k
        # The term h[i] * (1 - h[i + k]) contributes to overlap
        gradient = np.zeros(n_points)
        
        # Use the shifted indices for gradient computation
        # We are differentiating the max overlap term
        k_shift = max_idx - (n_points - 1)
        # Shift the indices accordingly to reach the right k
        for i in range(n_points):
            j = (i - k_shift) % n_points
            gradient[i] = 1 - h[j]
        
        return gradient, max_val, k_shift
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient and max overlap
        gradient, current_c5, k_shift = compute_gradient_and_max_overlap(y)
        
        # Update with projected gradient descent and Nesterov momentum
        v = momentum * v - step_size * gradient
        y = y + v
        
        # Clip to [0, 1]
        y = np.clip(y, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(y)
        if current_sum > 0:
            y = y * (target_sum / current_sum)
        
        # Evaluate current solution
        current_c5 = evaluate_overlap(y, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = y.copy()
            best_c5 = current_c5
            
        # Adaptive step size
        if i % 1000 == 0 and i > 0 and step_size > min_step_size:
            step_size *= 0.9
            
    # Final refinement using scipy.optimize
    from scipy.optimize import minimize
    
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
