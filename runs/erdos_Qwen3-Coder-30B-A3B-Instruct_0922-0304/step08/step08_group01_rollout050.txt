import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().astype(np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # SLSQP optimizer
    def objective(h):
        return compute_c5(h, dx)
    
    # Run optimization
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 5000, 'ftol': 1e-9, 'gtol': 1e-9})
        if res.success:
            h = res.x
        else:
            # fallback to initial solution if optimization failed
            h = best_h
    except Exception as e:
        # fallback to initial solution if optimization failed
        h = best_h
    
    # Refine with local search if time allows
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 1.0:
        # Local refinement with gradient approximation and line search
        max_iter = min(5000, int(remaining_time * 10))
        iter_count = 0
        while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
            # Compute gradient approximate
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
            indices = np.where(np.abs(overlap) == current_max)[0]
            
            # Approximate gradient for each h[i]
            gradients = np.zeros(n_points)
            for i in range(n_points):
                grad = 0
                for k in range(n_points):
                    if i + k < n_points:
                        grad += (1 - 2 * h[i]) * (1 - h[i + k])
                    if i - k >= 0:
                        grad += (1 - 2 * h[i]) * (1 - h[i - k])
                gradients[i] = grad
            
            # Update h using gradient descent with backtracking line search
            step_size = 0.02
            updated = False
            
            for i in np.argsort(np.abs(gradients))[::-1][:n_points//10]:
                old_h_i = h[i]
                h[i] = h[i] + step_size * gradients[i]
                h[i] = np.clip(h[i], 0, 1)
                
                # Check if update improved C5
                h_sum = np.sum(h)
                if abs(h_sum - n_points/2) > 1e-6:
                    # Re-normalize
                    h = h * (n_points / 2) / h_sum
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    updated = True
                else:
                    h[i] = old_h_i
            
            if not updated:
                step_size *= 0.5  # Reduce step size
            
            iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
