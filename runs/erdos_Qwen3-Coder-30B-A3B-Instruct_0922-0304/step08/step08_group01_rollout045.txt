import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Start with a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.999
    max_iter = 100000
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Create a copy of current solution
        h_new = h.copy()
        
        # Choose a perturbation type
        pert_type = np.random.choice(['flip', 'adjust', 'shuffle'], p=[0.5, 0.3, 0.2])
        
        if pert_type == 'flip':
            # Flip a random element
            i = np.random.randint(n_points)
            h_new[i] = 1 - h_new[i]
        elif pert_type == 'adjust':
            # Small random adjustment
            i = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        elif pert_type == 'shuffle':
            # Shuffle a random segment
            length = np.random.randint(1, min(5, n_points))
            i = np.random.randint(n_points - length + 1)
            segment = h_new[i:i+length]
            np.random.shuffle(segment)
            h_new[i:i+length] = segment
        
        # Ensure constraints are satisfied
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_e = new_c5 - best_c5
        if delta_e < 0 or np.exp(-delta_e / temperature) > np.random.rand():
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
        # Cool down
        temperature *= cooling_rate
        
        # Occasionally reheat if no progress
        if iter_count % 1000 == 0 and iter_count > 0:
            if best_c5 <= 0.3807:
                break
                
        iter_count += 1
        
    # Final computation
    final_c5 = compute_c5(best_h, dx)  
    
    return (best_h.tolist(), final_c5, n_points)
