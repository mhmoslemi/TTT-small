```python
import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # MCMC parameters
    T0 = 0.1
    T_min = 1e-6
    alpha = 0.999
    sigma = 0.05
    tau = 1000
    no_improvement_limit = 500
    no_improvement_count = 0
    
    # Annealing schedule
    T = T0
    iteration = 0
    
    def project_to_feasible(h):
        # Clamp values to [0,1]
        h = np.clip(h, 0, 1)
        # Scale to meet sum constraint
        h = h * (target_sum / np.sum(h))
        return h
    
    while time.time() - start_time < budget_s - 1:
        # Current C5
        c5_current = compute_c5(h_current, dx)
        
        # Propose move
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, sigma)
        h_new[idx] = h_new[idx] + delta
        h_new = project_to_feasible(h_new)
        
        # New C5
        c5_new = compute_c5(h_new, dx)
        
        # Accept/Reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < exp(-delta_c5 / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Adapt temperature and perturbation
        if no_improvement_count > no_improvement_limit:
            sigma *= 0.9
            no_improvement_count = 0
            
        T = max(T_min, T0 * (alpha ** (iteration / tau)))
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
