import numpy as np
from scipy.signal import correlate
import time
from math import exp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h."""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - np.array(h), mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create a good starting point
        h_current = np.zeros(n_points)
        # Place a few bumps to avoid trivial solutions
        h_current[10:20] = 0.7
        h_current[40:50] = 0.3
        h_current = h_current * (target_sum / np.sum(h_current))
    
    # MCMC with simulated annealing
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    n_steps = 2000  # For resetting perturbation
    reset_factor = 0.8
    converge_count = 0
    converge_threshold = 50
    
    c5_current = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    T = T0
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Propose new state
        h_proposed = h_current.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.normal(0, sigma)
        h_proposed[i] += delta
        h_proposed[i] = np.clip(h_proposed[i], 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_proposed = h_proposed * (target_sum / np.sum(h_proposed))
        
        # Compute new C5
        c5_proposed = evaluate_erdos_solution(h_proposed)
        delta_c5 = c5_proposed - c5_current
        
        # Accept or reject
        if delta_c5 < 0 or np.random.random() < exp(-delta_c5 / max(T, T_min)):
            h_current = h_proposed
            c5_current = c5_proposed
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                converge_count = 0
            else:
                converge_count += 1
        else:
            converge_count += 1
        
        # Update temperature
        T = max(T_min, T * alpha)
        
        # Adaptive perturbation adjustment
        if converge_count > converge_threshold:
            sigma *= reset_factor
            converge_count = 0
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
