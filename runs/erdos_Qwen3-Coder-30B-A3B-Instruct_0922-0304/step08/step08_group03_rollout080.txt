import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip values to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n_points):
    """Create a triangular ramp function."""
    half = n_points // 2
    return np.concatenate([np.linspace(0, 1, half), np.linspace(1, 0, n_points - half)])

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Momentum parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    velocity = np.zeros(n_points)
    
    # Early stopping
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    max_iter = 100000
    stall_count = 0
    max_stall = 1000
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute gradient
        # Create shifted version of h for position max_pos
        shift_amount = max_pos - (n_points - 1)
        h_shifted = np.roll(h, shift_amount)
        gradient = (1 - 2 * h) * (1 - h_shifted)
        
        # Apply momentum
        velocity = beta * velocity + (1 - beta) * gradient
        h_candidate = h - alpha * velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept if better
        if new_c5 < overlap_max:
            h = h_candidate.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            alpha *= gamma
            stall_count = 0
        else:
            alpha *= gamma
            stall_count += 1
            if stall_count > max_stall:
                break
        
        iter_count += 1
        
        # Occasionally switch to coordinate-wise optimization
        if iter_count % 1000 == 0:
            h = clip_and_renormalize(h, target_sum)
            if best_c5 <= 0.3807:
                break
            
            # Coordinate-wise refinement
            update_indices = np.random.choice(n_points, size=max(1, n_points // 5), replace=False)
            for i in update_indices:
                # Simple local search for coordinate i
                h_save = h[i]
                step_sizes = [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]
                best_step = 0.0
                best_value = overlap_max
                
                for step in step_sizes:
                    h[i] = h_save + step
                    h[i] = np.clip(h[i], 0, 1)
                    h = clip_and_renormalize(h, target_sum)
                    new_overlap = correlate(h, 1 - h, mode="full") * dx
                    new_max = np.max(new_overlap)
                    if new_max < best_value:
                        best_value = new_max
                        best_step = step
                h[i] = h_save + best_step
                h[i] = np.clip(h[i], 0, 1)
                h = clip_and_renormalize(h, target_sum)
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
