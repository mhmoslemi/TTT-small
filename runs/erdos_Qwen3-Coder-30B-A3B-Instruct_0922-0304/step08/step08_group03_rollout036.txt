import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function for initial guess."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Multi-resolution grid refinement (coarse, medium, fine)
    resolutions = [30, 50, 70]
    
    # Try grid refinement first
    for r in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
        if r > n_points:
            continue
        h_refined = np.interp(np.linspace(0, 2, r), np.linspace(0, 2, n_points), h)
        h_refined = project_h(h_refined, r / 2)
        c5_refined = compute_c5(h_refined, 2.0 / r)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Now optimize with projected gradient descent
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 2000
    lr = 0.001
    momentum = 0.9
    
    v = np.zeros_like(h)  # velocity for momentum
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient approximately using finite differences
        # We consider the influence of each coordinate
        grad = np.zeros_like(h)
        eps = 1e-4
        
        h_plus = h.copy()
        h_minus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = np.clip(h[i] + eps, 0, 1)
            h_minus[i] = np.clip(h[i] - eps, 0, 1)
            
            # Renormalize
            h_plus = project_h(h_plus, target_sum)
            h_minus = project_h(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            grad[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps)
            
            # Reset
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Update with momentum
        v = momentum * v - lr * grad
        h_new = h + v
        
        # Project and renormalize
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < current_max:
            h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
                
        iter_count += 1
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Final refinement with hill climbing for local search
    max_hill_climb_iter = 1000
    for _ in range(max_hill_climb_iter):
        if time.time() - start_time > budget_s - 1:
            break
        improved = False
        for i in np.random.permutation(n_points):  # Random order for exploration
            if time.time() - start_time > budget_s - 1:
                break
            h_trial = h.copy()
            step = np.random.uniform(-0.05, 0.05)
            h_trial[i] = np.clip(h[i] + step, 0, 1)
            h_trial = project_h(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                h = h_trial.copy()
                best_c5 = c5_trial
                improved = True
                break
        if not improved:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
