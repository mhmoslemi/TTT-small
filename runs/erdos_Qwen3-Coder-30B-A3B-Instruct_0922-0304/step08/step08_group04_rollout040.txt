import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Track best solution
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Grid search parameters
    grid_steps = [0.05, 0.02, 0.01]
    
    # Local search with adaptive grid
    max_iter = 50000
    no_improvement = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different segment counts
        n_segments = np.random.randint(2, min(11, n_points//2 + 1))
        
        # Create approximate piecewise approximation
        segment_starts = np.linspace(0, 2, n_segments + 1)
        segment_widths = np.diff(segment_starts)
        
        # Sample h values for each segment
        segment_vals = np.random.rand(n_segments)
        
        # Adjust segment values to satisfy constraint
        total_adjusted = np.sum(segment_vals * segment_widths)
        if total_adjusted > 0:
            segment_vals = segment_vals * (n_points / 2) / total_adjusted
        
        # Interpolate to full h
        h_new = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(segment_starts[i] / dx)
            end_idx = int(segment_starts[i+1] / dx)
            if end_idx > start_idx:
                h_new[start_idx:end_idx] = segment_vals[i]
        
        # Ensure values are clipped to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize the full h to target sum
        h_new = normalize_h(h_new, n_points / 2)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement > max_no_improvement:
            # Occasionally try a different approach to escape local minima
            if np.random.random() < 0.1:
                # Slightly perturb the current best
                h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = normalize_h(h_perturbed, n_points / 2)
                new_c5 = evaluate_c5(h_perturbed, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_perturbed.copy()
                    no_improvement = 0

    # Final validation and return
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    else:
        # If nothing was found, return initial construction
        best_h = h
        best_c5 = evaluate_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
