import numpy as np
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h using correlation."""
    # Using scipy's correlate for full correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain."""
    conj_1_minus_H = 1 - np.conj(H)
    return np.abs(H * conj_1_minus_H)

def spectral_update(H, H_best, learning_rate, freq_indices):
    """Update H in selected frequency bins to reduce surrogate."""
    # Only update selected frequency bins
    H_update = H.copy()
    
    # Apply gradient-like update
    for i in freq_indices:
        # Reduce the magnitude in frequency bins contributing to max surrogate
        H_update[i] = H[i] - learning_rate * np.real(H[i] * (1 - np.conj(H[i])) / (1e-12 + np.abs(H[i])**2 + np.abs(1 - np.conj(H[i]))**2))
    
    # Blend with best transform to keep stability
    H_update = 0.8 * H_update + 0.2 * H_best
    
    return H_update

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from parent construction
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        # Random construction
        h_current = np.random.uniform(0, 1, n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Frequency domain approach
        H = fft(h_current)
        
        # Compute surrogate objective
        surrogate = compute_surrogate(H)
        
        # Find top contributing frequency bins
        # We want to reduce the highest surrogate values
        freq_indices = np.argsort(-surrogate)[:max(1, n_points // 10)]
        
        # Perform spectral update
        H_new = spectral_update(H, H, learning_rate, freq_indices)
        
        # Reconstruct h and project to feasible region
        h_new = np.real(ifft(H_new))
        h_new = project_h(h_new, target_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Update best if improved
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            learning_rate = max(0.001, learning_rate * 0.995)  # Adaptive learning rate
            
        h_current = h_new.copy()
        iteration += 1
        
        # Periodic reset to keep solution stable
        if iteration % 50 == 0:
            H = fft(h_current)
            learning_rate = max(0.001, learning_rate * 0.95)
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
    # Final refinement
    h_final = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_final.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_final, dx):
            h_final = h_temp.copy()
            
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
