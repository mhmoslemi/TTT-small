import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0.0
    # For each possible shift k
    for k in range(n):
        overlap = 0.0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Optimization using SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        result = minimize(objective, h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 50, 'ftol': 1e-10})
        
        if result.success:
            h = np.clip(result.x, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 0:
                h = h * (target_sum / current_sum)
            current_c5 = evaluate_overlap(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Try a small random perturbation if optimization failed
            perturb = np.random.uniform(-1e-4, 1e-4, n_points)
            h_pert = h + perturb
            h_pert = np.clip(h_pert, 0, 1)
            current_sum = np.sum(h_pert)
            if current_sum > 0:
                h_pert = h_pert * (target_sum / current_sum)
            current_c5 = evaluate_overlap(h_pert, dx)
            if current_c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = current_c5
            h = h_pert
        
        iter_count += 1
        
        # Early stopping if no improvement over last 100 iterations
        # (improvement criterion would be slightly more complex)
    
    # Local refinement
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-10})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
