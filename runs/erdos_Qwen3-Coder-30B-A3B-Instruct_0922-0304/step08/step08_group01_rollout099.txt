import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.choice([0.0, 1.0], size=n_points, p=[0.5, 0.5])
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Simulated Annealing parameters
    start_temp = 0.1
    cooling_rate = 0.999
    current_temp = start_temp
    max_iter = 100000
    
    iter_count = 0
    last_improvement = 0
    
    def perturb(h):
        """Apply a random perturbation to h."""
        op = np.random.choice(['flip', 'adjust', 'shuffle'], 
                              p=[0.5, 0.3, 0.2])
        
        if op == 'flip':
            i = np.random.randint(len(h))
            h[i] = 1 - h[i]
        elif op == 'adjust':
            i = np.random.randint(len(h))
            delta = np.random.uniform(-0.1, 0.1)
            h[i] += delta
        elif op == 'shuffle':
            length = np.random.randint(1, 5)
            i = np.random.randint(len(h) - length + 1)
            segment = h[i:i+length]
            np.random.shuffle(segment)
            h[i:i+length] = segment
    
    # Main loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Store current state
        current_h = h.copy()
        current_c5 = compute_c5(h, dx)
        
        # Perturb
        perturb(h)
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            # Always accept better solutions
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                last_improvement = iter_count
        else:
            # Accept worse solutions with probability
            delta_e = new_c5 - current_c5
            if np.exp(-delta_e / current_temp) > np.random.rand():
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    last_improvement = iter_count
            else:
                # Revert to previous state if rejected
                h = current_h.copy()
        
        # Adaptive cooling
        if iter_count - last_improvement > 1000:
            current_temp = max(0.01, current_temp * 2)
            last_improvement = iter_count
        
        # Update temperature
        current_temp *= cooling_rate
        
        iter_count += 1
        
        # Occasionally recompute to double check
        if iter_count % 1000 == 0:
            if compute_c5(best_h, dx) < best_c5:
                best_c5 = compute_c5(best_h, dx)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
