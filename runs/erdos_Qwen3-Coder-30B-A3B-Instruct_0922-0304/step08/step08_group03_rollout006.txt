import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_gradient(h, k, dx):
    """Compute approximate gradient for specific overlap position k."""
    # We compute gradient for h[k]
    h_plus = h.copy()
    h_minus = h.copy()
    eps = 1e-4
    
    h_plus[k] = min(1.0, h[k] + eps)
    h_plus = clip_and_renormalize(h_plus, len(h)/2)
    
    h_minus[k] = max(0.0, h[k] - eps)
    h_minus = clip_and_renormalize(h_minus, len(h)/2)
    
    overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
    overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
    
    max_plus = np.max(overlap_plus)
    max_minus = np.max(overlap_minus)
    
    return (max_plus - max_minus) / (2 * eps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Phase 1: Gradient-based with momentum
    momentum = 0.9
    alpha = 0.001
    gamma = 0.995
    velocity = np.zeros(n_points)
    
    # Phase 2: Coordinate-wise sampling
    p = 0.2  # Proportion to sample
    no_improvement_count = 0
    max_no_improvement = 500
    
    iter_count = 0
    phase = 1
    
    while iter_count < 20000 and time.time() - start_time < budget_s - 1:
        if phase == 1:
            # Gradient descent
            overlap = correlate(h, 1 - h, mode="full") * dx
            max_idx = np.argmax(overlap)
            max_val = overlap[max_idx]
            
            # Compute proxy gradient
            h_shifted = np.roll(h, max_idx - n_points + 1)
            grad = (1 - 2 * h) * (1 - h_shifted)
            
            velocity = momentum * velocity + (1 - momentum) * grad
            h_candidate = h - alpha * velocity
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            
            new_c5 = compute_c5(h_candidate, dx)
            
            if new_c5 < best_c5:
                h = h_candidate.copy()
                best_c5 = new_c5
                best_h = h.copy()
                alpha *= gamma
                no_improvement_count = 0
            else:
                h = clip_and_renormalize(h, target_sum)
                alpha *= gamma
                no_improvement_count += 1
                
            if no_improvement_count > max_no_improvement:
                phase = 2
                no_improvement_count = 0
                
        elif phase == 2:
            # Coordinate-wise updates
            num_to_update = max(1, int(p * n_points))
            indices = np.random.choice(n_points, num_to_update, replace=False)
            improved = False
            
            for i in indices:
                # Compute local gradient
                grad = compute_gradient(h, i, dx)
                
                # Update step
                step_size = 0.001 * np.sign(grad)
                h_new = h.copy()
                h_new[i] = np.clip(h[i] - step_size, 0, 1)
                h_new = clip_and_renormalize(h_new, target_sum)
                
                new_c5 = compute_c5(h_new, dx)
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    best_h = h.copy()
                    improved = True
                    
            if not improved:
                no_improvement_count += 1
                if no_improvement_count > max_no_improvement:
                    break
            else:
                no_improvement_count = 0

        iter_count += 1

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
