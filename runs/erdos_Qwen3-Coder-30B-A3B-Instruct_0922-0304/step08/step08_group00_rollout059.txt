import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    # Normalize sum constraint exactly
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Define the optimization problem
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Define constraints
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try SLSQP optimization
    max_iterations = 1000
    iteration_count = 0
    
    while iteration_count < max_iterations and time.time() - start_time < budget_s - 5:
        try:
            # Run optimization
            result = minimize(objective, h, method='SLSQP',
                            bounds=bounds, constraints=constraints,
                            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
                    h = refined_h
                else:
                    # Perturb slightly for diversity
                    h = best_h.copy()
                    perturbation = np.random.normal(0, 1e-4, n_points)
                    h += perturbation
                    h = np.clip(h, 0, 1)
                    current_sum = np.sum(h)
                    if current_sum > 0:
                        h = h * (target_sum / current_sum)
            else:
                # If optimization fails, try another perturbation and restart
                h = best_h.copy()
                perturbation = np.random.normal(0, 1e-4, n_points)
                h += perturbation
                h = np.clip(h, 0, 1)
                current_sum = np.sum(h)
                if current_sum > 0:
                    h = h * (target_sum / current_sum)
                    
        except Exception:
            # For any error, restart with best solution
            h = best_h.copy()
        
        iteration_count += 1
    
    # Final validation and refinement
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
