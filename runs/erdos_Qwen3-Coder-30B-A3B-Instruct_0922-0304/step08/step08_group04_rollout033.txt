import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform uniform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate h by adding noise."""
    if random.random() > mutation_rate:
        return h.copy()
    
    noise = np.random.normal(0, noise_magnitude, len(h))
    h_mut = h + noise
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual via tournament selection."""
    indices = random.sample(range(len(population)), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def genetic_algorithm(n_points, population_size=50, generations=1000, 
                      crossover_rate=0.8, mutation_rate=0.3, 
                      initial_h=None, budget_s=600):
    """Genetic algorithm for solving the Erdős minimum overlap problem."""
    start_time = time.time()
    
    # Initialize population
    if initial_h is not None:
        # Use initial solution as base, add noise to create diversity
        base_h = np.array(initial_h)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            noise_h = mutate(base_h, mutation_rate, 0.05)
            population.append(noise_h)
    else:
        # Random initialization
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, n_points / 2)
            population.append(h)
    
    dx = 2.0 / n_points
    best_solution = None
    best_fitness = float('inf')
    
    for generation in range(generations):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of population
        fitnesses = Parallel(n_jobs=1)(
            delayed(evaluate_c5)(h, dx) for h in population
        )
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_solution = population[min_fitness_idx].copy()
            
        if best_fitness <= 0.3807:
            break
            
        # Selection, crossover, and mutation
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            child1 = mutate(child1, mutation_rate, 0.05 * (0.99 ** (generation // 10)))
            child2 = mutate(child2, mutation_rate, 0.05 * (0.99 ** (generation // 10)))
            new_population.extend([child1, child2])
            
        # Re-normalize and ensure feasibility
        for i, h in enumerate(new_population):
            h = np.clip(h, 0, 1)
            h = normalize_h(h, n_points / 2)
            new_population[i] = h
            
        population = new_population
    
    return best_solution, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    
    # Initialize from the parent construction if available
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(
        n_points, 
        population_size=50, 
        generations=1000,
        initial_h=initial_h,
        budget_s=budget_s
    )
    
    # Final evaluation for exact value
    if best_h is not None:
        dx = 2.0 / n_points
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    # Ensure the result is feasible
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, n_points / 2)
    
    return (best_h.tolist() if best_h is not None else [0.5] * n_points, 
            best_c5 if best_c5 != float('inf') else 1.0, 
            n_points)
