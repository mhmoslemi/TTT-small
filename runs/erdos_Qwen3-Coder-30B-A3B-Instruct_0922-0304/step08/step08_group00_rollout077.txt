import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try convex optimization approach
    # We'll solve a convex relaxation and use it as seed for local refinement
    
    # Variable definition
    h_var = cp.Variable(n_points)
    
    # Objective:
    # We want to minimize max_k int h(x)(1-h(x+k)) dx = max_k (h * (1-h))[k]
    # where * is convolution
    
    # The overlap integral is the maximum convolution entry
    # This is a non-convex problem, but we can get an approximate solution
    
    # Let's use a different approach: first try optimizing with CVXPY as much as possible
    # Set up the constraint that sum(h) = target_sum
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Note: we have an integer constraint in the objective that's non-convex
    # We can instead sample a candidate solution, then do local refinement
    
    # Determine a good starting point by solving a relaxation
    # We know the sum constraint, and bounds are [0,1]
    
    # Use the existing h as a starting point for local refinement
    
    # Local refinement approach using scipy
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints_eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    # Try different local optimization methods
    methods = ['SLSQP', 'L-BFGS-B']
    for method in methods:
        if time.time() - start_time > budget_s - 10:
            break
        try:
            result = minimize(objective, h, method=method, 
                             bounds=bounds, constraints=constraints_eq, 
                             options={'maxiter': 200, 'ftol': 1e-8})
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
        except:
            continue
    
    # Additional local search if time allows
    # Try to find a better solution by perturbing the best solution
    attempt_count = 0
    while time.time() - start_time < budget_s - 10 and attempt_count < 10:
        # Perturb the solution slightly
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        
        # Try local refinement on the perturbed solution
        try:
            result = minimize(objective, h_perturbed, method='SLSQP', 
                             bounds=bounds, constraints=constraints_eq, 
                             options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
        except:
            pass
        
        attempt_count += 1
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
