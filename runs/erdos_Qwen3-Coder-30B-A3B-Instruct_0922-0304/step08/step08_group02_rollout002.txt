import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to ensure it's in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h with given rate."""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.1)
    return h_mut

def local_optimize(h, target_sum, dx, steps=50):
    """Perform local optimization on h to reduce C5."""
    h_opt = h.copy()
    for _ in range(steps):
        if np.random.random() < 0.7:  # 70% chance of gradient move
            # Simple gradient-like move
            idx = np.random.randint(len(h_opt))
            delta = np.random.normal(0, 0.02)
            h_test = h_opt.copy()
            h_test[idx] += delta
            h_test = repair_h(h_test, target_sum)
            if compute_c5(h_test, dx) < compute_c5(h_opt, dx):
                h_opt = h_test
        else:
            # Random perturbation
            h_test = h_opt.copy()
            idx = np.random.randint(len(h_test))
            h_test[idx] = np.clip(h_test[idx] + np.random.normal(0, 0.1), 0, 1)
            h_test = repair_h(h_test, target_sum)
            if compute_c5(h_test, dx) < compute_c5(h_opt, dx):
                h_opt = h_test
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = repair_h(h_init, target_sum)
        population.append(h_init)
    else:
        # Create a baseline
        h_init = np.ones(n_points) * 0.5
        h_init = repair_h(h_init, target_sum)
        population.append(h_init)
    
    while len(population) < n_pop:
        h = np.random.random(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate population
        fitness = [compute_c5(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Local optimization on top individuals
        top_individuals = population[:3]
        for h in top_individuals:
            h_refined = local_optimize(h, target_sum, dx)
            c5_refined = compute_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        
        # Create new generation
        new_pop = [population[0].copy()]  # Elitism
        
        # Selection (top 20% as parents)
        parents = population[:n_pop // 5]
        
        while len(new_pop) < n_pop:
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, 0.05)
            child1 = repair_h(child1, target_sum)
            child2 = mutate(child2, 0.05)
            child2 = repair_h(child2, target_sum)
            
            new_pop.append(child1)
            if len(new_pop) < n_pop:
                new_pop.append(child2)
        
        # Replace population
        population = new_pop
        generation += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    if best_h is not None:
        final_h = local_optimize(best_h, target_sum, dx, 100)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    if best_h is None:
        # Fallback to the initial best try
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = repair_h(h, target_sum)
        else:
            h = np.ones(n_points) * 0.5
            h = repair_h(h, target_sum)
        best_h = h
        best_c5 = compute_c5(h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
