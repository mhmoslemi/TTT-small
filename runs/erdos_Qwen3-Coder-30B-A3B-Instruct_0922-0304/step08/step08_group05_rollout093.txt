import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Parameters
    max_iter = 100000
    epsilon = 0.01
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    # Iterative refinement
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5_exact(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if best_c5 <= 0.3807:
            break
        
        # Compute overlaps and find the most problematic lag
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_lag = max_idx - (n_points - 1)  # Convert to actual lag
        
        # Perturb h to reduce overlap at this lag
        # Adjust elements around the worst lag
        delta = epsilon
        for i in range(n_points):
            # Adjust in both directions to reduce overlap
            new_h = h.copy()
            
            # Try decreasing h[i], increasing h[i+lag]
            if i + max_lag >= 0 and i + max_lag < n_points:
                new_h[i] = max(0, h[i] - delta)
                new_h[i + max_lag] = min(1, h[i + max_lag] + delta)
                new_h = new_h * (target_sum / np.sum(new_h))
                
                new_c5 = compute_c5_exact(new_h, dx)
                if new_c5 < current_c5:
                    h = new_h
                    break
                    
            # Try decreasing h[i+lag], increasing h[i]
            if i - max_lag >= 0 and i - max_lag < n_points:
                new_h = h.copy()
                new_h[i] = max(0, h[i] - delta)
                new_h[i - max_lag] = min(1, h[i - max_lag] + delta)
                new_h = new_h * (target_sum / np.sum(new_h))
                
                new_c5 = compute_c5_exact(new_h, dx)
                if new_c5 < current_c5:
                    h = new_h
                    break
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Adaptive learning rate
        if last_improvement > 100:
            epsilon *= 0.99
    
    final_c5 = compute_c5_exact(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
