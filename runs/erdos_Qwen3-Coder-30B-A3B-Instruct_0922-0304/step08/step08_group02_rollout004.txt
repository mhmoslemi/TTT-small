import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def is_valid_solution(h, target_sum):
    """Check if solution is valid"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - target_sum) < 1e-10)

def repair_solution(h, target_sum):
    """Repair solution to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents"""
    if np.random.random() > crossover_rate:
        return deepcopy(parent1), deepcopy(parent2)
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate solution"""
    h_mut = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            delta = np.random.normal(0, 0.05)
            h_mut[i] = np.clip(h_mut[i] + delta, 0, 1)
    return h_mut

def genetic_algorithm(seed, budget_s, n_points, n_pop=20, max_gen=1000):
    """Run genetic algorithm"""
    np.random.seed(seed)
    start_time = time.time()
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        population.append(h_init)
    
    # Fill population
    while len(population) < n_pop:
        h = np.random.random(n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    gen = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Elitism
        new_population = population[:n_pop//5]
        
        # Generate offspring
        while len(new_population) < n_pop:
            # Selection (tournament selection)
            parent1_idx = np.random.randint(0, n_pop//2)
            parent2_idx = np.random.randint(0, n_pop//2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, 0.05)
            child2 = mutate(child2, 0.05)
            
            # Repair
            child1 = repair_solution(child1, target_sum)
            child2 = repair_solution(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        population = new_population[:n_pop]
        gen += 1
        if best_c5 <= 0.3807:
            break
    
    # Local refinement
    if best_h is not None:
        h_refined = best_h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Multiple refinements
            for _ in range(50):
                idx = np.random.randint(n_points)
                delta = np.random.normal(0, 0.01)
                h_temp = h_refined.copy()
                h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
                h_temp = h_temp * (target_sum / np.sum(h_temp))
                c5_temp = compute_c5(h_temp, dx)
                if c5_temp < compute_c5(h_refined, dx):
                    h_refined = h_temp.copy()
            # Check if any improvement
            if compute_c5(h_refined, dx) < compute_c5(best_h, dx):
                best_h = h_refined.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    
    h_result, c5_result = genetic_algorithm(seed, budget_s, n_points)
    
    if h_result is None:
        if 'initial_h_values' in globals():
            h_result = np.array(initial_h_values)
            h_result = np.clip(h_result, 0, 1)
            h_result = h_result * (n_points / 2) / np.sum(h_result)
        else:
            h_result = np.ones(n_points) * 0.5
        c5_result = compute_c5(h_result, 2.0 / n_points)
    
    return (h_result.tolist(), c5_result, n_points)
