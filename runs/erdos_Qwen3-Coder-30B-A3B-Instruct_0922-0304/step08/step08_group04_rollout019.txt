import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters
    max_iter = 10000
    patience = 100
    no_improvement = 0
    
    # Search with adaptive steps
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a small random perturbation
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new[idx] += delta
        
        # Keep in bounds
        h_new = np.clip(h_new, 0, 1)
        # Ensure integral constraint
        h_new = normalize_h(h_new, n_points / 2)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept better solution or with some probability (simulated annealing)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
        # Reset if no improvement for a while
        if no_improvement > patience:
            # Reinitialize with better random sample
            h = np.random.rand(n_points).astype(np.float64)
            h = normalize_h(h, n_points / 2)
            no_improvement = 0
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
