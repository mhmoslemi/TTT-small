import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    """Generate a triangular ramp function."""
    x = np.linspace(0, 2, n)
    h = np.minimum(x, 2-x)
    return h / np.sum(h) * (n/2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
    
    # Initial computation
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Hyperparameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    p = 0.2  # Coordinate sampling rate
    max_iter = 20000
    iter_count = 0
    momentum = np.zeros(n_points)
    no_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
            alpha *= gamma  # Decay learning rate
        else:
            no_improvement += 1
            if no_improvement > 500:
                # Switch to coordinate-wise if no progress
                break
        
        # Gradient descent phase
        overlap = correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlap)
        
        # Approximate gradient by finite difference
        eps = 1e-4
        grad = np.zeros(n_points)
        h_plus = h.copy()
        h_minus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            
            h_plus = clip_and_renormalize(h_plus, target_sum)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            grad[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps)
            
            # Restore h
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Update with momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_candidate = h - alpha * momentum
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < current_c5:
            h = h_candidate
        else:
            alpha *= gamma  # Reduce learning rate
            
        iter_count += 1
    
    # Coordinate-wise refinement phase
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Select random coordinates to update
        update_indices = np.random.choice(n_points, size=int(n_points * p), replace=False)
        
        improved = False
        for i in update_indices:
            # Gradient estimation at i
            h_plus = h.copy()
            h_minus = h.copy()
            
            eps = 1e-4
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            
            h_plus = clip_and_renormalize(h_plus, target_sum)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            grad_i = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps)
            
            # Update
            step = 0.001 * grad_i
            h_new = h.copy()
            h_new[i] = np.clip(h[i] - step, 0, 1)
            h_new = clip_and_renormalize(h_new, target_sum)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
                improved = True
        
        if not improved:
            break
            
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
