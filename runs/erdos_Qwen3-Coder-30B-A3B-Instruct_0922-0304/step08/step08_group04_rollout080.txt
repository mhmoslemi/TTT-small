```python
import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def feasibility_repair(h, target_sum):
    """Repair h to stay within bounds and maintain sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def crossover(parent1, parent2, rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, noise_level):
    """Apply mutation to an individual."""
    mutation_mask = np.random.random(len(h)) < 0.3
    noise = np.random.normal(0, noise_level, len(h))
    h = h + noise * mutation_mask
    return h

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx].copy()

def evaluate_population_parallel(population, dx):
    """Evaluate all individuals in population in parallel."""
    fitnesses = Parallel(n_jobs=1)(
        delayed(evaluate_c5)(h, dx) for h in population
    )
    return fitnesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, target_sum)
        population.append(parent)
        # Generate diverse individuals
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            individual = parent + noise
            individual = feasibility_repair(individual, target_sum)
            population.append(individual)
    else:
        # Random initialization
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, target_sum)
            population.append(individual)
    
    best_fitness = float('inf')
    best_individual = None
    
    # GA parameters
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.3
    noise_level = 0.05
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = evaluate_population_parallel(population, dx)
        
        # Track best individual
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
        
        # Early termination
        if best_fitness <= 0.3807:
            break
            
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Selection, crossover, mutation for new generation
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, noise_level)
            child2 = mutate(child2, noise_level)
            
            child1 = feasibility_repair(child1, target_sum)
            child2 = feasibility_repair(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        population = new_population
        
        # Decrease noise level over time
        noise_level *= 0.99
        
        # Periodic check
        if generation % 10 == 0:
            if time.time() - start_time > budget_s - 1:
                break
    
    # Final evaluation
    if best_individual is not None:
        final_c5 = evaluate_c5(best_individual, dx)
        if final_c5 < best_fitness:
            best_fitness = final_c5
    
    return (best_individual.tolist(), best_fitness, n_points)
```
