import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    return h

def crossover(parent1, parent2, crossover_point):
    """Perform crossover between two parents."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    child1[crossover_point:] = parent2[crossover_point:]
    child2[crossover_point:] = parent1[crossover_point:]
    return child1, child2

def mutate(h, mutation_rate, max_mutation):
    """Mutate h with given rate."""
    h_mut = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            delta = np.random.uniform(-max_mutation, max_mutation)
            h_mut[i] = h_mut[i] + delta
    return h_mut

def genetic_algorithm(seed, budget_s, n_points, n_pop=20, elite_ratio=0.2, 
                      crossover_rate=0.8, mutation_rate=0.05, max_mutation=0.2):
    """Evolutionary algorithm to minimize C5."""
    np.random.seed(seed)
    start_time = time.time()
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (target_sum / np.sum(h0))
        population.append(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        population.append(h0)
    
    # Fill initial population
    for _ in range(n_pop - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
        # Add random variation
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite_size = max(1, int(elite_ratio * n_pop))
        new_population = sorted_population[:elite_size]
        
        # Create offspring
        while len(new_population) < n_pop:
            # Selection (tournament)
            tournament_size = 3
            tournament_indices = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            tournament_indices = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1, child2 = crossover(parent1, parent2, crossover_point)
                child1 = repair_h(child1, target_sum)
                child2 = repair_h(child2, target_sum)
                new_population.extend([child1, child2])
            else:
                new_population.extend([parent1.copy(), parent2.copy()])
        
        # Trim to exact population size
        new_population = new_population[:n_pop]
        
        # Mutation
        for i in range(len(new_population)):
            # Decaying mutation rate
            actual_mutation_rate = mutation_rate * exp(-generation / 100)
            new_population[i] = mutate(new_population[i], actual_mutation_rate, max_mutation)
            # Repair after mutation
            new_population[i] = repair_h(new_population[i], target_sum)
            
        population = new_population
        generation += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    # Local refinement
    refined_h = best_h.copy()
    best_refined_c5 = best_c5
    
    # Do gradient ascent refinement on top 3 solutions, but let's sample just a few different ones
    for i in range(3):
        if time.time() - start_time > budget_s - 2:
            break
        h = refined_h.copy()
        
        # Gradient-based refinement
        for _ in range(50):
            if time.time() - start_time > budget_s - 2:
                break
                
            # For gradient estimation, we can perturb h at random points
            idx = np.random.randint(n_points)
            delta = np.random.normal(0, 0.01)
            h[idx] = np.clip(h[idx] + delta, 0, 1)
            h = h * (target_sum / np.sum(h))
            
            c5_new = compute_c5(h, dx)
            if c5_new < best_refined_c5:
                best_refined_c5 = c5_new
                refined_h = h.copy()
    
    return refined_h, best_refined_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    h, c5 = genetic_algorithm(seed, budget_s, n_points)
    return (h.tolist(), c5, n_points)
