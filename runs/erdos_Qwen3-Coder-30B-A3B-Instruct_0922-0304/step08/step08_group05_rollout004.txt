import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_individual(n_points, target_sum, use_initial=False):
    """Create a valid individual (normalized h vector)."""
    if use_initial and 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Generate random valid individual
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
    return np.clip(h, 0, 1)

def mutate_individual(h, mutation_rate=0.1, sigma=0.1):
    """Apply mutation to an individual."""
    h = h + np.random.normal(0, sigma, len(h)) * (np.random.rand(len(h)) < mutation_rate)
    h = np.clip(h, 0, 1)
    return h

def crossover_individuals(h1, h2):
    """Create offspring from two parents."""
    # Average crossover
    offspring = (h1 + h2) / 2.0
    return offspring

def genetic_algorithm_run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Parameters
    population_size = 50
    generations = 1000
    tournament_size = max(2, population_size // 10)
    mutation_rate = 0.1
    sigma = 0.1
    
    # Initialize population
    population = []
    for i in range(population_size):
        use_initial = (i == 0) and 'initial_h_values' in globals()
        h = create_individual(n_points, target_sum, use_initial)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(population):
        f = compute_c5_exact(h, dx)
        fitness.append(f)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    last_improvement = 0
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Tournament selection
        selected_indices = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create new population
        new_population = [population[best_idx].copy()]
        
        # Elitism: keep best individual
        while len(new_population) < population_size:
            # Select two individuals
            idx1, idx2 = np.random.choice(selected_indices, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            offspring = crossover_individuals(parent1, parent2)
            
            # Mutation
            offspring = mutate_individual(offspring, mutation_rate, sigma)
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            f = compute_c5_exact(h, dx)
            new_fitness.append(f)
            
        # Update best
        best_new_idx = np.argmin(new_fitness)
        if new_fitness[best_new_idx] < best_c5:
            best_c5 = new_fitness[best_new_idx]
            best_h = new_population[best_new_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Replace population
        population = new_population
        fitness = new_fitness
        
        if gen % 100 == 0:
            print(f"Generation {gen}, Best C5: {best_c5:.6f}")
    
    return (best_h.tolist(), best_c5, n_points)
    
def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm_run(seed, budget_s, **kwargs)
