import numpy as np
from scipy.fft import fft, ifft
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain"""
    H_conj = np.conj(H)
    return np.abs(H * H_conj)

def spectral_update(H, learning_rate=0.01):
    """Update H to reduce high-frequency components"""
    # Compute surrogate in frequency domain
    S = compute_surrogate(H)
    
    # Find indices with highest contributions
    # Target middle frequencies to reduce main lobe
    n = len(H)
    middle_idx = n // 2
    window_size = max(1, n // 10)  # Adaptive window size
    target_indices = np.arange(max(0, middle_idx - window_size), 
                              min(n, middle_idx + window_size))
    
    # Make small adjustments to reduce those peaks
    for idx in target_indices:
        if S[idx] > 0:
            # Simple gradient descent in frequency domain
            H[idx] -= learning_rate * H[idx] * (S[idx] / np.max(S))
    
    return H

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.rand(n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Fourier optimization approach
        H = fft(h_current)
        
        # Update in spectral domain
        H_updated = spectral_update(H.copy(), learning_rate)
        
        # Reconstruct
        h_reconstructed = np.real(ifft(H_updated))
        
        # Apply projection to maintain constraints
        h_refined = project_h(h_reconstructed, target_sum)
        
        # Compute C5
        c5_refined = compute_c5(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
        # Update current solution
        h_current = h_refined.copy()
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.5
            
        iteration += 1
        
        # Early termination if we're near target
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    h_final = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp = h_final.copy()
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_final, dx):
            h_final = h_temp.copy()

    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
