import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, max_overlap_idx):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Find the lag that gives max overlap
    k = max_overlap_idx
    
    # For each position i, the gradient is (1 - h[(i-k)%n]) 
    for i in range(n):
        prev_idx = (i - k) % n
        grad[i] = 1 - h[prev_idx]
    
    return grad * dx

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    h = project_to_constraint(h, target_sum)
    
    # Initialize momentum
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Parameters
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    improvement_tol = 1e-6
    restart_patience = 5000
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current maximum overlap and its index
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_val = np.max(overlap)
        max_overlap_idx = np.argmax(overlap)
        
        # Check if we have improved
        if max_overlap_val < best_c5 - improvement_tol:
            best_c5 = max_overlap_val
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h, dx, max_overlap_idx)
        
        # Update velocity with Nesterov momentum
        v = momentum * v - step_size * grad
        h_new = h + v
        
        # Project to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_constraint(h_new, target_sum)
        
        # Update h
        h = h_new
        
        # Adjust step size
        if iteration > 0 and iteration % 100 == 0:
            # Simple adaptive step size
            step_size = max(min_step_size, step_size * 0.95)

        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        grad = compute_gradient(h, dx, max_overlap_idx)
        
        h = h - 0.01 * grad
