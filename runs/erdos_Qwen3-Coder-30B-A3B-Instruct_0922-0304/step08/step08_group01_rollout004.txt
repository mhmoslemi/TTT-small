import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Define bounds (0 to 1) and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    def objective(h):
        return compute_c5(h, dx)
    
    # Set up optimization
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    # Set up optimization
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Need to monitor time
    remaining_time = budget_s - 1
    
    # SLSQP optimization
    try:
        result = minimize(
            objective, 
            h.copy(), 
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options=options,
            callback=lambda x: None
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            # Normalize
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            c5_opt = compute_c5(h_opt, dx)
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                
    except:
        pass
    
    # Continue with a local search if needed
    h = best_h.copy()
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:

        h_new = h.copy()

            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        # Renormalize
        h_new = h_new * (target_sum / np.sum(h_new))
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        iter_count += 1
        
        # Occasionally recompute for precision
        if iter_count % 100 == 0:
            c5_test = compute_c5(h, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
