import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (target_sum / np.sum(h))
    
    # Create CVXPY variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == target_sum)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints for each shift
    for k in range(-n_points+1, n_points):
        # Compute indices for the convolution
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i in range(n_points - k)
            expr = 0
            for i in range(n_points - k):
                expr += h_var[i] * (1 - h_var[i+k])
        else:
            # h[i+k] * (1 - h[i]) for i in range(-k, n_points)
            expr = 0
            for i in range(-k, n_points):
                expr += h_var[i+k] * (1 - h_var[i])
        
        constraints.append(expr <= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with ECOS solver
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_bound = t.value
        else:
            # Fallback to the starting point if solver fails
            h_opt = h
            c5_bound = evaluate_c5(h_opt, dx)
    except:
        # Fallback to the starting point if solver fails
        h_opt = h
        c5_bound = evaluate_c5(h_opt, dx)
    
    # Ensure constraints are met
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (target_sum / np.sum(h_opt))
    c5_bound = evaluate_c5(h_opt, dx)
    
    # Final check if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        final_c5 = evaluate_c5(h_opt, dx)
        if final_c5 < c5_bound:
            c5_bound = final_c5
    
    return (h_opt.tolist(), c5_bound, n_points)
