import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.choice([0.0, 1.0], size=n_points, p=[0.5, 0.5])
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Simulated Annealing parameters
    start_temp = 0.1
    cooling_rate = 0.999
    current_temp = start_temp
    max_iterations = 100000
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    def perturb(h):
        # Randomly select operation
        op = np.random.choice(['flip', 'adjust', 'shuffle'], p=[0.5, 0.3, 0.2])
        
        if op == 'flip':
            i = np.random.randint(len(h))
            h[i] = 1 - h[i]
        elif op == 'adjust':
            i = np.random.randint(len(h))
            delta = np.random.uniform(-0.1, 0.1)
            h[i] += delta
        elif op == 'shuffle':
            length = np.random.randint(1, 5)
            i = np.random.randint(len(h) - length + 1)
            segment = h[i:i+length]
            np.random.shuffle(segment)
            h[i:i+length] = segment

    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        # Make a perturbation
        h_new = h.copy()
        perturb(h_new)
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Acceptance criteria
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            delta_e = new_c5 - best_c5
            if np.exp(-delta_e / current_temp) > np.random.rand():
                h = h_new
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Adaptive cooling
        if no_improvement_count > max_no_improvement:
            current_temp = max(0.01, current_temp * 1.5)  # Reheat
            no_improvement_count = 0
            
        current_temp *= cooling_rate
        
        iter_count += 1
        
        # Occasionally recompute overlap for precision
        if iter_count % 1000 == 0:
            current_c5 = compute_c5(h, dx)
            best_c5 = min(best_c5, current_c5)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
