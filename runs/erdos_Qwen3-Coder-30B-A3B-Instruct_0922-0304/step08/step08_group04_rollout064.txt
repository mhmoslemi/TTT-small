import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # CVXPY convex relaxation
    h_vars = Variable(n_points)
    t = Variable()
    
    # Constraints for each shift k
    constraints = []
    for k in range(2 * n_points - 1):
        # Compute the overlap constraint for shift k
        if k < n_points:
            # Forward overlap
            overlap_term = sum(h_vars[i] * (1 - h_vars[i + k]) for i in range(n_points - k))
        else:
            # Backward overlap
            overlap_term = sum(h_vars[i] * (1 - h_vars[i + k - n_points]) for i in range(n_points - (k - n_points + 1)))
        
        constraints.append(overlap_term <= t)
        
    # Sum constraint
    constraints.append(sum(h_vars) == n_points / 2)
    
    # Bounds
    for i in range(n_points):
        constraints.append(h_vars[i] >= 0)
        constraints.append(h_vars[i] <= 1)
    
    # Objective
    objective = Minimize(t)
    
    # Problem
    prob = Problem(objective, constraints)
    
    try:
        # Solve with ECOS
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            h = np.array(h_vars.value).flatten()
    except:
        pass
    
    # Gradient descent refinement
    max_iter = 10000
    lr = 0.01
    decay = 0.999
    tolerance = 1e-8
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute loss and gradient
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Check termination conditions
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 1000:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient (approximated by finite differences)
        grad = np.zeros_like(h)
        eps = 1e-6
        for j in range(n_points):
            h_plus_eps = h.copy()
            h_plus_eps[j] += eps
            h_plus_eps = np.clip(h_plus_eps, 0, 1)
            h_plus_eps = h_plus_eps * (n_points / 2) / np.sum(h_plus_eps)
            
            grad[j] = (evaluate_c5(h_plus_eps, dx) - evaluate_c5(h, dx)) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
