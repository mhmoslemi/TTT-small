import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to ensure it satisfies the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def gaussian_mutation(h, mutation_rate, target_sum):
    """Apply Gaussian mutation to h."""
    mutated = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return repair_h(mutated, target_sum)

def crossover(h1, h2, target_sum):
    """Perform uniform crossover between two individuals."""
    n_points = len(h1)
    mask = np.random.random(n_points) > 0.5
    child = np.where(mask, h1, h2)
    return repair_h(child, target_sum)

def gradient_descent_step(h, dx, step_size=0.01):
    """Apply one step of gradient descent to reduce C5."""
    # Compute current overlap
    overlap = correlate(h, 1 - h, mode="full") * dx
    max_position = np.argmax(overlap)
    # Simple GD step with fixed step size
    gradient = 1 - 2 * h  # Approximation of gradient
    h_new = h - step_size * gradient
    # Ensure feasibility
    h_new = np.clip(h_new, 0, 1)
    current_sum = np.sum(h_new)
    if current_sum > 0:
        h_new = h_new * (len(h_new) / 2) / current_sum
    h_new = np.clip(h_new, 0, 1)
    return h_new
