import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def convex_relaxation(n_points, dx):
    """Solve convex relaxation to get initial guess."""
    # Variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n_points / 2]
    constraints += [h >= 0, h <= 1]
    
    # Add overlap constraints for each shift k
    for k in range(2 * n_points - 1):
        # Compute correlation for shift k
        expr = 0
        for x in range(n_points):
            if x + k - (n_points - 1) < n_points and x + k - (n_points - 1) >= 0:
                expr += h[x] * (1 - h[x + k - (n_points - 1)])
            elif x + k - (n_points - 1) >= n_points:
                expr += h[x] * 0  # boundary condition
            else:
                expr += h[x] * 1  # boundary condition
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            return np.array(h.value).flatten()
    except:
        pass
    return None

def compute_gradient(h, dx, n_points):
    """Compute approximate gradient of the C5 objective."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient based on the maximum overlap
    for i in range(n_points):
        total_grad = 0.0
        for k in range(2 * n_points - 1):
            # Calculate actual indices considering circular boundary
            pos1 = i
            pos2 = i + k - (n_points - 1)  # This corresponds to k in full correlation
            
            # Only consider valid indices
            if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                # Based on derivative of integration
                total_grad += 2 * (1 - 2 * h[pos1])
        grad[i] = total_grad
    
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.001, decay=0.999, tolerance=1e-8):
    """Optimize using gradient descent with adaptive learning rate."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Compute loss and gradient
        if i % 10 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Check termination conditions
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 100:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Attempt convex relaxation to get initial guess
    relaxed_h = convex_relaxation(n_points, dx)
    if relaxed_h is not None:
        h = relaxed_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimize the function using gradient descent
    best_h, best_c5 = optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.001, decay=0.999)
    
    # Make sure we have the exact best solution within the recorded time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        # Do one final check with exact computation
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
