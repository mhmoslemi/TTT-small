import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum_entries, norm, Minimize, sum_squares
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * 0.5
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe algorithm
    max_iter = 10000
    tolerance = 1e-8
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Approximate gradient - extract the effect of limiting each point
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Compute contribution of j'th point to maximum overlap
            contrib = 0.0
            for k in range(2 * n_points - 1):
                # Find the corresponding opposite point, i.e., the one we're varying
                idx2 = (j + k - (n_points - 1)) % n_points
                if 0 <= idx2 < n_points:
                    contrib += (1 - 2 * h[j]) * (1 - h[idx2])
            grad[j] = contrib
        
        # Solve linear subproblem to get the descent direction
        # min_v grad^T v s.t. sum(v) = n_points/2, 0 <= v <= 1
        # This is a standard LP, so we'll solve it via CP
        v = Variable(n_points)
        obj = Minimize(grad.T @ v)
        constraints = [
            sum_entries(v) == n_points / 2,
            v >= 0,
            v <= 1
        ]
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver='ECOS', verbose=False)
            v_vals = np.array(v.value).flatten()
        except:
            # Fallback: use simple method on all values
            v_vals = np.zeros(n_points)
            # Minimize grad.T * v subject to sum(v) = n/2, 0 <= v[i] <= 1:
            # This is solved by making v[i] equal to 0 if grad[i] > 0 and 1 otherwise,
            # then scale to satisfy the sum.
            # But let's just create a simple version for fallback
            weighted = grad.copy() * (-1)
            idx_sorted = np.argsort(weighted)
            total = 0.0
            for k in idx_sorted:
                if total < n_points / 2 and v_vals[k] < 1:
                    cap = min(1.0, n_points / 2 - total)
                    v_vals[k] = cap
                    total += cap
                if total >= n_points / 2:
                    break
            if total < n_points / 2:
                # fill last values
                for k in idx_sorted:
                    if total >= n_points / 2:
                        break
                    free = min(1.0 - v_vals[k], n_points / 2 - total)
                    if free > 0:
                        v_vals[k] += free
                        total += free
            if total < n_points / 2:
                v_vals += (n_points / 2 - total) / n_points
        
        # Compute step size with line search (Armijo rule)
        # try several: 0.1, 0.2, ..., 0.9
        step_sizes = np.linspace(0.1, 0.9, 9)
        best_step = 0.1
        best_val = float('inf')
        for eta in step_sizes:
            h_new = (1 - eta) * h + eta * v_vals
            h_new = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_val:
                best_val = new_c5
                best_step = eta
                
        # update
        h = (1 - best_step) * h + best_step * v_vals
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        if best_val < best_c5:
            best_c5 = best_val
            best_h = h.copy()
            
        # early stopping
        if best_val < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
