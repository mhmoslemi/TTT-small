import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, target_sum)
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Perturb using frequency-domain updates
        h_temp = h_current.copy()
        
        # Use a more sophisticated perturbation
        if iteration % 10 == 0:
            # Small random changes to different regions
            for _ in range(5):
                i = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.03, 0.03)
                h_temp[i] = np.clip(h_temp[i] + delta, 0, 1)
        else:
            # Local gradient-like step
            perturb_mag = 0.02 * np.exp(-iteration / 50)
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-perturb_mag, perturb_mag)
            h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            
        h_temp = project_h(h_temp, target_sum)
        
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < best_c5:
            best_h = h_temp.copy()
            best_c5 = c5_temp
            learning_rate = max(0.001, learning_rate * 0.995)
        else:
            learning_rate = max(0.001, learning_rate * 0.999)
            
        h_current = h_temp.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with gradient-like search
    h_refined = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Try small local search
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
