import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute approximate gradient of the C5 objective."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient based on the maximum overlap
    # For each position, accumulate contribution to gradient
    # Delay computation to avoid redundant work
    for i in range(n_points):
        # Calculate the contribution from the maximum overlap term
        # Gradient w.r.t. h[i] is 1 - 2*h[i] weighted by the number of times it appears
        # in a term of maximum overlap
        grad[i] = 0.0
        for k in range(2 * n_points - 1):
            pos2 = i + k - (n_points - 1)
            if 0 <= pos2 < n_points:
                # x_k is contribution from correlation
                grad[i] += overlap[k] * (1 - 2 * h[pos2])
    
    # Normalize gradient (optional)
    return grad

def solve_linear_subproblem(grad, n_points):
    """Solve the linear subproblem: min_v <grad, v> s.t. sum(v) = n/2, 0 <= v <= 1."""
    # This is a linear program where we minimize <grad, v>
    # subject to sum(v) = n/2 and 0 <= v <= 1
    # This can be solved easily since the objective is linear
    # Simple approach: sort gradient and assign values accordingly
    # to satisfy sum(v) = n/2 and keep v in [0,1]
    
    # Create the input vector for optimization
    v = np.zeros(n_points)
    # Sort the gradient in increasing order to make greedy assignment easier
    sorted_indices = np.argsort(grad)
    
    # Start filling with minimum elements first to minimize the objective
    # We want to assign as many elements as possible at their "left" values
    remaining_sum = n_points / 2.0
    for i in range(n_points):
        idx = sorted_indices[i]
        # Assign to v[idx] the value that minimizes the total sum
        # If we need more sum, just take the minimum of 1 or remaining_sum
        v[idx] = min(1.0, remaining_sum)
        remaining_sum -= v[idx]
        if remaining_sum <= 0:
            break
    
    # Normalize if we have remaining sum (should mostly not happen)
    if remaining_sum > 0:
        # This case should almost never happen but just in case
        # we only replace elements away if they are not already 1
        v = np.clip(v, 0, 1)
        total = np.sum(v)
        if total > 0:
            v = v * (n_points / 2) / total
    
    return v

def frank_wolfe_step(h, dx, n_points, max_iter=1000, tolerance=1e-8):
    """Frank-Wolfe optimization for the minimal overlap problem."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Decreasing step size schedule
    η_schedule = [0.1, 0.05, 0.01, 0.005, 0.001]
    
    for iter in range(max_iter):
        # Early exit if too close to target
        if best_c5 <= 0.3807:
            break
            
        # Check time budget
        if iter % 100 == 0:

                pass  # In a more sophisticated version we could abort early
                
        # Compute the gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Solve the linear sub problem
        v = solve_linear_subproblem(grad, n_points)
        
        # Instead of a fixed η, let's try a coordinate search
        # Start with a reasonable learning rate and adapt
        η = 0.1
        if iter < len(η_schedule):
            η = η_schedule[iter]
            
        # Update h
        h_new = (1 - η) * h + η * v
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2.0)
        
        # Check if we improved
        new_c5 = evaluate_c5(h_new, dx)

                pass
            
        # Track the best solution within the constraints
        if new_c5 < best_c5:
