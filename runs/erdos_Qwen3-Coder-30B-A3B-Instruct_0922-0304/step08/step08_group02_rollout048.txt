```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure it satisfies constraints"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Renormalize
    target_sum = n_points / 2
    h = h * (target_sum / np.sum(h))
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of possible shifts"""
    # Sample shifts from -n+1 to n-1, excluding 0
    shifts = np.arange(-n+1, n)
    # Remove 0 to avoid trivial correlation
    shifts = shifts[shifts != 0]
    # Sample without replacement
    selected = np.random.choice(shifts, size=min(sample_size, len(shifts)), replace=False)
    return selected

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Start with a simple uniform sample
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Adaptive parameters
    max_shifts = 50
    target_overlap = 0.38085
    patience = 0
    max_patience = 200
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        # Sample some shifts
        shifts = sample_shifts(n_points, max_shifts)
        
        # Use convex optimization to reduce maximum overlap
        h = Variable(n_points)
        # Objective is a relaxation, we minimize a proxy for max overlap
        # We'll create constraints for selected shifts
        constraints = [h >= 0, h <= 1]
        # Sum constraint
        constraints.append(sum(h) == target_sum)
        
        # Add constraints for selected shifts
        for k in shifts:
            # Compute the correlation coefficient and add constraint
            # This is a proxy for the true constraint in a convex form
            if k > 0:
                # overlap with right shift
                corr = sum(h[i] * (1 - h[i+k]) for i in range(n_points-k) if i+k < n_points)
            else:
                # overlap with left shift
                corr = sum(h[i] * (1 - h[i+k]) for i in range(n_points+k) if i+k >= 0)
            # add constraint to bound this overlap
            constraints.append(corr <= target_overlap)
            
        # Minimize a proxy for the maximum overlap in those sampled shifts
        # This is actually an approximation - in convex form we can't directly access max
        # For now we'll just try to reduce control points through constrained QP
        # A simpler approach: minimize L2 norm or sum of squares
        objective = Minimize(sum(h**2))
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False)  # Use CPLEX if available
        except:
            try:
                prob.solve(solver=OSQP, verbose=False)  # Fallback to OSQP
            except:
                prob.solve(solver=ECOS, verbose=False)  # Final fallback
        
        if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
            h_new = np.array(h.value).flatten()
            h_new = project_h(h_new, n_points)
            
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
                # Update dynamic target
                if best_c5 < 0.3807:
                    target_overlap = best_c5 * 0.99
                else:
                    target_overlap = best_c5 * 0.995
            else:
                patience += 1
        else:
            # If optimization fails, do local perturbation
            h_new = h_current.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
            h_new = project_h(h_new, n_points)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
                target_overlap = max(0.3805, best_c5 * 0.99)
            else:
                patience += 1
                
        h_current = h_new.copy()
        iteration += 1
        
        # Check for early stopping if we've hit a target
        if best_c5 <= 0.3807:
            break
        
    # Final refinement
    for i in range(30):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = best_h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(best_h, dx):
            best_h = h_temp.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
