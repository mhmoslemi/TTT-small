import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Solve as QP problem
    h_var = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints for each shift k
    # For k from 0 to n_points-1, compute correlation
    # We compute h_i * (1 - h_{i+k}) for valid indices
    for k in range(n_points):
        # Compute the overlap at shift k
        expr = 0
        for i in range(n_points):
            # Equivalent to h[i] * (1 - h[(i+k) % n_points])
            expr += h_var[i] * (1 - h_var[(i + k) % n_points])
        constraints.append(expr <= t)
    
    # Form and solve problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            t_value = t.value
        else:
            # Fallback to the parent solution if optimization fails
            h_opt = h.copy()
            t_value = np.max(correlate(h_opt, 1 - h_opt, mode="full")) * dx
    except:
        # Fallback to the parent solution if optimization fails
        h_opt = h.copy()
        t_value = np.max(correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    # Verify that it satisfies all constraints
    h_final = np.clip(h_opt, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final C5 computation
    c5_bound = np.max(correlate(h_final, 1 - h_final, mode="full")) * dx
    
    # Make sure we return within the budget with the best result found
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        # If timeout is close, return what we have
        pass
    
    return (h_final.tolist(), float(c5_bound), n_points)
