import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_constraints(h, target_sum):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def generate_neighborhood(h, target_sum, n_candidates=10):
    """Generate neighborhood of solutions."""
    candidates = []
    n = len(h)
    
    for _ in range(n_candidates):
        # Randomly choose between flip and swap
        if np.random.rand() < 0.3:
            # Flip
            new_h = h.copy()
            i, j = np.random.randint(0, n, 2)
            new_h[i] = max(0, min(1, new_h[i] * 0.97 + np.random.normal(0, 0.01)))
            new_h[j] = max(0, min(1, new_h[j] * 0.97 + np.random.normal(0, 0.01)))
        else:
            # Swap
            new_h = h.copy()
            i, j = np.random.randint(0, n, 2)
            new_h[i], new_h[j] = new_h[j], new_h[i]
        
        new_h = repair_constraints(new_h, target_sum)
        candidates.append(new_h)
    
    return candidates

def tabu_search(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    temperature = 1.0
    cooling_rate = 0.999
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Generate neighborhood
        candidates = generate_neighborhood(h, target_sum, 10)
        
        # Evaluate candidates
        candidate_values = []
        for candidate in candidates:
            c5 = compute_c5_exact(candidate, dx)
            candidate_values.append((candidate, c5))
        
        # Find best non-tabu candidate
        best_candidate = None
        best_candidate_c5 = float('inf')
        
        for candidate, c5 in candidate_values:
            # Check if in tabu list
            is_tabu = False
            for tabu_state in tabu_list:
                if np.allclose(candidate, tabu_state, atol=1e-6):
                    is_tabu = True
                    break
            
            # Apply aspiration criteria 
            if not is_tabu or c5 < best_c5:
                if c5 < best_candidate_c5:
                    best_candidate = candidate
                    best_candidate_c5 = c5
        
        if best_candidate is not None:
            # Update tabu list
            tabu_list.append(h.copy())
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            h = best_candidate
            if best_candidate_c5 < best_c5:
                best_c5 = best_candidate_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Cooling
            temperature *= cooling_rate
            
            # Early termination
            if best_c5 <= 0.3807:
                break
        else:
            no_improvement_count += 1
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return tabu_search(seed, budget_s)
