import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Optimization parameters
    step_size = 0.01
    max_iter = 10000
    epsilon = 1e-6
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 5000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Find the worst overlap lag
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_lag = max_idx - (n_points - 1)
        
        # Adjust h to reduce the worst overlap
        h_new = h.copy()
        adjusted = False
        
        # Try to decrease overlap by adjusting neighbors of max_lag
        # Check top 5 worst lags
        top_lags = np.argsort(overlap)[-5:][::-1]
        
        for lag_idx in top_lags:
            lag = lag_idx - (n_points - 1)
            # Samples that contribute to this lag
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    # Try to decrease h[i] and increase h[j] 
                    # to reduce the term h[i] * (1 - h[j])
                    if h[i] > epsilon and h[j] < 1 - epsilon:
                        delta = min(step_size, h[i], 1 - h[j])
                        h_new[i] -= delta
                        h_new[j] += delta
                        adjusted = True
        
        if adjusted:
            # Normalize
            h_new = h_new * (target_sum / np.sum(h_new))
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            h = h_new
            
        iteration += 1
        
        # Adaptive step size
        if iteration % 100 == 0 and step_size > 1e-6:
            step_size *= 0.999
            
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
