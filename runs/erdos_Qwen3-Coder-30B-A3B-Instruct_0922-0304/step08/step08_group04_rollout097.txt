```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 0.5)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe algorithm
    max_iter = 10000
    tolerance = 1e-8
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the gradient (approximate)
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Create gradient vector
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx = i + max_idx - (n_points - 1)  # index in 1-h
            if 0 <= idx < n_points:
                grad[i] = (1 - 2 * h[i]) * (1 - h[idx])
        
        # Solve linear subproblem: min v^T * grad s.t. sum(v) = n_points/2, 0 <= v <= 1
        # Since sum(v) = n_points/2 and we want to minimize grad^T * v,
        # we should set v = max(0, 1/n_points * (n_points/2 - grad)) where grad is adjusted
        # Actually, we can solve this in closed form: 
        # Select the smallest component of grad and set v as unit vector with same index
        # This is a canonical Frank-Wolfe step for the simplex constraint.
        v = np.zeros(n_points)
        idx = np.argmin(grad)
        v[idx] = 1.0  # set one element to 1
        
        # Normalize v to have sum = n_points / 2
        v *= (n_points / 2) / np.sum(v)
        
        # Check if v violates constraints
        v = np.clip(v, 0, 1)
        v_sum = np.sum(v)
        if v_sum > 0:
            v = v * (n_points / 2) / v_sum
        
        # Line search parameter
        eta = 0.1
        h_new = (1 - eta) * h + eta * v
        
        # Project h_new back to feasible region (sum = n_points/2, bounds [0,1])
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new.copy()
        
        if no_improvement_count > 1000:
            break
            
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
