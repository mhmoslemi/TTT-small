import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n_points):
    """Create a triangular ramp function for initialization."""
    x = np.linspace(0, 2, n_points)
    h = np.minimum(x, 2 - x)
    return h / np.sum(h) * (n_points / 2)

def compute_c5(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region with sum = target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradients(h, dx):
    """Compute gradients of the overlap with respect to h."""
    # This is an approximation of gradient of maximum overlap
    # We compute gradient using finite differences
    n_points = len(h)
    gradient = np.zeros(n_points)
    eps = 1e-4
    
    # Compute gradient by finite differences along each coordinate
    base_overlap = correlate(h, 1 - h, mode="full") * dx
    base_max = np.max(base_overlap)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + eps)
        h_plus = project_h(h_plus, len(h) / 2)
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
        max_plus = np.max(overlap_plus)
        
        gradient[i] = (max_plus - base_max) / eps
    
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_h(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Set up nonlinear optimization
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    learning_rate = 0.001
    momentum = 0.9
    
    velocity = np.zeros(n_points)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current c5 and gradient
        current_c5 = compute_c5(h, dx)
        grad = compute_gradients(h, dx)
        
        # Update velocity with momentum
        velocity = momentum * velocity - learning_rate * grad
        
        # Update h
        h_new = h + velocity
        h_new = project_h(h_new, target_sum)
        
        # Compute new c5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept improvement or small improvement
        if new_c5 <= current_c5:
            h = h_new.copy()
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        iter_count += 1
        
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
