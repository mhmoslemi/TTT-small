```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Start with a simple strategy: local search with adaptive grid
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    improvement_count = 0
    no_improvement_iterations = 0
    
    # Try different configurations: limited number of segments
    segment_configs = [3, 4, 5, 6, 7, 8, 9, 10, 12]  # fewer segments to keep it fast
    for i in range(2000):  # iterations limit
        if time.time() - start_time > budget_s - 1:
            break
        
        # Pick a random segment count
        n_segments = np.random.choice(segment_configs)
        
        # Create even segmented intervals
        segments = np.linspace(0, 2, n_segments + 1)
        
        # Initialize segment values
        h_try = np.zeros(n_points)
        segment_h = np.random.rand(n_segments).astype(np.float64)
        
        # Map segments to points
        for j in range(n_segments):
            start_idx = int(segments[j] / dx)
            end_idx = int(segments[j+1] / dx)
            if end_idx >= n_points: end_idx = n_points - 1
            if start_idx < 0: start_idx = 0
            h_try[start_idx:end_idx] = segment_h[j]
        
        # Clip to [0, 1]
        h_try = np.clip(h_try, 0, 1)
        
        # Normalize
        h_try = normalize_h(h_try, n_points / 2)
        
        # Evaluate
        c5_try = evaluate_c5(h_try, dx)
        
        # Update if better
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            improvement_count += 1
            no_improvement_iterations = 0
        else:
            no_improvement_iterations += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        if no_improvement_iterations > 200:
            # Try intensifying near current best
            h_try = best_h + np.random.normal(0, 0.02, n_points)
            h_try = np.clip(h_try, 0, 1)
            h_try = normalize_h(h_try, n_points / 2)
            c5_try = evaluate_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        
        # Occasionally, do more random searches
        if i % 50 == 0:
            for _ in range(10):
                h_trial = np.random.rand(n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = normalize_h(h_trial, n_points / 2)
                c5_trial = evaluate_c5(h_trial, dx)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
    
    # Final re-evaluation to make sure
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
