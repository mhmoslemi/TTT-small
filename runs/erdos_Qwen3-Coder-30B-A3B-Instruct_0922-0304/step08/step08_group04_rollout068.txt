import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Partitioning + local search strategy
    n_segments = 4
    max_iter = 1000
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adapt number of segments
        if iteration % 50 == 0 and n_segments < 10:
            n_segments += 1
            
        # Create breakpoints
        breakpoints = np.linspace(0, 2, n_segments + 1)
        
        # Initialize segment values
        segment_values = np.random.rand(n_segments)
        segment_values = normalize_h(segment_values, 1.0)
        
        # Local grid search
        best_segment_values = segment_values.copy()
        best_segment_c5 = float('inf')
        
        # Grid search for segment values
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Create new candidate
            new_segment_values = best_segment_values + np.random.normal(0, 0.05, n_segments)
            new_segment_values = np.clip(new_segment_values, 0, 1)
            new_segment_values = normalize_h(new_segment_values, 1.0)
            
            # Convert segment values to full h
            h_candidate = np.zeros(n_points)
            for i in range(n_segments):
                start_idx = int(breakpoints[i] / dx)
                end_idx = int(breakpoints[i+1] / dx)
                if end_idx > start_idx:
                    h_candidate[start_idx:end_idx] = new_segment_values[i]
            
            # Ensure the last segment covers remaining points
            if end_idx < n_points:
                h_candidate[end_idx:] = new_segment_values[-1]
                
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, n_points / 2)
            
            # Evaluate
            c5_candidate = evaluate_c5(h_candidate, dx)
            
            if c5_candidate < best_segment_c5:
                best_segment_c5 = c5_candidate
                best_segment_values = new_segment_values.copy()
        
        # Convert best segment values to h
        h_result = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(breakpoints[i] / dx)
            end_idx = int(breakpoints[i+1] / dx)
            if end_idx > start_idx:
                h_result[start_idx:end_idx] = best_segment_values[i]
                
        # Ensure the last segment covers remaining points
        if end_idx < n_points:
            h_result[end_idx:] = best_segment_values[-1]
            
        h_result = np.clip(h_result, 0, 1)
        h_result = normalize_h(h_result, n_points / 2)
        
        # Check improvement
        c5_result = evaluate_c5(h_result, dx)
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if c5_result <= 0.3807:
            break
            
        if no_improvement_count > 100:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
