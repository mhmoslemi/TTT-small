import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def feasibility_repair(h, target_sum):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.3, noise_magnitude=0.05):
    """Apply mutation to an individual."""
    if random.random() > mutation_rate:
        return individual.copy()
    
    n = len(individual)
    mutated = individual.copy()
    
    # Add noise to some positions
    num_mutations = random.randint(1, n // 4)
    positions = random.sample(range(n), num_mutations)
    
    for pos in positions:
        mutated[pos] += random.gauss(0, noise_magnitude)
    
    return mutated

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    selected_indices = random.sample(range(len(population)), tournament_size)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitnesses)]
    return population[winner_idx].copy()

def evaluate_population(population, dx, n_points):
    """Evaluate all individuals in the population."""
    fitnesses = Parallel(n_jobs=-1)(
        delayed(evaluate_c5)(ind, dx) for ind in population
    )
    return fitnesses

def genetic_algorithm(seed=42, budget_s=600, n_points=70):
    """Run genetic algorithm to optimize h."""
    np.random.seed(seed)
    start_time = time.time()
    
    target_sum = n_points / 2
    pop_size = 50
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.3
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, target_sum)
        population = [parent.copy()]
        # Add diverse variants
        for _ in range(pop_size - 1):
            variant = parent + np.random.normal(0, 0.02, n_points)
            variant = feasibility_repair(variant, target_sum)
            population.append(variant)
    else:
        population = []
        for _ in range(pop_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, target_sum)
            population.append(individual)
            
    dx = 2.0 / n_points
    
    best_individual = None
    best_fitness = float('inf')
    
    for generation in range(max_generations):
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = evaluate_population(population, dx, n_points)
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Early termination
        if best_fitness <= 0.3807:
            break
            
        # Select parents and generate new population
        new_population = []
        
        # Elitism: keep the best
        elite_idx = np.argmin(fitnesses)
        new_population.append(population[elite_idx].copy())
        
        # Generate rest of population
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate, 0.05 * (0.99 ** (generation // 10)))
            child2 = mutate(child2, mutation_rate, 0.05 * (0.99 ** (generation // 10)))
            
            child1 = feasibility_repair(child1, target_sum)
            child2 = feasibility_repair(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final evaluation of best individual
    if best_individual is not None:
        final_fitness = evaluate_c5(best_individual, dx)
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = genetic_algorithm(seed, budget_s)
    
    # Ensure feasibility
    n_points = len(best_h)
    dx = 2.0 / n_points
    best_h = feasibility_repair(best_h, n_points / 2)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
