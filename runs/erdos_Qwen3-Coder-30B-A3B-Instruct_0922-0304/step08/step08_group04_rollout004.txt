import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # QP formulation
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        # Compute overlap for shift k
        rhs = 0.0
        lhs = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            if 0 <= j < n_points:
                lhs += h_var[i] * (1 - h_var[j])
        constraints.append(lhs <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            solution_h = h_var.value
            c5_bound = t.value
        else:
            # Fallback to previous method if QP fails
            solution_h = h.copy()
            c5_bound = evaluate_c5(solution_h, dx)
    except:
        # If QP fails, fall back to baseline
        solution_h = h.copy()
        c5_bound = evaluate_c5(solution_h, dx)

    # Ensure final constraints
    solution_h = np.clip(solution_h, 0, 1)
    solution_h_sum = np.sum(solution_h)
    if solution_h_sum > 0:
        solution_h = solution_h * (n_points / 2) / solution_h_sum

    # Re-evaluate final C5 using exact evaluator
    final_c5 = evaluate_c5(solution_h, dx)
    
    # Make sure we return the best solution within time
    return (solution_h.tolist(), final_c5, n_points)
