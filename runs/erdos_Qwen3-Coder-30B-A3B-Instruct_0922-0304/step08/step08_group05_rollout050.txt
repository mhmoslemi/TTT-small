import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def init_population(pop_size, n_points, initial_h_values=None):
    """Initialize population with diverse individuals."""
    population = []
    
    # If initial solution available, use it
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = normalize_h(h, n_points / 2)
        population.append(h.copy())
    
    # Generate random individuals
    for i in range(pop_size - len(population)):
        # Start with a random distribution
        h = np.random.rand(n_points)
        # Add some Gaussian noise to make it more diverse
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        population.append(h.copy())
    
    return population

def selection(population, fitnesses, tournament_size=5):
    """Select individuals using tournament selection."""
    selected = []
    n = len(population)
    for _ in range(n):
        # Tournament
        tournament_indices = random.sample(range(n), min(tournament_size, n))
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Perform average-based crossover."""
    child = (parent1 + parent2) / 2.0
    return child

def mutate(individual, mutation_strength=0.1, n_points=70):
    """Apply Gaussian mutation."""
    mutated = individual + np.random.normal(0, mutation_strength, n_points)
    mutated = np.clip(mutated, 0, 1)
    return normalize_h(mutated, n_points / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.1
    
    # Initialize population
    population = init_population(pop_size, n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5_exact(h, dx)
            fitnesses.append(c5)
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = [population[current_best_idx].copy()]  # Keep best
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected, 2)
            child = crossover(parent1, parent2)
            if random.random() < mutation_rate:
                child = mutate(child, mutation_rate, n_points)
            new_population.append(child)
        
        population = new_population[:pop_size]
        generation += 1
        
    # Final computation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
