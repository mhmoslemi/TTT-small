import numpy as np
from scipy.signal import correlate
import time
from math import floor

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2):
    """Blend two parents to create offspring."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate(h, mutation_rate=0.05, max_mutation=0.1):
    """Apply mutation to h."""
    h_new = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            delta = np.random.uniform(-max_mutation, max_mutation)
            h_new[i] = h_new[i] + delta
    return h_new

def optimize_local(h, dx, steps=50):
    """Apply local gradient descent to refine solution."""
    h_best = h.copy()
    best_c5 = compute_c5(h_best, dx)
    
    for _ in range(steps):
        if np.random.random() < 0.5:
            # Try small perturbations
            h_temp = h_best.copy()
            idx = np.random.randint(len(h_temp))
            delta = np.random.normal(0, 0.01)
            h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = repair_h(h_temp, len(h_temp) / 2)
            c5_temp = compute_c5(h_temp, dx)
            if c5_temp < best_c5:
                h_best = h_temp.copy()
                best_c5 = c5_temp
        else:
            # Gradient-like update
            h_temp = h_best.copy()
            idx = np.random.randint(len(h_temp))
            # Simple gradient direction towards reducing max overlap
            # Use finite difference estimate of gradient
            eps = 0.01
            h_plus = h_temp.copy()
            h_plus[idx] = np.clip(h_plus[idx] + eps, 0, 1)
            h_plus = repair_h(h_plus, len(h_plus) / 2)
            h_minus = h_temp.copy()
            h_minus[idx] = np.clip(h_minus[idx] - eps, 0, 1)
            h_minus = repair_h(h_minus, len(h_minus) / 2)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            # Approximate gradient: (f(x+eps) - f(x-eps)) / (2*eps)
            grad = (c5_plus - c5_minus) / (2 * eps)
            
            # Move in direction of negative gradient
            if grad != 0:
                step_size = 0.05
                new_val = h_temp[idx] - step_size * grad
                h_temp[idx] = np.clip(new_val, 0, 1)
                h_temp = repair_h(h_temp, len(h_temp) / 2)
                c5_temp = compute_c5(h_temp, dx)
                if c5_temp < best_c5:
                    h_best = h_temp.copy()
                    best_c5 = c5_temp
    return h_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = []
    
    # Include the initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    else:
        # Create random valid solution
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    # Fill population with diverse random solutions
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            
        # Update best solution
        for i, c5 in enumerate(fitnesses):
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
                no_improvement_count = 0
                
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        
        # Elitism: keep top 20%
        n_elite = max(2, n_pop // 5)
        elites = population[:n_elite]
        
        # Generate new population
        new_population = elites.copy()
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            # Select parents
            parent1 = elites[np.random.randint(0, len(elites))]
            parent2 = elites[np.random.randint(0, len(elites))]
            
            # Crossover
            if np.random.random() < 0.8:
                child = crossover(parent1, parent2)
            else:
                # Just copy one parent
                child = parent1.copy()
                
            # Mutation
            child = mutate(child, mutation_rate=0.05, max_mutation=0.1)
            
            # Repair to constraint
            child = repair_h(child, target_sum)
            
            new_population.append(child)
            
        population = new_population
        
        # Local refinement of top solutions
        top_solutions = population[:3]
        for h in top_solutions:
            h_refined = optimize_local(h, dx, steps=50)
            c5_refined = compute_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
                
        generation += 1
        no_improvement_count += 1
        
    if best_h is None:
        # Fall back to last population member if no improvement
        best_h = population[-1].copy()
        best_c5 = compute_c5(best_h, dx)
        
    # Final refinement
    best_h = optimize_local(best_h, dx, steps=100)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
