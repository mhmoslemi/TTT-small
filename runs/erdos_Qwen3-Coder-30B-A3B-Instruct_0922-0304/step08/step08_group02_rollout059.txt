import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h onto the feasible region: [0,1] and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, shift, n_points):
    """Compute the gradient w.r.t. the maximum overlap shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Normalize initial
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Compute gradient for max overlap shift
        shift = max_overlap_idx - (n_points - 1)  # shift of the max overlap
        
        grad = compute_gradient(h_current, shift, n_points)
        
        # Update h using negative gradient
        h_new = h_current - step_size * grad
        
        # Project h_new
        h_new = project_h(h_new, n_points)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current h
        h_current = h_new.copy()
        
        # Adaptive step size
        if no_improvement_count % 100 == 0 and no_improvement_count > 0:
            step_size *= 0.5
            
        # Occasionally perturb to escape local minima
        if no_improvement_count > 10 and np.random.rand() < 0.05:
            h_current = h_current + np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current, n_points)
            
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_current + np.random.normal(0, 0.005, n_points)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_current, dx):
            h_current = h_temp.copy()
    
    final_c5 = compute_c5(h_current, dx)
    return (h_current.tolist(), final_c5, n_points)
