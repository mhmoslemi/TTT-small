import numpy as np
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute C5 value using correlation"""
    # Using scipy's correlate for maximum efficiency
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def compute_surrogate(H):
    """Compute surrogate in frequency domain"""
    H_conj = np.conj(H)
    # H * (1 - H_conj) 
    surrogate = np.abs(H * (1 - H_conj))
    return surrogate

def spectral_update(H, target_bins, learning_rate):
    """Update H in frequency domain to reduce surrogate value"""
    H_new = H.copy()
    for bin_idx in target_bins:
        H_new[bin_idx] -= learning_rate * np.real(H_new[bin_idx])
        # Enforce conjugate symmetry for real signal
        if bin_idx != 0 and bin_idx != len(H) - 1:
            H_new[-bin_idx] = np.conj(H_new[bin_idx])
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.uniform(0, 1, n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    decay_rate = 0.99
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= decay_rate
        
        # Fourier transform of h
        H = fft(h_current)
        
        # Compute surrogate values
        surrogate = compute_surrogate(H)
        
        # Find bins with high surrogate values
        target_bins = np.argsort(surrogate)[-10:]  # Top 10 bins
        
        # Update in frequency domain
        H_new = spectral_update(H, target_bins, learning_rate)
        
        # Reconstruct h
        h_new = np.real(ifft(H_new))
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h_current = h_new.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
