import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate the individual with noise."""
    if np.random.rand() > mutation_rate:
        return h.copy()
    
    noise = np.random.normal(0, noise_magnitude, len(h))
    mutated = h + noise
    mutated = np.clip(mutated, 0, 1)
    
    return mutated

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    selected_indices = np.random.choice(len(population), tournament_size)
    best_index = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_index].copy()

def genetic_algorithm(n_points, dx, population_size=50, max_generations=1000, 
                      crossover_rate=0.8, mutation_rate=0.3, noise_magnitude=0.05):
    """Run the genetic algorithm to minimize C5."""
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            individual = parent + np.random.normal(0, 0.01, n_points)
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    else:
        population = []
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    best_individual = None
    best_fitness = float('inf')
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = Parallel(n_jobs=1)(delayed(evaluate_c5)(ind, dx) for ind in population)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Early termination if target reached
        if best_fitness <= 0.3807:
            break
            
        # Selection, crossover, and mutation
        new_population = []
        
        # Keep the best individual
        new_population.append(best_individual)
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation
            child1 = mutate(child1, mutation_rate, noise_magnitude)
            child2 = mutate(child2, mutation_rate, noise_magnitude)
            
            # Repair feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, n_points / 2)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, n_points / 2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Decrease noise magnitude over time
        noise_magnitude *= 0.99
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    best_h, best_c5 = genetic_algorithm(n_points, dx, max_generations=1000)
    
    # Evaluate one final time to ensure accuracy
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
