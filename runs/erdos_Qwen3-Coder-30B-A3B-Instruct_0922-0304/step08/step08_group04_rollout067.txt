import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def optimize_with_partitioning(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Partition and search approach
    max_iter = 10000
    no_improvement_count = 0
    partition_sizes = [3, 4, 5, 6, 7, 8, 9, 10]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive partitioning
        n_segments = partition_sizes[iteration % len(partition_sizes)]
        
        # Initialize segments
        segment_breaks = np.linspace(0, 2, n_segments + 1)
        segment_values = np.random.rand(n_segments).astype(np.float64)
        segment_values = normalize_h(segment_values, target_sum)
        
        # Create piecewise function
        h_candidate = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(segment_breaks[i] / dx)
            end_idx = int(segment_breaks[i+1] / dx)
            if end_idx > start_idx:
                h_candidate[start_idx:end_idx] = segment_values[i]
        
        # Normalize the candidate
        h_candidate = normalize_h(h_candidate, target_sum)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Evaluate
        c5_candidate = evaluate_c5(h_candidate, dx)
        
        # Update best solution if improved
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > 1000:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return optimize_with_partitioning(seed, budget_s)
