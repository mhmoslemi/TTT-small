import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function with sum = n/2"""
    x = np.linspace(0, 2, n)
    h = 1 - np.abs(x - 1)
    h = h * (n/2) / np.sum(h)
    return h

def project(h, target_sum):
    """Project h to feasible region: clip to [0,1] and renormalize"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_c5(h, dx):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Multi-resolution grid search
    resolutions = [20, 40, 70]
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        if res < n_points:
            # Interpolate to lower resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h)
            h_new = project(h_new, target_sum)
        else:
            h_new = h.copy()
            
        # Gradient descent optimization on this resolution
        learning_rate = 0.001
        momentum = 0.9
        velocity = np.zeros_like(h_new)
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Approximate gradients
            grad = np.zeros_like(h_new)
            eps = 1e-5
            for i in range(len(h_new)):
                h_plus = h_new.copy()
                h_plus[i] = min(1.0, h_plus[i] + eps)
                h_plus = project(h_plus, target_sum)
                
                h_minus = h_new.copy()
                h_minus[i] = max(0.0, h_minus[i] - eps)
                h_minus = project(h_minus, target_sum)
                
                c5_plus = compute_c5(h_plus, dx)
                c5_minus = compute_c5(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with momentum
            velocity = momentum * velocity - learning_rate * grad
            h_new = h_new + velocity
            h_new = project(h_new, target_sum)
            
            # Check improvement
            current_c5 = compute_c5(h_new, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
                
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
