import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, shift, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        idx = i + shift
        if 0 <= idx < n_points:
            grad[i] = 1 - h[idx]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    last_improvement_iter = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        if iter_count > 0 and iter_count % 100 == 0:
            step_size *= 0.5
            
        # Compute the cross-correlation to find the maximum overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Determine the shift that causes maximum overlap
        # Since the correlation is full mode, the maximum is at idx = n_points - 1
        # The actual shift is: max_overlap_idx - (n_points - 1) = max_overlap_idx - n_points + 1
        shift = max_overlap_idx - n_points + 1
        
        # Compute gradient
        gradient = compute_gradient(h_current, shift, n_points)
        
        # Update step
        h_new = h_current - step_size * gradient
        
        # Project back to feasible region
        h_new = project_h(h_new, n_points)
        
        # Check improvement
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement_iter = iter_count
            
        # Perturb if no improvement in a while
        if iter_count - last_improvement_iter > 100:
            noise_magnitude = 0.1 * np.mean(h_new)
            perturbation = np.random.normal(0, noise_magnitude, n_points)
            h_new = h_current + perturbation
            h_new = project_h(h_new, n_points)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                last_improvement_iter = iter_count
                
        h_current = h_new.copy()
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
