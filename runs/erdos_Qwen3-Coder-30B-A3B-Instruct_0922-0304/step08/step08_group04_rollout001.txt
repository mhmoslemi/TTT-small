import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # Prepare QP formulation
    h_var = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Constraints
    constraints = []
    
    # Add sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Add overlap constraints for all shifts
    # Precompute all valid index pairs for overlaps
    for k in range(-n_points + 1, n_points):
        # For shift k, compute sum h_i * (1 - h_{i+k})
        # This is equivalent to sum h_i - sum h_i * h_{i+k}
        
        # Create constraint for shift k
        # This is more complex to compute exactly through matrix
        # So we use a different approach: We'll form it properly as a linear constraint
        # Effectively, we want h_i * (1 - h_{i+k}) for valid pairs
        
        # Create constraint manually for efficiency
        # This is sum h_i * (1 - h_{i+k}) <= t
        # Or sum h_i - sum h_i * h_{i+k} <= t
        # Rearranged: sum h_i * h_{i+k} >= sum h_i - t
        
        # Handle the circular buffer correctly
        terms = []
        for i in range(n_points):
            i_k = (i + k) % n_points
            terms.append(h_var[i] * h_var[i_k])
        
        # We can't directly create bilinear terms in CVXPY in this simple way
        # So we approximate using linear over-approximation or simply construct parts
        # Or reformulate properly as bilinear constraints
        
        # Better approach: Write out exact quadratic constraints
        # For each k, sum(h[i] * (1 - h[i+k])) <= t 
        # For (i+k) in [0, n_points-1] and i in [0, n_points-1]
        
        # Let's define constraints individually
        if k <= 0:
            # For k <= 0, we look backwards (so k-th element is at i+k, i >= -k)
            exprs = []
            for i in range(n_points):
                i_k = i + k
                if i_k >= 0:
                    exprs.append(h_var[i] * (1 - h_var[i_k]))
            if exprs:
                constraints.append(cp.sum(exprs) <= t)
        else:
            # For k > 0, i_k = i + k, need i+k < n_points => i < n_points-k
            exprs = []
            for i in range(n_points - k):
                exprs.append(h_var[i] * (1 - h_var[i+k]))
            if exprs:
                constraints.append(cp.sum(exprs) <= t)
    
    # Formulate objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            t_opt = t.value
            c5_bound = t_opt
        else:
            # If solver failed, fallback to simple optimization approach
            h_opt = h.copy()
            c5_bound = evaluate_c5(h_opt, dx)
    except:
        # Last resort: use the current h
        h_opt = h.copy()
        c5_bound = evaluate_c5(h_opt, dx)
    
    # Validate constraints
    h_opt = np.clip(h_opt, 0, 1)
    actual_sum = np.sum(h_opt)
    if abs(actual_sum - n_points / 2) > 1e-5:
        # Renormalize if needed
        h_opt = h_opt * (n_points / 2) / actual_sum
    
    # Verify final C5
    final_c5 = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
