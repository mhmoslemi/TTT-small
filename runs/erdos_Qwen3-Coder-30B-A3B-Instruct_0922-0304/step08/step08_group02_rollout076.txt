import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_point):
    """Perform crossover between two parents."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    child1[crossover_point:] = parent2[crossover_point:]
    child2[crossover_point:] = parent1[crossover_point:]
    return child1, child2

def mutate(h, mutation_rate):
    """Mutate h with given rate."""
    h_mut = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            delta = np.random.normal(0, 0.05)
            h_mut[i] = np.clip(h_mut[i] + delta, 0, 1)
    return h_mut

def genetic_algorithm(n_points, target_sum, dx, budget_s, seed=42):
    """Run genetic algorithm to find better h."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_pop = 20
    mutation_rate = 0.05
    crossover_prob = 0.8
    elitism_rate = 0.2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
        population.append(h_init)
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Fill initial population
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        # Perturb some values to add diversity
        for i in range(int(0.1 * n_points)):
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.1)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_pop = population[:int(elitism_rate * n_pop)]
        
        # Generate offspring
        while len(new_pop) < n_pop:
            # Tournament selection
            parent1_idx = np.random.randint(0, int(0.3 * n_pop))
            parent2_idx = np.random.randint(0, int(0.3 * n_pop))
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.random() < crossover_prob:
                crossover_point = np.random.randint(1, n_points)
                child1, child2 = crossover(parent1, parent2, crossover_point)
                child1 = repair_h(child1, target_sum)
                child2 = repair_h(child2, target_sum)
                if len(new_pop) < n_pop:

                    new_pop.append(child2)

                child2 = parent2.copy()
                if len(new_pop) < n_pop:

                if len(new_pop) < n_pop:
                    new_pop.append(child2)

        generation += 1

def local_refinement(h, dx, target_sum, budget_s):
    """Apply local refinement to improve solution."""
    start_time = time.time()
    h_refined = h.copy()
    best_h = h_refined.copy()
    best_c5 = compute_c5(h_refined, dx)
    
    current_memory = 0
    
    while time.time() - start_time < budget_s - 1 and current_memory < 5000:
        # Simple gradient-like search
        i = np.random.randint(0, len(h_refined))
        previous_c5 = compute_c5(h_refined, dx)
        delta = np.random.normal(0, 0.01)
        h_temp = h_refined.copy()
        h_temp[i] = np.clip(h_temp[i] + delta, 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < best_c5:
            best_h = h_temp.copy()
            best_c5 = c5_temp
        if c5_temp < previous_c5:
            h_refined = h_temp.copy()
        current_memory += 1
    
    return best_h, best_c5

    h_final, c5_final = local_refinement(h_best_ga, dx, target_sum, budget_s)
    
    # Ensure final h satisfies constraints
