import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_coefficients(h, n_basis):
    """Compute Fourier coefficients for the given function."""
    # Use normalized sinusoids as basis functions
    c = np.zeros(n_basis)
    x = np.linspace(0, 2, len(h))
    # Precompute basis functions
    for i in range(n_basis):
        basis_func = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        basis_func = basis_func / np.linalg.norm(basis_func)  # Normalize
        c[i] = np.dot(h, basis_func)
    return c

def backtransform(c, n_basis, n_points):
    """Transform coefficients back to function values."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_basis):
        basis_func = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        basis_func = basis_func / np.linalg.norm(basis_func)  # Normalize
        h += c[i] * basis_func
    return h

def project(h, target_sum):
    """Project h to feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_basis = n_points // 4  # Number of basis functions

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
        
    h = project(h, target_sum)

    # Keep track of best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)

    # Optimization loop
    max_iter = 50000
    alpha = 0.001
    momentum = 0.9
    v = np.zeros(n_basis)  # velocity for momentum
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 2000

    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current overlap and gradient
        overlap_vals = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap_vals)
        max_idx = np.argmax(overlap_vals)
        
        # Approximate gradient using finite differences
        grad = np.zeros(n_basis)
        eps = 1e-4
        for i in range(n_basis):
            c_plus = np.copy(h)
            c_minus = np.copy(h)
            c_plus[i] += eps
            c_minus[i] -= eps
            
            # Transform back to h space
            h_plus = backtransform(c_plus, n_basis, n_points)
            h_minus = backtransform(c_minus, n_basis, n_points)
            
            # Project to feasible region
            h_plus = project(h_plus, target_sum)
            h_minus = project(h_minus, target_sum)
            
            # Eval overlap
            o_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            o_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            grad[i] = (np.max(o_plus) - np.max(o_minus)) / (2 * eps)
        
        # Update coefficients with momentum
        v = momentum * v - alpha * grad
        new_c = compute_coefficients(h, n_basis) + v
        
        # Transform back to h space
        h_new = backtransform(new_c, n_basis, n_points)
        h_new = project(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # Update current h with projection
        h = h_new.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
