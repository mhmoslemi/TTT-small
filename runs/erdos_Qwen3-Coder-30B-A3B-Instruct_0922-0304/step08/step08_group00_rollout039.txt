import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using correlation with flipped h
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 100
    threshold = 1e-5
    delta = 1e-3
    
    # Initial solution from parent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create basic starting solution
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Perturb solution slightly to introduce variation
        perturbed = h + np.random.uniform(-delta, delta, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum > 0:
            perturbed = perturbed * (target_sum / current_sum)
        
        # Choose even and odd indices for partitioning
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Alternate optimizing even and odd indices
        for idx_set, name in [(even_indices, "even"), (odd_indices, "odd")]:
            # Separate variables
            fixed_vars = np.delete(perturbed, idx_set)
            fixed_sum = np.sum(fixed_vars)
            
            defobjective = lambda var: evaluate_overlap(np.concatenate([fixed_vars, var]), dx)
            
            bounds = [(0, 1) for _ in idx_set]
            
            # Create constraint for sum
            def constraint_func(var):
                return np.sum(var) + fixed_sum - target_sum
            
            # Initial guess for optimizing variables
            init_guess = np.ones(len(idx_set)) * 0.5
            init_guess = np.clip(init_guess, 0, 1)
            
            cons = {'type': 'eq', 'fun': constraint_func}
            
            # Use SLSQP optimization
            try:
                result = minimize(defobjective, init_guess, method='SLSQP', 
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
                if result.success:
                    # Reconstruct solution
                    new_h = perturbed.copy()
                    new_h[idx_set] = result.x
                    # Renormalize
                    current_sum = np.sum(new_h)
                    if current_sum > 0:
                        new_h = new_h * (target_sum / current_sum)
                    
                    eval_c5 = evaluate_overlap(new_h, dx)
                    if eval_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = eval_c5
                        delta *= 0.98  # shrink perturbation as we proceed
            except:
                pass  # Skip infeasible cases
        
        # Periodic full perturbation to avoid local optima
        if iteration % 10 == 0:
            full_perturb = h + np.random.uniform(-delta/2, delta/2, n_points)
            full_perturb = np.clip(full_perturb, 0, 1)
            current_sum = np.sum(full_perturb)
            if current_sum > 0:
                full_perturb = full_perturb * (target_sum / current_sum)
            
            h = full_perturb.copy()

    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(final_h)
    if current_sum > 0:
        final_h = final_h * (target_sum / current_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
