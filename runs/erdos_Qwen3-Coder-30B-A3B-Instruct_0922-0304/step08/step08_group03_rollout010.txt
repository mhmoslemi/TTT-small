import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function for initialization."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to the feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Multi-resolution grid search
    resolution_levels = [20, 40, 70]
    
    for r in resolution_levels:
        if time.time() - start_time > budget_s - 1:
            break
        n = r
        dx_r = 2.0 / n
        target_sum_r = n / 2
        
        h_r = np.linspace(h[0], h[-1], n)
        h_r = project_h(h_r, target_sum_r)
        
        # Simple gradient descent on the projected space
        iterations = 1000
        learning_rate = 0.005
        momentum = 0.9
        velocity = np.zeros_like(h_r)
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute overlap and gradients
            overlap = correlate(h_r, 1 - h_r, mode="full") * dx_r
            max_overlap = np.max(overlap)
            
            # Gradient approximation for each point
            grad = np.zeros_like(h_r)
            eps = 1e-4
            for i in range(n):
                h_plus = h_r.copy()
                h_minus = h_r.copy()
                h_plus[i] = min(1.0, h_r[i] + eps)
                h_minus[i] = max(0.0, h_r[i] - eps)
                
                h_plus = project_h(h_plus, target_sum_r)
                h_minus = project_h(h_minus, target_sum_r)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx_r
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx_r
                
                grad[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps)
            
            # Update using momentum
            velocity = momentum * velocity - learning_rate * grad
            h_r = h_r + velocity
            h_r = project_h(h_r, target_sum_r)
            
            current_c5 = compute_c5(h_r, dx_r)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_r.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Final refinement: direct local search
    if time.time() - start_time < budget_s - 1:
        h_local = best_h.copy()
        iterations = 5000
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb one element at a time
            i = np.random.randint(0, n_points)
            old_val = h_local[i]
            step = np.random.uniform(-0.01, 0.01)
            h_local[i] = old_val + step
            h_local = project_h(h_local, target_sum)
            
            new_c5 = compute_c5(h_local, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_local.copy()
            else:
                # Restore if didn't improve
                h_local[i] = old_val
    
    return (best_h.tolist(), best_c5, n_points)
