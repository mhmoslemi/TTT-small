import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Try multiple random initializations
    for attempt in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize from the parent construction if available
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, n_points / 2)
        else:
            # Use a different initialization strategy
            h = np.random.rand(n_points).astype(np.float64)
            h = normalize_h(h, n_points / 2)
        
        # Simple local search with random perturbations
        for iter_no in range(5000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Save current state
            current_h = h.copy()
            current_c5 = evaluate_c5(h, dx)
            
            # Make a small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_perturbed = h + perturbation
            
            # Apply bounds and normalization
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, n_points / 2)
            
            # Evaluate the perturbed solution
            new_c5 = evaluate_c5(h_perturbed, dx)
            
            # Accept better solutions
            if new_c5 < current_c5:
                h = h_perturbed
                current_c5 = new_c5
                
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early stopping if we've hit our target
            if best_c5 <= 0.3807:
                break
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    # Return the best solution found
    if best_h is None:
        # Fallback to simple initialization
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, n_points / 2)
        else:
            best_h = np.ones(n_points) * 0.5
            best_h = normalize_h(best_h, n_points / 2)
        best_c5 = evaluate_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
