import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Use a structured pattern for initial solution
        h = np.zeros(n_points)
        # Create a step function that is 1 on the middle portion
        mid_start = n_points // 4
        mid_end = 3 * n_points // 4
        h[mid_start:mid_end] = 1.0
        # Add some noise for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    T = 1.0  # initial temperature
    cooling_rate = 0.999
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Simulated Annealing Phase
        # Generate candidate solution by perturbing h
        candidate_h = h.copy()
        
        # Randomly select 10% of positions to perturb
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            if np.random.random() < 0.5:
                # Flip the value
                candidate_h[i] *= 0.97
            else:
                # Swap values (just for diversity)
                j = np.random.randint(0, n_points)
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Clip and normalize
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = normalize_h(candidate_h, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        # Accept or reject the candidate using SA criteria
        if candidate_c5 < best_c5 or np.random.random() < np.exp(-(candidate_c5 - best_c5) / T):
            h = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Update temperature
        T *= cooling_rate
        
        # Cutting-plane refinement (on every 10th iteration)
        if iteration % 10 == 0:
            # Compute overlap for all lags
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            
            # Refine by adjusting h to reduce overlapping
            # Identify influential indices around maximum overlap
            k = max_idx - (n_points - 1)  # lag corresponding to max overlap
            if abs(k) < n_points:
                # Adjust values
                epsilon = 0.005
                for i in range(n_points):
                    if 0 <= i + k < n_points:
                        h[i] = max(0, h[i] - epsilon)
                        h[i + k] = min(1, h[i + k] + epsilon)
            
            # Ensure normalization
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
