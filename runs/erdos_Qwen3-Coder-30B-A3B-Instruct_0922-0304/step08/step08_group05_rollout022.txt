import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with an orthogonal pattern similar to previous work
        h = np.zeros(n_points)
        h[n_points//4:n_points*3//4] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    
    # For exploration, use simulated annealing approach
    T = 1.0
    cooling_rate = 0.999
    epsilon = 0.01
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate candidate solution
        candidate = h.copy()
        
        # Mutate by randomly selecting a fraction of indices
        num_mutate = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, num_mutate, replace=False)
        
        for i in indices:
            # Randomly decide to flip or modify
            if np.random.random() < 0.5:
                # Scale down by some amount
                candidate[i] *= 0.95
            else:
                # Add random Gaussian noise
                candidate[i] += np.random.normal(0, 0.03)
        
        # Clip and enforce normalization
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        
        # Evaluate C5 for candidate
        candidate_c5 = compute_c5_exact(candidate, dx)
        
        # Acceptance probability
        if candidate_c5 < best_c5 or np.random.random() < np.exp(-(candidate_c5 - best_c5) / T):
            h = candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        # Cool down
        T *= cooling_rate
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with local search on the best solution
    final_h = best_h.copy()
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Apply small perturbations
        candidate = final_h.copy()
        k = np.random.randint(0, n_points)
        perturb_factor = np.random.normal(0, 0.01)
        candidate[k] += perturb_factor
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        
        new_c5 = compute_c5_exact(candidate, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            final_h = candidate.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5_exact(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
