import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Temperature and parameters
    start_temp = 0.1
    cooling_rate = 0.999
    current_temp = start_temp
    max_iter = 100000
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Make a copy of current solution
        new_h = h.copy()
        
        # Randomly choose perturbation
        choice = np.random.choice(['flip', 'adjust', 'shuffle'], 
                                 p=[0.5, 0.3, 0.2])
        
        if choice == 'flip':
            i = np.random.randint(n_points)
            new_h[i] = 1 - new_h[i]
        elif choice == 'adjust':
            i = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            new_h[i] += delta
        elif choice == 'shuffle':
            length = np.random.randint(1, min(5, n_points // 2))
            i = np.random.randint(n_points - length + 1)
            segment = new_h[i:i+length].copy()
            np.random.shuffle(segment)
            new_h[i:i+length] = segment
        
        # Enforce constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new C5
        new_c5 = compute_c5(new_h, dx)
        
        # Accept or reject the new solution
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            delta_e = new_c5 - best_c5
            if np.exp(-delta_e / current_temp) > np.random.rand():
                h = new_h.copy()
        
        # Update temperature
        current_temp *= cooling_rate
        
        # Occasionally increase temperature if no improvement
        if iter_count % 1000 == 0 and iter_count > 0:
            if new_c5 >= best_c5:
                current_temp = max(0.01, current_temp * 1.1)
        
        iter_count += 1
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
