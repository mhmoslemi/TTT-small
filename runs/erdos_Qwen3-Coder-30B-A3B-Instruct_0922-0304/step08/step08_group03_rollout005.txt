import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    if n % 2 == 0:
        return np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        return np.concatenate([np.arange(1, half + 2), np.arange(half, 0, -1)])

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    # Parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    p = 0.2  # sampling percentage
    
    # Variables for momentum
    velocity = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iters_since_improvement = 0
    max_iters_no_improvement = 5000
    
    # Phase 1: Gradient descent with momentum
    phase = 1
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Find maximum
        max_pos = np.argmax(overlap)
        
        # Compute gradient for current maximum
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Create shifted version manually for index k
            k = max_pos - n_points + 1  # due to full correlation
            shifted_val = 0
            if 0 <= i - k < n_points:
                shifted_val = h[i - k]
            gradient[i] = (1 - 2 * h[i]) * (1 - shifted_val)
        
        # Apply momentum
        velocity = beta * velocity + (1 - beta) * gradient
        h_candidate = h - alpha * velocity
        
        # Project to feasible set
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Check if improvement
        new_c5 = compute_c5(h_candidate, dx)
        if new_c5 < overlap_max:
            h = h_candidate.copy()
            alpha *= gamma
            iters_since_improvement = 0
        else:
            # Revert
            iters_since_improvement += 1
            alpha *= gamma
        
        # Decision to switch phases
        if phase == 1 and iters_since_improvement > 2000:
            phase = 2
            iters_since_improvement = 0
        
        if phase == 2:
            # Coordinate-wise updates
            num_updates = max(1, int(p * n_points))
            indices = np.random.choice(n_points, num_updates, replace=False)
            
            for i in indices:
                # Try both increasing and decreasing
                original = h[i]
                best_change = 0
                best_val = original
                
                # Try small positive step
                h_test = h.copy()
                h_test[i] = min(1.0, original + 0.01)
                h_test = clip_and_renormalize(h_test, target_sum)
                test_c5 = compute_c5(h_test, dx)
                
                if test_c5 < overlap_max:
                    best_change = 0.01
                    best_val = min(1.0, original + 0.01)
                
                # Try small negative step
                h_test = h.copy()
                h_test[i] = max(0.0, original - 0.01)
                h_test = clip_and_renormalize(h_test, target_sum)
                test_c5 = compute_c5(h_test, dx)
                
                if test_c5 < overlap_max:
                    if best_change == 0 or test_c5 < compute_c5(h.copy(), dx):
                        best_change = -0.01
                        best_val = max(0.0, original - 0.01)
                
                # Apply best change if any
                if best_change != 0:
                    h[i] = best_val
                    h = clip_and_renormalize(h, target_sum)
        
        iteration += 1
        
        # Termination conditions
        if iters_since_improvement > max_iters_no_improvement:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
