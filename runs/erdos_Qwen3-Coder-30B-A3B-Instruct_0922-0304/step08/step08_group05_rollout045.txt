import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_h(h, dx, target_sum):
    """Generate a perturbed version of h."""
    n = len(h)
    new_h = h.copy()
    
    # Flip 10% of the values
    indices = np.random.choice(n, size=int(0.1 * n), replace=False)
    for i in indices:
        new_h[i] *= 0.97
        if new_h[i] < 0:
            new_h[i] = 0
    
    # Add noise
    noise = np.random.normal(0, 0.01, n)
    new_h += noise
    
    # Clip and normalize
    new_h = np.clip(new_h, 0, 1)
    new_h = normalize_h(new_h, target_sum)
    
    return new_h

def cutting_plane_update(h, dx, target_sum):
    """Perform a cutting-plane update on h to reduce overlap."""
    n = len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Find the peak and some nearby significant values
    for i in range(max(0, max_idx - 5), min(n, max_idx + 5)):
        if i < n:
            # Decrease h[i] and increase the appropriate h[i+k] to reduce overlap
            if i < n - 1:
                h[i] = max(0, h[i] - 0.005)
                h[i + 1] = min(1, h[i + 1] + 0.005)
    
    # Normalize and clip
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        # Initialize with a simple pattern
        h = np.zeros(n_points)
        h[n_points//4 : 3*n_points//4] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    temp = 1.0
    cooling_rate = 0.999
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Simulated Annealing Phase
        candidate_h = perturb_h(h, dx, target_sum)
        current_c5 = compute_c5_exact(h, dx)
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        if candidate_c5 < current_c5:
            h = candidate_h
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.exp(-delta / temp) > np.random.rand():
                h = candidate_h
        
        # Cutting-plane update phase
        if iteration % 20 == 0:
            h = cutting_plane_update(h, dx, target_sum)
        
        # Constraint repair
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        temp *= cooling_rate
        last_improvement += 1
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
