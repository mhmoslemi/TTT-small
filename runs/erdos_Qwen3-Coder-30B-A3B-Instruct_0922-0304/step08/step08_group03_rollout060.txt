import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp from 0 to 1 and back to 0 over [0, 2]."""
    x = np.linspace(0, 2, n)
    h = np.zeros_like(x)
    for i in range(n):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    return h

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_target."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h with triangular ramp or initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = triangular_ramp(n_points) * (target_sum / np.sum(triangular_ramp(n_points)))
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Precompute grid for basis functions
    x_grid = np.linspace(0, 2, n_points)
    
    # Multi-resolution optimization
    resolutions = [50, 70, 80]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use smaller resolution for first few steps
        if res < n_points:
            x_small = np.linspace(0, 2, res)
            h_resampled = np.interp(x_small, x_grid, h)
            h = project_h(h_resampled, target_sum)
        else:
            h = project_h(h, target_sum)
            
        # Coordinate-wise updates with Wasserstein gradient
        iter_count = 0
        max_iter = 5000
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 1:
            # Evaluate current overlap
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
            
            if current_max < best_c5:
                best_c5 = current_max
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Simple coordinate update based on greedy heuristic
            improved = False
            
            # Try many possible updates
            for i in np.random.permutation(n_points):
                # Try small increases
                h_new = h.copy()
                if h[i] < 1:
                    h_new[i] = min(1, h[i] + 0.01)  
                    h_new = project_h(h_new, target_sum)
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < current_max:
                        h = h_new
                        improved = True
                        break
                        
                # Try small decreases
                h_new = h.copy()
                if h[i] > 0:
                    h_new[i] = max(0, h[i] - 0.01)
                    h_new = project_h(h_new, target_sum)
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < current_max:
                        h = h_new
                        improved = True
                        break
                        
            if not improved:
                break
                
            iter_count += 1
            
        # Polyak step-size refinement
        refined = False
        for i in range(4):  # at most 4 refinements
            if time.time() - start_time > budget_s - 1 or refined:
                break
            h_try = h.copy()
            for j in range(n_points):
                step = 0.005 * np.random.randn()
                if h_try[j] + step > 0 and h_try[j] + step < 1:
                    h_try[j] += step
            h_try = project_h(h_try, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
                refined = True
                # Early exit if we've improved significantly
                if best_c5 < 0.3807:
                    break
                    
        # Standard restart with random perturbation if needed
        if best_c5 <= 0.3807:
            break
            
        if np.random.random() < 0.1 and iter_count > 0:
            # Restart with random modest perturbation
            h = project_h(h + (np.random.rand(n_points) - 0.5) * 0.02, target_sum)
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
