import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    # Create a triangular ramp from 0 to 1
    ramp = np.zeros(n)
    mid = n // 2
    ramp[:mid] = np.linspace(0, 1, mid)
    ramp[mid:] = np.linspace(1, 0, n - mid)
    return ramp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
        
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Momentum parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    
    # Initialize momentum
    velocity = np.zeros(n_points)
    
    # Optimization parameters
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 5000
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Gradient estimation by finite differences
        gradients = np.zeros(n_points)
        epsilon = 1e-4
        
        # Sample a subset of coordinates for efficiency
        sample_size = max(1, n_points // 5)
        sample_indices = np.random.choice(n_points, sample_size, replace=False)
        
        for i in sample_indices:
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + epsilon)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - epsilon)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            gradients[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * epsilon)
            
        # Compute momentum
        velocity = beta * velocity + (1 - beta) * gradients
        
        # Gradient descent step
        h_candidate = h - alpha * velocity
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < current_max:
            h = h_candidate.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            alpha *= gamma  # Decay learning rate
            no_improvement_count = 0
        else:
            # No improvement, decrease step size
            alpha *= gamma
            no_improvement_count += 1
            
        # Switch to coordinate-wise optimization if stagnating
        if no_improvement_count > max_no_improvement:
            # Large random modifications
            for _ in range(100):
                if time.time() - start_time > budget_s - 1:
                    break
                i = np.random.randint(0, n_points)
                h_trial = h.copy()
                step = np.random.uniform(-0.01, 0.01)
                h_trial[i] = np.clip(h[i] + step, 0, 1)
                h_trial = clip_and_renormalize(h_trial, target_sum)
                trial_c5 = compute_c5(h_trial, dx)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
