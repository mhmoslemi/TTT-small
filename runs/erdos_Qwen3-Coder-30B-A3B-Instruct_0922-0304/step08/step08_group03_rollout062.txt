import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n_points):
    """Create a triangular ramp function as initial guess."""
    x = np.linspace(0, 2, n_points)
    h = np.clip(1 - np.abs(x - 1), 0, 1)
    return h

def compute_coefficients(h, n_basis):
    """Compute Fourier coefficients for h using windowed sinusoids."""
    # Precompute basis functions
    x = np.linspace(0, 2, len(h))
    basis = np.zeros((n_basis, len(h)))
    for i in range(n_basis):
        basis[i] = np.sin((i+1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        basis[i] /= np.sqrt(np.sum(basis[i]**2))  # Normalize
    
    # Compute coefficients
    coeffs = np.zeros(n_basis)
    for i in range(n_basis):
        coeffs[i] = np.dot(h, basis[i])
    return coeffs, basis

def backtransform(coeffs, basis):
    """Transform coefficients back to h."""
    h = np.dot(coeffs, basis)
    return h

def project(h, target_sum):
    """Project h to feasible region with target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute C5 value."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

    h = project(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Multi-resolution grid search
    resolutions = [10, 20, 35, 50, 70]
    # Base optimization with small number of basis functions
    n_basis = min(15, n_points // 4)
    
    # Optimization with multiple restarts
    for restart in range(3):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization
        if restart > 0:
            h = triangular_ramp(n_points)
            h = project(h, target_sum)
            
        # Fourier basis optimization
        coeffs, basis = compute_coefficients(h, n_basis)
        momentum = np.zeros_like(coeffs)
        alpha = 0.002
        beta = 0.9
        
        # Optimization loop
        max_iter = 10000
        iter_count = 0
        stagnation_count = 0
        max_stagnation = 200
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 1:
            # Compute overlap and gradient
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
            if current_max < best_c5:
                best_c5 = current_max
                best_h = h.copy()
                
            # Compute gradient
            # Approximate: -2 * h * (1-shift_k(h))
            # Simplified gradient normalized by n_points
            grad = np.zeros_like(coeffs)
            
            # Estimate gradient by perturbing each coefficient
            eps = 1e-4
            for i in range(len(coeffs)):
                coeffs_plus = coeffs.copy()
                coeffs_plus[i] += eps
                h_plus = backtransform(coeffs_plus, basis)
                h_plus = project(h_plus, target_sum)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                max_plus = np.max(overlap_plus)
                
                grad[i] = (max_plus - current_max) / eps
                
            # Update with momentum
            momentum = beta * momentum + (1 - beta) * grad
            coeffs = coeffs - alpha * momentum
            
            # Back-transform and project
            h = backtransform(coeffs, basis)
            h = project(h, target_sum)
            
            iter_count += 1
            
            if current_max <= best_c5 + 1e-8:
                stagnation_count += 1
            else:
                stagnation_count = 0
                
            if stagnation_count > max_stagnation:
                break
                
        # Grid refinement loop
        for level_idx, level_points in enumerate(resolutions):
            if time.time() - start_time > budget_s - 1:
                break

                x_old = np.linspace(0, 2, n_points)

                h_new = project(h_new, level_points / 2)

                current_h = h_new.copy()

                    if current_max < best_c5:
                        best_c5 = current_max
                        best_h = current_h.copy()
                        
                    # Simple update via projected gradient descent in space
                    grad_h = np.zeros_like(current_h)
                    small_delta = 1e-4
                    
                    for i in range(len(current_h)):
                        h_plus = current_h.copy()
                        h_plus[i] += small_delta
                        h_plus = project(h_plus, level_points / 2)

                        max_plus = np.max(overlap_plus)
                        
                        h_minus = current_h.copy()
                        h_minus[i] -= small_delta
                        h_minus = project(h_minus, level_points / 2)

                        max_minus = np.max(overlap_minus)
                        
                        grad_h[i] = (max_plus - max_minus) / (2 * small_delta)
                        
                    # Projected update
                    current_h = current_h - 0.01 * grad_h
                    current_h = project(current_h, level_points / 2)
                    
                    if current_max <= best_c5 + 1e-8:
                        break
                        
                if current_h is not None and time.time() - start_time < budget_s - 1:
                    h = current_h.copy()
                    
                    # Final check with full resolution
                    final_overlap = correlate(h, 1 - h, mode="full") * dx
                    final_c5 = np.max(final_overlap)
                    if final_c5 < best_c5:
                        best_c5 = final_c5
                        best_h = h.copy()
                        
    return (best_h.tolist(), best_c5, n_points)
