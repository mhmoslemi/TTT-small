import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def initialize_population(n_points, target_sum, initial_h=None):
    """Initialize population with diverse individuals."""
    population = []
    
    # Add initial solution if available
    if initial_h is not None:
        population.append(np.array(initial_h))
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Generate diverse random individuals
    for _ in range(49):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    return population

def selection(population, fitness, tournament_size=5):
    """Tournament selection."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2, target_sum):
    """Average-based crossover with normalization."""
    child = (parent1 + parent2) / 2.0
    child = normalize_h(child, target_sum)
    child = np.clip(child, 0, 1)
    child = normalize_h(child, target_sum)
    return child

def mutate(individual, mutation_rate=0.1, sigma=0.1, target_sum=35.0):
    """Gaussian mutation with constraint repair."""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual = individual + noise
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    population = initialize_population(n_points, target_sum, initial_h)
    
    # Evaluation
    fitness = [compute_c5_exact(ind, dx) for ind in population]
    
    # Tracking
    best_c5 = min(fitness)
    best_individual = population[np.argmin(fitness)].copy()
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Selection
        selected = selection(population, fitness)
        
        # Create new population
        new_population = [best_individual.copy()]
        while len(new_population) < len(population):
            # Crossover
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(parent1, parent2, target_sum)
            
            # Mutation
            child = mutate(child, target_sum=target_sum)
            new_population.append(child)
        population = new_population[:len(population)]
        
        # Evaluate
        fitness = [compute_c5_exact(ind, dx) for ind in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_individual = population[current_best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        generation += 1
    
    # Final check
    final_c5 = compute_c5_exact(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
