import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum initially
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Warm up: slightly adjust to enforce sum constraint
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Search parameters
    max_iter = 100000
    epsilon = 0.01
    decay_factor = 0.9999
    patience = 0
    max_patience = 500
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        current_c5 = overlap[max_idx] * dx
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Adjust h to reduce the maximum overlap
        # Get the best lag that produces maximum overlap
        lag = max_idx - (n_points - 1)
        # Find indices where the maximum overlap is achieved
        max_positions = np.where(overlap == overlap[max_idx])[0]
        # Pick one arbitrarily
        max_pos = max_positions[0] if len(max_positions) > 0 else max_idx
        k = max_pos - (n_points - 1)

        # Compute adjusted h that reduces overlap the most
        # We need to increase h[i] and decrease h[i-k] for some i
        new_h = h.copy()
        
        # Find settings that can influence the overlap
        # Focus on the positions involved in maximum overlap
        # Roughly, for positions i and i+k
        if lag > 0:
            # This means the overlap is due to h[i]*1-h[i+lag]
            # So we adjust i to reduce and i+lag to increase
            # Or equivalently, adjust the earlier position
            pos1 = 0
            pos2 = pos1 + lag
            
            # Check safe adjustments
            # Increasing h[pos1] gives more to subtract from overlap
            # If we reduce overlap_1 = h[pos1] * (1 - h[pos1 + lag])
            # Then we adjust h[pos1] to reduce this value
            # The derivative is (1 - h[pos1 + lag]) - h[pos1] = 
            # 1 - h[pos1 + lag] - h[pos1]
            # So for small changes, we increase h[pos1] to reduce overlap_1
            if pos1 < n_points and pos2 < n_points:
                if new_h[pos1] < 1.0 and new_h[pos2] > 0.0:
                    # Small adjustment
                    delta = epsilon if k >= 0 else -epsilon
                    new_h[pos1] = min(1.0, new_h[pos1] + 0.5 * delta)
                    new_h[pos2] = max(0.0, new_h[pos2] - 0.5 * delta)
        else:
            # lag < 0, so overlap due to position i and i+|lag|
            pos1 = 0
            pos2 = pos1 - lag  # positive value
            if pos1 < n_points and pos2 < n_points:
                if new_h[pos1] < 1.0 and new_h[pos2] > 0.0:
                    delta = epsilon if k <= 0 else -epsilon
                    new_h[pos1] = min(1.0, new_h[pos1] + 0.5 * delta)
                    new_h[pos2] = max(0.0, new_h[pos2] - 0.5 * delta)
        
        # Normalize and clip
        new_h = new_h * (target_sum / np.sum(new_h))
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (target_sum / np.sum(new_h))
        
        # Only use the new h if it improved
        new_c5 = compute_c5_exact(new_h, dx)
        if new_c5 < current_c5:
            h = new_h.copy()
        else:
            # Adjust epsilon for successive non-improvements
            epsilon *= decay_factor
            
        if last_improvement > max_patience:
            break
    
    # Final check
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
