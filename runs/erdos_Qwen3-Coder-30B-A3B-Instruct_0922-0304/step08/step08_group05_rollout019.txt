import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:

        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:

        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")

        max_idx = np.argmax(overlap)

        # Use a more direct adjustment based on the maximum overlap lag
        # and neighboring elements
        
        # The idea is to modify h[j] to reduce contribution of max_overlap
        # For details see how the convolution works
        # The maximum overlap happens at index max_idx, which is in [0, 2*n_points-2]

        lag = max_idx - (n_points - 1)

        # We want to reduce h[i] and increase h[i+lag] where possible,
        # But we need to respect h bounds
        
        # Let's sweep around, but limit to reasonable number of elements
        search_range = min(20, n_points // 2)
        best_i = -1
        best_improvement = -1
        
        # Evaluate changes for both directions
        for i in range(max(0, max_idx - search_range), min(n_points, max_idx + search_range)):

            if i + lag < 0 or i + lag >= n_points:
