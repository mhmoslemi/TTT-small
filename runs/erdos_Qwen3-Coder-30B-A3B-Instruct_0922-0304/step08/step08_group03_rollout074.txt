import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def gradient_descent_step(h, dx, max_iter=100, step_size=0.01):
    """Apply gradient descent refinement to reduce C5."""
    h = h.copy()
    target_sum = len(h) / 2
    
    for _ in range(max_iter):
        current_c5 = compute_c5(h, dx)
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(overlap)
        
        # Simple gradient descent using finite differences
        eps = 1e-4
        h_plus = h.copy()
        h_plus[max_idx % len(h)] = min(1.0, h[max_idx % len(h)] + eps)
        h_plus = repair_h(h_plus, target_sum)
        
        h_minus = h.copy()
        h_minus[max_idx % len(h)] = max(0.0, h[max_idx % len(h)] - eps)
        h_minus = repair_h(h_minus, target_sum)
        
        grad = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
        
        # Update step
        if grad != 0:
            new_val = h[max_idx % len(h)] - step_size * grad
            h[max_idx % len(h)] = np.clip(new_val, 0, 1)
            h = repair_h(h, target_sum)
        
        # Check if improvement
        if compute_c5(h, dx) < current_c5:
            continue
        else:
            break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h

    else:

    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic algorithm with gradient descent refinement
    population_size = 50
    num_generations = 2000
    crossover_prob = 0.7
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    for _ in range(population_size):
        individual = np.random.rand(n_points)
        individual = repair_h(individual, target_sum)
        population.append(individual)
    
    # Seed with current best
    if 'initial_h_values' in globals():
        population[0] = np.array(initial_h_values)
        population[0] = repair_h(population[0], target_sum)
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Selection (top 50%)
        selected = population[:population_size//2]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child = repair_h(child, target_sum)
                new_population.append(child)
            
            # Mutation
            elif np.random.rand() < mutation_rate:
                individual = np.random.choice(selected)
                individual = individual.copy()
                mutation_idx = np.random.randint(n_points)
                individual[mutation_idx] = np.random.rand()
                individual = repair_h(individual, target_sum)
                new_population.append(individual)
                
            # No operator, copy selected
            else:
                individual = np.random.choice(selected)
                new_population.append(individual.copy())
                
        # Refine top individuals using gradient descent
        for i in range(min(10, len(new_population))):
            new_population[i] = gradient_descent_step(new_population[i], dx)
            
        population = new_population[:population_size]
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
