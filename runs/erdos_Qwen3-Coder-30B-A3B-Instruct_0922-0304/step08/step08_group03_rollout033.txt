import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n_points):
    half = n_points // 2
    ramp = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    return ramp / (half + 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h properly
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with momentum
    momentum = 0.9
    velocity = np.zeros(n_points)
    learning_rate = 0.001
    decay_rate = 0.99
    
    # Coordinate-wise updates for later phase
    coord_sample_rate = 0.2
    max_coord_updates = int(coord_sample_rate * n_points)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Phase 1: Gradient descent with momentum
    phase1_iter = 0
    while iter_count < 10000 and time.time() - start_time < budget_s - 1:
        if time.time() - start_time > budget_s - 1:
            break
            
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute gradient (approximated)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            gradient[i] = (max_plus - max_minus) / 0.02
        
        velocity = momentum * velocity + (1 - momentum) * gradient
        h_candidate = h - learning_rate * velocity
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate
            learning_rate *= decay_rate
            no_improvement_count = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert and reduce learning rate
            learning_rate *= 0.9
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
                
        iter_count += 1
        phase1_iter += 1
    
    # Phase 2: Coordinate-wise optimization
    if time.time() - start_time < budget_s - 1:
        max_coord_iter = 10000
        coord_iter = 0
        
        while coord_iter < max_coord_iter and time.time() - start_time < budget_s - 1:
            # Randomly sample coordinates
            idxs = np.random.choice(n_points, size=max_coord_updates, replace=False)
            
            improved = False
            for i in idxs:
                # Approximate gradient for i
                h_plus = h.copy()
                h_plus[i] = min(1.0, h[i] + 0.01)
                h_plus = clip_and_renormalize(h_plus, target_sum)
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                max_plus = np.max(overlap_plus)
                
                h_minus = h.copy()
                h_minus[i] = max(0.0, h[i] - 0.01)
                h_minus = clip_and_renormalize(h_minus, target_sum)
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
                max_minus = np.max(overlap_minus)
                
                grad = (max_plus - max_minus) / 0.02
                
                # Adjust based on gradient
                if abs(grad) > 0.0001:
                    step = 0.01 * grad / abs(grad)
                    new_h = h.copy()
                    new_h[i] = np.clip(h[i] - step, 0.0, 1.0)
                    new_h = clip_and_renormalize(new_h, target_sum)
                    new_c5 = compute_c5(new_h, dx)
                    if new_c5 < best_c5:
                        h = new_h
                        best_c5 = new_c5
                        improved = True
                        
            if not improved:
                no_improvement_count += 1
                if no_improvement_count > max_no_improvement:
                    break
            else:
                no_improvement_count = 0
            
            coord_iter += 1
            iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
