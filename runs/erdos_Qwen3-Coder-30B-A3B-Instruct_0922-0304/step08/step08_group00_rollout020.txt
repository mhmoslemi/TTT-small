import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iterations = 1000
    w = 0.8
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.05
    tolerance = 1e-6
    
    # Initialize particles
    particles = [h.copy()]
    for i in range(n_particles - 1):
        new_h = np.random.uniform(0, 1, n_points)
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        particles.append(new_h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Evaluate initial particles
    fitness = [evaluate_overlap(p, dx) for p in particles]
    
    # Initialize best positions
    p_best = particles.copy()
    p_best_fitness = fitness.copy()
    
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    # Keep track of stagnation
    stagnation_count = 0
    previous_best = best_c5
    
    # PSO main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Ensure constraints
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            if current_sum > 0:
                particles[i] = particles[i] * (target_sum / current_sum)
            
            # Evaluate fitness
            new_fitness = evaluate_overlap(particles[i], dx)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = new_fitness
                    stagnation_count = 0
                    if g_best_fitness < best_c5:
                        best_h = g_best.copy()
                        best_c5 = g_best_fitness
            else:
                stagnation_count += 1
        
        # Apply shaking if no improvement
        if stagnation_count > 50:
            for i in range(n_particles):
                # Add small random perturbation
                perturbation = np.random.normal(0, 0.01, n_points)
                particles[i] += perturbation
                # Ensure constraints after perturbation
                particles[i] = np.clip(particles[i], 0, 1)
                current_sum = np.sum(particles[i])
                if current_sum > 0:
                    particles[i] = particles[i] * (target_sum / current_sum)
            stagnation_count = 0
            
        # Check for early exit
        if abs(previous_best - best_c5) < tolerance:
            stagnation_count += 1
        else:
            stagnation_count = 0
        previous_best = best_c5
        
        if stagnation_count > 100:
            break
    
    # Local refinement on best solution
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
