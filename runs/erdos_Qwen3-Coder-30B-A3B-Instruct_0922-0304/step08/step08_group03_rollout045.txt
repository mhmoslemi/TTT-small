import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy the integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute the gradient of C5 approximation."""
    # Simplified gradient assuming uniform distribution influence
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
    
    population.append(h.copy())
    
    # Initialize others with variations
    for _ in range(population_size - 1):
        h_var = h + np.random.normal(0, 0.05, n_points)
        h_var = repair_h(h_var, target_sum)
        population.append(h_var.copy())
    
    # Parameters for GA
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_generations = 2000
    improvement_threshold = 100
    no_improvement_count = 0
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    for gen in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        fitness_sorted = [fitness[i] for i in sorted_indices]
        population_sorted = [population[i] for i in sorted_indices]
        
        # Update best
        if fitness_sorted[0] < best_c5:
            best_c5 = fitness_sorted[0]
            best_h = population_sorted[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if no_improvement_count > improvement_threshold:
            mutation_rate = min(0.3, mutation_rate * 1.1)
            no_improvement_count = 0
            
        # Selection (top 50%)
        selected = population_sorted[:population_size//2]
        
        # Create new population
        new_population = [selected[0].copy()]  # Keep best
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.03)
            
            # Repair
            child = repair_h(child, target_sum)
            
            # Local refinement via gradient descent
            # Compute initial gradient
            grad = compute_gradient(child, n_points)
            # Take small step in negative gradient direction
            step_size = 0.001 / (np.max(np.abs(grad)) + 1e-8)  # Avoid division by zero
            child -= grad * step_size
            child = repair_h(child, target_sum)
            
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Final computation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
