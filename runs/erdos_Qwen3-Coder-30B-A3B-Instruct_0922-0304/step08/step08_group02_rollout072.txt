```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample shifts for QP constraints"""
    shifts = np.random.choice(range(-n+1, n), sample_size, replace=False)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Initialize variables for adaptive strategy
    max_shifts = 50
    num_iter = 0
    patience = 0
    min_improvement = 1e-6
    target_overlap = 0.38085  # Start with known upper bound
    
    while time.time() - start_time < budget_s - 1:
        # Sample subset of shifts
        shifts = sample_shifts(n_points, max_shifts)
        
        # Build QP problem
        h_vars = cp.Variable(n_points)
        overlap_vars = cp.Variable(len(shifts))
        
        # Objective
        objective = cp.Minimize(cp.max(overlap_vars))
        
        # Constraints
        constraints = []
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # Overlap constraints
        for i, shift in enumerate(shifts):
            # Compute the correlation integral for this shift
            if shift >= 0:
                corr = np.concatenate([np.zeros(shift), h_current[:n_points-shift]]) * (1 - h_current[shift:])
            else:
                corr = h_current[-shift:] * (1 - np.concatenate([np.zeros(-shift), h_current[:n_points+shift]]))
            # Only consider valid part
            if shift >= 0:
                total_corr = np.sum(corr) * dx
            else:
                total_corr = np.sum(corr) * dx
                
            constraints.append(overlap_vars[i] >= total_corr)
        
        # Formulate and solve QP problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            h_new = h_vars.value
        except:
            h_new = h_current.copy()
            
        # Project to feasible region
        h_new = project_h(h_new, n_points)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
            # Adjust target after improvement
            target_overlap = best_c5 * 0.99
        else:
            patience += 1
            
        # Update current solution
        h_current = h_new.copy()
        
        num_iter += 1
        
        # Early stopping if no improvement for a while
        if patience > 200:
            break
            
        # Adaptive target update
        if num_iter % 10 == 0:
            target_overlap = max(best_c5 * 0.99, 0.3805)
            
    # Final projection and validation
    final_h = project_h(best_h, n_points)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
