```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure it's within [0,1] and sums to n_points/2."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of possible shifts."""
    # Sample shifts from the full correlation range (negative to positive)
    shifts = np.arange(1 - n, n)
    selected = np.random.choice(shifts, size=min(sample_size, len(shifts)), replace=False)
    return selected

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Initialize QP variables
    h_vars = cp.Variable(n_points)
    target_overlap = 0.38085  # Initial estimate
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Sample shifts
        shifts = sample_shifts(n_points, 50)
        
        # Create a QP with sampled overlaps
        constraints = []
        # Add bounds
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        # Add sum constraint
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # Add overlap constraints
        # For now, we use a simplified version approximating maximum overlap
        objective = cp.Minimize(cp.sum_squares(h_vars))  # Simple penalization for smoothness
        
        # Solve QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                # Project to feasible region
                h_new = project_h(h_new, n_points)
                c5_new = compute_c5(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    no_improvement_count = 0
                    target_overlap = min(target_overlap, best_c5 * 0.99)
                else:
                    no_improvement_count += 1
        except:
            pass  # If QP solver fails, continue

        # Update iteration
        h_current = best_h.copy()
        iteration += 1
        
        # Final check for stopping condition
        if best_c5 <= 0.3807:
            break

    # Final refinement with a simple step optimization
    h_refined = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_refined.copy()
        # Perturb one element
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
```
