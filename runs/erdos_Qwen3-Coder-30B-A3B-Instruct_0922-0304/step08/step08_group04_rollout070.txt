import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def frank_wolfe_step(h, dx, n_points, max_iter=1000, tolerance=1e-8):
    """Frank-Wolfe optimization."""
    target_sum = n_points / 2.0
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        if i % 100 == 0:
            # Recheck the loss from current.
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 200:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss

        # Compute gradient
        grad = compute_gradient(h, dx, n_points) 
        # Solve linear subproblem
        v = solve_linear_subproblem(grad, n_points, target_sum)
        
        # Determine step size (line search or fixed)
        # Armijo rule or direct:
        eta = 0.1
        h = (1 - eta) * h + eta * v
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
        
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5
