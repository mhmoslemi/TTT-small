```python
import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h to satisfy sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Initial evaluation
    best_c5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    
    # Try convex optimization approach
    try:
        # Create variables
        h_var = cp.Variable(n_points)
        
        # Objective: minimize max overlap
        # We'll realize this by minimizing max over all possible shifts
        # This approach uses a two-stage method:
        # Stage 1: convex optimization to find a good baseline
        # Stage 2: local refinement using SLSQP
        
        # First, we create a constrained optimization:
        # minimize max_k integral h(x) * (1 - h(x+k)) dx
        # subject to: 0 <= h_i <= 1, sum(h) = target_sum
        
        # For now, let's iteratively improve with local search
        # But first try a direct CVXPY approach
        
        # The constraint is linear: sum(h) = target_sum
        # But we want to minimize max_k of overlapped integrals
        # This makes it more complex for direct CVXPY
        
        # Instead, use a simpler convex feasibility approach:
        # Goal: minimize C5 where we bound C5 = max_k integral
        # Reformulate as: minimize t such that max_k integral <= t for all k
        # But also add penalty constraints to manage overlaps computationally
        
        # Just do alternate approach - hybrid gradient/convex
        
        # ALTERNATIVE APPROACH: Stochastic gradient descent with constraints
        # Hard constraints will be handled in the optimize
        
        # Define a new approach - relax-to-convex + refined with SLSQP
        
        # Step 1: Use relaxed convex thinking (using matrix visualization) 
        # but remain constrained to avoid over-constraining
        
        # Step 2: Use trust-constr for robust optimization with exact constraints
        
        # Iteratively refine the solution
        max_iterations = 1000
        no_improvement = 0
        best_c5 = evaluate_overlap(h, dx)
        
        # Easy SLSQP refinement (slower but more precise)
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
    
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s - 2:
                break
                
            result = minimize(objective, h, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 50, 'ftol': 1e-8})
            
            if result.success:
                h = np.clip(result.x, 0, 1)
                current_sum = np.sum(h)
                if current_sum > 0:
                    h = h * (target_sum / current_sum)
                
                # Re-evaluate
                current_c5 = evaluate_overlap(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    no_improvement = 0
                else:
                    no_improvement += 1
                    
                if no_improvement > 100:
                    break
            
    except Exception as e:
        #若是失败，回退到原来的GM方法
        pass

    # Final refinement
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)

    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
