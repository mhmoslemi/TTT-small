import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the C5 value with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute full overlap
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each position i, compute how much it contributes to max overlap
    for i in range(n):
        for k in max_indices:
            # k corresponds to a lag in the correlation
            # h[i] contributes to overlap at lag k
            # when h[j] occurs at position j, and at position j+k, it's (1-h[j+k])
            # so contribution is 1-h[j+k] if i = j, else 0
            # More carefully:
            # overlap[k] = sum_{j=0}^{n-1} h[j] * (1 - h[j+k]) 
            # where we have periodicity, so j+k may wrap around
            
            # Only consider when h[i] contributes: i = j, so j = i
            # the term in overlap at lag k is: h[i] * (1 - h[i+k])
            # So we contribute to the gradient when h[i] appears to have an effect
            # For the gradient of overlap[k]:
            #  d/dh[i] [sum_j h[j]*(1-h[j+k])] = 1 - h[i+k] if j=i, 0 otherwise
            # But we're considering the max over all k
            # If overlap[k_0] = max overlap:
            # Gradient of max overlaps = (1 - h[i+k_0]) for k_0 that achieves max
            # And for all other indices j, gradient component is 0, except when
            # we might be computing the gradient of h[j] in other terms
            
            # Simplified: Handle the specific case where the maximum is achieved
            # and assume this represents a reasonable approximation
            if i+k < n:
                grad[i] += (1 - h[i+k]) * dx
            if i-k >= 0:
                grad[i] += (1 - h[i-k]) * dx
    
    # Check the sign of the gradient (should be negative to decrease C5)
    # But since we're minimizing the maximum overlap, we want a direction
    # that reduces the maximum value. Since subgradients can be very conservative,
    # normalize the gradient to prevent large jumps
    if np.linalg.norm(grad) > 0:
        grad = grad / np.linalg.norm(grad)
    
    # Apply constraint gradient for the sum constraint
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Work with small problem size for speed and better local search 
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Optimizer parameters
    learning_rate = 0.01
    momentum = 0.9
    v = np.zeros(n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 500:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum
        v = momentum * v + grad
        h = h - learning_rate * v
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.995
            
    # Final computation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
