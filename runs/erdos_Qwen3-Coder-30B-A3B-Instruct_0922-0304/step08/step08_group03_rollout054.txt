import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n_points = 70 if initial_h_values is available, otherwise 60
    if 'initial_h_values' in globals():
        n_points = 70
        h = np.array(initial_h_values)
    else:
        n_points = 60
        h = triangular_ramp(n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure h is valid
    h = clip_and_renormalize(h, target_sum)
    
    # Initialize momentum
    velocity = np.zeros(n_points)
    alpha = 0.9  # momentum coefficient
    beta = 0.001  # learning rate
    gamma = 0.5  # step size reduction factor
    max_iter = 20000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute exact gradient: proportional to (1 - 2*h) 
        h_gradient = (1 - 2 * h) * np.sign(overlap_max) / n_points
        
        # Momentum update
        velocity = alpha * velocity + beta * h_gradient
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
        else:
            # Reduce step size and retry
            beta *= gamma
            if beta < 1e-8:
                break
                
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
