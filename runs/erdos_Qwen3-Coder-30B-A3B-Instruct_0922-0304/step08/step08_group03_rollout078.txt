import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Gradient-based optimization with momentum
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    velocity = np.zeros(n_points)
    max_iter = 20000
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute the overlap and find max
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute the gradient with respect to maximizing overlap
        # This is based on the idea that ∇C₅ ≈ sign(overlap_max) * (1 - 2h) / n_points
        gradient = (1 - 2 * h) * np.sign(overlap_max) / n_points
        
        # Update with momentum
        velocity = momentum * velocity + learning_rate * gradient
        h_candidate = h + velocity
        
        # Project feasible (clip and renormalize)
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new overlap
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            h = h_candidate.copy()
            velocity = np.zeros(n_points)  # Reset momentum on improvement
        else:
            # Reduce learning rate and retry
            learning_rate *= shrink_factor
            if learning_rate < 1e-8:
                break
            velocity *= shrink_factor
        
        iter_count += 1
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
