import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a structured approach
        h = np.zeros(n_points)
        # Place a peak in the middle
        mid = n_points // 2
        for i in range(max(0, mid - 10), min(n_points, mid + 10)):
            h[i] = 1.0
        h = normalize_h(h, target_sum)
        # Add some noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Parameters
    temp = 1.0
    cooling_rate = 0.999
    perturbation_strength = 0.05
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Simulated Annealing Phase
        candidate_h = h.copy()
        
        # Perturb 10% of the values
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            if np.random.random() < 0.5:
                # Flip with small change
                candidate_h[i] = max(0, min(1, candidate_h[i] * 0.97))
            else:
                # Swap with random position
                j = np.random.randint(0, n_points)
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Ensure constraints
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = normalize_h(candidate_h, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        # Accept with probability based on temperature
        if candidate_c5 < best_c5 or np.random.random() < np.exp(-(candidate_c5 - best_c5) / temp):
            h = candidate_h
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Cool temperature
        temp *= cooling_rate
        
        iteration += 1
        
        # Occasional refinement through a simple cutting-plane approach
        if iteration % 50 == 0:
            # Identify the worst lag
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            max_lag = max_idx - (n_points - 1)  # Convert to lag
            
            # Try to reduce the worst case by adjusting nearby values
            # This is a simple heuristic to refine the solution
            # We will adjust values around the peak of the correlation
            adjustment_magnitude = 0.01
            for i in range(n_points):
                if abs(i - (max_idx - n_points + 1)) < 5:  # Around the peak
                    if i + max_lag < n_points and i + max_lag >= 0:
                        h[i] = max(0, h[i] - adjustment_magnitude)
                        h[i + max_lag] = min(1, h[i + max_lag] + adjustment_magnitude)
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Recompute after adjustment
            current_c5 = compute_c5_exact(h, dx)
            if current_c5 < best_c5:

                best_h = h.copy()
                last_improvement = 0
    
    # Final compute C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
