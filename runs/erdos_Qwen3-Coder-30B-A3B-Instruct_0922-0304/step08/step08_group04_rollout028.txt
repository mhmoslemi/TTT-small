import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # Formulate as convex QP
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints for each shift k
    for k in range(-n_points + 1, n_points):
        # Calculate overlap for shift k: sum h_i * (1 - h_{i+k})
        if k >= 0:
            # h_i * (1 - h_{i+k}) for i = 0 to n_points-1-k
            # where valid indices are i and i+k in [0, n_points-1]
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h_var[i] * (1 - h_var[i + k])
        else:
            # k < 0, h_i * (1 - h_{i+k}) where i+k >= 0, i < n_points
            overlap_sum = 0
            for i in range(-k, n_points):
                overlap_sum += h_var[i] * (1 - h_var[i + k])
                
        constraints.append(overlap_sum <= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers, favoring ECOS
    solver_options = {
        'ecos': {'max_iters': 1000, 'reltol': 1e-6},
        'cvxopt': {'max_iters': 2000, 'reltol': 1e-6},
    }
    
    solver_used = 'ecos'
    try:
        prob.solve(solver=cp.ECOS, **solver_options['ecos'])
        if prob.status != cp.OPTIMAL:
            solver_used = 'cvxopt'
            prob.solve(solver=cp.CVXOPT, **solver_options['cvxopt'])
    except:
        try:
            solver_used = 'cvxopt'
            prob.solve(solver=cp.CVXOPT, **solver_options['cvxopt'])
        except:
            pass
    
    # Get the solution
    if prob.status == cp.OPTIMAL:
        h_result = h_var.value
        c5_bound = t.value
    else:
        # Fallback: use original h and recompute
        h_result = h.copy()
        c5_bound = evaluate_c5(h_result, dx)
    
    # Ensure bounds and normalization
    h_result = np.clip(h_result, 0, 1)
    h_sum = np.sum(h_result)
    if h_sum > 0:
        h_result = h_result * (n_points / 2) / h_sum
    
    # Final evaluation
    final_c5 = evaluate_c5(h_result, dx)
    
    # Return best result found
    return (h_result.tolist(), final_c5, n_points)
