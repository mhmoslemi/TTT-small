import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try convex optimization approach
    try:
        # Create optimization variables
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
            t >= 0
        ]
        
        # For the overlap constraint, we need to add all shifted overlaps
        # Since the overlap is max over k, we impose t >= overlap_k for all k
        # We'll use a simplified constraint: add max constraint for a subset of k values
        for k in range(n_points):
            # Compute shifted correlation for k
            if k == 0:
                overlap_k = cp.sum(cp.multiply(h_var, 1 - h_var))
            elif k < n_points:
                # Pad and compute overlap
                pad_before = np.zeros(k)
                pad_after = np.zeros(n_points - k)
                h_shifted = np.concatenate([pad_before, h_var[:n_points-k], pad_after])
                overlap_k = cp.sum(cp.multiply(h_var, 1 - h_shifted))
            else:
                overlap_k = 0
                
            constraints.append(t >= overlap_k)
            
        # The objective is to minimize t, which is equivalent to minimizing max overlap
        objective = cp.Minimize(t)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            optimized_h = np.array(h_var.value).flatten()
            # Normalize to ensure exact sum
            normalized_h = optimized_h * (target_sum / np.sum(optimized_h))
            normalized_h = np.clip(normalized_h, 0, 1)
            optimized_c5 = evaluate_overlap(normalized_h, dx)
            if optimized_c5 < best_c5:
                best_h = normalized_h
                best_c5 = optimized_c5
    except:
        pass
        
    # If convex optimization doesn't help much, use local search on our best solution
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Try SLSQP with best solution so far
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
        
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
