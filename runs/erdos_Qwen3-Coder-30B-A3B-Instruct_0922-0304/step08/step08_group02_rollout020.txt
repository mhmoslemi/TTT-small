```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def sample_shifts(n, sample_size):
    """Sample shifts for QP constraints"""
    # Sample shifts from -n+1 to n-1 (valid shift range)
    shifts = np.random.choice(range(-n+1, n), size=sample_size, replace=False)
    return shifts

def evaluate_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.ones(n_points) * 0.5
        
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current, dx)
    
    # Adaptive parameters
    max_shifts = 50
    target_overlap = best_c5
    no_improvement_count = 0
    max_no_improvement = 200
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Sample shifts
        shifts = sample_shifts(n_points, max_shifts)
        
        # Create QP problem to minimize maximum overlap
        h = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Objective: minimize t such that overlap <= t for all sampled shifts
        constraints = [
            0 <= h[i] for i in range(n_points)
        ]
        constraints += [
            h[i] <= 1 for i in range(n_points)
        ]
        constraints += [
            cp.sum(h) == target_sum
        ]
        
        # Add overlap constraints for sampled shifts
        for k in shifts:
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in range(n_points-k)
                left = h[:n_points-k]
                right = 1 - h[k:]
                overlap = cp.sum(cp.multiply(left, right))
            else:
                # h[i] * (1 - h[i+k]) for i in range(-k, n_points)
                left = h[-k:]
                right = 1 - h[:n_points+k]
                overlap = cp.sum(cp.multiply(left, right))
            
            constraints.append(overlap <= t)
        
        # Solve the QP
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h.value
                h_new = project_h(h_new, n_points)
            else:
                # Fallback to simple perturbation
                h_new = h_current.copy()
                for i in range(5):
                    idx = np.random.randint(0, n_points)
                    delta = np.random.normal(0, 0.01)
                    h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
                h_new = project_h(h_new, n_points)
        except:
            # Fallback to simple perturbation
            h_new = h_current.copy()
            for i in range(5):
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, 0.01)
                h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
            h_new = project_h(h_new, n_points)
            
        # Check improvement
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
            # Adaptively adjust target for next iteration
            target_overlap = 0.95 * target_overlap + 0.05 * best_c5
        else:
            no_improvement_count += 1
            
        h_current = h_new.copy()
        iteration += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final refinement 
    h_refined = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Simple local perturbations
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = evaluate_c5(h_temp, dx)
        if c5_temp < evaluate_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = evaluate_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
```
