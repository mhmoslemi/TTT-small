import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

        for i in range(mid, n_points):
            h[i] = 1.0 - (i - mid) / mid
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)

        h_trial = np.clip(h_trial, 0, 1)
        h_trial = normalize_h(h_trial, target_sum)

        trial_c5 = evaluate_overlap(h_trial, dx)
