import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using correlation to compute all shifts efficiently
    correlation = np.correlate(h, 1 - h, mode="full")
    # Only consider the relevant part (shifts from -n+1 to n-1)
    n = len(h)
    mid = len(correlation) // 2
    # Extract valid shifts: from -n+1 to n-1, but we only care about positive shifts
    # Actually, we need to extract shifts from 0 to 2*n-2, but take max over all shifts
    # In correlation result, index corresponds to shift k (negative shifts are at start)
    return np.max(correlation) * dx

    target_sum = len(h) / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:

    h = np.clip(h, 0, 1)
    
    # Initial evaluation
    dx = 2.0 / len(h)
    best_c5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    
    # Use scipy.optimize.minimize with SLSQP for better convergence properties
    def obj_func(h_vec):
        return evaluate_overlap(h_vec, dx)
    
    # Define constraints
    def sum_constraint(h_vec):
        return np.sum(h_vec) - target_sum
    
    constraints = [{'type': 'eq', 'fun': sum_constraint}]

    bounds = [(0, 1) for _ in range(len(h))]

    max_iter = 1000
    for _ in range(100):  # Limit number of attempts to keep it fast

        h_perturbed = h + np.random.normal(0, 1e-4, len(h))
