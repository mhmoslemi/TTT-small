import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.1):
    """Apply Gaussian mutation with constraint repair."""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h_mut[mask] += np.random.normal(0, sigma, np.sum(mask))
    h_mut = np.clip(h_mut, 0, 1)
    return normalize_h(h_mut, np.sum(h))

def crossover_individuals(h1, h2):
    """Create offspring via averaging with normalization."""
    h_offspring = (h1 + h2) / 2.0
    return normalize_h(h_offspring, np.sum(h1))

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual via tournament selection."""
    indices = np.random.choice(len(population), tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        population.append(h)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill rest of population with random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Evolutionary algorithm
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate fitness for all individuals
        fitnesses = [compute_c5_exact(h, dx) for h in population]
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection for reproduction
        new_population = [population[min_idx]]  # Keep best
        
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover_individuals(parent1, parent2)
            
            # Mutation
            offspring = mutate_individual(offspring)
            
            new_population.append(offspring)
        
        # Replace population
        population = new_population[:population_size]
    
    # Return the best solution found
    if best_h is None:
        # Fallback: return the best of last population
        fitnesses = [compute_c5_exact(h, dx) for h in population]
        min_idx = np.argmin(fitnesses)
        best_h = population[min_idx]
        best_c5 = fitnesses[min_idx]
    
    return (best_h.tolist(), best_c5, n_points)
