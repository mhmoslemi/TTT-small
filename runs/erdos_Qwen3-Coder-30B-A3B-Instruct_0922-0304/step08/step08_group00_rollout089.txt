import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Perturbation-based alternating optimization
    max_iter = 100
    perturbation_delta = 1e-3
    improvement_threshold = 1e-5
    perturbation_decay = 0.98
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 10:
        # Create a slightly perturbed version to explore
        h_perturbed = h + np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed_sum = np.sum(h_perturbed)
        if h_perturbed_sum > 0:
            h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)
        
        # Partition indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Optimize even indices, fixing odd
        def objective_even(h_even):
            h_full = best_h.copy()
            h_full[even_indices] = h_even
            return evaluate_overlap(h_full, dx)
        
        bounds_even = [(0, 1) for _ in even_indices]
        # By fixing O, the constraint on odd elements is hard-coded
        # We must ensure overall sum constraint: sum(E) + sum(O) = n/2
        # This is automatically handled by the update below
        
        # Optimization step for even indices
        h_even_init = best_h[even_indices]
        try:
            result = minimize(objective_even, h_even_init, 
                            method='SLSQP', bounds=bounds_even, 
                            options={'ftol': 1e-10, 'maxiter': 100})
            if result.success:
                h_even_final = np.clip(result.x, 0, 1)
                h_candidate = best_h.copy()
                h_candidate[even_indices] = h_even_final
                sum_odd = np.sum(best_h[odd_indices])
                sum_even = np.sum(h_even_final)
                scale_factor = (target_sum - sum_odd) / sum_even
                if scale_factor > 0:
                    h_candidate[even_indices] *= scale_factor
                c5_candidate = evaluate_overlap(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
        
        # Optimize odd indices, fixing even
        def objective_odd(h_odd):
            h_full = best_h.copy()
            h_full[odd_indices] = h_odd
            return evaluate_overlap(h_full, dx)
        
        bounds_odd = [(0, 1) for _ in odd_indices]
        
        # Optimization step for odd indices
        h_odd_init = best_h[odd_indices]
        try:
            result = minimize(objective_odd, h_odd_init, 
                            method='SLSQP', bounds=bounds_odd, 
                            options={'ftol': 1e-10, 'maxiter': 100})
            if result.success:
                h_odd_final = np.clip(result.x, 0, 1)
                h_candidate = best_h.copy()
                h_candidate[odd_indices] = h_odd_final
                sum_even = np.sum(best_h[even_indices])
                sum_odd = np.sum(h_odd_final)
                scale_factor = (target_sum - sum_even) / sum_odd
                if scale_factor > 0:
                    h_candidate[odd_indices] *= scale_factor
                c5_candidate = evaluate_overlap(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
        
        # Periodic perturbation to escape local optima
        if iteration % 5 == 0:
            perturbation_delta *= perturbation_decay
            h_perturbed = best_h + np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum > 0:
                h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)
            best_h = h_perturbed.copy()
            best_c5 = evaluate_overlap(best_h, dx)
            
        iteration += 1
    
    # Final refinement
    def final_objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(final_objective, best_h, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
