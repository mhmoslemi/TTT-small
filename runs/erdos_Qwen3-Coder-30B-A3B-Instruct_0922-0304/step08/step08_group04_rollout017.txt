import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():

    else:

        h = normalize_h(h, n_points / 2)
    
    # Optimization parameters
    max_iter = 50000
    tolerance = 1e-6
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    prev_c5 = best_c5
    no_improvement_count = 0
    
    # Start with a slightly different initialization strategy
    if 'initial_h_values' not in globals():
        # Try a piecewise constant approach with 3 segments
        segments = 3
        segment_size = n_points // segments
        h = np.zeros(n_points)
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < segments - 1 else n_points
            h[start_idx:end_idx] = 0.5 + 0.3 * np.random.random()  # Vary the values
        h = normalize_h(h, n_points / 2)
    
    # Local search with perturbations
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        trial_h = h.copy()
        
        # Random perturbation
        idx = np.random.randint(0, n_points)
        perturbation = (np.random.random() - 0.5) * 0.1
        trial_h[idx] = np.clip(trial_h[idx] + perturbation, 0, 1)

        trial_h = normalize_h(trial_h, n_points / 2)

            prev_c5 = trial_c5
            
        # Early stopping
        if best_c5 <= 0.3807:
            break

            # Perturb more aggressively
            h = np.clip(h + (np.random.random(n_points) - 0.5) * 0.05, 0, 1)
            h = normalize_h(h, n_points / 2)
            no_improvement_count = 0
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
