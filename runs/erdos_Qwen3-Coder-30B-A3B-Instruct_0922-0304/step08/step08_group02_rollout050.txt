import numpy as np
from scipy.signal import correlate
import time
from math import exp
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def perturb_h(h, magnitude):
    """Create a perturbed version of h."""
    h_new = h.copy()
    idx = np.random.randint(0, len(h))
    delta = np.random.uniform(-magnitude, magnitude)
    h_new[idx] += delta
    h_new = np.clip(h_new, 0, 1)
    return h_new

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def genetic_algorithm_step(population, fitnesses, n_points, target_sum, dx, 
                          mutation_rate=0.05, crossover_prob=0.8, elitism_rate=0.2):
    """Perform one step of genetic algorithm."""
    # Sort population by fitness
    sorted_indices = np.argsort(fitnesses)
    population = [population[i] for i in sorted_indices]
    fitnesses = [fitnesses[i] for i in sorted_indices]
    
    # Elitism
    n_elite = int(elitism_rate * len(population))
    new_population = population[:n_elite]
    
    # Generate offspring
    while len(new_population) < len(population):
        if np.random.rand() < crossover_prob:
            # Tournament selection
            parent1_idx = np.random.randint(0, n_elite)
            parent2_idx = np.random.randint(0, n_elite)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Uniform crossover
            child1 = parent1.copy()
            child2 = parent2.copy()
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child1[i], child2[i] = child2[i], child1[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.clip(child1[i] + np.random.normal(0, 0.05), 0, 1)
                if np.random.rand() < mutation_rate:
                    child2[i] = np.clip(child2[i] + np.random.normal(0, 0.05), 0, 1)
            
            # Repair
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < len(population):
                new_population.append(child2)
        else:
            # Random mutation of elite
            parent_idx = np.random.randint(0, n_elite)
            parent = population[parent_idx]
            child = perturb_h(parent, 0.1)
            child = repair_h(child, target_sum)
            new_population.append(child)
    
    return new_population

def local_refinement(h, dx, steps=50):
    """Apply local refinement to improve solution."""
    h_refined = h.copy()
    best_h = h_refined.copy()
    best_c5 = compute_c5(h_refined, dx)
    
    for _ in range(steps):
        h_new = h_refined.copy()
        # Focus on areas likely to contribute to high overlap
        overlap = correlate(h_refined, 1 - h_refined, mode="full")
        max_pos = np.argmax(overlap)
        # Perturb around the max overlap position
        for _ in range(5):
            idx = np.random.randint(max(0, max_pos - 10), min(len(h_refined), max_pos + 10))
            delta = np.random.normal(0, 0.01)
            h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        h_new = repair_h(h_new, len(h_refined) / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h_refined = h_new.copy()
    
    return best_h, best_c5

    n_pop = 20

        population.append(h_current)
    
    # Add random individuals
    for _ in range(n_pop - len(population)):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(population[0], dx)
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitnesses = [compute_c5(h, dx) for h in population]
        
        # Check for improvement
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitnesses.index(current_best)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Local refinement of top individuals
        top_indices = np.argsort(fitnesses)[:3]
        for idx in top_indices:
            h_refined, c5_refined = local_refinement(population[idx], dx, 50)
            if c5_refined < fitnesses[idx]:
                population[idx] = h_refined
                fitnesses[idx] = c5_refined
        
        # Genetic algorithm step
        population = genetic_algorithm_step(population, fitnesses, n_points, target_sum, dx,
