import numpy as np
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.02
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Random perturbation for local search
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.02)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Renormalize
        h_new = project_h(h_new, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Take a small step in a random direction
        if np.random.random() < 0.7:
            h_new = h_current.copy()
            for _ in range(3):  # Multiple small steps
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, learning_rate)
                h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                else:
                    h_new = h_current.copy()
        else:
            h_new = best_h.copy()
            # Random small steps
            for _ in range(5):
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, learning_rate * 0.5)
                h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                else:
                    h_new = best_h.copy()
        
        h_current = h_new.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with gradient-like search
    h_refined = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
