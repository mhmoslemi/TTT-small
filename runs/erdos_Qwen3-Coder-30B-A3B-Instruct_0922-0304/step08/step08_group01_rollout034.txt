import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach with rounding heuristic
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Stage 1: Convex relaxation
        # To enhance performance, we use a direct coordinate descent method with adaptive steps
        
        # Sample a small subset of points for adjustment
        update_count = max(1, n_points // 20)
        selected_idxs = np.random.choice(n_points, update_count, replace=False)
        
        for i in selected_idxs:
            old_h_i = h[i]
            # Simple hill climbing with local exploration
            best_step = 0
            best_h_i = old_h_i
            best_val = best_c5
            
            # Test several small perturbations
            steps = np.linspace(-0.1, 0.1, 11)
            for step in steps:
                new_h_i = old_h_i + step
                new_h_i = np.clip(new_h_i, 0, 1)
                
                # Update h temporarily
                h[i] = new_h_i
                h = normalize_h(h, target_sum)
                
                # Compute new C5
                new_c5 = compute_c5(h, dx)
                
                if new_c5 < best_val:
                    best_val = new_c5
                    best_h_i = new_h_i
                    best_step = step
        
        # Apply the best update
        h[i] = best_h_i
        h = normalize_h(h, target_sum)
        
        # Keep the best solution found so far
        if best_val < best_c5:
            best_c5 = best_val
            best_h = h.copy()
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    # Recheck the actual maximum overlap to make sure it's properly computed
    overlap = correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(overlap)
    
    return (best_h.tolist(), final_c5, n_points)
