import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Projected gradient descent with Nesterov momentum
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tolerance = 1e-6
    
    # Initialize momentum variables
    v = np.zeros_like(h)
    h_prev = h.copy()
    
    # Evaluate initial overlap
    current_c5 = evaluate_overlap(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient direction
        # We approximate gradient by computing the gradient vector for
        # current h and then compute the max gradient direction
        # Since gradient of C_k w.r.t h_i is (1 - h_{i+k}) with k for max C_k
        
        # Compute all overlaps to find maximal one
        overlap_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_full)
        # max_idx represents the shift index where maximum overlap occurs
        if max_idx < n_points:
            k = max_idx
        else:
            k = max_idx - n_points
        
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k) % n_points
            gradient[i] = 1 - h[j]
        
        # Nesterov momentum update
        v = momentum * v - step_size * gradient
        h_new = h_prev + v
        
        # Project to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate new overlap
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Adaptive step size
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            step_size *= 0.9
            if step_size < min_step_size:
                step_size = min_step_size
        
        # Update for next iteration
        h_prev = h.copy()
        
        # Check for convergence
        if abs(current_c5 - best_c5) < tolerance:
            break
    
    # Final refinement with local optimization
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    from scipy.optimize import minimize
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Ensure final solution is valid
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
