import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.1):
    """Mutate an individual with Gaussian noise"""
    h_new = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    h_new[indices] += np.random.normal(0, sigma, len(indices))
    h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover_individuals(h1, h2):
    """Create offspring from two parents by averaging"""
    offspring = (h1 + h2) / 2.0
    return offspring

def evaluate_population(population, dx):
    """Evaluate all individuals in the population"""
    fitness = []
    for h in population:
        fit = compute_c5_exact(h, dx)
        fitness.append(fit)
    return np.array(fitness)

def selection_tournament(population, fitness, tournament_size=5):
    """Perform tournament selection"""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    
    # Initialize population
    population = []
    
    # Add initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Add random individuals to complete population
    while len(population) < population_size:
        h = np.random.rand(n_points)
        # Add some Gaussian noise to smooth diversity
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Tracking
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate current population
        fitness = evaluate_population(population, dx)
        
        # Track best individual
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected_population = selection_tournament(population, fitness)
        
        # Crossover and mutation to create new population
        new_population = [best_h.copy()]  # Keep best individual (elitism)
        
        while len(new_population) < population_size:
            # Choose two parents
            parent1_idx = np.random.randint(0, len(selected_population))
            parent2_idx = np.random.randint(0, len(selected_population))
            
            # Crossover
            offspring = crossover_individuals(selected_population[parent1_idx], 
                                               selected_population[parent2_idx])
            
            # Mutation
            offspring = mutate_individual(offspring, mutation_rate=0.3, sigma=0.05)
            
            # Ensure constraints
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
        
        population = new_population
        generation += 1
    
    if best_h is None:
        # Final evaluation of last population
        fitness = evaluate_population(population, dx)
        min_fitness_idx = np.argmin(fitness)
        best_c5 = fitness[min_fitness_idx]
        best_h = population[min_fitness_idx]
        
    return (best_h.tolist(), best_c5, n_points)
