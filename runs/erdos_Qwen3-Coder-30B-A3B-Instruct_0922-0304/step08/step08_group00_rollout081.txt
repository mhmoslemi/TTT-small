import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0.0
    # For each shift k from 0 to n-1
    for k in range(n):
        overlap = 0.0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Define the optimization problem
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Inequality constraints: each constraint is h_i <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Equality constraint: sum(h) = target_sum
    def sum_constraint(h_vector):
        return np.sum(h_vector) - target_sum
    
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Use SLSQP solver with bounds and constraints
    result = minimize(objective, h, method='SLSQP', 
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12})
    
    # If optimization fails or is not better, try to perturb the initial solution
    if not result.success or result.fun >= best_c5:
        # Try to make a small random adjustment to the initial solution
        perturbed_h = h + np.random.normal(0, 1e-4, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        current_sum = np.sum(perturbed_h)
        if current_sum > 0:
            perturbed_h = perturbed_h * (target_sum / current_sum)
        perturbed_c5 = evaluate_overlap(perturbed_h, dx)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = perturbed_c5
            
        # Also try another random initialization
        random_h = np.random.uniform(0, 1, n_points)
        random_h = np.clip(random_h, 0, 1)
        current_sum = np.sum(random_h)
        if current_sum > 0:
            random_h = random_h * (target_sum / current_sum)
        random_c5 = evaluate_overlap(random_h, dx)
        if random_c5 < best_c5:
            best_h = random_h.copy()
            best_c5 = random_c5
            
    else:
        # Use the result from optimization
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)

    for _ in range(5):

        perturbed_h = final_h + np.random.normal(0, 1e-4, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        current_sum = np.sum(perturbed_h)
        if current_sum > 0:
            perturbed_h = perturbed_h * (target_sum / current_sum)
        perturbed_c5 = evaluate_overlap(perturbed_h, dx)
        
        if perturbed_c5 < final_c5:
            final_h = perturbed_h
            final_c5 = perturbed_c5
    
    # Make sure final solution is valid
    final_h = np.clip(final_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
