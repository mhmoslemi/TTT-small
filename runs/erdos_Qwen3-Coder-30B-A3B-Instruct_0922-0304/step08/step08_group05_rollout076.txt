import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, sigma=0.1):
    """Apply Gaussian mutation to an individual."""
    h_mut = h + np.random.normal(0, sigma, len(h))
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover_parents(p1, p2):
    """Create offspring from two parents by averaging."""
    offspring = (p1 + p2) / 2.0
    return offspring

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population size
    pop_size = 50
    mutation_sigma = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.random.rand(n_points) * 0.5
    
    h_start = normalize_h(h_start, target_sum)
    population.append(h_start)
    
    # Fill the rest of the population with random individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Main evolution loop
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate fitness for all individuals
        fitnesses = [compute_c5_exact(h, dx) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[min_fitness_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[min_fitness_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = [population[min_fitness_idx]]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover_parents(parent1, parent2)
            
            # Mutation
            offspring = mutate_individual(offspring, mutation_sigma)
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
            
        population = new_population
        generation += 1
        
        # Adapt mutation rate
        if generation % 10 == 0:
            mutation_sigma *= 0.99
        
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5_exact(best_h, dx)
    else:
        # Fallback to best individual from last generation if needed
        last_fitnesses = [compute_c5_exact(h, dx) for h in population]
        min_fitness_idx = np.argmin(last_fitnesses)
        best_h = population[min_fitness_idx]
        final_c5 = last_fitnesses[min_fitness_idx]
        
    return (best_h.tolist(), final_c5, n_points)
