import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    # For large n, approximate gradient as (1 - 2h) / n_points
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h with either initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # If already good enough, return early
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic Algorithm parameters
    pop_size = 50
    max_generations = 2000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_size = 5
    
    # Population generation
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
        individual = repair_h(individual, target_sum)
        population.append(individual)
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    for generation in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluation
        fitness = []
        for individual in population:
            c5 = evaluate_h(individual, dx)
            fitness.append(c5)
        
        # Check for best solution
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = population[:elite_size]
        
        # Crossover and Mutation
        while len(new_pop) < pop_size:
            # Selection (tournament)
            parent1 = population[np.random.randint(0, pop_size//2)]
            parent2 = population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.clip(child1[i] + np.random.normal(0, 0.05), 0, 1)
                if np.random.rand() < mutation_rate:
                    child2[i] = np.clip(child2[i] + np.random.normal(0, 0.05), 0, 1)
            
            # Repair
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            # Gradient descent refinement
            child1_refined = child1.copy()
            grad1 = compute_gradient(child1_refined)
            step_size = 0.01
            child1_refined -= step_size * grad1
            child1_refined = repair_h(child1_refined, target_sum)
            
            child2_refined = child2.copy()
            grad2 = compute_gradient(child2_refined)
            child2_refined -= step_size * grad2
            child2_refined = repair_h(child2_refined, target_sum)
            
            new_pop.append(child1_refined)
            if len(new_pop) < pop_size:
                new_pop.append(child2_refined)
        
        population = new_pop
        
        # Adaptive parameters
        no_improvement_count += 1
        if len(population) > 0:
            new_fitness = [evaluate_h(ind, dx) for ind in population]
            new_min = min(new_fitness)
            if new_min < best_c5:
                no_improvement_count = 0
                
        if no_improvement_count > 100:
            mutation_rate = min(0.5, mutation_rate * 1.1)
            crossover_prob = max(0.3, crossover_prob * 0.95)
        elif no_improvement_count > 50:
            mutation_rate = min(0.3, mutation_rate * 1.05)
        
        if generation % 50 == 0:
            print(f"Gen {generation}: Best C5 = {best_c5:.6f}")
    
    # Final refinement of best solution
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
