```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Start with a structured pattern
        h = np.zeros(n_points)
        # Create a step-like function
        start = n_points // 4
        end = 3 * n_points // 4
        h[start:end] = 1.0
        h = normalize_h(h, target_sum)
        # Add slight noise to break symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = compute_c5_exact(h, dx)
    best_h = h.copy()
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    last_improvement = 0
    max_no_improvement = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < max_no_improvement:
        # Generate candidate using perturbation
        candidate = h.copy()
        
        # Perturb 10% of positions
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Randomly flip or swap
            if np.random.random() < 0.5 and candidate[i] > 0:
                # Flip
                candidate[i] *= 0.95
            else:
                # Swap with another position
                j = np.random.choice(n_points)
                candidate[i], candidate[j] = candidate[j], candidate[i]
        
        # Apply constraints
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5_exact(candidate, dx)
        
        # Accept or reject
        delta = candidate_c5 - best_c5
        if delta < 0 or np.exp(-delta / temperature) > np.random.random():
            h = candidate
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Cool temperature
        temperature *= cooling_rate
        if temperature < min_temp:
            temperature = min_temp
            
        # Occasionally apply a refined update based on overlap gradient
        if iteration % 50 == 0 and iteration > 0:
            overlap = correlate(h, 1 - h, mode="full")
            max_lag_idx = np.argmax(overlap)
            
            # Perturb around areas contributing to maximum overlap
            for i in range(max(0, max_lag_idx - 5), min(len(h), max_lag_idx + 5)):
                if np.random.random() < 0.1 and h[i] > 0:
                    h[i] = max(0, h[i] - 0.01)
                    h[i] = min(1, h[i] + 0.01)
            
            h = normalize_h(h, target_sum)
            new_c5 = compute_c5_exact(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improvement = 0
                
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final normalization
    best_h = normalize_h(best_h, target_sum)
    
    return (best_h.tolist(), best_c5, n_points)
```
