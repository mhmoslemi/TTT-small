import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, noise_magnitude):
    """Add noise to an individual."""
    noise = np.random.normal(0, noise_magnitude, len(h))
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, len(h) / 2)
    return h_new

def crossover_individuals(h1, h2, crossover_rate=0.8):
    """Perform crossover between two individuals."""
    if random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = random.randint(1, len(h1) - 1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    h1_new = normalize_h(h1_new, len(h1) / 2)
    h2_new = normalize_h(h2_new, len(h1) / 2)
    
    return h1_new, h2_new

def fitness(individual):
    """Fitness function - lower is better."""
    return evaluate_c5(individual, dx)

def genetic_algorithm(seed, n_points, max_generations, pop_size, crossover_rate, mutation_rate, noise_magnitude):
    """Genetic algorithm for optimizing the Erdős minimum overlap problem."""
    np.random.seed(seed)
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        population = [parent.copy()]
        for _ in range(pop_size - 1):
            individual = mutate_individual(parent, noise_magnitude)
            population.append(individual)
    else:
        population = []
        for _ in range(pop_size):
            individual = np.random.rand(n_points).astype(np.float64)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    best_individual = None
    best_fitness = float('inf')
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitness_values = Parallel(n_jobs=1)(delayed(fitness)(ind) for ind in population)
        
        # Track best individual
        for i, fit in enumerate(fitness_values):
            if fit < best_fitness:
                best_fitness = fit
                best_individual = population[i].copy()
                
        if best_fitness <= 0.3807:
            break
            
        # Selection - tournament
        new_population = []
        for _ in range(pop_size // 2):
            # Tournament selection
            tourn_size = 2
            idxs = np.random.choice(len(population), tourn_size)
            best_idx = idxs[np.argmin([fitness_values[i] for i in idxs])]
            new_population.append(population[best_idx].copy())
            
            # Crossover
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            h1, h2 = crossover_individuals(population[idx1], population[idx2], crossover_rate)
            new_population.append(h1)
            new_population.append(h2)
            
        # Mutation
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                new_population[i] = mutate_individual(new_population[i], noise_magnitude)
                
        population = new_population
        
        # Reduce noise magnitude every 10 generations
        if generation % 10 == 0 and generation > 0:
            noise_magnitude *= 0.99
            
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Set GA parameters
    max_generations = 1000
    pop_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.3
    noise_magnitude = 0.05
    
    # Run GA
    best_individual, best_fitness = genetic_algorithm(
        seed, n_points, max_generations, pop_size, 
        crossover_rate, mutation_rate, noise_magnitude
    )
    
    # Final evaluation to ensure accuracy
    final_fitness = evaluate_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
