import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Generate a random feasible solution
        h_current = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        h_current[ones_indices] = 1.0
        # Add random perturbation
        perturb_indices = np.random.choice(n_points, size=int(0.1*n_points), replace=False)
        for idx in perturb_indices:
            h_current[idx] += np.random.normal(0, 0.1)
        # Clamp and renormalize
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    
    # Original approach but more methodical
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    t = 0
    T0 = 0.1
    T_min = 1e-6
    alpha = 0.999
    sigma = 0.05
    n_steps_no_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        
        # Temperature schedule
        T = max(T_min, T0 * (alpha ** (t / 1000)))
        
        # Proposal: perturb one element
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        
        # Generate perturbation
        delta = np.random.normal(0, sigma)
        h_new[idx] += delta
        
        # Clamp
        h_new[idx] = max(0, min(1, h_new[idx]))
        
        # Renormalize to meet constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute C5
        c5_new = compute_c5(h_new, dx)
        c5_current = compute_c5(h_current, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        
        if delta_c5 < 0 or np.random.rand() < exp(-delta_c5 / T):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                n_steps_no_improvement = 0
            else:
                n_steps_no_improvement += 1
        else:
            n_steps_no_improvement += 1
            
        # Adaptive cooling
        if n_steps_no_improvement > max_no_improvement:
            sigma *= 0.9
            n_steps_no_improvement = 0
            
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
