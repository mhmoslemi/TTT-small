import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function on [0, 2] with n points."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_coefficients(h, basis_funcs):
    """Compute Fourier coefficients of h."""
    n = len(h)
    M = len(basis_funcs)
    coeffs = np.zeros(M)
    for i in range(M):
        coeffs[i] = np.dot(h, basis_funcs[i]) / n
    return coeffs

def backtransform(coeffs, basis_funcs):
    """Transform coefficients back to h."""
    h = np.zeros(len(basis_funcs[0]))
    for i, c in enumerate(coeffs):
        h += c * basis_funcs[i]
    return h

def project(h, target_sum):
    """Project h onto feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute C5 for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = project(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Create the basis functions
    M = max(1, n_points // 4)
    basis_funcs = []
    x = np.linspace(0, 2, n_points)
    for i in range(1, M + 1):
        # Windowed sine functions
        basis = np.sin(i * np.pi * x / 2) * (1 - np.abs(x - 1))
        basis_funcs.append(basis)
        
    # Initial fourier coefficients
    coeffs = compute_coefficients(h, basis_funcs)
    
    # Gradient descent in Fourier space
    max_iter = 10000
    learning_rate = 0.001
    momentum = 0.9
    velocity = np.zeros_like(coeffs)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
        
        h = backtransform(coeffs, basis_funcs)
        h = project(h, target_sum)
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient with respect to coefficients
        # For simplicity, we approximate gradient with finite difference
        eps = 1e-5
        grad = np.zeros_like(coeffs)
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = backtransform(coeffs_plus, basis_funcs)
            h_plus = project(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = backtransform(coeffs_minus, basis_funcs)
            h_minus = project(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update coefficients using momentum
        velocity = momentum * velocity - learning_rate * grad
        coeffs += velocity
    
    # Final check
    final_h = backtransform(coeffs, basis_funcs)
    final_h = project(final_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
