import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Simple initialization
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    sigma = 0.05
    anneal_steps = 1000
    no_improve_steps = 500
    no_improve_count = 0
    step = 0
    
    T = T0
    
    while time.time() - start_time < budget_s - 1:
        # Current state
        c5_current = compute_c5(h_current, dx)
        
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, sigma)
        h_new[idx] += delta
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to satisfy sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute new state C5
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < exp(-delta_c5 / max(T, 1e-10)):
            h_current = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Update temperature
        step += 1
        T = max(T_min, T0 * (alpha ** (step / anneal_steps)))
        
        # Adaptive perturbation scaling
        if no_improve_count > no_improve_steps:
            sigma *= 0.9
            no_improve_count = 0
            if sigma < 1e-4:
                sigma = 1e-4
                
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
