import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of C5 with respect to h."""
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # The gradient is derived from the maximum overlap term
    grad = np.zeros_like(h)
    for i in range(n_points):
        # For the lag that achieves maximum overlap
        # For each position i, the contribution to gradient at i
        # depends on where h[i] is involved in the overlap with (1-h)
        # The maximum overlap occurs over indices that sum to same value.
        # For index k at max overlap, deriv is -2 * h[i] for those k values.
        # Easier to compute via explicit gradient.
        grad[i] = 0
        # Manually compute exact gradient for max overlap case
        # We compute: 
        # d/dh[i] max_k [sum_j h[j] * (1 - h[j+k])] 
        # Involves the k that achieves maximum overlap
        max_k = max_idx - (n_points - 1)  # corresponds to k in full correlation
        for j in range(n_points):
            if 0 <= j + max_k < n_points:
                grad[i] += (1 - 2 * h[i]) * (1 - h[j + max_k])
    return grad * dx

def solve_linear_subproblem(grad, n_points, target_sum):
    """Solve min_v grad^T v s.t. sum(v) = target_sum, 0 <= v <= 1 using LP."""
    # minimize c^T * v, subject to
    # A_eq * v = b_eq
    # A_ub * v <= b_ub
    # lb <= v <= ub
    
    # Set up the linear program
    c = grad
    A_eq = np.array([np.ones(n_points)])
    b_eq = np.array([target_sum])
    A_ub = np.vstack((np.eye(n_points), -np.eye(n_points)))
    b_ub = np.hstack((np.ones(n_points), np.zeros(n_points)))
    bounds = [(0, 1) for _ in range(n_points)]

    # Solve using scipy linprog
    result = linprog(c, A_eq=A_eq, b_eq=b_eq, A_ub=A_ub, b_ub=b_ub, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback
        v = np.zeros(n_points)
        v[np.argmin(grad)] = target_sum
        return v

def line_search(h, v, grad, dx, n_points, max_trials=20):
    """Line search for step size η that reduces C5."""
    # Try to find an η such that C5 of (h + η*(v-h)) improves
    # We start with relatively large step and shrink
    def compute_c5_new(eta):
        h_new = h + eta * (v - h)
        return evaluate_c5(h_new, dx)
    
    eta = 1.0
    current_c5 = evaluate_c5(h, dx)
    c5_new = compute_c5_new(eta)
    
    tries = 0
    while c5_new >= current_c5 and tries < max_trials:
        eta *= 0.5
        c5_new = compute_c5_new(eta)
        tries += 1
    
    if tries >= max_trials:
        # Default case
        return 0.1
    else:
        return eta

def frank_wolfe_better(h, dx, n_points, max_iter=10000):
    """Frank-Wolfe for optimizing C5."""
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    prev_c5 = float('inf')
    
    for t in range(max_iter):
        if t % 100 == 0:
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if current_c5 <= 0.3807:
                break
            if abs(prev_c5 - current_c5) < 1e-8:
                break
            prev_c5 = current_c5
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Solve linear subproblem
        v = solve_linear_subproblem(grad, n_points, target_sum)
        
        # Line search
        eta = line_search(h, v, grad, dx, n_points)
        
        # Update
        h = (1 - eta) * h + eta * v
        
        # Project
        h = project_h(h, target_sum)
        
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points / 2)
    
    # Optimize the function using Frank-Wolfe
    best_h, best_c5 = frank_wolfe_better(h, dx, n_points, max_iter=10000)
    
    # Make sure we have the exact best solution within the recorded time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        # Do one final check with exact computation
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
