```python
import numpy as np
from scipy.signal import correlate
import time

def repair_h(h, target_sum):
    """Ensure h satisfies constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # GA parameters
    population_size = 50
    max_generations = 2000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_count = 5
    
    population = [h.copy()]
    for _ in range(population_size - 1):
        candidate = np.random.rand(n_points)
        candidate = repair_h(candidate, target_sum)
        population.append(candidate)
    
    last_improvement = 0
    improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append((c5, individual))
            
        fitness.sort(key=lambda x: x[0])
        best_in_generation = fitness[0][0]
        
        if best_in_generation < best_c5:
            best_c5 = best_in_generation
            best_h = fitness[0][1].copy()
            improvement_count += 1
            last_improvement = generation
            
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = [ind for _, ind in fitness[:population_size//2]]
        
        # Crossover and mutation
        new_population = selected[:elite_count]
        
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                child1 = repair_h(child1, target_sum)
                child2 = repair_h(child2, target_sum)
                
                # Gradient-based refinement
                if np.random.rand() < 0.8:
                    child1 = refine_with_gradient(child1, dx, target_sum)
                if np.random.rand() < 0.8:
                    child2 = refine_with_gradient(child2, dx, target_sum)
                    
                if len(new_population) < population_size:
                    new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Mutation
                parent = np.random.choice(selected)
                child = parent.copy()
                mutate_indices = np.random.choice(n_points, 
                                                 size=int(n_points * mutation_rate), 
                                                 replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.05)
                child = repair_h(child, target_sum)
                if np.random.rand() < 0.5:
                    child = refine_with_gradient(child, dx, target_sum)
                new_population.append(child)
                
        population = new_population
        
        # Adapt mutation rate
        if generation - last_improvement > 100:
            mutation_rate = min(0.3, mutation_rate * 1.1)
        else:
            mutation_rate = max(0.01, mutation_rate * 0.99)
            
    return (best_h.tolist(), best_c5, n_points)

def refine_with_gradient(h, dx, target_sum):
    """Perform a small gradient step."""
    # This is an approximation using the sensitivity of overlap to h
    # We find all maxima in the correlation with 1-h
    h_copy = h.copy()
    overlap_max = np.max(correlate(h_copy, 1 - h_copy, mode="full")) * dx
    
    # Compute approximate gradient for each point
    # In practice, we make tiny steps in the direction of decreasing overlap
    step_size = 0.001
    best_h = h_copy.copy()
    best_c5 = overlap_max
    
    for i in range(len(h_copy)):
        # Try slightly increasing h[i]
        h_plus = h_copy.copy()
        h_plus[i] = min(1.0, h_plus[i] + step_size)
        h_plus = repair_h(h_plus, target_sum)
        c5_plus = compute_c5(h_plus, dx)
        
        # Try slightly decreasing h[i]
        h_minus = h_copy.copy()
        h_minus[i] = max(0.0, h_minus[i] - step_size)
        h_minus = repair_h(h_minus, target_sum)
        c5_minus = compute_c5(h_minus, dx)
        
        # Choose the better step if it reduces C5
        if c5_plus < best_c5 and c5_plus < c5_minus:
            h_copy = h_plus.copy()
            best_c5 = c5_plus
        elif c5_minus < best_c5:
            h_copy = h_minus.copy()
            best_c5 = c5_minus
            
    return h_copy
```
