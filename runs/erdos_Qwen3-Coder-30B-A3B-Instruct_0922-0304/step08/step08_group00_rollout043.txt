import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Projected gradient descent with Nesterov momentum
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    nesterov_momentum = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current overlap
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Compute gradient for each point
        # Using overlap integral for each lag to get the index of max overlap
        overlap_full = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap_full)
        max_shift = max_shift_idx - (n_points - 1)  # Convert to actual shift
        if max_shift < 0:
            max_shift += n_points
        
        # Direct computation of gradient components
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = (i - max_shift) % n_points
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]
        
        # Apply Nesterov momentum
        nesterov_momentum = momentum * nesterov_momentum + gradient
        # Compute Nesterov accelerated update
        h_nesterov = h - step_size * nesterov_momentum
        
        # Project to feasible set [0,1] and rescale
        h_nesterov = np.clip(h_nesterov, 0, 1)
        current_sum = np.sum(h_nesterov)
        if current_sum > 0:
            h_nesterov = h_nesterov * (target_sum / current_sum)
        
        # Update h
        h = h_nesterov
        
        # Adjust learning rate
        if abs(current_c5 - best_c5) < tol:
            step_size *= 0.9
        if step_size < min_step_size:
            step_size = min_step_size
    
    # Final validation and local refinement
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
