import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed, target_sum):
    """Convert relaxed solution to binary vector with exact sum constraint."""
    # Round to binary using median threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet exact sum constraint
    current_sum = np.sum(h_binary)
    diff = current_sum - target_sum
    
    if diff > 0:
        # Need to flip diff 1's to 0's
        # Flip the smallest values that are >= threshold
        indices = np.where((h_binary == 1) & (h_relaxed >= threshold))[0]
        flip_indices = np.argsort(h_relaxed[indices])[:int(diff)]
        flip_indices = indices[flip_indices]
        h_binary[flip_indices] = 0
    elif diff < 0:
        # Need to flip |diff| 0's to 1's
        # Flip the largest values that are < threshold
        indices = np.where((h_binary == 0) & (h_relaxed < threshold))[0]
        flip_indices = np.argsort(h_relaxed[indices])[-int(-diff):]  # last (-diff) elements
        flip_indices = indices[flip_indices]
        h_binary[flip_indices] = 1
        
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, target_sum)
    
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Construct optimization problem using CVXPY
    iterations = 0
    max_iterations = 10
    solver = 'MOSEK'
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        # Set up the relaxed problem
        h_relaxed = cp.Variable(n_points, name="h")
        z = cp.Variable(name="z")
        
        constraints = []
        
        # Add sum constraint
        constraints.append(cp.sum(h_relaxed) == target_sum)
        
        # Add bounds
        constraints.append(h_relaxed >= 0)
        constraints.append(h_relaxed <= 1)
        
        # Add overlap constraints
        for k in range(-n_points+1, n_points):
            if k >= 0:
                expr = cp.sum(cp.multiply(h_relaxed[:n_points-k], 1 - h_relaxed[k:]))
            else:
                expr = cp.sum(cp.multiply(h_relaxed[-k:], 1 - h_relaxed[:n_points+k]))
            constraints.append(expr <= z)
        
        # Define and solve the problem
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False, max_iters=100)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_solution = h_relaxed.value
                h_binary = round_to_binary(h_solution, target_sum)
                
                c5_sol = compute_c5(h_binary, dx)
                if c5_sol < best_c5:
                    best_h = h_binary.copy()
                    best_c5 = c5_sol
                    
        except Exception as e:
            pass  # Just continue if LP fails
        
        iterations += 1
        
        # For final refinement, try gradient descent on the best solution
        if iterations == max_iterations - 1:
            # Gradient descent on best_h
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            alpha = 0.01
            max_iter = 1000
            for _ in range(max_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                # Compute approximate gradient
                overlap = correlate(h, 1 - h, mode="full") * dx
                max_val = np.max(overlap)
                max_idx = np.argmax(overlap)
                gradient = np.zeros_like(h)
                
                # Simple gradient estimate by finite differences
                eps = 1e-3
                for i in range(n_points):
                    h_p = h.copy()
                    h_p[i] = min(1, h[i] + eps)
                    ov_p = correlate(h_p, 1 - h_p, mode="full") * dx
                    h_m = h.copy()
                    h_m[i] = max(0, h[i] - eps)
                    ov_m = correlate(h_m, 1 - h_m, mode="full") * dx
                    gradient[i] = (np.max(ov_p) - np.max(ov_m)) / (2 * eps)
                
                # Update
                h = h - alpha * gradient
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
