from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using correlate from scipy to compute the overlap efficiently
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx
