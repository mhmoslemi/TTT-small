import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available

    else:

    h = h * (target_sum / np.sum(h))
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    learning_rate = 0.01
    epsilon = 1e-8
    max_iter = int(1e5)
    
    # Iterative adjustment algorithm
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Identify the worst lag (midpoint of full correlation array)
        worst_lag = max_idx - (n_points - 1)
        
        # Compute sensitivity for gradient-like adjustment
        delta = 0.001
        
        # Perturb h to reduce the overlap at worst lag
        h_new = h.copy()
        
        # Adjust for worst lag by decreasing h[i] and increasing h[i + worst_lag]
        if worst_lag >= 0:
            # Positive lag, move right
            for i in range(n_points - worst_lag):
                if h[i] > delta and h[i + worst_lag] < 1 - delta:
                    h_new[i] -= delta
                    h_new[i + worst_lag] += delta
        else:
            # Negative lag, move left
            for i in range(-worst_lag, n_points):
                if h[i] > delta and h[i + worst_lag] < 1 - delta:
                    h_new[i] -= delta
                    h_new[i + worst_lag] += delta
                    
        # Ensure h_new remains in [0,1] and sum to target_sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # If solution improved substantially, accept it; otherwise, continue
        new_c5 = compute_c5_exact(h_new, dx)
        if new_c5 < current_c5 - epsilon:
            h = h_new
        else:
            # Use small random perturbations to escape local minima
            perturb = np.random.normal(0, 0.001, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            
    # Final computation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
