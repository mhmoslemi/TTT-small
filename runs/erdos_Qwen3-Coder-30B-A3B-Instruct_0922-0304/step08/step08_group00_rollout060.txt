import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable starting point
        h = np.ones(n_points) * 0.5
    
    # Ensure feasibility and normalization
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    best_c5 = float(best_c5)
    
    # Perturbation-based alternating optimization
    max_iter = 100
    perturbation_delta = 1e-3
    decay_factor = 0.98
    improvement_threshold = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb solution slightly
        h_perturbed = h + np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        
        # Split into even and odd indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Alternate fixing odd and even
        for fix_odd in [True, False]:
            # Create fixed values
            if fix_odd:
                fixed_vals = h_perturbed[odd_indices]
                free_vars = h_perturbed[even_indices]
                free_indices = even_indices
            else:
                fixed_vals = h_perturbed[even_indices]
                free_vars = h_perturbed[odd_indices]
                free_indices = odd_indices
            
            # Define objective: evaluate overlap with fixed vals
            def objective(free_vals):
                test_h = h_perturbed.copy()
                test_h[free_indices] = free_vals
                
                # Normalize
                current_sum = np.sum(test_h)
                if current_sum > 0:
                    test_h = test_h * (target_sum / current_sum)
                
                # Compute overlap
                overlap = correlate(test_h, 1 - test_h, mode="full")
                return np.max(overlap) * dx
            
            # Constraints
            bounds = [(0, 1) for _ in range(len(free_vars))]
            
            # Optimize
            result = minimize(objective, free_vars, method='L-BFGS-B', bounds=bounds, 
                            options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10})
            
            if result.success and result.fun < best_c5:
                if fix_odd:
                    test_h = h_perturbed.copy()
                    test_h[even_indices] = result.x
                else:
                    test_h = h_perturbed.copy()
                    test_h[odd_indices] = result.x
                
                current_sum = np.sum(test_h)
                if current_sum > 0:
                    test_h = test_h * (target_sum / current_sum)
                
                overlap_val = np.max(correlate(test_h, 1 - test_h, mode="full")) * dx
                if overlap_val < best_c5:
                    best_h = test_h
                    best_c5 = overlap_val
            
        # Update perturbation
        perturbation_delta *= decay_factor
        
        # Periodically update best
        current_overlap = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
        if current_overlap < best_c5:
            best_c5 = current_overlap
    
    # Final refinement
    def final_objective(h_vector):
        return np.max(correlate(h_vector, 1 - h_vector, mode="full")) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(final_objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-12})

    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = np.max(correlate(refined_h, 1 - refined_h, mode="full")) * dx
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
