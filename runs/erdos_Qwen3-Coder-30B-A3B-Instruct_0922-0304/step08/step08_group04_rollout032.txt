import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fitness(individual, dx):
    """Evaluate the fitness (C5) of an individual."""
    h = np.array(individual)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points / 2)
    return evaluate_c5(h, dx)

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Single point crossover
    crossover_point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate an individual by adding noise."""
    if random.random() > mutation_rate:
        return individual.copy()
    
    mutated = individual + np.random.normal(0, noise_magnitude, len(individual))
    return mutated

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    tournament_indices = random.sample(range(len(population)), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def genetic_algorithm(budget_s):
    """Run genetic algorithm to find optimal h."""
    start_time = time.time()
    
    # Parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.3
    noise_magnitude = 0.05
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            individual = parent + noise
            population.append(individual)
    else:
        population = []
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    best_individual = None
    best_fitness = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of entire population in parallel
        fitnesses = Parallel(n_jobs=1)(
            delayed(fitness)(individual, dx) for individual in population
        )
        
        # Track best individual
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
            
        if best_fitness <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation
            child1 = mutate(child1, mutation_rate, noise_magnitude)
            child2 = mutate(child2, mutation_rate, noise_magnitude)
            
            # Feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, n_points / 2)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, n_points / 2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Update noise magnitude
        noise_magnitude *= 0.99
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    global n_points, dx
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(budget_s)
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
