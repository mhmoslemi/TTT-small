import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure it satisfies constraints."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample shifts for QP constraints."""
    return np.random.randint(0, 2 * n - 1, sample_size)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = np.max(correlate(h_current, 1 - h_current, mode="full")) * dx
    
    # Parameters
    max_iterations = 1000
    sample_size = 50
    target_c5 = 0.38085
    improvement_threshold = 1e-6
    no_improvement_count = 0
    max_no_improvement = 200
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sample shifts to include in QP
        shift_indices = sample_shifts(n_points, sample_size)
        
        # Compute overlap values for sampled shifts
        overlap_values = []
        h_shifted = np.concatenate([h_current, np.zeros(n_points - 1)])
        for shift_idx in shift_indices:
            offset = shift_idx - n_points + 1
            if offset >= 0:
                overlap = np.sum(h_current * (1 - h_shifted[offset:offset + n_points]))
            else:
                overlap = np.sum(h_current[-offset:] * (1 - h_shifted[:n_points + offset]))
            overlap_values.append(overlap * dx)
        
        # Create QP problem with sampled constraints
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(h_vars))
        
        # Build constraints for sampled shifts
        constraints = []
        for i, shift_idx in enumerate(shift_indices):
            offset = shift_idx - n_points + 1
            if offset >= 0:
                temp_h = h_shifted[offset:offset + n_points]
            else:
                temp_h = h_shifted[:n_points + offset]
            constraint = cp.sum(cp.multiply(h_vars, temp_h)) <= overlap_values[i] + 1e-8
            constraints.append(constraint)
        
        # Add bounds constraints
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Add sum constraint
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # Solve QP
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                h_new = project_h(h_new, n_points)
                
                # Compute C5
                c5_new = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                if best_c5 <= 0.3807:
                    break
                    
        except Exception as e:
            # If solver fails, just proceed
            pass
        
        # Early stopping condition
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement
    h_refined = best_h.copy()
    for i in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        # Simple local search
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = np.max(correlate(h_temp, 1 - h_temp, mode="full")) * dx
        if c5_temp < np.max(correlate(h_refined, 1 - h_refined, mode="full")) * dx:
            h_refined = h_temp.copy()
    
    final_c5 = np.max(correlate(h_refined, 1 - h_refined, mode="full")) * dx
    return (h_refined.tolist(), final_c5, n_points)
