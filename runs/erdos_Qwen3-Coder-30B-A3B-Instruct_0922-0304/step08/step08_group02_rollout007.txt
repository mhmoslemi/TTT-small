import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient vector with respect to the given shift k*"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        idx = i + shift
        if 0 <= idx < n_points:
            grad[i] = 1 - h[idx]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap 
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        # The shift k* is the offset corresponding to the maximum overlap
        shift = max_overlap_idx - (n_points - 1)  # Convert from full to valid shift indices
        
        # Compute gradient w.r.t. the identified shift
        gradient = compute_gradient(h_current, shift, n_points)
        
        # Update h using projected gradient descent
        h_new = h_current - step_size * gradient
        h_new = project_h(h_new, n_points)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current solution
        h_current = h_new.copy()
        
        # Adjust learning rate if no improvement for many iterations
        if no_improvement_count >= 100:
            step_size *= 0.5
            no_improvement_count = 0
            
        # Occasionally perturb to escape local minima
        if iteration % 20 == 0 and no_improvement_count > 10:
            perturb_magnitude = 0.1 * np.mean(h_current)
            noise = np.random.normal(0, perturb_magnitude, n_points)
            h_current += noise
            h_current = project_h(h_current, n_points)
            
        iteration += 1
        
        # Early stopping if we're near target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
