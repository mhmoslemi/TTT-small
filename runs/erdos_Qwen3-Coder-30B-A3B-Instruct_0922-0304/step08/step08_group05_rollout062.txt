import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute full overlap
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each index that contributes to maximum overlap
    for lag_idx in max_indices:
        # Convert lag index to actual lag value
        # The overlap at index k corresponds to lag k - (n-1)
        lag = lag_idx - (n - 1)
        
        # Add gradient contribution from this lag
        if lag >= 0:
            # h[i] contributes to overlap at lag = k
            # where h[i] and (1-h[i+k]) overlap
            for i in range(n - lag):
                grad[i] += 2 * (1 - 2 * h[i + lag])
        else:
            # lag < 0
            # h[i] contributes to overlap at lag = k where k < 0
            # i.e., h[i] and (1-h[i+k]) where k = -lag
            for i in range(-lag, n):
                grad[i] += 2 * (1 - 2 * h[i + lag])
    
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    # Apply constraint gradient (sum = target_sum)
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update velocity with momentum
        velocity = momentum * velocity - learning_rate * grad
        
        # Update h
        h = h + velocity
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.995
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
