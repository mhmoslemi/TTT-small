import numpy as np
from scipy.signal import correlate
import time
import random

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    h = np.zeros(n)
    mid = n // 2
    for i in range(mid):
        h[i] = i / mid
        h[n-1-i] = i / mid
    h[mid] = 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Momentum parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    velocity = np.zeros(n_points)
    
    # Phase 1: Gradient descent with momentum
    max_iter_phase1 = 10000
    iter_count = 0
    
    while iter_count < max_iter_phase1 and time.time() - start_time < budget_s - 1:
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_idx = np.argmax(overlap)
        overlap_max = overlap[overlap_idx]
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute approximate gradient
        h_plus = h + 0.001
        h_plus = clip_and_renormalize(h_plus, target_sum)
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
        grad = (overlap_plus - overlap) / 0.001
        
        # Update velocity and h
        velocity = beta * velocity + (1 - beta) * grad
        h_candidate = h - alpha * velocity
        
        # Project to feasible set
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate.copy()
            alpha *= gamma
        else:
            alpha *= gamma * 0.9
            
        iter_count += 1
    
    # Phase 2: Coordinate-wise optimization
    max_iter_phase2 = 10000
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iter_count < max_iter_phase2 + max_iter_phase1 and time.time() - start_time < budget_s - 1:
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        overlap_idx = np.argmax(overlap)
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Sample coordinates to update
        update_fraction = 0.2
        n_update = max(1, int(update_fraction * n_points))
        indices = random.sample(range(n_points), n_update)
        
        improved = False
        for idx in indices:
            # Try to increase and decrease the current value
            h_plus = h.copy()
            h_plus[idx] = min(1.0, h[idx] + 0.01)
            h_plus = clip_and_renormalize(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[idx] = max(0.0, h[idx] - 0.01)
            h_minus = clip_and_renormalize(h_minus, target_sum)
            
            c5_minus = compute_c5(h_minus, dx)
            
            # Select the better change for this coordinate
            if c5_plus < overlap_max and c5_plus < c5_minus:
                h = h_plus.copy()
                improved = True
            elif c5_minus < overlap_max:
                h = h_minus.copy()
                improved = True
                
        if not improved:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
                
        iter_count += 1
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
