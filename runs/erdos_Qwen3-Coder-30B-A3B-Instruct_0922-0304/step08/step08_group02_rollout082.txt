import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to ensure sum = target_sum and values in [0,1]."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient for a given shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Compute gradient and update using projected gradient descent
        gradient = compute_gradient(h_current, max_overlap_idx - (n_points - 1), n_points)
        h_new = h_current - step_size * gradient
        
        # Project onto feasible set
        h_new = project_h(h_new, target_sum)
        
        # Check C5 value
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adjust step size if needed
        if no_improvement_count > 50:
            step_size *= 0.9
            no_improvement_count = 0
            
        # Random perturbation if stuck
        if no_improvement_count > max_no_improvement:
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h_current + perturbation
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            no_improvement_count = 0
            
        h_current = h_new.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
