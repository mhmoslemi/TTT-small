def project_h(h, n_points):
    """Project h to satisfy bounds and sum constraint."""
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def sample_shifts(n, sample_size):
    """Sample random shifts for QP constraints."""

    shift_range = 2 * n - 2
    shifts = np.random.randint(-n + 1, n, size=sample_size)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx
