import numpy as np
from scipy.fft import fft, ifft
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1], scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain."""
    H_conj = np.conj(H)
    conj_term = 1 - H
    S = np.abs(H * conj_term)
    return S

def spectral_update(H, S, learning_rate, target_bins):
    """Update H in frequency domain to reduce surrogate objective."""
    # Create update mask 
    update_mask = np.zeros_like(H)
    update_mask[target_bins] = 1
    
    # Compute gradient 
    grad = H * (1 - 2 * H) * update_mask
    
    # Apply update
    H_new = H - learning_rate * grad
    
    # Ensure stability (real values for real input)
    H_new = np.real(H_new)
    
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.random.uniform(0, 1, n_points)
    
    h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Initialize
    H_current = fft(h_current)
    
    iter_count = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Update learning rate
        if iter_count % 100 == 0:
            learning_rate *= 0.5
            
        # Compute surrogate
        S_current = compute_surrogate(H_current)
        
        # Select high-impact frequency bins
        # Target top 10% of frequency bins by contribution
        freq_indices = np.argsort(-S_current)[:max(1, n_points // 10)]
        
        # Apply spectral update
        H_new = spectral_update(H_current, S_current, learning_rate, freq_indices)
        
        # Reconstruct and project
        h_new = np.real(ifft(H_new))
        h_new = project_h(h_new, target_sum)
        
        # Compute C5
        c5_new = compute_c5(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Reset for stability
            H_current = fft(h_new)
        else:
            # Otherwise keep evolving
            H_current = H_new
            
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    final_h = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = final_h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(final_h, dx):
            final_h = h_temp.copy()
            
    final_c5 = compute_c5(final_h, dx)
    return (final_h.tolist(), final_c5, n_points)
