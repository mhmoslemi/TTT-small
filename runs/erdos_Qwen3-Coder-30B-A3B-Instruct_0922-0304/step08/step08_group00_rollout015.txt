import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Projected gradient descent with Nesterov acceleration
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    stagnation_count = 0
    max_stagnation = 1000
    
    # For Nesterov momentum
    v = np.zeros(n_points)
    y = h.copy()

        # Compute gradient of max_k C_k (approximated)
        # This is a simplified approach:
        # Compute gradients for each shift and use the one that contributes most
        grad = np.zeros(n_points)
        # For simplicity, we'll approximate this by including all relevant contributions
        for k_shift in range(n_points):
            # Compute C_k
            c_k = np.sum(h_vec * np.roll(1 - h_vec, k_shift))
            # Gradient (simplified - assuming max shift contributes)
            if k_shift == max_k:
                for i in range(n_points):
                    grad[i] += (1 - h_vec[(i + k_shift) % n_points])
        return grad
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:

        current_c5 = evaluate_overlap(h, dx)

            best_c5 = current_c5

        correlation = correlate(h, 1 - h, mode="full")

        # Compute gradient (simplified computation for now)
        # Actually, we compute gradient analytically w.r.t. max_k
        grad = np.zeros(n_points)

        k = max_k
        for i in range(n_points):
            # d/dh_i of h_i * (1 - h_{i+k})
            if i < n_points - k:
                grad[i] += (1 - h[i + k])
            else:
                grad[i] += (1 - h[i + k - n_points])
                
        # Gradient descent step
        v = momentum * v - step_size * grad
        y = h + v
        
        # Projection step
        y = np.clip(y, 0, 1)
        y_sum = np.sum(y)

    return (final_h.tolist(), final_c5, n_points)
