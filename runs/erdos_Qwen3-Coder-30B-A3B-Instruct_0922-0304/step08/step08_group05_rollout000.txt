import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.1):
    """Mutate individual with Gaussian noise."""
    h_new = h.copy()
    for i in range(len(h_new)):
        if random.random() < mutation_rate:
            h_new[i] += np.random.normal(0, sigma)
    return np.clip(h_new, 0, 1)

def crossover_parents(p1, p2):
    """Average-based crossover between two parents."""
    offspring = (p1 + p2) / 2.0
    return offspring

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population initialization
    population_size = 50
    population = []
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    else:
        # Generate a random feasible solution
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Generate remaining individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Tracking
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Evaluate fitness
        fitness_scores = []
        for individual in population:
            c5 = compute_c5_exact(individual, dx)
            fitness_scores.append(c5)
        
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        current_best_c5 = fitness_scores[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection: top 50% participate in mating
        elite_size = population_size // 2
        selected = population[:elite_size]
        
        # Generate new population
        new_population = selected.copy()
        
        # Fill remaining slots with offspring
        while len(new_population) < population_size:
            # Tournament selection to pick parents
            parent1 = selected[random.randint(0, len(selected)-1)]
            parent2 = selected[random.randint(0, len(selected)-1)]
            
            # Crossover
            offspring = crossover_parents(parent1, parent2)
            
            # Mutation
            offspring = mutate_individual(offspring, mutation_rate=0.1, sigma=0.05)
            
            # Ensure constraints
            offspring = normalize_h(offspring, target_sum)
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
        
        population = new_population[:population_size]
        generation += 1
    
    # Final check
    if best_h is None:
        # Fallback to the best solution from final population
        final_fitness = [compute_c5_exact(h, dx) for h in population]
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        best_c5 = final_fitness[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
