import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Local search on the segments
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Try different numbers of segments
    segment_counts = [3, 4, 5, 6, 7, 8, 9, 10]
    
    for segments in segment_counts:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Create segment boundaries
        boundaries = np.linspace(0, 2, segments + 1)
        
        # Try grid search on segment values
        step_size = 0.05
        grid_points = np.arange(0, 1.001, step_size)
        
        if len(grid_points) > 1000:  # Prevent explosion
            grid_points = grid_points[::10]
        
        best_seg_c5 = float('inf')
        best_seg_h = None
        
        # Try combinations of segment values
        from itertools import product
        num_combinations = len(grid_points) ** segments
        if num_combinations > 5000:  # Control complexity
            # Sample instead of full grid
            sample_indices = np.random.choice(len(grid_points), 
                                            size=min(1000, num_combinations // 100), 
                                            replace=False)
            grid_combinations = [tuple(grid_points[i] for i in sample_indices)]
        else:
            # Full grid for small cases
            grid_combinations = product(grid_points, repeat=segments)
            
        for combo in list(grid_combinations)[:1000]:  # Sample first 1000 or so
            if time.time() - start_time > budget_s - 2:
                break
                
            # Create h from the segment values
            h_seg = np.zeros(n_points)
            h_vals = list(combo)
            
            for i in range(segments):
                start_idx = int(boundaries[i] / dx)
                end_idx = int(boundaries[i+1] / dx)
                if end_idx > start_idx:
                    h_seg[start_idx:end_idx] = h_vals[i]
                elif i == segments - 1 and end_idx == start_idx:
                    h_seg[start_idx:] = h_vals[i]
            
            # Normalize
            h_seg = normalize_h(h_seg, n_points / 2)
            h_seg = np.clip(h_seg, 0, 1)
            
            # Evaluate
            c5 = evaluate_c5(h_seg, dx)
            
            if c5 < best_seg_c5:
                best_seg_c5 = c5
                best_seg_h = h_seg.copy()
                
        # Update best if found a better one
        if best_seg_c5 < best_c5:
            best_c5 = best_seg_c5
            best_h = best_seg_h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with some gradient steps if time allows
    if best_h is not None and time.time() - start_time < budget_s - 5:
        # Perform few gradient-based refinements
        max_iter = 1000
        lr = 0.001
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute gradient for max overlap
            overlap = correlate(best_h, 1 - best_h, mode="full")
            max_idx = np.argmax(overlap)
            
            # Estimate gradient with Finite differences
            eps = 1e-4
            grad = np.zeros_like(best_h)
            
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = normalize_h(h_plus, n_points / 2)
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                
                h_minus = best_h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = normalize_h(h_minus, n_points / 2)
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (overlap_plus[max_idx] - overlap_minus[max_idx]) / (2 * eps)
            
            # Apply update
            best_h = best_h - lr * grad
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, n_points / 2)
            
            # Recompute
            c5 = evaluate_c5(best_h, dx)
            if c5 < best_c5:
                best_c5 = c5
            else:
                break
                
            # Early stopping if getting close to target
            if best_c5 <= 0.3807:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
