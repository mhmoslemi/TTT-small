```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to feasible region: 0≤h≤1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of possible shifts"""
    shifts = np.random.choice(range(-n+1, n), sample_size, replace=False)
    return shifts

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = np.max(correlate(h_current, 1 - h_current, mode="full")) * dx
    
    iteration = 0
    last_improvement = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < max_no_improvement:
        # Sample shifts for QP constraints
        sample_shifts_list = sample_shifts(n_points, min(50, n_points))
        
        # Compute the overlap matrix for sampled shifts
        overlap_matrix = []
        overlap_targets = []
        
        for k in sample_shifts_list:
            # For shift k, compute the overlap integral
            if k >= 0:
                # h[i] * (1 - h[i+k]) 
                overlap = np.zeros(n_points)
                for i in range(n_points - k):
                    overlap[i] = h_current[i] * (1 - h_current[i+k])
                overlap = np.sum(overlap) * dx
            else:
                overlap = np.zeros(n_points)
                for i in range(-k, n_points):
                    overlap[i] = h_current[i] * (1 - h_current[i+k])
                overlap = np.sum(overlap) * dx
            overlap_matrix.append(overlap)
            overlap_targets.append(overlap)
            
        # Create QP formulation
        h_vars = cp.Variable(n_points)
        h_vars.value = h_current
        
        # Express overlap in terms of dot products
        # This model is simplified and doesn't exactly match the discrete problem,
        # but combined with projection it works reasonably.
        # We solve a QP to minimize sum of squared overlaps (in the sampled shifts) 
        
        # Simplified approach: minimize one overlap approximated as a QP constraint
        # We form a surrogate for the main objective
        
        # Instead, we'll optimize a simpler, but focused approach:
        # Minimize the maximum overlap among sampled shifts, using a sliding QP
        
        # Create objective to minimize largest overlap among sample
        # Keep a single constraint that tries to reduce the maximum overlap
        max_overlap = cp.Variable()
        
        # Constraints that the maximum overlap upper bounded by max_overlap  
        # For each sampled shift, we constraint that overlap ≤ max_overlap
        # But we don't make the constraints tight because they're not linear a la the continuous form
        
        # Redefine a better objective: minimize max overlap among sampled overlaps
        # Using a robust QP solver efficiently
        
        # For implementation simplicity:
        # Sample overlaps for which we want to enforce constraints.
        # Approximate overall max overlap by minimizing total overlap of sampled shifts.
        
        # So we use a proxy: minimize sum of correlation values over sampled shifts
        # This is amenable to projection QP  
        
        # Even simpler and more robust approach: return to local search with constraints
        # Use gradient method but do so with        careful constraint handling
        
        h_old = h_current.copy()
        
        # Sample a small number of random directions for gradient-like descent
        n_directions = 20  
        best_h_new = h_current.copy()
        min_c5 = best_c5
        
        for _ in range(n_directions):
            if time.time() - start_time > budget_s - 1:
                break
            # Generate a random perturbation
            delta = np.random.normal(0, 0.01, n_points)
            h_new = h_old + delta
            
            # Project to feasible domain
            h_new = project_h(h_new, n_points)
            
            # Compute new C5
            c5_new = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
            if c5_new < min_c5:
                min_c5 = c5_new
                best_h_new = h_new.copy()
                
        # Update current solution
        h_current = best_h_new
        current_c5 = min_c5
        
        # Update best solution if necessary
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            last_improvement = iteration
            
        iteration += 1
        
        # Early termination if target is met
        if best_c5 <= 0.3807:
            break
            
    # Final exit with best solution
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
```
