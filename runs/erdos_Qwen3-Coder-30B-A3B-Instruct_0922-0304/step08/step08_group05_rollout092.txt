import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Parameters
    epsilon = 0.01
    max_iter = 10000
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < max_iter:
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        
        # Get the optimal lag for maximum overlap
        lag = max_idx - (n_points - 1)
        
        # We only adjust for the main worst lag
        # Use a more refined approach to update h
        h_new = h.copy()
        
        # Adjust h to reduce the overlap by modifying values around
        # the positions that contribute most to the overlap
        for i in range(n_points):
            # Compute sensitivity at position i
            sensitivity = 0.0
            for k in range(-n_points+1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    sensitivity += 2 * (1 - 2 * h[j])
            
            # Apply adjustment for position i
            if abs(sensitivity) > 1e-10:
                adjustment = epsilon * sensitivity
                h_new[i] -= adjustment
                
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Test new solution
        current_c5 = compute_c5_exact(h_new, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if best_c5 <= 0.3807:
            break
            
        h = h_new
        iteration += 1
        
    # Final recomputation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
