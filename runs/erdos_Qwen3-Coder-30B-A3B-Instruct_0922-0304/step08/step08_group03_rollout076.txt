import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if np.abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    # Optimization parameters
    momentum = 0.9
    step_size = 0.001
    shrink_factor = 0.5
    max_iter = 20000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    velocity = np.zeros(n_points)
    
    for i in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        # Update best if improved
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        # We approximate the gradient of max overlap w.r.t. h
        # Using the fact that overlap at position k is sum(h[j] * (1 - h[j+k]))
        gradient = np.zeros(n_points)
        for k in range(len(overlap)):
            # k corresponds to shift
            shift = k - (n_points - 1)
            if shift <= 0:
                # right shift case
                grad_part = np.zeros(n_points)
                for j in range(n_points):
                    if j + shift >= 0:
                        grad_part[j] = 1 - h[j + shift]
                gradient += grad_part
            else:
                # left shift case  
                grad_part = np.zeros(n_points)
                for j in range(n_points):
                    if j - shift < n_points:
                        grad_part[j] = h[j - shift]
                gradient -= grad_part
            
        # Normalize gradient
        gradient = gradient / n_points
        
        # Update with momentum
        velocity = momentum * velocity + step_size * gradient
        h_candidate = h - velocity
        
        # Project to feasibility
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate new candidate
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate
        else:
            # Try smaller step
            step_size *= shrink_factor
            if step_size < 1e-10:
                # Reset if step sizes get too small
                h = clip_and_renormalize(h, target_sum)
                step_size = 0.001
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
