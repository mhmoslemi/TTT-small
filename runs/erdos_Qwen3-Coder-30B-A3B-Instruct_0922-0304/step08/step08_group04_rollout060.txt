import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def solve_convex_relaxation(h_init, n_points, dx):
    """Solve convex relaxation to get initial feasible point."""
    # Create variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1
    ]
    
    # Add overlap constraints
    # For each shift k, we want sum h[i] * (1 - h[i+k]) <= t
    for k in range(2 * n_points - 1):
        # Compute the sum for this shift
        expr = 0
        for i in range(n_points):
            j = i + k - (n_points - 1)
            if 0 <= j < n_points:
                expr += h[i] * (1 - h[j])
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, FEASIBLE]:
            return np.array(h.value).reshape(-1)
    except:
        pass
    
    # Fallback: return normalized initial guess
    return h_init

def compute_gradient(h, dx, n_points):
    """Compute approximate gradient of the C5 objective."""
    # Compute overlap values using scipy for speed in batch
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    max_val = overlap[max_idx]
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # For each position compute the gradient contribution
    for i in range(n_points):
        total_grad = 0.0
        for k in range(2 * n_points - 1):
            # Position in the full correlation
            pos1 = i
            pos2 = i + k - (n_points - 1)
            
            # Only consider valid indices
            if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                # The derivative of h[i]*(1-h[i+k]) with respect to h[i]
                if k == 0:
                    grad[i] += (1 - 2 * h[i]) * (1 - h[pos2])
                else:
                    grad[i] += h[pos2] - h[i] * h[pos2] - h[i] * (1 - h[pos2])
    
    grad = grad * 2.0 / dx
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999):
    """Optimize using gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Periodic loss evaluation
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Early termination condition
            if loss <= 0.3807:
                break
                
            # Check for convergence
            if abs(prev_loss - loss) < 1e-8:
                no_improvement_count += 1
                if no_improvement_count > 1000:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Solve convex relaxation to improve initial point
    relaxed_h = solve_convex_relaxation(h, n_points, dx)
    if relaxed_h is not None:
        h = relaxed_h
    
    # Optimize with gradient descent
    best_h, best_c5 = optimize_with_gradient_descent(h, dx, n_points)
    
    # Final check with exact computation
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
