```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed(h, n_points, dx, max_time=10):
    """Solve the relaxed version using CVXPY."""
    start_time = time.time()
    h_r = cp.Variable(n_points)
    z = cp.Variable()

    # Constraints
    constraints = [0 <= h_r, h_r <= 1, cp.sum(h_r) == n_points / 2]

    # Objective: minimize z such that z >= overlap for all k
    # We'll handle this constraint-wise
    objective = cp.Minimize(z)

    # Build overlap constraints for all shifts
    for k in range(1, n_points):  # k = 1 to n_points-1
        if time.time() - start_time > max_time:
            break
        # For this shift, compute the sum
        # overlap_k = sum h_r[i] * (1 - h_r[i + k]) for i = 0 to n_points - k - 1
        idxs = np.arange(n_points - k)
        overlap_sum = cp.sum(cp.multiply(h_r[idxs], 1 - h_r[idxs + k]))
        constraints.append(z >= overlap_sum * dx)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False, time_limit=max_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h_r.value
        else:
            h_solution = h.copy()
    except:
        # Fallback to original
        h_solution = h.copy()
    return h_solution

def round_to_binary(h_relaxed, target_sum):
    """Convert relaxed solution to binary-like."""
    # Use median threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed > threshold).astype(float)
    current_sum = np.sum(h_binary)
    
    # Adjust to match target
    if current_sum == target_sum:
        return h_binary
    
    diff = int(current_sum - target_sum)
    if diff > 0:
        # Need to reduce by setting some 1s to 0s
        # Instead of just taking largest, we identify where we have
        # the most "overshot" value (larger values that should be 0)
        candidates = np.where(h_binary > 0)[0]
        # Sort by values in descending order
        sorted_indices = np.argsort(h_relaxed[candidates])[::-1]
        indices_to_zero = candidates[sorted_indices[:diff]]
        h_binary[indices_to_zero] = 0
    elif diff < 0:
        # Need to increase by setting some 0s to 1s
        candidates = np.where(h_binary == 0)[0]
        # Sort by values in ascending order
        sorted_indices = np.argsort(h_relaxed[candidates])
        indices_to_one = candidates[sorted_indices[:abs(diff)]]
        h_binary[indices_to_one] = 1

    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Initialize with uniform distribution scaled
        h = np.random.rand(n_points) * 0.5  # Avoid extreme values
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)

    # Initialize best
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    # Iteration with alternating stages
    max_iter = 50
    iter_count = 0
    no_improvement_count = 0

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Stage 1: Relaxation
        if not (iter_count % 3 == 0):
            # Occasionally do relaxation, otherwise just use previous best
            relaxed_h = h.copy()
        else:
            relaxed_h = solve_relaxed(h, n_points, dx, max_time=5)
        
        # Stage 2: Rounding
        h_binary = round_to_binary(relaxed_h, target_sum)
        h_binary = np.clip(h_binary, 0, 1)

        # Stage 3: Optional adjustment to be more careful about overlap
        # This is a simplified way to reduce overlap a bit more
        # Compute overlap to know the current structure
        coh = correlate(h_binary, 1 - h_binary, mode="full") * dx
        current_max = np.max(coh)

        # if we ran into convergence early, also just try small changes
        h_new = h_binary.copy()
        if current_max < best_c5 * 0.999:
            # Try to make a local adjustment in strategies
            for i in range(n_points):
                if np.random.rand() < 0.05:  # tiny probability of change
                    old_h = h_new[i]
                    h_new[i] = max(0, min(1, old_h + np.random.normal(0, 0.05)))
                    h_new = normalize_h(h_new, target_sum)
                    new_c5 = evaluate_c5(h_new, dx)
                    if new_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = new_c5
                    h_new = h_binary.copy()  # revert for next iteration

        # Overall update if improvement
        new_c5 = evaluate_c5(h_binary, dx)
        if new_c5 < best_c5:
            best_h = h_binary.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update iteration state
        h = h_binary.copy()
        iter_count += 1
        
        # Early stop if few progress
        if no_improvement_count > 10:
            break

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
