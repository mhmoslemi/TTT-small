import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def sample_shifts(n_points, sample_size):
    """Sample shifts for QP formulation"""
    # Sample shifts from -n_points+1 to n_points-1
    shifts = np.random.randint(-n_points + 1, n_points, size=sample_size)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Adaptive parameters
    sample_size = 50
    target_overlap = 0.38085  # Start from known best
    max_iter = int(budget_s * 2)  # Rough estimate
    no_improvement_count = 0
    max_no_improvement = 200
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample shifts for this iteration
        shifts = sample_shifts(n_points, sample_size)
        
        # Formulate QP for these shifts
        h = Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
        ]
        
        # Add overlap constraints
        # We compute overlaps for each shift
        max_overlap = 0
        for k in shifts:
            # For shift k, we compute sum h(x) * (1 - h(x+k))
            # This is a bit tricky in cvxpy due to indexing
            # Let's compute this directly in numpy for simplicity
            # We treat this as a constraint for the QP
            
            # Compute columns for matrix representation
            if k >= 0:
                # h(i) * (1 - h(i+k)) for valid indices
                # valid range is [0, n_points-k-1] for h(i+k) to exist
                cols = np.arange(n_points - k)
                rows = cols + k
                if len(cols) > 0:
                    # h[i] * (1 - h[i+k]) = h[i] - h[i] * h[i+k]
                    # So we're constraining: sum(h[i] - h[i] * h[i+k]) <= target_overlap
                    # This is non-convex, so we approximate
                    # Simple approach: linearize or just sample
                    pass # Skip QP formulation for now for simplicity
            else:
                # k < 0, so h(i+k) becomes h(i+k)
                # valid range is [0, n_points+k-1] to have h(i+k) defined
                cols = np.arange(n_points + k)
                rows = cols - k
                if len(cols) > 0:
                    # Same process
                    pass # Skip QP formulation 

        # Since QP approach with all overlap constraints is complex due to non-convexity,
        # we fall back to a simpler heuristic optimization:
        
        # Sample perturbations to improve on current solution
        candidate_h = h_current.copy()
        # Apply a small random perturbation
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            candidate_h[idx] = np.clip(candidate_h[idx] + delta, 0, 1)
        
        # Renormalize
        candidate_h = candidate_h * (target_sum / np.sum(candidate_h))
        candidate_c5 = compute_c5(candidate_h, dx)
        
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update h_current to be the best solution so far for next iteration
        h_current = best_h.copy()
        
        # Update target_overlap adaptively
        target_overlap = max(0.3800, best_c5 - 0.00001 * iteration)
        
        if no_improvement_count > max_no_improvement:
            break
            
        iteration += 1
        
        # Early stop if good solution found
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_h = project_h(best_h, n_points, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
