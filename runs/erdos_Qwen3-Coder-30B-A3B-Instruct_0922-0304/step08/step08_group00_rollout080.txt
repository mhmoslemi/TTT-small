import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using correlation with flipped h to compute the overlap
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Define the optimization problem
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    def sum_constraint(h_vector):
        return np.sum(h_vector) - target_sum
    
    # Define constraint and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Use SLSQP solver
    options = {'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
    
    # Try optimization multiple times with different initializations
    for attempt in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # SLSQP optimization
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options=options)
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
        
        # Perturb solution slightly for next attempt
        h_perturbed = best_h + np.random.normal(0, 1e-4, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        best_h = h_perturbed
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
