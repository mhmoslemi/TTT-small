import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    """Create a triangular ramp as initial guess."""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Phase 1: Gradient descent with momentum
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    velocity = np.zeros(n_points)
    
    # Phase 2: Coordinate-wise optimization
    coord_sample_rate = 0.2
    coord_sample_size = max(1, int(n_points * coord_sample_rate))
    
    phase1_iter = 0
    phase2_iter = 0
    max_phase1_iter = 10000
    max_phase2_iter = 10000
    early_stop_count = 0
    max_early_stop = 2000
    
    while time.time() - start_time < budget_s - 1:
        # Phase 1: Gradient Descent with Momentum
        if phase1_iter < max_phase1_iter:
            # Compute overlap and find maximum
            overlap = correlate(h, 1 - h, mode="full") * dx
            overlap_max = np.max(overlap)
            max_idx = np.argmax(overlap)
            
            # Compute approximate gradient (elements that contribute most to overlap)
            gradient = np.zeros(n_points)
            for i in range(n_points):
                # Approximate contribution to overlap at max_idx
                if i + max_idx < n_points:
                    gradient[i] += (1 - 2 * h[i]) * (1 - h[i + max_idx])
                else:
                    # Handle wrap around for full correlation
                    idx_wrap = i + max_idx - n_points
                    if idx_wrap >= 0:
                        gradient[i] += (1 - 2 * h[i]) * (1 - h[idx_wrap])
            
            # Apply momentum
            velocity = beta * velocity + (1 - beta) * gradient
            h_candidate = h - alpha * velocity
            
            # Project to feasible region
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            
            # Evaluate new C5
            new_c5 = compute_c5(h_candidate, dx)
            
            if new_c5 < overlap_max:
                h = h_candidate.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                alpha *= gamma
                early_stop_count = 0
            else:
                # Reset
                alpha *= gamma
                early_stop_count += 1
            phase1_iter += 1
        else:
            # Switch to coordinate-wise optimization
            if phase2_iter < max_phase2_iter:
                # Randomly sample coordinates to update
                indices = np.random.choice(n_points, size=coord_sample_size, replace=False)
                
                original_h = h.copy()
                improved = False
                
                for i in indices:
                    # Compute current overlap
                    current_overlap = correlate(h, 1 - h, mode="full") * dx
                    current_max = np.max(current_overlap)
                    
                    # Try adjustments
                    h_plus = h.copy()
                    h_plus[i] = min(1.0, h[i] + 0.01)
                    h_plus = clip_and_renormalize(h_plus, target_sum)
                    overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                    max_plus = np.max(overlap_plus)
                    
                    h_minus = h.copy()
                    h_minus[i] = max(0.0, h[i] - 0.01)
                    h_minus = clip_and_renormalize(h_minus, target_sum)
                    overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
                    max_minus = np.max(overlap_minus)
                    
                    # Increment if beneficial
                    if max_plus < current_max and max_plus < max_minus:
                        h = h_plus.copy()
                        improved = True
                    elif max_minus < current_max:
                        h = h_minus.copy()
                        improved = True
                    
                    if improved:
                        break
                        
                if not improved:
                    early_stop_count += 1
                else:
                    early_stop_count = 0
                    
                phase2_iter += 1
            else:
                # Both phases exhausted
                break

        if early_stop_count >= max_early_stop or best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
