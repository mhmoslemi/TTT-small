import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Build QP formulation
    h_var = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Feasibility constraints
    constraints.append(cp.sum(h_var) == n_points / 2)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints
    # For each possible shift k, we need to form the overlap constraint
    # This includes all valid shifts in the full correlation
    for k in range(-n_points + 1, n_points):
        # Build constraint for shift k
        # When k >= 0: h[i] * (1 - h[i+k]) 
        # When k < 0: h[i] * (1 - h[i+k+n_points])
        expr = 0
        for i in range(n_points):
            if k >= 0 and i + k < n_points:
                expr += h_var[i] * (1 - h_var[i + k])
            elif k < 0 and i + k >= 0:
                expr += h_var[i] * (1 - h_var[i + k])
            elif k < 0 and i + k + n_points < n_points:
                expr += h_var[i] * (1 - h_var[i + k + n_points])
            else:
                continue
        constraints.append(expr <= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try solving with ECOS
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, reltol=1e-6, abstol=1e-6)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_bound = t.value
        else:
            # Fallback to initial solution if optimization fails
            h_opt = h.copy()
            c5_bound = evaluate_c5(h_opt, dx)
    except:
        # Fallback to initial solution if optimization fails
        h_opt = h.copy()
        c5_bound = evaluate_c5(h_opt, dx)
    
    # Make sure that we have a valid solution
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Evaluate once more for final check
    final_c5 = evaluate_c5(h_opt, dx)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        # Final check of the rounded solution
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        final_c5 = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
