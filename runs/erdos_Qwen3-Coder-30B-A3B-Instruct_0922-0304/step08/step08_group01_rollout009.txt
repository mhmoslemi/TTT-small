import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # MINLP approach begins
    max_iter = 500  # Reduced iterations for budget match
    iter_count = 0
    
    # Convex relaxation variables
    h_vars = cp.Variable(n_points)
    z = cp.Variable()
    
    # Constraint definitions - Convex relaxation approach
    constraints = [0 <= h_vars, h_vars <= 1, cp.sum(h_vars) == target_sum, z >= 0]
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # First, try convex relaxation approach to get better starting point
        relaxed_constraints = constraints[:]
        for k in range(n_points):
            # Simplified overlay of constraints from model rules
            # Instead of full correlation (-∞ to +∞), consider a bounded window
            overlap_k_l = 0.0
            overlap_k_r = 0.0
            for x in range(n_points):
                if x + k < n_points:
                    overlap_k_l += h_vars[x] * (1 - h_vars[x + k])
                if x - k >= 0:
                    overlap_k_r += h_vars[x] * (1 - h_vars[x - k])
            relaxed_constraints.append(overlap_k_l <= z)
            # Handle the LHS case separately for tighter definition
            # In practice, we approximate both sides as equal for conservative bound
            # This could be made more precise by using bilinear constraints        
            
        # Relaxation approach with relaxation of bilinear terms
        # But in this simple discrete framework, let's do direct optimization
        h_new = h.copy()
        step_size = 0.01
        
        # Gradient-based refinement
        for i in range(n_points):
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            
            h_new[i] += step_size * grad
            h_new[i] = np.clip(h_new[i], 0, 1)
            
        # Renormalize
        h_new = h_new * (target_sum / np.sum(h_new))
        h = h_new
        
        # Evaluate improvement
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
