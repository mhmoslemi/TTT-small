import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5

    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    current_c5 = best_c5

    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    sigma = 0.05
    max_no_improvement = 1000
    no_improvement_count = 0
    temperature = T0
    iteration = 0

    while time.time() - start_time < budget_s - 1:
        # Perturb one element
        h_new = h_current.copy()
        idx = random.randint(0, n_points - 1)
        delta = np.random.normal(0, sigma)
        h_new[idx] = h_new[idx] + delta
        h_new = project_to_feasible(h_new, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        delta_c5 = new_c5 - current_c5
        
        # Accept with Metropolis-Hastings criterion
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / temperature):
            h_current = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Annealing
        temperature = max(T_min, T0 * (alpha ** (iteration / 1000)))
        
        # Adaptive perturbation scaling
        if no_improvement_count > max_no_improvement:
            sigma *= 0.9
            no_improvement_count = 0
            
        iteration += 1
        
        # Early stopping if we're very close to the target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
