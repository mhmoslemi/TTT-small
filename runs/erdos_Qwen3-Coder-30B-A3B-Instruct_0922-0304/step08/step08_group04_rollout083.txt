import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate the individual."""
    if random.random() > mutation_rate:
        return h.copy()
    
    h = h + np.random.normal(0, noise_magnitude, len(h))
    return np.clip(h, 0, 1)

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    best_idx = random.randint(0, len(population) - 1)
    for _ in range(tournament_size - 1):
        idx = random.randint(0, len(population) - 1)
        if fitnesses[idx] < fitnesses[best_idx]:
            best_idx = idx
    return population[best_idx].copy()

def genetic_algorithm(seed=42, budget_s=600, n_points=70):
    """Run genetic algorithm to minimize C5."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        for _ in range(pop_size):
            individual = mutate(parent.copy(), noise_magnitude=0.05)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    else:
        for _ in range(pop_size):
            individual = np.random.rand(n_points).astype(np.float64)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    dx = 2.0 / n_points
    best_individual = None
    best_c5 = float('inf')
    no_improvement = 0
    
    max_generations = 1000
    for generation in range(max_generations):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness in parallel
        fitnesses = Parallel(n_jobs=1)(
            delayed(evaluate_c5)(ind, dx) for ind in population
        )
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = fitnesses.index(min_fitness)
            best_individual = population[best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early termination if target met
        if best_c5 <= 0.3807:
            break
            
        # Stop if no improvement for too long
        if no_improvement > 100:
            break
            
        # Create new generation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Generate rest through selection, crossover, and mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, noise_magnitude=0.05 * (0.99 ** (generation // 10)))
            child2 = mutate(child2, noise_magnitude=0.05 * (0.99 ** (generation // 10)))
            child1 = normalize_h(child1, n_points / 2)
            child2 = normalize_h(child2, n_points / 2)
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Occasionally noisy restarts
        if generation % 50 == 0 and generation > 0:
            for i in range(pop_size // 5):
                idx = random.randint(0, len(population) - 1)
                population[idx] = np.random.rand(n_points).astype(np.float64)
                population[idx] = normalize_h(population[idx], n_points / 2)
                
    if best_individual is None:
        # If no improvement found, try to optimize final solution
        fitnesses = [evaluate_c5(ind, dx) for ind in population]
        min_fitness = min(fitnesses)
        best_idx = fitnesses.index(min_fitness)
        best_individual = population[best_idx]
        best_c5 = min_fitness
    
    return best_individual, best_c5

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = genetic_algorithm(seed, budget_s)
    n_points = len(best_h)
    
    # Final validation
    dx = 2.0 / n_points
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Ensure bounds and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, n_points / 2)
    
    return (best_h.tolist(), best_c5, n_points)
