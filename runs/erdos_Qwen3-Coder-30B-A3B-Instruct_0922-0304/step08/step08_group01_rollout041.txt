import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try convex relaxation approach
    max_iter_lp = 10
    iter_count = 0
    
    for _ in range(max_iter_lp):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Solve relaxed problem using CVXPY
        try:
            h_r = cp.Variable(n_points)
            z = cp.Variable()
            
            # Constraints
            constraints = []
            constraints.append(cp.sum(h_r) == target_sum)
            constraints.append(h_r >= 0)
            constraints.append(h_r <= 1)
            
            # Overlap constraints
            for k in range(-n_points + 1, n_points):
                if k >= 0:
                    expr = cp.sum(cp.multiply(h_r[:n_points-k], 1 - h_r[k:]))
                else:
                    expr = cp.sum(cp.multiply(h_r[-k:], 1 - h_r[:n_points+k]))
                constraints.append(expr * dx <= z)
            
            # Objective
            objective = cp.Minimize(z)
            prob = cp.Problem(objective, constraints)
            
            # Try different solvers
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_r.value
                # Rounding heuristic
                h_binary = (h_relaxed >= np.median(h_relaxed)).astype(float)
                current_sum = np.sum(h_binary)
                diff = current_sum - target_sum
                
                # Adjust binary vector to match target sum
                if diff > 0:
                    # Need to decrease sum
                    indices = np.argsort(h_binary)
                    for i in range(int(diff)):
                        h_binary[indices[i]] = 0
                elif diff < 0:
                    # Need to increase sum
                    indices = np.argsort(h_binary)
                    for i in range(int(-diff)):
                        h_binary[indices[-(i+1)]] = 1
                    
                # Ensure exact sum constraint
                h_binary = normalize_h(h_binary, target_sum)
                h_binary = np.clip(h_binary, 0, 1)
                
                new_c5 = compute_c5(h_binary, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_binary.copy()
    
        except:
            pass
            
        iter_count += 1
    
    # Final gradient-based refinement
    h = best_h.copy()
    max_iter = 50000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Identify indices where maximum overlap occurs
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        
        # Simple gradient descent approach
        gradients = np.zeros(n_points)
        for i in range(n_points):
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            gradients[i] = grad
        
        # Update top 5% coordinates
        idxs = np.argsort(np.abs(gradients))[::-1]
        update_count = max(1, n_points // 20)
        selected_idxs = idxs[:update_count]
        
        for i in selected_idxs:
            old_h_i = h[i]
            grad = gradients[i]
            step_size = 0.005 if grad != 0 else 0.001
            
            if grad > 0:
                h[i] = max(0, h[i] - step_size)
            else:
                h[i] = min(1, h[i] + step_size)
        
        # Renormalize
        h = normalize_h(h, target_sum)
        
        # Check improvement
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
