import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover_parents(p1, p2):
    """Blend two parent solutions"""
    n = len(p1)
    crossover_point = np.random.randint(1, n-1)
    child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    return child

def mutate_h(h, mutation_rate=0.05):
    """Mutate h by randomly adjusting some elements"""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            delta = np.random.normal(0, 0.05)
            h[i] = h[i] + delta
    return h

def local_descent(h, target_sum, dx, max_iter=50):
    """Apply local gradient descent to refine solution"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        if np.random.random() < 0.5:
            # Try smaller random perturbation
            idx = np.random.randint(len(h))
            perturb = np.random.uniform(-0.02, 0.02)
            h_temp = h.copy()
            h_temp[idx] = np.clip(h_temp[idx] + perturb, 0, 1)
            h_temp = repair_h(h_temp, target_sum)
            c5_temp = compute_c5(h_temp, dx)
            if c5_temp < best_c5:
                best_h = h_temp.copy()
                best_c5 = c5_temp
        else:
            # Small perturbation to make progression more stable
            h_mod = h.copy()
            idx = np.random.randint(len(h))
            perturb = np.random.normal(0, 0.01)
            h_mod[idx] = np.clip(h_mod[idx] + perturb, 0, 1)
            h_mod = repair_h(h_mod, target_sum)
            c5_mod = compute_c5(h_mod, dx)
            if c5_mod < best_c5:
                h = h_mod.copy()
                best_c5 = c5_mod
                
        if best_c5 < compute_c5(h, dx):
            h = best_h.copy()
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    population = []
    
    # Add initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (target_sum / np.sum(h0))
        population.append(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        population.append(h0)
        
    # Fill up population
    while len(population) < n_pop:
        # Create a new random solution
        h = np.random.rand(n_points)
        # Make it such that sum is correct
        h = repair_h(h, target_sum)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h, dx)
    
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all solutions
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # If no improvements for a while, break early
        if stagnation_count > 100:
            break
            
        # Keep top 20% (elitism)
        num_keep = max(1, n_pop // 5)
        keep_indices = sorted_indices[:num_keep]
        new_population = [population[i] for i in keep_indices]
        
        # Create offspring
        while len(new_population) < n_pop:
            # Selection: tournament
            tournament_size = 3
            tournament_indices = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            tournament_indices = np.random.choice(n_pop, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.random() < 0.8:  # Crossover probability
                child = crossover_parents(parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutation
            child = mutate_h(child)
            
            # Repair
            child = repair_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
        
        # Local refinement on top solutions (more expensive but beneficial)
        for i in range(min(3, len(population))):
            h_refined = local_descent(population[i], target_sum, dx)
            c5_refined = compute_c5(h_refined, dx)
            if c5_refined < fitness[i]:
                population[i] = h_refined
                fitness[i] = c5_refined
    
    # Final refinement of best solution
    best_h = local_descent(best_h, target_sum, dx, max_iter=100)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
