import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the C5 objective at current h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps and find maximum
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    
    # Find all indices where maximum overlap occurs
    active_lags = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # Compute subgradient with respect to h[i]
    # For each active lag k, we have contribution to gradient
    for k in active_lags:
        for i in range(n):
            if 0 <= i + k < n:
                # (1 - h[i + k]) contributes to overlap[i]
                grad[i] += 2 * (1 - h[i + k]) * dx
            if 0 <= i - k < n:
                # h[i - k] contributes to overlap[i]  
                grad[i] -= 2 * h[i - k] * dx
    
    # Normalize by n (since we are effectively computing average gradient)
    grad = grad / n
    
    # Apply constraint gradient to ensure sum = target_sum
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values.copy())
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum  # Normalize after clipping
    
    # Parameters
    learning_rate = 0.01
    momentum = 0.9
    v = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum
        v = momentum * v + grad
        h = h - learning_rate * v
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        
        iteration += 1
        
        # Adjust learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.995
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
