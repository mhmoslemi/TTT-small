import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed_lp(h_start, n_points, dx):
    """Solve relaxed LP for Erdos problem"""
    # Variables
    h_r = cp.Variable(n_points)
    z = cp.Variable()
    
    # Constraints
    constraints = []
    # Sum constraint
    constraints.append(cp.sum(h_r) == n_points / 2)
    
    # Bound constraints
    constraints.append(h_r >= 0)
    constraints.append(h_r <= 1)
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # Forward overlap
            expr = 0
            for x in range(n_points - k):
                expr += h_r[x] * (1 - h_r[x + k])
            constraints.append(expr * dx <= z)
        else:
            # Backward overlap
            expr = 0
            for x in range(-k, n_points):
                expr += h_r[x] * (1 - h_r[x + k])
            constraints.append(expr * dx <= z)
    
    # Objective
    objective = cp.Minimize(z)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Round to binary
            h_r_sol = h_r.value
            if h_r_sol is not None:
                threshold = np.median(h_r_sol)
                h_b = (h_r_sol >= threshold).astype(float)
                # Adjust to meet sum constraint
                current_sum = np.sum(h_b)
                deficit = current_sum - n_points / 2
                
                if deficit > 0:
                    # Need to turn some 1s to 0s
                    # Find indices of 1s sorted by value (descending)
                    idxs = np.argsort(h_r_sol)[::-1]
                    for i in range(int(deficit)):
                        if h_b[idxs[i]] == 1:
                            h_b[idxs[i]] = 0
                elif deficit < 0:
                    # Need to turn some 0s to 1s
                    # Find indices of 0s sorted by value (ascending)
                    idxs = np.argsort(h_r_sol)
                    for i in range(int(-deficit)):
                        if h_b[idxs[i]] == 0:
                            h_b[idxs[i]] = 1

                return h_b, z.value
    except:
        pass

    # Fallback: return original
    return h_start, np.inf

    np.random.seed(seed)

    n_points = 70

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 50
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try LP relaxation first
        if iter_count % 3 == 0:
            h_lp, c5_lp = solve_relaxed_lp(h, n_points, dx)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
            # Use LP solution as initial for actual optimization
            h = h_lp
        else:
            # Gradient descent approach
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
            
            # Identify the maximum conflict indices
            max_indices = np.where(np.abs(overlap) == current_max)[0]
            
            # Simple gradient descent with momentum
            step_size = 0.02
            gradients = np.zeros(n_points)
            
            # Approximate gradient for overlap with respect to h[i]
            for i in range(n_points):
                grad = 0
                for k in range(n_points):
                    if i + k < n_points:
                        grad += (1 - 2 * h[i]) * (1 - h[i + k])
                    if i - k >= 0:
                        grad += (1 - 2 * h[i]) * (1 - h[i - k])
                gradients[i] = grad
            
            # Update using gradient descent
            h_old = h.copy()
            h = h - step_size * gradients
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Check improvement
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            
            # Early stopping if target achieved
            if best_c5 <= 0.3807:

        iter_count += 1
        
        # Occasionally recompute overlap for precision
        if iter_count % 10 == 0:
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
