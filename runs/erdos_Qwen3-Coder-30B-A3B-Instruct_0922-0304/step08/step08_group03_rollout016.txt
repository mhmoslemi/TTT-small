import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():

    else:

        segments = 8
        segment_length = n_points // segments
        h = np.zeros(n_points)
        for i in range(segments):
            start_idx = i * segment_length
            end_idx = (i + 1) * segment_length if i < segments - 1 else n_points
            if i % 2 == 0:
                h[start_idx:end_idx] = np.linspace(0, 1, end_idx - start_idx)
            else:
                h[start_idx:end_idx] = np.linspace(1, 0, end_idx - start_idx)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Optimization loop with alternating updates
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    update_frequency = max(1, n_points // 10)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Identify critical points for update
        # Use gradient estimation or peak detection for crucial coordinates
        gradients = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            gradients[i] = (max_plus - max_minus) / 0.02
            
        # Pick top coordinates to update
        update_indices = np.argsort(np.abs(gradients))[-update_frequency:][::-1]
        
        improved = False
        # Update selection with momentum
        for i in update_indices:
            # Gradient-based update
            step_size = 0.01
            if gradients[i] < 0:
                # Reduce h[i] to decrease overlap
                new_val = max(0.0, h[i] - step_size)
            else:
                # Increase h[i] to reduce overlap
                new_val = min(1.0, h[i] + step_size)
            
            h_new = h.copy()
            h_new[i] = new_val
            
            # Maintain sum constraint
            h_new = normalize_h(h_new, target_sum)
            h_new = np.clip(h_new, 0, 1)
            
            # Check improvement
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_max:

                improved = True
                break
                
        if not improved:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:

                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
        iter_count += 1

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
