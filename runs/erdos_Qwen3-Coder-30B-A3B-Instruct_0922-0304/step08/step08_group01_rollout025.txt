import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function for scipy.optimize
    def objective(h):
        return compute_c5(h, dx)
    
    # Optimization with SLSQP
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9},
            callback=lambda x: None  # Placeholder for future use
        )
        
        if result.success:
            h_opt = result.x
            # Ensure bounds are respected
            h_opt = np.clip(h_opt, 0, 1)
            # Re-normalize to correct sum
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            new_c5 = compute_c5(h_opt, dx)
            if new_c5 < best_c5:
                best_h = h_opt
                best_c5 = new_c5
    except Exception:
        pass  # Continue with original best_h if optimization fails
    
    # Refinement with local search
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Random perturbation
        idxs = np.random.choice(n_points, size=n_points//10, replace=False)
        for i in idxs:
            old_val = best_h[i]
            # Small random perturbation
            pert = np.random.normal(0, 0.01)
            best_h[i] = old_val + pert
            best_h[i] = np.clip(best_h[i], 0, 1)
            
        # Re-normalize
        best_h = best_h * (target_sum / np.sum(best_h))
        
        # Evaluate
        new_c5 = compute_c5(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
