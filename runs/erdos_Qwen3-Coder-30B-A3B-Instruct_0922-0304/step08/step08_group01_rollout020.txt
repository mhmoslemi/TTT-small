import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb(h, n_points):
    """Apply a random perturbation to h."""
    op = np.random.choice(['flip', 'adjust', 'shuffle'], p=[0.5, 0.3, 0.2])
    
    if op == 'flip':
        i = np.random.randint(n_points)
        h[i] = 1 - h[i]
    elif op == 'adjust':
        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h[i] += delta
    elif op == 'shuffle':
        length = np.random.randint(1, 5)
        i = np.random.randint(n_points - length + 1)
        segment = h[i:i+length]
        np.random.shuffle(segment)
        h[i:i+length] = segment

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simulated Annealing parameters
    start_temp = 0.1
    cooling_rate = 0.999
    current_temp = start_temp
    max_iterations = 100000
    
    iter_count = 0
    improvement_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        # Store current state
        old_h = h.copy()
        old_c5 = best_c5
        
        # Apply perturbation
        perturb(h, n_points)
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        
        # Accept or reject the new solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            improvement_count += 1
            last_improvement = iter_count
        else:
            # Accept worse solution with probability
            delta_e = new_c5 - old_c5
            if np.exp(-delta_e / current_temp) > np.random.rand():
                best_h = h.copy()
                best_c5 = new_c5
            else:
                # Revert to old state
                h = old_h.copy()
        
        # Update temperature
        current_temp *= cooling_rate
        
        # Reheat if no improvement in a while
        if iter_count - last_improvement > 1000:
            current_temp = max(0.01, current_temp * 2)
            last_improvement = iter_count
            
        iter_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
