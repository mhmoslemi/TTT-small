import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.8
    c1 = c2 = 2.0
    inertia_min = 0.4
    inertia_max = 0.9
    velocity_max = 0.05
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.uniform(-0.1, 0.1, n_points)
        particle = np.clip(particle, 0, 1)
        current_sum = np.sum(particle)
        if current_sum > 0:
            particle = particle * (target_sum / current_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_fitness.append(evaluate_overlap(particle, dx))
    
    # Global best
    best_idx = np.argmin(p_best_fitness)
    g_best = p_best[best_idx].copy()
    g_best_fitness = p_best_fitness[best_idx]
    
    # PSO loop
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time < budget_s - 10):
        # Update inertia weight
        inertia = inertia_max - (inertia_max - inertia_min) * (iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize
            current_sum = np.sum(particles[i])
            if current_sum > 0:
                particles[i] = particles[i] * (target_sum / current_sum)
            
            # Evaluate fitness
            fitness = evaluate_overlap(particles[i], dx)
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = fitness
                    last_improvement = iteration
        
        iteration += 1
        
        # Shaking if no improvement for 50 iterations
        if iteration - last_improvement > 50:
            for i in range(n_particles):
                if np.random.random() < 0.3:  # 30% chance to shake
                    particles[i] += np.random.normal(0, 0.02, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    current_sum = np.sum(particles[i])
                    if current_sum > 0:
                        particles[i] = particles[i] * (target_sum / current_sum)
    
    # Local refinement with SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Test with the global best
    result = minimize(objective, g_best, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < g_best_fitness:
            g_best = final_h
            g_best_fitness = final_c5
    
    # Final validation
    final_h = np.clip(g_best, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
