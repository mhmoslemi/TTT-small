import numpy as np
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1], scale to target sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_surrogate(H):
    """Compute surrogate objective from frequency domain."""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, freq_indices, learning_rate):
    """Update H in specified frequency bins."""
    H_new = H.copy()
    for i in freq_indices:
        H_new[i] -= learning_rate * np.real(H_new[i])
    return H_new

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, target_sum)
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Fourier domain approach
        H = np.fft.fft(h_current)
        surrogate = compute_surrogate(H)
        
        # Identify bins with highest contribution to C5
        # Use the real part of the inverse FFT of surrogate as proxy
        # for the most influential frequency components
        try:
            # Check the frequencies in the forward half
            freq_mags = np.abs(surrogate[:n_points]) 
            # Find indices of highest contributions
            top_freqs = np.argsort(freq_mags)[-5:]  # Take top 5 bins
            # Update only those bins
            H_updated = spectral_update(H, top_freqs, learning_rate)
            
            # Reconstruct and project
            h_updated = np.real(np.fft.ifft(H_updated))
            h_updated = project_h(h_updated, target_sum)
            
            # Evaluate
            c5_new = compute_c5(h_updated, dx)
            if c5_new < best_c5:
                best_h = h_updated.copy()
                best_c5 = c5_new
                
            h_current = h_updated.copy()
            
            # Adapt learning rate
            if iteration % 100 == 0 and iteration > 0:
                learning_rate *= 0.5
                
        except Exception:
            # Fall back to simple gradient descent if FFT fails
            h_temp = h_current.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
            h_temp = project_h(h_temp, target_sum)
            c5_temp = compute_c5(h_temp, dx)
            if c5_temp < compute_c5(h_current, dx):
                h_current = h_temp.copy()
                
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
