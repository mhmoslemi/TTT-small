import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Set up the convex optimization problem
    h_vars = cp.Variable(n_points)
    t = cp.Variable()

    # Objective
    objective = cp.Minimize(t)

    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_vars) == n_points / 2)
    
    # Bound constraints
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    
    # Overlap constraints for all shifts
    for k in range(1 - n_points, n_points):
        shift_sum = 0
        for i in range(n_points):
            j = (i + k) % n_points
            shift_sum += h_vars[i] * (1 - h_vars[j])
        constraints.append(shift_sum <= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_vars.value
            c5_bound = t.value
        else:
            # If optimization failed, return the original h
            h_opt = h.copy()
            c5_bound = evaluate_c5(h_opt, dx)
    except:
        # If optimization fails, return the original h
        h_opt = h.copy()
        c5_bound = evaluate_c5(h_opt, dx)
    
    # Final verification and return
    final_c5 = evaluate_c5(h_opt, dx)
    if final_c5 < c5_bound:
        c5_bound = final_c5
    
    return (h_opt.tolist(), c5_bound, n_points)
