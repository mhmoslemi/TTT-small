import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    sigma = 0.05
    tau = 1000
    
    # Initialize
    best_h = h_current.copy()
    current_c5 = compute_c5(h_current, dx)
    best_c5 = current_c5
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        current_c5 = compute_c5(h_current, dx)
        
        # Propose new state
        h_proposed = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, sigma)
        h_proposed[idx] = np.clip(h_proposed[idx] + delta, 0, 1)
        
        # Maintain sum constraint
        old_sum = np.sum(h_current)
        new_sum = np.sum(h_proposed)
        scale_factor = old_sum / new_sum
        h_proposed = h_proposed * scale_factor
        
        # Compute new C5
        proposed_c5 = compute_c5(h_proposed, dx)
        
        # Accept or reject
        delta_c5 = proposed_c5 - current_c5
        if delta_c5 < 0 or np.random.random() < exp(-delta_c5 / T):
            h_current = h_proposed.copy()
            current_c5 = proposed_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (iteration / tau)))
        
        iteration += 1
        
        # Early stopping if we've achieved target
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
