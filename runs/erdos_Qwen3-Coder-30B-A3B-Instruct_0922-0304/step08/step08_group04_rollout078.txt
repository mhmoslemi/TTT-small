import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Solve as a convex QP
    h_var = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Bound constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # Sum over i: h[i] * (1 - h[i+k])
            expr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            # Sum over i: h[i] * (1 - h[i+k])
            expr = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:n_points+k]))
        
        constraints.append(expr <= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_bound = t.value
        else:
            # Fallback to initial solution
            h_opt = h
            c5_bound = evaluate_c5(h, dx)
    except:
        # Fallback to initial solution
        h_opt = h
        c5_bound = evaluate_c5(h, dx)
    
    # Final verification
    final_c5 = evaluate_c5(h_opt, dx)
    if final_c5 < c5_bound:
        c5_bound = final_c5
    
    return (h_opt.tolist(), float(c5_bound), n_points)
