import numpy as np
from scipy.signal import correlate
import time
from math import exp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Initialize with a better starting point
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1.0
        # Add some randomization
        for i in range(n_points//10):
            idx = np.random.randint(n_points)
            h_current[idx] = np.random.rand()
        h_current = h_current * (target_sum / np.sum(h_current))
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    sigma = 0.05
    N_steps = 500
    patience = 0
    last_improvement = 0
    
    T = T0
    t = 0
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, sigma)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Renormalize to satisfy constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute C5 values
        c5_current = compute_c5(h_current)
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < exp(-delta_c5 / T):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
                last_improvement = t
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (t / 1000)))
        
        # Adaptive adjustments
        if patience > N_steps:
            sigma *= 0.9
            patience = 0
            if T < T_min:
                T = T0 * 0.1
                
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
