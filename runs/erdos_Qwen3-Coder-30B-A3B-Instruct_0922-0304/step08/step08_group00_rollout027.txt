import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use cvxpy to solve the convex optimization problem
    # Variables
    h_vars = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == target_sum
    ]
    
    # Objective - We want to minimize the maximum correlation
    # This is a bit tricky to formulate exactly convexly, so we'll do it
    # iteratively with a max approach or use an iterative SDP approximation
    # For now we'll use a simpler but also intentional iterative refinement
    
    # Simple grid search refinement
    best_c5 = evaluate_overlap(best_h, dx)
    
    # Local refinement using SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints_eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    # Run a local search on the best solution
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints_eq, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Fine-tune with a more structured search
    # We'll run a few iterations of gradient-based optimization starting from
    # the initial solution to escape any local minima
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        # Random perturbation
        perturbation = np.random.normal(0, 0.01, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        current_sum = np.sum(perturbed_h)
        if current_sum > 0:
            perturbed_h = perturbed_h * (target_sum / current_sum)
        
        # Local optimization
        result = minimize(objective, perturbed_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints_eq, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            optimized_h = np.clip(result.x, 0, 1)
            optimized_sum = np.sum(optimized_h)
            if optimized_sum > 0:
                optimized_h = optimized_h * (target_sum / optimized_sum)
            optimized_c5 = evaluate_overlap(optimized_h, dx)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
