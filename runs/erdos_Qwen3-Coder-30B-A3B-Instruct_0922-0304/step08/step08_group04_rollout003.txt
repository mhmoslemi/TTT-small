import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # Create QP variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints
    for k in range(1 - n_points, n_points):
        # Build the constraint for shift k
        expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h_var[i] * (1 - h_var[j])
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, reltol=1e-6)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            opt_h = h_var.value
            opt_t = t.value
            # Ensure constraints are met
            opt_h = np.clip(opt_h, 0, 1)
            h_sum = np.sum(opt_h)
            if h_sum > 0:
                opt_h = opt_h * (n_points / 2) / h_sum
            else:
                opt_h = h
        else:
            # Fallback to the best solution we had
            opt_h = h
            opt_t = evaluate_c5(h, dx)
    except Exception as e:
        # Fallback to simple optimization if QP fails
        opt_h = h
        opt_t = evaluate_c5(h, dx)
    
    # Final verification
    final_c5 = evaluate_c5(opt_h, dx)
    
    # Return the best solution found
    return (opt_h.tolist(), final_c5, n_points)
