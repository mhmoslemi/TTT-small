import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def clip_and_normalize_h(h, target_sum):
    """Clip h to [0,1] and normalize to target_sum."""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def tabu_search_erdos(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []  # List of recent solutions (hashed)
    tabu_tenure = 100
    temperature = 1.0
    cooling_rate = 0.999
    aspiration_tolerance = 1e-10
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Iteration tracking
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate neighborhood by making small changes to current solution
        neighbors = []
        
        # Generate 10 candidates using local moves
        for _ in range(10):
            h_new = h.copy()
            
            # Randomly select type of move
            move_type = random.random()
            
            if move_type < 0.7:
                # Swap two random positions
                i, j = random.sample(range(n_points), 2)
                h_new[i], h_new[j] = h_new[j], h_new[i]
            else:
                # Flip two random positions with noise
                i, j = random.sample(range(n_points), 2)
                h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
                h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
            
            # Ensure feasibility and normalize
            h_new = clip_and_normalize_h(h_new, target_sum)
            
            # Compute overlap for this candidate
            c5 = compute_c5_exact(h_new, dx)
            neighbors.append((h_new, c5))
        
        # Find best non-tabu neighbor
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor_h, neighbor_c5 in neighbors:
            # Hash the neighbor to store in tabu list
            h_hash = tuple(np.round(neighbor_h, decimals=5))
            
            if h_hash not in tabu_list or neighbor_c5 < best_c5 - aspiration_tolerance:
                if neighbor_c5 < best_neighbor_c5:
                    best_neighbor = neighbor_h
                    best_neighbor_c5 = neighbor_c5
        
        # Update tabu list and solution
        if best_neighbor is not None:
            # Add current solution to tabu list (hash it)
            current_h_hash = tuple(np.round(h, decimals=5))
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(current_h_hash)
            
            # Update solution
            h = best_neighbor
            if best_neighbor_c5 < best_c5:
                best_h = h.copy()
                best_c5 = best_neighbor_c5
                last_improvement = 0
            else:
                last_improvement += 1
        else:
            last_improvement += 1
        
        # Update temperature
        temperature *= cooling_rate
        
        iteration += 1
        
        # Early termination if we've found a good enough solution
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return tabu_search_erdos(seed, budget_s)
