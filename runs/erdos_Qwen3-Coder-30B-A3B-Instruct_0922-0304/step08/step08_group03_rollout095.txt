import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    # Not exact but a reasonable approximation
    # Gradient is derived from influence of each h_i on the overlap
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h - from initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Parameters for the evolutionary algorithm
    population_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_generations = 2000
    
    # Population initialization
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = h + np.random.normal(0, 0.02, n_points)
        h_mut = repair_h(h_mut, target_sum)
        population.append(h_mut)
    
    no_improvement_count = 0
    best_overall_c5 = best_c5
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append((c5, ind))
        
        # Sort by fitness (lower is better)
        fitness.sort(key=lambda x: x[0])
        best_in_gen = fitness[0][0]
        
        if best_in_gen < best_overall_c5:
            best_overall_c5 = best_in_gen
            best_h = fitness[0][1].copy()
            
        if best_overall_c5 <= 0.3807:
            break
            
        # Selection: take top half
        selected = [ind for _, ind in fitness[:population_size//2]]
        
        # Create new population via crossover and mutation
        new_population = [best_h.copy()]
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                # Mutation
                parent = np.random.choice(selected)
                child = parent + np.random.normal(0, 0.02, n_points)
            
            child = repair_h(child, target_sum)
            # Add small gradient-based refinement
            grad = compute_gradient(child)
            step_size = 0.01
            child = child - step_size * grad
            child = repair_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
        
        # Adaptive parameters
        if best_in_gen >= best_overall_c5:
            no_improvement_count += 1
            if no_improvement_count > 100:
                mutation_rate = min(0.5, mutation_rate * 1.1)
        else:
            no_improvement_count = 0
            mutation_rate = max(0.02, mutation_rate * 0.95)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
