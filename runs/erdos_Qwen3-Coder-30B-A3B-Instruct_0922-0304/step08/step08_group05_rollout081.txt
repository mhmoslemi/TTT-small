import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def flip_elements(h, indices):
    """Flip selected elements (multiply by 0.97)"""
    h_copy = h.copy()
    for i in indices:
        h_copy[i] *= 0.97
    return h_copy

def swap_elements(h, i, j):
    """Swap two elements"""
    h_copy = h.copy()
    h_copy[i], h_copy[j] = h_copy[j], h_copy[i]
    return h_copy

def generate_neighborhood(h, n_candidates=10, n_points=70):
    """Generate neighborhood solutions"""
    candidates = []
    for _ in range(n_candidates):
        h_candidate = h.copy()
        # Randomly decide flip or swap
        if np.random.random() < 0.3:
            # Flip 2 random points
            indices = np.random.choice(n_points, 2, replace=False)
            h_candidate = flip_elements(h_candidate, indices)
        else:
            # Swap 2 random points
            i, j = np.random.choice(n_points, 2, replace=False)
            h_candidate = swap_elements(h_candidate, i, j)
        candidates.append(h_candidate)
    return candidates

def enforce_constraints(h, target_sum):
    """Enforce box constraints and sum constraint"""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h
