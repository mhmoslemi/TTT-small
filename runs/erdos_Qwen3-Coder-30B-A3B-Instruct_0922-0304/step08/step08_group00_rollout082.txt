import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a base function
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    def evaluate_overlap(h_vec):
        # Compute overlap integral for all shifts
        h = np.array(h_vec)
        # We need to compute: max_k integral of h(x)(1 - h(x+k))dx
        # Using correlation with flipped function
        # This computes: sum_{i,k} h[i] * (1 - h[(i+k)%n]) 
        # We are doing full convolution with flipped h
        n = len(h)
        # Creating the flipped version
        flipped_h = 1 - h
        # Use numpy.correlate with full mode
        correlations = np.correlate(h, flipped_h, mode="full")
        
        # The indices in the full correlation result correspond to shifts
        # We want k = 0 to n-1, where k is the shift, and the correlation
        # at index k corresponds to shift k (at least for positive shifts)
        # The overlap integral for shift k is correlations[n-1+k] * dx
        max_overlap = -1e30
        for k in range(n):
            overlap = correlations[n - 1 + k] * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Objective function for optimization
    def objective(c):
        # This is the representation of the optimization problem
        # We're minimizing the maximum overlap, but the constraint is that the
        # overlap should be bounded by c
        # This can be done by optimizing the actual overlap function directly
        return evaluate_overlap(h)
    
    # Constraint ensuring sum(h) = 1  
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for optimization
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try SLSQP to optimize the overlaps with the bounds
    maxiter = 2000
    iter_count = 0
    best_c5 = evaluate_overlap(h)
    best_h = h.copy()
    
    # Small optimization step: try local descent from the initial solution
    # First, optimize with bounds only  
    try:
        result = minimize(objective, h.copy(), method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': 1e-10, 'gtol': 1e-10})
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            h_sum = np.sum(h_opt)
            if h_sum > 0:
                h_opt = h_opt * (target_sum / h_sum)
            c5_opt = evaluate_overlap(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception as e:
        pass
    
    # If no improvement, make very small perturbations to see if we 
    # can come slightly closer to the theoretical minimum.
    for _ in range(10):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_pert = h.copy() + np.random.normal(0, 1e-4, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_sum = np.sum(h_pert)
        if h_sum > 0:
            h_pert = h_pert * (target_sum / h_sum)
        c5_pert = evaluate_overlap(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert
    
    # Final refinement using scipy optimizer if still under time limit
    if time.time() - start_time < budget_s - 10:
        # Final simplified approach, direct optimization with more control
        try:
            # More constrained optimization to avoid local minima
            result = minimize(objective, best_h, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12})
            if result.success:
                h_final = np.clip(result.x, 0, 1)
                h_sum = np.sum(h_final)
                if h_sum > 0:
                    h_final = h_final * (target_sum / h_sum)
                c5_final = evaluate_overlap(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass

    return (best_h.tolist(), best_c5, n_points)
