import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighbors(h, n_neighbors=10, n_points=70):
    """Generate neighbor solutions."""
    neighbors = []
    for _ in range(n_neighbors):
        h_new = h.copy()
        # 30% chance of flipping, 70% chance of swapping
        if np.random.random() < 0.3:
            # Flip two random points
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new[i] *= 0.97 + np.random.normal(0, 0.01)
            h_new[j] *= 0.97 + np.random.normal(0, 0.01)
        else:
            # Swap two random points
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2)
        neighbors.append(h_new)
    
    return neighbors

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Random initialization with slight perturbation
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Tabu search setup
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    current_temp = 1.0
    alpha = 0.999
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Generate neighbors
        neighbors = generate_neighbors(h, 10, n_points)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor in neighbors:
            neighbor_c5 = compute_c5_exact(neighbor, dx)
            # Check if neighbor is not tabu or if it's better (aspiration)
            if tuple(neighbor) not in tabu_list or neighbor_c5 < best_c5:
                if neighbor_c5 < best_neighbor_c5:
                    best_neighbor_c5 = neighbor_c5
                    best_neighbor = neighbor
        
        if best_neighbor is not None:
            # Accept the neighbor
            h = best_neighbor
            if best_neighbor_c5 < best_c5:
                best_c5 = best_neighbor_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Add current state to tabu list with tenure
            tabu_list.append(tuple(h))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Update temperature
            current_temp *= alpha
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
