import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

                h[i] = i / mid
            else:
                h[i] = 1 - (i - mid) / (n_points - mid)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

        return (best_h.tolist(), best_c5, n_points)

        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute accurate gradient for each coordinate using finite differences
        # But apply smarter steps to prevent overshooting
        gradient = np.zeros(n_points)

            h_plus = h.copy()

            if delta > 0:

                h_plus = normalize_h(h_plus, target_sum)
                h_plus = np.clip(h_plus, 0, 1)
                max_plus = np.max(correlate(h_plus, 1 - h_plus, mode="full") * dx)
            else:
                max_plus = np.inf
            
            # Perturb backward
            delta = min(eps, h[i])
            if delta > 0:
                h_minus[i] = h[i] - delta
                h_minus = normalize_h(h_minus, target_sum)
                h_minus = np.clip(h_minus, 0, 1)
                max_minus = np.max(correlate(h_minus, 1 - h_minus, mode="full") * dx)
            else:
                max_minus = np.inf

            # Estimate gradient
            if max_plus < np.inf and max_minus < np.inf:
                gradient[i] = (max_plus - max_minus) / (2 * delta)
            else:
                gradient[i] = 0.0

        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)

        lr = 0.01

        h_new = h - lr * gradient
