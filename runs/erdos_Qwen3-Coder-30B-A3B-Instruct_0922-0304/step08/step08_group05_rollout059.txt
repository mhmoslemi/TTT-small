import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_constraints(h, target_sum):
    """Repair constraints by clipping and normalizing."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def generate_neighborhood(h, target_sum, n_candidates=10):
    """Generate neighborhood by flipping and swapping."""
    n = len(h)
    candidates = []
    
    for _ in range(n_candidates):
        h_new = h.copy()
        
        # Randomly choose between flip or swap
        if np.random.rand() < 0.3:
            # Flip two points
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
            h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
        else:
            # Swap two points
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Apply constraints
        h_new = repair_constraints(h_new, target_sum)
        candidates.append(h_new)
    
    return candidates

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70

    h = repair_constraints(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Cooling schedule
    T = 1.0
    alpha = 0.999
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < max_stagnation:
        # Generate neighborhood
        candidates = generate_neighborhood(h, target_sum)
        
        # Evaluate each candidate
        candidate_c5 = [compute_c5_exact(c, dx) for c in candidates]
        
        # Find best non-tabu candidate
        best_candidate_idx = -1
        best_candidate_c5 = float('inf')
        
        for i, (c, c5) in enumerate(zip(candidates, candidate_c5)):
            # Check if candidate is tabu
            if tuple(c) in tabu_list:
                # Allow if it's better than best
                if c5 < best_c5:
                    best_candidate_idx = i
                    best_candidate_c5 = c5
            else:
                if c5 < best_candidate_c5:
                    best_candidate_idx = i
                    best_candidate_c5 = c5
        
        if best_candidate_idx == -1:
            # No valid candidates found, try accepting any
            for i, (c, c5) in enumerate(zip(candidates, candidate_c5)):
                if c5 < best_candidate_c5:
                    best_candidate_idx = i
                    best_candidate_c5 = c5
        
        if best_candidate_idx != -1:
            h = candidates[best_candidate_idx]
            
            # Update tabu list
            tabu_list.append(tuple(h))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Update best solution
            if best_candidate_c5 < best_c5:

                best_h = h.copy()
                stagnation_count = 0
            else:

        else:
            stagnation_count += 1
            
        # Cooling
        T *= alpha
        
        # Early exit if good enough
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5
