import numpy as np
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to sum=target_sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    # Using scipy's correlate for reference, but we'll optimize in frequency domain
    from scipy.signal import correlate
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain"""
    H_conj = np.conj(H)
    # H * (1 - H_conj) and take magnitude
    prod = H * (1 - H_conj)
    return np.abs(prod)

def spectral_update(H, learning_rate, high_freq_mask):
    """Update H to reduce high frequency contributions"""
    # Modify frequency bins based on mask
    H_new = H.copy()
    H_new[high_freq_mask] -= learning_rate * H[high_freq_mask] / (1 + np.abs(H[high_freq_mask])**2)
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, target_sum)
    
    # Initialize tracking variables
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # For frequency domain approach
    learning_rate = 0.01
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0 and iteration > 0:
            learning_rate *= 0.95
        
        # FFT approach
        H_current = fft(h_current)
        surrogate = compute_surrogate(H_current)
        
        # Identify high-frequency components that contribute most to C5
        high_freq_idx = np.argsort(surrogate)[-n_points//4:]  # Top 1/4 frequencies
        
        # Update in frequency domain
        H_new = spectral_update(H_current, learning_rate, high_freq_idx)
        
        # Inverse transform and project back to feasible region
        h_new = np.real(ifft(H_new))
        h_new = project_h(h_new, target_sum)
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current for next iteration
        h_current = h_new.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with local search for robustness
    h_refined = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
