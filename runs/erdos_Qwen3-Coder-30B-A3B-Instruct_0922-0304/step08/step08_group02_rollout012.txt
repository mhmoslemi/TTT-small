import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create initial point with some variation
        h_current = np.zeros(n_points)
        for i in range(n_points // 2):
            h_current[i] = 1.0
        h_current = np.random.permutation(h_current)
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    tau = 1000
    no_improve_steps = 0
    max_no_improve = 200

    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        
        # Perturb one element
        delta = np.random.normal(0, sigma)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Maintain sum constraint
        old_sum = np.sum(h_current)
        new_sum = np.sum(h_new)
        scale_factor = old_sum / new_sum
        h_new = h_new * scale_factor
        
        # Ensure exact sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        c5_current = compute_c5(h_current, dx)
        
        delta_c5 = c5_new - c5_current
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < exp(-delta_c5 / T):
            h_current = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve_steps = 0
            else:
                no_improve_steps += 1
        else:
            no_improve_steps += 1
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (iteration / tau)))
            
        # Adaptive adjustments
        if no_improve_steps > max_no_improve:
            sigma *= 0.9
            no_improve_steps = 0
            if T < T_min * 10:
                T = T0
                
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
