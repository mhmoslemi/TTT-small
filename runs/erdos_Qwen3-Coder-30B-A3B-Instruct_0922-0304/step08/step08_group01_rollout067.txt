import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed):
    """Round relaxed solution to binary with sum constraint."""
    h_binary = (h_relaxed >= np.median(h_relaxed)).astype(float)
    current_sum = np.sum(h_binary)
    target_sum = len(h_binary) / 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        diff = int(current_sum - target_sum)
        # Flip the smallest values
        indices = np.argsort(h_relaxed)[:diff]
        h_binary[indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        diff = int(target_sum - current_sum)
        # Flip the largest values that are 0
        indices = np.argsort(h_relaxed)[::-1][:diff]
        h_binary[indices] = 1
    
    return h_binary

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try convex relaxation approach
    attempts = 0
    max_attempts = 10
    
    while time.time() - start_time < budget_s - 1 and attempts < max_attempts:
        # Relaxation
        h_relaxed = h.copy()
        
        # Create the optimization variables
        h_var = cp.Variable(n_points, lower_bound=0, upper_bound=1)
        
        # Constraint: sum = target_sum
        sum_constraint = cp.Equality(cp.sum(h_var), target_sum)
        
        # Objective: minimize z
        z = cp.Variable()
        
        # Add overlap constraints
        constraints = [sum_constraint, z >= 0]
        
        # Precompute full overlap with relaxed solution
        overlap_vals = []
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                overlap_val = cp.sum(cp.multiply(h_var[:n_points-k], 1-h_var[k:]))
            else:
                overlap_val = cp.sum(cp.multiply(h_var[-k:], 1-h_var[:n_points+k]))
            overlap_vals.append(overlap_val)
            constraints.append(overlap_val <= z)
        
        # Formulate and solve the QP
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=100)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_relaxed = h_var.value
                if h_relaxed is not None:
                    h_relaxed = np.clip(h_relaxed, 0, 1)
                    h_binary = round_to_binary(h_relaxed)
                    
                    # Compute new C5
                    new_c5 = compute_c5(h_binary, dx)
                    if new_c5 < best_c5:
                        best_h = h_binary.copy()
                        best_c5 = new_c5
                        
                    # For next iteration, start with binary
                    h = h_binary.copy()
                else:
                    # If no good solution from solver, proceed with regular method
                    pass
            else:
                # If solver fails, use current h
                pass
        except Exception:
            # If solver crashes, fall back to standard method

            grad = np.zeros(n_points)
            eps = 1e-5

                h_plus[j] += eps

                # Gradient is approximated by the effect on max overlap
                grad[j] = np.mean(diff)
                
            # Update with gradient descent step
            step_size = 0.001
            h_new = h - step_size * grad
            
            # Ensure bound constraints
            h_new = np.clip(h_new, 0, 1)

            # Stop early if target reached

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
