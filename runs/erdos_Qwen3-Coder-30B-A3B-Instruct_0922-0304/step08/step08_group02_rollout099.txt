```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of possible shifts"""
    # We can have shifts from -(n-1) to (n-1), but we only care about positive shifts for correlation
    # For n points, we have 2*n-1 possible shifts in full correlation mode
    # Sample between 0 and 2*n-2
    shifts = np.random.choice(2*n-1, sample_size, replace=False)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    # Ensure initial h is feasible
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    target_overlap = 0.38085  # Start with known bound
    
    # Initialize progress tracking
    prev_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Sample random shift indices
        if iteration < 100:
            sample_size = 50  # Sample more early on
        else:
            sample_size = 30  # Then sample less
        
        shifts = sample_shifts(n_points, sample_size)
        
        # Create the QP constraint matrix
        # The constraints are: (h * (1 - h_shift))_i <= target_overlap for i in shifts
        # We're trying to minimize max(h * (1-h_shift)) over shifts
        
        # Simplification: instead of constraint matrix, we model the maximum overlap
        h_refined = h_current.copy()
        
        # Try different perturbation strategies 
        if iteration % 5 == 0:
            # Do a more aggressive search
            for _ in range(20):
                if time.time() - start_time > budget_s - 1:
                    break
                idx = np.random.randint(n_points)
                delta = np.random.normal(0, 0.05)
                h_refined[idx] = np.clip(h_refined[idx] + delta, 0, 1)
        else:
            # Random perturbations
            for _ in range(10):
                if time.time() - start_time > budget_s - 1:
                    break
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.02, 0.02)
                h_refined[idx] = np.clip(h_refined[idx] + delta, 0, 1)
        
        h_refined = project_h(h_refined, n_points)
        c5_new = compute_c5(h_refined, dx)
        
        if c5_new < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_new
            no_improvement_count = 0
            prev_c5 = best_c5
        else:
            no_improvement_count += 1
            
        # Keep trying smaller steps for final refinement  
        h_current = h_refined.copy()
        iteration += 1
        
        # Adapt target overlap based on progress
        if iteration % 10 == 0:
            if abs(prev_c5 - best_c5) < 1e-5:
                target_overlap *= 0.99
            else:
                target_overlap = max(target_overlap * 0.999, best_c5)
        
        # Early stopping if good enough
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_final.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_final, dx):
            h_final = h_temp.copy()
            
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
```
