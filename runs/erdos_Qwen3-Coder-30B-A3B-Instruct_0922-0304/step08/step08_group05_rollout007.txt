import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_constraints(h, target_sum):
    """Repair constraints: clip and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def tabu_search(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = repair_constraints(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    last_improvement = 0
    temperature = 1.0
    cooling_rate = 0.999
    
    # For generating neighborhood
    max_candidates = 10
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate neighborhood
        candidates = []
        for _ in range(max_candidates):
            candidate = h.copy()
            
            # Randomly choose neighborhood move
            if np.random.random() < 0.3:
                # Flip two random points
                i, j = np.random.choice(n_points, 2, replace=False)
                candidate[i] *= 0.97 + np.random.normal(0, 0.01)
                candidate[j] *= 0.97 + np.random.normal(0, 0.01)
            else:
                # Swap two random points
                i, j = np.random.choice(n_points, 2, replace=False)
                candidate[i], candidate[j] = candidate[j], candidate[i]
            
            # Apply constraints
            candidate = repair_constraints(candidate, target_sum)
            c5 = compute_c5_exact(candidate, dx)
            candidates.append((candidate, c5))
        
        # Find best non-tabu candidate
        best_candidate = None
        best_c5_candidate = float('inf')
        
        for candidate, c5 in candidates:
            # Check if candidate is in tabu list
            is_tabu = any(np.array_equal(candidate, tabu_state) for tabu_state in tabu_list)
            
            # Aspiration criteria
            if c5 < best_c5 or not is_tabu:
                if c5 < best_c5_candidate:
                    best_c5_candidate = c5
                    best_candidate = candidate
        
        # Update
        if best_candidate is not None:
            h = best_candidate
            current_c5 = best_c5_candidate
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
            # Update tabu list
            tabu_list.append(h.copy())
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
                
            # Cooling
            temperature *= cooling_rate
            
            # Early stopping
            if best_c5 <= 0.3807:
                break
        else:
            # No good candidate found, continue
            last_improvement += 1
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = tabu_search(seed, budget_s)
    n_points = 70
    return (best_h.tolist(), best_c5, n_points)
