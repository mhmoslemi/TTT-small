import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Generate initial feasible solution
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1.0
        # Add some randomness
        for i in range(n_points//10):
            idx = np.random.randint(0, n_points)
            h_current[idx] = np.random.rand()
        h_current = h_current * (target_sum / np.sum(h_current))
    
    # Temperature schedule parameters
    T0 = 0.1
    alpha = 0.999
    T_min = 1e-6
    tau = 1000
    sigma = 0.05
    n_steps_no_improvement = 0
    max_no_improvement = 500
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    T = T0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        c5_current = compute_c5(h_current, dx)
        
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, sigma)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        h_new = h_new * (target_sum / current_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / max(T, T_min)):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
                n_steps_no_improvement = 0
            else:
                n_steps_no_improvement += 1
        else:
            n_steps_no_improvement += 1
        
        # Update temperature
        T = max(T_min, T0 * (alpha ** (iteration / tau)))
        
        # Adaptive perturbation scaling
        if n_steps_no_improvement > max_no_improvement:
            sigma *= 0.9
            n_steps_no_improvement = 0
            if T < T_min:
                T = T0
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(best_h, dx):
            best_h = h_temp.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
