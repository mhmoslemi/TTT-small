import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Ensure h sums to target_sum and is clipped to [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute approximate gradient of C5 with respect to h."""
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from either global or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Hybrid GA + gradient descent approach
    population_size = 50
    max_generations = 2000
    mutation_rate = 0.1
    crossover_prob = 0.7
    
    # Initialize population
    population = [repair_h(np.random.rand(n_points), target_sum) for _ in range(population_size)]
    if 'initial_h_values' in globals():
        population[0] = np.array(initial_h_values)
        population[0] = repair_h(population[0], target_sum)
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    for generation in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate population
        fitness = [compute_c5(individual, dx) for individual in population]
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        selected = population[:population_size // 2]
        
        # Crossover and mutation
        new_population = [selected[0].copy()]  # Elitism
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_prob and len(selected) > 1:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = np.random.choice(selected, 1)[0].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in mutate_indices:
                    # Small random perturbation
                    perturb = np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i] + perturb, 0, 1)
                child = repair_h(child, target_sum)
            
            # Local gradient step to refine
            if np.random.rand() < 0.3:  # 30% chance of local refinement
                grad = compute_gradient(child, n_points)
                # Take step in descent direction
                step_size = 0.01 / (1 + np.linalg.norm(grad) * 10)
                child -= step_size * grad
                child = repair_h(child, target_sum)
                
            new_population.append(child)
            
        population = new_population
        
        # Adaptive parameters
        if fitness[0] < best_c5:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                mutation_rate = min(0.5, mutation_rate * 1.1)
                crossover_prob = max(0.4, crossover_prob * 0.9)
                no_improvement_count = 0
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
