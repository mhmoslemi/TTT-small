import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    # Compute correlation for all shifts (full mode)
    overlap = np.correlate(h, 1 - h, mode="full")
    # Extract the overlapping parts (shifts from -n+1 to n-1)
    # The valid overlap region corresponds to shifts from 0 to 2*n-2
    # The overlaps we care about are from shift 0 to 2*n-2, but we want shift from 0 to n-1
    # Actually, for shift k in [0, n-1], range of indices for correlation is [k, n-1] and [0, n-1-k]
    # So we take overlapping part for all k. Since we want maximum, take max of all k
    # Correlation of length 2n-1, we want elements at indices n-1 to 2n-2 for shift in [0, n-1]
    # So we take elements from index n-1:2n-1
    max_overlap = np.max(overlap[n-1:2*n-1])  # the overlap for shifts 0 to n-1
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization method
    max_iterations = 100
    improvement_threshold = 1e-6
    perturbation_delta = 1e-3
    decay_factor = 0.98
    
    for it in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Perturb the current solution slightly
        perturbed_h = h + np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Renormalize
        current_sum = np.sum(perturbed_h)
        if current_sum > 0:
            perturbed_h = perturbed_h * (target_sum / current_sum)
        h = perturbed_h
        
        # Partition the variables into even and odd indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Alternating optimization loop
        old_c5 = best_c5 + 1
        for cycle in range(5):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Optimize even indices fixing odd ones
            def objective_even(even_vals):
                temp_h = h.copy()
                temp_h[even_indices] = even_vals
                return evaluate_overlap(temp_h, dx)
            
            bounds = [(0, 1) for _ in even_indices]
            # Constraint for sum: even_vals + odd_vals = target_sum
            # Or equivalently, sum(even_vals) = target_sum - sum(odd_vals)
            # Since we're holding the odd values fixed, use:
            # sum(even_vals) = target_sum - sum(h[odd_indices])
            constraint_value = target_sum - np.sum(h[odd_indices])
            
            def constraint_fun(even_vals):
                return np.sum(even_vals) - constraint_value
            
            constraints = [{'type': 'eq', 'fun': constraint_fun}]
            
            result_even = minimize(objective_even, h[even_indices], method='SLSQP',
                                   bounds=bounds, constraints=constraints,
                                   options={'ftol': 1e-10, 'maxiter': 100})
            
            if result_even.success:
                h[even_indices] = np.clip(result_even.x, 0, 1)
                current_sum = np.sum(h)
                if current_sum > 0:
                    h = h * (target_sum / current_sum)
                current_c5 = evaluate_overlap(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
            # Optimize odd indices fixing even ones
            def objective_odd(odd_vals):
                temp_h = h.copy()
                temp_h[odd_indices] = odd_vals
                return evaluate_overlap(temp_h, dx)
            
            bounds = [(0, 1) for _ in odd_indices]
            constraint_value = target_sum - np.sum(h[even_indices])
            
            def constraint_fun(odd_vals):
                return np.sum(odd_vals) - constraint_value
            
            constraints = [{'type': 'eq', 'fun': constraint_fun}]
            
            result_odd = minimize(objective_odd, h[odd_indices], method='SLSQP',
                                  bounds=bounds, constraints=constraints,
                                  options={'ftol': 1e-10, 'maxiter': 100})
            
            if result_odd.success:
                h[odd_indices] = np.clip(result_odd.x, 0, 1)
                current_sum = np.sum(h)
                if current_sum > 0:
                    h = h * (target_sum / current_sum)
                current_c5 = evaluate_overlap(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        
        # Decay the perturbation delta
        perturbation_delta *= decay_factor
        
        # Early stopping if no improvement or no time left
        if abs(best_c5 - old_c5) < improvement_threshold:
            break
    
    # Refine final solution with local optimization
    def refine_objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(refine_objective, best_h, method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    # Local refinement result
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
