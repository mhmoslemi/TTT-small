import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def compute_coefficients(h, n_basis):
    """Compute Fourier coefficients of h."""
    x = np.linspace(0, 2, len(h))
    coeffs = np.zeros(n_basis)
    for i in range(n_basis):
        basis = np.sin((i + 1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        coeffs[i] = np.dot(h, basis) / np.dot(basis, basis)
    return coeffs

def backtransform(coeffs, n_basis, n_points):
    """Transform coefficients back to h."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_basis):
        basis = np.sin((i + 1) * np.pi * x / 2) * (1 - np.abs(x - 1))
        h += coeffs[i] * basis
    return h

def project(h, target_sum):
    """Project h to feasible region."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_basis = max(1, n_points // 4)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    h = project(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Multi-resolution optimization
    grid_levels = [1.0, 0.5, 0.25]
    lr = 0.001
    momentum = 0.9
    v = np.zeros(n_basis)
    
    max_iter = 50000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Step 1: Fourier space optimization
        coeffs = compute_coefficients(h, n_basis)
        
        # Compute overlap and its max
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Approximate gradients
        grad = np.zeros(n_basis)
        eps = 1e-6
        for i in range(n_basis):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = backtransform(coeffs_plus, n_basis, n_points)
            h_plus = project(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (np.max(overlap_plus) - max_overlap) / eps
            
        # Update coefficients with momentum
        v = momentum * v + lr * grad
        coeffs -= v
        
        # Transform back and project
        h_candidate = backtransform(coeffs, n_basis, n_points)
        h_candidate = project(h_candidate, target_sum)
        
        # Evaluate the new solution
        new_c5 = compute_c5(h_candidate, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            
        # Accept the improvement
        h = h_candidate.copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1

    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
