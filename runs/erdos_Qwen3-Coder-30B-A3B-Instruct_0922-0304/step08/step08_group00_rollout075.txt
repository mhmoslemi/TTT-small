import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Convex optimization approach
    # Define optimization variables
    h_vars = cp.Variable(n_points)
    
    # Build the full overlap matrix
    # This is a bit complex but we'll compute it once
    # Build the constraint sum(h) == target_sum
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == target_sum]
    
    # For the objective, we want to minimize max_k integral
    # Since this is not directly convex, we use a relaxation approach
    
    # We can just use a sophisticated local solver on the best solution
    # First try to solve with CVXPY using a relaxed approach for maximum
    # Let's first double-check the constraints are correct for our starting point
    
    # Use an efficient solver approach directly
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Try multiple starting points to avoid local optima
    best_h = None
    best_c5 = float('inf')
    
    # Try 5 different initializations
    for i in range(5):
        # Initialize with a different method for each
        if i == 0:
            # Use the existing solution if available
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
            else:
                h = np.zeros(n_points)
                h[:n_points//2] = 1
                h += np.random.normal(0, 0.02, n_points)
                h = np.clip(h, 0, 1)
        elif i == 1:
            # More random
            h = np.random.uniform(0, 1, n_points)
        elif i == 2:
            # Smooth wave
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.3 * np.sin(3 * np.pi * x) + 0.2 * np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
        elif i == 3:
            # Step function with randomization
            h = np.zeros(n_points)
            mid = n_points // 2 + np.random.randint(-5, 6)
            h[:mid] = 1
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        else:  # i == 4
            # Random smooth
            h = np.random.normal(0.5, 0.2, n_points)
            h = np.clip(h, 0, 1)
            
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Optimize
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        try:
            result = minimize(objective, h, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-8})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = refined_h.copy()
        except:
            continue
            
        if time.time() - start_time > budget_s - 10:
            break
    
    # If still no solution, fall back to a simple approach
    if best_h is None:
        # Start simple and uniform
        h = np.ones(n_points) * 0.5
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        best_h = h
        best_c5 = evaluate_overlap(h, dx)
        # Refine
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation and extra refinement known to be good
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
