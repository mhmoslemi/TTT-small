import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Convex optimization approach using CVXPY
    # Variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize maximum overlap
    # This is a bit tricky in convex form, we will do it differently
    
    # Let's use a structured approach: 
    # We use CVXPY to solve a constrained optimization directly on the overlap
    
    def solve_convex_optimization(h_start):
        """Solve the convex optimization problem."""
        # Create the variable
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Note: If we want to minimize the max overlap, we use a different approach
        # we can set up a constraint that the overlap values are bounded
        # but since this is not straightforward to express in CVXPY directly, 
        # we will solve the optimization using a constrained program that's
        # more direct
        
        # Let's approach this using projected gradient descent on the simplex
        # as a better alternative to pure random sampling / GA
        
        # Just solve the original problem with bounds and sum constraint
        return h_var, constraints
    
    # Simple Frank-Wolfe or projected gradient approach
    # we'll implement projected gradient decent with line search
    step_size = 0.01
    max_iter = 2000
    
    h_current = h_start = best_h.copy()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate gradient of the overlap function (approximate)
        # For a generic h, we want to compute the gradient of the max overlap
        # We'll use finite differences on the overlap integral to get a gradient-like estimate
        
        # Gradient estimation with finite differences
        epsilon = 1e-5
        grad = np.zeros(n_points)
        base_overlap = evaluate_overlap(h_current, dx)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            current_sum = np.sum(h_plus)
            if current_sum > 0:
                h_plus = h_plus * (target_sum / current_sum)
                
            plus_overlap = evaluate_overlap(h_plus, dx)
            grad[i] = (plus_overlap - base_overlap) / epsilon
            
        # Negative gradient descent step
        h_new = h_current - step_size * grad
        
        # Project to feasible set: [0,1] and sum = target_sum
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Check improvement
        new_c5 = evaluate_overlap(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            step_size = min(0.1, step_size * 1.05)  # adaptive step size
        else:
            step_size *= 0.9
            
        # Project to feasible set
        h_current = h_new.copy()
        
        # Early stopping strategy
        if iter_num > 100 and abs(best_c5 - 0.3807) < 1e-6:
            break
            
    # Local refinement with SLSQP to fine-tune
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
