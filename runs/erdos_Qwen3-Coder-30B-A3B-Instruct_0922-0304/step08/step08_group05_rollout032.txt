import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighbor(h, n_points):
    """Generate a neighboring solution."""
    new_h = h.copy()
    
    # 30% chance of point flipping
    if np.random.random() < 0.3:
        idx = np.random.randint(0, n_points)
        new_h[idx] = max(0, min(1, new_h[idx] * 0.97 + np.random.normal(0, 0.01)))
    else:
        # 70% chance of point swapping
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    tabu_tenure = 100
    max_no_improvement = 1000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create initial solution
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Validate initial solution
    assert np.abs(np.sum(h) - target_sum) < 1e-10
    assert np.all(h >= 0) and np.all(h <= 1)
    
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Tabu list as a queue of recent moves
    tabu_list = []
    tabu_moves = set()
    
    # State tracking
    no_improvement_count = 0
    temperature = 1.0
    alpha = 0.999
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Generate candidates
        candidates = []
        for _ in range(10):  # Generate 10 neighbors
            neighbor_h = generate_neighbor(h, n_points)
            # Ensure constraints
            neighbor_h = np.clip(neighbor_h, 0, 1)
            neighbor_h = normalize_h(neighbor_h, target_sum)
            
            c5 = compute_c5_exact(neighbor_h, dx)
            candidates.append((neighbor_h, c5))
        
        # Sort candidates by C5
        candidates.sort(key=lambda x: x[1])
        
        # Find a non-tabu candidate
        selected_h, selected_c5 = None, float('inf')
        for candidate_h, candidate_c5 in candidates:
            # Check if candidate's move is tabu
            # For tracking moves, we use a simple hash of the difference (simplified)
            candidate_tuple = tuple(candidate_h.round(6))
            if candidate_tuple not in tabu_moves:
                selected_h, selected_c5 = candidate_h, candidate_c5
                break
        
        # If all candidates are tabu, use best one anyway (aspiration)
        if selected_h is None:
            selected_h, selected_c5 = candidates[0]
            
        # Update best
        if selected_c5 < best_c5:
            best_c5 = selected_c5
            best_h = selected_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Update tabu list
        h_tuple = tuple(h.round(6))
        if len(tabu_moves) >= tabu_tenure:
            # Remove oldest
            tabu_moves.remove(tabu_list.pop(0))
        tabu_list.append(h_tuple)
        tabu_moves.add(h_tuple)
        
        # Update current solution
        h = selected_h
        
        # Update temperature
        temperature *= alpha
    
    # Final validation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
