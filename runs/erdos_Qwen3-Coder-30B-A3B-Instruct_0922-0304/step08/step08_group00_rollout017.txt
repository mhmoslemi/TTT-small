import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Initialize momentum variables
    v = np.zeros_like(h)
    theta = 1.0
    
    # Parameters for gradient descent
    learning_rate = 0.1
    momentum_factor = 0.9
    min_lr = 1e-5
    tolerance = 1e-6
    max_iter = int(1e5)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient computation function
    def compute_gradient(h_vec):
        # Compute all correlations
        corr_h_1mh = correlate(h_vec, 1 - h_vec, mode="full")
        # Find the maximum overlap index
        max_idx = np.argmax(corr_h_1mh)
        # Compute gradient, this is a simplified version
        # We're actually computing the gradient of the maximum correlation
        grad = np.zeros_like(h_vec)
        k = max_idx - (len(h_vec) - 1)  # shift
        for i in range(len(h_vec)):
            j = (i - k) % len(h_vec)
            grad[i] += (1 - h_vec[j])
        return grad
    
    # Project function to maintain constraint sum(h) = target_sum
    def project(h_vec):
        h_proj = h_vec.copy()
        current_sum = np.sum(h_proj)
        if current_sum > 0:
            h_proj = h_proj * (target_sum / current_sum)
        return h_proj
    
    # Clip and project
    def clip_and_project(h_vec):
        h_clip = np.clip(h_vec, 0, 1)
        return project(h_clip)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Nesterov accelerated gradient
        theta_new = (1 + np.sqrt(1 + 4 * theta**2)) / 2
        v = momentum_factor * v + compute_gradient(h)
        h_nesterov = h - learning_rate * v
        h_new = h_nesterov + (theta - 1) / theta_new * (h_nesterov - h)
        
        # Apply projection and clipping
        h_new = clip_and_project(h_new)
        
        # Evaluate new objective
        c5 = evaluate_overlap(h_new, dx)
        
        # Update if improvement
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            # Adaptively adjust learning rate based on improvement
            if abs(c5 - best_c5) > 0:
                learning_rate = min(learning_rate * 1.05, 1.0)
        else:
            # Reduce learning rate if no improvement
            learning_rate = max(learning_rate * 0.99, min_lr)
        
        # Update for next iteration
        h = h_new
        theta = theta_new
    
    # Final validation and return
    final_h = clip_and_project(best_h)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
