```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of possible shifts"""
    # We sample from -n+1 to n-1 (full overlap range)
    shifts = np.random.randint(-n+1, n, size=sample_size)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller size for faster optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5

    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Adaptive parameters
    target_overlap = best_c5
    max_shifts = 50
    min_improvement = 1e-6
    patience = 0
    max_patience = 200
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Sample a subset of shifts
        shifts = sample_shifts(n_points, max_shifts)
        # Build constraints for QP
        h = cp.Variable(n_points)
        # Set up QP
        # Minimize max overlap over sampled shifts
        # We add one constraint for each shift to control overlap
        
        # Create overlap constraints for each sampled shift
        overlaps = []
        for k in shifts:
            overlap = 0
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    overlap += h[i] * (1 - h[i + k])
            overlaps.append(overlap)
        
        # Create the QP problem - minimize the maximum overlap
        max_overlap = cp.Variable()
        constraints = [h[i] >= 0 for i in range(n_points)]
        constraints += [h[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h) == target_sum]
        # Each overlap should be bounded by target_overlap
        for ov in overlaps:
            constraints.append(ov <= max_overlap)
        
        # Minimize the maximum overlap
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        # Solve QP
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = np.array(h.value).flatten()
                h_solution = project_h(h_solution, n_points)
            else:  
                h_solution = h_current
        except:
            h_solution = h_current
            
        # Compute new C5 for this solution
        h_current = h_solution
        c5_new = compute_c5(h_current, dx)
        
        # Update best solution if improved
        if c5_new < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
            
        # Adapt target overlap based on improvement
        if iteration % 10 == 0:
            if patience > 0 and patience % 50 == 0:
                # Reduce target if stagnated
                target_overlap = max(best_c5 * 0.995, 0.3805)
        
        # Early stopping condition
        if best_c5 <= 0.3807 or patience > max_patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
