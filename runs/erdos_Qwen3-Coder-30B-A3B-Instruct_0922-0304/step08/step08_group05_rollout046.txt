import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_solution(h, n_points, target_sum):
    """Generate a perturbed solution for simulated annealing."""
    new_h = h.copy()
    
    # Randomly select indices to modify
    n_modify = max(1, int(0.1 * n_points))
    indices = np.random.choice(n_points, n_modify, replace=False)
    
    # Apply perturbations
    for i in indices:
        if np.random.random() < 0.5:
            # Scale down and normalize
            new_h[i] *= 0.97
        else:
            # Add noise
            new_h[i] += np.random.normal(0, 0.05)
    
    # Ensure values are [0,1]
    new_h = np.clip(new_h, 0, 1)
    
    # Restore sum constraint
    new_h = normalize_h(new_h, target_sum)
    
    return new_h

def cutting_plane_update(h, dx, target_sum, max_iter=5):

    new_h = h.copy()

                new_h[idx] = max(0, new_h[idx] - 0.01)

        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, target_sum)
        
        # Early exit if no improvement expected
        if compute_c5(new_h, dx) > compute_c5(h, dx):
            break
            
    return new_h
