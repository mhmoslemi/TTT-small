```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure it's in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample shifts for QP constraints"""
    # Sample shifts from -n+1 to n-1 (valid shift range)
    shifts = np.random.randint(-n+1, n, sample_size)
    return shifts

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Adaptive parameters
    target_overlap = best_c5
    patience = 0
    max_patience = 200
    min_improvement = 1e-6
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Sample shifts
        shifts = sample_shifts(n_points, 50)
        
        # Setup QP
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Compute overlaps for sampled shifts
        overlaps = []
        for k in shifts:
            # Compute convolution coefficients for shift k
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0..n_points-k-1 
                # For fixed k: selective multiplication of elements
                overlap = 0
                for i in range(n_points - k):
                    overlap += h[i] * (1 - h[i+k])
                overlaps.append(overlap)
            else:
                # k < 0: h[i+k] * (1 - h[i]) for i = -k..n_points-1
                overlap = 0
                for i in range(-k, n_points):
                    overlap += h[i+k] * (1 - h[i])
                overlaps.append(overlap)
        
        # Objective: minimize max overlap
        max_overlap = cp.Variable()
        for i, overlap in enumerate(overlaps):
            constraints.append(overlap <= max_overlap)
            
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve QP (for large-scale, this might be slow but we'll see)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h.value
                h_new = project_h(h_new, n_points)
                c5_new = compute_c5(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    patience = 0
                    # Adapt target overlap with a small decrement
                    target_overlap = best_c5 * 0.995
                else:
                    patience += 1
                    # Gradually increase target overlap if no improvement
                    if patience > 50:
                        target_overlap = np.minimum(target_overlap * 1.05, best_c5)
                        
                h_current = h_new.copy()
                
        except Exception as e:
            # Rollback to previous solution if solver failed
            pass
        
        iteration += 1
        
        if patience >= max_patience or best_c5 <= 0.3807:
            break
            
    # Final computation and validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
