import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a symmetric triangular ramp as starting point
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = 1.0 - (i - mid) / mid
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Simple gradient descent with adaptive step size
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 2000
    step_size = 0.01
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient approximation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / (2 * eps)
        
        # Update h using gradient descent
        h_new = h - step_size * grad
        h_new = normalize_h(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        
        # Check if new solution is better
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_max:
            h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            step_size *= 0.95  # Reduce step size if no improvement
            
        if no_improvement_count > max_no_improvement:
            # Try random perturbation
            h = np.random.rand(n_points) * 0.5  # Random between 0 and 0.5
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            no_improvement_count = 0
            step_size = 0.01
            
        iter_count += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
