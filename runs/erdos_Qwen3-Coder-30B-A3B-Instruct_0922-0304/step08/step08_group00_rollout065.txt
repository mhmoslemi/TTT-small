import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function (step function)
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add small noise for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.1
    momentum = 0.9
    min_lr = 1e-5
    max_iterations = 100000
    improvement_tolerance = 1e-6
    stagnation_patience = 1000
    
    # Nesterov momentum variables
    v = np.zeros(n_points)
    y = h.copy()
    
    # Track progress
    last_improvement = 0
    last_c5 = best_c5
    
    # Gradient descent loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current overlap for all shifts to find max
        overlap = correlate(y, 1 - y, mode="full")
        max_shift_idx = np.argmax(overlap)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient for the maximum overlap shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - max_shift) % n_points
            grad[i] = 1 - y[j]
        
        # Nesterov momentum update
        v = momentum * v - learning_rate * grad
        y = y + v
        
        # Project onto feasible set
        y = np.clip(y, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(y)
        if current_sum > 0:
            y = y * (target_sum / current_sum)
        
        # Evaluate current solution
        current_c5 = evaluate_overlap(y, dx)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_h = y.copy()
            best_c5 = current_c5
            last_improvement = iteration
            learning_rate = 0.1  # Reset learning rate on improvement
        else:
            # Adaptive learning rate
            if abs(current_c5 - last_c5) < improvement_tolerance:
                learning_rate *= 0.9
                if learning_rate < min_lr:
                    learning_rate = min_lr
            
        last_c5 = current_c5
        
        # Check for stagnation
        if iteration - last_improvement > stagnation_patience:
            # Restart with current best
            y = best_h.copy()
            learning_rate = 0.1
            last_improvement = iteration
    
    # Final optimization with local search
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    # Clip and rescale final result
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
