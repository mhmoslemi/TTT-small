import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        
        # Focus on top few worst lags
        # We select the maximum overlap lag
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)
        
        # Modify h to reduce this overlap
        # We try moving mass from i to i+k to reduce (1-h[i+k])*h[i]
        new_h = h.copy()
        
        # Determine which indices to modify
        i_start = max(0, -k)
        i_end = min(n_points, n_points - k)
        
        # Try moving small amounts of mass
        if i_start < i_end and k != 0:
            # Try moving from i to i+k
            for i in range(i_start, i_end):
                if new_h[i] > epsilon and new_h[i + k] < 1 - epsilon:
                    # Move a small amount
                    move_amount = min(epsilon, new_h[i], 1 - new_h[i + k])
                    new_h[i] -= move_amount
                    new_h[i + k] += move_amount
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * (target_sum / np.sum(new_h))
                    
                    # Check if this improves C5
                    new_c5 = compute_c5_exact(new_h, dx)
                    if new_c5 < current_c5:
                        h = new_h.copy()
                        break
            else:
                # If no improvement was found, adjust epsilon
                epsilon *= 0.95
        
        # Occasionally try more aggressive moves
        if iteration % 10 == 0 and epsilon > 1e-6:
            # Try to evenly distribute lags
            new_h = np.copy(h)
            # Distribute weights more evenly to reduce peaks
            sorted_indices = np.argsort(new_h)
            new_h[sorted_indices[:n_points//4]] = np.maximum(0, new_h[sorted_indices[:n_points//4]] - 0.02)
            new_h[sorted_indices[-n_points//4:]] = np.minimum(1, new_h[sorted_indices[-n_points//4:]] + 0.02)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (target_sum / np.sum(new_h))
            new_c5 = compute_c5_exact(new_h, dx)
            if new_c5 < current_c5:
                h = new_h.copy()
        
        iteration += 1
        if iteration % 100 == 0:
            epsilon = max(epsilon * 0.98, 1e-6)
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
