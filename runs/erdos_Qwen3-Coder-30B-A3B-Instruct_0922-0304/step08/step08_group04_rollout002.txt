import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h_values, dx):
    """Evaluate the C5 value for given h_values"""
    h = np.array(h_values)
    # Compute overlaps for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 70 points to match parent resolution
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        # Normalize to sum to n_points/2
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Otherwise, initialize uniformly
        h_init = np.ones(n_points) * 0.5
    
    # Create CVXPY variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints: sum(h) = n_points/2, 0<=h<=1
    constraints = [sum(h) == n_points / 2,
                   h >= 0,
                   h <= 1]
    
    # Generate overlap constraints for all shifts
    # For shift k, we have: sum h_i * (1 - h_{i+k}) <= t
    # This is: sum h_i - sum h_i * h_{i+k} <= t
    # So: sum h_i * h_{i+k} >= sum h_i - t
    
    # Using a more direct approach for the quadratic constraint
    # We handle the bilinear terms using CVXPY's built-in support for quadratic constraints
    # We can express overlap_k = sum h_i * (1 - h_{i+k}) = sum(h_i) - sum(h_i * h_{i+k})
    # Generalized constraint: for each shift k:
    # sum(h_i * (1 - h_{i+k})) <= t
    
    # Generate all shift constraints
    for shift in range(1 - n_points, n_points):
        # Compute valid indices for the shift
        # For shift 'shift', h_i contributes to h_{i+shift} 
        # This creates a cyclic boundary, can be handled by modular indexing
        # Build the constraint sum_{indices i} h_i * (1 - h_{(i+shift)%n_points}) <= t
        
        # Directly calculate the contribution in a loop for clarity
        # The constraint is: sum(h[i] * (1 - h[(i+shift)%n_points]) <= t
        if shift >= 0:
            # indices i in [0, n_points-shift-1] map to (i+shift) in [shift, n_points-1]
            # We create the constraint for this specific shift
            # Use a more efficient handling:
            indices_i = np.arange(n_points - shift)
            indices_shift = (indices_i + shift) % n_points
            # The term is h_i * (1 - h_{i+shift}) = h_i - h_i * h_{i+shift}
            # So sum_{i in indices_i} (h_i - h_i * h_{i+shift}) <= t
            # This means sum_{i} h_i - sum_{i} h_i * h_{i+shift} <= t
            # Or sum_{i} h_i * h_{i+shift} >= sum_{i} h_i - t
            
            # Note: the above is the non-linear version
            # Let's use terms carefully: h_i * (1 - h_{i+shift})
            terms = h[indices_i] * (1 - h[indices_shift])
            constraint = sum(terms) <= t
        else:
            # shift is negative, so (i+shift) should be handled carefully
            # indices i in [-shift, n_points-1] map to (i+shift) in [0, n_points+shift-1] 
            # Left over indices are [0, -shift-1], they map to [shift, -1]
            indices_i = np.arange(-shift, n_points)
            # indices_shift=indices_i+shift
            indices_shift = (indices_i + shift) % n_points
            terms = h[indices_i] * (1 - h[indices_shift])
            constraint = sum(terms) <= t
        
        constraints.append(constraint)
    
    # Create and solve problem
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    try:
        # Try multiple solvers with limits
        prob.solve(solver=ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
        else:
            # Fallback to local solution
            h_opt = h_init.copy()
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            t_opt = evaluate_erdos_solution(h_opt.tolist(), dx)
    except:
        # Last resort, use the initialization
        h_opt = h_init.copy()
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        t_opt = evaluate_erdos_solution(h_opt.tolist(), dx)
    
    # Final validation
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)  # Ensure sum = n_points/2
    c5_bound = evaluate_erdos_solution(h_opt.tolist(), dx)
    
    return (h_opt.tolist(), float(c5_bound), n_points)
