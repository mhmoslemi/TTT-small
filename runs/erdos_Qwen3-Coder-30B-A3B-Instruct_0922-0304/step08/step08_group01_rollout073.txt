import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.choice([0.0, 1.0], size=n_points, p=[0.5, 0.5])
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Temperature schedule
    start_temp = 0.1
    cooling_rate = 0.999
    current_temp = start_temp
    
    # Perturbation probabilities
    prob_flip = 0.5
    prob_adjust = 0.3
    prob_shuffle = 0.2
    
    max_iter = 100000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Make a copy of current solution
        h_new = h.copy()
        
        # Choose perturbation type
        choice = np.random.choice(['flip', 'adjust', 'shuffle'], p=[prob_flip, prob_adjust, prob_shuffle])
        
        if choice == 'flip':
            i = np.random.randint(n_points)
            h_new[i] = 1 - h_new[i]
        elif choice == 'adjust':
            i = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        elif choice == 'shuffle':
            length = np.random.randint(1, min(6, n_points))
            i = np.random.randint(n_points - length + 1)
            segment = h_new[i:i+length]
            np.random.shuffle(segment)
            h_new[i:i+length] = segment
        
        # Maintain constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        else:
            delta_e = new_c5 - best_c5
            if np.exp(-delta_e / current_temp) > np.random.rand():
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature
        current_temp *= cooling_rate
        
        # Adaptive reheating
        if iter_count % 1000 == 0 and current_temp < 0.01:
            current_temp = 0.1
            
        iter_count += 1
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
