import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    epsilon = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Get all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # For efficiency, only consider top few worst cases
        top_worst = 5
        lag_indices = np.argsort(overlap)[-top_worst:]
        # Focus on the single worst overlapping lag
        worst_lag = lag_indices[-1] - (n_points - 1)  # Convert full index to lag
        
        # Adjust h to penalize the worst overlap
        # Find positions contributing to this overlap
        for i in range(n_points):
            # Skipping boundary conditions for simplicity, could be improved
            if 0 <= i + worst_lag < n_points:
                # Increment an appropriate amount
                delta = epsilon * (1 - h[i]) * h[i + worst_lag]
                h[i] = min(1.0, h[i] + delta)
                h[i + worst_lag] = max(0.0, h[i + worst_lag] - delta)
        
        # Normalize again after adjustment
        h = h * (target_sum / np.sum(h))
        
        iter_count += 1
        
        # Adjust epsilon based on progress
        if iter_count % 100 == 0:
            epsilon *= 0.98
    
    # Final recomputation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
