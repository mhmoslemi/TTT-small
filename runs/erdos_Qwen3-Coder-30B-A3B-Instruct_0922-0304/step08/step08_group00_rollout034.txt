```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using numpy's correlate with proper handling
    # The mode='full' gives us all shifts, we need max of these
    overlap = np.correlate(h, 1 - h, mode="full")
    # Get the maximum value in the overlap array
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n_points = 70 as mentioned in the problem
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a basic construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Define equality constraint: sum(h) = target_sum
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use SLSQP method for optimization
    options = {'maxiter': 1000, 'ftol': 1e-10}
    
    # Try the optimization
    try:
        result = minimize(objective, h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options=options)
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except Exception as e:
        # If optimization fails, continue with current best
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
