```python
import numpy as np
from scipy.fftpack import fft, ifft
import time
from cvxpy import *
from scipy.signal import correlate

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to target sum"""
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute frequency-domain surrogate objective"""
    H_conj = np.conj(H)
    conj_term = 1 - H
    S = np.abs(H * conj_term)
    return S

def spectral_update(H, S, learning_rate=0.01):
    """Update spectrum to reduce high-frequency components"""
    # Find indices with highest surrogate values
    idxs = np.argsort(S)[::-1][:max(1, len(S) // 8)]
    
    # Update those frequencies
    for idx in idxs:
        H[idx] -= learning_rate * (H[idx] - np.conj(H[idx])) * S[idx]
    
    return H

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_h(h_current, target_sum)
    
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Transform to frequency domain
        H_current = fft(h_current)
        
        # Compute surrogate objective in frequency domain
        S = compute_surrogate(H_current)
        
        # Apply spectral update
        H_new = H_current.copy()
        H_new = spectral_update(H_new, S, learning_rate)
        
        # Reconstruct in time domain
        h_new = np.real(ifft(H_new))
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Check improvement
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update for next iteration
        h_current = h_new.copy()
        iteration += 1
        
        # Adaptively decrease learning rate
        if iteration % 100 == 0 and learning_rate > 0.001:
            learning_rate *= 0.5
            
        if best_c5 <= 0.3807:
            break
    
    # Local refinement
    h_refined = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.005)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
```
