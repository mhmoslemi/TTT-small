import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Ensure sum constraint
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function for minimization
    def objective(x):
        return compute_c5(x, dx)
    
    # Optimization with SLSQP
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Run optimization
    result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options=options, 
                      callback=lambda x: None)
    
    if result.success:
        h_opt = result.x
        if np.all(h_opt >= 0) and np.all(h_opt <= 1):
            h = h_opt.copy()
            c5_bound = compute_c5(h, dx)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    # Ensure the solution meets constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    
    return (best_h.tolist(), final_c5, n_points)
