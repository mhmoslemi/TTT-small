import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed_lp(h_init, n_points, dx, target_sum):
    """Solve relaxed LP to get a continuous solution."""
    # Create decision variables
    h = cp.Variable(n_points)
    
    # Objective: minimize maximum overlap
    z = cp.Variable()
    
    # Constraints
    constrs = []
    
    # Sum constraint
    constrs.append(cp.sum(h) == target_sum)
    
    # Bound constraints
    constrs.append(h >= 0)
    constrs.append(h <= 1)
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # Sum over valid x indices
            expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
        else:
            expr = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
        constrs.append(expr <= z)
    
    # Create and solve problem
    prob = cp.Problem(cp.Minimize(z), constrs)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    return h_init

def round_to_binary(h_relaxed, target_sum):
    """Round relaxed solution to binary with correct sum."""
    # Use median threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet target sum
    current_sum = np.sum(h_binary)
    deficit = int(current_sum - target_sum)
    
    if deficit > 0:
        # Need to set some 1s to 0
        indices = np.argsort(h_relaxed)
        for i in range(deficit):
            idx = indices[i]
            h_binary[idx] = 0
    elif deficit < 0:
        # Need to set some 0s to 1
        indices = np.argsort(-h_relaxed)
        for i in range(-deficit):
            idx = indices[i]
            h_binary[idx] = 1
            
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Iterate to alternate between LP relaxation and rounding
    max_iter = 10
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Solve relaxed LP
        h_relaxed = solve_relaxed_lp(h, n_points, dx, target_sum)
        
        if h_relaxed is not None:
            # Round to binary
            h_binary = round_to_binary(h_relaxed, target_sum)
            
            # Validate and compare
            new_c5 = compute_c5(h_binary, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_binary.copy()
                
            # Use binary as new starting point
            h = h_binary
            
            # Early stopping if target achieved
            if best_c5 <= 0.3807:
                break
                
        iter_count += 1

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
