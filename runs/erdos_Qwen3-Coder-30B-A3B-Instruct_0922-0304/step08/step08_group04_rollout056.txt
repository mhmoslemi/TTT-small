import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, noise_magnitude):
    """Apply Gaussian noise mutation to an individual."""
    noise = np.random.normal(0, noise_magnitude, len(h))
    h_mut = h + noise
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, len(h) / 2)
    return h_mut

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Crossover two parents to create offspring."""
    if random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    # Simple uniform crossover
    mask = np.random.rand(len(p1)) > 0.5
    child1 = np.where(mask, p1, p2)
    child2 = np.where(mask, p2, p1)
    
    # Normalize children to maintain sum constraint
    child1 = normalize_h(child1, len(p1) / 2)
    child2 = normalize_h(child2, len(p2) / 2)
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def evaluate_population(population, dx):
    """Evaluate the entire population and return fitnesses."""
    def evaluate_individual(h):
        return evaluate_c5(h, dx)
    
    fitnesses = Parallel(n_jobs=1)(
        delayed(evaluate_individual)(h) for h in population
    )
    return fitnesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.3
    noise_magnitude_start = 0.05
    noise_decay = 0.99
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            individual = mutate_individual(parent, noise_magnitude_start)
            population.append(individual)
    else:
        # Initialize with random individuals
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, n_points / 2)
            population.append(h)
    
    best_fitness = float('inf')
    best_individual = None
    
    # GA iterations
    for gen in range(generations):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 2:
            break
            
        # Evaluate fitness
        fitnesses = evaluate_population(population, dx)
        
        # Update best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            
        if best_fitness <= 0.3807:
            break
            
        # Selection, crossover, and mutation
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(best_individual.copy())
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover_parents(parent1, parent2, crossover_rate)
            
            # Mutation
            if random.random() < mutation_rate:
                child1 = mutate_individual(child1, noise_magnitude_start * (noise_decay ** gen))
            if random.random() < mutation_rate:
                child2 = mutate_individual(child2, noise_magnitude_start * (noise_decay ** gen))
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population[:population_size]
    
    # Final evaluation
    if best_individual is not None:
        final_c5 = evaluate_c5(best_individual, dx)
    else:
        final_c5 = evaluate_c5(population[0], dx)
        
    return (best_individual.tolist(), final_c5, n_points)
