import numpy as np
from scipy.fft import fft, ifft
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to target sum"""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h using numpy correlate"""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute magnitude of frequency-domain surrogate objective"""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, surrogate_grad, learning_rate):
    """Modify H in frequency domain to reduce surrogate"""
    # Gradient ascent in frequency domain
    H_up = H + learning_rate * surrogate_grad
    return H_up

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        h_current = np.random.uniform(0, 1, n_points)
        h_current = project_h(h_current, target_sum)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    learning_rate = 0.01
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Forward pass: compute Fourier transform
        H_current = fft(h_current)
        
        # Compute surrogate (frequency domain objective)
        surrogate = compute_surrogate(H_current)
        
        # Find dominant frequency bins contributing to max
        idxs = np.argsort(surrogate)[::-1][:n_points//4]  # Top 25% bins
        
        # Construct gradient in frequency domain (reduce high surrogate values)
        surrogate_grad = np.zeros_like(H_current)
        for i in idxs:
            surrogate_grad[i] = -np.sign(H_current[i]) * np.abs(surrogate[i])
        
        # Update H in frequency domain
        H_new = spectral_update(H_current, surrogate_grad, learning_rate)
        
        # Reconstruct h in time domain
        h_new = np.real(ifft(H_new))
        
        # Apply feasibility projection
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update current solution
        h_current = h_new.copy()
        
        # Adaptive learning rate decay
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.5
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    h_refined = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp = h_refined.copy()
        h_temp[idx] += delta
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
    
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
