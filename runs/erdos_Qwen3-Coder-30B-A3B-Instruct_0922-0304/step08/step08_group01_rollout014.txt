import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed, target_sum):
    """Convert relaxed solution to binary-like with correct sum"""
    h_binary = (h_relaxed > np.median(h_relaxed)).astype(float)
    
    current_sum = np.sum(h_binary)
    diff = int(current_sum - target_sum)
    
    if diff > 0:
        # Need to flip some 1s to 0s
        idxs = np.argsort(h_relaxed)[::-1][:diff]
        for i in idxs:
            h_binary[i] = 0
    elif diff < 0:
        # Need to flip some 0s to 1s
        idxs = np.argsort(h_relaxed)[:abs(diff)]
        for i in idxs:
            h_binary[i] = 1
    
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 50
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Relaxation Phase
        # Create a simple LP relaxation
        h_relaxed = h.copy()  # Start with current solution
        
        # Simple gradient-inspired heuristic: perturb in direction of reduction
        for i in range(n_points):
            # Approximate gradient
            grad = 0
            # Consider elements shifted by different k values
            for k in range(1, n_points):
                if i+k < n_points:
                    grad += (h[i] - 0.5) * (1 - h[i+k])
                if i-k >= 0:
                    grad += (h[i] - 0.5) * (1 - h[i-k])
            
            step = 0.01
            h_relaxed[i] += step * grad
            h_relaxed[i] = np.clip(h_relaxed[i], 0, 1)
        
        # Normalize relaxed solution
        h_relaxed = normalize_h(h_relaxed, target_sum)
        
        # Rounding Phase: convert to binary-like structure
        h_binary = round_to_binary(h_relaxed, target_sum)
        
        # Validate and compute C5
        h_binary = normalize_h(h_binary, target_sum)
        new_c5 = compute_c5(h_binary, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_binary.copy()
        
        # Update h for next iteration
        h = h_binary.copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Final validation using exact computation method
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
