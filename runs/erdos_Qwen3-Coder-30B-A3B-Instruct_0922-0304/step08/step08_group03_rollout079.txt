```python
import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    mid = n // 2
    return np.append(np.linspace(0, 1, mid), np.linspace(1, 0, n - mid))

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if np.sum(h) != target_sum or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Momentum parameters
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iterations = 20000
    
    velocity = np.zeros(n_points)
    no_improvement_count = 0
    max_no_improvement = 1000
    
    for iter_count in range(max_iterations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Find the index of maximum overlap
        max_idx = np.argmax(overlap)
        
        # Gradient based on the maximum overlap location
        # Approximating gradient: sign(overlap_max) * (1 - 2*h)
        sign_max = 1.0 if overlap[max_idx] >= 0 else -1.0
        gradient = sign_max * (1 - 2 * h) / n_points
        
        # Momentum update
        velocity = momentum * velocity + learning_rate * gradient
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < current_max:
            h = h_candidate.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            # Reduce step size and try again
            velocity *= shrink_factor
            h_candidate = h + velocity
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            new_c5 = compute_c5(h_candidate, dx)
            
            if new_c5 < current_max:
                h = h_candidate.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
```
