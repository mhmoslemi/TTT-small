import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    half = n // 2
    left = np.linspace(0, 1, half, endpoint=False)
    right = np.linspace(1, 0, n - half, endpoint=False)
    return np.append(left, right)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 20000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if np.abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    # Ensure constraints
    h = clip_and_renormalize(h, target_sum)
    
    # Optimization parameters
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    velocity = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_position = np.argmax(overlap)
        
        # Compute exact gradient (simplified)
        # Approximate gradient is sign(overlap_max) * (1 - 2*h) / n_points
        gradient_sign = 1.0 if overlap_max > 0 else -1.0
        
        # Gradient direction
        h_gradient = gradient_sign * (1 - 2 * h) / n_points
        h_gradient = np.nan_to_num(h_gradient)  # Handle any NaNs
        
        # Momentum update
        velocity = momentum * velocity + learning_rate * h_gradient
        
        # Candidate update
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept improvement or shrink learning rate
        if new_c5 < overlap_max:
            h = h_candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Too big a step, reduce learning rate
            learning_rate *= shrink_factor
            if learning_rate < 1e-6:
                break  # Stop if learning rate gets too small

        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
