import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize and clip
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 500
    iter_count = 0
    
    # Use CVXPY for convex relaxation approach
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # For binary-like handling, we will apply a smarter relaxation technique
        
        # Initialize the relaxed problem with CVXPY
        h_var = cp.Variable(n_points)
        z = cp.Variable()
        
        # Create constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
            z >= 0
        ]
        
        # Expose the actual overlap calculation within constraints
        # We model the overlap constraint for max k
        for k in range(2*n_points - 1):
            overlap_k = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    overlap_k += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_k <= z)
        
        # Minimize z (the maximum overlap)
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            h_relaxed = h_var.value
            if h_relaxed is not None:
                # Round the relaxed solution
                h_rounded = np.round(h_relaxed)
                
                # Ensure sum constraint
                current_sum = np.sum(h_rounded)
                diff = target_sum - current_sum
                
                # Adjust to maintain sum constraint
                if diff != 0:
                    # Find indices with highest relaxed values to increase
                    if diff > 0:
                        indices = np.argsort(h_relaxed)[-int(diff):]
                        for idx in indices:
                            if h_rounded[idx] < 1:
                                h_rounded[idx] = 1
                    else:
                        indices = np.argsort(h_relaxed)[:int(-diff)]
                        for idx in indices:
                            if h_rounded[idx] > 0:
                                h_rounded[idx] = 0
                
                # Ensure final clipping and normalization
                h_rounded = np.clip(h_rounded, 0, 1)
                
                # Renormalize if needed
                if np.sum(h_rounded) != target_sum:
                    h_rounded = h_rounded * (target_sum / np.sum(h_rounded))
                h_rounded = np.clip(h_rounded, 0, 1)
                
                # Compute new C5
                new_c5 = compute_c5(h_rounded, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_rounded.copy()
                
                # Early stopping if target achieved
                if best_c5 <= 0.3807:
                    break
                
                h = h_rounded.copy()
        except:
            # If convex relaxation fails, we fallback to simple optimization
            pass
            
        iter_count += 1
    
    # Final correction and re-evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
