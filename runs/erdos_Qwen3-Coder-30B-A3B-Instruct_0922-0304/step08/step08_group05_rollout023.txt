import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent or create structured construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    else:
        # Create a structured starting point
        h = np.zeros(n_points)
        # Place some mass in the middle
        mid_start = n_points // 4
        mid_end = 3 * n_points // 4
        h[mid_start:mid_end] = 1.0
        # Add some noise
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)

    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing variables
    temperature = 1.0
    cooling_rate = 0.999
    stagnation_count = 0
    max_stagnation = 1000
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        candidate_h = h.copy()
        
        # Perturb 10% of elements
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            # 50% chance to flip, 50% to swap
            if np.random.random() < 0.5:
                # Flip with small perturbation
                candidate_h[i] *= 0.97
            else:
                # Swap with another index
                j = np.random.randint(0, n_points)
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Ensure feasibility (bounds and sum)
        candidate_h = np.clip(candidate_h, 0, 1)
        current_sum = np.sum(candidate_h)
        if current_sum > 0:
            candidate_h = candidate_h * (target_sum / current_sum)
        
        # Compute C5 for candidate
        overlap = correlate(candidate_h, 1 - candidate_h, mode="full")
        candidate_c5 = np.max(overlap) * dx
        
        # Acceptance criterion
        if candidate_c5 < best_c5 or np.random.random() < np.exp(-(candidate_c5 - best_c5) / temperature):
            h = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h.copy()
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update temperature
        temperature *= cooling_rate
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Cutting plane direction
        if stagnation_count % 100 == 0 and stagnation_count > 0:
            # Perform more local refinement
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            max_lag = max_idx - n_points + 1
            
            # Adjust h to decrease overlap at max_lag
            # Reduce h[0] increase h[max_lag] if possible
            adjustment = 0.005
            h = h.copy()
            
            # Apply small adjustments
            for i in range(n_points):
                if (i + max_lag >= 0) and (i + max_lag < n_points):
                    if h[i] > adjustment and h[i + max_lag] < 1 - adjustment:
                        h[i] -= adjustment
                        h[i + max_lag] += adjustment
            
            # Ensure sum constraint
            current_sum = np.sum(h)
            if current_sum > 0:
                h = h * (target_sum / current_sum)
            
            # Re-evaluate
            overlap = correlate(h, 1 - h, mode="full")
            new_c5 = np.max(overlap) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Constraint repair
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        if stagnation_count > max_stagnation:
            break
    
    # Final computation
    overlap = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
