import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_individual(n_points, parent=None, noise_magnitude=0.05):
    """Create a new individual (h vector) with optional parent and noise."""
    if parent is not None and np.random.random() < 0.7:
        # Create offspring from parent with noise
        h = parent + np.random.normal(0, noise_magnitude, n_points)
    else:
        # Create random individual
        h = np.random.rand(n_points)
    
    # Clip to [0,1] and normalize
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points / 2)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform single-point crossover between two individuals."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, noise_magnitude=0.05):
    """Mutate an individual by adding random noise."""
    mutation = np.random.normal(0, noise_magnitude, len(h))
    h = h + mutation
    h = np.clip(h, 0, 1)
    h = normalize_h(h, len(h) / 2)
    return h

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def compute_fitness_parallel(population, dx):
    """Compute fitness of all individuals in parallel."""
    return Parallel(n_jobs=1)(delayed(evaluate_c5)(ind, dx) for ind in population)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    noise_magnitude = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population.append(parent.copy())
        for _ in range(population_size - 1):
            population.append(create_individual(n_points, parent, noise_magnitude))
    else:
        for _ in range(population_size):
            population.append(create_individual(n_points))
    
    best_c5 = float('inf')
    best_h = None
    
    # Evaluate initial population
    fitnesses = compute_fitness_parallel(population, dx)
    
    for gen in range(generations):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Keep best individual
        new_population.append(population[min_fitness_idx].copy())
        
        # Generate rest via selection, crossover, and mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, noise_magnitude)
            child2 = mutate(child2, noise_magnitude)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Update population and fitness
        population = new_population
        fitnesses = compute_fitness_parallel(population, dx)
        
        # Decrease noise magnitude over time
        noise_magnitude *= 0.99
        if noise_magnitude < 0.001:
            noise_magnitude = 0.001
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
