import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    h = np.clip(h, 0, 1)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization function
    def objective(x):
        return compute_c5(x, dx)
    
    # Run optimization with SLSQP
    result = minimize(
        objective,
        h,
        method='SLSQP',

    # Ensure constraints are met and compute final result
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))

    # Final validation and return
    return (best_h.tolist(), final_c5, n_points)
