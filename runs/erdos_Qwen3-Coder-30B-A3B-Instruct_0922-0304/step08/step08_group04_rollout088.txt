import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def convex_relaxation(n_points, dx):
    """Solve convex relaxation to get initial guess."""
    # Variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    constraints.append(cp.sum(h) == n_points / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Add overlap constraints for all shifts
    for k in range(2 * n_points - 1):
        # This is a simplified constraint ensuring linearization
        # In a real implementation we would properly linearize h*x terms
        # For now, we'll use a compromise approach
        pass
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return np.array(h.value).clip(0, 1)
    except:
        pass
    
    # Fallback to random if convex relaxation fails
    return np.random.rand(n_points).clip(0, 1)

def compute_gradient(h, dx, n_points):
    """Compute gradient of the C5 objective."""
    # Compute overlap
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Assume direct gradient with respect to h elements
    # We're looking at the first max input to the overlap calculation
    grad = np.zeros_like(h)
    
    # This is a gradient approximation via finite differences of the max overlap
    epsilon = 1e-6
    max_overlaps = []
    
    for i in range(n_points):
        # Small perturbation to compute directional derivative
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] = min(1.0, h[i] + epsilon)
        h_minus[i] = max(0.0, h[i] - epsilon)
        
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
        
        gradient_val = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * epsilon)
        grad[i] = gradient_val
    
    # Normalize the gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    return grad

def optimize_with_adam(h, dx, n_points, max_iter=5000, lr=0.01, decay=0.999, tolerance=1e-8):
    """Optimize h using Adam-like gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Compute loss and gradient
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Check termination conditions
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 500:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

        # Try convex relaxation first, then fallback
        h = convex_relaxation(n_points, dx)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # Optimize using Adam
    best_h, best_c5 = optimize_with_adam(h, dx, n_points, max_iter=5000, lr=0.01, decay=0.999)
    
    # Final validation
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
