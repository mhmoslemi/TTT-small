import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Use convex optimization approach
    h_var = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints
    # For each shift k, we compute the overlap integral
    # This involves summing h[i](1-h[i+k]) for valid indices
    for k in range(1 - n_points, n_points):
        # Compute the effective indices that have a contribution
        # In correlate with mode="full", when computing overlap for shift k,
        # we sum h[i] * (1 - h[i+k]) for valid pairs
        expr = 0
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                expr += h_var[i] * (1 - h_var[j])
        constraints.append(expr <= t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        # Solve using ECOS solver
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_var.value
            c5_bound = t.value
        else:
            # If optimization fails, fall back to the initial solution
            h_sol = h
            c5_bound = evaluate_c5(h_sol, dx)
    except Exception as e:
        # If there's any issue with solving, fall back to initial
        h_sol = h
        c5_bound = evaluate_c5(h_sol, dx)
    
    # Final refinement using the evaluator
    # Ensure the solution is within bounds
    h_sol = np.clip(h_sol, 0, 1)
    # Normalize to exactly sum to n_points / 2
    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_sol, dx)
    
    return (h_sol.tolist(), float(final_c5), n_points)
