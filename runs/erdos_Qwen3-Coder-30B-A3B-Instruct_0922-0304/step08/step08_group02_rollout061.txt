import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create a more informed initial starting point
        h_current = np.zeros(n_points)
        # Place 1's in dense regions, 0's elsewhere
        h_current[:n_points//2] = 1.0
        # Add some noise to escape trivial solutions
        noise = np.random.normal(0, 0.05, n_points)
        h_current += noise
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    annealing_scale = 1000
    max_no_improvement = 500
    
    # Tracking
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    current_c5 = best_c5
    T = T0
    no_improvement_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        
        # Generate perturbation
        delta = np.random.normal(0, sigma)
        h_new[idx] = max(0, min(1, h_new[idx] + delta))
        
        # Apply sum constraint projection
        current_sum = np.sum(h_new)
        scale_factor = target_sum / current_sum
        h_new = h_new * scale_factor
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Update temperature
        T = max(T_min, T0 * (alpha ** (iteration / annealing_scale)))
        
        # Adaptive perturbation
        if no_improvement_count > max_no_improvement:
            sigma *= 0.95
            no_improvement_count = 0
            
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
