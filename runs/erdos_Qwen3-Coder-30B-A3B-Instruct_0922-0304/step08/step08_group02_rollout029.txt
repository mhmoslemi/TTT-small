import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate h with given probability."""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.1)
    return h_mut

def genetic_algorithm_step(h_population, fitnesses, target_sum, n_points, n_pop, elite_ratio=0.2, mutation_rate=0.05):
    """Perform one generation of genetic algorithm."""
    sorted_indices = np.argsort(fitnesses)
    elite_count = int(n_pop * elite_ratio)
    
    new_population = [h_population[i] for i in sorted_indices[:elite_count]]
    
    while len(new_population) < n_pop:
        # Tournament selection
        idx1, idx2 = np.random.choice(sorted_indices[:elite_count], 2, replace=False)
        parent1 = h_population[idx1]
        parent2 = h_population[idx2]
        
        child1, child2 = crossover(parent1, parent2)
        child1 = repair_h(mutate(child1, mutation_rate), target_sum)
        child2 = repair_h(mutate(child2, mutation_rate), target_sum)
        
        new_population.extend([child1, child2])
    
    return new_population[:n_pop]

def gradient_refinement(h, target_sum, dx, max_steps=50):
    """Improve solution using gradient-like local search."""
    h_refined = h.copy()
    old_c5 = compute_c5(h_refined, dx)
    
    for _ in range(max_steps):
        overlap = correlate(h_refined, 1 - h_refined, mode="full")
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Find which values contribute most to this overlap
        # Look at the correlation structure to guide gradient steps
        if max_idx < len(h_refined):
            # Focus on segment that contributes to current max overlap
            start_idx = max_idx
            end_idx = min(start_idx + len(h_refined), len(overlap) - 1)
        else:
            # Overlap at the tail - check last few entries
            start_idx = max_idx - len(h_refined) + 1
            end_idx = len(h_refined) - 1
            
        # Modify values in area where overlap is maximized
        for i in range(start_idx, min(end_idx + 1, len(h_refined))):
            delta = np.random.normal(0, 0.01)
            h_refined[i] += delta
            
        h_refined = repair_h(h_refined, target_sum)
        
        new_c5 = compute_c5(h_refined, dx)
        if new_c5 < old_c5:
            old_c5 = new_c5
        else:
            # Reset if no improvement 
            h_refined = h.copy()
            
    return h_refined
