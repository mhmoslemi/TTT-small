import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Find the worst overlap lag
        lag = max_idx - (n_points - 1)
        
        # Adjust values to reduce overlap at lag
        # Focus on the primary configuration that contributes to maximum overlap
        temp_h = h.copy()
        
        # Find indices contributing to max overlap
        max_overlap = overlap[max_idx]
        
        # This adjustment aims to reduce the key overlap term by locally adjusting h values
        updates = np.zeros(n_points)
        
        # Target to reduce h[i]*h[i+lag] terms by moving mass from overlapping regions
        # We modify h to decrease overlap terms
        for i in range(n_points):
            j = i + lag
            # Consistent with max overlap index
            if 0 <= j < n_points:
                # Local adjustment to reduce the specific overlap
                updates[i] -= epsilon * (1 - 2 * temp_h[i]) * (1 - 2 * temp_h[j])
                updates[j] -= epsilon * (1 - 2 * temp_h[i]) * (1 - 2 * temp_h[j])
        
        # Apply update
        h = h + updates * 0.5
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        iteration += 1
        if iteration % 100 == 0:
            epsilon *= 0.995

    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
