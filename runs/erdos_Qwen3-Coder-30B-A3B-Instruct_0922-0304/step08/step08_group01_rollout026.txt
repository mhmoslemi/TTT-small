import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    def objective(h_vals):
        return compute_c5(h_vals, dx)
    
    # Perform optimization with SLSQP
    result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                      constraints=constraint, options={'maxiter': 10000})
    
    if result.success:
        h_opt = result.x
    else:
        # Fallback to initial if optimization failed
        h_opt = h
    
    # Final validation and computation
    final_c5 = compute_c5(h_opt, dx)
    
    # Ensure constraints are met
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (target_sum / np.sum(h_opt))
    
    # Recompute final C5
    final_c5 = compute_c5(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
