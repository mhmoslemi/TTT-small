import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform uniform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    mask = np.random.rand(n) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate h by adding noise."""
    if random.random() > mutation_rate:
        return h.copy()
    
    noise = np.random.normal(0, noise_magnitude, len(h))
    mutated = h + noise
    return np.clip(mutated, 0, 1)

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual via tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_index = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_index].copy()

def ga_optimize(initial_h, dx, n_points, pop_size=50, generations=1000, 
                crossover_rate=0.8, mutation_rate=0.3, noise_magnitude=0.05):
    """Genetic algorithm optimization for C5 minimization."""

        noise = np.random.normal(0, 0.01, n_points)
        individual = np.clip(initial_h + noise, 0, 1)
        individual = normalize_h(individual, n_points / 2)
        population.append(individual)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    for generation in range(generations):

        # Update best solution

            if fitnesses[i] < best_c5:
                best_c5 = fitnesses[i]

        if best_c5 <= 0.3807:
            break
            
        # Selection, crossover, mutation
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)

            child1 = mutate(child1, mutation_rate, noise_magnitude)
            child2 = mutate(child2, mutation_rate, noise_magnitude)
            
            # Feasibility repair
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, n_points / 2)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, n_points / 2)
            
            new_population.extend([child1, child2])
        
        population = new_population
        noise_magnitude *= 0.99
        if generation % 20 == 0:
            print(f"Generation {generation}: Best C5 = {best_c5:.6f}")
    
    return best_h, best_c5
