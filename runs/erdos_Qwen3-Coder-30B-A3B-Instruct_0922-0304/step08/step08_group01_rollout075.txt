import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    n_points = 70

    target_sum = n_points / 2

    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization using SLSQP
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9},
        callback=lambda x: None
    )
    
    # If optimization fails or takes too long, return initial solution
    if not result.success:
        best_h = h.copy()
        best_c5 = compute_c5(h, dx)
    else:

        best_c5 = result.fun
    
    # Final validation
    # Ensure constraints are still satisfied
    if abs(np.sum(best_h) - target_sum) > 1e-6:

        best_h = best_h * (target_sum / np.sum(best_h))
        best_h = np.clip(best_h, 0, 1)
    
    final_c5 = compute_c5(best_h, dx)
