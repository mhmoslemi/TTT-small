import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        # Validate and fix sum
        if abs(np.sum(h) - target_sum) > 1e-10:
            h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
    else:
        h = triangular_ramp(n_points)
    
    # Ensure sum constraint
    h = clip_and_renormalize(h, target_sum)
    
    # Optimizer state
    velocity = np.zeros(n_points)
    alpha = 0.9      # momentum
    beta = 0.001     # learning rate
    gamma = 0.5      # shrink factor
    max_iter = 20000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        # Find the position of maximum overlap
        max_pos = np.argmax(overlap)
        
        # Compute gradient direction: approximating sign of max overlap effect
        h_gradient = np.zeros(n_points)
        # For each index, we estimate how much changing h_i affects the maximum overlap
        for i in range(n_points):
            # Perturb h[i] up and down
            h_up = h.copy()
            h_up[i] = min(1.0, h[i] + 0.01)
            h_up = clip_and_renormalize(h_up, target_sum)
            up_overlap = correlate(h_up, 1 - h_up, mode="full") * dx
            up_max = np.max(up_overlap)
            
            h_down = h.copy()
            h_down[i] = max(0.0, h[i] - 0.01)
            h_down = clip_and_renormalize(h_down, target_sum)
            down_overlap = correlate(h_down, 1 - h_down, mode="full") * dx
            down_max = np.max(down_overlap)
            
            h_gradient[i] = (up_max - down_max) / 0.02  # finite diff approx
        
        # Update with momentum
        velocity = alpha * velocity + beta * h_gradient
        h_candidate = h + velocity
        
        # Project back to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new overlap
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            if best_c5 <= 0.3807:
                break
            h = h_candidate.copy()
        else:
            # Shrinking step
            beta *= gamma
            if beta < 1e-6:
                break  # give up if too small
                
        iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
