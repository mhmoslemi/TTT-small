import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: 0 <= h <= 1 and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of C5 with respect to h."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # For the maximum overlap, compute gradient
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Position in the full correlation array
        # When k = max_idx - n_points + 1, we get i + k = i + max_idx - n_points + 1
        # But we need to consider the circular nature
        # But actually max_idx corresponds to k such that [k] in full correlation 
        # overlap[k] = sum h[i] * (1 - h[i + k - (n_points - 1)]) for i = 0..n_points-1
        # i + k - (n_points - 1) = offset
        
        k = max_idx - (n_points - 1)
        # This is the effective k for the maximum overlap, possibly using wrapping
        # We'll compute when correlation matches the max
        c = 0
        # Compute contribution of h[i] to the gradient of overlapping term
        for j in range(n_points):
            # Effective position=j+k when handling circular convolution properly
            pos = (j + k) % n_points
            c += h[j] * (1 - 2 * h[i])
        # We are robustly using normalized gradient
        c = 2 * overlap[max_idx] / dx * (1 - 2 * h[i]) if overlap[max_idx] != 0 else 0
        grad[i] = 2 * (1 - 2 * h[i])  # Simplified gradient, but actually we need to compute the precise match
        
    # Actually take a simpler approach
    # Take the max overlap and just compute gradient for that
    max_idx = np.argmax(overlap)
    grad = np.zeros_like(h)
    # Compute derivative of the gradient of the overlap for that k
    # Let's just compute it more directly:
    for i in range(n_points):
        s = 0.0
        for j in range(n_points):
            # In the correlation: overlap[max_idx] = sum_{i=0}^{n-1} h[i]* (1-h[(i+max_idx-n+1)%n])*
            # For this index, its derivative in the direction of h[i]
            # delta_overlaps[k]/ delta_h[i] = 1-h[(i+k-n+1)%n] + h[i] * delta/ dh[i] (1-h[...])
            # Let's reconsider the derivative 
            k = max_idx - (n_points - 1)
            pos = (i + k) % n_points
            s += (1 - 2 * h[i])  # zero differentiate - this simplifies if overcorr is T or similar
        grad[i] = s * (2 / dx)  # gradient comes from correlation
        
    return grad

def solve_linear_subproblem(grad, n_points):
    """Solve linear subproblem: minimize grad^T v subject to sum v = n_points/2, 0 <= v <= 1."""
    # Since it is a linear cost and a simple sum constraint,
    # x = 1 yields gradient min over 0 <= v_i <= 1, sum(v) = n_points/2.
    # Let's solve using cvxpy.
    v = cp.Variable(n_points)
    constraints = [cp.sum(v) == n_points / 2, v >= 0, v <= 1]
    objective = cp.Minimize(grad @ v)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        v_sol = v.value
    except:
        # fallback to simpler method:
        v_sol = np.zeros(n_points)
        # Simple greedy heuristic
        sorted_indices = np.argsort(-grad)
        remaining_sum = n_points / 2
        for i in sorted_indices:
            if remaining_sum <= 0: break
            v_sol[i] = min(1.0, remaining_sum)
            remaining_sum -= v_sol[i]
        # rescale to exact sum
        v_sol = v_sol * (n_points / 2) / np.sum(v_sol)
        
    return np.array(v_sol)

def frank_wolfe_optimize(h, dx, n_points, max_iter=10000, tolerance=1e-8):
    """Frank-Wolfe algorithm for minimizing C5."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    prev_loss = float('inf')
    for i in range(max_iter):
        if (time.time() - start_time) > 590:
            break
        
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)

        # Solve linear subproblem
        v = solve_linear_subproblem(grad, n_points)

        # Line search to find best step size
        # Use simple grid search on eta
        eta_vals = np.linspace(0.01, 0.99, 10)
        best_eta = 0.01
        best_v_c5 = float('inf')
        for eta in eta_vals:
            h_new = (1 - eta) * h + eta * v
            h_new = project_h(h_new, n_points / 2)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_v_c5:
                best_v_c5 = c5_new
                best_eta = eta
        # Update h using best eta
        h = (1 - best_eta) * h + best_eta * v
        h = project_h(h, n_points / 2)
        
        # Update best if improved
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check for convergence
        if abs(prev_loss - c5) < tolerance:
            break
        prev_loss = c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)  # Uniform initialization
    
    # Optimize the function
    best_h, best_c5 = frank_wolfe_optimize(h, dx, n_points, max_iter=10000)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            
    return (best_h.tolist(), best_c5, n_points)
