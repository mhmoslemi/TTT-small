import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Relaxation and rounding approach
    max_iter = 100
    iteration = 0
    patience = 0
    max_patience = 10
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Relaxation step using a convex formulation
        try:
            # Relax problem to a convex program
            h_relaxed = cp.Variable(n_points)
            z = cp.Variable()
            constraints = [
                cp.sum(h_relaxed) == target_sum,
                h_relaxed >= 0,
                h_relaxed <= 1,
            ]
            
            # Add constraints for overlap terms
            # For efficiency, sample a subset of terms for consistency
            num_samples = min(20, n_points // 2)
            sample_indices = np.random.choice(n_points - num_samples, num_samples, replace=False)
            
            for i in sample_indices:
                # K = [i, i+1, ..., i+num_samples]
                if i + num_samples < n_points:
                    # Simple approximation: only looking at k=0 to num_samples-1
                    k_range = range(0, min(num_samples, n_points - i))
                    terms = [h_relaxed[j] * (1 - h_relaxed[j + k]) for j, k in zip(
                        range(i, i + len(k_range)), k_range)]
                    constraints.append(z >= cp.sum(terms) * dx)
            
            objective = cp.Minimize(z)
            prob = cp.Problem(objective, constraints)
            
            # Check remaining time before solving
            if time.time() - start_time > budget_s - 10:
                break
                
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value.copy()
                
                # Rounding heuristic
                # Threshold at median
                median_val = np.median(h_new)
                h_binary = (h_new > median_val).astype(float)
                
                # Adjust to meet constraint sum = target_sum
                current_sum = np.sum(h_binary)
                diff = int(current_sum - target_sum)
                
                if diff > 0:
                    # Need to flip diff ones to zeros
                    # Find the largest values to flip
                    idxs = np.argsort(h_binary)[::-1][:diff]
                    for i in idxs:
                        h_binary[i] = 0
                elif diff < 0:
                    # Need to flip -diff zeros to ones
                    # Find the smallest values to flip
                    idxs = np.argsort(h_binary)[:abs(diff)]
                    for i in idxs:
                        h_binary[i] = 1
                        
                # Verify sum constraint
                if abs(np.sum(h_binary) - target_sum) < 1e-4:
                    h = h_binary.copy()
                    new_c5 = compute_c5(h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                else:
                    # Infeasible adjustment
                    pass
                
                # Early stopping
                if best_c5 <= 0.3807:
                    break
                    
        except:
            # If LP fails, fall back to small random perturbation
            perturb = np.random.normal(0, 0.02, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        
        iteration += 1
        
        # Early stopping if no improvement for a while
        if patience >= max_patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
