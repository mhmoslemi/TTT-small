import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the C5 objective."""
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Gradient computation for the maximum overlap term
    grad = np.zeros_like(h)
    for i in range(n_points):
        # For each position compute contribution to max overlap
        grad[i] = 0.0
        for k in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - 2 * h[i]) * (1 - h[j])
    
    return grad * dx

def solve_linear_subproblem(grad, n_points):
    """Solve the linear subproblem to find descent direction."""
    # Minimize grad^T * v subject to sum(v) = n_points/2, 0 <= v <= 1
    # This is a linear program. We can solve it directly:
    # minimize c^T * x
    # subject to A_eq * x = b, A_ub * x <= b_ub
    
    c = grad
    A_eq = np.array([1.0] * n_points)
    b_eq = n_points / 2
    
    bounds = [(0, 1)] * n_points
    
    # Use linprog to solve
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        v = res.x
    else:
        # Fallback to simple approach: set v = 0 at max grad, 1 at min grad then renormalize
        v = np.zeros(n_points)
        # Simple heuristic: set values to 1 where grad is negative, 0 where positive
        v[grad < 0] = 1.0  # Try to go towards minimizing gradient
        # Renormalize to sum to n_points/2
        v = project_h(v, n_points / 2)
        
    return v

def frank_wolfe_optimization(h, dx, n_points, max_iter=10000, tolerance=1e-8):
    """Run Frank-Wolfe algorithm for optimization."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    prev_loss = float('inf')
    no_improvement_count = 0
    
    for i in range(max_iter):
        # Compute current objective value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if current_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Solve linear subproblem
        v = solve_linear_subproblem(grad, n_points)
        
        # Line search (simple backtracking)
        eta = 1.0
        min_eta = 0.01
        max_eta = 0.99
        
        while eta > min_eta:
            h_new = (1 - eta) * h + eta * v
            h_new = project_h(h_new, n_points / 2)
            test_c5 = evaluate_c5(h_new, dx)
            
            if test_c5 <= current_c5:
                h = h_new
                break
            else:
                eta *= 0.5
        
        if abs(prev_loss - current_c5) < tolerance:
            no_improvement_count += 1
            if no_improvement_count > 100:
                break
        else:
            no_improvement_count = 0
        prev_loss = current_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available

        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Uniform initial guess
    
    # Optimize the function
    best_h, best_c5 = frank_wolfe_optimization(h, dx, n_points, max_iter=10000)
    
    return (best_h.tolist(), best_c5, n_points)
