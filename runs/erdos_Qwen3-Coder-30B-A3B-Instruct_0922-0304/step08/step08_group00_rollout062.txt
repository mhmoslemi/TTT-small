import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Simple construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization with perturbation
    max_iter = 100
    perturb_magnitude = 1e-3
    improvement_threshold = 1e-5
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        prev_c5 = best_c5
        
        # Partition into even and odd indices
        even_indices = list(range(0, n_points, 2))
        odd_indices = list(range(1, n_points, 2))
        
        # Optimize even indices (fix odd)
        def even_objective(h_even):
            h_full = best_h.copy()
            h_full[even_indices] = h_even
            h_full = np.clip(h_full, 0, 1)
            current_sum = np.sum(h_full)
            if current_sum > 0:
                h_full = h_full * (target_sum / current_sum)
            return evaluate_overlap(h_full, dx)
            
        # Optimize odd indices (fix even)
        def odd_objective(h_odd):
            h_full = best_h.copy()
            h_full[odd_indices] = h_odd
            h_full = np.clip(h_full, 0, 1)
            current_sum = np.sum(h_full)
            if current_sum > 0:
                h_full = h_full * (target_sum / current_sum)
            return evaluate_overlap(h_full, dx)
            
        # Optimize even indices
        h_even = best_h[even_indices].copy()
        bounds = [(0, 1) for _ in range(len(even_indices))]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - np.sum(best_h[even_indices])}]
        
        result = minimize(even_objective, h_even, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            h_full = best_h.copy()
            h_full[even_indices] = result.x
            h_full = np.clip(h_full, 0, 1)
            current_sum = np.sum(h_full)
            if current_sum > 0:
                h_full = h_full * (target_sum / current_sum)
            c5 = evaluate_overlap(h_full, dx)
            if c5 < best_c5:
                best_h = h_full
                best_c5 = c5
                
        # Optimize odd indices
        h_odd = best_h[odd_indices].copy()
        bounds = [(0, 1) for _ in range(len(odd_indices))]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - np.sum(best_h[odd_indices])}]
        
        result = minimize(odd_objective, h_odd, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            h_full = best_h.copy()
            h_full[odd_indices] = result.x
            h_full = np.clip(h_full, 0, 1)
            current_sum = np.sum(h_full)
            if current_sum > 0:
                h_full = h_full * (target_sum / current_sum)
            c5 = evaluate_overlap(h_full, dx)
            if c5 < best_c5:
                best_h = h_full
                best_c5 = c5
                
        # Perturb every few iterations to escape local maxima
        if iter_count % 5 == 0:
            perturb_magnitude *= 0.98
            noise = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h_perturbed = best_h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            if current_sum > 0:
                h_perturbed = h_perturbed * (target_sum / current_sum)
            c5_perturbed = evaluate_overlap(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
                
        iter_count += 1
        if abs(best_c5 - prev_c5) < improvement_threshold:
            break
        
    # Final refinement
    def final_objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(final_objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-6})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
