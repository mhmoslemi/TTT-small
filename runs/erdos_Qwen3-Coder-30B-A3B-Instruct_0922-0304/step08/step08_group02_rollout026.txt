import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_prob:
        return deepcopy(parent1), deepcopy(parent2)
    
    n = len(parent1)
    point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a solution."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h

        population.append(h_init)
    
    # Fill population
    while len(population) < n_pop:
        h_candidate = np.random.rand(n_points)
        h_candidate = repair_h(h_candidate, target_sum)
        population.append(h_candidate)
    
    best_fitness = float('inf')
    best_individual = None
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []

        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        if fitnesses[0] < best_fitness:
            best_fitness = fitnesses[0]
            best_individual = population[0].copy()
        
        # Keep top 20%
        top_indices = sorted_indices[:n_pop // 5]
        top_pop = [population[i] for i in top_indices]
        
        # Generate new population
        new_pop = top_pop.copy()
        
        # Crossover and mutate
        while len(new_pop) < n_pop:
            parent1 = population[np.random.randint(0, n_pop // 2)]
            parent2 = population[np.random.randint(0, n_pop // 2)]
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, 0.05)
            child2 = mutate(child2, 0.05)
            
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_pop.extend([child1, child2])
        
        # Trim to population size
        population = new_pop[:n_pop]
        
        # Early stopping
        if gen > 100 and fitnesses[0] <= 0.3807:
            break
    
    return best_individual, best_fitness
