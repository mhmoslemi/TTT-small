import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 500
    iter_count = 0
    
    # Early stopping threshold
    target = 0.3807
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Create variables for CVXPY
        h_var = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum, z >= 0]
        
        # Add overlap constraints
        # For each shift k, create overlap constraint
        for k in range(2 * n_points - 1):
            # Compute indices and their contributions
            overlap_k = 0
            for i in range(n_points):
                j = i + k - (n_points - 1)
                if 0 <= j < n_points:
                    overlap_k += h_var[i] * (1 - h_var[j])
            
            constraints.append(overlap_k <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_relaxed = h_var.value
            else:
                h_relaxed = h
        except:
            h_relaxed = h

        # Rounding heuristic: threshold at 0.5 and adjust
        h_rounded = np.zeros_like(h_relaxed)
        for i in range(len(h_relaxed)):
            if h_relaxed[i] >= 0.5:
                h_rounded[i] = 1.0
            else:
                h_rounded[i] = 0.0
            
        # Ensure sum constraint
        current_sum = np.sum(h_rounded)
        diff = target_sum - current_sum
        
        # If we need to add more ones, flip zeros to ones (higher indices)
        if diff > 0:
            # Find zeros to turn into ones
            zero_indices = np.where(h_rounded == 0)[0]
            num_to_flip = min(len(zero_indices), int(diff))
            for i in range(num_to_flip):
                h_rounded[zero_indices[i]] = 1.0
        elif diff < 0:
            # Find ones to turn into zeros
            one_indices = np.where(h_rounded == 1)[0]
            num_to_flip = min(len(one_indices), int(-diff))
            for i in range(num_to_flip):
                h_rounded[one_indices[i]] = 0.0
        
        # Renormalize if needed
        h_rounded = h_rounded * (target_sum / np.sum(h_rounded))
        h_rounded = np.clip(h_rounded, 0, 1)
        
        # Evaluate
        new_c5 = compute_c5(h_rounded, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_rounded.copy()
            
        if best_c5 <= target:
            break
            
        # Greedy adjustments
        for i in range(n_points):
            # Try to adjust h[i]
            h_temp = h_rounded.copy()
            h_temp[i] = 1 - h_temp[i]
            
            # Renormalize
            h_temp = h_temp * (target_sum / np.sum(h_temp))
            h_temp = np.clip(h_temp, 0, 1)
            
            new_c5_temp = compute_c5(h_temp, dx)
            if new_c5_temp < new_c5:
                h_rounded = h_temp
                new_c5 = new_c5_temp
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_rounded.copy()
                    
                    if best_c5 <= target:
                        break
        
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
