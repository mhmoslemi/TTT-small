import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simplified search: perturb and check with local search
    max_iter = 30000
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
            
        # Random perturbation
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        perturbation = np.random.normal(0, 0.05)
        h_new[idx] += perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2)
        
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iter_count
            
        h = h_new
        iter_count += 1
        
        # Early stopping if no improvement in a while
        if iter_count - last_improvement > 5000:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
