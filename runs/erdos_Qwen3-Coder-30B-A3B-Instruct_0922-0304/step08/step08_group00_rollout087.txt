import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Manually compute correlation for better efficiency
    n = len(h)
    max_overlap = 0.0
    # For each shift k from -n+1 to n-1 (shifts that are meaningful)
    # But we only care about k from 0 to n-1 since we're computing max_k of 
    # integral_h(x)(1 - h(x+k)) dx over [0,2]
    # And notice that the Shifted Integral will be computed as correlation 
    # of h with (1-h) with full mode (not valid here) so let's recompute properly
    # Actually we should be using scipy.signal.correlate with mode='full'
    # but with efficient direct computation for max
    for shift in range(n):
        overlap = 0.0
        for i in range(n):
            j = i + shift
            if j < n:
                overlap += h[i] * (1.0 - h[j])
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function - simple step
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization procedure
    max_iter = 200
    improvement_threshold = 1e-6
    
    it = 0
    
    while it < max_iter and (time.time() - start_time) < budget_s - 5.0:
        old_best_c5 = best_c5
        # Partition into even and odd indices
        even_indices = [i for i in range(n_points) if i % 2 == 0]
        odd_indices = [i for i in range(n_points) if i % 2 == 1]
        
        # Warm-start with current solution
        current_h = best_h.copy()
        
        # 1. Optimize even indices, keep odds fixed
        def optimize_evens():
            # Set odd indices of current_h to be kept fixed
            odd_vals = current_h[odd_indices]
            
            # Define objective function that depends only on even indices 
            def objective(even_vals):
                # Reconstruct full h
                h_temp = current_h.copy()
                h_temp[even_indices] = even_vals
                # Project to [0,1] and normalize to sum=target_sum
                # We need to also make sure sum = target_sum
                # This is easier to do with the sum constraint directly in constraint
                return evaluate_overlap(h_temp, dx)
                
            # Bound constraints
            bounds = [(0, 1) for _ in even_indices]
            # Constraints: sum of h should equal target_sum
            def sum_constraint(even_vals):
                h_temp = current_h.copy()
                h_temp[even_indices] = even_vals
                sum_diff = np.sum(h_temp) - target_sum
                return sum_diff
            
            cons = {'type': 'eq', 'fun': sum_constraint}
            
            # Minimize using SLSQP
            try:
                result = minimize(objective, current_h[even_indices], 
                                method='SLSQP', bounds=bounds, constraints=cons,
                                options={'ftol': 1e-10, 'maxiter': 200})
                if result.success:
                    h_new = current_h.copy()
                    h_new[even_indices] = result.x
                    # Ensure constraint is met by slight normalization
                    actual_sum = np.sum(h_new)
                    if actual_sum > 0:
                        h_new = h_new * (target_sum / actual_sum)
                    return h_new
                else:
                    return current_h
            except:
                return current_h

        # 2. Optimize odd indices, keep evens fixed (after possibly updating evens)
        def optimize_odds():
            even_vals = current_h[even_indices]
            
            def objective(odd_vals):
                h_temp = current_h.copy()
                h_temp[odd_indices] = odd_vals
                return evaluate_overlap(h_temp, dx)
                
            # Bound constraints
            bounds = [(0, 1) for _ in odd_indices]
            # Constraints: sum of h should equal target_sum
            def sum_constraint(odd_vals):
                h_temp = current_h.copy() 
                h_temp[odd_indices] = odd_vals
                sum_diff = np.sum(h_temp) - target_sum
                return sum_diff
            
            cons = {'type': 'eq', 'fun': sum_constraint}
            
            try:
                result = minimize(objective, current_h[odd_indices], 
                                method='SLSQP', bounds=bounds, constraints=cons,
                                options={'ftol': 1e-10, 'maxiter': 200})
                if result.success:
                    h_new = current_h.copy()
                    h_new[odd_indices] = result.x
                    # Ensure constraint is met by slight normalization
                    actual_sum = np.sum(h_new)
                    if actual_sum > 0:
                        h_new = h_new * (target_sum / actual_sum)
                    return h_new
                else:
                    return current_h
            except:
                return current_h
        
        # Perform alternating optimization
        outer_iter = 0
        while outer_iter < 10 and (time.time() - start_time) < budget_s - 5.0:
            h_new = optimize_evens()
            if evaluate_overlap(h_new, dx) < best_c5:
                best_h = h_new.copy()
                best_c5 = evaluate_overlap(h_new, dx)
            h_new = optimize_odds()
            if evaluate_overlap(h_new, dx) < best_c5:
                best_h = h_new.copy()
                best_c5 = evaluate_overlap(h_new, dx)
            outer_iter += 1
            
        # Periodic perturbation
        if (it % 20 == 0) and (time.time() - start_time) < budget_s - 10.0:
            # Add small random perturbations to maintain diversity
            perturb = 1e-4
            for i in range(n_points):
                sign = 1 if np.random.random() < 0.5 else -1
                current_h[i] += sign * perturb
            # Project to [0,1]
            current_h = np.clip(current_h, 0, 1)
            # Adjust magnitude to enforce sum constraint
            actual_sum = np.sum(current_h)
            if actual_sum > 0:
                current_h = current_h * (target_sum / actual_sum)
            if evaluate_overlap(current_h, dx) < best_c5:
                best_h = current_h.copy()
                best_c5 = evaluate_overlap(current_h, dx)
        
        if abs(best_c5 - old_best_c5) < improvement_threshold:
            break
        it += 1
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    actual_sum = np.sum(final_h)
    if actual_sum > 0:
        final_h = final_h * (target_sum / actual_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
