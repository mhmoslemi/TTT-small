import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization with perturbation
    max_iterations = 100
    perturbation_delta = 1e-3
    improvement_threshold = 1e-5
    perturbation_decay = 0.98
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb solution to escape local optima
        if iteration > 0:
            perturb = np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
            h_perturbed = h + perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            if current_sum > 0:
                h_perturbed = h_perturbed * (target_sum / current_sum)
            h = h_perturbed
            perturbation_delta *= perturbation_decay
        
        # Partition variables
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Block coordinate descent approach
        for phase in range(2):
            if phase == 0:
                # Optimize even indices
                fixed_indices = odd_indices
                optimize_indices = even_indices
            else:
                # Optimize odd indices  
                fixed_indices = even_indices
                optimize_indices = odd_indices
            
            # Create objective function for optimization
            def optimize_objective(h_subset):
                # Reconstruct full vector
                h_full = h.copy()
                h_full[optimize_indices] = h_subset
                
                # Apply constraints
                current_sum = np.sum(h_full)
                if current_sum > 0:
                    h_full = h_full * (target_sum / current_sum)
                
                # Evaluate objective
                return evaluate_overlap(h_full, dx)
            
            # Define bounds and constraints
            bounds = [(0, 1) for _ in optimize_indices]
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(h[fixed_indices]) - target_sum}]

            h_optimize = h[optimize_indices].copy()
            
            # Optimize using SLSQP
            try:
                result = minimize(optimize_objective, h_optimize, method='SLSQP',
                                 bounds=bounds, constraints=constraints,
                                 options={'maxiter': 100, 'ftol': 1e-8})
                if result.success:
                    h[optimize_indices] = np.clip(result.x, 0, 1)
                    # Re-normalize to ensure sum constraint
                    current_sum = np.sum(h)
                    if current_sum > 0:
                        h = h * (target_sum / current_sum)
            except:
                # If optimization fails, use current values
                pass

            # Check for improvement
            current_c5 = evaluate_overlap(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        if best_c5 < improvement_threshold:
            break

    if time.time() - start_time < budget_s - 5:

            return evaluate_overlap(h_vector, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        final_result = minimize(final_objective, best_h, method='SLSQP',

                             options={'maxiter': 200, 'ftol': 1e-10})

            refined_h = np.clip(final_result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)

                best_c5 = final_c5

    # Ensure feasibility
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)

    return (final_h.tolist(), final_c5, n_points)
