import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n_points):
    half = n_points // 2
    ramp = np.concatenate([np.linspace(0, 1, half), np.linspace(1, 0, n_points - half)])
    return ramp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Phase 1: Gradient descent with momentum
    momentum = 0.9
    learning_rate = 0.001
    decay_rate = 0.99
    velocity = np.zeros(n_points)
    
    # Phase 2: Coordinate-wise optimization
    coordinate_sampling_rate = 0.2
    max_no_improvement = 1000
    no_improvement_count = 0
    
    iter_count = 0
    total_iter = 20000
    
    while iter_count < total_iter and time.time() - start_time < budget_s - 1:
        # Phase 1: Gradient descent
        if iter_count < 10000:
            # Compute overlap
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
            
            # Find maximum overlap and its position
            max_pos = np.argmax(overlap)
            
            # Compute gradient approximation
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Forward difference
                h_plus = h.copy()
                h_plus[i] = min(1.0, h[i] + 0.001)
                h_plus = clip_and_renormalize(h_plus, target_sum)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                max_plus = np.max(overlap_plus)
                
                grad[i] = (max_plus - current_max) / 0.001
                
            # Momentum update
            velocity = momentum * velocity + (1 - momentum) * grad
            h_candidate = h - learning_rate * velocity
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            
            new_c5 = compute_c5(h_candidate, dx)
            if new_c5 < current_max:
                h = h_candidate.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                learning_rate *= decay_rate
            else:
                learning_rate *= 0.9
                no_improvement_count += 1
        else:
            # Phase 2: Coordinate-wise optimization
            update_count = max(1, int(n_points * coordinate_sampling_rate))
            update_indices = np.random.choice(n_points, update_count, replace=False)
            
            improved = False
            for i in update_indices:
                current_val = h[i]
                best_step = 0
                best_c5_check = current_max = compute_c5(h, dx)
                
                for step_size in [0.001, 0.005, 0.01, 0.05]:
                    # Try positive step
                    h_test = h.copy()
                    h_test[i] = min(1.0, current_val + step_size)
                    h_test = clip_and_renormalize(h_test, target_sum)
                    
                    c5_test = compute_c5(h_test, dx)
                    if c5_test < best_c5_check:
                        best_c5_check = c5_test
                        best_step = step_size
                        
                    # Try negative step
                    h_test = h.copy()
                    h_test[i] = max(0.0, current_val - step_size)
                    h_test = clip_and_renormalize(h_test, target_sum)
                    
                    c5_test = compute_c5(h_test, dx)
                    if c5_test < best_c5_check:
                        best_c5_check = c5_test
                        best_step = -step_size
                
                # Apply best step found
                if best_step != 0:
                    h[i] = np.clip(current_val + best_step, 0, 1)
                    h = clip_and_renormalize(h, target_sum)
                    improved = True
                    break
            
            if not improved:
                no_improvement_count += 1
            else:
                no_improvement_count = 0
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            break
            
        iter_count += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
