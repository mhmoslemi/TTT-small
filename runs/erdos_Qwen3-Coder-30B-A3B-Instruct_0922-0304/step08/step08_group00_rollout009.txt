import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Compute full correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    # Only consider the relevant part (shifts from -n+1 to n-1)
    n = len(h)
    mid = len(overlap) // 2
    relevant = overlap[mid - n + 1 : mid + n - 1]
    return np.max(relevant) * dx

    target_sum = n_points / 2

        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Objective function to minimize (C)
    def objective(c_and_h):
        # c is first element of array
        c = c_and_h[0]
        h_vec = c_and_h[1:]
        # Compute overlap
        overlap = np.correlate(h_vec, 1 - h_vec, mode="full")
        n = len(h_vec)
        mid = len(overlap) // 2
        relevant = overlap[mid - n + 1 : mid + n - 1]
        max_overlap = np.max(relevant) * dx
        return max_overlap
    
    # Optimization
    # Create initial guess with first element as an upper bound
    initial_guess = np.concatenate([[0.5], h])
    
    # Solve using SLSQP
    try:
        result = minimize(objective, initial_guess, method='SLSQP', 
                         bounds=[(0, 1)] + bounds,
