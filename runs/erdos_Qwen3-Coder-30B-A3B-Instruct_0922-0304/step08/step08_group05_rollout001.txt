import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_population(n_points, target_sum, population_size):
    """Create an initial population of feasible solutions."""
    population = []
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        # Add a simple piecewise constant solution
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Generate the rest randomly
    for _ in range(population_size - len(population)):
        # Generate with some Gaussian perturbations to ensure variety
        h = np.random.normal(0.5, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = random.sample(range(len(population)), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2):
    """Create offspring via average crossover."""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(h, mutation_rate=0.1, sigma=0.1):
    """Mutate an individual with Gaussian noise."""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, sigma, len(h))
    h = h + noise * mask
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    population_size = 50
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_generations = 1000
    elite_size = 5
    
    # Initialize population
    population = create_initial_population(n_points, target_sum, population_size)
    
    # Evaluate initial population
    fitnesses = [compute_c5_exact(individual, dx) for individual in population]
    
    # Tracking
    best_c5 = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    last_improvement = 0
    
    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if sorted_fitnesses[0] < best_c5:
            best_c5 = sorted_fitnesses[0]
            best_individual = sorted_population[0].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep the best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutate(offspring, mutation_rate=0.1, sigma=0.05)
            
            # Ensure constraints
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            
            new_population.append(offspring)
        
        # Replace population
        population = new_population
        
        # Re-evaluate
        fitnesses = [compute_c5_exact(individual, dx) for individual in population]
        
        generation += 1
    
    # Final check
    final_fitnesses = [compute_c5_exact(individual, dx) for individual in population]
    final_best_idx = np.argmin(final_fitnesses)
    final_best_c5 = final_fitnesses[final_best_idx]
    
    if final_best_c5 < best_c5:
        best_c5 = final_best_c5
        best_individual = population[final_best_idx].copy()
    
    return (best_individual.tolist(), best_c5, n_points)
