import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function - step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add small noise to avoid local optima
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient descent with Nesterov momentum
    max_iter = int(1e5)
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    patience = 100
    
    # For Nesterov momentum
    v = np.zeros_like(h)
    y = h.copy()
    
    prev_c5 = best_c5
    no_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        # We need the maximum value of overlap
        overlap = correlate(y, 1 - y, mode="full")
        max_k = np.argmax(overlap)
        
        # Invert correlate to get gradients
        # dC/dh_i = sum over all k of (1 - h_{(i+k)%n})
        # But since we're at max_k, we use that shift
        gradient = np.zeros_like(y)
        
        # Compute gradient for the max overlap case
        # For shift k, we have: sum_{i} y_i (1 - y_{i+k})
        #Gradient contributes to: dC/dy_i = (1 - y_{i-k})
        # But use reverse indexing for Cauchy convolution, checked 
        # Equivalent to: derivative of overlap at max_k
        # This is equivalent to a specific correlation with 1-h
        # But we are dealing with a full convolution 
        # We actually want: sum_{i} y_i (1 - y_{i+k}) = sum_{i,k} y_i (1 - y_{i+k})
        # Let's directly compute what leads to max overlap - find direction of steepest descent
        
        # Let's simplify by taking direction that decreases max overlap most
        # Try: compute gradient of overlap with shift k that gives max
        
        max_shift = max_k - (2*n_points-1)//2 # shift to get index
        
        # Gradient wrt to position i
        for j in range(n_points):
            # For shift k=0, all h[j] contributes to h[j](1-h[j]) 
            # But we are dealing with correlation  = sum of h[i] * (1 - h[i+k])
            # So d C_k / d h_i = (1 - h[i+k]) if k gives max overlap
            
            # If k is the shift that maximizes overlap, we can use that explicitly
            # The contribution to max overlap from point j is:
            # dx = [0,1] - current h vs previous h
            # Exploring a simpler gradient method
            # Influence of h[j] on C_max: contribution is (1 - h[j+max_shift])

            # Actually, for shift k that maximizes C_k:
            # C_k = sum_i h[i] * (1 - h[i+k]) 
            # dC_k/dh[j] = (1 - h[j+k])
            # But we want the max over all k
            
            # gradient = sum_k dC_k/dh_j  = sum_k (1-h[j+k]) where we sum over all k
            # Hmm, but this is not the right interpretation.
            # We want to minimize max_k C_k = max_k sum_i h[i] (1 - h[i+k]) 
            # Gradient of max is tricky. Let's follow the direction of steepest descent:
            # Find max k, then compute gradient of that function
            
            # We'll compute the partial of overlap function which maximizes
            # d/dh_i of sum_j h_j (1 - h_{j + max_shift})  
            # = 1-h_{i+max_shift} - 0 -1-h_{i-max_shift}, but that's not right...
            # Advise: compute gradient directly as sum of derivatives of each term
            if max_shift >= 0:
                gradient[j] = 1 - y[(j - max_shift) % n_points] 
            else:
                # There is a complication with indexing 
                jshift = (j + max_shift) % n_points
                gradient[j] = 1 - y[jshift]
        
        # Nesterov momentum
        v = momentum * v - step_size * gradient
        y = y + v
        
        # Project onto feasible set [0,1]
        y = np.clip(y, 0, 1)
        
        # Correct the sum constraint
        y_sum = np.sum(y)
        if y_sum > 0:
            y = y * (target_sum / y_sum)
        
        # Evaluate
        current_c5 = evaluate_overlap(y, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = y.copy()
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Adjust step size
        if abs(current_c5 - prev_c5) < tol:
            step_size *= 0.5
            if step_size < min_step_size:
                step_size = min_step_size
        prev_c5 = current_c5
            
        # Early stop on stagnation
        if no_improvement > patience:
            # Restart with new initialization
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 0:
                h = h * (target_sum / current_sum)
            y = h.copy()
            v = np.zeros_like(h)
            no_improvement = 0
            step_size = 0.1
            
    # One final refinement with local optimization  
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Simple descent from best
    import scipy.optimize as opt
    res = opt.minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=constraints, 
                      options={'maxiter': 200, 'ftol': 1e-8})
    
    if res.success:
        refined_h = np.clip(res.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
