import numpy as np
from scipy.signal import correlate
import time

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_population(population, dx):
    """Evaluate all candidates in population."""
    fitness = []
    for h in population:
        c5 = compute_c5(h, dx)
        fitness.append(c5)
    return np.array(fitness)

def selection(population, fitness, num_parents):
    """Select top performers as parents."""
    indices = np.argsort(fitness)
    return [population[i] for i in indices[:num_parents]]

def crossover(parent1, parent2, crossover_prob):
    """Uniform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, mutation_rate, target_sum):
    """Mutate h with given rate."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] = np.random.rand()
    return repair_h(h, target_sum)

def gradient_descent_refine(h, dx, step_size=0.01):
    """Refine h using gradient descent."""
    h = h.copy()
    # Approximate gradient
    overlap = correlate(h, 1 - h, mode="full") * dx
    max_overlap = np.max(overlap)
    
    # Simple gradient ascent step
    grad = (1 - 2 * h) / len(h)
    h_new = h - step_size * grad
    h_new = repair_h(h_new, len(h) / 2)
    new_c5 = compute_c5(h_new, dx)
    
    if new_c5 < max_overlap:
        return h_new
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    num_generations = 2000
    crossover_prob = 0.7
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
        population.append(h)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    # Add variations of the seed solution
    while len(population) < pop_size:
        h = population[0].copy()
        # Add small random noise
        for i in range(len(h)):
            if np.random.rand() < 0.1:
                h[i] = np.random.rand()
        h = repair_h(h, target_sum)
        population.append(h)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Optimization loop
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while gen < num_generations and time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = evaluate_population(population, dx)
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Increase mutation rate to escape local minima
            mutation_rate = min(0.3, mutation_rate * 1.1)
            
        # Selection
        num_parents = pop_size // 2
        parents = selection(population, fitness, num_parents)
        
        # Create new population
        new_population = [parents[0].copy()]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            ch1, ch2 = crossover(p1, p2, crossover_prob)
            ch1 = mutate(ch1, mutation_rate, target_sum)
            ch2 = mutate(ch2, mutation_rate, target_sum)
            
            # Gradient descent refinement
            ch1 = gradient_descent_refine(ch1, dx)
            ch2 = gradient_descent_refine(ch2, dx)
            
            new_population.extend([ch1, ch2])
            
        # Keep only the required number of individuals
        population = new_population[:pop_size]
        
        gen += 1
    
    return (best_h.tolist(), best_c5, n_points)
