import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    
    # QP formulation
    # Variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points / 2]
    constraints += [0 <= h_var, h_var <= 1]
    
    # Overlap constraints for all shifts
    for k in range(-n_points + 1, n_points):
        # Compute valid indices
        expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h_var[i] * (1 - h_var[j])
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            t_opt = t.value
            final_c5 = t_opt
        else:
            # Fallback to parent if no solution found
            h_opt = h
            final_c5 = evaluate_c5(h_opt, dx)
    except:
        # Fallback to parent construction if error
        h_opt = h
        final_c5 = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
