import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_val = 0
    for k in range(n):
        # Compute convolution at shift k
        conv = 0
        for i in range(n):
            j = (i + k) % n
            conv += h[i] * (1 - h[j])
        max_val = max(max_val, conv)
    return max_val * dx
