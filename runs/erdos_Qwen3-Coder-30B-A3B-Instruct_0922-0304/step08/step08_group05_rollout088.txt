import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize and clip initial values
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5 and find active lags
        overlap = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlap) * dx
        max_idx = np.argmax(overlap)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient for the maximum overlap
        subgrad = np.zeros(n_points)
        
        # The subgradient of max_k (h * (1-h))[k] is the derivative of that term
        # that achieves the maximum at the current point
        for i in range(n_points):
            # Contribution to subgradient from different lag terms
            # We check which lags contribute to the maximum
            # This is a simplification of true subgradient calculation
            contrib = 0.0
            # Only contributing if indices are valid for both forward and backward
            for k in range(n_points):
                if 0 <= i+k < n_points:
                    contrib += (1 - 2 * h[i+k]) 
                if 0 <= i-k < n_points:
                    contrib += (1 - 2 * h[i-k])
            
            # We scale back based on which positions compute the actual max
            # For more precision, we utilize a smoothing approach
            subgrad[i] = contrib
        
        # Add momentum to descent
        velocity = momentum * velocity - learning_rate * subgrad
        h = h + velocity
        
        # Project back to constraints
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        iteration += 1
        
        # Adaptive learning rate decrease
        if iteration % 100 == 0:
            learning_rate *= 0.995
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
