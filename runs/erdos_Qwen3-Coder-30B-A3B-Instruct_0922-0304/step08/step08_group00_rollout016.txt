import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Gradient descent with Nesterov momentum
    max_iter = 100000
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    improvement_tolerance = 1e-6
    
    # For Nesterov momentum
    z = h.copy()
    y = h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    prev_c5 = best_c5 + 1.0  # Initialize to ensure at least one iteration
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current overlap for each lag
        overlap = correlate(y, 1 - y, mode="full")
        max_lag_idx = np.argmax(overlap)
        max_overlap = overlap[max_lag_idx] * dx
        
        # Compute gradient: dC/dh_i = sum of (1 - h_j) for all j where j = i - k (mod n)
        # But we focus on just the maximizer lag
        gradient = np.zeros_like(y)
        lag = max_lag_idx - (n_points - 1)
        for i in range(n_points):
            j = (i - lag) % n_points
            gradient[i] += (1 - y[j])
        
        # Nesterov update
        z = y - step_size * gradient
        y = z + momentum * (z - h)
        
        # Project onto feasible set [0,1]
        y = np.clip(y, 0, 1)
        
        # Rescale to satisfy sum constraint
        current_sum = np.sum(y)
        if current_sum > 0:
            y = y * (target_sum / current_sum)
        
        # Update h to the latest y (for next iteration's momentum)
        h = y.copy()
        
        # Evaluate the new function value
        current_c5 = evaluate_overlap(y, dx)
        
        # Track improvement
        if current_c5 < best_c5:
            best_h = y.copy()
            best_c5 = current_c5
            no_improvement_count = 0
            prev_c5 = best_c5
        else:
            no_improvement_count += 1
            # Gradually decrease step size
            step_size = max(step_size * 0.9995, min_step_size)
        
        # Deal with stagnation
        if no_improvement_count > 500:
            step_size = max(step_size * 0.5, min_step_size)
            if step_size < min_step_size:
                break
            no_improvement_count = 0
        
        if abs(prev_c5 - current_c5) < improvement_tolerance:
            step_size = max(step_size * 0.9, min_step_size)
            if step_size < min_step_size:
                break
            prev_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
