def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_numerical_gradient(h, dx, n_points):
    """Compute gradient for h in terms of overlap function."""
    overlap_full = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(overlap_full)
    
    # For gradient computation
    grad = np.zeros(n_points)
    
    # This is a simplified gradient of the max overlap
    # The actual update is more complex as overlap depends on convolution
    
    # Simple finite difference gradient estimator
    eps = 1e-6
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = np.clip(h_minus, 0, 1)
        h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
        
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    return grad

def local_refinement(h, dx, n_points, max_iter=5000, lr=0.005):
    """Gradient descent refinement"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if current_c5 <= 0.3807:
            break
            
        grad = compute_numerical_gradient(h, dx, n_points)
        h = h - lr * grad
        
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if i > 0 and i % 1000 == 0:
            lr *= 0.9
    
    return best_h, best_c5

def run(seed=42, budget_s=600):

    n_points = 100

        h = h * (n_points / 2) / np.sum(h)
    else:

        h = solve_convex_relaxation(n_points, dx)
        
    # Local refinement
    final_h, final_c5 = local_refinement(h, dx, n_points, max_iter=5000, lr=0.005)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s - 1 and final_c5 > 0.3807:
        # Try refinement again with different LR
        h_try = final_h.copy()
        _, new_c5 = local_refinement(h_try, dx, n_points, max_iter=3000, lr=0.001)
        if new_c5 < final_c5:
            final_h = h_try
            final_c5 = new_c5
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        # Final evaluation
        final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
