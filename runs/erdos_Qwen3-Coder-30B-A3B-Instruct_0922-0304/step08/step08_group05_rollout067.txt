import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Ensure the sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    epsilon = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Compute overlaps for all lags and find the maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        
        # Identify the worst contributing positions
        # Use the maximum overlap contribution
        max_val = overlap[max_lag_idx]
        
        # Perturb h at positions that contribute to the maximum overlap
        # To reduce this overlap, shift mass from h[i] to h[i+k] 
        # where k is the lag that causes max overlap
        # Find i such that h[i]*1 contributes most to this overlap
        
        # Quick greedy adjustment
        # We consider nearby indices that contribute
        half_size = n_points // 2
        # Check around the position that contributes most to overlap
        for i in range(max(0, max_lag_idx - half_size), min(n_points, max_lag_idx + half_size)):
            if i >= n_points: continue
            # For an overlap at lag=i, we're looking at h[j] * (1 - h[j+i]) 
            # Where j + i = max_lag_idx (since we go from -n to 2*n-2)
            # So we're dealing with j = max_lag_idx - i
            
            # Let's determine how to reduce the maximum overlap
            # One way: decrease h[j] by epsilon, increase h[j+i] by epsilon
            # To make this practical, we'll adjust 2 elements
            # The critical element depends on the maximum position of overlap
            
            # But a simpler approach to find most impacted index:
            # Find all positions that when used as the current one, 
            # could cause the maximum overlap
            
            # Look at the actual cause of max overlap: encode back with lag
            actual_lag = max_lag_idx - n_points + 1  # Pandas-like
            if actual_lag < 0:  # Positive lag (shift to right)
                j1 = 0
                j2 = j1 + actual_lag
            else:  # Negative or zero lag
                j2 = 0
                j1 = j2 - actual_lag  
            
            # Now we have two indices: j1 and j2 such that h[j1](1-h[j2]) gives max partial
            # But this is not very precise. Better: just greedily adjust based on more info
            
            # Let's just focus on absolute adjustments
            if i < n_points - 1:
                adjust_amount = min(epsilon, h[i] * 0.5)
                h[i] -= adjust_amount
                h[(i + actual_lag) % n_points] += adjust_amount     # handle wraparound
                
        # Ensure constraint after adjustments
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Occasionally reduce learning rate
        if iteration % 100 == 0:
            epsilon *= 0.995
            
        iteration += 1
        
        # Minor final pad adjustment
        if iteration % 50 == 0:
            # Enforce exact sum constraint
            h = h * (target_sum / np.sum(h))
    
    final_c5 = compute_c5_exact(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
