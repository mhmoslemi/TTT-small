import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # PSO parameters
    n_particles = 50
    max_iterations = 1000
    w = 0.8
    c1 = c2 = 2.0
    max_velocity = 0.05
    tolerance = 1e-6
    shaking_strength = 0.05
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(n_particles - 1):
        pert = np.random.uniform(-0.1, 0.1, n_points)
        new_h = h + pert
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        particles.append(new_h)
    
    # Evaluate initial fitness
    fitness = [evaluate_overlap(p, dx) for p in particles]
    
    # Initialize personal bests and global best
    p_best = particles.copy()
    p_best_fitness = fitness.copy()
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    # Velocity initialization
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # PSO loop
    no_improvement_iter = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            new_pos = particles[i] + velocities[i]
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = np.sum(new_pos)
            if current_sum > 0:
                new_pos = new_pos * (target_sum / current_sum)
            
            # Evaluate new fitness
            new_fitness = evaluate_overlap(new_pos, dx)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_pos.copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < g_best_fitness:
                    g_best = new_pos.copy()
                    g_best_fitness = new_fitness
                    no_improvement_iter = 0
            else:
                no_improvement_iter += 1
            
            particles[i] = new_pos
            
        # Apply shaking if no improvement
        if no_improvement_iter > 50:
            for i in range(n_particles):
                shake = np.random.uniform(-shaking_strength, shaking_strength, n_points)
                particles[i] = np.clip(particles[i] + shake, 0, 1)
                current_sum = np.sum(particles[i])
                if current_sum > 0:
                    particles[i] = particles[i] * (target_sum / current_sum)
            no_improvement_iter = 0
    
    # Local refinement using SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, g_best, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < g_best_fitness:
            g_best = refined_h
            g_best_fitness = final_c5
    
    # Final validation
    final_h = np.clip(g_best, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
