import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 1000
    iter_count = 0
    
    # Use the same optimization approach but try to make it more stable
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Start with a relaxed optimization approach
        # Create optimization variables
        h_var = cp.Variable(n_points, name='h')
        z = cp.Variable(name='z')
        
        # Constraints
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum, z >= 0]
        
        # Add overlap constraints using a simplified approach
        # We do this by approximating the convolution with linear constraints
        for k in range(n_points):
            # Approximate the shifted overlap as a linear expression
            # This is a rough approximation but good for optimization
            expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    expr += h_var[i] * (1 - h_var[i + k])
            constraints.append(expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(warm_start=True, verbose=False, solver=cp.GLPK)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed = h_var.value
                if h_relaxed is not None:
                    # Apply rounding heuristic to get a better feasible solution
                    h_new = h_relaxed.copy()
                    
                    # Simple rounding: shift towards binary values with some tolerance
                    # But maintain the sum constraint
                    h_new = np.clip(h_new, 0, 1)
                    current_sum = np.sum(h_new)
                    h_new = h_new * (target_sum / current_sum)
                    h_new = np.clip(h_new, 0, 1)
                    
                    # Density correction
                    if np.sum(h_new) < target_sum:
                        # Add some to the smallest values to make up the deficit
                        diff = target_sum - np.sum(h_new)
                        indices = np.argsort(h_new)
                        for i in range(int(diff)):
                            if indices[i] < len(h_new):
                                h_new[indices[i]] += 1.0 / n_points
                    elif np.sum(h_new) > target_sum:
                        # Remove some from the largest values to bring down
                        diff = np.sum(h_new) - target_sum
                        indices = np.argsort(h_new)[::-1]
                        for i in range(int(diff)):
                            if indices[i] < len(h_new):
                                h_new[indices[i]] -= 1.0 / n_points
                    h_new = np.clip(h_new, 0, 1)
                    
                    # Calculate overlap value for the new solution
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_new.copy()
                    
                    h = h_new.copy()
                    
                    # If solution is good, try a steepest ascent step to further improve
                    # Simple gradient algorithm to refine the solution
                    for _ in range(5):
                        if time.time() - start_time > budget_s - 1:
                            break
                        # Consider small gradient updates
                        gradients = np.zeros(n_points)
                        for i in range(n_points):
                            grad_sum = 0
                            for k in range(n_points):
                                if i + k < n_points:
                                    grad_sum += 1 - 2 * h[i] - (1 - 2 * h[i + k])
                                if i - k >= 0:
                                    grad_sum += 1 - 2 * h[i] - (1 - 2 * h[i - k])
                            gradients[i] = grad_sum
                        
                        # Update
                        update_step = 0.01
                        for i in range(n_points):
                            new_val = h[i] + update_step * gradients[i]
                            h[i] = np.clip(new_val, 0, 1)
                        
                        # Renormalize
                        h = h * (target_sum / np.sum(h))
                        
                        # Measure improvement
                        new_c5_check = compute_c5(h, dx)
                        if new_c5_check < best_c5:
                            best_c5 = new_c5_check
                            best_h = h.copy()
                        
                        if best_c5 <= 0.3807:
                            break
                    
                    if best_c5 <= 0.3807:
                        break
        except Exception as e:
            pass
        
        iter_count += 1
        # Occasionally check if we should recompute 
        if iter_count % 100 == 0:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
