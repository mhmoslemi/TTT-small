```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed(h_r, n_points, dx, max_time=10):
    """Solve relaxed optimization problem using CVXPY."""
    start_time = time.time()
    try:
        # Create variables
        h = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Objective
        # This is a bit complex because we need to minimize the max overlap term
        # We'll approximate by creating a constraint for large k
        # In practice, we only verify with a subset of k values for efficiency
        objective = cp.Minimize(z)
        
        # Add one constraint to bound from above
        # This is a simplified, approximate approach
        expr = 0
        for i in range(n_points//4, 3*n_points//4):
            expr += h[i] * (1 - h[i])
        constraints.append(z >= expr * dx)
        prob = cp.Problem(objective, constraints)
        
        # Solve with limited time
        prob.solve(solver=cp.GLPK, max_iters=5000, time_limit=max_time)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, prob.value, True
        else:
            return h_r, np.inf, False
            
    except Exception as e:
        return h_r, np.inf, False

def round_to_binary(h_r, target_sum):
    """Round continuous solution to binary solution."""
    h_b = np.zeros_like(h_r)
    median_val = np.median(h_r)
    h_b[h_r > median_val] = 1.0
    current_sum = np.sum(h_b)
    
    # Adjust to get exactly target_sum
    diff = int(current_sum - target_sum)
    
    if diff > 0:
        # Need to flip some 1s to 0s
        indices = np.argsort(h_r)
        for idx in indices[:diff]:
            h_b[idx] = 0.0
    elif diff < 0:
        # Need to flip some 0s to 1s
        indices = np.argsort(h_r)[::-1]
        for idx in indices[:abs(diff)]:
            h_b[idx] = 1.0
            
    return h_b

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 50
    no_improvement_count = 0
    
    for i in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Relaxation
        h_relaxed, relaxed_val, succeeded = solve_relaxed(h, n_points, dx)
        if not succeeded:
            h_relaxed = h.copy()
            
        # Rounding and adjustment
        h_binary = round_to_binary(h_relaxed, target_sum)
        
        # Validate and compute C5
        new_c5 = compute_c5(h_binary, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_binary.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update h for next iteration
        h = h_binary.copy()
        
        if best_c5 <= 0.3807:
            break
        
        if no_improvement_count >= 10:
            break
            
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
