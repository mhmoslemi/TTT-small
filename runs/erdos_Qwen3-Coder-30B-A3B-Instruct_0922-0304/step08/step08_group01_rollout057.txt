import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    iter_count = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Formulate the convex relaxation
        # Variables
        h_vars = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == target_sum]
        
        # Relax the overlap constraints
        # For each shift k, we have:
        # overlap_k = sum_{x} h[x] * (1 - h[x+k]) <= z
        # Approximate h[x] * h[x+k] with convex envelope
        for k in range(n_points):
            expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    expr += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            h_relaxed = h_vars.value
        except:
            h_relaxed = h.copy()
        
        # Rounding heuristic
        # Threshold rounding
        h_rounded = np.zeros(n_points)
        threshold = 0.5
        
        # Simple threshold
        for i in range(n_points):
            if h_relaxed[i] > threshold:
                h_rounded[i] = 1.0
            else:
                h_rounded[i] = 0.0
        
        # Adjust to meet the sum constraint
        current_sum = np.sum(h_rounded)
        diff = target_sum - current_sum
        
        # Make small adjustments to meet the sum
        if diff > 0:
            # Find indices with 0 values to change to 1
            zero_indices = np.where(h_rounded == 0)[0]
            if len(zero_indices) > 0:
                for i in range(int(diff)):
                    if i < len(zero_indices):
                        h_rounded[zero_indices[i]] = 1.0
        elif diff < 0:
            # Find indices with 1 values to change to 0
            one_indices = np.where(h_rounded == 1)[0]
            if len(one_indices) > 0:
                for i in range(int(-diff)):
                    if i < len(one_indices):
                        h_rounded[one_indices[i]] = 0.0
        
        # Ensure precise constraint satisfaction
        h_rounded = h_rounded * (target_sum / np.sum(h_rounded))
        h_rounded = np.clip(h_rounded, 0, 1)
        
        # Evaluate
        new_c5 = compute_c5(h_rounded, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_rounded.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Use the rounded solution as next iteration starting point
        h = h_rounded.copy()
        iter_count += 1
        
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
