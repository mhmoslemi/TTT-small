import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    # The gradient with respect to each component is approximately (1 - 2*h[i]) / n_points
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points) * 0.5
    h = repair_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic Algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_generations = 2000
    no_improvement_limit = 100
    
    # Initialize population with diverse individuals
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = np.random.rand(n_points) * 0.5
        individual = repair_h(individual, target_sum)
        population.append(individual)
    
    best_overall = best_c5
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_overall:
            best_overall = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_overall <= 0.3807:
            break
            
        if no_improvement_count > no_improvement_limit:
            # Increase mutation rate
            mutation_rate = min(0.5, mutation_rate * 1.2)
            no_improvement_count = 0
        
        # Selection (top 50%)
        selected = population[:population_size // 2]
        new_population = selected.copy()
        
        # Crossover and Mutation
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Repair
                offspring1 = repair_h(offspring1, target_sum)
                offspring2 = repair_h(offspring2, target_sum)
                
                # Gradient descent refinement
                gradient1 = compute_gradient(offspring1, dx)
                step_size = 0.001
                offspring1 -= step_size * gradient1
                offspring1 = repair_h(offspring1, target_sum)
                
                gradient2 = compute_gradient(offspring2, dx)
                offspring2 -= step_size * gradient2
                offspring2 = repair_h(offspring2, target_sum)
                
                new_population.extend([offspring1, offspring2])
            else:
                # Mutation
                individual = np.random.choice(selected)
                mutated = individual.copy()
                indices_to_mutate = np.random.choice(n_points, 
                                                   size=int(mutation_rate * n_points), 
                                                   replace=False)
                for i in indices_to_mutate:
                    mutated[i] += np.random.normal(0, 0.01)
                mutated = repair_h(mutated, target_sum)
                
                # Gradient descent refinement
                gradient = compute_gradient(mutated, dx)
                step_size = 0.001
                mutated -= step_size * gradient
                mutated = repair_h(mutated, target_sum)
                
                new_population.append(mutated)
        
        # Keep only population_size individuals
        population = new_population[:population_size]
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
