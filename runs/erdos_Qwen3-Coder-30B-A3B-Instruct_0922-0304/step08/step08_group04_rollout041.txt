import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Partition-based search approach
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Track improvements
    last_improvement = 0
    no_improvement_count = 0
    
    # Main optimization loop
    max_iter = 5000
    for iter_count in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Every 100 iterations, try random restart
        if iter_count % 100 == 0 and iter_count > 0:
            if no_improvement_count > 10:  # No improvement in recent iterations
                # New random restart
                h = np.random.rand(n_points).astype(np.float64)
                h = normalize_h(h, n_points / 2)
                no_improvement_count = 0
                
        # Vary number of segments for exploration
        n_segments = np.random.randint(3, min(12, n_points//4 + 1))
        
        # Create segments
        segments = np.linspace(0, 2, n_segments + 1)
        segment_values = np.random.rand(n_segments).astype(np.float64)
        
        # Fill h in segments
        h_new = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(segments[i] / dx)
            end_idx = int(segments[i+1] / dx)
            if end_idx > start_idx:
                h_new[start_idx:end_idx] = segment_values[i]
        
        # Ensure we have the right total sum
        h_new = normalize_h(h_new, n_points / 2)
        
        # Evaluate with current h
        current_c5 = evaluate_c5(h_new, dx)
        
        # Improvement check
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            last_improvement = iter_count
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping if we've hit target
        if best_c5 <= 0.3807:
            break
            
        # Occasionally, do some gradient-like adjustment
        if iter_count % 50 == 0:
            # Simple gradient estimation
            grad = np.zeros_like(h_new)
            eps = 1e-4
            h_plus = h_new.copy()
            for i in range(n_points):
                h_plus[i] += eps
                if h_plus[i] <= 1:
                    c5_plus = evaluate_c5(h_plus, dx)
                    grad[i] = (c5_plus - current_c5) / eps
                h_plus[i] = h_new[i]  # restore
            
            # Update with gradient
            update = 0.005 * grad
            h_new = h_new - update
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
