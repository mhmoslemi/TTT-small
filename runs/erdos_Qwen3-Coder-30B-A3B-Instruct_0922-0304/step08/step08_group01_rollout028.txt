import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

        return compute_c5(h, dx)

    def sum_constraint(h):
        return np.sum(h) - target_sum
    
    constraints = [
        {'type': 'eq', 'fun': sum_constraint}
    ]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
