```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient with respect to given shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlation
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        # Convert index back to shift
        shift = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h_current, shift, n_points)
        
        # Update
        h_new = h_current - step_size * grad
        h_new = project_h(h_new, target_sum)
        
        # Check improvement
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update h_current
        h_current = h_new
        
        # Adapt step size
        if no_improvement_count > 50:
            step_size *= 0.95
            
        # Occasionally add noise to escape local minima
        if no_improvement_count > 100 and np.random.random() < 0.1:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, target_sum)
            no_improvement_count = 0
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
