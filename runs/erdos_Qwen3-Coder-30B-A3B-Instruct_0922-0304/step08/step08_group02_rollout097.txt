import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample shifts for QP constraints"""
    return np.random.randint(0, 2*n - 1, size=sample_size)
