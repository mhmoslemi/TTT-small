import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, dx, target_sum):
    """Project h onto feasible set."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, dx, target_sum)
    else:
        # Initialize with a triangular ramp
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x <= 1:
                h[i] = x
            else:
                h[i] = 2 - x
        h = project_h(h, dx, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Define search parameters
    max_iter = 10000
    step_size = 0.02
    momentum = 0.5

    # Momentum buffer for updates
    velocity = np.zeros(n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Compute gradient for each coordinate
        gradients = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = project_h(h_plus, dx, target_sum)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            h_minus = project_h(h_minus, dx, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            gradients[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps)

        # Update with momentum
        velocity = momentum * velocity - step_size * gradients
        h_new = h + velocity
        h_new = project_h(h_new, dx, target_sum)
        
        # Check if the new h improves the solution
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_max:
            h = h_new.copy()
            step_size = min(0.1, step_size * 1.01)  # Adaptive step size
        else:
            step_size *= 0.9  # Reduce step size if no improvement
            
        iter_count += 1

    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
