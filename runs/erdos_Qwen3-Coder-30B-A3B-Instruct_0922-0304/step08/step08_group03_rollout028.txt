import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create triangular ramp function."""
    mid = n // 2
    return np.append(np.linspace(0, 1, mid), np.linspace(1, 0, n - mid))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose n_points
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available, else use triangular ramp
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    # Initialize optimization variables
    velocity = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iter = 20000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        overlap_max_pos = np.argmax(overlap)
        
        # Compute gradient
        gradient = (1 - 2 * h) * np.sign(overlap_max) / n_points
        
        # Apply momentum
        velocity = momentum * velocity + learning_rate * gradient
        
        # Try update
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
        else:
            # Reduce step size
            learning_rate *= shrink_factor
            if learning_rate < 1e-8:
                break
                
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
