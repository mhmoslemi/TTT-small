import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed, target_sum):
    """Convert relaxed solution to binary vector."""
    # Use median as threshold
    threshold = np.median(h_relaxed)
    h_binary = (h_relaxed >= threshold).astype(float)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h_binary)
    deficit = current_sum - target_sum
    
    if deficit > 0:
        # Need to flip deficit number of 1s to 0s
        # Flip the smallest values that are >= threshold
        indices = np.where(h_binary == 1)[0]
        # Sort by value in h_relaxed to find smallest ones
        sort_indices = np.argsort(h_relaxed[indices])
        to_flip = indices[sort_indices[:int(deficit)]]
        h_binary[to_flip] = 0
    elif deficit < 0:
        # Need to flip |deficit| number of 0s to 1s
        # Flip the largest values that are < threshold
        indices = np.where(h_binary == 0)[0]
        sort_indices = np.argsort(h_relaxed[indices])[::-1]
        to_flip = indices[sort_indices[:int(-deficit)]]
        h_binary[to_flip] = 1
    
    return h_binary

def solve_lp_relaxation(h_init, n_points, dx):
    """Solve LP relaxation for the problem."""
    h_r = Variable(n_points, name='h_r')
    z = Variable(1, name='z')
    
    # Objective
    objective = Minimize(z)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(sum(h_r) == n_points/2)
    
    # Bounding constraints
    constraints.append(0 <= h_r)
    constraints.append(h_r <= 1)
    
    # Overlap constraints for all shifts
    for k in range(-n_points+1, n_points):
        if k >= 0:
            # Positive shift
            expr = 0
            for x in range(n_points - k):
                expr += h_r[x] * (1 - h_r[x + k])
            constraints.append(expr * dx <= z)
        else:
            # Negative shift
            expr = 0
            for x in range(n_points + k):
                expr += h_r[x] * (1 - h_r[x - k])
            constraints.append(expr * dx <= z)
    
    # Formulate and solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=MOSEK, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return h_r.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 40
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Solve relaxed problem
        h_relaxed = solve_lp_relaxation(h, n_points, dx)
        if h_relaxed is not None:
            # Get binary version
            h_binary = round_to_binary(h_relaxed, target_sum)
            
            # Check if better
            c5_binary = compute_c5(h_binary, dx)
            if c5_binary < best_c5:
                best_c5 = c5_binary
                best_h = h_binary.copy()
                
            # Use this as new starting point
            h = h_binary.copy()
        else:
            # If LP fails, continue with current h
            pass
        
        # Small random perturbations to escape traps
        if iter_count % 5 == 0:
            # Perturb a few values randomly
            indices = np.random.choice(n_points, size=n_points//10, replace=False)
            for i in indices:
                perturb = np.random.normal(0, 0.05)
                h[i] = np.clip(h[i] + perturb, 0, 1)
            h = normalize_h(h, target_sum)
            
            c5_current = compute_c5(h, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
                
        iter_count += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
