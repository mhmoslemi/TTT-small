import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy().astype(np.float64)
    else:

    # Normalize to target sum
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    # Objective function for optimization
    def objective(h):
        return compute_c5(h, dx)
    
    # Set up optimization
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Run optimization
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options=options,
            callback=lambda x: None,  # No callback
        )
        
        if result.success:
            h_opt = result.x
            # Ensure the result is within bounds
            h_opt = np.clip(h_opt, 0, 1)
            # Renormalize
            current_sum = np.sum(h_opt)
            if current_sum > 0:
                h_opt = h_opt * (target_sum / current_sum)
            final_c5 = compute_c5(h_opt, dx)
            if final_c5 < best_c5:
                best_h = h_opt
                best_c5 = final_c5
    except:
        # If optimization fails, keep the initial solution
        pass
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
