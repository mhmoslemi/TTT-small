import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Ensure h is within [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n_points):
    """Create a triangular ramp as initial guess."""
    half = n_points // 2
    up = np.arange(1, half + 1)
    down = np.arange(half, 0, -1)
    if n_points % 2 == 1:
        h = np.concatenate([up, [half + 1], down])
    else:
        h = np.concatenate([up, down])
    return h / np.sum(h) * (n_points / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
    
    # Parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    p = 0.2  # Coordinate sampling rate
    max_iter_phase1 = 10000
    max_iter_phase2 = 10000
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Phase 1: Gradient descent with momentum
    for i in range(max_iter_phase1):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute pseudo-gradient for the position of max overlap
        # This is a simplified approximation
        if max_pos < len(h):
            # This is a tad imprecise but computationally cheap
            grad = np.zeros_like(h)
            # Simply increase gradient near position max_pos
            # based on where in the correlation it occurs
            # This is a heuristic to guide towards low correlation
            shift = max_pos
            for j in range(len(h)):
                if j < len(h) - shift:
                    grad[j] += (1 - 2 * h[j]) * (1 - h[j + shift])
                else:
                    # Handle wraparound; not quite right but fast for now
                    pass
        else:
            # This branch is degenerate but safe
            grad = np.ones_like(h) - h

        # Update with momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_candidate = h - alpha * momentum
        
        # Project
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate
        new_c5 = compute_c5(h_candidate, dx)
        if new_c5 < overlap_max:
            h = h_candidate
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            alpha *= gamma
        else:
            alpha *= gamma
        
        if best_c5 <= 0.3807:
            break

    # Phase 2: Coordinate-wise updates
    if time.time() - start_time < budget_s - 1:
        for i in range(max_iter_phase2):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Sample coordinates
            num_sample = max(1, int(p * n_points))
            sample_indices = np.random.choice(n_points, num_sample, replace=False)
            
            improved = False
            for idx in sample_indices:
                current_val = h[idx]
                # Try small perturbations
                perturbations = [-0.01, 0.01, -0.005, 0.005]
                
                best_h_new = h.copy()
                best_c5_new = compute_c5(h, dx)
                
                for delta in perturbations:
                    h_new = h.copy()
                    h_new[idx] = np.clip(current_val + delta, 0, 1)
                    h_new = clip_and_renormalize(h_new, target_sum)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5_new:
                        best_c5_new = c5_new
                        best_h_new = h_new.copy()
                        
                if best_c5_new < compute_c5(h, dx):
                    h = best_h_new
                    improved = True
                    if best_c5_new < best_c5:
                        best_c5 = best_c5_new
                        best_h = h.copy()
                    
            if not improved:
                break
                
            if best_c5 <= 0.3807:
                break

    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
