import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighborhood(h, n_candidates=10):
    """Generate neighborhood by flipping or swapping elements."""
    n = len(h)
    candidates = []
    
    for _ in range(n_candidates):
        h_new = h.copy()
        
        # 30% chance of flipping, 70% chance of swapping
        if np.random.random() < 0.3:
            # Flip operation
            i, j = np.random.choice(n, size=2, replace=False)
            h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
            h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
        else:
            # Swap operation  
            i, j = np.random.choice(n, size=2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
        # Repair constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n/2)
        
        candidates.append(h_new)
        
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization  
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Tracking
    last_improvement = 0
    iteration = 0
    temperature = 1.0
    alpha = 0.999
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate neighborhood
        candidates = generate_neighborhood(h, n_candidates=20)
        
        # Evaluate candidates
        candidate_c5s = [compute_c5_exact(c, dx) for c in candidates]
        
        # Find best non-tabu candidate
        best_candidate_idx = -1
        best_candidate_c5 = float('inf')
        best_candidate = None
        
        for i, (cand, c5) in enumerate(zip(candidates, candidate_c5s)):
            # Check if candidate is tabu
            if tuple(cand) in tabu_list:
                # Apply aspiration if better than best
                if c5 < best_c5:
                    best_candidate_idx = i
                    best_candidate_c5 = c5
                    best_candidate = cand
                continue
            
            if c5 < best_candidate_c5:
                best_candidate_idx = i
                best_candidate_c5 = c5
                best_candidate = cand
        
        # If no non-tabu candidate found, use a tabu candidate with aspiration
        if best_candidate is None and len(candidates) > 0:
            best_candidate_idx = np.argmin(candidate_c5s)
            best_candidate_c5 = candidate_c5s[best_candidate_idx]
            best_candidate = candidates[best_candidate_idx]
                
        if best_candidate is not None:
            # Update tabu list
            tabu_list.append(tuple(h))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Update current solution
            h = best_candidate
            current_c5 = best_candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
            
            # Allow early termination
            if best_c5 <= 0.3807:
                break
                
        # Update temperature
        temperature *= alpha
        iteration += 1
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
