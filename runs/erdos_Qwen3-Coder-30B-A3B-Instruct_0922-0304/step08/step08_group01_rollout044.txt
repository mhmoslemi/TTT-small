import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed_lp(h_r, n_points, dx, max_iter=10):
    """Solve the relaxed LP and return binary approximation."""
    # Create the optimization variables
    z = cp.Variable()
    h = cp.Variable(n_points)
    
    # Formulate constraints
    constraints = []
    constraints.append(cp.sum(h) == n_points / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Add overlap constraints
    # We'll use a representative sample for efficiency
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift + 1):
        if k >= 0:
            # Sum for positive shift
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i + k])
        else:
            # Sum for negative shift
            expr = 0
            for i in range(-k, n_points):
                expr += h[i + k] * (1 - h[i])
        constraints.append(expr * dx <= z)
    
    # Solve the problem
    prob = cp.Problem(cp.Minimize(z), constraints)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_star = h.value
            return h_star
    except:
        pass
    return h_r

def round_to_binary(h_relaxed, target_sum):
    """Convert relaxed h to binary-like h with proper sum."""
    # Threshold-based rounding with median
    median_val = np.median(h_relaxed)
    h_binary = (h_relaxed >= median_val).astype(float)
    
    # Ensure correct sum
    current_sum = np.sum(h_binary)
    diff = target_sum - current_sum
    
    if diff > 0:  # Need to add more 1s
        # Find positions that are 0 and would become 1 with minimal loss
        diff_indices = np.where(h_binary == 0)[0]
        if len(diff_indices) >= diff:
            # Flip the smallest elements in the relaxed solution
            to_flip = np.argsort(h_relaxed[diff_indices])[-diff:]
            flip_indices = diff_indices[to_flip]
            for idx in flip_indices:
                h_binary[idx] = 1
    elif diff < 0:  # Need to subtract 1s
        # Find positions that are 1 and would become 0 with minimal loss
        diff_indices = np.where(h_binary == 1)[0]
        if len(diff_indices) >= -diff:
            # Flip the smallest elements in the relaxed solution
            to_flip = np.argsort(h_relaxed[diff_indices])[:(-diff)]
            flip_indices = diff_indices[to_flip]
            for idx in flip_indices:
                h_binary[idx] = 0
    
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # CVXPY will make this code somewhat slow, so limit iterations
    max_iter = 20  # Drawback of using LP: slow, but also potentially better
    iter_count = 0
    warm_start = False
    
    while time.time() - start_time < budget_s - 2 and iter_count < max_iter:
        # Solve the relaxed LP to get better starting point
        if warm_start:
            h_lp = solve_relaxed_lp(h, n_points, dx)
        else:
            h_lp = h.copy()
        warm_start = True
        
        # Round the relaxed solution to get binary-like form
        h_binary = round_to_binary(h_lp, target_sum)
        h_binary = np.clip(h_binary, 0, 1)
        
        # Adjust if necessary
        h_binary = normalize_h(h_binary, target_sum)
        
        # Compute C5
        new_c5 = compute_c5(h_binary, dx)
        
        # Update if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_binary.copy()
            
        # If we're close, use this as our next solver initial guess
        h = h_binary.copy()
        
        iter_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    # Final validation & ensure target sum
    final_c5 = compute_c5(best_h, dx)
    final_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
