import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_individual(n_points, target_sum):
    """Create a valid random individual."""
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def evaluate_population(population, dx):
    """Evaluate all individuals in population."""
    fitness = []
    for h in population:
        c5 = compute_c5_exact(h, dx)
        fitness.append(c5)
    return fitness

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx].copy()

def crossover(parent1, parent2, target_sum):
    """Create offspring from two parents."""
    offspring = (parent1 + parent2) / 2.0
    offspring = normalize_h(offspring, target_sum)
    offspring = np.clip(offspring, 0, 1)
    offspring = normalize_h(offspring, target_sum)
    return offspring

def mutate(individual, mutation_rate, target_sum):
    """Mutate an individual with Gaussian noise."""
    noise = np.random.normal(0, mutation_rate, len(individual))
    mutated = individual + noise
    mutated = np.clip(mutated, 0, 1)
    mutated = normalize_h(mutated, target_sum)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    population = []
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = create_individual(n_points, target_sum)
    
    population.append(h.copy())
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = create_individual(n_points, target_sum)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    last_improvement = 0
    
    # Main evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = evaluate_population(population, dx)
        
        # Track best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        new_population = []
        
        # Elitism - keep best individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Create offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)
            
            offspring = crossover(parent1, parent2, target_sum)
            mutated = mutate(offspring, mutation_rate, target_sum)
            
            new_population.append(mutated)
            
        population = new_population[:population_size]
        
    # Final evaluation of best
    if best_h is not None:
        final_c5 = compute_c5_exact(best_h, dx)
    else:
        final_c5 = float('inf')
        
    return (best_h.tolist(), final_c5, n_points)
