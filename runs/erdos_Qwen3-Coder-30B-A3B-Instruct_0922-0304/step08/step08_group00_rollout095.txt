import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # PSO parameters
    n_particles = 50
    w = 0.8
    c1 = c2 = 2.0
    max_iter = 1000
    velocity_limits = 0.05
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(n_particles - 1):
        pert = np.random.uniform(-0.1, 0.1, n_points)
        p = np.clip(h + pert, 0, 1)
        current_sum = np.sum(p)
        if current_sum > 0:
            p = p * (target_sum / current_sum)
        particles.append(p)
    
    # Evaluate initial fitness
    fitness = [evaluate_overlap(p, dx) for p in particles]
    
    # Track personal bests and global best
    p_best = particles.copy()
    p_best_fitness = fitness.copy()
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fit = fitness[g_best_idx]
    
    # Velocity for each particle
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Main PSO loop
    no_improvement_iter = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(n_particles):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.random(2)
            vel_update = (w * velocities[i] + 
                         c1 * r1 * (p_best[i] - particles[i]) + 
                         c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            vel_update = np.clip(vel_update, -velocity_limits, velocity_limits)
            
            # Update position
            new_pos = particles[i] + vel_update
            new_pos = np.clip(new_pos, 0, 1)
            
            # Normalize to satisfy sum constraint
            current_sum = np.sum(new_pos)
            if current_sum > 0:
                new_pos = new_pos * (target_sum / current_sum)
            
            # Evaluate new fitness
            new_fit = evaluate_overlap(new_pos, dx)
            
            # Update personal best
            if new_fit < p_best_fitness[i]:
                p_best[i] = new_pos.copy()
                p_best_fitness[i] = new_fit
                
                # Update global best if needed
                if new_fit < g_best_fit:
                    g_best = new_pos.copy()
                    g_best_fit = new_fit
                    no_improvement_iter = 0
            
            # Update particle and velocity
            particles[i] = new_pos
            velocities[i] = vel_update
            
        no_improvement_iter += 1
        
        # Apply shaking if stagnated
        if no_improvement_iter > 50:
            for i in range(n_particles):
                # Add random perturbation
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                particles[i] = np.clip(particles[i] + perturbation, 0, 1)
                current_sum = np.sum(particles[i])
                if current_sum > 0:
                    particles[i] = particles[i] * (target_sum / current_sum)
            no_improvement_iter = 0
    
    # Local refinement with SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, g_best, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < g_best_fit:
            g_best = refined_h
            g_best_fit = final_c5
    
    # Final validation
    final_h = np.clip(g_best, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
