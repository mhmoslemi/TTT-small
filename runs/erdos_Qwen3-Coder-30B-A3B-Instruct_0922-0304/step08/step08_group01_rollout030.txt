import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try different configurations for search
    configs = [
        {"n_points": 70, "max_iter": 30000, "step_size": 0.01, "update_ratio": 0.1},
        {"n_points": 50, "max_iter": 50000, "step_size": 0.005, "update_ratio": 0.15},
        {"n_points": 100, "max_iter": 20000, "step_size": 0.015, "update_ratio": 0.05}
    ]
    
    best_solution = (best_h.tolist(), best_c5, n_points)
    
    # Multi-start approach
    for config in configs:
        if time.time() - start_time > budget_s - 1:
            break
            
        n_points_config, max_iter, step_size, update_ratio = (
            config["n_points"], config["max_iter"], config["step_size"], config["update_ratio"]
        )
        
        # Reset h for this config's search
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
            h = h * (target_sum / np.sum(h))
        else:
            h = np.random.rand(n_points_config).astype(np.float64)
            h = h * (target_sum / np.sum(h))
        
        h = np.clip(h, 0, 1)
        
        # Simple gradient-based method with restarts
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
            
            # Compute approximate gradients
            gradients = np.zeros(n_points_config)
            for i in range(n_points_config):
                grad = 0
                for k in range(n_points_config):
                    if i + k < n_points_config:
                        grad += (1 - 2 * h[i]) * (1 - h[i + k])
                    if i - k >= 0:
                        grad += (1 - 2 * h[i]) * (1 - h[i - k])
                gradients[i] = grad
            
            # Select top updates
            selected = np.argsort(np.abs(gradients))[::-1][:max(1, int(update_ratio * n_points_config))]
            
            # Update
            for i in selected:
                step = step_size * gradients[i]
                h[i] = np.clip(h[i] + step, 0, 1)
            
            # Renormalize
            h = h * (target_sum / np.sum(h))
            
            # Evaluate
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
        if best_c5 < best_solution[1]:
            best_solution = (best_h.tolist(), best_c5, n_points_config)
    
    # Final validation
    final_h, final_c5, final_n = best_solution
    final_h_array = np.array(final_h)
    final_c5 = compute_c5(final_h_array, 2.0 / final_n)
    
    return (final_h, final_c5, final_n)
