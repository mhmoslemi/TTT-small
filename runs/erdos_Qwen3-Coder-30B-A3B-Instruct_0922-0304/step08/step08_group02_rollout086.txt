import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        # Create a simple initial guess
        h_current = np.ones(n_points) * 0.5
    
    # Ensure exact sum constraint
    h_current = h_current * (target_sum / np.sum(h_current))
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # MCMC parameters
    T0 = 0.1
    alpha = 0.999
    sigma = 0.05
    T_min = 1e-6
    cooling_steps = 1000
    no_improvement_limit = 2000
    
    T = T0
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Propose new state
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        
        # Perturb one element
        delta = np.random.normal(0, sigma)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Maintain sum constraint by adjusting other elements
        old_sum = np.sum(h_current)
        new_sum = np.sum(h_new)
        scale_factor = old_sum / new_sum
        h_new = h_new * scale_factor
        
        # Ensure exact constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Compute change
        delta_c5 = c5_new - best_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Update temperature
        T = max(T_min, T * alpha)
        
        # Adaptive adjustments
        if no_improvement_count > no_improvement_limit:
            sigma *= 0.9
            no_improvement_count = 0
            T = min(T0, T * 1.5)
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
