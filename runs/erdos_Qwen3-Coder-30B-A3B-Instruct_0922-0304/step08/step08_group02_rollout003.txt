import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def constraint_satisfy(h, target_sum):
    """Ensure h is in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    return h * (target_sum / np.sum(h))

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.05):
    """Apply mutation to a solution."""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h_mut[i] = np.clip(h_mut[i] + np.random.normal(0, 0.1), 0, 1)
    return h_mut

def genetic_algorithm(rng, start_time, budget_s, n_points, target_sum):
    """Perform genetic algorithm optimization."""
    n_pop = 20
    mutation_rate = 0.05
    crossover_prob = 0.8
    elitism_rate = 0.2
    
    # Initialize population
    population = []
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = constraint_satisfy(h, target_sum)
        population.append(h)
    
    # Fill the population
    while len(population) < n_pop:
        h = rng.random(n_points)
        h = constraint_satisfy(h, target_sum)
        population.append(h)
        
    # Evolve
    generation = 0
    best_fitness = float('inf')
    best_individual = None
    
    while time.time() - start_time < budget_s - 2 and generation < 1000:
        # Evaluate fitness
        fitnesses = [compute_c5(h, 2.0/n_points) for h in population]
        sorted_indices = np.argsort(fitnesses)
        
        # Track best
        if fitnesses[sorted_indices[0]] < best_fitness:
            best_fitness = fitnesses[sorted_indices[0]]
            best_individual = population[sorted_indices[0]].copy()
            
        # Elitism
        new_population = [population[i] for i in sorted_indices[:int(n_pop * elitism_rate)]]
        
        # Generate next generation
        while len(new_population) < n_pop:
            # Selection (tournament)
            parent1 = population[np.random.choice(sorted_indices[:n_pop//2])]
            parent2 = population[np.random.choice(sorted_indices[:n_pop//2])]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mutation_rate)
            
            # Constraint repair
            child = constraint_satisfy(child, target_sum)
            new_population.append(child)
            
        population = new_population
        generation += 1
        
        # Early stopping if we're good enough
        if best_fitness <= 0.3807:
            break
            
    return best_individual, best_fitness

def local_refinement(h, dx, target_sum, iterations=50):
    """Refine a solution with gradient-like steps."""
    h_best = h.copy()
    best_c5 = compute_c5(h_best, dx)
    
    for _ in range(iterations):
        if time.time() < time.time() - 1:  # Basic time check
            break
        
        h_trial = h_best.copy()
        idx = np.random.randint(len(h_trial))
        
        # Small perturbation
        delta = np.random.normal(0, 0.02)
        h_trial[idx] = np.clip(h_trial[idx] + delta, 0, 1)
        h_trial = constraint_satisfy(h_trial, target_sum)
        
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_c5:
            h_best = h_trial.copy()
            best_c5 = c5_trial
            
    return h_best, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with genetic algorithm
    best_h, best_c5 = genetic_algorithm(rng, start_time, budget_s, n_points, target_sum)
    
    # Local refinement of top solutions
    refinement_trials = 3
    for _ in range(refinement_trials):
        if time.time() - start_time > budget_s - 2:
            break
        refined_h, refined_c5 = local_refinement(best_h, dx, target_sum)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return (best_h.tolist(), best_c5, n_points)
