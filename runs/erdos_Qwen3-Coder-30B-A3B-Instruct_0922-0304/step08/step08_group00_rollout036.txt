import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Using numpy's correlate with proper handling of circular convolution
    # We actually want the full convolution, as specified in problem
    # But this simulates the behavior from previous code
    n = len(h)
    # Create extended arrays for full convolution
    h_ext = np.concatenate([h, np.zeros(n-1)])
    h_reflect = np.concatenate([1 - h, np.zeros(n-1)])
    conv = np.convolve(h_ext, h_reflect, mode='full')
    # take only relevant part - shifts from -n+1 to n-1, which gives 2*n-1 elements
    # max shift k is n-1, shifts k=0 to 2*n-2 correspond to indices 0 to 2*n-2
    # we need values of overlap corresponding to shifts k=0 to n-1 (same as indices 0 to n-1)
    # and shifts k=1 to n-1 (indices n to 2*n-2 when measured from i+k)
    conv_vals = conv[n-1:]  # takes vals where shift k = 0 to n-1 (or equivalently from index = n-1 to 2*n-2)
    # but actually the right way to compute corr in our discrete case with shift 0 to n-1
    # is to compute np.correlate(h, 1-h, mode="full") and take max over all indices which correspond to 
    # shift from -n+1 through n-1, so we effectively want indices 0 to 2*n-2 and max over them
    conv_vals = conv
    
    # However, let's keep consistent with the old logic that's equivalent to what's required:
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Generate base feasible solution
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Parameters
    max_iter = 1000
    improvement_threshold = 1e-5
    perturbation_delta = 1e-3
    perturbation_decay = 0.98
    slsqp_tolerance = 1e-10
    slsqp_maxiter = 200
    
    def project_and_normalize(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            x = x * (target_sum / s)
        return x
    
    def optimize_subset(h_full, fixed_indices, var_indices, subset_h):
        # Build full h vector
        h_new = h_full.copy()
        h_new[var_indices] = subset_h
        
        # Define function to optimize (overlap for this h_new)
        def obj(h_var):
            # build full h again
            h_candidate = h_full.copy()
            h_candidate[var_indices] = h_var
            return evaluate_overlap(h_candidate, dx)
            
        # constraints: sum of h_candidate should be target_sum
        def sum_constraint(h_var):
            h_candidate = h_full.copy()
            h_candidate[var_indices] = h_var
            return np.sum(h_candidate) - target_sum
            
        # constraints for optimizing, just var_indices
        bounds = [(0, 1) for _ in var_indices]
        cons = {'type': 'eq', 'fun': sum_constraint}
        
        # solve
        result = minimize(obj, subset_h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': slsqp_maxiter, 'ftol': slsqp_tolerance})
        
        if result.success:
            return project_and_normalize(result.x)
        else:
            # fallback to original
            return subset_h.copy()
    
    # Main loop
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        iter_count += 1
        
        # partition to even/odd components for alternating optimization
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Optimization loop between even and odd indices
        last_c5 = best_c5
        
        # Optimize even indices, keeping odd fixed
        even_h = h[even_indices]
        h_even_updated = optimize_subset(h, odd_indices, even_indices, even_h)
        h[even_indices] = h_even_updated
        
        # Optimize odd indices, keeping even fixed
        odd_h = h[odd_indices]
        h_odd_updated = optimize_subset(h, even_indices, odd_indices, odd_h)
        h[odd_indices] = h_odd_updated
        
        # Recompute overlap
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # perturb solution to avoid getting stuck
        if iter_count % 10 == 0:
            perturb = np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
            h += perturb
            h = project_and_normalize(h)
            c5 = evaluate_overlap(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Decay perturbation strength
        perturbation_delta *= perturbation_decay
        
        if abs(best_c5 - last_c5) < improvement_threshold:
            break
    
    # Final refinement
    final_h = best_h.copy()
    # run one more local optimization on best solution
    def obj_final(h_vec):
        return evaluate_overlap(h_vec, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(obj_final, final_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # ensure it meets constraints
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
