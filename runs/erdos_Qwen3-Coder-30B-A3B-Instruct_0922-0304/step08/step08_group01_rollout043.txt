import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def round_to_binary(h_relaxed, target_sum):
    """Round relaxed solution to binary with sum constraint."""
    h_binary = (h_relaxed >= np.median(h_relaxed)).astype(float)
    current_sum = np.sum(h_binary)
    diff = current_sum - target_sum
    
    if diff > 0:
        # Too many 1s: flip some to 0
        flip_indices = np.argsort(h_relaxed)[:int(diff)]
        h_binary[flip_indices] = 0
    elif diff < 0:
        # Too many 0s: flip some to 1
        flip_indices = np.argsort(h_relaxed)[-int(-diff):]
        h_binary[flip_indices] = 1
    
    return h_binary

def solve_relaxed_lp(h_init, n_points, dx, target_sum):
    """Solve LP relaxation with continuous variables."""
    h = cp.Variable(n_points, name="h")
    z = cp.Variable(1, name="z")
    
    # Objective
    prob = cp.Problem(cp.Minimize(z), [])
    
    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
        z >= 0
    ]
    
    # Linearize overlap constraints
    for shift in range(-n_points+1, n_points):
        if shift >= 0:
            expr = cp.sum(cp.multiply(h[:n_points-shift], 1-h[shift:]))
        else:
            expr = cp.sum(cp.multiply(h[-shift:], 1-h[:n_points+shift]))
        constraints.append(expr <= z)
    
    prob.constraints = constraints
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    
    return h_init

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for _ in range(10):  # limited iterations to stay within time budget
        if time.time() - start_time >= budget_s - 1:
            break
        
        # Solve relaxed LP
        h_relaxed = solve_relaxed_lp(h, n_points, dx, target_sum)
        if h_relaxed is not None:
            h = h_relaxed.copy()
        
        # Apply rounding heuristic
        h_binary = round_to_binary(h, target_sum)
        h = h_binary.copy()
        
        # Normalize again and check improvement
        h = normalize_h(h, target_sum)
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
