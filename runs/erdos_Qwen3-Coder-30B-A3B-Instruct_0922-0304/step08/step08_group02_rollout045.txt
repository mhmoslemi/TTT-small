import numpy as np

def project_h(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def sample_shifts(n, sample_size):
    """Sample random shifts for overlap constraints"""
    return np.random.randint(0, 2*n - 1, sample_size)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx
