import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip values to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    """Create a triangular ramp function."""
    x = np.linspace(0, 2, n)
    h = np.minimum(x, 2 - x)
    return h / np.sum(h) * (n / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Phase 1: Gradient-based with momentum
    momentum = 0.9
    alpha = 0.001
    gamma = 0.99
    velocity = np.zeros(n_points)
    
    # Phase 2: Coordinate-wise updates
    coord_sample_rate = 0.2
    coord_update_count = max(1, int(n_points * coord_sample_rate))
    
    phase1_iters = 10000
    phase2_iters = 10000
    iter_count = 0
    
    # Track progress for switching between phases
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 500
    
    while iter_count < phase1_iters + phase2_iters and time.time() - start_time < budget_s - 1:
        if iter_count < phase1_iters:
            # Phase 1: Momentum-based gradient descent
            overlap = correlate(h, 1 - h, mode="full") * dx
            overlap_max = np.max(overlap)
            max_pos = np.argmax(overlap)
            
            # Compute gradient
            shifted = np.roll(1 - h, max_pos - n_points + 1)
            gradient = (1 - 2 * h) * shifted / n_points
            
            # Update velocity and h
            velocity = momentum * velocity + (1 - momentum) * gradient
            h_candidate = h - alpha * velocity
            
            # Project
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            
            new_c5 = compute_c5(h_candidate, dx)
            if new_c5 < overlap_max:
                h = h_candidate.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                alpha *= gamma
                last_improvement = iter_count
            else:
                h = clip_and_renormalize(h, target_sum)
                alpha *= gamma
                
        else:
            # Phase 2: Coordinate-wise updates
            # Sample coordinates
            indices = np.random.choice(n_points, coord_update_count, replace=False)
            improved = False
            
            # For each sampled coordinate
            for i in indices:
                current_val = h[i]
                # Try different step directions
                steps = [-0.01, -0.005, 0.005, 0.01]
                best_step = 0.0
                best_new_c5 = overlap_max = compute_c5(h, dx)
                
                h_new = h.copy()
                for step in steps:
                    h_new[i] = current_val + step
                    h_new = clip_and_renormalize(h_new, target_sum)
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_step = step
                
                if best_new_c5 < overlap_max:
                    h[i] = current_val + best_step
                    h = clip_and_renormalize(h, target_sum)
                    improved = True
                    if best_new_c5 < best_c5:
                        best_c5 = best_new_c5
                        best_h = h.copy()
            
            if not improved:
                stagnation_count += 1
                if stagnation_count > max_stagnation:
                    break
            else:
                stagnation_count = 0
                
        iter_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
