import numpy as np
from scipy.signal import correlate
import time
from math import exp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_solution(h, target_sum):
    """Ensure h is in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def crossover(parent1, parent2, n_points):
    """Blend two parents at random points."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    
    # Random crossover point
    point = np.random.randint(1, n_points-1)
    child1[point:] = parent2[point:]
    child2[point:] = parent1[point:]
    
    return child1, child2

def mutate(h, mutation_rate=0.05, max_mutation=0.1):
    """Add small random perturbations."""
    h_mut = h.copy()
    for i in range(len(h_mut)):
        if np.random.random() < mutation_rate:
            delta = np.random.uniform(-max_mutation, max_mutation)
            h_mut[i] += delta
    return h_mut

def local_improve(h, dx, steps=50):
    """Refine solution using small perturbations."""
    h_best = h.copy()
    c5_best = compute_c5(h_best, dx)
    
    for _ in range(steps):
        h_temp = h_best.copy()
        idx = np.random.randint(len(h_temp))
        delta = np.random.normal(0, 0.01)
        h_temp[idx] += delta
        h_temp = repair_solution(h_temp, len(h_temp)/2)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < c5_best:
            h_best = h_temp
            c5_best = c5_temp
            
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population[0] = np.clip(population[0], 0, 1)
        population[0] = population[0] * (target_sum / np.sum(population[0]))
    else:
        population = [np.ones(n_points) * 0.5]
    
    # Create initial population
    n_pop = 20
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    
    # Genetic algorithm parameters
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = int(0.2 * n_pop)
    
    best_fitness = float('inf')
    best_individual = None
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = compute_c5(h, dx)
            fitnesses.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses.sort()
        
        # Update best
        if fitnesses[0] < best_fitness:
            best_fitness = fitnesses[0]
            best_individual = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Select elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            a, b = np.random.choice(elite_size, size=2, replace=False)
            parent1, parent2 = population[a], population[b]
            
            if np.random.random() < crossover_prob:
                child1, child2 = crossover(parent1, parent2, n_points)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate, 0.1 * exp(-generation/100))
            child1 = repair_solution(child1, target_sum)
            
            child2 = mutate(child2, mutation_rate, 0.1 * exp(-generation/100))
            child2 = repair_solution(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population[:n_pop]
        generation += 1
        
        # Escape local minima
        if no_improvement > 50 and np.random.random() < 0.1:
            for i in range(len(population)):
                population[i] = np.random.rand(n_points)
                population[i] = population[i] * (target_sum / np.sum(population[i]))
    
    # Final refinement of top solutions
    top_solutions = population[:3]
    best_final_h = best_individual.copy()
    best_final_c5 = compute_c5(best_final_h, dx)
    
    for h in top_solutions:
        refined_h, refined_c5 = local_improve(h, dx, 50)
        if refined_c5 < best_final_c5:
            best_final_c5 = refined_c5
            best_final_h = refined_h.copy()
    
    return (best_final_h.tolist(), best_final_c5, n_points)
