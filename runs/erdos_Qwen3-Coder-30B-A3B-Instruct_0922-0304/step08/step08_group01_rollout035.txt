```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Alternating optimization: relaxation + rounding
    max_iter = 50
    iter_count = 0
    patience = 0
    max_patience = 10
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Relaxation stage - use CVXPY to solve convex relaxation
        try:
            h_relaxed = cp.Variable(n_points)
            z = cp.Variable()
            
            # Constraints
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == target_sum]
            
            # Objective
            # Use dummy correlation calculation to build constraints
            # We approximate the max overlap using a convex upper bound
            correlations = []
            for k in range(2*n_points - 1):
                # Compute correlation term based on k-shifted indices
                if k < n_points:
                    # First part: x = 0 to n_points-k-1
                    idx = slice(0, n_points-k)
                    idx_shifted = slice(k, n_points)
                else:
                    # Second part: x = 0 to k-n_points
                    idx = slice(k-n_points+1, n_points)
                    idx_shifted = slice(0, n_points-(k-n_points+1))
                
                # Only include corresponding terms
                corr_term = cp.sum(cp.multiply(h_relaxed[idx], 1 - h_relaxed[idx_shifted]))
                correlations.append(corr_term)
            
            for corr in correlations:
                constraints.append(z >= corr * dx)
            
            prob = cp.Problem(cp.Minimize(z), constraints)
            
            # Solve with timeout
            try:
                prob.solve(solver=CPLEX, verbose=False, time_limit=5.0)
            except:
                try:
                    prob.solve(solver=SCS, verbose=False, max_iters=1000)
                except:
                    pass  # Failed, continue with current h
            
            if h_relaxed.value is not None:
                h_r = np.array(h_relaxed.value)
                if np.all(h_r >= 0) and np.all(h_r <= 1):
                    h = h_r.copy()
                
        except Exception as e:
            pass  # Skip relaxation if anything fails
        
        # Rounding heuristic
        # Convert to binary-like suggestions
        tau = np.median(h)
        h_b = (h > tau).astype(float)
        h_b = normalize_h(h_b, target_sum)
        
        # Adjust binary to match target sum
        current_sum_hb = np.sum(h_b)
        diff = int(current_sum_hb - target_sum)
        if diff != 0:
            if diff > 0:
                # Flip some 1s to 0s (largest values)
                idxs = np.argsort(-h_b)
                for i in range(diff):
                    h_b[idxs[i]] = 0
            else:
                # Flip some 0s to 1s (smallest values)
                idxs = np.argsort(h_b)
                for i in range(-diff):
                    h_b[idxs[i]] = 1
        
        # Ensure no violation
        h_b = np.clip(h_b, 0, 1)
        
        # Refine using gradient-based local search
        # Compute full overlap for adjustment
        overlap = correlate(h_b, 1 - h_b, mode="full")
        current_max = np.max(overlap)
        if current_max < best_c5:
            best_c5 = current_max * dx
            best_h = h_b.copy()
            patience = 0
        else:
            patience += 1
        
        # Optional update
        if patience <= 5:
            # Local adaptation using gradient
            for _ in range(5):
                gradients = np.zeros(n_points)
                for i in range(n_points):
                    grad = 0
                    for k in range(1, n_points):
                        if i + k < n_points:
                            grad += (1 - 2 * h_b[i]) * (1 - h_b[i + k])
                        if i - k >= 0:
                            grad += (1 - 2 * h_b[i]) * (1 - h_b[i - k])
                    gradients[i] = grad
                
                # Update
                step = 0.01
                h_b = h_b + step * gradients * (1 - 2 * h_b)
                h_b = np.clip(h_b, 0, 1)
                h_b = normalize_h(h_b, target_sum)
                
                # Re-evaluate
                overlap = correlate(h_b, 1 - h_b, mode="full")
                if np.max(overlap) * dx < best_c5:
                    best_c5 = np.max(overlap) * dx
                    best_h = h_b.copy()
                    patience = 0
                else:
                    patience += 1

        iter_count += 1
        if patience >= max_patience:
            break
        
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
```
