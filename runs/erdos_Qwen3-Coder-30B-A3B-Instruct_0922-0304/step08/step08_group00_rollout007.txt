import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Apply SQP optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try optimization with SLSQP
    result = minimize(objective, h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 1000, 'ftol': 1e-10})
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # If no improvement, try with slight perturbation
    if best_c5 >= 0.38085:
        h_perturbed = h + np.random.normal(0, 1e-4, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        
        result = minimize(objective, h_perturbed, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 500, 'ftol': 1e-10})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final check and correction
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
