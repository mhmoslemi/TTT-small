import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp from 0 to 1 and back to 0 over [0,2]"""
    x = np.linspace(0, 2, n)
    return np.clip(1 - np.abs(x - 1), 0, 1)

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum==target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = project_h(h, target_sum)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Grid search for different initial configurations
    initial_configs = []
    # Original triangular
    initial_configs.append(triangular_ramp(n_points))
    # Random
    initial_configs.append(np.random.rand(n_points))
    # Step function
    h_step = np.zeros(n_points)
    h_step[:n_points//2] = 1.0
    h_step = project_h(h_step, target_sum)
    initial_configs.append(h_step)
    
    # Try each initial configuration
    for i, h_init in enumerate(initial_configs):
        h_current = h_init.copy()
        h_current = project_h(h_current, target_sum)
        c5_current = compute_c5(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        if best_c5 <= 0.3807:
            return (best_h.tolist(), best_c5, n_points)
    
    # Use coordinate-wise approach with modeled gradient
    # Augment with grid search technique
    
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current C5 and overlap
        overlap = correlate(h_current, 1 - h_current, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h_current.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute approximate gradient for each point
        # Simplified discrete gradient approximation
        gradients = np.zeros(n_points)
        step_size = 0.01
        
        for i in range(n_points):
            # try perturbation
            h_plus = h_current.copy()
            h_plus[i] = min(1.0, h_current[i] + step_size)
            h_plus = project_h(h_plus, target_sum)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h_current.copy()
            h_minus[i] = max(0.0, h_current[i] - step_size)
            h_minus = project_h(h_minus, target_sum)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            gradients[i] = (max_plus - max_minus) / (2.0 * step_size)
        
        # Find high gradient indices to update
        update_indices = np.argsort(np.abs(gradients))[-(n_points//10):][::-1]
        
        # Apply updates
        improved = False
        for i in update_indices:
            # Modify value in direction of negative gradient
            if gradients[i] != 0:
                # Direction of change based on gradient sign
                alpha = 0.001
                if gradients[i] < 0:  # If increasing C5
                    h_new = h_current.copy()
                    h_new[i] = max(0.0, h_current[i] - alpha)
                    h_new = project_h(h_new, target_sum)
                    
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < current_max:
                        h_current = h_new.copy()
                        improved = True
                        break
                        
                else:  # If decreasing C5 (gradient +ve)
                    h_new = h_current.copy()
                    h_new[i] = min(1.0, h_current[i] + alpha)
                    h_new = project_h(h_new, target_sum)
                    
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < current_max:
                        h_current = h_new.copy()
                        improved = True
                        break

        if not improved:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
        else:
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
