import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Define bounds (0, 1) for all variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: sum(h) = target_sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function for minimization
    def objective(h):
        return compute_c5(h, dx)
    
    # Set up optimization
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Minimize using SLSQP with bounds and constraint
        try:
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            
            if result.success:
                h = result.x.copy()
                h = np.clip(h, 0, 1)
                
                # Renormalize to ensure sum constraint
                h = h * (target_sum / np.sum(h))
                
                # Compute C5 for this result
                new_c5 = compute_c5(h, dx)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                
                # Early stopping if target achieved
                if best_c5 <= 0.3807:
                    break
            else:
                # If optimization fails, reinitialize with current best
                h = best_h.copy()
                
        except Exception as e:
            # Fallback: reinitialize with current best
            h = best_h.copy()
            
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
