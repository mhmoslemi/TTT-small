import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighbors(h, n_neighbors=10):
    """Generate neighbor solutions by flipping or swapping elements."""
    neighbors = []
    n = len(h)
    for _ in range(n_neighbors):
        h_new = h.copy()
        
        # Choose between flip and swap with probabilities 0.3 and 0.7
        if np.random.rand() < 0.3:
            # Flip operation: randomly select 2 points and modify their values
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
            h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
        else:
            # Swap operation: randomly select 2 points and swap their values
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Repair constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        neighbors.append(h_new)
    
    return neighbors

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

        h = normalize_h(h, target_sum)
    
    # Ensure constraints are satisfied
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Initialize tabu list (stores recent visited solutions)
    tabu_list = []
    tabu_tenure = 100
    aspiration_tol = 1e-6
    
    # Initialize temperature for simulated annealing-like behavior
    temperature = 1.0
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Generate neighbors
        neighbors = generate_neighbors(h, n_neighbors=10)
        
        # Evaluate neighbors
        neighbor_c5s = [compute_c5_exact(n, dx) for n in neighbors]
        
        # Find best non-tabu neighbor
        best_neighbor_idx = -1
        best_neighbor_c5 = float('inf')
        
        for i, (n, c5) in enumerate(zip(neighbors, neighbor_c5s)):
            # Check if neighbor is in tabu list
            if n.tolist() in tabu_list:
                # Apply aspiration criteria
                if c5 < best_c5 - aspiration_tol:
                    best_neighbor_idx = i
                    best_neighbor_c5 = c5
                continue
            
            if c5 < best_neighbor_c5:
                best_neighbor_idx = i
                best_neighbor_c5 = c5
        
        # If no valid neighbor found, continue
        if best_neighbor_idx == -1:
            # Randomly perturb current solution
            h = h * (1 + np.random.normal(0, 0.01, n_points))
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            last_improvement += 1
            iteration += 1
            continue
        
        # Accept the neighbor
        h = neighbors[best_neighbor_idx]
        current_c5 = best_neighbor_c5
        
        # Update tabu list
        tabu_list.append(h.tolist())
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Update temperature for simulated annealing-like behavior
        temperature *= 0.999
        
        iteration += 1

            h = np.clip(h, 0, 1)
