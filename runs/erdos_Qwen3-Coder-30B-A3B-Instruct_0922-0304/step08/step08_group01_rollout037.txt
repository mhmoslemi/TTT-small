import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def solve_relaxed(h_r, n_points, dx):
    """Solve a relaxed linear program for the overlap."""
    # Variables
    h = Variable(n_points)
    
    # Objective: minimize z where z >= sum of all overlaps
    z = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # For the purpose of relaxing, we approximate objective by max of
    # correlation entries. Since we cannot directly include all correlations in a single linear constraint
    # we optimize over a surrogate: linear in the variables and upper bound on overlap
    # Below is an educated workaround under linear relaxation bounds
    # Solve a simpler convex relaxation to get a better point for rounding
    
    # Just solve for the sum constraint and letting solver handle the rest
    prob = Problem(Minimize(z), constraints)
    
    # We don't compute overlaps directly in constraint in CVXPY due to quadratic nature
    # Instead, we target an upper bound via usable inequalities
    try:
        prob.solve(solver=ECOS, max_iters=200)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value
    except:
        pass
    # Fallback to original if not solved
    return h_r

def threshold_round(h_r, n_points):
    """Threshold round h_r to binary values."""
    # Compute median for thresholding
    median_h = np.median(h_r)
    
    # Using threshold around median
    h_b = (h_r > median_h).astype(float)
    
    # Adjust to match sum constraint
    current_sum = np.sum(h_b)
    target_sum = n_points / 2
    
    # If too many 1's, set some to 0
    if current_sum > target_sum:
        diff = int(current_sum - target_sum)
        # Get indices of highest values
        idxs = np.argsort(h_r)[::-1][:diff]
        h_b[idxs] = 0
    elif current_sum < target_sum:
        diff = int(target_sum - current_sum)
        # Get indices of lowest values
        idxs = np.argsort(h_r)[:diff]
        h_b[idxs] = 1
    
    return h_b

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Early stopping indicator
    no_improvement_count = 0
    max_no_improvement = 10
    
    iterations = 0
    max_iters = 50
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iters:
        # Relaxation Stage - CVXPY solution
        h_r = solve_relaxed(h, n_points, dx)
        if h_r is None:
            h_r = h  # fallback
            
        # Rounding Heuristic
        h_b = threshold_round(h_r, n_points)
        
        # Adjustment - ensure sum
        h_b = normalize_h(h_b, target_sum)
        h_b = np.clip(h_b, 0, 1)
        
        # Validate and update
        new_c5 = compute_c5(h_b, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_b.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early convergence
        if no_improvement_count == max_no_improvement or best_c5 <= 0.3807:
            break
            
        # Update h for next round
        h = h_b.copy()
        iterations += 1
        
        # Occasionally recompute overlap for precision
        if iterations % 5 == 0:
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
