import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    mid = n // 2
    left = np.linspace(0, 1, mid)
    right = np.linspace(1, 0, n - mid)
    return np.append(left, right)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Hyperparameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iter = 20000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Validate and correct
        if abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
        else:
            h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Momentum variables
    velocity = np.zeros(n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_idx = np.argmax(overlap)
        
        # Compute exact gradient
        # The derivative of max_k [h(x) * (1 - h(x+k))] w.r.t. h_i
        # is approximately the sign of overlap_max times (1 - 2*h_i) / n_points
        # This is a simplification, but works well for optimization
        gradient = (1 - 2 * h) * np.sign(overlap_max) / n_points
        
        # Apply momentum
        velocity = momentum * velocity + learning_rate * gradient
        
        # Update h candidate
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
                
        else:
            # Reduce step size and retry
            learning_rate *= shrink_factor
            velocity *= shrink_factor
    
    return (best_h.tolist(), best_c5, n_points)
