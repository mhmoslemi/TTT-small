import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    # For each shift k from 0 to n-1
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap * dx)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use sequential quadratic programming (SLSQP)
    def objective(c_and_h):
        # c is a dummy variable for the max constraint, but objective is always 0
        # We'll deal with the constraint differently
        return 0.0
    
    def constraint_sum(c_and_h):
        h_slice = c_and_h[1:]
        return np.sum(h_slice) - target_sum
    
    def constraint_max_overlap(c_and_h):
        # c is c_and_h[0] and h is c_and_h[1:]
        h = c_and_h[1:]
        max_overlap = 0
        n = len(h)
        for k in range(n):
            overlap = 0
            for i in range(n):
                j = (i + k) % n
                overlap += h[i] * (1 - h[j])
            max_overlap = max(max_overlap, overlap * dx)
        return max_overlap - c_and_h[0]
    
    # Bounds for h variables
    bounds = [(0, 1) for _ in range(n_points)]
    bounds = [(None, None)] + bounds  # First element is for c
    
    # Initial guess
    x0 = np.concatenate([[best_c5], h])
    
    # Constraints
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_max_overlap}
    ]
    
    # Use SLSQP
    try:
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if result.success:
            final_h = np.clip(result.x[1:], 0, 1)
            final_sum = np.sum(final_h)
            if final_sum > 0:
                final_h = final_h * (target_sum / final_sum)
            final_c5 = evaluate_overlap(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception as e:
        # If optimization fails, just return current best
        pass
    
    # Extra refinement with local optimization if time permits
    if time.time() - start_time < budget_s - 10:
        def refine_objective(h_vector):
            return evaluate_overlap(h_vector, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(refine_objective, best_h, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 1000, 'ftol': 1e-12})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
