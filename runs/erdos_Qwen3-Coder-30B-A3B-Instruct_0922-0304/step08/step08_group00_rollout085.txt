import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Initialize with a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Alternating optimization parameters
    max_iterations = 100
    improvement_threshold = 1e-5
    perturbation_delta = 1e-3
    perturbation_decay = 0.98
    
    # Helper to project onto feasible set
    def project_and_normalize(x):
        x = np.clip(x, 0, 1)
        current_sum = np.sum(x)
        if current_sum > 0:
            x = x * (target_sum / current_sum)
        return x
    
    # Iterative alternating optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Perturb solution
        h_perturbed = h + np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
        h_perturbed = project_and_normalize(h_perturbed)
        
        # Partition into even and odd indexed components
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Split variables
        h_even = h_perturbed[even_indices]
        h_odd = h_perturbed[odd_indices]
        
        # Optimization function for even components (fixing odd)
        def objective_even(h_even_part):
            # Reconstruct full vector with fixed odd values
            h_full = h_perturbed.copy()  # start from previous solution
            h_full[even_indices] = h_even_part
            h_full[odd_indices] = h_odd
            return evaluate_overlap(h_full, dx)
        
        # SLSQP optimization for even components
        bounds_even = [(0, 1) for _ in range(len(even_indices))]
        constraints_even = [{'type': 'eq', 'fun': lambda x: np.sum(x) + np.sum(h_odd) - target_sum}]
        result_even = minimize(objective_even, h_even, method='SLSQP', 
                              bounds=bounds_even, constraints=constraints_even,
                              options={'ftol': 1e-10, 'maxiter': 100})
        
        if result_even.success:
            h_even_opt = np.clip(result_even.x, 0, 1)
            h_full = h_perturbed.copy()
            h_full[even_indices] = h_even_opt
            h_full[odd_indices] = h_odd
            new_c5 = evaluate_overlap(h_full, dx)
            if new_c5 < best_c5:
                best_h = h_full.copy()
                best_c5 = new_c5

        # Optimization function for odd components (fixing even)
        def objective_odd(h_odd_part):
            # Reconstruct full vector with fixed even values
            h_full = h_perturbed.copy()
            h_full[even_indices] = h_even
            h_full[odd_indices] = h_odd_part
            return evaluate_overlap(h_full, dx)
        
        # SLSQP optimization for odd components
        bounds_odd = [(0, 1) for _ in range(len(odd_indices))]
        constraints_odd = [{'type': 'eq', 'fun': lambda x: np.sum(h_even) + np.sum(x) - target_sum}]
        result_odd = minimize(objective_odd, h_odd, method='SLSQP',
                             bounds=bounds_odd, constraints=constraints_odd,
                             options={'ftol': 1e-10, 'maxiter': 100})
        
        if result_odd.success:
            h_odd_opt = np.clip(result_odd.x, 0, 1)
            h_full = h_perturbed.copy()
            h_full[even_indices] = h_even
            h_full[odd_indices] = h_odd_opt
            new_c5 = evaluate_overlap(h_full, dx)
            if new_c5 < best_c5:
                best_h = h_full.copy()
                best_c5 = new_c5

        # Update iteration parameters
        perturbation_delta *= perturbation_decay
    
    # Final refinement
    def refine_objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(refine_objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
