import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Start with CVXPY approach
    # Variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize maximum overlap
    # This is a bit tricky to model directly in CVXPY due to max operation
    # We'll model it as a constrained optimization and use a proxy
    
    # For now, let's try optimizing via direct solver and local refinement
    # Use SLSQP for local refinement with constraints
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # First, use CVXPY to solve a relaxed version if possible
    # But let's make a direct approach with tighter algorithms
    
    # Refine with local search (SLSQP) for the best initial solution
    result = minimize(objective, h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 500, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Further refinement using a different local optimization approach with more steps
    # Run a few more steps of local optimization starting from the best solution
    max_refinements = 5
    for _ in range(max_refinements):
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 300, 'ftol': 1e-9})
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
