import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog
import cvxpy as cp


def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx


def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h


def frank_wolfe_step(h, dx, n_points):
    """Compute Frank-Wolfe step for the current h."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Gradient at h
    grad = np.zeros(n_points)
    # Compute gradient from the maximum overlap term
    for i in range(n_points):
        # For each position, compute the partial derivative 
        # of the maximum overlap term w.r.t. h[i]
        # This is the simplified gradient for the maximum overlap term
        grad[i] = 1 - 2 * h[i]
        # Adjust for other terms in overlap that depend on h[i]
        # Only add the contribution of the maximum overlap term
        # Add the linear term contribution from each active element
        for k in range(2 * n_points - 1):
            j = (i + k - (n_points - 1)) % n_points
            # This would involve more complex computation for actual derivative
            # But we simplify based on the contribution of the maximum term:
            if k == max_idx:
                grad[i] += 1 - 2 * h[j]
    
    # Solve linear subproblem: min v^T grad s.t. sum(v) = target_sum, 0 <= v <= 1
    # This is a standard LP problem equivalent to minimizing grad^T v with:
    # A_eq = [1, 1, ..., 1], b_eq = target_sum, bounds = [0, 1]
    target_sum = n_points / 2
    A_eq = np.array([1.0] * n_points)
    b_eq = target_sum
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Direct solution since it's simple
    # If all gradient elements are non-negative, 
    # the min is at v = 0 (but we need ∑v = target_sum)
    # Otherwise, use the algorithm that checks which variables to set to 0 or 1
    
    # For the linear subproblem, we solve min ∇h^T v subject to sum(v) = target_sum, 0 <= v <= 1
    
    # Quadratic programming approach with cvxpy for full generality
    v = cp.Variable(n_points)
    obj = cp.Minimize(grad @ v)
    constraints = [cp.sum(v) == target_sum, v >= 0, v <= 1]
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        v_opt = v.value
        if v_opt is None:
            # Fallback: use simple method
            v_opt = np.zeros(n_points)
            # Push to extremes (like Frank-Wolfe step, but in the direction of negative grad)
            # This is a heuristic: choose v such that v_i = 0 or 1 to minimize vs grad
            # Or try a simple gradient projection method
            
            # Simple method: set negative gradient directions to 1, positive to 0
            # But must still satisfy ∑v = target_sum
            v_dense = np.zeros(n_points)
            sorted_indices = np.argsort(grad)  # ascending order
            # Set smallest values to 1 and largest to 0
            v_dense[sorted_indices[:int(target_sum)]] = 1
            # Adjust to have exactly target_sum
            if np.sum(v_dense) < target_sum:
                # Set some elements to 1 more
                ones_indices = np.where(v_dense == 0)[0]
                for i in range(int(target_sum - np.sum(v_dense))):
                    v_dense[i] = 1
            elif np.sum(v_dense) > target_sum:
                # Set some to 0
                ones_indices = np.where(v_dense == 1)[0]
                for i in range(int(np.sum(v_dense) - target_sum)):
                    v_dense[ones_indices[i]] = 0
            
            v_opt = v_dense
        
    except:
        # Fallback simple optimization
        # Sort elements and take top target_sum
        v_opt = np.zeros(n_points)
        # Absorb gradient into a task 
        # Simple case: kind of greedy selection of smallest to set to 1
        idx = np.argsort(grad)  # ascending gradients
        # This approach targets the negative gradient direction strongly
        v_opt[idx[:int(target_sum)]] = 1
        sum_v = np.sum(v_opt)
        if sum_v < target_sum:
            # Need to increase some more
            for i in range(int(target_sum - sum_v)):
                v_opt[idx[int(target_sum) + i]] = 1
        elif sum_v > target_sum:
            # Too many ones, reduce
            for i in range(int(sum_v - target_sum)):
                v_opt[idx[len(v_opt) - 1 - i]] = 0
                
    return v_opt


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, target_sum)
    
    # Frank-Wolfe parameters
    max_iter = 10000
    tolerance = 1e-8
    best_c5 = float('inf')
    best_h = h.copy()
    
    for step in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if current_c5 <= 0.3807:
            break
            
        # Frank-Wolfe step
        v = frank_wolfe_step(h, dx, n_points)
        
        # Line search
        # Simple backtracking: find step size to reduce objective
        eta = 0.1  # initial step size
        
        # Try a few step sizes
        best_eta = 0.0
        best_val = evaluate_c5(h, dx)
        
        # Try steps
        for trial_eta in [0.1, 0.2, 0.3, 0.5, 0.8]:
            h_trial = (1 - trial_eta) * h + trial_eta * v
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = project_h(h_trial, target_sum)
            
            trial_c5 = evaluate_c5(h_trial, dx)
            if trial_c5 < best_val:
                best_val = trial_c5
                best_eta = trial_eta

        # Update
        eta = best_eta
        h = (1 - eta) * h + eta * v
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
        
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
