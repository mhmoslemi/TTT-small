import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    max_iter = 500
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Set up the convex relaxation problem
        h_var = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
            z >= 0
        ]
        
        # Compute overlap constraints
        for k in range(2 * n_points - 1):
            # Compute overlap for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = i + k - (n_points - 1)
                if 0 <= j < n_points:
                    overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= z)
        
        # Objective
        objective = cp.Minimize(z)
        prob = cp.Problem(objective, constraints)
        
        # Solve the convex relaxation
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            h_relaxed = h_var.value
        except:
            # If convex relaxation fails, revert to simple gradient descent
            h_relaxed = h.copy()
            
        # Rounding heuristic
        # Threshold at 0.5, then adjust to meet sum constraint
        h_rounded = np.zeros(n_points)
        for i in range(n_points):
            if h_relaxed[i] >= 0.5:
                h_rounded[i] = 1.0
            else:
                h_rounded[i] = 0.0
        
        # Adjust to meet sum constraint
        current_sum = np.sum(h_rounded)
        diff = target_sum - current_sum
        if diff != 0:
            # Find indices with 0 or 1 values to adjust
            zero_indices = np.where(h_rounded == 0)[0]
            one_indices = np.where(h_rounded == 1)[0]
            
            if diff > 0:  # Need to add more 1s
                # Pick indices with value 0 and set them to 1
                for i in range(int(diff)):
                    if len(zero_indices) > 0:
                        idx = zero_indices[i % len(zero_indices)]
                        h_rounded[idx] = 1.0
                    else:
                        break
            else:  # Need to remove 1s
                # Pick indices with value 1 and set them to 0
                for i in range(int(-diff)):
                    if len(one_indices) > 0:
                        idx = one_indices[i % len(one_indices)]
                        h_rounded[idx] = 0.0
                    else:
                        break
        
        # Normalize again to match target sum
        h_rounded = h_rounded * (target_sum / np.sum(h_rounded)) if np.sum(h_rounded) > 0 else h_rounded
        
        # Greedy improvement phase
        current_c5 = compute_c5(h_rounded, dx)
        improved = True
        max_local_iter = 100
        local_iter = 0
        while improved and local_iter < max_local_iter and time.time() - start_time < budget_s - 1:
            improved = False
            local_iter += 1
            # Try small changes to h_rounded
            for i in range(n_points):
                old_val = h_rounded[i]
                for new_val in [0.0, 1.0]:
                    if new_val != old_val:
                        h_rounded[i] = new_val
                        new_c5 = compute_c5(h_rounded, dx)
                        if new_c5 < current_c5:
                            current_c5 = new_c5
                            improved = True
                        else:
                            h_rounded[i] = old_val
                            
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_rounded.copy()
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        h = h_rounded.copy()
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
