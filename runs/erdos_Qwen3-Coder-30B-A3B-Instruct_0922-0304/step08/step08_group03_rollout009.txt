import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    ramp = np.arange(1, half + 1)
    if n % 2 == 0:
        return np.concatenate([ramp, ramp[::-1]]) / (half + 1)
    else:
        return np.concatenate([ramp, [half + 1], ramp[::-1]]) / (half + 1)

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = triangular_ramp(n_points)
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Momentum parameters
    alpha = 0.001
    beta = 0.9
    gamma = 0.99
    velocity = np.zeros(n_points)
    
    # Phase 1: Gradient descent with momentum
    max_iter_phase1 = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iter_count < max_iter_phase1 and time.time() - start_time < budget_s - 1:
        # Compute overlap and find max
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Estimate gradient via finite differences for max overlap position k
        max_k = np.argmax(overlap)
        shift_k = np.roll(1 - h, max_k)
        gradient = (1 - 2 * h) * shift_k / n_points
        
        # Update momentum
        velocity = beta * velocity + (1 - beta) * gradient
        h_candidate = h - alpha * velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            alpha *= gamma  # Decay learning rate
            no_improvement_count = 0
        else:
            h = clip_and_renormalize(h, target_sum)  # Revert, but ensure constraints
            alpha *= gamma  # Shrink step size
            no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            break
            
        iter_count += 1
    
    # Phase 2: Coordinate-wise refinement
    max_iter_phase2 = 10000
    iter_count = 0
    p = 0.2  # Percentage of coordinates to update
    update_count = max(1, int(p * n_points))
    
    while iter_count < max_iter_phase2 and time.time() - start_time < budget_s - 1:
        if best_c5 <= 0.3807:
            break
            
        # Sample random coordinates
        indices = np.random.choice(n_points, update_count, replace=False)
        
        improved = False
        for i in indices:
            # Try to reduce overlap by modifying h[i]
            current_val = h[i]
            
            # Try small increments
            step_sizes = [0.01, 0.005, 0.001]
            for step in step_sizes:
                h_test = h.copy()
                h_test[i] = min(1.0, current_val + step)
                h_test = clip_and_renormalize(h_test, target_sum)
                new_c5 = compute_c5(h_test, dx)
                if new_c5 < best_c5:
                    h = h_test.copy()
                    best_c5 = new_c5
                    improved = True
                    break
                    
                h_test[i] = max(0.0, current_val - step)
                h_test = clip_and_renormalize(h_test, target_sum)
                new_c5 = compute_c5(h_test, dx)
                if new_c5 < best_c5:
                    h = h_test.copy()
                    best_c5 = new_c5
                    improved = True
                    break
                
            if improved:
                break
            
        if not improved:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > max_no_improvement:
            break
            
        iter_count += 1
    
    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
