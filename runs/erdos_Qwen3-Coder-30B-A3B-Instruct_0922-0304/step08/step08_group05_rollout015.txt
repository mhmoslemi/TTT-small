import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    ε = 0.01
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Identify and adjust most influential indices
        # Focus on k around max_k, adjust to reduce that particular overlap
        for i in range(n_points):
            # Compute how changing h[i] affects overlap at k = max_k
            # Let's modulate amount based on current overlap
            sensitivity = 2 * (1 - 2 * h[i]) * dx
            delta = ε * sensitivity
            
            # Apply small adjustments at i and i + max_k (circular)
            h[i] -= delta
            if (i + max_k) < n_points:
                h[i + max_k] += delta
                
        # Keep values in bounds and re-normalize
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        # Dynamic learning rate
        if iteration % 100 == 0:
            ε *= 0.99
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
