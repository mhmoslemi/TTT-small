```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def project_h(h, n_points):
    """Project h to ensure it's in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def sample_shifts(n, sample_size):
    """Sample a subset of possible shifts"""
    max_shift = 2 * n - 1
    return np.random.choice(max_shift, sample_size, replace=False)

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    # Initialize with feasible h
    h_current = project_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    no_improvement_count = 0
    target_overlap = 0.38085
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 200:
        # Sample shifts for this iteration
        sample_size = min(50, n_points)
        shift_indices = sample_shifts(n_points, sample_size)
        
        # Build the QP problem
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(cp.abs(cp.conv(h_vars, 1-h_vars)[shift_indices])))
        
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == target_sum
        ]
        
        # Use CVXPY to solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                h_new = project_h(h_new, n_points)
                c5_new = compute_c5(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    no_improvement_count = 0
                    if best_c5 <= 0.3807:
                        break
                else:
                    no_improvement_count += 1
                h_current = h_new.copy()
            else:
                h_current = project_h(h_current, n_points)
                no_improvement_count += 1
        except:
            # fallback to simple gradient method
            h_current = project_h(h_current, n_points)
            no_improvement_count += 1
            
        iteration += 1
        # Adjust target based on progress
        if iteration % 10 == 0 and best_c5 < target_overlap:
            target_overlap = best_c5 * 0.999
        elif iteration % 20 == 0:
            target_overlap = max(target_overlap * 0.9995, 0.3807)
            
    # Final refinement step
    h_final = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_temp = h_final.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_final, dx):
            h_final = h_temp.copy()
    
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
```
