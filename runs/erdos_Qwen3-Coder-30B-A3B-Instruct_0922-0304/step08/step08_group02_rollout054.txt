import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_solution(h, target_sum):
    """Repair solution to ensure it meets constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_point):
    """Perform crossover between two parents."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    child1[crossover_point:] = parent2[crossover_point:]
    child2[crossover_point:] = parent1[crossover_point:]
    return child1, child2

def mutate(h, mutation_rate=0.05, max_mutation=0.1):
    """Apply mutation to solution."""
    h_new = h.copy()
    for i in range(len(h_new)):
        if np.random.random() < mutation_rate:
            delta = np.random.uniform(-max_mutation, max_mutation)
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
    return h_new

def gradient_descent_step(h, dx, target_sum, num_steps=50):
    """Refine solution using gradient descent."""
    h_refined = h.copy()
    for _ in range(num_steps):
        overlap = correlate(h_refined, 1 - h_refined, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_k = max_overlap_idx - (len(h_refined) - 1)  # Convert index to k
        
        # Adjust h to reduce overlap at this k
        # We can try to reduce h where (h+k) is 1, but this is a complex analysis
        # Instead, we use a simple approach: directional adjustment
        delta = 0.01 * np.exp(-_ / 10)
        modifier = np.zeros_like(h_refined)
        for i in range(len(h_refined)):
            if i + max_k >= 0 and i + max_k < len(h_refined):
                modifier[i] += -delta * (1 - h_refined[i + max_k])
        
        h_temp = h_refined + modifier
        h_temp = repair_solution(h_temp, target_sum)
        
        if compute_c5(h_temp, dx) < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
        else:
            break
    return h_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population size
    n_pop = 20
    
    # Initialize population
    population = []
    
    # Add initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
        population.append(h_current)
    else:
        # Create a random solution
        h_current = np.random.rand(n_points)
        h_current = repair_solution(h_current, target_sum)
        population.append(h_current)
    
    # Fill the rest of the population
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        h = repair_solution(h, target_sum)
        if not any(np.allclose(p, h) for p in population):
            population.append(h)
    
    # Track generation without improvement
    no_improvement_count = 0
    best_c5 = float('inf')
    best_h = None
    
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitnesses = [compute_c5(individual, dx) for individual in population]
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        current_best_solution = population[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_solution.copy()
            no_improvement_count = 0
        
        no_improvement_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        
        # Keep top 20% as elite
        elite_size = max(1, n_pop // 5)
        elite = sorted_pop[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Generate the rest through crossover and mutation
        while len(new_population) < n_pop:
            # Select two parents
            parent1 = sorted_pop[np.random.randint(0, n_pop//2)]
            parent2 = sorted_pop[np.random.randint(0, n_pop//2)]
            
            # Crossover
            if np.random.random() < 0.8:
                crossover_point = np.random.randint(1, n_points - 1)
                child1, child2 = crossover(parent1, parent2, crossover_point)
                # Apply mutation
                child1 = mutate(child1, 0.05, 0.1)
                child2 = mutate(child2, 0.05, 0.1)
                child1 = repair_solution(child1, target_sum)
                child2 = repair_solution(child2, target_sum)
                new_population.append(child1)
                if len(new_population) < n_pop:
                    new_population.append(child2)
            else:
                # Just mutate parent1
                child = mutate(parent1, 0.05, 0.1)
                child = repair_solution(child, target_sum)
                new_population.append(child)
                
        population = new_population
        
        # Local refinement: apply gradient descent to top solutions
        top_solutions = sorted_pop[:3]
        for i, h in enumerate(top_solutions):
            if time.time() - start_time > budget_s - 1:
                break
            refined_h = gradient_descent_step(h, dx, target_sum, 50)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < fitnesses[sorted_indices[i]]:
                population[sorted_indices[i]] = refined_h.copy()
        
        generation += 1
        
        if best_c5 <= 0.3807:
            break
    
    if best_h is None:
        # Final fallback
        fitnesses = [compute_c5(individual, dx) for individual in population]
        current_best_idx = np.argmin(fitnesses)
        best_h = population[current_best_idx].copy()
        best_c5 = fitnesses[current_best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
