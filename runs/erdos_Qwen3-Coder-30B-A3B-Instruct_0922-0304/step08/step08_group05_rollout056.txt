import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_neighborhood(h, n_candidates=10):
    """Generate neighborhood solutions by flipping or swapping elements."""
    n = len(h)
    candidates = []
    
    for _ in range(n_candidates):
        h_new = h.copy()
        
        # 30% chance of point flipping, 70% chance of swapping
        if np.random.random() < 0.3:
            # Flip two random points
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i] *= 0.97 + np.random.normal(0, 0.01)
            h_new[j] *= 0.97 + np.random.normal(0, 0.01)
        else:
            # Swap two random points
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Clip values
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize
        target_sum = n / 2
        h_new = normalize_h(h_new, target_sum)
        
        candidates.append(h_new)
    
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Cooling schedule
    temperature = 1.0
    cooling_rate = 0.999
    last_improvement = 0
    max_no_improvement = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and last_improvement < max_no_improvement:
        # Generate neighborhood
        candidates = generate_neighborhood(h, 10)
        
        # Evaluate candidates
        candidate_c5s = [compute_c5_exact(c, dx) for c in candidates]
        
        # Find best candidate not in tabu list (or allowed by aspiration)
        best_candidate_idx = -1
        best_candidate_c5 = float('inf')
        
        for i, (candidate, c5) in enumerate(zip(candidates, candidate_c5s)):
            # Check aspiration criteria
            aspiration = (c5 < best_c5)
            
            # Check if in tabu list
            in_tabu = False
            if len(tabu_list) > 0:
                for tabu_h in tabu_list:
                    if np.allclose(candidate, tabu_h):
                        in_tabu = True
                        break
            
            # Select if not tabu or aspiration
            if not in_tabu or aspiration:
                if c5 < best_candidate_c5:
                    best_candidate_c5 = c5
                    best_candidate_idx = i
        
        # Update if we found a better candidate
        if best_candidate_idx >= 0:
            h = candidates[best_candidate_idx]
            last_improvement = 0
            
            # Add current solution to tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(h.copy())
            
            # Update best solution
            if best_candidate_c5 < best_c5:
                best_c5 = best_candidate_c5
                best_h = h.copy()
        else:
            last_improvement += 1
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Update temperature
        temperature *= cooling_rate
        
        iteration += 1
        if iteration % 1000 == 0:
            cooling_rate *= 0.999
    
    # Final evaluation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
