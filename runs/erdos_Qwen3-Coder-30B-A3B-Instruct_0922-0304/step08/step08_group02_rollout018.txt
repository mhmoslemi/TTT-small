import numpy as np
from scipy.fftpack import fft, ifft
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to feasible region: clamp to [0,1] and scale to desired sum."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_surrogate(H):
    """Compute surrogate objective in frequency domain."""
    return np.abs(H * np.conj(1 - H))

def spectral_update(H, surrogate_grad, learning_rate):
    """Update frequency domain representation."""
    # Simplified gradient-like update
    H_new = H - learning_rate * surrogate_grad
    return H_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_h(h_current, target_sum)
    else:
        # Initialize with uniform distribution and random noise
        h_current = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h_current = project_h(h_current, target_sum)
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Frequency domain optimization
    iteration = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Transform to frequency domain
        H_current = fft(h_current)
        
        # Compute surrogate in frequency domain
        surrogate = compute_surrogate(H_current)
        
        # Identify dominant frequency components for update
        # Use a simplified approach: target high values in surrogate
        # and adjust spectrum to reduce those
        surrogate_grad = np.zeros_like(H_current)
        idx_max = np.argmax(surrogate)
        surrogate_grad[idx_max] = 1.0 if np.real(H_current[idx_max]) > 0.5 else -1.0
        
        # Apply update
        H_new = spectral_update(H_current, surrogate_grad, learning_rate)
        
        # Reconstruct in time domain
        h_new = np.real(ifft(H_new))
        
        # Apply feasibility constraints
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        # Accept better solution
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current solution
        h_current = h_new.copy()
        
        # Adjust learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.5
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    h_refined = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Random local search
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, target_sum)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
