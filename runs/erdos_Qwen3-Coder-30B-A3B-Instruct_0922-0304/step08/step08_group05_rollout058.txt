import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def get_neighborhood(h, n_points, target_sum, dx):
    """Generate neighborhood candidates using flips and swaps."""
    candidates = []
    # Generate 10 candidates
    for _ in range(10):
        h_new = h.copy()
        # Randomly choose operation: flip (30%) or swap (70%)
        if random.random() < 0.3:
            # Flip operation
            i, j = random.sample(range(n_points), 2)
            h_new[i] *= 0.97 + np.random.normal(0, 0.01)
            h_new[j] *= 0.97 + np.random.normal(0, 0.01)
        else:
            # Swap operation
            i, j = random.sample(range(n_points), 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        c5 = compute_c5_exact(h_new, dx)
        candidates.append((h_new, c5))
    
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    no_improvement = 0
    temperature = 1.0
    cooling_rate = 0.999
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and no_improvement < 1000:
        # Generate neighborhood
        candidates = get_neighborhood(h, n_points, target_sum, dx)
        
        # Sort by C5 value
        candidates.sort(key=lambda x: x[1])
        
        # Find best non-tabu candidate
        selected_h = None
        selected_c5 = float('inf')
        for h_candidate, c5_candidate in candidates:
            # Check if in tabu list (by checking if it's a similar configuration)
            # For simplicity, we use the tuple of values
            h_tuple = tuple(h_candidate.round(5))
            if h_tuple not in tabu_list:
                selected_h = h_candidate
                selected_c5 = c5_candidate
                break
            elif c5_candidate < best_c5 + 1e-10:  # Aspiration criterion
                selected_h = h_candidate
                selected_c5 = c5_candidate
                break
        
        # Update tabu list and temperature
        if selected_h is not None:
            h = selected_h
            h_tuple = tuple(h.round(5))
            tabu_list.append(h_tuple)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            temperature *= cooling_rate
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update global best
        current_c5 = compute_c5_exact(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final computation
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
