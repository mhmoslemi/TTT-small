import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Ensure h is normalized and clipped
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    # Objective function
    def objective(h):
        return compute_c5(h, dx)
    
    # Set up optimization parameters
    max_iter = 1000
    iter_count = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # SLSQP optimization
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': max_iter, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            c5_bound = compute_c5(h, dx)
            if c5_bound < best_c5:
                best_h = h.copy()
                best_c5 = c5_bound
    except Exception:
        pass  # Continue with fallback
    
    # Additional local refinement with gradient descent
    h = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Select top indices to update
        max_indices = np.where(np.abs(overlap) == current_max)[0]
        if len(max_indices) == 0:
            break
            
        # Simple gradient-like update
        update = np.zeros(n_points)
        for i in range(n_points):
            grad = 0
            for k in range(n_points):
                if i + k < n_points:
                    grad += (1 - 2 * h[i]) * (1 - h[i + k])
                if i - k >= 0:
                    grad += (1 - 2 * h[i]) * (1 - h[i - k])
            update[i] = grad
            
        # Update with adaptive step size
        step_size = 0.001
        h_new = h + step_size * update
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Check improvement
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
