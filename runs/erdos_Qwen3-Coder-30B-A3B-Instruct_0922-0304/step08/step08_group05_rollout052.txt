import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength=0.1):
    """Apply Gaussian mutation to an individual."""
    h_new = h + np.random.normal(0, mutation_strength, len(h))
    h_new = np.clip(h_new, 0, 1)
    return h_new

def crossover_parents(p1, p2):
    """Perform average-based crossover."""
    child = (p1 + p2) / 2.0
    return child

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    population_size = 50
    mutation_strength = 0.1
    
    # Initialize population
    population = []

    h = normalize_h(h, target_sum)
    population.append(h.copy())
    
    # Generate the rest randomly
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    # Track generations without improvement
    no_improvement_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 1000:
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            c5 = compute_c5_exact(h, dx)
            fitnesses.append(c5)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = []
        for _ in range(population_size):
            selected.append(tournament_selection(population, fitnesses))
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child = crossover_parents(parent1, parent2)
            child = normalize_h(child, target_sum)
            
            # Apply mutation
            child = mutate_individual(child, mutation_strength)
            child = np.clip(child, 0, 1)
            child = normalize_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Ensure final solution meets all constraints
    if best_h is not None:

        fitnesses = [compute_c5_exact(h, dx) for h in population]
