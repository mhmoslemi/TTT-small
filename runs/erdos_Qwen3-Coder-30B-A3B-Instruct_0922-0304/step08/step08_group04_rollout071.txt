import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize_scalar
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: 0 <= h <= 1, sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the C5 objective."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Compute gradient based on the maximum overlap
    grad = np.zeros_like(h)
    for i in range(n_points):
        # For each position, compute contribution to gradient from max overlap
        # When we shift h by max_idx positions, we get h[i] * (1 - h[i + max_idx])
        # So the gradient is approximately 1 - 2*h[i] for the active terms
        # But we just use a simplified approximation
        grad[i] = 1 - 2 * h[i]
        
    return grad

def frank_wolfe_step(grad, n_points):
    """Solve linear subproblem: min <grad, v> s.t. sum(v) = n_points/2, 0 <= v <= 1"""
    # Solve linear program to find v that minimizes grad^T * v
    # subject to sum(v) = n_points/2, 0 <= v[i] <= 1
    
    # Since we want to minimize grad^T * v, we assign to the most negative components
    # But we also need to ensure the sum constraint of n_points/2
    
    # Initialize v
    v = np.zeros(n_points)
    
    # Sort indices by gradient values
    sorted_indices = np.argsort(grad)
    
    # Fill with smallest elements first
    target_sum = n_points / 2
    remaining_sum = target_sum
    
    for i in range(n_points):
        idx = sorted_indices[i]
        if remaining_sum >= 1:
            v[idx] = 1
            remaining_sum -= 1
        elif remaining_sum > 0:
            v[idx] = remaining_sum
            remaining_sum = 0
        else:
            break
    
    # Normalize so we sum correctly if necessary
    total = np.sum(v)
    if total > 0:
        v = v * (target_sum / total)
    
    return v

def backtracking_line_search(h, v, grad, dx, n_points, alpha=1.0, beta=0.5, c=1e-4):
    """Perform backtracking line search."""
    t = alpha
    current_c5 = evaluate_c5(h, dx)
    
    while True:
        h_new = (1 - t) * h + t * v
        h_new = project_to_feasible(h_new, n_points / 2)
        new_c5 = evaluate_c5(h_new, dx)
        
        # Armijo condition: f(x+t*d) <= f(x) + c*t*grad^T*d
        if new_c5 <= current_c5 + c * t * np.dot(grad, v - h):
            return t
        else:
            t *= beta

def frank_wolfe_optimize(h, dx, n_points, max_iter=10000, tolerance=1e-8):
    """Optimize using Frank-Wolfe algorithm."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Solve linear subproblem
        v = frank_wolfe_step(grad, n_points)
        
        # Line search
        alpha = 1.0
        t = backtracking_line_search(h, v, grad, dx, n_points, alpha)
        
        # Update
        h = (1 - t) * h + t * v
        h = project_to_feasible(h, n_points / 2)
        
        # Evaluation
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping
        if i % 100 == 0:
            if abs(current_c5 - best_c5) < tolerance:
                break
        
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

    # Final check
