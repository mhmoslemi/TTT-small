import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, noise_magnitude):
    """Add noise to individual."""
    noise = np.random.normal(0, noise_magnitude, len(h))
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, len(h) / 2)
    return h_new

def crossover_parents(h1, h2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return h1.copy()
    point = random.randint(1, len(h1) - 1)
    child = np.concatenate([h1[:point], h2[point:]])
    return child

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), size=tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def genetic_algorithm(n_points, population_size=50, max_generations=1000,
                      crossover_rate=0.8, mutation_rate=0.3, 
                      initial_noise=0.05, noise_decay=0.99, 
                      target_c5=0.3807, budget_s=600):
    """Genetic algorithm to minimize C5."""
    start_time = time.time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            individual = mutate_individual(parent, initial_noise)
            population.append(individual)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, n_points / 2)
            population.append(h)
            
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_individual = None
    noise_magnitude = initial_noise
    stagnation_count = 0
    
    for generation in range(max_generations):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1:
            break
            
        # Evaluate fitness in parallel
        fitnesses = Parallel(n_jobs=1)(delayed(evaluate_c5)(ind, dx) for ind in population)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= target_c5:
            break
            
        if stagnation_count > 50:
            # Inject diversity by mutating best solution
            for i in range(population_size):
                if random.random() < 0.2:
                    population[i] = mutate_individual(best_individual, noise_magnitude * 2)
            stagnation_count = 0
            
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child = crossover_parents(parent1, parent2, crossover_rate)
            
            # Mutation
            if random.random() < mutation_rate:
                child = mutate_individual(child, noise_magnitude)
                
            new_population.append(child)
            
        population = new_population
        noise_magnitude *= noise_decay
        
    return best_individual, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    best_h, best_c5 = genetic_algorithm(n_points, budget_s=budget_s)
    
    # Final evaluation
    dx = 2.0 / n_points
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
