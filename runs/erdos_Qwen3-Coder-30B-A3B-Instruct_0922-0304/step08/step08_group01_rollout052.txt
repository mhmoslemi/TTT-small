import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return compute_c5(h, dx)
    
    # Optimization with SLSQP
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None  # Placeholder callback
        )
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))  # Renormalize to exact sum
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    except Exception:
        pass  # Fallback to original h
    
    # Final fine-tuning with coordinate descent
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        improved = False
        
        # Coordinate descent on each variable
        for i in range(n_points):
            orig_h_i = best_h[i]
            
            # Try small perturbation
            step_sizes = np.array([-0.01, -0.005, 0.005, 0.01])
            best_h_i = orig_h_i
            best_c5_i = best_c5
            
            for step in step_sizes:
                new_h_i = orig_h_i + step
                
                # Ensure bounds
                if 0 <= new_h_i <= 1:
                    # Temporarily update
                    test_h = best_h.copy()
                    test_h[i] = new_h_i
                    
                    # Adjust other entries to maintain sum
                    delta = new_h_i - orig_h_i
                    # Distribute delta among other entries (simple approach)
                    non_i_indices = [j for j in range(n_points) if j != i]
                    if non_i_indices:
                        distribute_amount = delta / len(non_i_indices)
                        for j in non_i_indices:
                            test_h[j] = np.clip(test_h[j] - distribute_amount, 0, 1)
                    
                    new_c5 = compute_c5(test_h, dx)
                    
                    if new_c5 < best_c5_i:
                        best_c5_i = new_c5
                        best_h_i = new_h_i
            
            # Apply the best change if it improves
            if best_h_i != orig_h_i:
                best_h[i] = best_h_i
                improved = True
                
        if not improved:
            break
            
        # Renormalize
        best_h = best_h * (target_sum / np.sum(best_h))
        
        # Update best if improved
        new_c5 = compute_c5(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    # Ensure final normalization
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
