import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create variables for the QP
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints for overlap
    constraints = []
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Compute overlap constraints for each shift
    for k in range(-n_points + 1, n_points):
        # Calculate overlap for shift k efficiently
        overlap_expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap_expr += h_var[i] * (1 - h_var[j])
        constraints.append(overlap_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with ECOS solver
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, reltol=1e-6)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_opt = h_var.value
            c5_bound = t.value
        else:
            # Fallback to original solution if optimization fails
            h_opt = h
            c5_bound = evaluate_c5(h, dx)
    except:
        h_opt = h
        c5_bound = evaluate_c5(h, dx)
    
    # Verify constraints
    if np.abs(np.sum(h_opt) - n_points / 2) > 1e-5:
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Evaluate final C5
    c5_final = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), float(c5_final), n_points)
