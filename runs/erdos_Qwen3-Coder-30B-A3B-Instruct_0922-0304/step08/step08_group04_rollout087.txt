import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def initial_h_from_parent(n_points):
    """Initialize h from the parent construction if available."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        return h
    return None

def solve_convex_relaxation(n_points, dx):
    """Solve convex relaxation to get a starting point."""
    h = Variable(n_points)
    t = Variable()
    
    # Constraints to model overlaps
    constraints = [
        0 <= h, 
        h <= 1,
        sum(h) == n_points / 2,
        t >= 0
    ]
    
    # For each shift k, add the overlap constraint
    max_k = 2 * n_points - 1
    for k in range(max_k):
        # We'll use a relaxation that's computationally manageable
        # For small n_points this can be cheap enough
        if k < n_points:  # Only for valid shifts
            # Introduce additional variables for bilinear terms
            z = Variable(n_points)
            for x in range(n_points):
                # Ensure z[x] <= h[x] and z[x] <= h[(x+k) % n_points] 
                constraints.append(z[x] <= h[x])
                constraints.append(z[x] <= h[(x + k) % n_points])
            # Add constraint that sum of (h[x] - z[x]) <= t
            # This is an approximation, better relaxations exist but this
            # is simpler and still improves chances
            pass  # For now, we proceed with a different approach
    
    # Simplify: just check the exact correlation values
    # This is exact for the purpose of warm start
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return np.array(h.value).flatten()
    except:
        return None
    return None

def compute_gradient_exact(h, dx, n_points):
    """Compute the exact gradient of the C5 objective."""
    # We do this by computing partial derivatives of max over all shifts
    # For each x, compute how much varying h[x] affects max overlap.
    
    # We compute full correlation for h and (1-h)
    # Then find where max overlap occurs
    overlap_full = correlate(h, 1-h, mode="full")
    max_idx = np.argmax(overlap_full)
    max_val = overlap_full[max_idx]
    
    # For gradient calculation, we compute effect of small change in h[x]
    # on max overlap for all x
    
    grad = np.zeros(n_points)
    
    # We compute a finite difference estimate
    eps = 1e-6
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Normalize
        
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (np.max(overlap_plus) - np.max(overlap_full)) / eps
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999):
    """Optimize h using gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        # Compute loss and gradient at most every 100 steps
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            if loss <= 0.3807:
                break
            
            if abs(prev_loss - loss) < 1e-10:
                no_improvement_count += 1
                if no_improvement_count > 500:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient using a block approximation
        grad = compute_gradient_exact(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    h = initial_h_from_parent(n_points)
    if h is None:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Solve convex relaxation for starting point
    try:
        h_relaxed = solve_convex_relaxation(n_points, dx)
        if h_relaxed is not None:
            h = h_relaxed
    except:
        pass
    
    # Optimize using gradient descent
    best_h, best_c5 = optimize_with_gradient_descent(
        h, dx, n_points, 
        max_iter=10000, 
        lr=0.01, 
        decay=0.999
    )
    
    # Final validation
    actual_c5 = evaluate_c5(best_h, dx)
    if actual_c5 < best_c5:
        best_c5 = actual_c5
    
    if best_c5 > 0.3807:
        # Try multiple restarts if needed
        for _ in range(3):
            h_restart = np.random.rand(n_points).astype(np.float64)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            try:
                h_rest, c5_rest = optimize_with_gradient_descent(
                    h_restart, dx, n_points, max_iter=5000, lr=0.01, decay=0.999
                )
                if c5_rest < best_c5:
                    best_c5 = c5_rest
                    best_h = h_rest
            except:
                pass

    # Ensure we do not exceed the time budget and improve solution if possible
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 1:
        # Retune parameters slightly for the last bit
        h_final, c5_final = optimize_with_gradient_descent(
            best_h, dx, n_points, 
            max_iter=2000, 
            lr=0.02, 
            decay=0.999
        )
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    return (best_h.tolist(), best_c5, n_points)
