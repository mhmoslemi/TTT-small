import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip values to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Generate a triangular ramp for initialization."""
    x = np.linspace(0, 2, n)
    h = np.minimum(x, 2 - x)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Parameters
    learning_rate = 0.001
    momentum = 0.9
    decay_rate = 0.99
    coord_sample_rate = 0.2
    max_iter = 10000
    
    # Precompute coordinate indices for sampling
    coord_indices = np.arange(n_points)
    
    velocity = np.zeros(n_points)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap and find maximum position
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Phase 1: Gradient Descent with Momentum
        # Find the most problematic position
        max_pos = np.argmax(overlap)
        # Compute approximate gradient wrt to position max_pos
        h_shifted = np.roll(h, max_pos)
        grad = (1 - 2 * h) * (1 - h_shifted)
        
        # Apply momentum
        velocity = momentum * velocity + (1 - momentum) * grad
        h_candidate = h - learning_rate * velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate candidate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        if candidate_c5 < overlap_max:
            h = h_candidate
            learning_rate *= decay_rate
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            learning_rate *= decay_rate
            if no_improvement_count > max_no_improvement:
                # Switch to coordinate-wise optimization
                break
        
        iter_count += 1
    
    # Phase 2: Coordinate-wise optimization
    max_iter = 20000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Sample coordinates to update
        num_to_update = max(1, int(n_points * coord_sample_rate))
        update_indices = np.random.choice(coord_indices, num_to_update, replace=False)
        
        current_c5 = compute_c5(h, dx)
        
        # Try to improve each sampled coordinate
        improved = False
        for i in update_indices:
            h_trial = h.copy()
            
            # Try small steps in both directions
            step_sizes = [0.001, 0.005, 0.01]
            best_h_trial = h_trial.copy()
            best_trial_c5 = current_c5
            
            for step in step_sizes:
                # Try increasing
                h_trial[i] = min(1.0, h[i] + step)
                h_trial = clip_and_renormalize(h_trial, target_sum)
                trial_c5 = compute_c5(h_trial, dx)
                if trial_c5 < best_trial_c5:
                    best_trial_c5 = trial_c5
                    best_h_trial = h_trial.copy()
                
                # Try decreasing
                h_trial[i] = max(0.0, h[i] - step)
                h_trial = clip_and_renormalize(h_trial, target_sum)
                trial_c5 = compute_c5(h_trial, dx)
                if trial_c5 < best_trial_c5:
                    best_trial_c5 = trial_c5
                    best_h_trial = h_trial.copy()
            
            if best_trial_c5 < current_c5:
                h = best_h_trial
                improved = True
                break
                
        if not improved:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > max_no_improvement:
            break
            
        iter_count += 1
        
        # Occasionally check for better solution
        if iter_count % 100 == 0:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if best_c5 <= 0.3807:
                break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
