import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_val = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_val = max(max_val, overlap)
    return max_val * dx

    start_time = time.time()
    
    n_points = 70

    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Perturbation parameters
    perturbation = 1e-3
    decay_factor = 0.98
    improvement_threshold = 1e-5
    max_iterations = 100
    
    # Alternating optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Clone and perturb for partition
        h_perturbed = h.copy()
        h_perturbed += np.random.uniform(-perturbation, perturbation, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        
        # Partition even and odd indices
        even_idx = np.arange(0, n_points, 2)
        odd_idx = np.arange(1, n_points, 2)
        
        # Alternative optimization: fix odd, optimize even
        def optimize_even_subproblem(h_fixed_odd):
            def objective(h_even):
                # Build full h vector
                h_full = h_fixed_odd.copy()
                h_full[even_idx] = h_even
                return evaluate_overlap(h_full, dx)
            
            # Constraints
            bounds = [(0, 1) for _ in range(len(even_idx))]
            
            # Initial guess: use current values
            x0 = h[even_idx].copy()
            
            # Minimize with SLSQP
            result = minimize(objective, x0, method='SLSQP', bounds=bounds,

            if result.success:
                h_full = h_fixed_odd.copy()
                h_full[even_idx] = result.x
                return h_full
            else:
                return h_fixed_odd
        
        # Fix even indices and optimize odd indices
        def optimize_odd_subproblem(h_fixed_even):
            def objective(h_odd):
                h_full = h_fixed_even.copy()
                h_full[odd_idx] = h_odd
                return evaluate_overlap(h_full, dx)
            
            bounds = [(0, 1) for _ in range(len(odd_idx))]
            
            # Initial guess: current values
            x0 = h[odd_idx].copy()
            
            result = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                            options={'ftol': 1e-10, 'maxiter': 100})

                h_full = h_fixed_even.copy()
                h_full[odd_idx] = result.x
                return h_full

        h_even_optimized = optimize_even_subproblem(h_perturbed[odd_idx])

        # Perturbation decay

    # Final normalization

        best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
