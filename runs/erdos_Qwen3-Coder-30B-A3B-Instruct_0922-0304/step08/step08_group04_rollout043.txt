```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def partition_search(initial_h, dx, n_points, max_time):
    """Perform partitioning search for improved configuration."""
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    start_time = time.time()
    
    # Try different numbers of segments
    for n_segments in range(2, 11):
        if time.time() - start_time > max_time - 2:
            break
            
        # Initialize breakpoints
        breakpoints = np.linspace(0, 2, n_segments + 1)
        
        # Try different segment values
        segment_values = np.linspace(0.1, 0.9, 10)  # Reduced search space for speed
        
        for _ in range(500):  # Limit iterations
            if time.time() - start_time > max_time - 2:
                break
                
            # Create new segment values
            h_trial = np.zeros(n_points)
            for i in range(n_segments):
                h_val = np.random.choice(segment_values)
                start_idx = int(breakpoints[i] / dx)
                end_idx = int(breakpoints[i+1] / dx)
                if end_idx > start_idx:
                    h_trial[start_idx:end_idx] = h_val
                    
            # Normalize
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = normalize_h(h_trial, n_points / 2)
            
            # Evaluate
            c5_trial = evaluate_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                
            # Early stopping
            if best_c5 <= 0.3807:
                break
                
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Run partitioning search
    best_h, best_c5 = partition_search(h, dx, n_points, budget_s)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
