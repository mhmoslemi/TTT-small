import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute gradient of C5 with respect to h."""
    return (1 - 2 * h) / n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
    
    # Keep track of the best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    max_gens = 2000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        new_h = np.random.rand(n_points)
        new_h = repair_h(new_h, target_sum)
        population.append(new_h)
    
    gen = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while gen < max_gens and time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we improved
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if best_c5 <= 0.3807:
            break
            
        if no_improvement > max_no_improvement:
            # If we're stuck, mix in some randomness
            mutation_rate = min(0.3, mutation_rate * 1.1)
            max_no_improvement = min(500, max_no_improvement * 1.1)
        
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        
        while len(new_population) < pop_size:
            # Selection (tournament)
            parent1 = population[np.random.randint(0, pop_size // 2)]
            parent2 = population[np.random.randint(0, pop_size // 2)]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Repair and refinement
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            # Local refinement with gradient descent
            if np.random.rand() < 0.5:
                for _ in range(10):
                    grad = compute_gradient(child1, n_points)
                    step_size = 0.01
                    child1 -= step_size * grad
                    child1 = repair_h(child1, target_sum)
            
            if np.random.rand() < 0.5:
                for _ in range(10):
                    grad = compute_gradient(child2, n_points)
                    step_size = 0.01
                    child2 -= step_size * grad
                    child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        gen += 1
    
    return (best_h.tolist(), best_c5, n_points)
