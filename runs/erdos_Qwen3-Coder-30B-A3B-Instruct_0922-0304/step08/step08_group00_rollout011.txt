import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    # Compute correlation of h with 1-h (with full mode)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Maximum value gives the maximum overlap
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use fixed size for efficiency
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Default construction
        h = np.ones(n_points) * 0.5
    
    # Normalize sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Initialize variables
    iter_count = 0
    max_iter = 200
    improvement_threshold = 1e-6
    decay_factor = 0.98
    perturbation_delta = 1e-3
    max_perturbation = 1e-2
    
    def project_and_normalize(x):
        """Project to [0,1] and normalize sum to target_sum."""
        x = np.clip(x, 0, 1)
        current_sum = np.sum(x)
        if current_sum > 0:
            x = x * (target_sum / current_sum)
        return x
    
    def solve_subproblem(indices, fixed_vals, var_vals, is_even):
        """Solve for a subset of variables, fixing the rest."""
        # Define variables to optimize
        free_indices = np.array(indices)
        fixed_vals = np.array(fixed_vals)
        var_vals = np.array(var_vals)
        
        def objective(h_var):
            # Recreate full h vector
            h_full = best_h.copy()
            h_full[free_indices] = h_var
            
            # Project and normalize
            h_full = project_and_normalize(h_full)
            
            # Evaluate overlap
            return evaluate_overlap(h_full, dx)
            
        bounds = [(0, 1) for _ in h_var]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Use SLSQP with the previous solution as initialization
        result = minimize(objective, var_vals, 
                         method='SLSQP', bounds=bounds, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            return result.x
        else:
            return var_vals
            
    # Main optimization loop
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Create a small perturbation
        pert_h = h + np.random.uniform(-perturbation_delta, perturbation_delta, n_points)
        pert_h = project_and_normalize(pert_h)
        
        # Partition variables into even and odd indices
        even_indices = np.arange(0, n_points, 2)
        odd_indices = np.arange(1, n_points, 2)
        
        # Extract values
        even_vals = h[even_indices]
        odd_vals = h[odd_indices]
        
        # Fix odd and optimize even
        new_even = solve_subproblem(even_indices, h[odd_indices], even_vals, True)
        
        # Reconstruct h
        h_new = h.copy()
        h_new[even_indices] = new_even
        
        # Fix even and optimize odd
        new_odd = solve_subproblem(odd_indices, h_new[even_indices], odd_vals, False)
        
        # Reconstruct h
        h_new[odd_indices] = new_odd
        
        # Project and normalize
        h_new = project_and_normalize(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Update best if improvement
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Perturb and continue
        h = h_new.copy()
        iter_count += 1
        
        # Adapt delta
        perturbation_delta *= decay_factor
        if perturbation_delta < 1e-6:
            perturbation_delta = 1e-6
            
    # Final refinement with full optimization
    def final_objective(h_full):
        return evaluate_overlap(h_full, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(final_objective, best_h, 
                     method='SLSQP', bounds=bounds, 
                     constraints=constraints, 
                     options={'maxiter': 500, 'ftol': 1e-10})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
