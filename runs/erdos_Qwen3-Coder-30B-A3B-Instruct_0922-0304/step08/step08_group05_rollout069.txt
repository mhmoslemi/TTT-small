import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Additional parameters
    epsilon = 0.01
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and last_improvement < max_iter:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)  # Convert index to lag
        
        # Adjust h to reduce the maximum overlap at lag max_k
        # Create a sensitivity vector based on positions that contribute to max_k
        sensitivity = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + max_k < n_points:
                # Influence on overlap at lag max_k
                sensitivity[i] += 1 - 2 * h[i + max_k]
            if 0 <= i - max_k < n_points:
                # Influence on overlap at lag -max_k 
                sensitivity[i] += 1 - 2 * h[i - max_k]
        
        # Make adjustments to reduce the worst overlap
        step_size = epsilon * dx
        delta = np.zeros(n_points)
        for i in range(n_points):
            if abs(sensitivity[i]) > 1e-10:
                delta[i] += -step_size * sensitivity[i]
        
        # Apply adjustments to h
        h_new = h + delta
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Accept new solution if it's better
        h = h_new
        
        iteration += 1
        if iteration % 100 == 0:
            epsilon *= 0.999
        
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
