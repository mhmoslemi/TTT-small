import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    # For each possible shift k
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx

    current_sum = np.sum(h)
    if current_sum > 0:

    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Function to compute objective with respect to C
    def obj_with_c(c_val, h_vec):
        # Compute overlap for current h
        n = len(h_vec)

        return max_overlap * dx - c_val

    def overlap_func(h_vec):
        return evaluate_overlap(h_vec, dx)

        return np.sum(h_vec) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
