import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint
    def constraint_fn(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_fn}
    
    # Use SLSQP optimizer
    max_iter = 1000
    iter_count = 0
    
    try:
        while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
            # Optimize
            result = minimize(
                lambda x: compute_c5(x, dx),
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 10, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))  # Renormalize
                
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                
                # Early stopping if target achieved
                if best_c5 <= 0.3807:
                    break
            else:
                # Fallback to simple gradient descent if optimizer fails
                gradients = np.zeros(n_points)
                for i in range(n_points):
                    grad = 0
                    for k in range(n_points):
                        if i + k < n_points:
                            grad += (1 - 2 * h[i]) * (1 - h[i + k])
                        if i - k >= 0:
                            grad += (1 - 2 * h[i]) * (1 - h[i - k])
                    gradients[i] = grad

                update_count = max(1, n_points // 10)
                idxs = np.argsort(np.abs(gradients))[::-1][:update_count]
                
                for i in idxs:
                    step_size = 0.01
                    grad = gradients[i]
                    if grad != 0:
                        step_size = min(1.0, max(0.0, -grad * 0.01))
                    h[i] = h[i] + step_size * (1 - 2 * h[i])
                    h[i] = np.clip(h[i], 0, 1)
                
                h = h * (target_sum / np.sum(h))
                
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                
                if best_c5 <= 0.3807:
                    break
            
            iter_count += 1
            
    except Exception:
        # If any error occurs, return best solution so far
        pass
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
