import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Convex optimization approach
    # Variables
    h_vars = cp.Variable(n_points)
    
    # Objective function
    # We want to minimize max_k integral of h(x)(1-h(x+k)) dx
    # This is equivalent to minimizing t subject to:
    # integral of h(x)(1-h(x+k)) dx <= t for all k
    
    # Let's build a complete convex formulation
    # Phrased as minimizing max overlap
    
    # Try using convex optimization solver
    t = cp.Variable()
    
    # Define constraints for all shifts 0 to n_points-1
    constraints = [h_vars >= 0, h_vars <= 1]
    constraints.append(cp.sum(h_vars) == target_sum)
    
    # For each shift k compute the overlap integral
    # We'll do this with a more direct method using scipy
    # and refine using local optimization around solution
    
    # Start by solving a simpler version using CVXPY
    # If the problem size is too large for direct convex, 
    # abandon this path and use local refinement
    
    # Instead, use a hybrid approach:
    # 1. Start with a good heuristic
    # 2. Use a local optimizer like SLSQP
    # 3. Steady-state gradual refinement
    
    # Let's try a direct convex formulation approach for small size:
    
    # Set up initial good solution and then refine
    h_solution = h.copy()
    
    # Local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Try multiple starting points
    best_local = h_solution.copy()
    best_local_c5 = evaluate_overlap(h_solution, dx)
    
    for trial in range(5):
        # Random perturbations
        h_trial = h_solution + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        current_sum = np.sum(h_trial)
        if current_sum > 0:
            h_trial = h_trial * (target_sum / current_sum)
        
        # Refine with scipy
        result = minimize(objective, h_trial, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            refined_c5 = evaluate_overlap(refined_h, dx)
            if refined_c5 < best_local_c5:
                best_local = refined_h
                best_local_c5 = refined_c5
        else:
            # Try with original if infeasible
            refined_c5 = evaluate_overlap(h_trial, dx)
            if refined_c5 < best_local_c5:
                best_local = h_trial
                best_local_c5 = refined_c5
    
    # Perform a few rounds of gradient descent
    h_opt = best_local.copy()
    step_size = 0.01
    iter_count = 0
    
    while time.time() - start_time < budget_s - 10:
        iter_count += 1
        if iter_count > 1000:
            break
            
        h_eval = h_opt.copy()
        h_eval = np.clip(h_eval, 0, 1)
        current_sum = np.sum(h_eval)
        if current_sum > 0:
            h_eval = h_eval * (target_sum / current_sum)
        
        # Compute gradients using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_overlap(h_eval, dx)
        
        for i in range(n_points):
            h_plus = h_eval.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            sum_plus = np.sum(h_plus)
            if sum_plus > 0:
                h_plus = h_plus * (target_sum / sum_plus)
            
            c5_plus = evaluate_overlap(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step
        h_new = h_eval - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Check improvement
        c5_new = evaluate_overlap(h_new, dx)
        if c5_new < current_c5:
            h_opt = h_new.copy()
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local = h_opt.copy()
        else:
            step_size *= 0.95
    
    # Final check
    final_h = np.clip(best_local, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
