import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Initialize relaxation variables
    h_relaxed = h.copy()
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Relaxation stage: solve a convex relaxation of the problem
        # We form a linear program to minimize the maximum correlation
        h_relaxed = h.copy()
        
        # Create variables for relaxed problem
        h_vars = cp.Variable(n_points)
        z = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add constraint for max correlation
        # For each k, we want to compute the correlation term
        # This is a simplified view - in practice, we evaluate it on a grid
        # We will try to solve a simpler relaxation: do policy iteration
        
        # Instead of full LP, we'll do a more practical alternating direction method
        # If we could construct a good h_b from h_relaxed, then compute
        # rounding and adjust to sum
        # Round to get good binary structure
        
        # Binary rounding heuristic
        median_h = np.median(h_relaxed)
        h_binary = (h_relaxed > median_h).astype(float)
        sum_binary = np.sum(h_binary)
        
        # Adjust binary to match constraint
        diff = sum_binary - target_sum
        if diff > 0:
            # Too many 1s: flip some to 0
            to_flip = np.argsort(h_binary)[::-1][:int(diff)]
            h_binary[to_flip] = 0
        elif diff < 0:
            # Too many 0s: flip some to 1
            to_flip = np.argsort(h_binary)[:int(-diff)]
            h_binary[to_flip] = 1
            
        # Check if this new binary vector improves the objective
        h_candidate = h_binary.copy()
        c5_candidate = compute_c5(h_candidate, dx)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            
        # Intensify search with local improvement
        # Small random mutations to explore nearby points
        h_mutation = h_candidate.copy()
        # Perturb some fraction of the points
        num_perturb = max(1, n_points // 20)
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in perturb_indices:
            # Add small Gaussian noise then clip
            h_mutation[i] = np.clip(h_mutation[i] + np.random.normal(0, 0.02), 0, 1)
            
        # Renormalize and check if it's better
        h_mutation = normalize_h(h_mutation, target_sum)
        c5_mutation = compute_c5(h_mutation, dx)
        
        if c5_mutation < best_c5:
            best_c5 = c5_mutation
            best_h = h_mutation.copy()
            
        # One step of gradient-based local search on h
        # Simply mutate each point a little then evaluate
        h_local = h.copy()
        for i in range(n_points):
            # Mutate one point
            old_val = h_local[i]
            new_val = np.clip(old_val + np.random.normal(0, 0.01), 0, 1)
            h_local[i] = new_val
            
        h_local = normalize_h(h_local, target_sum)
        c5_local = compute_c5(h_local, dx) 
        
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()
            
        iter_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
