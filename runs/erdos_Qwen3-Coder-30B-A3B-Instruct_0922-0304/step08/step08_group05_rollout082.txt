import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_constraints(h, target_sum):
    """Repair constraints by clipping and normalizing."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def generate_neighborhood(h, target_sum, n_candidates=10):
    """Generate neighborhood by flipping and swapping points."""
    candidates = []
    n = len(h)
    for _ in range(n_candidates):
        h_new = h.copy()
        
        # Randomly choose between flip or swap (70% swap)
        if random.random() < 0.7:
            # Swap two random positions
            i, j = random.sample(range(n), 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Flip two random positions
            i, j = random.sample(range(n), 2)
            h_new[i] = max(0, min(1, h_new[i] * 0.97 + np.random.normal(0, 0.01)))
            h_new[j] = max(0, min(1, h_new[j] * 0.97 + np.random.normal(0, 0.01)))
        
        # Repair constraints
        h_new = repair_constraints(h_new, target_sum)
        candidates.append(h_new)
    
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a more structured approach
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = repair_constraints(h, target_sum)
    
    # Initialize tabu list (stores recent solutions)
    tabu_list = []
    tabu_tenure = 100
    
    # Parameters
    temperature = 1.0
    cooling_rate = 0.999
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    no_improvement_count = 0
    iteration = 0
    
    # Main search loop
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 1000:
        # Generate neighborhood
        candidates = generate_neighborhood(h, target_sum, 10)
        
        # Evaluate candidates (but not the current solution if it's in the neighborhood)
        candidate_c5s = [compute_c5_exact(candidate, dx) for candidate in candidates]
        
        # Check aspiration criteria: if a candidate is better than current best, accept it
        best_candidate_idx = np.argmin(candidate_c5s)
        best_candidate_c5 = candidate_c5s[best_candidate_idx]
        best_candidate = candidates[best_candidate_idx]
        
        # Check tabu status and update
        if best_candidate_c5 < best_c5:
            # Accept better solution
            h = best_candidate
            best_c5 = best_candidate_c5
            best_h = h.copy()
            no_improvement_count = 0
            # Add to tabu list
            tabu_list.append(h.copy())
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        else:
            # If the best from neighborhood isn't better, we may still accept it with probability
            delta = best_candidate_c5 - best_c5
            if delta > 0 and random.random() < np.exp(-delta / temperature):
                h = best_candidate
                no_improvement_count = 0
                tabu_list.append(h.copy())
                if len(tabu_list) > tabu_tenure:
                    tabu_list.pop(0)
            else:
                no_improvement_count += 1
                
        # Update temperature
        temperature *= cooling_rate
        
        iteration += 1
        
        # Early termination condition
        if best_c5 <= 0.3807:
            break
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
