import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.3, noise_magnitude=0.05):
    """Mutate an individual with controlled noise."""
    if random.random() > mutation_rate:
        return individual.copy()
    
    mutated = individual.copy()
    noise = np.random.normal(0, noise_magnitude, len(mutated))
    mutated += noise
    mutated = np.clip(mutated, 0, 1)
    
    return mutated

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    selected_idx = random.randint(0, len(population)-1)
    for _ in range(tournament_size - 1):
        candidate_idx = random.randint(0, len(population)-1)
        if fitnesses[candidate_idx] < fitnesses[selected_idx]:
            selected_idx = candidate_idx
    return population[selected_idx]

def genetic_algorithm(n_points, population_size=50, generations=1000, 
                     crossover_rate=0.8, mutation_rate=0.3, time_budget=600):
    """Genetic algorithm to optimize h."""
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = normalize_h(parent, n_points / 2)
        population = [parent.copy()]
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            individual = parent + noise
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    else:
        population = []
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    # Evaluate initial population
    def compute_fitness(individual):
        return evaluate_c5(individual, dx)

    fitnesses = Parallel(n_jobs=1)(delayed(compute_fitness)(ind) for ind in population)
    
    best_individual = population[np.argmin(fitnesses)]
    best_fitness = min(fitnesses)
    
    noise_magnitude = 0.05
    
    for generation in range(generations):
        if time.time() - start_time > time_budget - 1:
            break
            
        if best_fitness <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Keep best individual
        new_population.append(best_individual.copy())
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation
            child1 = mutate(child1, mutation_rate, noise_magnitude)
            child2 = mutate(child2, mutation_rate, noise_magnitude)
            
            # Repair feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, n_points / 2)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, n_points / 2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Evaluate the new population
        fitnesses = Parallel(n_jobs=1)(delayed(compute_fitness)(ind) for ind in population)
        
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Decrease noise magnitude over time
        noise_magnitude *= 0.99
        
        if generation % 100 == 0:
            pass  # Logging could be added here if needed
    
    final_c5 = evaluate_c5(best_individual, dx)
    if final_c5 < best_fitness:
        best_fitness = final_c5
        
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(n_points, generations=1000, time_budget=budget_s)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
