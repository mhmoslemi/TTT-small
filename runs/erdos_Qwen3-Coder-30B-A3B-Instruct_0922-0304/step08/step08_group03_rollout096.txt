import numpy as np
from scipy.signal import correlate
import time
import random

def repair_h(h, n_points):
    """Repair h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        scale = (n_points / 2) / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def gradient_descent_step(h, dx, step_size=0.01):
    """Perform a gradient descent step to reduce C5."""
    # Compute gradient analytically for approximate improvement
    # For simple case, directly modify approach to minimize overlap
    # This is a basic gradient-based refinment to improve current solution
    h_new = h.copy()
    for i in range(len(h)):
        # Simple heuristic: move in direction of decreasing overlap
        # Try small changes in both directions and see which helps
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + step_size)
        h_plus = repair_h(h_plus, len(h))
        
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - step_size)
        h_minus = repair_h(h_minus, len(h))
        
        c5_plus = compute_c5(h_plus, dx)
        c5_minus = compute_c5(h_minus, dx)
        c5_current = compute_c5(h, dx)
        
        if c5_plus < c5_current and c5_plus < c5_minus:
            h_new[i] = h_plus[i]
        elif c5_minus < c5_current:
            h_new[i] = h_minus[i]
    
    return repair_h(h_new, len(h))

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Parameters for GA-inspired search
    population_size = 50
    generations = 2000
    mutation_rate = 0.1
    crossover_prob = 0.7
    no_improvement_limit = 100
    
    # Compute correlation vector once
    def compute_overlap(h_vec):
        return correlate(h_vec, 1 - h_vec, mode="full") * dx
    
    # Generate initial population
    def generate_population(size):
        population = []
        for _ in range(size):
            # Start with a variant of best solution or random
            if random.random() < 0.5:
                individual = h + np.random.normal(0, 0.05, n_points)
                individual = np.clip(individual, 0, 1)
            else:
                individual = np.random.rand(n_points)
                individual = np.clip(individual, 0, 1)
            individual = repair_h(individual, n_points)
            population.append(individual)
        return population
    
    # Selection based on C5 values
    def select_parents(population, fitnesses):
        indices = np.argsort(fitnesses)
        # Take top 50% 
        selected = [population[i] for i in indices[:len(population)//2]]
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_prob:
            return parent1.copy(), parent2.copy()
        
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return repair_h(child1, n_points), repair_h(child2, n_points)
    
    # Mutation
    def mutate(individual, rate=mutation_rate):
        individual = individual.copy()
        for i in range(n_points):
            if random.random() < rate:
                individual[i] = max(0, min(1, individual[i] + np.random.normal(0, 0.05)))
        return repair_h(individual, n_points)

        fitness = []

            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        return fitness

    population = generate_population(population_size)

            top_indices = np.argsort(fitnesses)[:10]

        parents = select_parents(population, fitnesses)

        next_generation = []

            next_generation.append(mutate(c1))
            next_generation.append(mutate(c2))
        
        # Keep best of old population to prioritize good solutions
        best_old = population[np.argmin(fitnesses)]
        if len(next_generation) < population_size:
            next_generation.append(best_old)
        elif compute_c5(best_old, dx) < fitnesses[np.argmax(fitnesses)]:
            next_generation[-1] = best_old
            
        population = next_generation[:population_size]
        
        iter_count += 1
    
    # Final refinement
    final_fitness = compute_c5(best_h, dx)
    return (best_h.tolist(), final_fitness, n_points)
