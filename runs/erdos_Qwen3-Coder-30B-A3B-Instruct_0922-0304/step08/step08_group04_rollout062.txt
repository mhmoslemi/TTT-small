import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)
    
    # Convex relaxation approach using CVXPY
    try:
        # Create variables
        h_var = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # For each shift k, add constraint for overlap
        for k in range(2 * n_points - 1):
            # Compute overlap for shift k using linearization
            # Use an auxiliary variable for the product of two h terms
            overlap_expr = 0
            for x in range(n_points):
                if x + k - (n_points - 1) < n_points and x + k - (n_points - 1) >= 0:
                    overlap_expr += h_var[x] * (1 - h_var[x + k - (n_points - 1)])
                elif x + k - (n_points - 1) >= n_points:
                    overlap_expr += h_var[x] * 0  # h value is 0 beyond bounds
            constraints.append(overlap_expr <= t)
        
        # Objective
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # Solve using ECOS solver
        prob.solve(solver=ECOS, verbose=False)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h = np.array(h_var.value).flatten()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    except Exception as e:
        # If convex optimization fails, fallback to simple gradient descent
        pass
    
    # Gradient descent refinement
    max_iter = 5000
    lr = 0.01
    decay = 0.999
    tolerance = 1e-8
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute loss and gradient
        if i % 10 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            # Check termination conditions
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 500:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            loss_plus = evaluate_c5(h_plus, dx)
            loss_minus = evaluate_c5(h_minus, dx)
            grad[j] = (loss_plus - loss_minus) / (2 * eps)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        # Update parameters
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
