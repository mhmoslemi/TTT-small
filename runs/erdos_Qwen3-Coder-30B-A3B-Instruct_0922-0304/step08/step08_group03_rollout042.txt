import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a symmetric triangular ramp
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = 1 - (i - mid) / (n_points - mid)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use a gradient-free optimization: simulated annealing
    # We'll not use scipy.optimize directly but implement a basic approach
    
    # Use coordinate-wise gradient estimation
    # We'll try to refine the solution
    max_iter = 20000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    step_size = 0.01
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        old_c5 = compute_c5(h, dx)
        
        # Sample random indices to modify
        indices = np.random.choice(n_points, size=min(5, n_points), replace=False)
        
        # Use finite differences to estimate the gradient
        for i in indices:
            # Try a small positive change
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + step_size)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = compute_c5(h_plus, dx)
            
            # Try a small negative change
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - step_size)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            c5_minus = compute_c5(h_minus, dx)
            
            # Estimate gradient
            grad = (c5_plus - c5_minus) / (2 * step_size)
            
            # Update if improving
            if abs(grad) > 1e-10:
                # Update in direction of negative gradient
                new_val = h[i] - 0.1 * grad
                h[i] = np.clip(new_val, 0, 1)
                h = normalize_h(h, target_sum)
                h = np.clip(h, 0, 1)
        
        # Sometimes we accept less improvements with probability to escape local minima  
        new_c5 = compute_c5(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        if new_c5 >= old_c5:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Accept a random perturbation with some probability
            if np.random.rand() < 0.1:
                rand_idx = np.random.choice(n_points)
                h[rand_idx] = np.clip(h[rand_idx] + (np.random.rand() - 0.5) * 0.01, 0, 1)
                h = normalize_h(h, target_sum)
                h = np.clip(h, 0, 1)
            else:
                # Reduce step size
                step_size *= 0.99
                if step_size < 1e-6:
                    break
                
    # Final refinement check using a local search if we're still near the target
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        old_c5 = compute_c5(best_h, dx)
        indices = np.random.choice(n_points, size=min(3, n_points), replace=False)
        for idx in indices:
            # Try small changes
            for delta in [-0.005, 0.005]:
                h_test = best_h.copy()
                h_test[idx] = np.clip(best_h[idx] + delta, 0, 1)
                h_test = normalize_h(h_test, target_sum)
                c5_test = compute_c5(h_test, dx)
                if c5_test < old_c5:
                    best_h = h_test.copy()
                    break
            else:
                continue
            break
        else:
            # No improvement found, stop
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
