import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a smooth triangular ramp
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x <= 1:
                h[i] = x
            else:
                h[i] = 2 - x
        h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Hyperparameters
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    update_frequency = max(1, n_points // 10)  # Update top 10% of coordinates
    
    # Start optimization
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Find the coordinate that contributes most to the maximum overlap
        # Approximate gradients by perturbing each point
        gradients = np.zeros(n_points)
        
        # Compute approximate gradient for each coordinate
        for i in range(n_points):
            # Perturb h[i] slightly
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            h_minus = normalize_h(h_minus, target_sum)
            h_minus = np.clip(h_minus, 0, 1)
            
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            max_minus = np.max(overlap_minus)
            
            gradients[i] = (max_plus - max_minus) / 0.02  # Approximate gradient
        
        # Identify top update_frequency coordinates to modify
        update_indices = np.argsort(np.abs(gradients))[-update_frequency:][::-1]
        
        # Try to improve by modifying these coordinates
        improved = False
        for i in update_indices:
            current_val = h[i]
            h_new = h.copy()
            
            # Convert gradient to update direction
            step_size = 0.005  # Adaptive step size
            direction = 1 if gradients[i] < 0 else -1  # Move in direction to reduce overlap
            # Adjust step size based on gradient magnitude
            if abs(gradients[i]) > 0.1:
                step_size = 0.02
            elif abs(gradients[i]) > 0.01:
                step_size = 0.01
            
            # Apply step
            h_new[i] = np.clip(current_val + direction * step_size * (1 - 2 * current_val), 0, 1)
            h_new = normalize_h(h_new, target_sum)
            h_new = np.clip(h_new, 0, 1)
            
            # Check if this improves the maximum overlap
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_max:
                h = h_new.copy()
                improved = True
                break
        
        if not improved:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
        else:
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final refinement check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
