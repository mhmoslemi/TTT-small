import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient w.r.t. the shift that maximizes correlation"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Initialize tracking variables
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Training parameters
    step_size = 0.1
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlap and find max shift
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        # Convert full correlation index to shift
        shift = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        gradient = compute_gradient(h_current, shift, n_points)
        
        # Update using projected gradient descent
        h_new = h_current - step_size * gradient
        
        # Project to feasible region
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current solution (even if not better for stability)
        h_current = h_new.copy()
        
        # Adapt step size
        if iterations > 0 and iterations % 100 == 0:
            if no_improvement_count > 50:
                step_size *= 0.5
                no_improvement_count = 0
                
        # Perturb occasionally to escape local minima
        if no_improvement_count > 100 and np.random.random() < 0.1:
            noise = np.random.normal(0, 0.02, n_points)
            h_current = h_current + noise
            h_current = project_h(h_current, n_points)
            no_improvement_count = 0
            
        iterations += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    # Final refinement 
    h_refined = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Do a simple local search
        h_temp = h_refined.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = project_h(h_temp, n_points)
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
