import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize_scalar
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip and normalize."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def frank_wolfe_step(h, dx, n_points):
    """Compute Frank-Wolfe step for current h."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Compute gradient: for the max overlap k, we compute 
    # gradient_i = (1 - 2 * h_i) * sum_j (1 - h_{(i+k) mod n})
    # This is an approximation to the gradient of overlap_{max_idx}
    grad = np.zeros_like(h)
    
    # Compute influence of each element on the max overlap
    a = max_idx - (n_points - 1)  # This gives the corresponding k index
    
    # Simple approach: for each element in h, calculate gradient based on influence
    # Update gradient approximation: depends on the exact overlap structure
    # Compute the gradient as if we take partial derivative of the max overlap
    # w.r.t. h[i], considering the value of 1 - h[j] at relevant positions
    k = max_idx - (n_points - 1)
    # Gather indices that contribute to the maximum overlap
    total_grad = np.zeros_like(h)
    for i in range(n_points):
        j = (i + k) % n_points
        # Use this term if it's active in computing the max overlap
        total_grad[i] = 1 - 2 * h[i]

    # The Frank-Wolfe subproblem is:
    # min_v <grad, v> s.t. \sum v = n_points/2, 0 <= v <= 1
    # Solution: set v[i] = min(1, (n_points/2 - sum_{j | v[j]=1} 1) * (1 - grad[i]) / sum_{j | v[j] < 1} (1 - grad[j])  
    # This is a simplified approximation based on gradient sign:
    # If grad[i] < 0, we want larger v[i] => set to 1
    # If grad[i] > 0, we want smaller v[i] => set to 0 or some large V_i
    # Let's just use CVXPY (more robust)
    
    # Solving the linear subproblem:
    v = cp.Variable(n_points)
    obj = cp.Minimize(grad.T @ v)
    constr = [cp.sum(v) == n_points / 2,
              v >= 0, v <= 1]
    prob = cp.Problem(obj, constr)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        v_sol = v.value
        if v_sol is None:
            # fallback to a naive step
            v_sol = np.zeros_like(h)
            # Set to 1 where grad[i] is negative (in tough case, try to decrease it)
            mask = grad < 0
            v_sol[mask] = 1.0
            # normalize to sum to n_points/2
            current_sum = v_sol.sum()
            if current_sum > 0:
                v_sol = v_sol * (n_points / 2) / current_sum
    except:
        # Fallback strategy
        v_sol = np.zeros_like(h)
        # Set values based on sign of gradient
        v_sol[grad < 0] = 1.0
        # normalize
        current_sum = v_sol.sum()
        if current_sum > 0:
            v_sol = v_sol * (n_points / 2) / current_sum
    
    return v_sol

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points) * 0.5  # Uniform initialization

    # Frank-Wolfe algorithm
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute Frank-Wolfe step
        v = frank_wolfe_step(h, dx, n_points)
        
        # Line search on eta to find best step size
        def line_search_obj(eta):
            new_h = (1 - eta) * h + eta * v
            new_h = project_h(new_h, target_sum)
            return evaluate_c5(new_h, dx)
        
        # Use scalar optimization to find good eta
        try:
            res = minimize_scalar(line_search_obj, bounds=(0, 1), method='bounded')
            eta = res.x if res.success else 0.5
        except:
            eta = 0.5
            
        # Update h
        h = (1 - eta) * h + eta * v
        h = project_h(h, target_sum)
        
        # Update best solution if improved
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
