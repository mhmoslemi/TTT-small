import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def optimize_with_convex_relaxation_and_gradient(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999, tolerance=1e-8):
    """Optimize h using convex relaxation followed by gradient descent."""
    # Convex relaxation with CVXPY first
    try:
        h_cvx = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [cp.sum(h_cvx) == n_points / 2]
        
        # Add constraints for each shift
        for k in range(2 * n_points - 1):
            # Compute valid indices for the shift k
            shift_indices = []
            for i in range(n_points):
                j = (i + k - (n_points - 1)) % n_points
                if 0 <= j < n_points:
                    shift_indices.append((i, j))
            
            # Sum over all valid pairs
            if len(shift_indices) > 0:
                constraint_expr = 0
                for i, j in shift_indices:
                    constraint_expr += h_cvx[i] * (1 - h_cvx[j])
                constraints.append(constraint_expr <= t)
        
        # Bound constraints
        constraints.extend([h_cvx >= 0, h_cvx <= 1])
        
        # Solve convex relaxation
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h = np.array(h_cvx.value).reshape(-1)
    except:
        pass
    
    # Gradient descent refinement
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < tolerance:
                no_improvement_count += 1
                if no_improvement_count > 1000:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss
        
        # Compute gradient for overlap maximization 
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Estimate gradient based on overlap
        grad = np.zeros_like(h)
        for k in range(2 * n_points - 1):
            # index for zero padding in full convolution
            offset = k - (n_points - 1)
            # indices in h
            a0 = np.arange(n_points)
            b0 = a0 + offset
            # mask to valid positions 
            valid = (b0 >= 0) & (b0 < n_points)
            a_valid = a0[valid]
            b_valid = b0[valid]
            # gradient contribution from this shift
            grad[a_valid] -= overlap[k] * (1 - 2 * h[b_valid])
        
        # Update with Adam-like method
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        
        if i % 1000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, n_points / 2)
    
    # Optimize the function
    best_h, best_c5 = optimize_with_convex_relaxation_and_gradient(h, dx, n_points, max_iter=10000, lr=0.01, decay=0.999)
    
    # Make sure we have the exact best solution within the recorded time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        # Do one final check with exact computation
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
