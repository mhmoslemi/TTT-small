import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Repair h to ensure it's in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return deepcopy(parent1), deepcopy(parent2)
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05, mutation_strength=0.1):
    """Mutate h with given probability and strength."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            delta = np.random.normal(0, mutation_strength)
            h[i] += delta
    return h

def local_refinement(h, dx, steps=50):
    """Perform gradient-like local refinement."""
    h_refined = h.copy()
    target_sum = len(h) / 2
    
    for i in range(steps):
        # Find where maximum overlap occurs
        overlap = correlate(h_refined, 1 - h_refined, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Get the indices that contribute to this overlap
        # This is heuristic: target the region around max_overlap_idx
        n = len(h_refined)
        start = max(0, max_overlap_idx - n//4)
        end = min(n, max_overlap_idx + n//4)
        
        # Smoothing approach in the vulnerable region
        h_temp = h_refined.copy()
        for j in range(start, end):
            if h_temp[j] > 0.01 and h_temp[j] < 0.99:  # avoid edges
                delta = np.random.normal(0, 0.01)
                h_temp[j] = np.clip(h_temp[j] + delta, 0, 1)
        
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        
        new_c5 = compute_c5(h_temp, dx)
        if new_c5 < compute_c5(h_refined, dx):
            h_refined = h_temp
    
    return h_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    n_pop = 20
    
    population = []
    
    # Add the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        population.append(h)
    
    # Fill the rest of the population with random valid solutions
    while len(population) < n_pop:
        # Generate a random valid solution
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    gen = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Elitism: keep top 20%
        elite_count = max(1, n_pop // 5)
        new_population = population[:elite_count]
        
        # Generate offspring
        while len(new_population) < n_pop:
            # Select parents
            parent1_idx = np.random.randint(0, n_pop // 2)
            parent2_idx = np.random.randint(0, n_pop // 2)
            
            child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            
            # Mutate
            child1 = mutate(child1, 0.05, 0.1 * np.exp(-gen/20))
            child2 = mutate(child2, 0.05, 0.1 * np.exp(-gen/20))
            
            # Repair
            child1 = repair_h(child1, target_sum)
            child2 = repair_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < n_pop:
                new_population.append(child2)
        
        population = new_population
        
        # Local refinement on top solutions
        for i in range(min(3, len(population))):
            refined_h = local_refinement(population[i], dx, 50)
            population[i] = refined_h
            
        gen += 1
        no_improvement_count += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    if best_h is not None:
        final_h = local_refinement(best_h, dx, 100)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    if best_h is None:
        # Default fallback
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (target_sum / np.sum(best_h))
        best_c5 = compute_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
