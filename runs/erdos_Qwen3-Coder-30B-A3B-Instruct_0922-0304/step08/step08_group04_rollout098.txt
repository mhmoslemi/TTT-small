import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum, multiply, norm
from cvxpy import Optional, Solver

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of C5 w.r.t. h."""
    overlap = correlate(h, 1 - h, mode="full")
    k_max = np.argmax(overlap)
    
    # For the maximum overlap, compute contribution to gradient
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Equivalent to computing partial derivative using chain rule
        # Gradient w.r.t. h[i] from (1-h[i+k]) * h[i] 
        # We only consider the part from the maximum overlap lag
        j = (i + k_max) % n_points
        grad[i] = 1 - 2 * h[i]  # This is a simplified approximation
        
    return grad

def frank_wolfe_step(h, grad, n_points, target_sum):
    """Solve linear subproblem to find optimal step direction."""
    # Minimize grad^T v subject to sum(v) = target_sum and 0 <= v <= 1
    # This is a classic LP: minimize c^T x subject to Ax = b, l <= x <= u  
    # With c = grad, A = ones(1, n), b = target_sum, l=0, u=1
    
    # We can solve this greedily: 
    # 1. Sort grad in descending order to identify largest components
    # 2. Set all negative gradient components to 0
    # 3. Set positive gradient components to min(1, x)
    
    # Try to set it up as a convex optimization problem
    v = Variable(n_points)
    objective = Minimize(grad.T @ v)
    
    constraints = [
        sum(v) == target_sum,
        v >= 0,
        v <= 1
    ]
    
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver='ECOS', verbose=False)
        return np.array(v.value).flatten()
    except:
        # fallback to greedy approach
        v_values = np.zeros(n_points)
        sorted_indices = np.argsort(-grad)  # descending order
        
        remaining_sum = target_sum
        for i in sorted_indices:
            if remaining_sum <= 0:
                break
            v_values[i] = min(1.0, remaining_sum)
            remaining_sum -= v_values[i]
            
        return v_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe parameters  
    max_iter = 10000
    tolerance = 1e-8
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current gradient 
        grad = compute_gradient(h, dx, n_points)
        
        # Compute Frank-Wolfe step (solve linear subproblem)
        v = frank_wolfe_step(h, grad, n_points, target_sum)
        
        # Line search to find best step size α ∈ [0,1] 
        # We try a few candidate values and keep the best
        best_alpha = 0
        best_new_c5 = best_c5
        candidate_alphas = np.linspace(0.01, 0.99, 20)
        
        for alpha in candidate_alphas:
            h_candidate = (1 - alpha) * h + alpha * v
            h_candidate = project_h(h_candidate, target_sum)
            new_c5 = evaluate_c5(h_candidate, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
                
        # Update with best found step
        h = (1 - best_alpha) * h + best_alpha * v
        h = project_h(h, target_sum)
        
        # Update best solution
        if best_new_c5 < best_c5:
            best_c5 = best_new_c5
            best_h = h.copy()
            
        # Early stopping 
        if abs(best_new_c5 - best_c5) < tolerance or best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy() 
    
    return (best_h.tolist(), best_c5, n_points)
