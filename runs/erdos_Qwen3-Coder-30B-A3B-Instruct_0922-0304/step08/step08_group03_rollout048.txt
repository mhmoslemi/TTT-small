import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def repair_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h):
    """Approximate gradient of C5 w.r.t. h."""
    # This is a simplified gradient approximation
    # For a more precise estimation, we'd compute actual derivatives
    # But this provides a reasonable direction for descent
    return 1 - 2 * h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize population
    population_size = 50
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, target_sum)
    
    # Initialize population with diverse variations
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_new = h + np.random.normal(0, 0.05, n_points)
        h_new = repair_h(h_new, target_sum)
        population.append(h_new)
    
    # Parameters
    max_gens = 2000
    mutation_rate = 0.1
    crossover_prob = 0.7
    no_improvement_count = 0
    best_c5 = float('inf')
    best_h = None
    
    for gen in range(max_gens):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Track best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection (rank-based)
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:population_size//2]]
        
        # Crossover and mutation
        new_population = [population[current_best_idx].copy()]  # Elitism
        
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob and len(parents) >= 2:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, p1, p2)
            else:
                # Mutation
                parent = np.random.choice(parents)
                child = parent.copy()
                mutate_idx = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                for i in mutate_idx:
                    child[i] += np.random.normal(0, 0.05)
            
            child = repair_h(child, target_sum)
            
            # Gradient descent refinement
            for _ in range(5):
                grad = compute_gradient(child)
                step_size = 0.01 / (1 + gen/100)
                child -= step_size * grad
                child = repair_h(child, target_sum)
                
            new_population.append(child)
            
        population = new_population[:population_size]
        
        # Adapt parameters
        if gen % 100 == 0:
            no_improvement_count += 1
            if no_improvement_count > 10:
                mutation_rate = min(0.5, mutation_rate * 1.1)
                no_improvement_count = 0
                
    # Final evaluation of best solution
    if best_h is None:
        # Return initial if no improvement
        best_h = h
        best_c5 = evaluate_h(best_h, dx)
    else:
        best_c5 = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
