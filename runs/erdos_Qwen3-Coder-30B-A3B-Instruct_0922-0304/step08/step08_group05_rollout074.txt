import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5_exact(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a structured pattern
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid//2:3*mid//2] = 1.0
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-6
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Simulated Annealing iteration
        candidate_h = h.copy()
        
        # Random perturbation: 10% of elements
        n_perturb = max(1, n_points // 10)
        perturb_indices = random.sample(range(n_points), n_perturb)
        
        for i in perturb_indices:
            if random.random() < 0.5:
                # Flip (multiply by small factor)
                candidate_h[i] *= 0.97
            else:
                # Swap with another element
                j = random.choice(range(n_points))
                candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = normalize_h(candidate_h, target_sum)
        
        # Compute new C5
        current_c5 = compute_c5_exact(h, dx)
        candidate_c5 = compute_c5_exact(candidate_h, dx)
        
        # Accept decision
        delta = candidate_c5 - current_c5
        if candidate_c5 < current_c5 or (temperature > min_temperature and 
                                        random.random() < np.exp(-delta / temperature)):
            h = candidate_h
            current_c5 = candidate_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive cooling
        temperature *= cooling_rate
        if temperature < min_temperature:
            temperature = min_temperature
            
        iteration += 1
        
        # Cutting-plane phase (occasionally)
        if iteration % 100 == 0:
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            
            # Adjust for gradient descent on the most problematic overlap
            if max_idx < n_points:
                # Modify h at positions that contribute to max overlap
                for i in range(n_points):
                    if 0 <= i + max_idx < n_points:
                        h[i] = max(0, h[i] - 0.01 * (1 - 2 * h[i + max_idx]))
            
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            
            # Update current C5
            current_c5 = compute_c5_exact(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    # Final constraint check and normaliztion
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
