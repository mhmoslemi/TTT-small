import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Projected gradient descent with Nesterov momentum
    # Hyperparameters
    step_size = 0.1
    momentum = 0.9
    min_step_size = 1e-5
    tol = 1e-6
    max_iter = 100000
    
    # Initialize
    x_k = h.copy()
    y_k = h.copy()
    v_k = np.zeros(n_points)
    
    prev_c5 = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient approximation
        # The gradient of C_k w.r.t. h_i is (1 - h_{i-k})
        # For max over all k, we need to find the corresponding k
        # We will compute overlaps for all lags and get max
        overlap = correlate(x_k, 1 - x_k, mode="full")
        max_k_idx = np.argmax(overlap)
        max_k = max_k_idx - (n_points - 1)
        
        # Construct gradient vector for max overlap
        grad = np.zeros(n_points)
        # Compute k = max_k, negative because we're shifting x_k to the left
        # i + max_k = j => j = i + max_k 
        if max_k <= 0:
            # For index i, we look backward
            for i in range(n_points):
                j = (i + max_k) % n_points
                grad[i] = 1 - x_k[j]
        else:
            # For index i, we look forward
            for i in range(n_points):
                j = (i - max_k) % n_points
                grad[i] = 1 - x_k[j]
        
        # Add gradient penalty for sum constraint: Lagrange multiplier lambda
        # Constraint is sum(h) = target_sum, so gradient is [1, 1, ...]
        grad += (target_sum - np.sum(x_k)) / target_sum
        
        # Nesterov momentum update
        v_k = momentum * v_k + grad
        y_k = x_k - step_size * v_k
        
        # Projection to [0, 1]
        y_k = np.clip(y_k, 0, 1)
        
        # Rescale to satisfy sum constraint
        y_sum = np.sum(y_k)
        if y_sum > 0:
            y_k = y_k * (target_sum / y_sum)
            
        # Update x_k
        x_k = y_k.copy()
        
        # Evaluate
        current_c5 = evaluate_overlap(x_k, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x_k.copy()
        
        # Adaptive step size adjustment
        if abs(prev_c5 - current_c5) < tol:
            step_size *= 0.9
            if step_size < min_step_size:
                break
        else:
            step_size = min(step_size * 1.05, 1.0)
            
        prev_c5 = current_c5
    
    # Local refinement with bounds constrained optimization
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    # Simple lbfgs-like local refinement for final polish
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        old_c5 = evaluate_overlap(best_h, dx)
        
        # Simple gradient approximation
        overlap = correlate(best_h, 1 - best_h, mode="full")
        max_k_idx = np.argmax(overlap)
        max_k = max_k_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        if max_k <= 0:
            for i in range(n_points):
                j = (i + max_k) % n_points
                grad[i] = 1 - best_h[j]
        else:
            for i in range(n_points):
                j = (i - max_k) % n_points
                grad[i] = 1 - best_h[j]
        
        # Add constraint gradient
        grad += (target_sum - np.sum(best_h)) / target_sum
        
        step_size = 0.05
        trial = best_h - step_size * grad
        trial = np.clip(trial, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > 0:
            trial = trial * (target_sum / trial_sum)
        
        new_c5 = evaluate_overlap(trial, dx)
        if new_c5 < old_c5:
            best_h = trial
            best_c5 = new_c5
        else:
            break
    
    # Ensure final normalization
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
