import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    """Create a triangular ramp function."""
    mid = n // 2
    left = np.linspace(0, 1, mid)
    right = np.linspace(1, 0, n - mid)
    return np.append(left, right)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Validate and correct
        if np.sum(h) != target_sum or np.any(h < 0) or np.any(h > 1):
            h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
        h = clip_and_renormalize(h, target_sum)
    
    # Optimization parameters
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iter = 20000
    
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        # Update best if improved
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient (based on approximate sensitivity)
        # The gradient of C5 w.r.t. h is approximately sign(overlap_max) * (1 - 2h) 
        # for the dominant overlap term
        # This is not exact but a good practical approximation
        h_gradient = (1 - 2 * h) * np.sign(overlap_max) / n_points
        velocity = momentum * velocity + learning_rate * h_gradient
        
        # Try update
        h_candidate = h + velocity
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Evaluate new C5
        new_overlap = correlate(h_candidate, 1 - h_candidate, mode="full") * dx
        new_c5 = np.max(new_overlap)
        
        # Accept if better
        if new_c5 < overlap_max:
            h = h_candidate
        else:
            # Shrink step and retry
            velocity *= shrink_factor
            h_candidate = h + velocity
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            new_overlap = correlate(h_candidate, 1 - h_candidate, mode="full") * dx
            new_c5 = np.max(new_overlap)
            if new_c5 < overlap_max:
                h = h_candidate
                
        iter_count += 1
    
    # Final validation
    final_overlap = correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(final_overlap)
    
    return (best_h.tolist(), final_c5, n_points)
