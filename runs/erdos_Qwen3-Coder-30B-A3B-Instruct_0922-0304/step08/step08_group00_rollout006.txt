import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap * dx)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Fallback: create a step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Define objective function for optimization
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Define constraints
    def sum_constraint(h_vector):
        return np.sum(h_vector) - target_sum
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Try optimization with SLSQP
    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    
    # First try with initial guess
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 500, 'ftol': 1e-10})
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
    
    # If no improvement, apply local perturbations
    if time.time() - start_time < budget_s - 10:
        for _ in range(50):
            if time.time() - start_time > budget_s - 10:
                break
            # Small random perturbations
            perturbed_h = best_h + np.random.normal(0, 1e-4, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            current_sum = np.sum(perturbed_h)
            if current_sum > 0:
                perturbed_h = perturbed_h * (target_sum / current_sum)
            c5_new = evaluate_overlap(perturbed_h, dx)
            if c5_new < best_c5:
                best_h = perturbed_h
                best_c5 = c5_new
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
