import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-10 or np.any(h < 0) or np.any(h > 1):
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Optimizer parameters
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iter = 20000
    
    velocity = np.zeros(n_points)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        
        if overlap_max < best_c5:
            best_c5 = overlap_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Compute exact gradient
        # This is a simplification that finds the gradient as the direction
        # that reduces overlap most at the maximum overlap position
        max_pos = np.argmax(overlap)
        max_val = overlap[max_pos]
        sign_val = 1.0 if max_val > 0 else -1.0
        
        # Approximate gradient computation
        # We'll take a directional derivative in the direction that reduces overlap
        gradient = (1 - 2 * h) * sign_val / n_points
        
        # Apply momentum
        velocity = momentum * velocity + learning_rate * gradient
        
        # Try update
        h_candidate = h + velocity
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Test new solution
        new_overlap = correlate(h_candidate, 1 - h_candidate, mode="full") * dx
        new_c5 = np.max(new_overlap)
        
        if new_c5 < overlap_max:
            h = h_candidate.copy()
        else:
            # Revert to original, reduce step size with momentum
            velocity *= shrink_factor
            h_candidate = h + velocity
            h_candidate = clip_and_renormalize(h_candidate, target_sum)
            
            new_overlap = correlate(h_candidate, 1 - h_candidate, mode="full") * dx
            new_c5 = np.max(new_overlap)
            
            if new_c5 < overlap_max:
                h = h_candidate.copy()
        
        iter_count += 1
    
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
