import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    else:

        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use a simulated annealing approach with adaptive steps
    max_iter = 10000
    iter_count = 0
    temp = 1.0
    min_temp = 1e-6
    cooling_rate = 0.9995
    no_improvement_count = 0
    max_no_improvement = 100
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Create a neighbor solution by perturbing a random subset
        h_new = h.copy()
        
        # Select random points to perturb
        num_perturb = max(1, n_points // 20)
        idxs = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in idxs:
            # Perturb the value
            delta = np.random.normal(0, 0.01 * temp)
            h_new[i] = np.clip(h[i] + delta, 0, 1)
        
        # Renormalize
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):

            if new_c5 < best_c5:
                best_c5 = new_c5

                no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Cool down temperature
        temp = max(min_temp, temp * cooling_rate)

            # Reset with a new random solution
            if 'initial_h_values' in globals():

            else:

                h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            no_improvement_count = 0
            
        iter_count += 1

    for _ in range(100):
        if time.time() - start_time >= budget_s - 1:

            h_plus[i] = min(1, h[i] + eps)

        step_size = 0.01

        h = np.clip(h, 0, 1)
