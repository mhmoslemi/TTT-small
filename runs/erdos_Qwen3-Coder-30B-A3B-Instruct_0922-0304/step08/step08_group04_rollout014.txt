import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def solve_convex_relaxation(h0, n_points, dx):
    """Solve convex relaxation to get a good starting point."""
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # Add overlap constraints for all shifts
    for k in range(2 * n_points - 1):
        # Compute how many points contribute to this shift
        total = 0
        for x in range(n_points):
            if 0 <= x + k - (n_points - 1) < n_points:
                total += 1
        
        # We sample some points to approximate
        if total > 0:
            # For efficiency, sample a few representative terms
            terms = []
            for x in range(n_points):
                if 0 <= x + k - (n_points - 1) < n_points:
                    terms.append(h[x] * (1 - h[x + k - (n_points - 1)]))

            if terms:
                # Approximate this constraint
                sum_terms = sum(terms)
                constraints.append(sum_terms <= t)
    
    # Objective
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return np.array(h.value).reshape(-1)
    except:
        pass
    return h0  # fallback to original

def compute_gradient(h, dx, n_points):
    """Compute gradient via finite differences or direct calculation."""
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Compute partial derivative w.r.t. h[i] 
        # This is derived from the correlation expression
        contribution = 0.0
        for k in range(2 * n_points - 1):
            # Relative position in the correlation
            pos1 = i
            pos2 = i + k - (n_points - 1)
            if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
                # h[i] appears in the term h[pos1] * (1 - h[pos2])
                if pos1 == i:
                    contribution += (1 - 2 * h[pos2])
                elif pos2 == i:
                    contribution -= h[pos1]
        grad[i] = contribution * dx
    
    return grad

def optimize_with_gradient_descent(h, dx, n_points, max_iter=5000, lr=0.01, decay=0.999):
    """Refine solution with gradient descent."""
    prev_loss = float('inf')
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adam-like parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iter):
        if i % 100 == 0:
            loss = evaluate_c5(h, dx)
            if loss < best_c5:
                best_c5 = loss
                best_h = h.copy()
            
            if loss <= 0.3807:
                break
                
            if abs(prev_loss - loss) < 1e-8:
                no_improvement_count += 1
                if no_improvement_count > 100:
                    break
            else:
                no_improvement_count = 0
            prev_loss = loss

        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = h - update
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # maintain integral = 1
        
        if i % 2000 == 0 and i > 0:
            lr *= decay
    
    final_c5 = evaluate_c5(best_h, dx)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # normalize
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = h * (n_points / 2) / np.sum(h)  # normalize

    # Try convex relaxation for better initial solution
    h_initial = solve_convex_relaxation(h, n_points, dx)
    
    # Refine with gradient descent
    best_h, best_c5 = optimize_with_gradient_descent(h_initial, dx, n_points, max_iter=5000)
    
    # Validate one final time
    actual_c5 = evaluate_c5(best_h, dx)
    if actual_c5 < best_c5:
        best_c5 = actual_c5
    
    return (best_h.tolist(), best_c5, n_points)
