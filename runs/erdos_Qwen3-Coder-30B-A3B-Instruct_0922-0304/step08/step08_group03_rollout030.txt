import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    """Clip h to [0,1] and renormalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def triangular_ramp(n):
    """Create a triangular ramp initialization."""
    x = np.linspace(0, 2, n)
    h = np.minimum(x, 2 - x)
    return h / np.sum(h) * (n / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_renormalize(h, target_sum)
    else:
        h = triangular_ramp(n_points)
    
    # Adaptive optimization parameters
    learning_rate = 0.001
    momentum = 0.9
    decay = 0.99
    coord_sampling_rate = 0.2
    coord_update_count = max(1, int(n_points * coord_sampling_rate))
    
    # Track momentum and best solution
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Check if we already have a good enough solution
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Phase 1: Gradient descent with momentum
    max_iter_phase1 = 10000
    iter_count = 0
    
    while iter_count < max_iter_phase1 and time.time() - start_time < budget_s - 1:
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute gradient
        # This is a simplification - gradients are computed with respect to each position
        # to find which variables give us the most decrease in overlap
        if max_pos < n_points:
            shifted_h = np.concatenate([h[max_pos:], h[:max_pos]])
        else:
            shifted_h = np.concatenate([h[max_pos - n_points:], h[:max_pos - n_points]])
        
        # Use approximate per-coordinate gradient
        grad = (1 - 2 * h) * (1 - shifted_h) / n_points
        
        # Momentum update
        velocity = momentum * velocity + (1 - momentum) * grad
        h_candidate = h - learning_rate * velocity
        
        # Project feasible
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Compute new overlap
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < overlap_max:
            h = h_candidate.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            # Adapt learning rate
            learning_rate *= decay
        else:
            # Revert and shrink step size
            # Reset momentum for reset compromise
            velocity = np.zeros(n_points)
            learning_rate *= 0.95
        
        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Phase 2: Coordinate-wise adaptive updates
    if time.time() - start_time < budget_s - 1:
        max_iter_phase2 = 10000
        iter_count = 0
        
        while iter_count < max_iter_phase2 and time.time() - start_time < budget_s - 1:
            # Sample coordinates to update
            update_indices = np.random.choice(n_points, coord_update_count, replace=False)
            
            improved = False
            for i in update_indices:
                if time.time() - start_time > budget_s - 1:
                    break
                    
                current_val = h[i]
                # Evaluate impact by pairwise shifts
                best_val = current_val
                best_c5_eval = compute_c5(h, dx)
                
                # Try small positive and negative steps
                for step in [0.01, -0.01, 0.005, -0.005]:
                    h_trial = h.copy()
                    h_trial[i] = np.clip(current_val + step, 0, 1)
                    h_trial = clip_and_renormalize(h_trial, target_sum)
                    c5_trial = compute_c5(h_trial, dx)
                    
                    if c5_trial < best_c5_eval:
                        best_c5_eval = c5_trial
                        best_val = h_trial[i]
                
                # Apply change if beneficial
                if best_val != current_val:
                    h[i] = best_val
                    improved = True
            
            # Renormalize if needed
            h = clip_and_renormalize(h, target_sum)
            
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if not improved and best_c5 <= 0.3807:
                break
                
            iter_count += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
