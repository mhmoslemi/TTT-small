import numpy as np
from scipy.signal import correlate
import time
from joblib import Parallel, delayed
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fitness(individual, dx):
    """Fitness is the negative of C5 (we minimize C5)."""
    return -evaluate_c5(individual, dx)

def mutate(individual, noise_magnitude):
    """Mutate by adding Gaussian noise."""
    mutated = individual + np.random.normal(0, noise_magnitude, len(individual))
    mutated = np.clip(mutated, 0, 1)
    return normalize_h(mutated, len(individual) / 2)

def crossover(parent1, parent2, crossover_rate=0.8):
    """Blend crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Simple blend crossover
    alpha = random.random()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    
    return normalize_h(child1, len(parent1) / 2), normalize_h(child2, len(parent2) / 2)

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), size=tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmax(tournament_fitnesses)]
    return population[winner_index].copy()

def genetic_algorithm(n_points, dx, population_size=50, max_generations=1000, 
                      crossover_rate=0.8, mutation_rate=0.3, initial_h=None):
    """Genetic algorithm to optimize the Erdos problem."""
    
    # Initialize population
    population = []
    if initial_h is not None:
        # Use initial solution and add variation
        population.append(initial_h)
        for _ in range(population_size - 1):
            individual = mutate(initial_h, 0.1)
            population.append(individual)
    else:
        # Random initialization
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, n_points / 2)
            population.append(individual)
    
    best_individual = None
    best_fitness = float('-inf')
    noise_magnitude = 0.05
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = Parallel(n_jobs=1)(delayed(fitness)(ind, dx) for ind in population)
        
        # Track best solution
        max_fitness_idx = np.argmax(fitnesses)
        if fitnesses[max_fitness_idx] > best_fitness:
            best_fitness = fitnesses[max_fitness_idx]
            best_individual = population[max_fitness_idx].copy()
        
        # Early termination
        if -best_fitness <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: Keep best individual
        new_population.append(best_individual.copy())
        
        # Generate rest of population
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation
            if random.random() < mutation_rate:
                child1 = mutate(child1, noise_magnitude)
            if random.random() < mutation_rate:
                child2 = mutate(child2, noise_magnitude)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        
        # Decay noise magnitude
        noise_magnitude *= 0.99
    
    return best_individual, -best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    best_h, best_c5 = genetic_algorithm(n_points, dx, initial_h=h)
    
    # Ensure we have a valid solution within time budget
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
