import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to [0,1] and scale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient for given shift k*."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        idx = i + shift
        if 0 <= idx < n_points:
            grad[i] = 1 - h[idx]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    step_size = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Convert full correlation index to shift
        shift = max_overlap_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h_current, shift, n_points)
        
        # Update h using projected gradient descent
        h_new = h_current - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        # Try to improve with small random perturbations
        if c5_new >= best_c5:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Random perturbation
                h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
                h_perturbed = project_h(h_perturbed, n_points)
                c5_perturbed = compute_c5(h_perturbed, dx)
                if c5_perturbed < best_c5:
                    h_new = h_perturbed
                    c5_new = c5_perturbed
                    no_improvement_count = 0
        else:
            no_improvement_count = 0
        
        # Update best solution
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Adaptive step size
        if no_improvement_count > 0 and no_improvement_count % 10 == 0:
            step_size *= 0.9
            
        h_current = h_new.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
