import numpy as np
from scipy.signal import correlate
import time

def clip_and_renormalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def triangular_ramp(n):
    half = n // 2
    return np.append(np.linspace(0, 1, half), np.linspace(1, 0, n - half))

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if np.abs(np.sum(h) - target_sum) < 1e-10 and np.all((h >= 0) & (h <= 1)):
            pass  # Valid initialization
        else:
            h = triangular_ramp(n_points)
    else:
        h = triangular_ramp(n_points)
    
    h = clip_and_renormalize(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Gradient-based optimization with momentum
    momentum = 0.9
    learning_rate = 0.001
    shrink_factor = 0.5
    max_iterations = 20000
    velocity = np.zeros(n_points)
    
    iter_count = 0
    while iter_count < max_iterations and time.time() - start_time < budget_s - 1:
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        overlap_max = np.max(overlap)
        max_pos = np.argmax(overlap)
        
        # Compute exact gradient (simplified version for discrete signals)
        # This approximates the direction that reduces the maximum overlap
        gradient = (1 - 2 * h) / n_points
        if overlap_max > 0:
            gradient *= np.sign(overlap_max)
            
        # Apply momentum
        velocity = momentum * velocity + learning_rate * gradient
        h_candidate = h + velocity
        
        # Project to feasible region
        h_candidate = clip_and_renormalize(h_candidate, target_sum)
        
        # Check improvement
        new_c5 = compute_c5(h_candidate, dx)
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
        else:
            # Reduce step size if no improvement
            learning_rate *= shrink_factor
            if learning_rate < 1e-6:
                break
                
        iter_count += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
