import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # Parameters
    T0 = 0.1  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.05  # Initial perturbation scale
    T_min = 1e-6  # Minimum temperature
    n_iter = 0
    n_steps_no_improve = 0
    max_steps_no_improve = 1000
    temp_decay = 500
    
    T = T0
    last_best_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1:
        # Propose new state by perturbing one element
        h_proposed = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, sigma)
        h_proposed[idx] = np.clip(h_proposed[idx] + delta, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(h_proposed)
        h_proposed = h_proposed * (target_sum / current_sum)
        
        # Compute new C5
        c5_proposed = compute_c5(h_proposed, dx)
        
        # Metropolis criterion
        delta_c5 = c5_proposed - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_proposed
            if c5_proposed < best_c5:
                best_h = h_proposed.copy()
                best_c5 = c5_proposed
                last_best_c5 = best_c5
                n_steps_no_improve = 0
            else:
                n_steps_no_improve += 1
        else:
            n_steps_no_improve += 1
            
        # Adaptive cooling and perturbation
        n_iter += 1
        if n_iter % temp_decay == 0:
            T = max(T_min, T * alpha)
            
        # Adjust perturbation scale if needed
        if n_steps_no_improve > max_steps_no_improve:
            sigma *= 0.9
            n_steps_no_improve = 0
            
        if best_c5 <= 0.3807:
            break
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
