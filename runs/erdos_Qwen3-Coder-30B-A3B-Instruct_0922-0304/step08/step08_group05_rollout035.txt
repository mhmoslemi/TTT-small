import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_subgradient(h, dx, target_sum):
    """Compute subgradient of the max overlap function."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_val = np.max(overlap)
    
    # Find all indices where maximum overlap occurs
    max_indices = np.where(np.abs(overlap - max_val) < 1e-10)[0]
    
    # For each position, compute influence of overlaps at max indices
    for i in range(n):
        for k in max_indices:
            # k is the lag index, i is the position in h
            
            # The overlap at lag k involves terms h[i] * (1 - h[i+k]) 
            # where both indices are valid
            if 0 <= i < n and 0 <= i + (k - (n-1)) < n:
                # Contribution from h[i]
                grad[i] += 1 - 2 * h[i + (k - (n-1))]
                
    # Normalize gradient to unit length to prevent explosion
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
        
    # Apply constraint gradient: sum should remain constant
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Subgradient descent parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    # Early termination if already good enough
    current_c5 = compute_c5_exact(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute subgradient
        grad = compute_subgradient(h, dx, target_sum)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.995
            
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
