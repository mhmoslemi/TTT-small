def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute approximate gradient of the C5 objective."""
    # Compute overlap values
    overlap = correlate(h, 1 - h, mode="full")
    
    # Find the index of maximum overlap
    max_idx = np.argmax(overlap)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # For the maximum overlap index, compute gradient
    # This is a simplified approximation
    for i in range(n_points):
        pos1 = i
        pos2 = (i + max_idx - (n_points - 1)) % n_points
        if 0 <= pos1 < n_points and 0 <= pos2 < n_points:
            grad[i] = 1 - 2 * h[pos1]
    
    return grad

def solve_subproblem(grad, n_points):
    """Solve the linear subproblem: min_v grad^T v s.t. sum(v) = n_points/2, 0 <= v <= 1."""
    # This is a linear programming problem
    # min sum_i grad[i] * v[i]
    # s.t. sum_i v[i] = n_points / 2
    #      0 <= v[i] <= 1
    
    # Since we're linear, we can solve this manually
    # Sort and adjust
    v = np.zeros(n_points)
    sorted_indices = np.argsort(grad)
    target_sum = n_points / 2
    
    # Set values to 0 or 1 in sorted order to get near the target sum
    current_sum = 0
    for i in sorted_indices:
        if current_sum + 1 <= target_sum:
            v[i] = 1
            current_sum += 1
        elif current_sum < target_sum:
            v[i] = target_sum - current_sum
            current_sum = target_sum

            break
    
    # Adjust for exact sum
    if np.abs(current_sum - target_sum) > 1e-8:
        # Scale the values to get exact sum
        scale = target_sum / current_sum
        v *= scale
    
    return v

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe algorithm

        # Compute gradient approximated via current maximum overlap
        grad = compute_gradient(best_h, dx, n_points)
        
        # Solve linear subproblem
        v = solve_subproblem(grad, n_points)
        
        # Line search to find optimal step size
        def objective(alpha):
            h_new = (1 - alpha) * best_h + alpha * v
            h_new = np.clip(h_new, 0, 1)
            h_new = project_h(h_new, n_points / 2)
            return evaluate_c5(h_new, dx)
        
        # Binary search for step size
        alpha_min, alpha_max = 0.0, 1.0
        for _ in range(20):  # Log scale search
            alpha = (alpha_min + alpha_max) / 2

                alpha_max = alpha
            else:
                alpha_min = alpha

        h_new = project_h(h_new, n_points / 2)
        
        current_c5 = evaluate_c5(h_new, dx)

            best_h = h_new
            best_c5 = current_c5
        else:
            # Early stopping if no improvement
            if iteration > 10:
                break
            
    return best_h, best_c5
