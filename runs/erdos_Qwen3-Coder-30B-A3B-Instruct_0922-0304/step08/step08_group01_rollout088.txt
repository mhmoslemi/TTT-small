import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Sampling from the refinement approach
    max_iter = 50
    iter_count = 0
    improved_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Phase 1: Relaxation via convex optimization
        # Prepare optimization variables and constraints
        h_relaxed = cp.Variable(n_points, name="h_relaxed")
        z = cp.Variable(name="z")

        # Objective: minimize z
        objective = cp.Minimize(z)
        
        # Constraints: z >= sum of overlaps over all shifts
        constraints = []
        constraints.append(cp.sum(h_relaxed) == target_sum)
        constraints.append(h_relaxed >= 0)
        constraints.append(h_relaxed <= 1)
        
        # Compute full correlation sum constraints for all possible shifts k
        # For simplicity, we bound the maximum of overlap integrals
        # This form conservatively bounds without full correlation
        # A robust approach is to use a simplified LP relaxation version here
        # For this problem, we'll use gradient-based method for stability
        
        # Gradient ascent with adaptation
        h_current = h.copy()
        iter_inner = 0
        inner_budget = 300  # Max half a second per outer iteration
        inner_start = time.time()
        
        while time.time() - inner_start < inner_budget and iter_inner < 1000:
            c5_current = compute_c5(h_current, dx)
            
            # Approximate gradients with finite differences
            gradients = np.zeros(n_points)
            eps = 1e-3
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] = min(1, h_current[i] + eps)
                h_minus = h_current.copy()
                h_minus[i] = max(0, h_current[i] - eps)
                
                c5_plus = compute_c5(h_plus, dx)
                c5_minus = compute_c5(h_minus, dx)
                gradients[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step based on gradient
            step_size = 0.005
            for i in range(n_points):
                new_val = h_current[i] - step_size * gradients[i]
                h_current[i] = np.clip(new_val, 0, 1)
            
            # Renormalize
            h_current = normalize_h(h_current, target_sum)
            
            # Evaluate improvement
            c5_new = compute_c5(h_current, dx)
            if c5_new < c5_current:
                h = h_current.copy()

            iter_inner += 1
        
        # Evaluate best solution up to now
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improved_count = 0
        else:
            improved_count += 1
            
        # Early stopping after no improvement
        if improved_count > 10:
            break

        iter_count += 1

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
