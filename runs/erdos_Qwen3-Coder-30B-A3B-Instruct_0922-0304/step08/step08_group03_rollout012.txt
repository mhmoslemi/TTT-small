import numpy as np
from scipy.signal import correlate
import time

def triangular_ramp(n):
    """Create a triangular ramp function as initial guess."""
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

    start_time = time.time()

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Multi-start optimization
    max_trials = 10
    for trial in range(max_trials):

        # Random perturbation
        h_trial = h.copy()
        perturbation_magnitude = 0.1
        h_trial += np.random.normal(0, perturbation_magnitude, n_points)
        h_trial = project_h(h_trial, target_sum)
        
        # Gradient-like optimization with fixed learning rate
        for _ in range(5000):
            if time.time() - start_time > budget_s - 1:

            # Compute overlap
            overlap = correlate(h_trial, 1 - h_trial, mode="full") * dx
            current_max = np.max(overlap)
            
            if current_max < best_c5:
                best_c5 = current_max
                best_h = h_trial.copy()
                
            if best_c5 <= 0.3807:

            # Compute projected gradient: a simplified gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_trial.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, target_sum)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                max_plus = np.max(overlap_plus)
                
                grad[i] = (max_plus - current_max) / eps
            
            # Update using gradient
            learning_rate = 0.01
            h_new = h_trial - learning_rate * grad
            h_new = project_h(h_new, target_sum)
            
            # Accept improvement
            overlap_new = correlate(h_new, 1 - h_new, mode="full") * dx
            new_max = np.max(overlap_new)
            
            if new_max < current_max:
                h_trial = h_new.copy()
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
