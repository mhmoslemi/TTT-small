import numpy as np
import time
import math

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated annealing parameters
    T0 = 1.0  # initial temperature
    alpha = 0.99  # cooling rate
    T_min = 1e-8  # minimum temperature
    max_iter = 10000
    max_evals = 100000
    
    # initial evaluation
    current_C5 = best_C5
    current_h = h.copy()
    
    # fake "energy" as the C5 value to minimize
    energy = current_C5
    T = T0
    
    evals = 0
    iter_count = 0

    while T > T_min and evals < max_evals and time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        # make a small perturbation
        new_h = current_h.copy()
        
        # select subset of points to adjust
        n_change = max(1, int(0.05 * n_points))  # 5% of points or at least 1
        indices = np.random.choice(n_points, size=n_change, replace=False)
        
        # adjust selected points
        for i in indices:
            # perturbation: random change in [0, 0.2] or [-0.2, 0]
            delta = np.random.uniform(-0.2, 0.2)
            new_h[i] += delta
            
        # ensure bounds
        new_h = np.clip(new_h, 0, 1)
        
        # re-scale to maintain constraint sum(h) = n_points/2
        scale_factor = (n_points / 2.0) / np.sum(new_h)
        new_h *= scale_factor
        
        # compute new energy (C5)
        new_C5 = evaluate_erdos_solution(new_h, dx)
        evals += 1
        
        # acceptance probability
        delta_E = new_C5 - energy
        if delta_E < 0 or np.random.rand() < math.exp(-delta_E / T):
            current_h = new_h.copy()
            current_C5 = new_C5
            energy = new_C5
                
        # update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
            
        # decay temperature
        T *= alpha
    
    # Final check of the best solution
    final_C5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h, final_C5, n_points)
