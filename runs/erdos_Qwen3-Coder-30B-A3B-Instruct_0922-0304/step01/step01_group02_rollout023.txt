import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0  # Initial temperature
    alpha = 0.99  # Cooling rate
    min_T = 1e-8  # Minimum temperature
    max_iter = 10000
    max_evals = 100000
    
    # Energy function
    def energy(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Perturbation function to maintain constraints
    def perturb(h_vec, T):
        new_h = h_vec.copy()
        # Select subset to perturb (5-10%)
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Adjust values
        for i in indices:
            # Small random change
            delta = np.random.normal(0, T * 0.1)
            new_h[i] += delta
            
        # Ensure bounds and normalize
        new_h = np.clip(new_h, 0, 1)
        
        # Renormalize to satisfy integral constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2.0
        new_h = new_h * (target_sum / current_sum)
        
        return new_h
    
    # Simulated Annealing
    current_h = h.copy()
    current_E = energy(current_h)
    T = T0
    evals = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Generate neighbor solution
        new_h = perturb(current_h, T)
        new_E = energy(new_h)
        evals += 1
        
        # Accept or reject new solution
        if new_E < current_E or np.random.rand() < np.exp(-(new_E - current_E) / T):
            current_h = new_h
            current_E = new_E
            
            if current_E < best_C5:
                best_C5 = current_E
                best_h = current_h.copy()
        
        # Cool down
        T = max(min_T, T * alpha)
        
        if evals >= max_evals:
            break
    
    return (best_h, best_C5, n_points)
