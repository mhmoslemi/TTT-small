import numpy as np
import time
import math

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    max_iter = 50000
    min_T = 1e-8
    
    # perturbation strategies
    def perturb_h(h, T):
        h_new = h.copy()
        # select subset of points to modify
        n_modify = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        # adjust values
        for i in indices:
            # small random perturbation
            delta = np.random.normal(0, T * 0.1)
            h_new[i] += delta
            
        # project back to valid range
        h_new = np.clip(h_new, 0, 1)
        
        # maintain sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        if current_sum != 0:
            scale = target_sum / current_sum
            h_new = h_new * scale
            
        return h_new
    
    # Annealing loop
    T = T0
    current_h = h.copy()
    current_C5 = evaluate_erdos_solution(current_h, dx)
    
    iter_count = 0
    while T > min_T and time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Perturb
        h_trial = perturb_h(current_h, T)
        C5_trial = evaluate_erdos_solution(h_trial, dx)
        
        # Accept or reject
        if C5_trial < current_C5 or np.random.rand() < math.exp(-(C5_trial - current_C5) / T):
            current_h = h_trial
            current_C5 = C5_trial
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cool down
        if iter_count % 100 == 0:
            T *= alpha
    
    return (best_h, best_C5, n_points)
