import numpy as np
import time
from scipy.signal import correlate
import random

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Mutate h by adding Gaussian noise."""
    mutated = h + np.random.normal(0, noise_std, len(h))
    return mutated

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]
