import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    """Compute overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy ∫ h(x) dx = 1 and h ∈ [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_crossover(parent1, parent2, n_points):
    """Perform segment-wise crossover between two parents"""
    # Randomly select segment boundaries
    segments = np.sort(np.random.choice(n_points, size=3, replace=False))
    segments = np.concatenate([[0], segments, [n_points]])
    
    offspring = parent1.copy()
    for i in range(len(segments) - 1):
        if i % 2 == 1:  # Swap segments with odd indices
            start, end = segments[i], segments[i+1]
            offspring[start:end] = parent2[start:end]
    return offspring

def genetic_mutation(individual, mutation_rate, n_points):
    """Apply Gaussian noise mutation"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, 0.02, n_points)
        individual = individual + noise
        individual = project_to_feasible(individual, n_points)
    return individual

def solve_relaxation(h, n_points, dx):
    """Simple local optimization using gradient ascent on a convex approximation"""
    # Use gradient ascent on a smoothed version of the overlap
    h_copy = h.copy()
    step_size = 0.01
    iterations = 100
    
    for _ in range(iterations):
        # Compute overlap
        corr = correlate(h_copy, 1 - h_copy, mode="full")
        c5_val = np.max(corr) * dx
        
        # Approximate gradient via finite differences
        grad = np.zeros_like(h_copy)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_copy.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_val) / eps
        
        # Update
        h_copy -= step_size * grad
        h_copy = project_to_feasible(h_copy, n_points)
    
    return h_copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        # Generate a simple initial sequence
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        
    population.append(h.copy())
    
    # Generate diverse initial population
    for _ in range(pop_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = [evaluate_c5(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Selection: top half
        parents = population[:pop_size//2]
        parent_fitness = fitness[:pop_size//2]
        
        # Crossover and mutation to generate new population
        new_population = [population[0]]  # Keep best
        while len(new_population) < pop_size:
            if np.random.random() < crossover_rate and len(parents) >= 2:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                offspring = genetic_crossover(p1, p2, n_points)
            else:
                # Clone best parent (if no crossover)
                offspring = parents[0].copy()

            offspring = genetic_mutation(offspring, mutation_rate, n_points)
            new_population.append(offspring)
        
        # Replace population
        population = new_population
        
        # Evaluate new generation
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Update best if needed
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Periodic relaxation step
        if generation % relaxation_interval == 0:
            relaxed = solve_relaxation(best_h, n_points, dx)
            relaxed_c5 = evaluate_c5(relaxed, dx)
            if relaxed_c5 < best_c5:
                best_c5 = relaxed_c5
                best_h = relaxed.copy()
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
