import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Initialize with a sawtooth-like pattern
        h_init = np.array([1.0 - abs(np.sin(i)) for i in range(n_points)])
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to compute the constraint violations for optimization
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Perform optimization using scipy minimize
    method = 'SLSQP'
    
    # Initial evaluation
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Record the best solution found so far
    best_c5 = current_c5
    best_h = current_h.copy()
    
    # Optimization parameters
    maxiter = 500
    options = {'maxiter': maxiter, 'ftol': 1e-6, 'gtol': 1e-6}
    
    # Nonlinear optimization approach
    try:
        # Run multiple restarts
        for restart in range(3):
            # Perturb the initial point slightly
            perturbation = 0.01 * np.random.rand(n_points)
            h_start = np.clip(h_init + perturbation, 0, 1)
            h_start = h_start * target_sum / np.sum(h_start)
            
            result = minimize(
                objective,
                h_start,
                method=method,
                options=options,
                callback=lambda x: None,
            )
            
            if result.success:
                h_opt = result.x
            else:
                # Fallback to current h
                h_opt = h_start
            
            # Ensure non-negativity and bounds
            h_opt = np.abs(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
            
            # Evaluate
            c5_opt = evaluate_c5(h_opt, dx)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                
            if time.time() - start_time > budget_s * 0.95:
                break
                
    except Exception as e:
        pass

    # Final processing with best solution
    current_h = best_h.copy()
    current_c5 = best_c5

    # Ensure correctness
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    # Final validation step
    current_c5 = evaluate_c5(current_h, dx)

    return (current_h, current_c5, n_points)
