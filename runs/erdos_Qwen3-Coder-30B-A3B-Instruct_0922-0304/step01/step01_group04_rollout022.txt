import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper function to optimize
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def constraint_bounds(h):
        return np.minimum(h, 1 - h)  # Ensures h[i] ∈ [0,1]
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)},
    ]
    
    # Initial evaluation
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Perform optimization using scipy minimize with multiple methods
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    best_c5 = current_c5
    best_h = current_h.copy()
    elapsed = time.time() - start_time
    
    # Use randomized restarts and multiple methods to escape local optima
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Run a few optimizations with different starting points and methods
    for _ in range(3):  # 3 random restarts
        if elapsed >= budget_s * 0.9:  # leave some time for final processing
            break
            
        # Perturb the initial solution
        perturbed_h = current_h + np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        for method in methods:
            if elapsed >= budget_s * 0.9:
                break
                
            try:
                result = minimize(
                    objective,
                    perturbed_h,
                    method=method,
                    constraints=constraints,
                    options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-6},
                    callback=lambda x: None,
                )
                
                if result.success:
                    optimized_h = result.x
                else:
                    optimized_h = perturbed_h.copy()
                    
                # Adjust to maintain constraints
                optimized_h = np.clip(optimized_h, 0, 1)
                optimized_h = optimized_h * target_sum / np.sum(optimized_h)
                optimized_c5 = evaluate_c5(optimized_h, dx)
                
                if optimized_c5 < best_c5:
                    best_c5 = optimized_c5
                    best_h = optimized_h.copy()
                    
            except Exception as e:
                continue
                
            elapsed = time.time() - start_time
            
    # Final check and return
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Ensure correctness
    adjusted_sum = np.sum(final_h)
    if not np.isclose(adjusted_sum, target_sum, rtol=1e-4):
        final_h = final_h * target_sum / adjusted_sum
    
    # Final evaluation to ensure accuracy
    final_c5 = evaluate_c5(final_h, dx)
    
    assert np.isclose(np.sum(final_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in final_h), "Bounds constraint violated"
    
    return (final_h, final_c5, n_points)
