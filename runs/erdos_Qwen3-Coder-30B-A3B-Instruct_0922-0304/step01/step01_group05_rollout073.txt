import numpy as np
import time
from scipy.signal import correlate

def compute_C5(h, dx):
    """Compute C5 value for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select individuals via tournament selection"""
    selected = []
    for _ in range(len(population) // 2):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(individual, mutation_rate=0.1, noise_scale=0.1):
    """Mutate individual with Gaussian noise"""
    mask = np.random.rand(len(individual)) < mutation_rate
    noise = np.random.normal(0, noise_scale, len(individual))
    individual = individual + mask * noise
    return individual
