import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    T0 = 1e-3
    cooling_rate = 0.995
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    T = T0
    
    def get_max_overlap_and_lag(h):
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        return overlaps[max_k], max_k
    
    def compute_gradient(h, lag):
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            overlap_plus = get_max_overlap_and_lag(h_plus)[0]
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_minus = get_max_overlap_and_lag(h_minus)[0]
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        return grad
    
    def perturb_binary(h, ratio=0.1):
        h = h.copy()
        n = len(h)
        num_perturb = int(n * ratio)
        indices = np.random.choice(n, num_perturb, replace=False)
        h[indices] = 1 - h[indices]
        return h
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Get current max overlap and lag
        max_overlap, max_k = get_max_overlap_and_lag(h)
        
        # Compute gradient
        grad = compute_gradient(h, max_k)
        
        # Simulated Annealing update
        h_new = h - alpha * grad + T * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        scale = len(h) / 2 / np.sum(h_new)
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2 / np.sum(h_new))
        
        # Compute new overlap
        new_overlap = get_max_overlap_and_lag(h_new)[0]
        
        # Metropolis criterion
        delta = new_overlap - max_overlap
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if new_overlap < best_c5:
                best_c5 = new_overlap
                best_h = h.copy()
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
            
        # Adaptive learning rate
        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:
                alpha *= gamma
            iter_since_improvement = 0
            # Temperature decay
            T = max(T0 * (cooling_rate ** iteration), 1e-8)
            
        # Periodic binary perturbation
        if iteration % 1000 == 0 and iteration > 0:
            h = perturb_binary(h)
            T = T0
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
