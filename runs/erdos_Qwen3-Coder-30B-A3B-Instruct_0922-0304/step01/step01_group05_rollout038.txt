        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx

        # Normalize initial values
        h_init = initial_h.copy()

        # Variables for CVXPY

        # Constraint for the DC component (sum = n_points/2)
        constraints = [

            h_imag[0] == 0,
            h_real >= 0,
            h_imag >= 0,
            h_real <= 1,
            h_imag <= 1,
        ]
        
        # Objective: minimize max overlap
        max_overlap = cp.Variable()
        constraints.append(max_overlap >= 0)
        
        # Convert to proper constraints using PSD form
        # This approach: use heuristics
        return h_init

    def objective(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Use different optimizer: L-BFGS-B for better convergence properties
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    # Try multiple initializations to avoid local optima
    best_c5 = float('inf')

    for _ in range(5):
