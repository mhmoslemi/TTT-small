import numpy as np
import time
import math

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    max_iter = 10000
    max_evals = 100000
    
    # energy evaluations counter
    evals = 0
    iter_count = 0
    
    # keep track of current h and C5 values
    current_h = h.copy()
    current_C5 = best_C5
    
    # main annealing loop
    while evals < max_evals and iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
        # perturb h: select random points to adjust
        new_h = current_h.copy()
        # 5% of variables change per step
        num_changes = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, num_changes, replace=False)
        
        for idx in indices:
            # random change within bounds
            delta = np.random.normal(0, 0.1)
            new_h[idx] += delta
            
        # ensure bounds
        new_h = np.clip(new_h, 0, 1)
        
        # fix integral constraint
        h_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2.0) / h_sum
        
        # compute new C5
        new_C5 = evaluate_erdos_solution(new_h, dx)
        evals += 1
        iter_count += 1
        
        # acceptance probability
        if new_C5 < current_C5:
            current_h = new_h.copy()
            current_C5 = new_C5
        else:
            # accept with probability based on temperature
            delta_E = new_C5 - current_C5
            prob = math.exp(-delta_E / T)
            if np.random.rand() < prob:
                current_h = new_h.copy()
                current_C5 = new_C5
                
        # update best
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
        
        # cool down
        T = T * alpha
    
    return (best_h, best_C5, n_points)
