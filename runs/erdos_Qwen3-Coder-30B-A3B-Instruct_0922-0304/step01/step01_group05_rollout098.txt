import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import random

def evaluate_C5(h, dx):
    """Compute C5 for a given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def mutate_h(h, mutation_rate=0.1, noise_std=0.1):
    """Mutate h with Gaussian noise"""
    h_mut = h + np.random.normal(0, noise_std, len(h))
    return repair_h(h_mut, len(h))

def crossover_h(h1, h2):
    """Uniform crossover between two individuals"""
    mask = np.random.rand(len(h1)) > 0.5
    h_child = np.where(mask, h1, h2)
    return h_child

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]
