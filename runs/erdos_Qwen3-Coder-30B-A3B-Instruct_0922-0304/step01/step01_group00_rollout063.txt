import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Segment-wise crossover between two parents"""
    # Create offspring with segments from each parent
    offspring = np.zeros(n_points)
    segment_length = max(1, n_points // 4)  # Make at least one element
    start = np.random.randint(0, n_points - segment_length)
    end = min(start + segment_length, n_points)
    
    # Copy segments alternatingly from parents
    offspring[:start] = parent1[:start]
    offspring[start:end] = parent2[start:end]
    offspring[end:] = parent1[end:]
    
    return offspring

def mutate(individual, n_points, mutation_rate=0.2, noise_sigma=0.02):
    """Mutate with Gaussian noise"""
    if np.random.random() < mutation_rate:
        # Add Gaussian noise
        noise = np.random.normal(0, noise_sigma, n_points)
        individual = individual + noise
        individual = project_to_feasible(individual, n_points)
    return individual

def genetic_search(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 2000
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Fallback construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)

    # Add random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)

    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_c5(h, dx)
        fitness.append(c5)

    best_c5 = min(fitness)
    best_h = population[fitness.index(best_c5)].copy()

    print(f"Initial best C5: {best_c5:.6f}")

    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% as parents
        num_parents = population_size // 2
        parents = sorted_population[:num_parents]
        parent_fitness = sorted_fitness[:num_parents]
        
        # Create new population
        new_population = parents.copy()  # Elitism
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection: tournament
            parent1_idx = np.random.randint(0, len(parents))
            parent2_idx = np.random.randint(0, len(parents))
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring = crossover(parents[parent1_idx], parents[parent2_idx], n_points)
            else:
                offspring = deepcopy(parents[parent1_idx])
                
            # Mutation
            offspring = mutate(offspring, n_points, mutation_rate)
            
            new_population.append(offspring)
            
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Compute new fitness
        new_fitness = []
        for h in new_population:
            c5 = evaluate_c5(h, dx)
            new_fitness.append(c5)
            
        # Apply relaxation step occasionally
        if gen % relaxation_interval == 0:
            # Solve a convex relaxation using projective gradient descent
            best_idx = np.argmin(new_fitness)
            best_current = new_population[best_idx]
            relaxed_h = best_current.copy()
            
            # Simple gradient step: decrease overlap
            for _ in range(20):
                corr = correlate(relaxed_h, 1 - relaxed_h, mode="full")
                max_idx = np.argmax(corr)
                max_corr = corr[max_idx]
                
                # Approximate gradient via finite differences
                # For step size, proportional to max_corr
                step_size = 0.01 * max_corr
                if step_size < 1e-8:
                    break
                    
                # Directional derivative
                eps = 1e-5
                grad = np.zeros_like(relaxed_h)
                
                for i in range(n_points):
                    h_plus = relaxed_h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    grad[i] = (corr_plus[max_idx] - max_corr) / eps
                
                # Update in gradient direction to reduce max overlap
                relaxed_h += step_size * grad
                relaxed_h = project_to_feasible(relaxed_h, n_points)
                
            # Replace worst with relaxed
            worst_idx = np.argmax(new_fitness)
            new_population[worst_idx] = relaxed_h
            new_fitness[worst_idx] = evaluate_c5(relaxed_h, dx)
            
        # Update population and fitness
        population = new_population
        fitness = new_fitness
        
        # Check for improvement
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
        # Logging
        if gen % 50 == 0:
            print(f"Gen {gen}: Best C5 = {best_c5:.6f}")

        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_search(seed, budget_s)
