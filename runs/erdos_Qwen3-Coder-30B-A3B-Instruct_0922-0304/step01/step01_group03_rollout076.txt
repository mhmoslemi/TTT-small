import time
from scipy import signal

    np.random.seed(seed)
    start_time = time.time()

        np.random.shuffle(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    T0 = 1e-3
    cooling_rate = 0.995
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    temperature = T0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps using FFT for efficiency
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Use finite differences to compute gradient approximation 
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Simulated Annealing update with temperature
        h_new = h - alpha * grad + temperature * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        scale = np.sum(h_new) / (len(h) / 2.0)
        h_new = h_new / scale  # Renormalize
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update using Metropolis criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
            
        # Adaptive learning rate and temperature
        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:

            temperature *= cooling_rate

            h = h * (len(h) / 2.0) / np.sum(h)

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
