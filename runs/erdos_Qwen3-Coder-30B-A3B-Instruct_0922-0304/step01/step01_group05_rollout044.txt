import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # PSO parameters
    num_particles = 50
    max_iter = 500
    w_start = 0.9
    w_end = 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(num_particles):
        h = initial_h + 0.2 * (2 * np.random.rand(n_points) - 1)
        # Repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(0.1 * np.random.randn(n_points))
    
    # Evaluate initial particles
    personal_best = [p.copy() for p in particles]
    personal_best_values = [compute_C5(p) for p in particles]
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO loop
    iter_count = 0
    no_improvement_count = 0
    last_best = global_best_value
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_value = compute_C5(particles[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
                    no_improvement_count = 0
                    
        no_improvement_count += 1
        iter_count += 1
        
        # Perturbation if no improvement in last 20 iterations
        if no_improvement_count >= 20:
            for i in range(num_particles):
                if np.random.rand() < 0.3:  # 30% chance of perturbation
                    particles[i] += 0.05 * (2 * np.random.rand(n_points) - 1)
                    # Repair
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Re-evaluate and update local bests
            for i in range(num_particles):
                current_value = compute_C5(particles[i])
                if current_value < personal_best_values[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = current_value
                    if current_value < global_best_value:
                        global_best = particles[i].copy()
                        global_best_value = current_value
                        
            no_improvement_count = 0
    
    return (global_best.tolist(), global_best_value, n_points)
