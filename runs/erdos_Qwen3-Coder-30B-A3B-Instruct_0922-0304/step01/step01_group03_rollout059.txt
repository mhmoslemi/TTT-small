import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    n_points = 45
    if 'initial_h_values' in globals():

    else:

        h0[:23] = 1.0

    population = []
    population.append(h0.copy())
    
    # Generate diverse initial candidates
    for i in range(1, pop_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2 / np.sum(h))  # Renormalize
        population.append(h)
    
    dx = 2.0 / n_points
    
    # Evaluate initial population
    fitness = []
    for h in population:
        overlap = np.max(signal.correlate(h, 1 - h, mode='full')) * dx
        fitness.append(overlap)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution parameters
    generations = 2000
    elite_size = pop_size // 10
    crossover_rate = 0.8
    mutation_rate = 0.05
    noise_std = 0.05
    no_improvement_limit = 50
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection: keep top candidates
        selected_indices = np.argsort(fitness)[:pop_size//2]
        selected_pop = [population[i] for i in selected_indices]
        selected_fit = [fitness[i] for i in selected_indices]
        
        # Elitism: preserve best n individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_pop = [population[i].copy() for i in elite_indices]
        
        # Crossover and mutation
        offspring = []
        
        while len(offspring) < pop_size // 2:
            if np.random.rand() < crossover_rate and len(selected_pop) >= 2:
                # Select two random parents
                p1, p2 = np.random.choice(len(selected_pop), size=2, replace=False)
                parent1, parent2 = selected_pop[p1], selected_pop[p2]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for child in [child1, child2]:
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child[i] += np.random.normal(0, noise_std)
                    # Ensure constraints
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2 / np.sum(child))
                    offspring.append(child)
            else:
                # Just mutate random individual
                idx = np.random.randint(len(selected_pop))
                child = selected_pop[idx].copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, noise_std)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2 / np.sum(child))
                offspring.append(child)
        
        # Replace worst half of population with offspring
        # Keep elite candidates
        population = elite_pop + offspring
        
        # Re-evaluate
        fitness = []
        for h in population:
            overlap = np.max(signal.correlate(h, 1 - h, mode='full')) * dx
            fitness.append(overlap)
        
        # Check for improvement
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            
        # Diversification
        if gen % no_improvement_limit == 0 and gen > 0:
            for i in range(int(0.2 * pop_size)):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2 / np.sum(h))
                population[np.random.randint(len(population))] = h
            noise_std *= 0.9  # Reduce noise
    
    return (best_h.tolist(), best_c5, len(best_h))
