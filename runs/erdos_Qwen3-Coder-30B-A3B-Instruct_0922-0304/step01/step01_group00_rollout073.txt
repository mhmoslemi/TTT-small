import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        
        # Identify worst overlap index
        max_idx = np.argmax(corr)
        
        # Compute gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_val) / eps
            
        # Simple gradient descent step with projection
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5_val = evaluate_c5(h, dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Reduce step size gradually
        step_size *= 0.995

    return (best_h.tolist(), best_c5, n_points)
