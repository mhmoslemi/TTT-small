import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import warnings
warnings.filterwarnings('ignore')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        # Default construction if no initial_h_values
        n_points = 42
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure initial values are valid
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Objective function helper
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute overlap using correlation
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Optimization with different strategies
    best_C5 = float('inf')
    best_h = None
    
    # Strategy 1: Gradient-based local optimization
    def objective(h):
        return compute_C5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Try different starting points to escape local minima
    starting_points = [initial_h]
    # Add some random perturbations for exploration
    for _ in range(3):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = initial_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        starting_points.append(h_perturbed)
    
    # Run optimizations from different starting points
    for i, h_init in enumerate(starting_points):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        try:
            res = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 500, 'ftol': 1e-6})
            
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                C5 = compute_C5(h_opt)
                
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_opt.copy()
        except:
            continue
    
    # Strategy 2: A more robust approach using direct optimization with bounds
    if best_h is None:
        # If no improvement in first phase, try another variant with bounds adjustments
        try:
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            h_init = initial_h.copy()
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            
            res = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-6})
            
            if res.success:
                h_opt = res.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                C5 = compute_C5(h_opt)
                
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final correction
    if best_h is None:
        # Fallback to initial
        best_h = initial_h.copy()
        best_C5 = compute_C5(best_h)
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
