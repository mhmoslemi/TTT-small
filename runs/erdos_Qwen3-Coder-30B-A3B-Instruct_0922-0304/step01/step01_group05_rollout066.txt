import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C5 for given h values."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute all correlations for different lags
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum overlap scaled by dx
    return np.max(corr) * dx
