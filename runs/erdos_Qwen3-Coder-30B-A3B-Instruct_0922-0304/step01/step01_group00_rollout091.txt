import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, k_max):
    """Compute the gradient of L(k_max) with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_max < n:
            grad[i] += 1 - h[i + k_max]
        if i >= k_max:
            grad[i] -= h[i]
    return grad

def line_search(h, i_star, grad, k_max, dx):
    """Perform line search to minimize L(k_max) along direction of h[i_star]"""
    n = len(h)
    # Determine the change in h that maintains sum constraint
    # We want to decrease h[i_star] and possibly increase h[i_star + k_max] 
    # to maintain sum(h) = n/2
    h_new = h.copy()
    
    # Let's parameterize the change: delta = h_new[i_star] - h[i_star]
    # If we change h[i_star], we must also change h[i_star + k_max]
    # to preserve sum. Let delta = h_new[i_star] - h[i_star]
    # Then we need: h_new[i_star + k_max] = (h[i_star + k_max] - delta) 
    # constraint: sum(h_new) = sum(h) so delta = 0 for all cases
    # Actually the change of variables needs to be more sophisticated
    
    # To adjust just one index while preserving the sum
    # h_new[i_star] = h[i_star] + delta
    # h_new[i_star + k_max] = h[i_star + k_max] - delta
    # If i_star + k_max >= len(h), this should not happen, so we directly adjust
    # But since our overlap is already fixed, let's see for exact case
    
    # Instead, we just reduce h[i_star] and let k_max shift be fixed by convention
    # We want to reduce overlap by changing h[i_star], while preserving:
    # sum(h) = n/2
    
    # So we adjust both h[i_star] and h[i_star + k_max] to keep sum equal
    # A simpler way: update just one element and renormalize

    best_i_star = i_star
    best_delta = 0.0
    min_overlap = float('inf')
    
    # Vectorized direction search
    directions = np.linspace(0, 0.5, 50)  # Reduce max change to avoid going out of bounds
    
    for delta in directions:
        h_trial = h.copy()
        h_trial[i_star] = max(0, min(1, h[i_star] - delta))  # This is more precise
        if i_star + k_max < len(h):
            h_trial[i_star + k_max] = max(0, min(1, h[i_star + k_max] + delta))
        h_trial = h_trial * (len(h) / 2) / np.sum(h_trial)  # Renormalize
            
        # Calculate overlap for current k_max
        overlaps = correlate(h_trial, 1 - h_trial, mode="full")
        overlap_val = np.max(overlaps) * dx
            
        if overlap_val < min_overlap:
            min_overlap = overlap_val
            best_i_star = i_star
            best_delta = delta
            
    return best_i_star, best_delta

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iterations = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with a simple binary pattern plus noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps and maximum
        overlaps = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break

        # Compute gradient to guide coordinate selection
        # Compute gradient of maximum overlap
        grad = compute_gradient(h, max_k)
        # The next update will be in the direction of steepest decrease of max overlap
        i_star = np.argmin(grad)  # Minimizing the gradient means increasing this element to reduce overlap
        
        # We approximately update at coordinate i_star
        # Let's do a small step that's safe
        step_size = 0.01
        h_new = h.copy()
        h_new[i_star] = max(0, min(1, h_new[i_star] - step_size))
        
        # Adjust its counterpart to maintain sum
        if i_star + max_k < n_points:
            h_new[i_star + max_k] = max(0, min(1, h_new[i_star + max_k] + step_size))
        
        # Renormalize to meet sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
