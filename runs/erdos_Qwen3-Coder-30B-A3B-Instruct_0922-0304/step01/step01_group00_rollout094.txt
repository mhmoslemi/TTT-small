import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 10000
    line_search_precision = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break

        # Compute gradient of overlap w.r.t h for the max overlap k
        # For overlap at shift k: 
        # L(k) = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
        # Gradient at h[i] = 1 - h[i+k] if i+k < n and -h[i] if i < k
        grad = np.zeros(n_points)
        
        # Simplified gradient calculation for the selected shift k_max
        k_max = max_k - (n_points - 1)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max]
        
        # Find i that minimizes gradient (i.e., most negative)
        i_star = np.argmin(grad)
        
        # Line search along direction of negative gradient
        # We want to reduce overlap at shift k_max, so we move h[i_star]
        # The update is to h[i_star] = h[i_star] + alpha * step_direction
        # Need to be careful not to bring h[i_star] out of [0,1] or break sum constraint
        
        # Determine range for step
        # min_step: to keep h[i_star] in [0,1]
        # max_step: to keep other h values from going negative
        
        current = h[i_star]
        min_h = 0.0
        max_h = 1.0
        
        # To move h[i_star] in direction of steepest descent for overlap
        step_dir = -grad[i_star]
        step_size = 0.01
        step_acc = 1e-6
        step_max = 1.0

        # Perform simple line search using gradient info
        # We will decrease h[i_star] if gradient is positive (we want to minimize overlap)
        if step_dir > 0:
            step = min(max_h - current, step_size * step_dir)
            step = max(step, step_acc)
        elif step_dir < 0:
            step = min(current - min_h, -step_size * step_dir)
            step = max(step, step_acc)
        else:
            step = 0.01
            
        # Update h[i_star]
        h[i_star] = current + step
        h[i_star] = np.clip(h[i_star], 0, 1)
        
        # Adjust all other h values so that sum remains preserved
        # Adjust by the opposite amount of the change of h[i_star]:
        # if h[i_star] decreased by delta, then we need to increase others by delta/n_remaining (approximation)
        # Better: use the sum constraint to adjust the entire h vector
        old_sum = np.sum(h)
        h = h * (n_points/2) / old_sum
        
        # For stability: check if any h[i] became nan or inf
        if np.any(np.isnan(h)) or np.any(np.isinf(h)):
            h = best_h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
