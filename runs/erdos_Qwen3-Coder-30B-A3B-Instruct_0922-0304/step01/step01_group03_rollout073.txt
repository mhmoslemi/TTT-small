import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0
        np.random.shuffle(h)
        
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 5000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_frequency = 1000
    
    # Tree structure - dictionary mapping h to (c5, visits, children)
    tree = {}
    root_h = tuple(h)
    tree[root_h] = (best_c5, 1, {})
    
    def ucb_score(node_h, parent_visits):
        c5, visits, _ = tree[node_h]
        if visits == 0:
            return float('inf')
        return c5 + exploration_weight * np.sqrt(2 * np.log(parent_visits) / visits)
    
    def mutate_h(current_h):
        h_new = current_h.copy()
        n_mutate = max(1, int(len(h_new) * mutation_rate))
        indices = np.random.choice(len(h_new), n_mutate, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.05)
        h_new = np.clip(h_new, 0, 1)
        scale_factor = len(h_new) / 2 / np.sum(h_new)
        h_new = h_new * scale_factor
        return h_new
    
    def get_best_node():
        best_node = None
        best_value = float('inf')
        for node_h, (c5, visits, _) in tree.items():
            if c5 < best_value:
                best_value = c5
                best_node = node_h
        return best_node
    
    iter_count = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
        iter_count += 1
        
        # Selection: traverse tree using UCB
        current_node = root_h
        path = [current_node]
        while current_node in tree and tree[current_node][2]:  # while has children
            _, _, children = tree[current_node]
            best_child = max(children, key=lambda x: ucb_score(x, tree[current_node][1]))
            current_node = best_child
            path.append(current_node)
        
        # Expansion: generate children
        current_h = np.array(current_node)
        children = []
        for _ in range(5):
            new_h = mutate_h(current_h)
            new_h_tuple = tuple(new_h)
            children.append(new_h_tuple)
            # Add to tree
            if new_h_tuple not in tree:
                c5_new = evaluate_c5(new_h, dx)
                tree[new_h_tuple] = (c5_new, 0, {})
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = new_h.copy()
        
        # Simulation: evaluate children and update tree
        for child_tuple in children:
            c5_val, visits, _ = tree[child_tuple]
            tree[child_tuple] = (c5_val, visits+1, {})
        
        # Backpropagation: update all nodes in path
        for node in reversed(path):
            c5_val, visits, children = tree[node]
            tree[node] = (min(c5_val, min(tree[child][0] for child in children)), 
                         visits + 1, children)
        
        # Periodic reset to encourage exploration
        if iter_count % reset_frequency == 0:
            for node_h in list(tree.keys()):
                if random.random() < 0.1:  # Reset 10% of nodes
                    reset_h = np.random.rand(len(h))
                    reset_h = reset_h * (len(h) / 2 / np.sum(reset_h))
                    tree[node_h] = (evaluate_c5(reset_h, dx), 0, {})
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
