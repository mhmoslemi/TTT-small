import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T_min = 1e-8
    max_iter = 10000
    max_evals = 100000
    
    current_h = h.copy()
    current_C5 = best_C5
    T = T0
    
    eval_count = 0
    iter_count = 0
    
    # Main optimization loop
    while T > T_min and eval_count < max_evals and iter_count < max_iter:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Perturb the current solution
        new_h = current_h.copy()
        
        # Modify a random subset of points
        n_modify = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        for i in indices:
            # Modify by small random amount
            delta = np.random.normal(0, 0.05 * T)
            new_h[i] += delta
            
        # Maintain bounds and integral constraint
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        target_sum = n_points / 2.0
        if current_sum > 0:
            new_h = new_h * target_sum / current_sum
            new_h = np.clip(new_h, 0, 1)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(new_h, dx)
        eval_count += 1
        
        # Accept or reject based on Metropolis criterion
        delta_E = new_C5 - current_C5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h = new_h
            current_C5 = new_C5
            
            # Update best solution
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cool down
        T *= alpha
        iter_count += 1
    
    return (best_h, best_C5, n_points)
