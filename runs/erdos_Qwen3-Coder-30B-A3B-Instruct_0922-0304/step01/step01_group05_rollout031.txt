import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        initial_h = np.array(initial_h_values)
    except NameError:
        # Fallback: create a simple symmetric construction
        initial_h = np.ones(n_points) * 0.5
        
    # Normalize initial guess
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Define objective function using correlation
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Constraint: sum of h = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Constraint: bounds
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Objective function for scipy
    def objective(h):
        return compute_C5(h)
    
    # Constraints for scipy
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization with different starting points to escape local optima
    best_h = initial_h.copy()
    best_C5 = compute_C5(best_h)
    
    # Try to escape local minima with multiple random restarts
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation of the initial solution
        h_perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 500, 'ftol': 1e-6})
            if res.success:
                h_candidate = res.x
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                candidate_C5 = compute_C5(h_candidate)
                if candidate_C5 < best_C5:
                    best_h = h_candidate
                    best_C5 = candidate_C5
        except:
            pass
    
    # Final refinement using a more precise solver approach
    if time.time() - start_time < budget_s * 0.95:
        try:
            # Use CVXPY for more precise optimization with constraints
            h_vars = cp.Variable(n_points)
            constraints_cvx = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            # Compute the maximum overlap using a relaxed approach
            # The goal is to minimize the maximum correlation term
            c_max = cp.Variable()
            # We'll use a different approach to enforce the max constraint
            objective_cvx = cp.Minimize(c_max)
            prob = cp.Problem(objective_cvx, constraints_cvx)
            
            # Try with fewer iterations to save time
            prob.solve(solver=CPLEX, verbose=False, time_limit=10)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_final = h_vars.value
                if h_final is not None:
                    h_final = h_final * (n_points / 2) / np.sum(h_final)
                    final_C5 = compute_C5(h_final)
                    if final_C5 < best_C5:
                        best_h = h_final
                        best_C5 = final_C5
        except:
            pass
    
    # Ensure valid return values even if optimization fails
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
