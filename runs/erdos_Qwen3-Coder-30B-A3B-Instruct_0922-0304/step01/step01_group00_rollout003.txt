import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Hyperparameters
    max_iter = 1000
    step_size = 0.01
    inner_iterations = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

        # Local gradient-based optimization
        h_temp = h.copy()
        for _ in range(inner_iterations):
            # Compute overlap
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            c5_val = np.max(corr) * dx
            
            # Gradient via finite differences (simplified)
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_val) / eps
            
            # Update h
            h_temp -= step_size * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)

        h = h_temp
        
        # Periodic feasibility repair
        if iteration % 20 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final verification
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
