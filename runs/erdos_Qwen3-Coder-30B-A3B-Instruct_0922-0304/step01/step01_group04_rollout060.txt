import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):

    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Prepare CVXPY variables
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(name="t")
    
    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
        t >= 0
    ]
    
    # Add overlap constraints

    # Let's get the correlation size first: 2*n_points - 1
    corr_size = 2 * n_points - 1
    # For each shift k in range[-(n_points-1), n_points-1]
    # Store all the constraints
    for k in range(-(n_points-1), n_points):
        # Find indices: i, j such that j-i=k (i in [0, n_points-1], j in [0, n_points-1])
        # This means:
        # i in [0, n_points-1], j = i + k
        # so i in [0, n_points-1] and i+k in [0, n_points-1]
        # => i in [max(0, -k), min(n_points-1, n_points-1-k)]
        
        i_start = max(0, -k)
        i_end = min(n_points-1, n_points-1-k)
        if i_start <= i_end:
            # Sum over these pairs: h[i] * (1 - h[i+k])
            # This is equivalent to sum over those indices:  h[i] - h[i]*h[i+k]
            # But we want to bound the sum of h[i]*(1-h[i+k])
            # So the constraint becomes: sum h[i] - sum h[i]*h[i+k] <= t
            # which rearranges to: sum h[i]*h[i+k] >= sum h[i] - t
            
            # Better approach: make constraints that sum h[i]*(1-h[i+k]) <= t
            # which is: sum h[i] - sum h[i]*h[i+k] <= t
            # Rearranging: sum h[i]*h[i+k] >= sum h[i] - t
            
            # Instead of rewriting, explicitly use the convolution-style constraints
            # Let's build the indices for shift k
            i_indices = np.arange(i_start, i_end+1)
            j_indices = i_indices + k
            
            # We have h[i]*(1-h[j]) <= t, which means h[i] - h[i]*h[j] <= t
            # Let's make a linear approximation or use a more direct approach:
            # Build i,j pairs and force constraints
            
            # Direct interpretation of correlation
            # The sum for shift k is sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) = sum h[i] - sum h[i]*h[i+k]
            # So: sum h[i] - sum h[i]*h[i+k] <= t
            # => sum h[i]*h[i+k] >= sum h[i] - t
            
            # But a better way: make constraints like:
            # h[i] - h[i]*h[i+k] <= t  for all valid i
            # Which gives:
            # h[i] - t <= h[i]*h[i+k]
            # This is a non-convex constraint, so we approximate
            
            # The best we can do with convex constraints is to **
            # use relaxation for h[i]*(1-h[i+k]), 
            # Use a formulation: h[i] + h[i+k] - h[i]*h[i+k] <= 1 + (h[i] + h[i+k] - 1) 
            # Actually: 
            # h[i]*(1-h[i+k]) <= 1/2 * (h[i]^2 + (1-h[i+k])^2) - 1/2 =: upper_bound

            # Let z[i,k] = h[i] * (1 - h[i+k])

            # For all i: h[i] - h[i]*h[i+k] <= t  # which is h[i] + h[i+k] - h[i]*h[i+k] <= h[i] + h[i+k] - t
            # Use this: 
            # sum h[i] - sum h[i]*h[i+k] <= t
            # Let g[k] = sum h[i]*h[i+k] where i is according to bounds.
            # We constrain h[i]*h[i+k] >= h[i] + h[i+k] - 1  (McCormick)
            # So : g[k] = sum h[i]*h[i+k] >= sum (h[i] + h[i+k] - 1) = sum h[i] + sum h[i+k] - n
            # But a more correct approach is verification block-wise below:
            
            # A simpler cube relaxation: 
            # z[i,k] = h[i]*h[i+k], constraint z[i,k] <= min(h[i], h[i+k])
            # z[i,k] >= h[i] + h[i+k] - 1

                # Build constraint iteratively for this shift k
                left_side = cp.sum(cp.multiply(h[i_start:(i_end+1)],

                constraints.append(left_side <= t)

                i_start_alt = -k  # corrected from your original

                    left_side = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try solving

    # Ensure correctness
