import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h)
        
    # Ensure sum equals n_points/2
    h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturb_freq = 1000
    
    # Initialize simplex
    simplex = np.array([h.copy()])
    for _ in range(n_points):
        h_perturb = h + simplex_size * np.random.normal(0, 1, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (22.5 / np.sum(h_perturb))
        simplex = np.vstack([simplex, h_perturb])
    
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate all simplex points
        values = []
        for i in range(len(simplex)):
            val = evaluate_c5(simplex[i], dx)
            values.append(val)
        values = np.array(values)
        
        # Order by value
        order = np.argsort(values)
        simplex = simplex[order]
        values = values[order]
        
        # Check if improved
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > max_no_improvement:
            break
        
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        reflected = centroid + reflection_factor * (centroid - simplex[-1])
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (22.5 / np.sum(reflected))
        reflected_val = evaluate_c5(reflected, dx)
        
        # Check if reflected point is better than worst
        if values[0] <= reflected_val < values[-2]:
            simplex[-1] = reflected
        elif reflected_val < values[0]:
            # Expansion
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (22.5 / np.sum(expanded))
            expanded_val = evaluate_c5(expanded, dx)
            
            if expanded_val < reflected_val:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            if reflected_val < values[-1]:
                # Outside contraction
                contracted = centroid + contraction_factor * (reflected - centroid)
            else:
                # Inside contraction
                contracted = centroid + contraction_factor * (simplex[-1] - centroid)
                
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (22.5 / np.sum(contracted))
            contracted_val = evaluate_c5(contracted, dx)
            
            if contracted_val < values[-1]:
                simplex[-1] = contracted
            else:
                # Shrink simplex
                simplex = simplex[0:1] + (simplex[0] + simplex[1:] ) / 2.0
                
        # Periodic perturbation
        if iter_since_improvement % perturb_freq == 0 and iter_since_improvement > 0:
            best_idx = np.argmin(values)
            perturb_mask = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_perturb = simplex[best_idx].copy()
            h_perturb[perturb_mask] += np.random.normal(0, 0.01, len(perturb_mask))
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (22.5 / np.sum(h_perturb))
            simplex[-1] = h_perturb
            
    # Final evaluation
    best_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
