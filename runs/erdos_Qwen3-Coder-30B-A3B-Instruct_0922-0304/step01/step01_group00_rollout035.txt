import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    # Project to sum = n_points/2 and bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def segment_crossover(parent1, parent2, n_points):
    # Perform segment-wise crossover
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, sigma=0.02):
    # Mutate with Gaussian noise
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, sigma, len(h))
    h = h + mask * noise
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 0
    max_generations = 1200  # budget allows ~1200 generations
    
    # Initialize population
    population = []
    
    # If initial construction exists, use it
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points, dx)
    else:
        # Fallback to structured binary construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points, dx)
        
    population.append(h.copy())
    
    # Generate diverse initial population
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points, dx)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_c5(h, dx)
        fitness.append(c5)
    
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx].copy()
    
    # Main evolution loop
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        generations += 1
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)
        selected_indices = sorted_indices[:population_size//2]
        selected_population = [population[i] for i in selected_indices]
        selected_fitness = [fitness[i] for i in selected_indices]
        
        # Create new population through crossover and mutation
        new_population = [best_h.copy()]
        new_fitness = [best_c5]
        
        while len(new_population) < population_size:
            # Crossover
            parent1_idx = np.random.choice(len(selected_population))
            parent2_idx = np.random.choice(len(selected_population))
            child1, child2 = segment_crossover(
                selected_population[parent1_idx], 
                selected_population[parent2_idx], 
                n_points
            )
            
            # Mutate
            child1 = mutate(child1)
            child1 = project_to_feasible(child1, n_points, dx)
            child2 = mutate(child2)
            child2 = project_to_feasible(child2, n_points, dx)
            
            # Evaluate
            c5_1 = evaluate_c5(child1, dx)
            c5_2 = evaluate_c5(child2, dx)
            
            new_population.extend([child1, child2])
            new_fitness.extend([c5_1, c5_2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        new_fitness = new_fitness[:population_size]
        
        # Check for improvement
        min_fitness_index = np.argmin(new_fitness)
        if new_fitness[min_fitness_index] < best_c5:
            best_c5 = new_fitness[min_fitness_index]
            best_h = new_population[min_fitness_index].copy()
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Perform relaxation every 10 generations
        if generations % 10 == 0:
            # Try to improve the best individual using optimization
            try:
                # Simple local optimization step
                h_opt = best_h.copy()
                step = 0.01
                
                for _ in range(50):
                    corr = correlate(h_opt, 1 - h_opt, mode="full")
                    current_c5 = np.max(corr) * dx
                    
                    grad = np.zeros_like(h_opt)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_opt.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - current_c5) / eps
                    
                    h_opt -= step * grad
                    h_opt = np.clip(h_opt, 0, 1)
                    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                
                # Check if this gives improvement
                c5_new = evaluate_c5(h_opt, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_opt.copy()
                    
            except:
                pass
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
