import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_frequency = 1000
    
    # MCTS tree structure: dictionary with nodes
    tree = {}
    root_node = tuple(h.round(5))
    tree[root_node] = {
        'h': h.copy(),
        'visits': 0,
        'value': float('inf'),
        'children': []
    }
    
    # Simple optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Periodic reset
        if iteration % reset_frequency == 0 and iteration > 0:
            # Reset a portion of nodes
            for node in list(tree.keys()):
                if random.random() < 0.1:
                    h_reset = np.random.rand(len(h))
                    h_reset = h_reset * (np.sum(h) / np.sum(h_reset))
                    tree[node] = {
                        'h': h_reset,
                        'visits': 0,
                        'value': float('inf'),
                        'children': []
                    }
        
        # Selection: choose a node to expand
        current_node = root_node
        while current_node in tree and tree[current_node]['children']:
            children = tree[current_node]['children']
            ucb_values = []
            total_visits = tree[current_node]['visits']
            
            for c in children:
                if tree[c]['visits'] == 0:
                    ucb_values.append(float('inf'))
                else:
                    ucb = tree[c]['value'] + exploration_weight * np.sqrt(np.log(total_visits) / tree[c]['visits'])
                    ucb_values.append(ucb)
            
            # Choose best child
            best_child_idx = np.argmin(ucb_values)
            current_node = children[best_child_idx]
        
        # Expand: create new node
        if current_node in tree:
            # Perform mutation
            h_mutated = tree[current_node]['h'].copy()
            mutate_indices = np.random.choice(len(h_mutated), 
                                              size=int(mutation_rate * len(h_mutated)), 
                                              replace=False)
            for idx in mutate_indices:
                h_mutated[idx] += np.random.normal(0, 0.05)
            h_mutated = np.clip(h_mutated, 0, 1)
            # Renormalize
            scale_factor = np.sum(tree[current_node]['h']) / np.sum(h_mutated)
            h_mutated = h_mutated * scale_factor
            h_mutated = np.clip(h_mutated, 0, 1)
            
            new_node = tuple(h_mutated.round(5))
            if new_node not in tree:
                # Evaluate C5
                c5_val = evaluate_c5(h_mutated, dx)
                # Add new node
                tree[new_node] = {
                    'h': h_mutated.copy(),
                    'visits': 0,
                    'value': c5_val,
                    'children': []
                }
                tree[current_node]['children'].append(new_node)
                
                # Backpropagate
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_mutated.copy()
                
                # Update parent
                cur_node_ptr = current_node
                while cur_node_ptr in tree:
                    tree[cur_node_ptr]['visits'] += 1
                    if tree[cur_node_ptr]['value'] > c5_val:
                        tree[cur_node_ptr]['value'] = c5_val
                    if 'parent' in tree[cur_node_ptr]:
                        cur_node_ptr = tree[cur_node_ptr]['parent']
                    else:
                        break
    
    # Final evaluation of best solution
    c5_final = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), c5_final, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
