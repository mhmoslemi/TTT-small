import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Gradient-based optimization with better initial guess
    def objective(h):
        return compute_C5(h)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess from parent construction
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Try multiple starting points to escape local optima
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Try several random perturbations of the initial solution
    for i in range(10):
        if time.time() - start_time > budget_s - 5:
            break
        h_pert = h_init + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        # Local optimization from this point
        res = minimize(objective, h_pert, method='SLSQP', 
                       bounds=[(0, 1) for _ in range(n_points)],
                       constraints={'type': 'eq', 'fun': constraint_sum},
                       options={'maxiter': 200, 'ftol': 1e-6})
        
        if res.success:
            new_h = res.x
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_C5 = compute_C5(new_h)
            if new_C5 < best_C5:
                best_h = new_h
                best_C5 = new_C5
    
    # Final optimization with more precision
    if time.time() - start_time < budget_s - 10:
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=[(0, 1) for _ in range(n_points)],
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_C5 = compute_C5(final_h)
            if final_C5 < best_C5:
                best_h = final_h
                best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
