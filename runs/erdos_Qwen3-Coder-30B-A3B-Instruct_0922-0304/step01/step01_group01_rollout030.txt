import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Scale to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Rerun scaling due to clipping
    h = h * (n_points / 2) / np.sum(h)
    return h
