import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Objective function to minimize
    def objective(h):
        h = np.clip(h, 0, 1)  # Ensure h is in [0,1]
        # Ensure sum constraint
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Apply constraints using a penalty function
    def penalty(h):
        sum_violation = abs(np.sum(h) - target_sum)
        return 1000 * sum_violation if sum_violation > 1e-3 else 0
    
    def penalized_objective(h):
        return objective(h) + penalty(h)
    
    # Run differential evolution with timeout
    try:
        result = differential_evolution(
            penalized_objective,
            bounds,
            maxiter=1000,
            popsize=40,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False,
            timeout=budget_s - (time.time() - start_time) - 1
        )
        
        if result.success:
            current_h = result.x
        else:
            # Fallback to the best iteration so far
            current_h = h_init.copy()
            
        # Final clipping and normalization
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = evaluate_c5(current_h, dx)
        
    except Exception as e:
        # Fallback to initial construction
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)
    
    return (current_h, current_c5, n_points)
