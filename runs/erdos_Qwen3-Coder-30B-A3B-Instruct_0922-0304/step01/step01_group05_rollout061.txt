import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Frequency domain optimization approach
    def compute_C5_freq(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Use FFT to compute overlap efficiently
        H = np.fft.fft(h)
        H_conj = np.conj(H)
        # Convolution in frequency domain
        overlap_fft = H * (H_conj - 1)  # This is incorrect
        # Correct approach:
        # Compute h * (1 - h) correlation
        h1 = np.concatenate([h, np.zeros(n_points)])
        h2 = np.concatenate([np.zeros(n_points), 1 - h])
        # Compute full convolution using fft
        corr_fft = np.fft.fft(h1) * np.fft.fft(h2)
        # Extract valid correlation values
        corr_vals = np.fft.ifft(corr_fft).real[:2*n_points-1]
        return np.max(corr_vals) * dx
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Objective function using frequency domain
    def objective_freq(h):
        return compute_C5_freq(h)
    
    # Original algorithm approach using scipy minimization
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Constraint and bounds
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize
    res = minimize(objective_freq, h_init, method='SLSQP', 
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 1000, 'ftol': 1e-6},
                   callback=lambda x: None)
    
    best_h = res.x
    # Normalize
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final check of C5
    final_C5 = compute_C5_freq(best_h)
    
    # Make sure it's within bounds
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Compute final C5
    final_C5 = compute_C5_freq(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
