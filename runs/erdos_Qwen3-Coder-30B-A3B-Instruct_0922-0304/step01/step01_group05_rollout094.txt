import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx=2.0/42):
    """Evaluate the Erdős constant for given h values."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    pbest_positions = [p.copy() for p in particles]

        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / 1000)  # Assuming max iter ~1000

            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +

            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            # Evaluate current particle
            current_value = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
