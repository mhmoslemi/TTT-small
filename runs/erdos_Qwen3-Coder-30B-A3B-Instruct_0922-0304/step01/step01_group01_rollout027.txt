import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_algorithm(seed, budget_s, n_points):
    np.random.seed(seed)
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3
    stagnation_generations = 20
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = project_to_feasible(initial_h_values.copy(), n_points)
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_feasible(h_init, n_points)
        population.append(h_init)
    
    # Fill rest of population with random feasible solutions
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    
    while (time.time() - start_time < budget_s * 0.95) and (generation < max_generations):
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Find best in current population
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # If stagnated, reset
        if stagnation_count > stagnation_generations:
            break
            
        # Selection and reproduction
        new_population = [best_h.copy()]
        
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent = population[winner_idx].copy()
            
            # Crossover (simple arithmetic average with another parent)
            if np.random.rand() < 0.8:  # 80% crossover rate
                other_idx = np.random.choice(pop_size, 1)[0]
                if other_idx != winner_idx:
                    child = (parent + population[other_idx]) / 2.0
                else:
                    child = parent
            else:
                child = parent
                
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_to_feasible(child, n_points)
            
            # Ensure feasibility
            child = project_to_feasible(child, n_points)
            
            new_population.append(child)
            
        population = new_population
        generation += 1
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Use small number of points to keep optimization fast and within limits
    n_points = 42
    
    best_h, best_c5 = genetic_algorithm(seed, budget_s, n_points)
    
    # Final validation
    dx = 2.0 / n_points
    final_c5 = evaluate_c5(best_h, dx)
    
    # Make sure the final solution is really valid
    best_h = project_to_feasible(best_h.copy(), n_points)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h, final_c5, n_points)
