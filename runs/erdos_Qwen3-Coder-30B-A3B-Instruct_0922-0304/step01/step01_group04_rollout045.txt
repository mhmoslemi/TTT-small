import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Initialize with a sawtooth shape
        h_init = np.array([0.5 + 0.5 * np.sin(i * np.pi / n_points) for i in range(n_points)])
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize the initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        return evaluate_c5(h, dx)
    
    # Starting point
    current_h = h_init.copy()
    current_c5 = compute_c5(current_h)

    # Optimization parameters
    max_iter = 500
    tol = 1e-6
    step_size = 0.01
    
    # Gradient descent with projection
    for i in range(max_iter):
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Compute gradient numerically
        grad = np.zeros_like(current_h)
        eps = 1e-5
        base_c5 = compute_c5(current_h)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            
            grad[j] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
            
        # Update step
        new_h = current_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        
        # Check improvement
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
        else:
            step_size *= 0.99
            if step_size < 1e-8:
                break

    # Final computations
    final_h = np.clip(current_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(final_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in final_h), "Bounds constraint violated"
    
    return (final_h, final_c5, n_points)
