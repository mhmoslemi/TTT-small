import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: smooth step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated annealing parameters
    T = 0.05
    cooling_factor = 0.9998
    max_iter = 100000
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Track acceptance rate
    accepted = 0
    total = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a new candidate solution
        h_new = h.copy()
        
        # Randomly choose between perturbation strategies
        if np.random.random() < 0.7:
            # Binary-like perturbation: swap two values
            i, j = np.random.choice(n_points, 2, replace=False)
            # Swap values and adjust to maintain sum constraint
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Add small noise to two random points
            i, j = np.random.choice(n_points, 2, replace=False)
            noise_i = np.random.normal(0, 0.01)
            noise_j = np.random.normal(0, 0.01)
            h_new[i] += noise_i
            h_new[j] += noise_j
            # Adjust both to maintain sum
            adjustment = -(noise_i + noise_j) * (n_points / 2) / np.sum(h_new)
            h_new[i] += adjustment / 2
            h_new[j] += adjustment / 2
        
        # Clip to ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        c5_old = evaluate_c5(h, dx)
        
        # Accept or reject based on simulated annealing
        total += 1
        if c5_new < c5_old or np.random.random() < np.exp((c5_old - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = iteration
                
        # Cooling schedule
        T *= cooling_factor
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
