import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h[i] <= 1"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2):
    """Perform segment-wise crossover"""
    n = len(parent1)
    # Random crossover point
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.02):
    """Apply Gaussian noise to some genes"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, noise_std)
    return h

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Create structured initial population
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, n_points)
    
    population.append(h.copy())
    
    # Generate remaining individuals
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, n_points)
        population.append(h.copy())
    
    # Evaluate initial population
    population_fitness = []
    for h in population:
        c5 = evaluate_c5(h, dx)
        population_fitness.append(c5)
    
    best_c5 = min(population_fitness)
    best_h = population[np.argmin(population_fitness)].copy()
    
    # Parameters
    generation = 0
    relaxation_interval = 10
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(population_fitness)
        top_indices = sorted_indices[:population_size // 2]
        parents = [population[i] for i in top_indices]
        parent_fitness = [population_fitness[i] for i in top_indices]
        
        # Generate new population
        new_population = parents.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            
            # Crossover
            if np.random.random() < 0.8:  # Crossover rate
                child1, child2 = crossover(parent1, parent2)
                child1 = project_h(child1, n_points)
                child2 = project_h(child2, n_points)
                
                # Mutation
                child1 = mutate(child1, 0.2, 0.02)
                child1 = project_h(child1, n_points)
                
                child2 = mutate(child2, 0.2, 0.02)
                child2 = project_h(child2, n_points)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate one parent
                child = mutate(parent1, 0.2, 0.02)
                child = project_h(child, n_points)
                new_population.append(child)
        
        # Update population
        population = new_population[:population_size]
        
        # Evaluate new population
        population_fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            population_fitness.append(c5)
        
        # Update best solution
        current_best_c5 = min(population_fitness)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[np.argmin(population_fitness)].copy()
        
        # Relaxation step
        if generation % relaxation_interval == 0:
            # Solve a simple convex relaxation
            # Use best individual to narrow search
            h_relaxed = best_h.copy()
            
            # Perturb slightly to find better local minima
            h_relaxed += np.random.normal(0, 0.01, n_points)
            h_relaxed = project_h(h_relaxed, n_points)
            
            # Local optimization using simple gradient
            step_size = 0.01
            for _ in range(20):
                # Gradient estimation
                grad = np.zeros_like(h_relaxed)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_relaxed.copy()
                    h_plus[i] += eps
                    h_plus = project_h(h_plus, n_points)
                    c5_plus = evaluate_c5(h_plus, dx)
                    grad[i] = (c5_plus - current_best_c5) / eps
                
                # Move along gradient
                h_relaxed -= step_size * grad
                h_relaxed = project_h(h_relaxed, n_points)
                
                # Recompute
                current_best_c5 = evaluate_c5(h_relaxed, dx)
            
            # Replace worst individual with relaxed solution
            worst_idx = np.argmax(population_fitness)
            population[worst_idx] = h_relaxed
            population_fitness[worst_idx] = current_best_c5
            
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = h_relaxed.copy()
        
        # Early exit if promising
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
