import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Identify maximum overlap shift
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (overlap_plus[k_max] - overlap[k_max]) / eps
        
        # Hessian approximation (diagonal)
        hessian_diag = 1 - 2 * h
        
        # Quadratic approximation of the overlap at k_max
        # Minimize: grad^T * d + 0.5 * d^T * H * d subject to h+d in [0,1], sum(d) = 0
        # Simple projected gradient step
        step_size = 0.1
        d = -step_size * grad
        
        # Project d to ensure sum(d) = 0
        d = d - np.mean(d)
        
        # Update h
        h_new = h + d
        
        # Projection to [0,1]^n
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to meet integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
