import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    max_iter = 10000
    max_evals = 100000
    
    current_h = h.copy()
    current_C5 = best_C5
    T = T0
    
    eval_count = 0
    iter_count = 0
    
    while eval_count < max_evals and time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        if iter_count > max_iter:
            break
            
        # Create a new candidate solution by perturbing current solution
        new_h = current_h.copy()
        
        # Choose a random subset to modify
        n_modify = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        # Perturb chosen points
        for idx in indices:
            # Random perturbation
            delta = np.random.normal(0, 0.1 * T)
            new_h[idx] += delta
            
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        
        # Adjust to meet sum constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2.0
        diff = target_sum - current_sum
        
        if abs(diff) > 1e-6:
            # Distribute the difference
            if diff > 0:
                # Need to increase values
                for i in range(n_points):
                    if new_h[i] < 1:
                        add = min(diff, 1 - new_h[i])
                        new_h[i] += add
                        diff -= add
                        if diff <= 0:
                            break
            else:
                # Need to decrease values
                for i in range(n_points):
                    if new_h[i] > 0:
                        subtract = min(-diff, new_h[i])
                        new_h[i] -= subtract
                        diff += subtract
                        if diff >= 0:
                            break
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(new_h, dx)
        eval_count += 1
        
        # Accept with simulated annealing probability
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            current_h = new_h
            current_C5 = new_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cool down temperature
        T = T * alpha
    
    # Final validation of best solution
    final_C5 = evaluate_erdos_solution(best_h, dx)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h, best_C5, n_points)
