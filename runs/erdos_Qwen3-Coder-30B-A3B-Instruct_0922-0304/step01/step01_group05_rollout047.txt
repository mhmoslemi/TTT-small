import numpy as np
import time
from scipy.signal import correlate
import random

def evaluate_h(h, dx):
    """Compute C5 for given h values."""
    # Ensure h is properly normalized
    h = np.array(h)
    h = h * (2.0 / dx) / np.sum(h)
    # Compute overlap integral for all lags
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points, dx):
    """Repair h to ensure it's in [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    child = np.array(parent1)
    mask = np.random.rand(len(parent1)) > 0.5
    child[mask] = parent2[mask]
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Apply mutation to h."""
    h = np.array(h)
    mask = np.random.rand(len(h)) < mutation_rate
    h[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]
