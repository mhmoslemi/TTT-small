import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values or create fallback
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:
        # Fallback: binary pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h_val):
        corr = correlate(h_val, 1 - h_val, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simulated Annealing parameters
    T = 0.05  # Initial temperature
    cooling_rate = 0.9998
    min_T = 1e-6
    max_iter = int(1e5)
    
    # Perturbation strategies
    def perturb_swap(h_val):
        h_new = h_val.copy()
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        # Swap values
        h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        return h_new
    
    def perturb_noise(h_val):
        h_new = h_val.copy()
        # Select two indices to modify
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        # Add noise
        noise1 = np.random.normal(0, 0.01)
        noise2 = -noise1  # Balance to preserve sum
        h_new[i1] += noise1
        h_new[i2] += noise2
        # Clamp
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Main optimization loop
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # Choose perturbation type
        if np.random.rand() < 0.7:
            h_new = perturb_swap(best_h)
        else:
            h_new = perturb_noise(best_h)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update temperature
        T = max(T * cooling_rate, min_T)
        
        # Early exit if target achieved
        if best_c5 <= target_c5:
            break
    
    return (best_h.tolist(), best_c5, n_points)
