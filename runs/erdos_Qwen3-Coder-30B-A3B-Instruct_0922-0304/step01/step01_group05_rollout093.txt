import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 5000
    num_particles = 50
    w_max, w_min = 0.9, 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initialize particles around initial_h_values
    initial_h = np.array(initial_h_values)
    h = np.zeros((num_particles, n_points))
    for i in range(num_particles):
        h[i] = initial_h + 0.2 * (2 * np.random.rand(n_points) - 1)
        h[i] = np.clip(h[i], 0, 1)
        h[i] *= (n_points / 2) / np.sum(h[i])
    
    # Initialize velocities
    v = 0.1 * np.random.randn(num_particles, n_points)
    
    # Compute initial overlaps
    def compute_overlap(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Repair function
    def repair(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum != 0:
            h_vec *= (n_points / 2) / current_sum
        else:
            h_vec = np.ones_like(h_vec) * (n_points / 2) / n_points
        return h_vec
    
    # Initialize personal bests and global best
    p_best = h.copy()
    p_best_scores = np.array([compute_overlap(h[i]) for i in range(num_particles)])
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO loop
    iter_no_improve = 0
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (elapsed / budget_s)
        
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v[i] = (w * v[i] + 
                   c1 * r1 * (p_best[i] - h[i]) + 
                   c2 * r2 * (g_best - h[i]))
            
            # Clip velocities
            v[i] = np.clip(v[i], -v_max, v_max)
            
            h[i] += v[i]
            
            # Repair solution
            h[i] = repair(h[i])
            
            # Evaluate new solution
            score = compute_overlap(h[i])
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = h[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < g_best_score:
                    g_best = h[i].copy()
                    g_best_score = score
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
        
        # Perturbation mechanism
        if iter_no_improve > 20:
            for i in range(num_particles):
                if np.random.rand() < 0.2:  # 20% chance of perturbation
                    h[i] = h[i] + 0.1 * (2 * np.random.rand(n_points) - 1)
                    h[i] = repair(h[i])
            iter_no_improve = 0
    
    # Final evaluation
    final_C5 = compute_overlap(g_best)
    
    return (g_best.tolist(), final_C5, n_points)
