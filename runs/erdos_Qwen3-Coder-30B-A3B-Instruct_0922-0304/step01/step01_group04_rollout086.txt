import numpy as np
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Convert to CVXPY problem
    # h variables
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(name="t")  # Upper bound on overlap

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]

    # For each shift k from -n_points+1 to n_points-1
    # we need to ensure sum(h[i] * (1 - h[i+k])) <= t
    # where the sum is over valid pairs and we treat out-of-bounds as 0
    # For k=0: h[i] * (1 - h[i]) = h[i] - h[i]^2 <= t
    # For k!=0: we sum over valid i's
    
    # Compute appropriate shifts and corresponding constraints
    for k in range(-n_points + 1, n_points):
        if k == 0:
            # sum(h[i] - h[i]^2) <= t
            # This is equivalent to sum(h[i]) - sum(h[i]^2) <= t
            # i.e., target_sum - cp.sum(cp.power(h, 2)) <= t
            constraints.append(target_sum - cp.sum(cp.power(h, 2)) <= t)
        else:
            # Build indices for h[i] and h[i+k] for valid i (such that i+k is in [0, n_points-1])
            total_shifted = 0
            valid_pairs_sum = 0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    total_shifted += 1
                    valid_pairs_sum += h[i] * (1 - h[j])
            constraints.append(valid_pairs_sum <= t)

    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)

    # Solve with MOSEK (if available) or fallback to ECOS if needed for compatibility
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-8, reltol=1e-8, abstol=1e-8)
    except:
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                # Fallback: simply use the initial solution if optimization fails
                current_h = h_init.copy()
                current_c5 = evaluate_c5(current_h, dx)
                return (current_h, current_c5, n_points)

    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_solution = h.value
        t_solution = t.value
    else:
        # If optimization fails, return the best non-convex result we can get
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)
        return (current_h, current_c5, n_points)

    # Post-process: ensure h lies in [0,1] and sum is correct
    h_solution = np.clip(h_solution, 0, 1)
    sum_h = np.sum(h_solution)
    h_solution = h_solution * target_sum / sum_h

    # compute final C5
    final_c5 = evaluate_c5(h_solution, dx)

    # Ensure correctness
    assert np.isclose(np.sum(h_solution), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_solution), "Bounds constraint violated"

    # Return
    return (h_solution, final_c5, n_points)
