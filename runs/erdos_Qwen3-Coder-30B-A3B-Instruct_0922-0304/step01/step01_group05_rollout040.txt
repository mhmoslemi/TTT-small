def compute_C5(h):
    # Normalize h to sum to n_points/2
    h = np.array(h)
    n_points = len(h)
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    # Compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Return max correlation (scaled by dx)
    return np.max(corr) * dx

def repair(h, n_points):
    # Clip h[i] to [0,1] and rescale to maintain sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    desired_sum = n_points / 2
    if current_sum != 0:
        h *= desired_sum / current_sum
    else:
        h = np.ones_like(h) * (desired_sum / n_points)
    return h
