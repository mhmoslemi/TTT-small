import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Create a new optimization approach: use a spectral approach
    # with convex optimization to minimize the maximum overlap
    def new_objective(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Constraint that sum of h equals n_points/2
    def sum_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimize using a variation of trust-region optimization
    # Init with simple grid
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Add some randomness to escape local optima
    np.random.seed(seed)
    h_init = h_init + np.random.normal(0, 0.01, n_points)
    h_init = np.clip(h_init, 0, 1)
    
    # Refine the solution using scipy minimize with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': sum_constraint}
    
    res = minimize(new_objective, h_init, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-8})
    
    best_h = res.x
    
    # Final normalization to ensure sum = n_points/2
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Compute final C5
    final_C5 = compute_C5(best_h)
    
    # Ensure solution is within bounds
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), final_C5, n_points)
