import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Initialize with a sawtooth pattern
        h_init = 1.0 - np.abs(np.sin(np.arange(n_points) * np.pi / n_points))
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Project h to [0,1] and normalize
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * target_sum / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Advanced optimization with trust region adjustments
    max_iter = 50
    trust_region = 0.3
    prev_c5 = float('inf')
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 2:
            break
            
        # Perform a local optimization step
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            return evaluate_c5(h, dx)
        
        # Find the optimal step within trust region
        try:
            # Linearize the problem around current solution
            delta = trust_region * np.random.rand(n_points) - trust_region/2
            h_trial = current_h + delta
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * target_sum / np.sum(h_trial)
            
            trial_c5 = objective(h_trial)
            if trial_c5 < current_c5:
                current_h = h_trial.copy()
                current_c5 = trial_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
            # Adjust trust region
            improvement = (prev_c5 - current_c5) / max(prev_c5, 1e-10)
            if improvement < 0.01:
                trust_region *= 0.9
            else:
                trust_region = min(trust_region * 1.1, 0.5)
                
            prev_c5 = current_c5
            
            # Small perturbations to escape local minima
            if iteration % 5 == 0 and iteration > 0:
                noise = np.random.normal(0, 0.01, n_points)
                h_perturb = current_h + noise
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * target_sum / np.sum(h_perturb)
                perturb_c5 = objective(h_perturb)
                if perturb_c5 < current_c5:
                    current_h = h_perturb.copy()
                    current_c5 = perturb_c5
                    
        except Exception as e:
            pass
            
        # Occasionally re-initialize from global best
        if iteration % 10 == 0 and iteration > 0:
            current_h = best_h.copy()
            current_c5 = best_c5
    
    # Final validation
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    # Return best solution found
    return (current_h, current_c5, n_points)
