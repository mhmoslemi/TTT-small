import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h_init)
    else:

        h_init = 1.0 - np.abs(np.sin(np.arange(n_points) * 0.5))
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Use a trust region approach with quadratic approximations
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Parameters
    max_iter = 500
    tol = 1e-6
    trust_radius = 0.5
    
    # Trust region optimization loop
    for iteration in range(max_iter):
        # Time check
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Create quadratic approximation of the objective
        # Linearization around current point
        h_current = current_h.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * target_sum / np.sum(h_current)
        
        # Determine the negative gradient direction manually
        # by finite differencing
        eps = 1e-4
        grad_h = np.zeros_like(h_current)
        for i in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad_h[i] = (c5_plus - current_c5) / eps
            
        # Find direction that minimizes a quadratic model
        # We are minimizing a quadratic model in a trust region
        
        # Use a heuristic step that considers the gradient
        step_direction = -grad_h
        step_norm = np.linalg.norm(step_direction)
        if step_norm > 0:
            step_direction = step_direction / step_norm * trust_radius
            
        # Try a step
        h_candidate = h_current + step_direction
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        
        c5_candidate = evaluate_c5(h_candidate, dx)
        
        # Accept or reject based on if it improves
        if c5_candidate < current_c5:
            current_h = h_candidate.copy()
            current_c5 = c5_candidate
            trust_radius = min(trust_radius * 1.5, 1.0)
        else:
            trust_radius *= 0.5
            
        # For stability, periodically reproject
        if iteration % 5 == 0:
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = evaluate_c5(current_h, dx)
    
    # Final check and normalization
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    final_c5 = evaluate_c5(current_h, dx)
    
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, final_c5, n_points)
