import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential convex programming approach
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute overlaps and find the worst case
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-4
        h_copy = h.copy()
        
        for i in range(n_points):
            h_plus = h_copy.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_plus = corr_plus[max_idx] * dx
            grad[i] = (max_plus - max_overlap) / eps
            
        # Formulate and solve convex subproblem for descent direction
        # We approximate Hessian as diagonal with (1 - 2*h[i])
        hessian_diag = 1 - 2 * h
        hessian_diag = np.clip(hessian_diag, 1e-6, 1e6)  # Prevent zero or negative
        
        # Use the negative gradient for descent, weighted by a small amount
        d = -grad / (1.0 + np.linalg.norm(grad) * 1e-3)
        
        # Ensure d is direction that reduces the maximum overlap
        d = d * 0.01  # Step size adjustment
        
        # Apply the step
        h_new = h + d
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new

    return (best_h.tolist(), best_c5, n_points)
