import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 45
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2.0 / np.sum(h))
    population.append(h.copy())
    
    # Generate random initial population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0 / np.sum(h))
        population.append(h.copy())
        
    best_h = population[0].copy()
    best_c5 = evaluate_c5(population[0], 2.0 / n_points)
    
    # Evolution parameters
    generations = 2000
    mutation_rate = 0.05
    mutation_noise = 0.05
    elite_size = population_size // 10
    no_improvement_count = 0
    max_no_improvement = 50
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append((c5, h))
        
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        best_fitness = fitness[0][0]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = fitness[0][1].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Selection: Top 50% as parents
        parents = [ind for _, ind in fitness[:population_size//2]]
        
        # Create new population
        new_population = []
        
        # Elitism: keep top 10%
        for i in range(elite_size):
            new_population.append(fitness[i][1].copy())
            
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            parent1 = parents[np.random.randint(0, len(parents))]
            parent2 = parents[np.random.randint(0, len(parents))]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, mutation_noise)
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            scale = (n_points / 2.0) / np.sum(child)
            child = child * scale
            
            new_population.append(child)
            
        population = new_population[:population_size]
        
        # Diversity maintenance
        if no_improvement_count > max_no_improvement:
            for i in range(population_size // 5):
                idx = np.random.randint(0, population_size)
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2.0 / np.sum(h))
                population[idx] = h
            mutation_noise *= 0.9
            no_improvement_count = 0
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
