import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Settings
    n_points = 42
    dx = 2.0 / n_points
    max_k = n_points - 1
    
    # Construct initial guess from provided values if available
    try:
        h_init = np.array(initial_h_values)
    except NameError:
        # Fallback: create a simple symmetric construction
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial guess
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_C5 = float('inf')
    best_h = h_init.copy()
    
    # Define optimization variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C >= 0]
    
    # For each possible lag k, add quadratic constraint
    for k in range(-max_k, max_k + 1):
        # Compute valid indices for correlation
        if k >= 0:
            i_indices = range(n_points - k)
            j_indices = range(k, n_points)
        else:
            i_indices = range(-k, n_points)
            j_indices = range(n_points + k)
        
        if len(i_indices) > 0:
            # Build the sum of h[i] * (1 - h[i + k])
            expr = sum(h[i] * (1 - h[j]) for i, j in zip(i_indices, j_indices))
            constraints.append(expr <= C * dx)
    
    # Objective
    objective = Minimize(C)
    
    # Create problem
    prob = Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = ['SCS', 'ECOS']
    
    for solver in solvers_to_try:
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            prob.solve(solver=solver, verbose=False, max_iters=1000)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                current_C5 = C.value
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.value.copy()
        except:
            continue
    
    # Local refinement with scipy if needed
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        
        def compute_C5(h_vals):
            # Normalize
            h_vals = np.array(h_vals)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            # Compute overlap
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
        
        def objective(h_vals):
            return compute_C5(h_vals)
        
        # Use best solution as starting point
        h_refine = best_h.copy()
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        # Limit bounds for numerical stability
        bounds = [(0, 1) for _ in range(n_points)]
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        res = minimize(objective, h_refine, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 500, 'ftol': 1e-8})
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_C5 = compute_C5(final_h)
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = final_h
    
    return (best_h.tolist(), best_C5, n_points)
