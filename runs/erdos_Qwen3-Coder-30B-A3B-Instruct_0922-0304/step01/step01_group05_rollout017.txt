import numpy as np
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    num_particles = 50
    max_iter = 500
    w_max, w_min = 0.9, 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initialize particles around initial_h_values
    initial_h = np.array(initial_h_values)
    particles = []
    for i in range(num_particles):
        h = initial_h + 0.2 * (2 * np.random.rand(n_points) - 1)
        # Repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    velocities = 0.1 * np.random.randn(num_particles, n_points)
    
    # Compute initial overlaps
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (n_points / 2) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [compute_overlap(p) for p in particles]
    
    # Initialize global best
    best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]
    
    # PSO loop
    no_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (iteration / max_iter)
        
        # Update particles
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair
            particles[i] = repair(particles[i])
            
            # Evaluate
            score = compute_overlap(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    no_improvement = 0
            else:
                no_improvement += 1
                
        # Perturb if no improvement
        if no_improvement >= 20:
            for i in range(num_particles):
                particles[i] += 0.01 * np.random.randn(n_points)
                particles[i] = repair(particles[i])
            no_improvement = 0
    
    return (global_best.tolist(), global_best_score, n_points)
