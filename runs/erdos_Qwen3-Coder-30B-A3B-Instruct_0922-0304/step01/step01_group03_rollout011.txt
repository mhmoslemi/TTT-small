import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: create a binary-like vector
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
    
    # Renormalize to sum to n_points/2
    dx = 2.0 / len(h)
    h = h * (len(h) / 2.0) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    no_improvement_limit = 1000
    perturb_freq = 1000
    
    # Create initial simplex
    n_points = len(h)
    simplex = [h.copy()]
    for _ in range(n_points):
        pert = h + simplex_size * np.random.normal(0, 1, n_points)
        pert = np.clip(pert, 0, 1)
        pert = pert * (len(pert) / 2.0) / np.sum(pert)
        simplex.append(pert)
    
    iter_since_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate all simplex points
        values = []
        for point in simplex:
            val = evaluate_c5(point, dx)
            values.append(val)
        
        # Sort by value
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Check improvement
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Termination conditions
        if iter_since_improvement > no_improvement_limit:
            break
            
        # Check simplex size
        if np.std(values) < 1e-6:
            break
            
        # Create new simplex using Nelder-Mead
        centroid = np.mean(simplex[:-1], axis=0)
        worst = simplex[-1]
        
        # Reflection
        reflected = centroid + reflection_factor * (centroid - worst)
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (len(reflected) / 2.0) / np.sum(reflected)
        reflected_val = evaluate_c5(reflected, dx)
        
        if values[0] <= reflected_val < values[-2]:
            simplex[-1] = reflected
        elif reflected_val < values[0]:
            # Expansion
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (len(expanded) / 2.0) / np.sum(expanded)
            expanded_val = evaluate_c5(expanded, dx)
            if expanded_val < reflected_val:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            if reflected_val < values[-1]:
                contracted = centroid + contraction_factor * (reflected - centroid)
            else:
                contracted = centroid + contraction_factor * (worst - centroid)
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (len(contracted) / 2.0) / np.sum(contracted)
            contracted_val = evaluate_c5(contracted, dx)
            if contracted_val <= values[-1]:
                simplex[-1] = contracted
            else:
                # Shrink simplex
                best = simplex[0]
                for i in range(1, len(simplex)):
                    simplex[i] = best + 0.5 * (simplex[i] - best)
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (len(simplex[i]) / 2.0) / np.sum(simplex[i])
        
        # Periodic perturbations
        if iteration % perturb_freq == 0 and iteration > 0:
            best_h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            best_h_perturbed = np.clip(best_h_perturbed, 0, 1)
            best_h_perturbed = best_h_perturbed * (len(best_h_perturbed) / 2.0) / np.sum(best_h_perturbed)
            # Replace worst vertex of simplex with perturbed best
            simplex[-1] = best_h_perturbed
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
