import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Solver settings
    solver = 'SCS'
    solver_opts = {'verbose': False, 'max_iters': 5000, 'eps': 1e-6, 'warm_start': True}
    
    best_C5 = float('inf')
    best_h = None
    
    # Try different initializations
    initial_guesses = [initial_h]
    
    # Add some random perturbations to explore
    for _ in range(3):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        initial_guesses.append(perturbed)
    
    for i, h_init in enumerate(initial_guesses):
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        try:
            # Define CVXPY variables
            h = Variable(n_points)
            C = Variable()
            
            # Define constraints
            constraints = [
                h >= 0,
                h <= 1,
                sum(h) == n_points / 2,
                C >= 0
            ]
            
            # Add overlap constraints for all lags
            for k in range(-n_points + 1, n_points):
                if k < 0:
                    # Overlap for negative k (h shifted right)
                    valid_indices = range(0, n_points + k)
                    expr = sum(h[i] * (1 - h[i - k]) for i in valid_indices)
                else:
                    # Overlap for positive k (h shifted left)
                    valid_indices = range(k, n_points)
                    expr = sum(h[i] * (1 - h[i - k]) for i in valid_indices)
                
                constraints.append(expr <= C * dx)
            
            # Define objective
            objective = Minimize(C)
            
            # Create and solve problem
            prob = Problem(objective, constraints)
            prob.solve(solver=solver, **solver_opts)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                current_C5 = C.value
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.value.copy()
                    
                    # Early termination if we're close to target
                    if best_C5 < 0.3807:
                        break
                        
        except Exception as e:
            continue
            
        # Time check
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # If no solution was found, return initial guess
    if best_h is None:
        best_h = initial_h
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_C5 = 0  # placeholder
        # Compute actual C5
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_C5 = np.max(corr) * dx
    
    # Normalize final solution
    if best_h is not None:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        # Final C5 computation
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_C5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_C5, n_points)
