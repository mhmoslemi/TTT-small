import numpy as np
import time

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T_min = 1e-8
    max_iters = 10000
    
    T = T0
    iteration = 0
    
    # Ensure initial h satisfies the integral constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2.0) / h_sum
    
    while T > T_min and iteration < max_iters and (time.time() - start_time) < budget_s * 0.95:
        # Create a new candidate by perturbing current solution
        h_new = h.copy()
        
        # Select a random subset of points to modify
        num_modify = max(1, int(0.1 * n_points))  # 5-10% of points
        indices = np.random.choice(n_points, size=num_modify, replace=False)
        
        # Apply small random perturbations
        for i in indices:
            # Small random change in [0, 0.2]
            delta = np.random.uniform(-0.2, 0.2)
            h_new[i] += delta
        
        # Ensure h_new stays within [0,1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum = n_points / 2
        h_sum_new = np.sum(h_new)
        adjustment = (n_points / 2.0 - h_sum_new) / n_points  # one per element
        h_new += adjustment
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if C5_new < best_C5 or np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        
        # Cool down
        T *= alpha
        iteration += 1
    
    return (best_h, best_C5, n_points)
