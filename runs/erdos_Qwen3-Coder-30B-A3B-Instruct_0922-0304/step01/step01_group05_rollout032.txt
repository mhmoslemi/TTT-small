import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Gradient-based optimization
    def objective(h):
        return compute_C5(h)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different optimization approaches
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Start with a different optimization algorithm
    try:
        # Try L-BFGS-B which typically handles bounds better
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1000, 'ftol': 1e-8})
        if res.success:
            candidate_h = res.x
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            candidate_C5 = compute_C5(candidate_h)
            if candidate_C5 < best_C5:
                best_h = candidate_h
                best_C5 = candidate_C5
    except:
        pass
    
    # Try a few random restarts to escape local minima
    for i in range(10):
        if time.time() - start_time > budget_s - 5:
            break
        # Random perturbation
        perturbed = best_h + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        
        # Project onto feasible set with sum constraint
        for _ in range(10):
            # Gradient step in direction of negative gradient 
            current_C5 = compute_C5(perturbed)
            # Small finite difference approximation of gradient
            eps = 1e-6
            grad = np.zeros_like(perturbed)
            for j in range(n_points):
                perturbed_plus = perturbed.copy()
                perturbed_plus[j] += eps
                perturbed_plus = perturbed_plus * (n_points / 2) / np.sum(perturbed_plus)
                grad[j] = (compute_C5(perturbed_plus) - current_C5) / eps
            
            # Update with negative gradient step
            perturbed = perturbed - 0.01 * grad
            # Project onto feasible set
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        
        new_C5 = compute_C5(perturbed)
        if new_C5 < best_C5:
            best_h = perturbed
            best_C5 = new_C5
    
    # Final optimization with SLSQP to fine-tune
    if time.time() - start_time < budget_s - 5:
        try:
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_h = final_h * (n_points / 2) / np.sum(final_h)
                final_C5 = compute_C5(final_h)
                if final_C5 < best_C5:
                    best_h = final_h
                    best_C5 = final_C5
        except:
            pass
    
    return (best_h.tolist(), best_C5, n_points)
