import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Define variables for CVXPY optimization in frequency domain
    def compute_C5_cvx(h_vals):
        # Ensure h_vals is a CVXPY variable
        h = cp.Variable(n_points)
        h.value = h_vals
        
        # Constraints
        constraints = [cp.sum(h) == n_points / 2, h >= 0, h <= 1]
        
        # Objective: minimize max overlap
        # This will be handled by the max constraint in the optimization
        
        # Manually compute the correlations to get a bound
        h_array = np.array(h_vals)
        h_array = h_array * (n_points / 2) / np.sum(h_array)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        return np.max(corr) * dx
    
    # Simplified approach with gradient-based optimization
    def objective(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Use a hybrid approach: start with initial guess, then refine
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Post-process: refine using a local optimizer
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try optimization with different methods
    methods = ['SLSQP', 'L-BFGS-B']
    
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    for method in methods:
        if time.time() - start_time > budget_s - 10:
            break
        try:
            res = minimize(objective, h_init, method=method, 
                          bounds=bounds, options={'maxiter': 500})
            if res.success:
                h_candidate = res.x
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = compute_C5(h_candidate)
                if c5_candidate < best_C5:
                    best_C5 = c5_candidate
                    best_h = h_candidate
        except:
            pass
            
        if time.time() - start_time > budget_s - 10:
            break
            
    # Final refinement step with smaller step size
    if time.time() - start_time < budget_s - 10:
        try:
            res = minimize(objective, best_h, method='Nelder-Mead', 
                          options={'maxiter': 200, 'xatol': 1e-6})
            if res.success:
                h_candidate = res.x
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = compute_C5(h_candidate)
                if c5_candidate < best_C5:
                    best_C5 = c5_candidate
                    best_h = h_candidate
        except:
            pass
    
    # Final boundary check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
