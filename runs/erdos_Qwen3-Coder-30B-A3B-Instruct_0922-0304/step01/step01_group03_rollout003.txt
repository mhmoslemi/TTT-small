import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    T0 = 1e-3
    gamma = 0.995
    alpha = 0.1
    max_iter = 10000
    
    T = T0
    iter_since_improvement = 0
    max_no_improvement = 200
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Gradient computation using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Simulated annealing update
        h_new = h - alpha * grad + T * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        scale = len(h) / 2.0 / np.sum(h_new)
        h_new = h_new * scale
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update using Metropolis criterion
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
            iter_since_improvement += 1

        # Cooling schedule
        if iter_since_improvement > max_no_improvement:
            T *= gamma
            iter_since_improvement = 0
            
        # Periodic binary perturbation
        if iteration % 5000 == 0 and iteration > 0:
            h = np.zeros_like(h)
            block_size = np.random.randint(5, 10)
            num_blocks = len(h) // block_size
            for i in range(num_blocks):
                if np.random.rand() > 0.5:
                    start_idx = i * block_size
                    end_idx = min(start_idx + block_size, len(h))
                    h[start_idx:end_idx] = 1.0
            h = np.clip(h, 0, 1)
            scale = len(h) / 2.0 / np.sum(h)
            h = h * scale

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
