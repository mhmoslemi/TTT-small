import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, k_max):
    """Compute gradient of overlap at shift k_max w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_max < n:
            grad[i] += 1 - h[i + k_max]
        if i >= k_max:
            grad[i] -= h[i]
    return grad

def line_search(h, k_max, i_star, grad, dx):
    """Perform line search along coordinate i_star to decrease overlap k_max"""
    n = len(h)
    # Determine range for alpha
    alpha_min = 0
    alpha_max = 1
    
    # Check bounds
    if grad[i_star] > 0:
        # We're increasing h[i_star] - clip at 1
        alpha_max = min(alpha_max, (1 - h[i_star]) / grad[i_star])
    else:
        # We're decreasing h[i_star] - clip at 0
        alpha_max = min(alpha_max, -h[i_star] / grad[i_star])
    
    # Binary search for best alpha
    tol = 1e-6
    min_alpha = alpha_min
    max_alpha = alpha_max
    best_alpha = 0.5 * (min_alpha + max_alpha)
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simple binary search up to 10 iterations
    for _ in range(10):
        h_test = h.copy()
        h_test[i_star] += best_alpha * grad[i_star]
        if i_star + k_max < n:
            h_test[i_star + k_max] -= best_alpha * grad[i_star]
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (len(h_test) / 2) / np.sum(h_test)
        c5_test = evaluate_c5(h_test, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
        if grad[i_star] > 0:
            max_alpha = best_alpha
        else:
            min_alpha = best_alpha
        best_alpha = 0.5 * (min_alpha + max_alpha)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Optimization parameters
    max_iter = 10000
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(corr)
        k_max = max_overlap_idx - (n_points - 1)
        current_c5 = np.max(corr) * dx
        
        # Early termination
        if current_c5 <= 0.3807:
            best_h = h.copy()
            best_c5 = current_c5
            break
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h, k_max)
        
        # Find index to update
        i_star = np.argmin(grad)
        
        # Perform line search
        h, c5_val = line_search(h, k_max, i_star, grad, dx)
        
        # Ensure constraint maintained
        h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
