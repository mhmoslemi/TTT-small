import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to desired sum
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient for the k-th lag"""
    gradient = np.zeros(n_points)
    for i in range(n_points):
        if k >= 0:
            # h[i] contributes to overlap with (i-k)th element in 1-h
            # For 1-h, the contribution is -1*sign(h[i] - 0.5)
            j = (i - k) % n_points
            gradient[i] = 1 - 2 * h[j]
        else:
            # Corresponding case when k < 0
            j = (i + k) % n_points
            gradient[i] = 1 - 2 * h[j]
    return gradient
