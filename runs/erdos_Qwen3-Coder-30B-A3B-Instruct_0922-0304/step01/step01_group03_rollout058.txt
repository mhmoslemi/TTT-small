import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    n_points = 45
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        population.append(h.copy())
    else:
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2.0 / np.sum(h))
        population.append(h.copy())
    
    # Generate random initial population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0 / np.sum(h))
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    # Best solution tracking
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    generations = 2000
    elite_size = pop_size // 10
    mutate_rate = 0.05
    mutate_noise = 0.05
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check for stagnation
        if no_improvement_count >= max_no_improvement:
            # Introduce diversity
            for i in range(pop_size // 5):
                idx = np.random.randint(pop_size // 2, pop_size)
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0 / np.sum(h))
                h = np.clip(h, 0, 1)
                population[idx] = h.copy()
            mutate_noise *= 0.9
            no_improvement_count = 0
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection (tournament)
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutate_rate:
                    child[i] += np.random.normal(0, mutate_noise)
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            scale = (n_points / 2.0) / np.sum(child)
            child = child * scale
            
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
