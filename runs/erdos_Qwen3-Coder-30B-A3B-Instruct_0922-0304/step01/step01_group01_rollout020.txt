import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    particles = []
    velocities = []

    h_init = project_to_feasible(h_init, n_points)
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))

    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_fitness = [evaluate_c5(p, 2.0 / n_points) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[global_best_idx].copy()

            r1, r2 = np.random.rand(2)

            velocities[i] = np.clip(velocities[i], velocity_clamp[0], velocity_clamp[1])
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)

            fitness = evaluate_c5(particles[i], 2.0 / n_points)
