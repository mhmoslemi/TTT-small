import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: structured binary initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Simulated Annealing approach
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T = 0.05
    cooling_rate = 0.9998
    min_T = 1e-6
    
    # Iteration counter
    iterations = 0
    max_iterations = int(budget_s * 100)  # Rough estimate
    
    while iterations < max_iterations and time.time() - start_time < budget_s - 1:
        # Perturb h
        h_new = current_h.copy()
        
        # Choose perturbation type (70% swaps, 30% noise)
        if np.random.rand() < 0.7:
            # Swap operation: flip a random bit and adjust another to maintain sum
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            # Swap some value
            val = h_new[i]
            h_new[i] = h_new[j]
            h_new[j] = val
            
            # Ensure feasibility by adjusting to maintain sum constraint
            diff = np.sum(h_new) - (n_points / 2)
            if abs(diff) > 1e-6:
                # Adjust one element to make sum correct
                idx = np.random.randint(0, n_points)
                h_new[idx] = np.clip(h_new[idx] - diff, 0, 1)
        else:
            # Add noise with small perturbations
            # Select two indices to perturb
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            noise = np.random.normal(0, 0.01, 2)
            h_new[i1] += noise[0]
            h_new[i2] += noise[1]
            
            # Clamp to [0,1] and re-normalize to sum constraint
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to meet sum = n_points/2
            current_sum = np.sum(h_new)
            if abs(current_sum - n_points/2) > 1e-6:
                # Adjust one element to fix sum
                idx = np.random.randint(0, n_points)
                adjustment = (n_points/2 - current_sum)
                h_new[idx] = np.clip(h_new[idx] + adjustment, 0, 1)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            current_h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
                # Early termination if target reached
                if best_c5 <= 0.3807:
                    break
        
        # Cool down
        T = max(min_T, T * cooling_rate)
        
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
