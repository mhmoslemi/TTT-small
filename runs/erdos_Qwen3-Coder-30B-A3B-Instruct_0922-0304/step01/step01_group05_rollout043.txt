import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 5000
    num_particles = 50
    w_init = 0.9
    w_min = 0.5
    c1 = c2 = 2.0
    v_max = 0.5
    perturb_freq = 20
    
    # Initial setup
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    personal_best = np.zeros((num_particles, n_points))
    personal_best_scores = np.full(num_particles, np.inf)
    global_best = None
    global_best_score = np.inf
    
    # Initialize particles
    for i in range(num_particles):
        # Start near initial solution with small random perturbations
        particles[i] = h_init + 0.2 * (np.random.rand(n_points) - 0.5)
        particles[i] = np.clip(particles[i], 0, 1)
        # Rescale to maintain sum constraint
        particles[i] *= (n_points / 2) / np.sum(particles[i])
        
        # Set personal best to initial position
        personal_best[i] = particles[i].copy()
        score = compute_overlap(particles[i])
        personal_best_scores[i] = score
        
        if score < global_best_score:
            global_best = particles[i].copy()
            global_best_score = score

    # PSO loop
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia weight
        w = w_init - (w_init - w_min) * (iteration / max_iter)
        
        # Update particles
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair: clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = compute_overlap(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
                
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    last_improvement = iteration
        
        # Perturbation to escape local optima
        if iteration - last_improvement > perturb_freq:
            for i in range(num_particles):
                if np.random.rand() < 0.3:  # Perturb 30% of particles
                    particles[i] += 0.1 * (np.random.rand(n_points) - 0.5)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] *= (n_points / 2) / np.sum(particles[i])
                    
                    score = compute_overlap(particles[i])
                    if score < personal_best_scores[i]:
                        personal_best_scores[i] = score
                        personal_best[i] = particles[i].copy()
                        
                        if score < global_best_score:
                            global_best = particles[i].copy()
                            global_best_score = score
                            last_improvement = iteration
    
    # Return best solution found
    final_h = global_best
    final_c5 = global_best_score
    
    return (final_h.tolist(), final_c5, n_points)

def compute_overlap(h):
    """Compute the maximum overlap for given h"""
    dx = 2.0 / len(h)
    # Using correlation with reflected h (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
