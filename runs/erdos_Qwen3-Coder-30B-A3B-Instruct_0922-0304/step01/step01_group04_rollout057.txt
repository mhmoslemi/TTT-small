import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function for optimization
    def objective(h):
        h = np.clip(h, 0, 1)  # Clamp to [0, 1]
        h = h * target_sum / np.sum(h)  # Normalize to desired sum
        return evaluate_c5(h, dx)
    
    # Constraint for sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution phase
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence=None: None
    )
    
    if de_result.success:
        best_h = de_result.x
    else:
        best_h = h_init
    
    # Refine with L-BFGS-B
    def objective_lbfgs(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    try:
        # Create bounds for L-BFGS
        bounds_lbfgs = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            objective_lbfgs,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            options={'maxiter': 500}
        )
        
        if result.success:
            best_h = result.x
    except:
        pass
    
    # Final repair
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure final constraints
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass
    
    return (best_h, final_c5, n_points)
