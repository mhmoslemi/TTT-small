import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    def objective(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Define bounds (h[i] ∈ [0, 1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Differential Evolution to explore global minima
    de_result = differential_evolution(
        de_objective,
        bounds,
        seed=seed,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        disp=False
    )
    
    # Refine with L-BFGS-B
    h_refined = de_result.x.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * target_sum / np.sum(h_refined)
    
    # Local optimization
    try:
        result = minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if result.success:
            h_refined = result.x
    except:
        pass
    
    # Final repair
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * target_sum / np.sum(h_refined)
    c5_bound = evaluate_c5(h_refined, dx)
    
    # Validate constraints
    assert np.isclose(np.sum(h_refined), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_refined), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass
        
    return (h_refined, c5_bound, n_points)
