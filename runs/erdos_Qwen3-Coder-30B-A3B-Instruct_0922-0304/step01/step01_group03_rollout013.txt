import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    
    # Normalize to sum to n_points/2
    h = h * (len(h) / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    no_improvement_limit = 1000
    iter_since_improvement = 0
    
    # Initial simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        h_pert = h + simplex_size * np.random.randn(n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (len(h_pert) / 2.0) / np.sum(h_pert)
        simplex.append(h_pert)
    
    def evaluate_simplex(simplex):
        values = []
        for point in simplex:
            value = evaluate_c5(point, dx)
            values.append(value)
        return np.array(values)
    
    def project_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2.0) / np.sum(h)
        return h
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate simplex
        values = evaluate_simplex(simplex)
        indices = np.argsort(values)
        simplex = [simplex[i] for i in indices]
        values = values[indices]
        
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > no_improvement_limit:
            # Perturb best solution slightly
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_pert = best_h.copy()
            h_pert[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h_pert = project_and_normalize(h_pert)
            simplex[-1] = h_pert
            iter_since_improvement = 0
        
        # Calculate centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_reflect = centroid + reflection_factor * (centroid - simplex[-1])
        h_reflect = project_and_normalize(h_reflect)
        c_reflect = evaluate_c5(h_reflect, dx)
        
        if values[0] <= c_reflect < values[-2]:
            simplex[-1] = h_reflect
        elif c_reflect < values[0]:
            # Expansion
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = project_and_normalize(h_expand)
            c_expand = evaluate_c5(h_expand, dx)
            if c_expand < values[0]:
                simplex[-1] = h_expand
            else:
                simplex[-1] = h_reflect
        else:
            # Contraction
            h_contract = centroid + contraction_factor * (simplex[-1] - centroid)
            h_contract = project_and_normalize(h_contract)
            c_contract = evaluate_c5(h_contract, dx)
            if c_contract < values[-1]:
                simplex[-1] = h_contract
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_and_normalize(simplex[i])
        
        # Periodic perturbation
        if iteration % 1000 == 0 and iteration > 0:
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_pert = best_h.copy()
            h_pert[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h_pert = project_and_normalize(h_pert)
            simplex[-1] = h_pert
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
