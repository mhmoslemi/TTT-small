import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = signal.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0  # initial temperature
    alpha = 0.99  # cooling rate
    T_min = 1e-6  # minimum temperature
    max_iter = 100000
    
    # current state
    current_h = h.copy()
    current_C5 = best_C5
    
    # Temperature schedule
    T = T0
    iteration = 0
    
    while T > T_min and iteration < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Perturb the current solution
        new_h = current_h.copy()
        
        # Select random points to modify
        n_modify = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        # Modify these points with small random changes
        for idx in indices:
            # Random change with bounds adjustment
            delta = np.random.normal(0, 0.1)
            new_h[idx] += delta
            
        # Ensure boundaries
        new_h = np.clip(new_h, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2.0
        if current_sum > 0:
            scale_factor = target_sum / current_sum
            new_h = new_h * scale_factor
            
        # Compute new energy
        new_C5 = evaluate_erdos_solution(new_h, dx)
        
        # Metropolis acceptance criterion
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            current_h = new_h
            current_C5 = new_C5
            
            # Update best if better
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cool down
        T *= alpha
        iteration += 1
    
    return (best_h, best_C5, n_points)
