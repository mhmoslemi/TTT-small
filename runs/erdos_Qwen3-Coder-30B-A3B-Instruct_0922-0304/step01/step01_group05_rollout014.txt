import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h with the provided construction
    h = np.array(initial_h_values)
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Fourier domain optimization
    def optimize_fourier():
        # Initialize with current solution
        h0 = h.copy()
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
        # Fourier formulations
        H = fft.fft(h0)
        H_conj = np.conj(H)
        # Generate the frequency domain representation of (1-h)
        one_minus_h = 1 - h0
        one_minus_H = fft.fft(one_minus_h)
        
        # The overlap in frequency domain
        C_freq = H * one_minus_H
        
        # Optimization variables in frequency domain
        H_var = Variable(n_points, complex=True)
        
        # Real and imaginary parts
        H_real = Variable(n_points)
        H_imag = Variable(n_points)
        
        # Constraints
        # H_var = H_real + 1j * H_imag
        constraints = []
        
        # Magnitude constraints
        for i in range(n_points):
            constraints.append(H_var[i] == H_real[i] + 1j * H_imag[i])
            
        # Sum constraint in frequency domain (DC component)
        constraints.append(H_var[0].real == n_points / 2)
        constraints.append(H_var[0].imag == 0)
        
        # Bounds on h values
        h_values = H_real + 1j * H_imag
        h_real = Variable(n_points)
        for i in range(n_points):
            constraints.append(h_real[i] == real(H_var[i]))
        
        # h in [0, 1] constraints
        for i in range(n_points):
            constraints.append(h_real[i] >= 0)
            constraints.append(h_real[i] <= 1)
        
        # Objective: minimize maximum magnitude of overlap
        max_overlap = Variable()
        for i in range(n_points):
            constraints.append(Real(H_var[i] * conj(H_var[i])) <= max_overlap)
        
        # Create and solve problem
        objective = Minimize(max_overlap)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        except:
            return h0
            
        # Extract solution
        h_sol = np.real(h_values.value)
        return np.clip(h_sol, 0, 1)
    
    # Local refinement using gradient descent
    def refine_solution(h):
        best_h = h.copy()
        best_C5 = compute_C5(best_h)
        
        # Try multiple random restarts
        for _ in range(20):
            h_restart = np.random.rand(n_points) * 0.5
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart) 
            h_restart = np.clip(h_restart, 0, 1)
            
            # Gradient-free iterative refinement
            for _ in range(20):
                h_test = h_restart.copy()
                # Small random perturbations
                for i in range(n_points):
                    perturb = np.random.normal(0, 0.05)
                    if 0 <= h_test[i] + perturb <= 1:
                        h_test[i] += perturb

                h_test = h_test * (n_points / 2) / np.sum(h_test)
                C5 = compute_C5(h_test)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_test.copy()
        
        return best_h, best_C5

    # Main optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Two stage approach
    # Stage 1: Fourier domain optimization
    h_fourier = optimize_fourier()
    C5_fourier = compute_C5(h_fourier)
    if C5_fourier < best_C5:
        best_h = h_fourier
        best_C5 = C5_fourier

    # Stage 2: Local refinement with restarts
    h_refined, C5_refined = refine_solution(best_h)
    if C5_refined < best_C5:
        best_h = h_refined
        best_C5 = C5_refined

    return (best_h.tolist(), best_C5, n_points)
