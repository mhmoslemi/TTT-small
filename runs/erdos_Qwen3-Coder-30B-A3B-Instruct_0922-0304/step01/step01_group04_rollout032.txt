import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def repair_h(h, target_sum):
    # Clamp to [0, 1] and renormalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s - 5  # Leave buffer for cleanup
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize initial h if needed
    current_h = repair_h(h_init, target_sum)
    current_c5 = evaluate_c5(current_h, dx)
    
    # DE parameters
    popsize = 40
    mutation = 0.8
    crossover = 0.9
    maxiter = 100
    
    # Define bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization
    def de_objective(h):
        h = repair_h(h, target_sum)
        return evaluate_c5(h, dx)
    
    # Run differential evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        crossover=crossover,
        strategy='best1bin',
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False,
    )
    
    if de_result.success:
        h_de = repair_h(de_result.x, target_sum)
        c5_de = evaluate_c5(h_de, dx)
        if c5_de < current_c5:
            current_h = h_de
            current_c5 = c5_de

    # Local refinement with L-BFGS-B
    def lbfgs_objective(h):
        h = repair_h(h, target_sum)
        return evaluate_c5(h, dx)
    
    # Optimization with constraints
    # Note: L-BFGS-B doesn't naturally support equality constraints,
    # but we refine towards the constraint satisfying solution
    
    # Perform L-BFGS-B optimization starting from our best solution
    # Use bounds and initial point from DE
    try:
        # Use scipy minimize with L-BFGS-B
        result = minimize(
            lbfgs_objective,
            current_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None,
        )
        
        if result.success:
            h_local = repair_h(result.x, target_sum)
            c5_local = evaluate_c5(h_local, dx)
            if c5_local < current_c5:
                current_h = h_local
                current_c5 = c5_local
    except Exception:
        pass

    # Ensure final solution meets criteria
    current_h = repair_h(current_h, target_sum)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Validate
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    # Early return if time limit nears
    elapsed = time.time() - start_time
    if elapsed >= time_limit:
        pass

    return (current_h, current_c5, n_points)
