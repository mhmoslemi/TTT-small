import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper function to ensure bounds and sum constraint
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Objective function for optimization
    def objective(h):
        h = np.abs(h)
        h = repair_h(h)
        return evaluate_c5(h, dx)

    # Differential Evolution with custom bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    best_h = repair_h(de_result.x)
    best_c5 = de_result.fun

    # Local refinement with L-BFGS-B
    def constraint_sum(h):
        return np.sum(h) - target_sum

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        # Optimize with L-BFGS-B using the DE result as initial point
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=constraints,
            options={'maxiter': 500},
            callback=lambda x: None
        )
        
        if result.success:
            final_h = repair_h(result.x)
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    # Final validation
    best_h = repair_h(best_h)
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"

    return (best_h, best_c5, n_points)
