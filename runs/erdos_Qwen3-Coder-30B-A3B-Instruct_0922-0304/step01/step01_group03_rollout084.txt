import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    n_points = 45
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a basic initialization
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2.0 / np.sum(h))
    
    # Add initial solution to population
    population.append(h.copy())
    
    # Generate rest of population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0 / np.sum(h))
        # Ensure values are in [0, 1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    # Evolution parameters
    generations = 2000
    elite_size = pop_size // 10  # Top 10%
    crossover_rate = 0.8
    mutation_rate = 0.05
    mutation_noise = 0.05
    no_improvement_limit = 50
    no_improvement_count = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate all solutions
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Diversification
        if no_improvement_count > no_improvement_limit:
            # Random restart for 20% of population
            for i in range(int(0.2 * pop_size)):
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0 / np.sum(h))
                h = np.clip(h, 0, 1)
                population[-(i+1)] = h
            mutation_noise *= 0.9
            
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection (tournament)
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = single_point_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutate(child1, mutation_rate, mutation_noise)
            mutate(child2, mutation_rate, mutation_noise)
            
            # Add to population
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx

def tournament_selection(population, fitness, k):
    """Tournament selection of k individuals."""
    idx = np.random.choice(len(population), k)
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return population[best_idx].copy()

def single_point_crossover(parent1, parent2):
    """Single-point crossover of two parents."""
    n = len(parent1)
    point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Renormalize
    child1 = child1 * (len(child1) / 2.0 / np.sum(child1))
    child2 = child2 * (len(child2) / 2.0 / np.sum(child2))
    
    # Clip and renormalize
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    child1 = child1 * (len(child1) / 2.0 / np.sum(child1))
    child2 = child2 * (len(child2) / 2.0 / np.sum(child2))
    
    return child1, child2

def mutate(h, rate, noise):
    """Mutate vector h with given rate and noise."""
    for i in range(len(h)):
        if np.random.rand() < rate:
            h[i] += np.random.normal(0, noise)
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Renormalize to sum to n/2
    h = h * (len(h) / 2.0 / np.sum(h))
    
    # Ensure non-negative
    h = np.maximum(h, 0)
