import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

    alpha_initial = 1e-3
    alpha_decay = 0.999

        if time.time() - start_time > budget_s * 0.95:

        # Compute current C5 and dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)  # convert to lag
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i - k >= 0:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i - k]
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        
        # Bias correction
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        # Update h
        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate and track best
        c5 = evaluate_c5(h, 2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Decay learning rate
        alpha *= alpha_decay
        
        # Early stopping
        if t > 100 and abs(c5 - evaluate_c5(best_h, 2.0 / n_points)) < 1e-5:
            break

    return (best_h, best_c5, n_points)
