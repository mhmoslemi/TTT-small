import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Normalize initial guess to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_C = float('inf')
    best_h = h_init.copy()
    
    # Optimizer with CVXPY for convex QCP formulation
    def solve_with_cvxpy(h0, n_points, dx):
        # Variables
        h = Variable(n_points)
        C = Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C >= 0]
        
        # Generate overlap constraints for all possible lags
        # Consider lags from -(n_points-1) to (n_points-1)
        for k in range(-n_points + 1, n_points):
            # Determine valid indices for correlation
            if k >= 0:
                # Positive shift: h[i] * (1 - h[i+k]) for i in range(n_points-k)
                corr_terms = [h[i] * (1 - h[i+k]) for i in range(n_points - k)]
            else:
                # Negative shift: h[i] * (1 - h[i+k]) for i in range(-k, n_points)
                corr_terms = [h[i] * (1 - h[i+k]) for i in range(-k, n_points)]
                
            # Sum over valid terms
            overlap_expr = sum(corr_terms)
            # Scale by dx and add constraint
            constraints.append(overlap_expr <= C * dx)
            
        # Objective
        objective = Minimize(C)
        
        # Create and solve the problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-6, 
                       normalize=True, use_indirect=False, 
                       warm_start=True, max_iter=2000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                return h.value, C.value
        except:
            pass
        return h0, float('inf')
    
    # Initial solve with CVXPY
    h_sol, C_sol = solve_with_cvxpy(h_init, n_points, dx)
    if C_sol < best_C and np.all(h_sol >= 0) and np.all(h_sol <= 1):
        best_C = C_sol
        best_h = h_sol.copy()
    
    # Local refinement with scipy if still time
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        
        def obj(h):
            h = np.array(h)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        # Try different starting points for local search
        for i in range(5):
            if time.time() - start_time >= budget_s - 10:
                break
            h_start = h_sol + np.random.normal(0, 0.01, n_points) * (i+1)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
            try:
                res = minimize(obj, h_start, method='SLSQP', bounds=bounds, 
                               constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    C_refined = obj(h_refined)
                    if C_refined < best_C:
                        best_C = C_refined
                        best_h = h_refined.copy()
            except:
                continue
    
    return (best_h.tolist(), float(best_C), n_points)
