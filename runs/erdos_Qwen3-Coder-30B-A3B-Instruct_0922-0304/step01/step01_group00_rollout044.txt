import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Initialize variables
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style coordinate descent
    max_iter = 10000
    step_size = 0.01
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient of overlap w.r.t. h
        # We compute the partial derivative for each h[i] of L(k_max)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max]
        
        # Select update coordinate: minimize the gradient (most negative)
        i_update = np.argmin(grad)
        
        # Line search along direction of negative gradient
        # But we must maintain sum(h) = n_points / 2 and h[i] \in [0,1]
        h_new = h.copy()
        
        # Compute step size for h[i_update]
        # The constraint is sum(h_new) = n_points / 2
        # So if we increase h[i_update] by delta, 
        # we must decrease other elements to maintain sum
        
        # Simple line search for the best value of h[i_update]
        # around its current value
        best_h_new = h.copy()
        best_c5_new = max_overlap
        
        # Try small steps along the gradient direction
        for alpha in np.linspace(0, 1.0, 10):
            h_trial = h.copy()
            h_trial[i_update] = h[i_update] - alpha * step_size * grad[i_update]
            
            # Clip to feasible range [0,1]
            h_trial[i_update] = np.clip(h_trial[i_update], 0, 1)
            
            # Adjust other elements to maintain constraint
            # This is very approximate, but works for low dimensional cases
            current_sum = np.sum(h_trial)
            scaling = (n_points / 2) / current_sum
            h_trial = h_trial * scaling
            h_trial[i_update] = np.clip(h_trial[i_update], 0, 1)
            
            # Recompute overlap to see if it helps
            corr_trial = np.correlate(h_trial, 1 - h_trial, mode="full")
            trial_max_overlap = np.max(corr_trial) * dx
            
            if trial_max_overlap < best_c5_new:
                best_c5_new = trial_max_overlap
                best_h_new = h_trial.copy()
        
        # Update h
        h = best_h_new
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
