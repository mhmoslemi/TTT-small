import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Use a better starting point: sawtooth like pattern
        h_init = np.array([1.0 - abs(np.sin(i)) for i in range(n_points)])
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Ensure h is within bounds
    h_init = np.clip(h_init, 0, 1)
    
    # Helper function to optimize
    def objective(h):
        h = np.clip(h, 0, 1)  # Ensure within bounds
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Use simulated annealing-like approach with constraint handling
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Apply gradient-based optimization with projection
    def project_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Try different optimization approaches
    methods = ['L-BFGS-B', 'TNC', 'SLSQP']
    
    for method in methods:
        if time.time() - start_time > budget_s * 0.8:
            break
            
        try:
            result = minimize(
                objective,
                current_h,
                method=method,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8},
                callback=lambda x: None,
            )
            
            if result.success:
                new_h = project_and_normalize(result.x)
                new_c5 = evaluate_c5(new_h, dx)
                if new_c5 < current_c5:
                    current_h = new_h
                    current_c5 = new_c5
        except:
            pass
    
    # If time is still available, try local search with perturbations
    if time.time() - start_time < budget_s * 0.9:
        best_h = current_h.copy()
        best_c5 = current_c5
        
        for iter in range(1000):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Small random perturbations
            perturbation = np.random.normal(0, 0.01, n_points)
            new_h = best_h + perturbation
            new_h = project_and_normalize(new_h)
            new_c5 = evaluate_c5(new_h, dx)
            
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                
        current_h = best_h
        current_c5 = best_c5
    
    # Final validation
    current_h = project_and_normalize(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
                
    return (current_h, current_c5, n_points)
