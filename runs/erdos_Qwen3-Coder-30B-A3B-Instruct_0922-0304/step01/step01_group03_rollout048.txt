import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
        
    dx = 2.0 / len(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    max_children = 10
    reset_frequency = 1000
    
    # Store nodes as dictionaries
    nodes = {}
    root_key = tuple(h)
    nodes[root_key] = {
        'h': h.copy(),
        'visited': 0,
        'c5': best_c5,
        'children': []
    }
    
    # Track current best
    current_best = best_h.copy()
    current_best_c5 = best_c5
    
    # Tree-based MCTS loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        if iteration % reset_frequency == 0 and iteration > 0:
            # Reset some nodes randomly to encourage exploration
            keys_to_reset = random.sample(list(nodes.keys()), min(len(nodes)//10, 50))
            for key in keys_to_reset:
                node_h = list(nodes[key]['h'])
                # Mutate by randomizing 5% of elements
                idxs = random.sample(range(len(node_h)), int(len(node_h)*mutation_rate))
                for i in idxs:
                    node_h[i] = np.random.random()  # Random between 0 and 1
                # Renormalize
                node_h = np.array(node_h)
                node_h = node_h * (22.5 / np.sum(node_h))
                nodes[key] = {
                    'h': node_h,
                    'visited': 0,
                    'c5': float('inf'),
                    'children': []
                }
        
        # Selection
        node_key = root_key
        depths = 0
        while True:
            node = nodes[node_key]
            if node['children'] == []:
                break  # Leaf node
            if len(node['children']) < max_children:
                # Expand if at capacity
                break
            
            # UCB selection
            ucb_scores = []
            for child_key in node['children']:
                child = nodes[child_key]
                if child['visited'] == 0:
                    ucb = float('inf')  # Never visited
                else:
                    ucb = child['c5'] + exploration_weight * np.sqrt(np.log(node['visited']) / child['visited'])
                ucb_scores.append(ucb)
            
            if all(score == float('inf') for score in ucb_scores):
                break  # All unvisited
            
            best_child_idx = np.argmax(ucb_scores)
            node_key = node['children'][best_child_idx]
            depths += 1
            if depths > 100:  # Safety check
                break
        
        # Expansion
        current_h = nodes[node_key]['h']
        new_h = current_h.copy()
        mutated_indices = random.sample(range(len(current_h)), int(len(current_h)*mutation_rate))
        for i in mutated_indices:
            new_h[i] = max(0, min(1, new_h[i] + np.random.normal(0, 0.1)))
        
        # Renormalize
        new_h = new_h * (22.5 / np.sum(new_h))
        new_key = tuple(new_h)
        
        # Evaluation
        new_c5 = evaluate_c5(new_h, dx)
        
        # Create new node if doesn't exist
        if new_key not in nodes:
            nodes[new_key] = {
                'h': new_h.copy(),
                'visited': 0,
                'c5': float('inf'),
                'children': []
            }
            
        # Update parent node
        nodes[node_key]['children'].append(new_key)
        
        # Backpropagate
        current_key = node_key
        while True:
            if current_key not in nodes:
                break
            n = nodes[current_key]
            n['visited'] += 1
            if new_c5 < n['c5']:
                n['c5'] = new_c5
                if new_c5 < current_best_c5:
                    current_best = new_h.copy()
                    current_best_c5 = new_c5
            parent_keys = [k for k, v in nodes.items() if current_key in v['children']]
            if not parent_keys:
                break
            # We only propagate parent of first found one
            current_key = parent_keys[0]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
