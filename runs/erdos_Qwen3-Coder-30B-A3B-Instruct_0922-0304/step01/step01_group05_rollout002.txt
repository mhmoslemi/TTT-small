import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    best_C5 = float('inf')
    best_h = None
    
    # Use CVXPY for convex optimization
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Compute all overlap constraints (quadratic)
    # For each shift k, compute the overlap sum
    max_k = n_points - 1
    
    # Generate all constraints
    # Note: For each valid k and i, we have: h[i] * (1 - h[i + k]) 
    for k in range(-max_k, max_k + 1):
        # Determine valid indices for this k
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i in [0, n_points-1-k]
            valid_indices = list(range(n_points - k))
            expr = 0
            for i in valid_indices:
                expr += h[i] * (1 - h[i + k])
        else:
            # h[i] * (1 - h[i+k]) for i in [-k, n_points-1]
            valid_indices = list(range(-k, n_points))
            expr = 0
            for i in valid_indices:
                expr += h[i] * (1 - h[i + k])
        constraints.append(expr <= C * dx)
        
    # Objective: minimize C
    objective = Minimize(C)
    
    # Solve the QCP
    prob = Problem(objective, constraints)
    
    # Solve with SCS
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-6)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_values = h.value
            final_C5 = C.value
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = h_values
    except Exception as e:
        pass
    
    # If no solution from CVXPY, fallback to simple gradient-free method
    if best_h is None:
        # Try random restarts with simple local search
        def compute_C5(h_vals):
            h_vals = np.array(h_vals)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
            
        def local_search(h0, max_iter=200):
            h = np.array(h0)
            h = h * (n_points / 2) / np.sum(h)
            best_val = compute_C5(h)
            best_h = h.copy()
            
            for _ in range(max_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                # Small perturbations
                h_new = h.copy()
                for i in range(n_points):
                    if np.random.rand() < 0.3:  # 30% chance to modify
                        delta = np.random.normal(0, 0.01)
                        h_new[i] = np.clip(h_new[i] + delta, 0, 1)
                
                # Ensure integral constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                new_val = compute_C5(h_new)
                if new_val < best_val:
                    h = h_new
                    best_val = new_val
                    best_h = h.copy()
            
            return best_h, best_val
            
        h_init = initial_h.copy()
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        final_h, final_C5 = local_search(h_init)
        best_h = final_h
        best_C5 = final_C5
    
    # Final correction to make sure sum == n_points/2
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), float(best_C5), n_points)
