import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and h[i] in [0,1]"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def segment_crossover(parent1, parent2):
    """Perform segment-wise crossover between two parents"""
    n_points = len(parent1)
    # Choose random crossover point
    crossover_idx = np.random.randint(1, n_points)
    # Create offspring by combining segments
    child1 = np.concatenate([parent1[:crossover_idx], parent2[crossover_idx:]])
    child2 = np.concatenate([parent2[:crossover_idx], parent1[crossover_idx:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.02):
    """Apply mutation to h"""
    if np.random.random() < mutation_rate:
        # Add Gaussian noise
        h = h + np.random.normal(0, noise_std, len(h))
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
    return h

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    population_size = 50
    generations = 1200
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create a simple binary-like starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Fill population
    while len(population) < population_size:
        # Start with binary-like structures
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_c5(h, 2.0 / n_points)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection (tournament selection)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            if np.random.random() < crossover_rate:
                child1, child2 = segment_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = []
        for h in population:
            fit = evaluate_c5(h, 2.0 / n_points)
            fitness.append(fit)
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Periodic relaxation step
        if gen % relaxation_interval == 0 and gen > 0:
            # Simple gradient descent on the best individual
            h = population[current_best_idx].copy()
            dx = 2.0 / n_points
            step_size = 0.01
            
            for _ in range(50):  # Small number of updates
                corr = correlate(h, 1 - h, mode="full")
                max_k = np.argmax(corr)
                max_val = corr[max_k] * dx
                
                # Simple gradient estimation (finite difference)
                grad = np.zeros_like(h)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    max_val_plus = np.max(corr_plus) * dx
                    grad[i] = (max_val_plus - max_val) / eps
                
                # Gradient step
                h = h - step_size * grad
                h = project_to_feasible(h, n_points)
                
                # Check if it's better
                c5_candidate = evaluate_c5(h, dx)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h.copy()
        
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
