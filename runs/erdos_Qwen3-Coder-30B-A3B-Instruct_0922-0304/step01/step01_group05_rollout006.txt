import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx

    # Second optimization approach: direct constraint handling with warm start
    def objective(h):
        return compute_C5(h)
    
    # Gradient-based optimization with better scaling and warm start
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Custom optimization to avoid local minima
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Try different starting points to avoid local optima
    for i in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation of initial point
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Use less strict method 
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        
        try:
            res = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 500, 'ftol': 1e-6})
            if res.success:
                new_h = res.x
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_C5 = compute_C5(new_h)
                if new_C5 < best_C5:
                    best_h = new_h
                    best_C5 = new_C5
        except:
            pass
    
    # Final fine-tuning with constrained optimization
    if time.time() - start_time < budget_s * 0.95:
        # Use cvxpy for final refinement
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Approximate the max overlap constraint
        max_overlap = cp.Variable()
        # For a more precise constraint handling, we'd need extensive correlation constraints
        # But for now, minimize the overall expression
        
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                refined_h = h_var.value
                refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                final_C5 = compute_C5(refined_h)
                if final_C5 < best_C5:
                    best_h = refined_h
                    best_C5 = final_C5
        except:
            pass
    
    # Final computation of C5
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
