```python
import numpy as np
import time
from cvxpy import *
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 500
    num_particles = 50
    w_max, w_min = 0.9, 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initial h from parent construction
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Repair function to ensure constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (n_points / 2) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # Compute overlap for a given h
    def compute_C5(h):
        h = np.array(h)
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize particles
    particles = []
    velocities = []
    
    for _ in range(num_particles):
        h = h_init + 0.2 * (2 * np.random.rand(n_points) - 1)
        particles.append(repair(h))
        velocities.append(0.1 * np.random.randn(n_points))
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_scores = [compute_C5(h) for h in particles]
    
    # Initialize global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO loop
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (elapsed / budget_s)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair position
            particles[i] = repair(particles[i])
            
            # Evaluate new position
            score = compute_C5(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    last_improvement = iter_count
        
        iter_count += 1
        
        # Perturbation if no improvement for 20 iterations
        if iter_count - last_improvement > 20:
            for i in range(num_particles):
                if np.random.rand() < 0.3:  # 30% chance to perturb
                    particles[i] += 0.05 * (2 * np.random.rand(n_points) - 1)
                    particles[i] = repair(particles[i])
            last_improvement = iter_count
    
    # Final check
    final_score = compute_C5(gbest)
    
    return (gbest.tolist(), final_score, n_points)
```
