    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Repair function to ensure constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (n_points / 2) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # Initialize personal bests
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [compute_overlap(p) for p in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / max_iter)

            r1, r2 = np.random.rand(2)

                            c2 * r2 * (global_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair to feasible region
            particles[i] = repair(particles[i])
            
            # Evaluate
            current_score = compute_overlap(particles[i])
            
            # Update personal best
            if current_score < personal_best_scores[i]:

                    global_best_score = current_score
