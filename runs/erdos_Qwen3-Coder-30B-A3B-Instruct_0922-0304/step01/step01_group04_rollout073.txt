import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Use the simpler gradient-based method but with better initialization and parameter tuning
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Create bounds constraints for scipy
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B with bounds
    method = 'L-BFGS-B'
    options = {'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
    
    try:
        result = minimize(
            objective,
            current_h,
            method=method,
            bounds=bounds,
            options=options,
        )
        
        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = evaluate_c5(current_h, dx)
        else:
            # Fallback to initial construction
            pass
            
    except Exception as e:
        # Fallback to initial construction
        pass

    # Additional local search with random perturbations
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for i in range(50):
        if time.time() - start_time >= budget_s * 0.95:
            break
            
        # Add small Gaussian noise
        noise = np.random.normal(0, 0.02, n_points)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
        
        new_c5 = evaluate_c5(perturbed_h, dx)
        
        if new_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = new_c5

    return (best_h, best_c5, n_points)
