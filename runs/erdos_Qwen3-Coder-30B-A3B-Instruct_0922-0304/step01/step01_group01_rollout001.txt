import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation at all lags
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to exactly n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Find best individual in current generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= stagnation_generations:
            break

        # Selection using tournament
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent = population[winner_idx].copy()
            
            # Crossover with random parent
            parent2_idx = np.random.choice(len(population))
            parent2 = population[parent2_idx]
            child = (parent + parent2) / 2.0
            child = project_to_feasible(child, n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_to_feasible(child, n_points)
                
            new_population.append(child)
        
        population = new_population
