import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum
    def constraint_sum(x):
        return np.sum(x) - target_sum

    # Helper function to optimize
    def objective(h):
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        seed=seed,
        maxiter=100,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6,
        polish=True
    )
    
    # Get the best solution
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    c5_bound = evaluate_c5(best_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass
        
    return (best_h, c5_bound, n_points)
