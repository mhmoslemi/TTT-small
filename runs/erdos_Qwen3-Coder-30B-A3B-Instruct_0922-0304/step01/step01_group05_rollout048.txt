import numpy as np
import time
from scipy.signal import correlate
import random

def evaluate_C5(h, dx):
    """Compute the maximum overlap integral C5 for given h"""
    # Ensure h is properly normalized
    h = np.array(h)
    h = h * (len(h) / 2) / np.sum(h)
    
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    
    # Return max scaled by dx
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.array(h)
    # Clip values
    h = np.clip(h, 0, 1)
    # Normalize sum
    h = h * (n_points / 2) / np.sum(h)
    return h.tolist()

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    n = len(parent1)
    child = []
    for i in range(n):
        if random.random() < 0.5:
            child.append(parent1[i])
        else:
            child.append(parent2[i])
    return child

def mutate(h, mutation_rate=0.1, sigma=0.1):

    h = np.array(h)
    mask = np.random.random(len(h)) < mutation_rate

    # Repair
    return repair_h(h, len(h))

    return population[winner_index]
