import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

    # ADAM optimizer parameters
    alpha = 1e-3

    m = np.zeros(n_points)

    t = 0

        # Compute current C5 and dominant lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        # The lag at which maximum overlap occurs
        lag = max_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradient[i] = 1 - 2 * best_h[i] - 2 * best_h[j]
        
        # Update ADAM
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        # Update h
        best_h = best_h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        best_h = np.clip(best_h, 0, 1)

        if c5_val < best_c5:
            best_c5 = c5_val
            last_improvement = t
        
        # Update learning rate
        alpha *= alpha_decay
        
        t += 1
        
        # Early stopping if no improvement in a while
        if t - last_improvement > 1000:
            break
