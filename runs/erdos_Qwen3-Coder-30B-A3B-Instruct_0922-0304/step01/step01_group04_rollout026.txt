import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Define bounds (0,1) for each variable
    bounds = [(0, 1) for _ in range(n_points)]

    # Define constraint for sum
    def sum_constraint(x):
        return np.sum(x) - target_sum
    
    # Constraint dictionary
    constraints = {'type': 'eq', 'fun': sum_constraint}

    # Evaluate the initial solution first
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Objective function to minimize
    def objective(x):
        return evaluate_c5(x, dx)
    
    # Set DE parameters
    maxiter = 100
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.7
    
    # Initialize best solution
    best_x = current_h.copy()
    best_c5 = current_c5
    
    # Differential evolution with early termination
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            args=(dx,),
            maxiter=maxiter,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            constraints=constraints,
            callback=lambda x, convergence=None: None,
            disp=False,
            tol=1e-8
        )
        
        if de_result.success:
            trial_h = de_result.x
            trial_c5 = de_result.fun
            if trial_c5 < best_c5:
                best_x = trial_h
                best_c5 = trial_c5
                
    except Exception as e:
        pass
    
    # Ensure the best solution meets the sum constraint exactly
    scaled_h = best_x * target_sum / np.sum(best_x)
    clamped_h = np.clip(scaled_h, 0, 1)
    
    # Validate final solution
    final_c5 = evaluate_c5(clamped_h, dx)
    
    # Final check
    final_sum = np.sum(clamped_h)
    assert np.isclose(final_sum, target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in clamped_h), "Bounds constraint violated"
    
    return (clamped_h, final_c5, n_points)
