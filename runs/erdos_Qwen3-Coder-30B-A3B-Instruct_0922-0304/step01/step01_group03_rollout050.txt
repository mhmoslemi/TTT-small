import time
from scipy import signal
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1e-3
    gamma = 0.995
    alpha = 0.1
    alpha_min = 1e-6
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 200
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current overlaps with DFT for speed
        # h_conv = signal.correlate(h, 1-h, mode='full')
        # overlaps = h_conv * dx
        # max_k = np.argmax(overlaps)
        # max_overlap = overlaps[max_k]
        
        # Fast DFT-based computation of overlaps
        n = len(h)
        # pad h and (1-h) to size 2*n -1
        h_padded = np.pad(h, (0, n-1), 'constant')
        h1_padded = np.pad(1-h, (0, n-1), 'constant')
        corr_fft = ifft(fft(h_padded) * np.conj(fft(h1_padded)))
        overlaps = np.real(corr_fft) * dx
        # take only the valid correlations (shift from -n+1 to n-1)
        max_k = np.argmax(overlaps[n-1:])
        max_overlap = overlaps[n-1+max_k]
        
        # Compute approximate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            # compute overlaps for both cases
            h_plus_padded = np.pad(h_plus, (0, n-1), 'constant')
            h_minus_padded = np.pad(h_minus, (0, n-1), 'constant')
            corr_plus_fft = ifft(fft(h_plus_padded) * np.conj(fft(h1_padded)))
            corr_minus_fft = ifft(fft(h_minus_padded) * np.conj(fft(h1_padded)))
            
            overlap_plus = np.real(corr_plus_fft)[n-1+max_k] * dx
            overlap_minus = np.real(corr_minus_fft)[n-1+max_k] * dx
            
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Simulated Annealing Update
        T = T0 * (gamma ** iteration)
        noise = np.random.normal(0, T/2.0, len(h))
        
        h_new = h - alpha * grad + noise
        h_new = np.clip(h_new, 0, 1)
        scale = (len(h) / 2.0) / np.sum(h_new)
        h_new = h_new * scale
        
        new_c5 = evaluate_c5(h_new, dx)
        
        # Metropolis acceptance criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1

        # Adaptive learning rate
        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:
                alpha *= 0.95
            iter_since_improvement = 0
            
        # Periodic reset to binary random configuration
        if iteration % 5000 == 0 and iteration > 0:
            h_new = np.random.rand(n_points) > 0.5

            scale = (len(h_new) / 2.0) / np.sum(h_new)
            h_new = h_new * scale
            h = h_new
            # h = np.random.rand(len(h)) > 0.5

            # h = h * scale
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
