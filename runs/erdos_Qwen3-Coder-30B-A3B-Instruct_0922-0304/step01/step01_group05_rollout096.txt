import numpy as np
import time
from scipy.signal import correlate
import random

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Uniform crossover of two parents"""
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Mutation with Gaussian noise"""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = np.array(h) + noise * mask
    return h

def select_tournament(population, fitnesses, k=5):
    """Tournament selection"""
    selected = []
    for _ in range(len(population) // 2):
        tournament_indices = np.random.choice(len(population), k)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected
