import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturbation_freq = 1000
    max_no_improvement = 1000
    
    # Create initial simplex
    def normalize_h(h_vec):
        scaled = h_vec * (n_points / 2.0) / np.sum(h_vec)
        return np.clip(scaled, 0, 1)
    
    simplex = [h.copy()]
    for _ in range(n_points):
        perturbed = h + simplex_size * (np.random.rand(n_points) - 0.5)
        simplex.append(normalize_h(perturbed))
    
    iter_since_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate simplex
        evaluations = []
        for vertex in simplex:
            eval_c5 = evaluate_c5(vertex, dx)
            evaluations.append(eval_c5)
        
        # Sort vertices by evaluation (ascending)
        sorted_indices = np.argsort(evaluations)
        sorted_simplex = [simplex[i] for i in sorted_indices]
        sorted_evals = [evaluations[i] for i in sorted_indices]
        
        if sorted_evals[0] < best_c5:
            best_h = sorted_simplex[0].copy()
            best_c5 = sorted_evals[0]
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
        
        if iter_since_improvement > max_no_improvement:
            break
        
        # Create new simplex
        best = sorted_simplex[0]
        worst = sorted_simplex[-1]
        second_worst = sorted_simplex[-2]
        
        centroid = np.mean(sorted_simplex[:-1], axis=0)
        
        # Reflection
        reflected = centroid + reflection_factor * (centroid - worst)
        reflected = normalize_h(reflected)
        reflected_c5 = evaluate_c5(reflected, dx)
        
        if sorted_evals[0] <= reflected_c5 < sorted_evals[-2]:
            simplex[-1] = reflected
        elif reflected_c5 < sorted_evals[0]:
            # Expansion
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = normalize_h(expanded)
            expanded_c5 = evaluate_c5(expanded, dx)
            if expanded_c5 < reflected_c5:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            if reflected_c5 < sorted_evals[-1]:
                contracted = centroid + contraction_factor * (worst - centroid)
                contracted = normalize_h(contracted)
                contracted_c5 = evaluate_c5(contracted, dx)
                if contracted_c5 < sorted_evals[-1]:
                    simplex[-1] = contracted
                else:
                    # Shrink
                    for i in range(1, len(simplex)):

            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = best + 0.5 * (simplex[i] - best)
        
        # Perturb best solution if necessary
        if len(simplex) % perturbation_freq == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            perturbed_best = best + perturbation
            perturbed_best = normalize_h(perturbed_best)
            simplex[-1] = perturbed_best
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
