import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Use CVXPY to solve the convex relaxation
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(name="t")
    
    # Constraint that sum(h) = target_sum
    constraints = [cp.sum(h) == target_sum]
    
    # Constraint that h in [0, 1]
    constraints += [h >= 0, h <= 1]
    
    # Constraint that t is an upper bound on all overlaps
    # For each possible shift k from -n_points+1 to n_points-1
    for k in range(-n_points+1, n_points):
        # Compute the overlap integral for this shift
        # This is sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k)%n_points])
        # which equals sum_{i=0}^{n_points-1} h[i] - sum_{i=0}^{n_points-1} h[i] * h[(i+k)%n_points]
        # We write it as: sum(h) - sum(h[i] * h[(i+k)%n_points])
        if k >= 0:
            # h[i] * h[i+k] for i=0..n_points-k-1
            # So h[i] * h[(i+k)%n_points]
            # explictly, h[i] * h[i+k]
            # only valid when i+k < n_points
            # For each i, we want h[i] * h[i+k] to appear in the sum
            # So for the sum of h[i] * (1 - h[i+k]), we can say
            # h[i] * (1 - h[i+k]) = h[i] - h[i] * h[i+k]
            # Note that we are summing over all valid i to get correlation
            
            # So we can construct correlation sums manually
            # Overlap = sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k]) 
            # = sum h - sum_{i=0}^{n_points-k-1} h[i] * h[i+k]
            # i+k < n_points  so i < n_points-k
            corr_sum = 0
            # For i from 0 to n_points-1
            # if i+k >= n_points, ignore (longer shift)
            # if i < n_points-k, it's valid
            # But we can also use negative shifts correctly
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h[i] * h[j]
                
            # overlap is sum(h) - corr_sum
            overlap = target_sum - corr_sum
            constraints.append(overlap <= t)
        else:
            # For negative k, the shift goes backwards
            # h[i] * (1 - h[i+k]) for valid i = 0 to n_points + k - 1
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h[i] * h[j]
                
            # overlap is sum(h) - corr_sum
            overlap = target_sum - corr_sum
            constraints.append(overlap <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with MOSEK
    try:
        # Provide a more permissive time limit for solving to get better results
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-7, reltol=1e-7, abstol=1e-7)
        
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_opt = h.value.copy()
            c5_bound = t.value
        else:
            # Fallback to initial values
            h_opt = h_init.copy()
            c5_bound = np.max(np.correlate(h_opt, 1-h_opt, mode="full")) * dx
            
    except Exception as e:
        # If solvers fail, fall back to initial construction
        h_opt = h_init.copy()
        c5_bound = np.max(np.correlate(h_opt, 1-h_opt, mode="full")) * dx

    # Ensure h_opt is in [0, 1] and sum equals target_sum
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    if current_sum != target_sum:
        h_opt = h_opt * target_sum / current_sum
    c5_bound = np.max(np.correlate(h_opt, 1-h_opt, mode="full")) * dx
    
    # Validate
    assert np.isclose(np.sum(h_opt), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_opt), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (h_opt, c5_bound, n_points)
