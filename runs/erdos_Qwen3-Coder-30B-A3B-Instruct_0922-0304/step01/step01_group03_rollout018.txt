import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to n_points / 2
        h = h * (n_points / 2.0 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimize using convex relaxation
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Create optimization problem using CVXPY
        h_var = cp.Variable(n_points)
        
        # Objective: minimize C
        C = cp.Variable()
        
        # Constraints
        # 0 <= h[i] <= 1
        constraints = [h_var >= 0, h_var <= 1]
        
        # sum(h) = n_points / 2
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Overlap constraints
        # For each lag k, we need to bound the overlap sum
        # This is a linear relaxation of the bilinear terms
        # We can sample only a subset of lags, since only a few are likely critical
        
        # Use a subset of lags to avoid too many constraints
        lag_indices = list(range(0, n_points))  # consider lags 0 to n_points-1
        for k_idx, k in enumerate(lag_indices):
            # Compute the overlap as a linear expression, without the bilinear terms
            # We use a conservative linear bound for this
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                # Use h[i] * (1 - h[j]) <= h[i] - h[i]*h[j]
                # For optimization, we use the linearization of the bilinear term:
                # h[i]*h[j] <= (h[i] + h[j])/2 but this is too loose
                # We instead add constraints for all lag combinations in a sample
                overlap_sum += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_sum <= C)

        # Create and solve the problem
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if h_new is not None:
                    # Renormalize to exact sum
                    scale = (n_points / 2.0) / np.sum(h_new)
                    h_new = h_new * scale
                    h_new = np.clip(h_new, 0, 1)
                    
                    # Evaluate this new solution
                    c5_new = evaluate_c5(h_new, dx)
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        iter_since_improvement = 0
                    else:
                        iter_since_improvement += 1
                        if iter_since_improvement > max_no_improvement:
                            # Restart with a random perturbation
                            h_new = np.random.rand(n_points)
                            h_new = h_new * (n_points / 2.0 / np.sum(h_new))
                            h_new = np.clip(h_new, 0, 1)
                            iter_since_improvement = 0
            else:
                # If infeasible, perturb and continue
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2.0 / np.sum(h_new))
        except Exception:
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2.0 / np.sum(h_new))
        
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
