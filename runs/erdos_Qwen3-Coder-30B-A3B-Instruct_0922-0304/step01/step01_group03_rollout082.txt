import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        population.append(h)
    else:
        # Create random binary-like vector
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))
        population.append(h)
    
    # Fill population with diverse initial solutions
    for i in range(population_size - 1):
        h = np.zeros(n_points)
        # Randomly determine how many 1s to place (around 22-23)
        n_ones = np.random.randint(20, 27)
        indices = np.random.choice(n_points, n_ones, replace=False)
        h[indices] = 1.0
        h = h * (22.5 / np.sum(h))
        population.append(h)
    
    def evaluate_c5(h):
        """Evaluate C5 given h and dx."""
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * dx
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)
    
    # Evolution parameters
    max_generations = 2000
    elitism_rate = 0.1
    mutation_rate = 0.05
    noise_level = 0.05
    no_improvement_limit = 50
    generation_without_improvement = 0
    
    # Main evolutionary loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate population
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness_scores.append(c5)
        
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            generation_without_improvement = 0
        else:
            generation_without_improvement += 1
        
        # Check for diversification trigger
        if generation_without_improvement > no_improvement_limit:
            # Introduce diversity by resetting 20% of population
            num_reset = int(0.2 * population_size)
            for i in range(num_reset):
                h = np.zeros(n_points)
                n_ones = np.random.randint(20, 27)
                indices = np.random.choice(n_points, n_ones, replace=False)
                h[indices] = 1.0
                h = h * (22.5 / np.sum(h))
                population[-(i+1)] = h
            noise_level *= 0.9  # Reduce noise
            generation_without_improvement = 0
        
        # Selection: keep top 50% as parents
        num_parents = population_size // 2
        parents = population[:num_parents]
        
        # Elitism: keep top 10%
        elite_num = int(elitism_rate * population_size)
        new_population = population[:elite_num]
        
        # Crossover and mutation to create children
        while len(new_population) < population_size:
            # Select two random parents
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_indices = np.random.choice(n_points, 
                                               size=int(mutation_rate * n_points), 
                                               replace=False)
            for i in mutation_indices:
                child[i] += np.random.normal(0, noise_level)
            
            # Constraint enforcement (clip and renormalize)
            child = np.clip(child, 0, 1)
            child = child * (22.5 / np.sum(child))
            
            new_population.append(child)
        
        population = new_population
    
    return (best_h.tolist(), best_c5, len(best_h))
