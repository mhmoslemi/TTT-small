import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))
        
    dx = 2.0 / len(h)
    
    # Nelder-Mead parameters
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturb_freq = 1000
    noise_std = 0.01
    max_no_improvement = 1000
    simplex_tolerance = 1e-6
        
    # Initialize simplex
    n_points = len(h)
    simplex = [h.copy()]
    for _ in range(n_points):
        perturbed = h + simplex_size * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        scale = 22.5 / np.sum(perturbed)
        perturbed = perturbed * scale
        simplex.append(perturbed)
    
    # Evaluate initial simplex
    simplex_values = []
    for vertex in simplex:
        val = np.max(signal.correlate(vertex, 1 - vertex, mode='full')) * dx
        simplex_values.append(val)
    
    best_idx = np.argmin(simplex_values)
    best_h = simplex[best_idx].copy()
    best_c5 = simplex_values[best_idx]
    
    iter_since_improvement = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 0.1:
        # Sort simplex
        sorted_indices = np.argsort(simplex_values)
        sorted_simplex = [simplex[i] for i in sorted_indices]
        sorted_values = [simplex_values[i] for i in sorted_indices]
        
        # Check if we should terminate
        if iter_since_improvement > max_no_improvement:
            break
            
        # Compute centroid of best (n_points) points
        centroid = np.mean(sorted_simplex[:-1], axis=0)
        
        # Reflect worst point
        worst = sorted_simplex[-1]
        worst_value = sorted_values[-1]
        
        h_reflected = centroid + reflection_factor * (centroid - worst)
        h_reflected = np.clip(h_reflected, 0, 1)
        scale = 22.5 / np.sum(h_reflected)
        h_reflected = h_reflected * scale
        c5_reflected = np.max(signal.correlate(h_reflected, 1 - h_reflected, mode='full')) * dx
        
        # Check if it's better than the second worst
        if sorted_values[-2] > c5_reflected > sorted_values[0]:
            sorted_simplex[-1] = h_reflected
            sorted_values[-1] = c5_reflected
        elif c5_reflected < sorted_values[0]:
            # Expansion
            h_expanded = centroid + expansion_factor * (h_reflected - centroid)
            h_expanded = np.clip(h_expanded, 0, 1)
            scale = 22.5 / np.sum(h_expanded)
            h_expanded = h_expanded * scale
            c5_expanded = np.max(signal.correlate(h_expanded, 1 - h_expanded, mode='full')) * dx
            
            if c5_expanded < c5_reflected:
                sorted_simplex[-1] = h_expanded
                sorted_values[-1] = c5_expanded
            else:
                sorted_simplex[-1] = h_reflected
                sorted_values[-1] = c5_reflected
        else:
            # Contraction
            h_contracted = centroid + contraction_factor * (worst - centroid)
            h_contracted = np.clip(h_contracted, 0, 1)
            scale = 22.5 / np.sum(h_contracted)
            h_contracted = h_contracted * scale
            c5_contracted = np.max(signal.correlate(h_contracted, 1 - h_contracted, mode='full')) * dx
            
            if c5_contracted < worst_value:
                sorted_simplex[-1] = h_contracted
                sorted_values[-1] = c5_contracted
            else:
                # Shrink
                best = sorted_simplex[0]
                for i in range(1, len(sorted_simplex)):
                    sorted_simplex[i] = best + 0.5 * (sorted_simplex[i] - best)
                    sorted_simplex[i] = np.clip(sorted_simplex[i], 0, 1)
                    scale = 22.5 / np.sum(sorted_simplex[i])
                    sorted_simplex[i] = sorted_simplex[i] * scale
                    sorted_values[i] = np.max(signal.correlate(sorted_simplex[i], 1 - sorted_simplex[i], mode='full')) * dx
        
        # Update best
        min_idx = np.argmin(sorted_values)
        if sorted_values[min_idx] < best_c5:
            best_c5 = sorted_values[min_idx]
            best_h = sorted_simplex[min_idx].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Update simplex
        simplex = sorted_simplex
        simplex_values = sorted_values
        
        # Periodic perturbations to escape local minima
        if iter_since_improvement % perturb_freq == 0 and iter_since_improvement > 0:
            perturb_idx = np.random.choice(len(h), size=int(0.05 * len(h)), replace=False)
            best_h[perturb_idx] += np.random.normal(0, noise_std, len(perturb_idx))
            best_h = np.clip(best_h, 0, 1)
            scale = 22.5 / np.sum(best_h)
            best_h = best_h * scale
            # Replace worst in simplex
            max_idx = np.argmax(simplex_values)
            simplex[max_idx] = best_h.copy()
            simplex_values[max_idx] = np.max(signal.correlate(best_h, 1 - best_h, mode='full')) * dx
        
        # Periodically reset if no improvement
        if iter_since_improvement % (perturb_freq * 2) == 0:
            # Re-initialize simplex from best solution with noise
            simplex = [best_h.copy()]
            for _ in range(n_points):
                perturbed = best_h + simplex_size * np.random.randn(n_points)
                perturbed = np.clip(perturbed, 0, 1)
                scale = 22.5 / np.sum(perturbed)
                perturbed = perturbed * scale
                simplex.append(perturbed)
            
            # Re-evaluate
            simplex_values = []
            for vertex in simplex:
                val = np.max(signal.correlate(vertex, 1 - vertex, mode='full')) * dx
                simplex_values.append(val)
            
            # Update best
            min_idx = np.argmin(simplex_values)  
            best_c5 = simplex_values[min_idx]
            best_h = simplex[min_idx].copy()
            iter_since_improvement = 0
            
        iter_since_improvement += 1
            
    return (best_h.tolist(), best_c5, len(best_h))
