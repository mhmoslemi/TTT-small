def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    
    # ADMM-like optimization with convex subproblems
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Convex optimization to reduce this specific overlap
        h_var = Variable(n_points)
        
        # Objective: minimize max overlap by optimizing h
        # We want to solve min h s.t. (h ⋆ (1-h))[k_max] <= C for some C

        # Simplified approach - update with gradient descent
        step = 0.01
        iterations = 100
        
        h_temp = h.copy()
        for _ in range(iterations):
            # Compute overlap to get gradient direction
            corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
            c5_temp = np.max(corr_temp) * dx
            
            # Compute gradient of overlap w.r.t h via finite difference approximation
            grad = np.zeros_like(h_temp)

            # Update h

        corr = correlate(h, 1 - h, mode="full")

        if current_c5 < best_c5:

            best_h = h.copy()
