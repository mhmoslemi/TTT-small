import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

            best_c5 = c5_current

        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        lag = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradient[i] = 1 - 2 * h[i] - 2 * h[j]
            else:
                gradient[i] = 1 - 2 * h[i]
        
        # Update ADAM
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)

        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        h = np.clip(h, 0, 1)

        # Decay learning rate
        alpha *= alpha_decay
        
    return (best_h, best_c5, n_points)
