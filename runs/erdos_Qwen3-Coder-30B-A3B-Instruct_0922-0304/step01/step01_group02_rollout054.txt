import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create a structured initial function based on Barker sequence ideas
        n_points = 42
        # Create a Barker-like pattern: 1s and 0s with minimal autocorrelation
        h = np.zeros(n_points)
        # Use a pattern that alternates 1s and 0s in a way to reduce overlap
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0
        # normalize so integral is 1
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Helper to normalize h to integral constraint
    def normalize_h(h_vec):
        return h_vec * (n_points / 2.0) / np.sum(h_vec)
    
    # Concentrate on fewer optimizations but use better strategies
    max_iters = 500
    max_time = budget_s * 0.95
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points/2.0
    
    # Try multiple local optimizations with different starting points
    for trial in range(20):
        if time.time() - start_time > max_time:
            break
            
        # Start with different random initializations
        # Supplement with structured initialization
        if trial == 0:
            h_init = h.copy()
        elif trial == 1:
            # Try perturbed version of initial
            h_init = h + np.random.normal(0, 0.05, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = normalize_h(h_init)
        else:
            # Various time-efficient constructions
            h_init = np.random.rand(n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = normalize_h(h_init)
        
        # Try optimization with multiple methods
        methods = ['SLSQP', 'Nelder-Mead']
        for method in methods:
            if time.time() - start_time > max_time:
                break
            try:
                if method == 'SLSQP':
                    result = optimize.minimize(
                        objective, 
                        h_init, 
                        method='SLSQP', 
                        bounds=bounds,
                        constraints={'type': 'eq', 'fun': constraint_func},
                        options={'maxiter': max_iters//2, 'ftol': 1e-8, 'gtol': 1e-6}
                    )
                else:  # Nelder-Mead
                    result = optimize.minimize(
                        objective, 
                        h_init, 
                        method='Nelder-Mead',
                        options={'maxiter': max_iters//2, 'ftol': 1e-8, 'xtol': 1e-6}
                    )
                
                if result.success:
                    h_trial = result.x
                    # ensure it respects bounds and constraint
                    h_trial = np.clip(h_trial, 0, 1)
                    h_trial = normalize_h(h_trial)
                    C5_trial = evaluate_erdos_solution(h_trial, dx)
                    if C5_trial < best_C5:
                        best_C5 = C5_trial
                        best_h = h_trial.copy()
            except:
                continue
                
    # Final refinement on best solution
    try:
        result = optimize.minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final)
            C5_final = evaluate_erdos_solution(h_final, dx)
            if C5_final < best_C5:
                best_C5 = C5_final
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h, best_C5, n_points)
