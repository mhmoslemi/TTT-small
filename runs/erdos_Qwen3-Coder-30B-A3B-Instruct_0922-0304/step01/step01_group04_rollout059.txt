import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper function to project back to feasible set
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Objective function for optimization
    def objective(h):
        h = project_h(h)
        return evaluate_c5(h, dx)

    # Differential evolution for global search
    bounds = [(0, 1) for _ in range(n_points)]
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False,
        polish=False,
    )
    
    best_h = project_h(de_result.x)
    best_c5 = evaluate_c5(best_h, dx)
    
    # Local optimization with L-BFGS-B
    def lbfgs_objective(h):
        h = project_h(h)
        return evaluate_c5(h, dx)
    
    # Use the DE result as starting point for L-BFGS-B
    try:
        result = minimize(
            lbfgs_objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None,
        )
        
        if result.success:
            refined_h = project_h(result.x)
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Ensure final solution meets constraints
    final_h = project_h(best_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Validate constraints
    assert np.isclose(np.sum(final_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in final_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (final_h, final_c5, n_points)
