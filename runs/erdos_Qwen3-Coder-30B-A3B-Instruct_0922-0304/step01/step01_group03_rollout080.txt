import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from provided construction or create a random one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h_init = np.zeros(n_points)
        h_init[:23] = 1.0
        np.random.shuffle(h_init)
        
    dx = 2.0 / len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 2000
    elite_size = pop_size // 10
    crossover_rate = 0.8
    mutation_rate = 0.05
    noise_std = 0.05
    
    # Create initial population
    population = []
    population.append(h_init)
    for _ in range(pop_size - 1):
        h = np.random.rand(len(h_init))
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2.0 / np.sum(h))
        population.append(h)
    
    # Track best solution
    best_fitness = best_c5
    no_improvement_count = 0
    max_no_improvement = 100
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Create new population
        new_population = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection: tournament selection
            parent1 = population[np.random.randint(0, pop_size//2)]
            parent2 = population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, len(parent1) - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(len(child)):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, noise_std)
                        
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (len(child) / 2.0 / np.sum(child))
            
            new_population.append(child)
            
        # Apply diversity restoration if no improvement
        if no_improvement_count > 50:
            for i in range(int(0.2 * pop_size)):
                idx = np.random.randint(elite_size, pop_size)
                h = np.random.rand(len(h_init))
                h = np.clip(h, 0, 1)
                h = h * (len(h) / 2.0 / np.sum(h))
                new_population[idx] = h
            noise_std *= 0.9
            
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_fitness, len(best_h))

def evaluate_c5(h, dx): 
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
