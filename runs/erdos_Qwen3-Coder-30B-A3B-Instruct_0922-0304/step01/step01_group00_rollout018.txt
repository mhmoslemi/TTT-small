import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Initialize optimization parameters
    max_iter = 10000
    line_search_precision = 1e-4
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style coordinate-wise optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # If we've improved, update best
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient for maximum overlap
        # The gradient of L(k_max) wrt h[i] = 1-h[i+k_max] if i+k_max < n
        #                                            -h[i] if k_max >= i
        grad = np.zeros(n_points)
        
        # If we're optimizing with k_max = max_k, compute gradient correctly
        # This is for the overlap integral at shift k_max
        for i in range(n_points):
            if i + max_k < n_points:
                grad[i] += 1 - h[i + max_k]
            if max_k <= i:
                grad[i] -= h[i]
                
        # Choose coordinate to reduce: smallest gradient (i.e. negative gradient)
        i_star = np.argmin(grad)
        
        # Compute step size for line search
        if grad[i_star] >= 0:
            continue  # Already at good point
            
        # Line search along the gradient direction 
        alpha = 1.0
        best_alpha = 0.0
        best_val = max_overlap
        
        # Line search with small steps
        step_size = 0.01
        for _ in range(100):
            new_h = h.copy()
            new_h[i_star] += alpha * step_size
            
            # Check if we need to adjust other coordinate to keep sum constraint
            # We need to move the value to respect both bounds and sum constraint
            if new_h[i_star] < 0:
                new_h[i_star] = 0
            elif new_h[i_star] > 1:
                new_h[i_star] = 1
                
            # Maintain sum constraint
            delta = new_h[i_star] - h[i_star]
            if abs(delta) > 1e-10:
                # Adjust another coordinate in the same direction to preserve sum
                if i_star + max_k < n_points:
                    # Check if we can decrease the other coordinate
                    other_i = i_star + max_k
                    if new_h[other_i] > delta:
                        new_h[other_i] -= delta
                        # Clip
                        new_h[other_i] = np.clip(new_h[other_i], 0, 1)
                    else:
                        delta2 = delta - new_h[other_i]
                        new_h[other_i] = 0
                        # Clip difference to maintain sum
                        if new_h[i_star] + delta2 > 1:
                            new_h[i_star] = 1
                        else:
                            new_h[i_star] += delta2
                        new_h[i_star] = np.clip(new_h[i_star], 0, 1)
                else:
                    # If other_i is out of range, adjust i_star only
                    new_h[i_star] = np.clip(new_h[i_star], 0, 1)
            
            # Enforce sum constraint properly
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
            
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_val:
                best_val = new_c5
                best_alpha = alpha
            alpha -= step_size
            if alpha < 0:
                break
                
        # Apply best update step
        if best_alpha > 0:
            new_h = h.copy()
            new_h[i_star] += best_alpha * step_size
            
            if new_h[i_star] < 0:
                new_h[i_star] = 0
            elif new_h[i_star] > 1:
                new_h[i_star] = 1
                
            # Adjust neighbor to preserve sum constraint
            if i_star + max_k < n_points:
                other_i = i_star + max_k
                delta = new_h[i_star] - h[i_star]
                if new_h[other_i] >= delta:
                    new_h[other_i] -= delta
                else:
                    delta2 = delta - new_h[other_i]
                    new_h[other_i] = 0
                    if new_h[i_star] + delta2 <= 1:
                        new_h[i_star] += delta2
                    else:
                        new_h[i_star] = 1
            # Enforce sum constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
                    
            h = new_h
            
    return (best_h.tolist(), best_c5, n_points)
