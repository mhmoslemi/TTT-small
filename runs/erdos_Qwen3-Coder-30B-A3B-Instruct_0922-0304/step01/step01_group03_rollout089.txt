import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # Set first 23 elements to 1
    
    # Normalize to sum to 22.5 (n_points/2)
    h = h * (22.5 / np.sum(h))
    
    # Define Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 50000
    perturbation_freq = 1000
    noise_level = 0.01
    no_improvement_limit = 1000
    
    # Create simplex
    def create_simplex(h_init, size, n_points):
        simplex = [h_init.copy()]
        for _ in range(n_points):
            h_pert = h_init + size * np.random.rand(n_points) - size/2
            h_pert = np.clip(h_pert, 0, 1)
            scale = 22.5 / np.sum(h_pert)
            h_pert = h_pert * scale
            simplex.append(h_pert)
        return simplex
    
    simplex = create_simplex(h, simplex_size, n_points)
    dx = 2.0 / n_points
    
    def evaluate_c5(h):
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * dx
    
    # Evaluate all simplex points
    values = [evaluate_c5(v) for v in simplex]
    best_idx = np.argmin(values)
    best_h = simplex[best_idx].copy()
    best_value = values[best_idx]
    
    iter_since_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort simplex by value
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Compute centroid excluding the worst point
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflect
        worst = simplex[-1]
        h_reflect = centroid + reflection_factor * (centroid - worst)
        h_reflect = np.clip(h_reflect, 0, 1)
        scale = 22.5 / np.sum(h_reflect)
        h_reflect = h_reflect * scale
        
        value_reflect = evaluate_c5(h_reflect)
        
        if values[0] <= value_reflect < values[-2]:
            # Accept reflection
            simplex[-1] = h_reflect
            values[-1] = value_reflect
        elif value_reflect < values[0]:
            # Expand
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = np.clip(h_expand, 0, 1)
            scale = 22.5 / np.sum(h_expand)
            h_expand = h_expand * scale
            value_expand = evaluate_c5(h_expand)
            if value_expand < value_reflect:
                simplex[-1] = h_expand
                values[-1] = value_expand
            else:
                simplex[-1] = h_reflect
                values[-1] = value_reflect
        else:
            # Contract
            if value_reflect < values[-1]:
                h_contract = centroid + contraction_factor * (h_reflect - centroid)
            else:
                h_contract = centroid + contraction_factor * (worst - centroid)
            h_contract = np.clip(h_contract, 0, 1)
            scale = 22.5 / np.sum(h_contract)
            h_contract = h_contract * scale
            value_contract = evaluate_c5(h_contract)
            if value_contract < values[-1]:
                simplex[-1] = h_contract
                values[-1] = value_contract
            else:
                # Shrink simplex
                best = simplex[0]
                for i in range(1, len(simplex)):
                    simplex[i] = best + 0.5 * (simplex[i] - best)
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    scale = 22.5 / np.sum(simplex[i])
                    simplex[i] = simplex[i] * scale
                values = [evaluate_c5(v) for v in simplex]
        
        if values[0] < best_value:
            best_h = simplex[0].copy()
            best_value = values[0]
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > no_improvement_limit:
            # Perturb best solution to escape local minima
            idx_perturb = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            best_h[idx_perturb] += np.random.normal(0, noise_level, len(idx_perturb))
            best_h = np.clip(best_h, 0, 1)
            scale = 22.5 / np.sum(best_h)
            best_h = best_h * scale
            
            # Replace worst simplex with perturbed version
            simplex[-1] = best_h.copy()
            values[-1] = evaluate_c5(best_h)
            iter_since_improvement = 0
        
        # Periodic perturbation
        if iteration % perturbation_freq == 0 and iteration > 0:
            idx_perturb = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            best_h[idx_perturb] += np.random.normal(0, noise_level, len(idx_perturb))
            best_h = np.clip(best_h, 0, 1)
            scale = 22.5 / np.sum(best_h)
            best_h = best_h * scale
            
            # Replace worst simplex with perturbed best
            simplex[-1] = best_h.copy()
            values[-1] = evaluate_c5(best_h)
    
    return (best_h.tolist(), best_value, n_points)
