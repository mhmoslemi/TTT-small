import numpy as np
import time
import math

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T_min = 1e-8
    max_iters = 100000
    
    # Perturbation parameters
    max_perturb = max(1, n_points // 20)  # at most 5% of points
    
    # Run simulated annealing
    T = T0
    iters = 0
    
    while T > T_min and time.time() - start_time < budget_s * 0.95:
        if iters > max_iters:
            break
            
        # Create perturbation
        h_new = h.copy()
        
        # Select random subset to modify
        n_modify = np.random.randint(1, max_perturb + 1)
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        # Modify values - try to keep sum constraint
        sum_old = np.sum(h_new)
        sum_to_add = 0
        
        for idx in indices:
            # Perturb within bounds
            change = np.random.normal(0, 0.1)
            h_new[idx] += change
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            sum_to_add += h_new[idx] - h[idx]
        
        # Adjust other values to maintain sum constraint
        if sum_old != np.sum(h_new):
            # Distribute the difference to other values
            delta_sum = np.sum(h_new) - sum_old
            to_distribute = delta_sum / (n_points - n_modify)
            for idx in range(n_points):
                if idx not in indices:
                    h_new[idx] -= to_distribute
                    h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on temperature
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Accept with probability based on temperature
            delta = C5_new - best_C5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
        
        T *= alpha
        iters += 1

    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure final constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points/2.0) < 1e-10
    
    return (best_h, final_C5, n_points)
