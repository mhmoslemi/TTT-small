import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1
        np.random.shuffle(h)
        
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 5000
    exploration_weight = 1.0  # UCB exploration weight
    num_mutations_per_node = 5
    reset_frequency = 1000
        
    # Tree representation: dict of nodes with structure:
    # {h_tuple: {'c5': float, 'visits': int, 'children': dict, 'parent': h_tuple}}
    tree = {}
    root_h_tuple = tuple(h)
    tree[root_h_tuple] = {'c5': best_c5, 'visits': 0, 'children': {}, 'parent': None}
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection phase: traverse tree to leaf using UCB
        current_h_tuple = root_h_tuple
        path = [current_h_tuple]
        
        while current_h_tuple in tree and tree[current_h_tuple]['children']:
            children = tree[current_h_tuple]['children']
            best_child = None
            best_ucb = -float('inf')
            
            for child_h_tuple in children:
                child_info = tree[child_h_tuple]
                ucb = child_info['c5'] + exploration_weight * np.sqrt(np.log(tree[current_h_tuple]['visits'] + 1) / (child_info['visits'] + 1))
                if ucb > best_ucb:
                    best_ucb = ucb
                    best_child = child_h_tuple
                    
            current_h_tuple = best_child
            path.append(current_h_tuple)
        
        # Expansion phase: add children to this node
        current_h_array = np.array(current_h_tuple)
        current_node_info = tree[current_h_tuple]
        
        if current_node_info['visits'] == 0:
            # Add children
            for _ in range(num_mutations_per_node):
                # Generate mutated h
                h_mutated = current_h_array.copy()
                num_to_mutate = max(1, int(len(h_mutated) * 0.05))
                indices_to_mutate = np.random.choice(len(h_mutated), num_to_mutate, replace=False)
                for i in indices_to_mutate:
                    h_mutated[i] += np.random.normal(0, 0.1)
                h_mutated = np.clip(h_mutated, 0, 1)
                # Renormalize
                h_mutated = h_mutated * (len(h_mutated)/2 / np.sum(h_mutated))
                h_mutated = np.clip(h_mutated, 0, 1)
                
                h_mutated_tuple = tuple(h_mutated)
                
                # Check if already in tree
                if h_mutated_tuple not in tree:
                    tree[h_mutated_tuple] = {'c5': float('inf'), 'visits': 0, 'children': {}, 'parent': current_h_tuple}
                
                current_node_info['children'][h_mutated_tuple] = h_mutated_tuple
                
            # Simulate: evaluate all children and update best found c5
            c5_values = []
            for child_h_tuple in current_node_info['children']:
                child_h_array = np.array(child_h_tuple)
                c5_val = evaluate_c5(child_h_array, dx)
                tree[child_h_tuple]['c5'] = c5_val
                c5_values.append(c5_val)
                
            # Update node's c5 to best found in expansion
            current_node_info['c5'] = min(c5_values)
            
        # Backpropagation: update all nodes in path
        for h_tuple in path:
            tree[h_tuple]['visits'] += 1
            if h_tuple == root_h_tuple:
                continue
            parent_h_tuple = tree[h_tuple]['parent']
            if parent_h_tuple:
                tree[parent_h_tuple]['children'][h_tuple] = h_tuple  # Ensure link exists
                
        # Update best solution
        if current_node_info['c5'] < best_c5:
            best_c5 = current_node_info['c5']
            best_h = current_h_array.copy()
            
        # Periodic random reset to escape local optima
        if iteration % reset_frequency == 0 and iteration > 0:
            random_reset_count = max(1, int(0.05 * len(tree)))
            reset_indices = random.sample(list(tree.keys()), random_reset_count)
            for h_tuple in reset_indices:
                if h_tuple == root_h_tuple:
                    continue
                new_h = np.random.rand(len(h))
                new_h = np.clip(new_h, 0, 1)
                # Renormalize
                new_h = new_h * (len(new_h)/2 / np.sum(new_h))
                new_h = np.clip(new_h, 0, 1)
                tree[h_tuple] = {'c5': evaluate_c5(new_h, dx), 'visits': 0, 'children': {}, 'parent': tree[h_tuple]['parent']}
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
