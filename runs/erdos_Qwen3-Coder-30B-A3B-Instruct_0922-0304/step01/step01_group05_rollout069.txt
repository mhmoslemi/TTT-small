import numpy as np
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 500
    num_particles = 50
    w_start = 0.9
    w_end = 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initialize particles near initial construction
    h_init = np.array(initial_h_values)
    particles = []
    for _ in range(num_particles):
        h = h_init + 0.2 * (2 * np.random.rand(n_points) - 1)
        particles.append(h)
    
    # Initialize velocities
    velocities = 0.1 * np.random.randn(num_particles, n_points)
    
    # Repair function to ensure constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= (n_points / 2) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # Compute overlap function
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial repair and evaluation
    for i in range(num_particles):
        particles[i] = repair(particles[i])
        
    # Compute initial overlaps
    best_overlaps = [compute_overlap(p) for p in particles]
    best_positions = [p.copy() for p in particles]
    global_best_idx = np.argmin(best_overlaps)
    global_best_overlap = best_overlaps[global_best_idx]
    global_best_position = best_positions[global_best_idx].copy()
    
    # PSO loop
    last_improvement_iter = 0
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s * 0.98:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (best_positions[i] - particles[i])
                           + c2 * r2 * (global_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair position
            particles[i] = repair(particles[i])
            
            # Evaluate new overlap
            current_overlap = compute_overlap(particles[i])
            
            # Update personal best
            if current_overlap < best_overlaps[i]:
                best_overlaps[i] = current_overlap
                best_positions[i] = particles[i].copy()
                
                # Update global best
                if current_overlap < global_best_overlap:
                    global_best_overlap = current_overlap
                    global_best_position = particles[i].copy()
                    last_improvement_iter = iteration
                    
        # Perturbation if no improvement in last 20 iterations
        if iteration - last_improvement_iter > 20:
            for i in range(num_particles):
                if np.random.rand() < 0.3:  # 30% chance to perturb
                    noise = 0.1 * (2 * np.random.rand(n_points) - 1)
                    particles[i] += noise
                    particles[i] = repair(particles[i])
            last_improvement_iter = iteration
    
    # Final check and return best result
    final_h = global_best_position
    final_C5 = global_best_overlap
    
    return (final_h.tolist(), final_C5, n_points)
