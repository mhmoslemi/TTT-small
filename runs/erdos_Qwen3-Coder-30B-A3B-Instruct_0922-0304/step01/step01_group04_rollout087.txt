import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Prepare for convex optimization
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Constraint: sum(h) = target_sum
    sum_constraint = cp.Equality(cp.sum(h), target_sum)
    
    # Constraint: h[i] in [0,1]
    bounds_constraints = [h >= 0, h <= 1]
    
    # Constraints for overlap integral
    # For k from -n_points+1 to n_points-1
    # We shift by k: sum(h[i] * (1 - h[i+k])) <= t
    # This is equivalent to sum(h[i] - h[i]*h[i+k]) <= t
    # We'll scan over valid k ranges for the correlation
    
    max_k = n_points - 1
    constraints = [sum_constraint] + bounds_constraints
    
    # Generate overlap constraints
    # For each shift k from -max_k to max_k, we consider the integral
    # This involves computing sum over all valid i,j pairs where j = i + k
    # Also note that for k positive, valid i range is [0, n_points - k - 1]
    # For k negative, valid i range is [-k, n_points - 1]
    
    # We will compute all valid shifts for correlation, then apply to constraints
    
    # Evaluate cross-correlation manually to extract all valid shifts
    # Ignore zero shift case first, as it's not relevant to the maximum overlap
    
    # Manually enumerate all valid correlations
    for k in range(-max_k, max_k + 1):
        if k == 0:
            continue
        # We'll form constraint for this shift of k:
        # sum_{i} h[i] * (1 - h[i+k]) <= t
        # For positive k: i ranges from 0 to n_points - k - 1
        # For negative k: i ranges from -k to n_points - 1
        if k > 0:
            # sum over i from 0 to n_points - k - 1: h[i] * (1 - h[i+k]) <= t
            expr_left = cp.sum(cp.multiply(h[0:n_points-k], (1.0 - h[k:])))
        else:
            # sum over i from 0 to -k - 1: h[i] * (1 - h[i+k]) <= t
            expr_left = cp.sum(cp.multiply(h[-k:], (1.0 - h[0:-k])))
        constraints.append(expr_left <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Package problem
    prob = cp.Problem(objective, constraints)
    
    # Solve using MOSEK (or similar for real performance)
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
    except Exception:
        # Fallback to another solver
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except Exception:
            # Last resort: use initial solution if all else fails
            current_h = h_init.copy()
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
            return (current_h, current_c5, n_points)
            
    # Get optimized h
    h_opt = h.value
    t_opt = t.value
    
    if h_opt is None or t_opt is None:
        current_h = h_init.copy()
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
        return (current_h, current_c5, n_points)
    
    # Adjust h to match sum constraint exactly
    h_opt = h_opt * target_sum / np.sum(h_opt)
    
    # Check it's in [0, 1]
    h_opt = np.clip(h_opt, 0, 1)
    
    # Final evaluate C5
    current_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    # Return
    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.95:
        pass
    
    return (h_opt, current_c5, n_points)
