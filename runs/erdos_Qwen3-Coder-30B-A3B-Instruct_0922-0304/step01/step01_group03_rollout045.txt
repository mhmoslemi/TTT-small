            # Simple check: if we've done a few mutations, we assume fully expanded
            return len(self.children) > 20

            return self.visits > 1000  # Arbitrary limit to prevent infinite expansion
        
        def ucb_value(self, parent_visits):
            if self.visits == 0:
                return float('inf')
            exploitation = self.c5
            exploration = exploration_weight * np.sqrt(np.log(parent_visits) / self.visits)
            return exploitation + exploration

            self.children.append(child_node)

def evaluate_c5(h, dx):
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx

def mutate_h(h, mutation_rate):
    # Create a copy and mutate a small percentage of elements
    new_h = h.copy()
    n_points = len(h)
    n_mutate = max(1, int(n_points * mutation_rate))
    
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        # Add normal noise
        new_h[i] += np.random.normal(0, 0.1)
        # Clip to [0, 1]
        new_h[i] = np.clip(new_h[i], 0, 1)
    
    # Renormalize to satisfy sum = 22.5
    scale = 22.5 / np.sum(new_h)
    new_h = new_h * scale
    new_h = np.clip(new_h, 0, 1)
    
    return new_h
