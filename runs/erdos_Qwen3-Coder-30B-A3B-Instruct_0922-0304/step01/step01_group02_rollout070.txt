import numpy as np
import time

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    max_iter = 10000
    max_evals = 100000
    eval_count = 0
    
    # Start with the current solution
    current_h = h.copy()
    current_C5 = best_C5
    
    T = T0
    
    # Use the same bounds as before but also ensure integral constraint
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb current_h
        new_h = current_h.copy()
        
        # Select subset of indices to modify
        n_modify = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        # Modify those indices
        for idx in indices:
            # Add small random perturbation
            delta = np.random.normal(0, 0.05) * T
            new_h[idx] += delta
            
        # Ensure bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Adjust to keep sum constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2.0
        adjustment = (target_sum - current_sum) / n_points
        new_h += adjustment
        
        # Keep within bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(new_h, dx)
        eval_count += 1
        
        # Accept condition for simulated annealing
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            current_h = new_h
            current_C5 = new_C5
            
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
        
        # Cooling schedule
        T *= alpha
        
        # Sometimes do larger jumps for good exploration
        if np.random.rand() < 0.05:
            T = min(T * 1.1, T0)
            
        # Time-based check
        if eval_count >= max_evals:
            break
    
    return (best_h, best_C5, n_points)
