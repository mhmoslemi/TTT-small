import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Objective function for optimization
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    def sum_constraint(x):
        return np.sum(x) - target_sum
    
    constraints = {'type': 'eq', 'fun': sum_constraint}
    
    # Differential evolution with timeout control
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Try different DE strategies
    strategies = ['best1bin', 'best2bin', 'rand1bin']
    
    for strategy in strategies:
        if time.time() - start_time >= budget_s * 0.95:
            break
            
        try:
            result = differential_evolution(
                objective,
                bounds,
                strategy=strategy,
                maxiter=50,
                popsize=20,
                tol=1e-6,
                constraints=constraints,
                seed=seed,
                callback=lambda x, convergence: None
            )
            
            if result.success:
                current_h = result.x
                current_c5 = objective(current_h)
                
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
                    
        except Exception:
            pass
            
    # Final refinement with local search if time permits
    if time.time() - start_time < budget_s * 0.8:
        # Use scipy's minimize to refine
        from scipy.optimize import minimize
        
        def objective_refine(h):
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            return evaluate_c5(h, dx)
            
        # Check bounds constraint
        bounds_refine = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            objective_refine,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_refine,
            options={'maxiter': 50}
        )
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * target_sum / np.sum(refined_h)
            refined_c5 = evaluate_c5(refined_h, dx)
            
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, final_c5, n_points)
