import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, norm, sum as cp_sum, maximum, power
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using projected gradient descent with adaptive learning rate
    max_iter = 2500
    step_size = 0.01
    decay_rate = 0.9995
    min_step_size = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Gradient computation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        corr_base = correlate(h, 1 - h, mode="full")
        base_max_k = np.argmax(corr_base)
        base_max_val = corr_base[base_max_k] * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            plus_max_k = np.argmax(corr_plus)
            plus_max_val = corr_plus[plus_max_k] * dx
            grad[i] = (plus_max_val - base_max_val) / eps
            
        # Update h with gradient descent
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        step_size *= decay_rate
        step_size = max(step_size, min_step_size)
        
        # Occasionally reinitialize with a perturbed version if stagnating
        if iteration % 200 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
