import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    # L1-ball projection to ensure sum = n_points/2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2):
    # Segment-wise crossover
    n_points = len(parent1)
    split_point = random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.02):
    h = h.copy()
    if random.random() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(h))
        h += noise
        h = project_feasible(h, len(h))
    return h

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    
    # Population setup
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points)
        population.append(h)
        
    # Fill population
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Early exit check
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        num_parents = population_size // 2
        parents = population[:num_parents]
        
        # Create new population
        new_population = parents.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if random.random() < crossover_rate and len(parents) >= 2:
                p1, p2 = random.sample(parents, 2)
                child1, child2 = crossover(p1, p2)
                child1 = project_feasible(child1, n_points)
                child2 = project_feasible(child2, n_points)
                new_population.append(mutate(child1, mutation_rate))
                if len(new_population) < population_size:
                    new_population.append(mutate(child2, mutation_rate))
            else:
                # Copy from parents
                parent = random.choice(parents)
                new_population.append(mutate(parent, mutation_rate))
        
        # Relaxation step
        if generation % relaxation_interval == 0 and len(parents) > 0:
            # Try to optimize the best solution
            best_individual = parents[0]
            improved = False
            
            # Simple local optimization
            for _ in range(10):
                # Gradient approximation
                h0 = best_individual.copy()
                c5_base = evaluate_c5(h0, 2.0 / n_points)
                step = 1e-3
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h0.copy()
                    h_plus[i] += step
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate_c5(h_plus, 2.0 / n_points)
                    grad[i] = (c5_plus - c5_base) / step
                
                # Update
                h0 -= step * grad
                h0 = project_feasible(h0, n_points)
                if evaluate_c5(h0, 2.0 / n_points) < c5_base:
                    best_individual = h0
                    improved = True
            
            if improved:
                # Replace worst with improved solution
                new_population[-1] = best_individual
        
        population = new_population
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
