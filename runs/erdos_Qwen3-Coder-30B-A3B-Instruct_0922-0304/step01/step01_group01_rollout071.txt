import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones_like(h) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:

        n_points = len(h_init)
    except:
        n_points = 42
        h_init = np.ones(n_points) * 0.5

    # Ensure initial construction meets constraints
    h_init = project_to_feasible(h_init, n_points)
    
    # PSO parameters
    n_particles = 50
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_iterations = 1000
    max_stagnation = 50
    velocity_clamp = (-0.5, 0.5)
    
    # Initialize swarm
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    # Add initial solution as one particle
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    p_best.append(h_init.copy())
    p_best_fitness.append(evaluate_c5(h_init, 2.0 / n_points))
    
    # Fill the rest with random particles
    for _ in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        p_best.append(h.copy())
        p_best_fitness.append(evaluate_c5(h, 2.0 / n_points))
    
    # Global best
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]
    
    # PSO main loop
    start_time = time.time()
    stagnation_counter = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (p_best[i] - particles[i]) + 
                           social * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], velocity_clamp[0], velocity_clamp[1])
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(particles[i], 2.0 / n_points)
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best_fitness[i] = fitness
                p_best[i] = particles[i].copy()
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = fitness
                    stagnation_counter = 0
                else:
                    stagnation_counter += 1
                    
        # Early termination for stagnation
        if stagnation_counter >= max_stagnation:
            break
    
    # Final evaluation of best solution
    best_c5 = g_best_fitness
    best_h = g_best
    
    # Verification
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points / 2) < 1e-10
    
    return (best_h, best_c5, n_points)
