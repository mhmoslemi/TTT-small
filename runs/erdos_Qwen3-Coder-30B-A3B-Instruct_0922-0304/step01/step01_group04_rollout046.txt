import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to compute overlap
    def obj_func(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for the optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Method 1: Try different optimization approaches
    best_h = h_init.copy()
    best_c5 = obj_func(best_h)
    
    # Try local optimization
    method = 'SLSQP'
    options = {'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6}
    
    try:
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        result = minimize(
            obj_func,
            best_h.copy(),
            method=method,
            constraints=constraints,
            options=options
        )
        
        if result.success:
            improved_h = result.x
            improved_h = np.clip(improved_h, 0, 1)
            improved_h = improved_h * target_sum / np.sum(improved_h)
            improved_c5 = evaluate_c5(improved_h, dx)
            
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5
    except:
        pass

    # Try a second pass with different starting points
    # Generate a sawtooth pattern as alternative starting point
    sawtooth = np.array([1.0 - abs(2 * (i / (n_points - 1)) - 1) for i in range(n_points)])
    sawtooth = sawtooth * target_sum / np.sum(sawtooth)
    
    try:
        result = minimize(
            obj_func,
            sawtooth,
            method=method,
            constraints=constraints,
            options=options
        )
        
        if result.success:
            improved_h = result.x
            improved_h = np.clip(improved_h, 0, 1)
            improved_h = improved_h * target_sum / np.sum(improved_h)
            improved_c5 = evaluate_c5(improved_h, dx)
            
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5
    except:
        pass

    # Try a third pass with random perturbations
    for _ in range(5):
        # Create a slightly randomized version
        h_pert = np.random.normal(best_h, 0.1)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        c5_new = evaluate_c5(h_pert, dx)
        
        if c5_new < best_c5:
            best_h = h_pert
            best_c5 = c5_new

        # Check time budget
        if time.time() - start_time > budget_s * 0.9:
            break

    # Final check
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
