import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style coordinate-wise updates
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Compute gradient for maximum overlap shift
        grad = np.zeros(n_points)
        k = k_max - (n_points - 1)
        
        # Gradient computation: ∂L(k)/∂h[i]
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] = 1 - h[i + k]
            if i - k >= 0 and i - k < n_points:
                grad[i] -= h[i - k]
        
        # Select index to decrease (minimize gradient, but prefer small h)
        i_star = np.argmin(grad)
        
        # Perform line search along direction e_{i_star}
        alpha_max = 1.0
        if i_star + k >= 0 and i_star + k < n_points:
            alpha_max = min(alpha_max, h[i_star + k])
        alpha_max = min(alpha_max, h[i_star])
        
        # Small step
        alpha = 0.01 * alpha_max
        
        # Update h
        h_copy = h.copy()
        h_copy[i_star] -= alpha
        if i_star + k >= 0 and i_star + k < n_points:
            h_copy[i_star + k] += alpha
            
        # Project onto constraints
        h_copy = np.clip(h_copy, 0, 1)
        sum_h = np.sum(h_copy)
        h_copy = h_copy * (n_points / 2) / sum_h
        
        h = h_copy

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
