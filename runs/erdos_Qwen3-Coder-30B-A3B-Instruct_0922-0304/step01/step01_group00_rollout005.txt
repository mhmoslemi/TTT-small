import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure it's feasible
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: structured binary
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Simulated annealing setup
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    temp = 0.05  # Initial temperature
    cooling_rate = 0.9998
    min_temp = 1e-6
    max_iter = 100000
    
    for iteration in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make perturbation
        h_new = h.copy()
        
        # Choose perturbation type
        if np.random.rand() < 0.3:
            # Add Gaussian noise to two random points and adjust to keep sum constraint
            i1, i2 = np.random.randint(0, n_points, 2)
            noise = np.random.normal(0, 0.01, 2)
            h_new[i1] += noise[0]
            h_new[i2] += noise[1]
            
            # Adjust to preserve sum
            sum_diff = np.sum(h_new) - n_points / 2
            if abs(sum_diff) > 1e-10:
                # Distribute adjustment to two random points
                j1, j2 = np.random.randint(0, n_points, 2)
                adjustment = sum_diff / 2.0
                h_new[j1] -= adjustment
                h_new[j2] -= adjustment
        else:
            # Flip a random bit
            idx = np.random.randint(0, n_points)
            # Simple approach: flip sign in a probabilistic way
            h_new[idx] = 1 - h_new[idx]
            
        # Clip values to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Decide whether to accept
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temp):
            h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cool down temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Check if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
