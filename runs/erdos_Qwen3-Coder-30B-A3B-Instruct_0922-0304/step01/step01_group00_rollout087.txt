import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project onto sum constraint ∫h = 1
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2):
    # Segment-wise crossover
    n = len(parent1)
    split_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.02):
    mask = np.random.random(len(h)) < mutation_rate
    h = h.copy()
    h[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return h

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Fallback: structured binary construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
        
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Genetic algorithm loop
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Check target
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50%
        new_population = population[:population_size//2]
        
        # Crossover (80%)
        remaining = population_size - len(new_population)
        for _ in range(remaining // 2):
            if time.time() - start_time > budget_s - 1:
                break
            p1, p2 = np.random.choice(len(population)//2), np.random.choice(len(population)//2)
            child1, child2 = crossover(population[p1], population[p2])
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            new_population.extend([child1, child2])
            
        # Mutation (20%)
        for i in range(len(new_population)):
            if time.time() - start_time > budget_s - 1:
                break
            if np.random.random() < 0.2:
                new_population[i] = mutate(new_population[i], 0.2, 0.02)
                new_population[i] = project_to_feasible(new_population[i], n_points)
                
        population = new_population[:population_size]
        
        # Periodic relaxation
        if generation % 10 == 0 and len(population) > 0:
            # Relaxation: improve best solution via gradient-like steps 
            try:
                best_in_pop = population[0]  # Already sorted
                h_current = best_in_pop.copy()
                for _ in range(50):
                    # Simple gradient descent on the overlap
                    corr = correlate(h_current, 1 - h_current, mode="full")
                    current_c5 = np.max(corr) * dx
                    # Approximate gradient
                    grad = np.zeros_like(h_current)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_current.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - current_c5) / eps
                        
                    # Update
                    h_current -= 0.01 * grad
                    h_current = np.clip(h_current, 0, 1)
                    h_current = h_current * (n_points / 2) / np.sum(h_current)
                    
                # Replace worst individual
                worst_idx = np.argmax([evaluate_c5(h, dx) for h in population])
                population[worst_idx] = h_current
            except:
                pass  # Fall back to GA

        generation += 1
    
    return best_h, best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    h, c5, n_points = genetic_algorithm(seed, budget_s)
    return (h.tolist(), c5, n_points)
