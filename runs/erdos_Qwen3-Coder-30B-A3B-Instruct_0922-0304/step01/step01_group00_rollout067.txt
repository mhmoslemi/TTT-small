import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise Frank-Wolfe style updates
    max_iter = 10000
    line_search_precision = 1e-4
    prioritization_factor = 2.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Track best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
            
        # Compute Jacobian of overlap at k_max (gradient w.r.t. h)
        grad = np.zeros(n_points)
        if k_max < n_points:
            # For i < n_points - k_max, the term is h[i](1 - h[i+k])
            # Gradient w.r.t. h[i] is (1 - h[i+k])
            for i in range(n_points - k_max):
                grad[i] += (1 - h[i + k_max])
            # Gradient w.r.t. h[i + k_max] is -h[i]  
            for i in range(n_points - k_max):
                grad[i + k_max] -= h[i]
        else:
            # k_max >= n_points so we only consider the first part of h
            # The term is h[i](1 - h[i+k]) where i + k >= n_points, hence overlap 
            # relies on the final part with shifted values
            for i in range(n_points - (k_max - n_points + 1)):
                grad[i + (k_max - n_points + 1)] -= h[i]
        
        # Find coordinate that gives largest decrease
        i_star = np.argmin(grad)
        
        # Line search to find best step in direction i_star
        h_old = h[i_star]
        h_new = h.copy()
        
        # A simple line search: test a few steps
        best_step = 0.0
        best_h_i_star = h[i_star]
        best_c5_i = max_overlap
        
        for step_size in np.linspace(0, 1, 100):
            candidate_h_i = h[i_star] + step_size * (0 if grad[i_star] >= 0 else -1) 
            candidate_h_i = np.clip(candidate_h_i, 0, 1)
            
            h_new[i_star] = candidate_h_i
            
            # Adjust sum
            sum_diff = np.sum(h_new) - n_points / 2
            if abs(sum_diff) > 1e-10:
                # Adjust another coordinate to compensate (choose the one least influential)
                j = np.argmin(np.abs(grad))
                if j != i_star:
                    adjustment = -sum_diff
                    h_new[j] += adjustment
                    h_new[j] = np.clip(h_new[j], 0, 1)
            
            # Compute new overlap
            overlap_new = correlate(h_new, 1 - h_new, mode="full")
            new_max_overlap = np.max(overlap_new) * dx
            
            if new_max_overlap < best_c5_i:
                best_c5_i = new_max_overlap
                best_h_i_star = candidate_h_i
                best_step = step_size
        
        # Apply best step
        h[i_star] = best_h_i_star
        
        # Adjust other coordinate if needed
        sum_diff = np.sum(h) - n_points / 2
        if abs(sum_diff) > 1e-10:
            j = (i_star + 1) % n_points
            h[j] += -sum_diff
            h[j] = np.clip(h[j], 0, 1)
            
    return (best_h.tolist(), best_c5, n_points)
