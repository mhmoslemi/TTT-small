import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Create a smooth decreasing function as starting point
        h_init = np.linspace(0.9, 0.1, n_points)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure initial h satisfies constraints
    h = h_init.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.9995
    T_min = 1e-5
    max_iterations = 50000
    
    T = T0
    iteration = 0
    
    # Main Annealing Loop
    while iteration < max_iterations and T > T_min:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Create new candidate solution
        h_trial = h.copy()
        
        # Perturb 5% of points
        num_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_trial[i] += delta
            
        # Project to [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize to maintain sum constraint
        h_trial = h_trial * target_sum / np.sum(h_trial)
        
        # Evaluate
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject
        delta = c5_trial - current_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_trial
            current_c5 = c5_trial
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        T *= alpha
        iteration += 1
    
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    assert best_c5 == evaluate_c5(best_h, dx), "C5 computation mismatch"
    
    return (best_h, best_c5, n_points)
