import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    n_simulations = 5000
    max_depth = 1000
    
    class Node:
        def __init__(self, h_config, parent=None, action=None):
            self.h = h_config.copy()
            self.parent = parent
            self.action = action
            self.children = []
            self.visits = 0
            self.value = float('inf')
            self.best_c5 = evaluate_c5(self.h, dx)
            
        def ucb1(self, parent_visits):
            if self.visits == 0:
                return float('inf')
            exploitation = self.value
            exploration = np.sqrt(2 * np.log(parent_visits) / self.visits)
            return exploitation - exploration  # Lower is better, so we subtract exploration
            
        def is_fully_expanded(self):
            return len(self.children) > 0
            
        def is_terminal(self):
            return self.visits > max_depth

    def select_node(node):
        while node.is_fully_expanded() and not node.is_terminal():
            # Select child with best UCB1
            best_child = None
            best_ucb = float('inf')
            for child in node.children:
                ucb = child.ucb1(node.visits)
                if ucb < best_ucb:
                    best_ucb = ucb
                    best_child = child
            if best_child is None:
                break
            node = best_child
        return node
    
    def expand_node(node):
        n_mutations = max(1, int(0.05 * len(node.h)))  # 5% mutations
        new_h_list = []
        for _ in range(10):
            h_copy = node.h.copy()
            indices = np.random.choice(len(h_copy), size=n_mutations, replace=False)
            for i in indices:
                noise = np.random.normal(0, 0.1)
                h_copy[i] += noise
            h_copy = np.clip(h_copy, 0, 1)
            h_copy = h_copy * (22.5 / np.sum(h_copy))  # Renormalize
            new_h_list.append(h_copy)
            
        children = []
        for new_h in new_h_list:
            child = Node(new_h, parent=node, action="mutation")
            children.append(child)
            node.children.append(child)
        return children
    
    def simulate_and_evaluate(node):
        c5 = evaluate_c5(node.h, dx)
        return c5
    
    def backpropagate(node, c5):
        current = node
        while current is not None:
            current.visits += 1
            current.value = min(current.value, c5)
            current = current.parent

    def mcts_search():
        root = Node(h)
        best_c5 = root.best_c5
        best_h = root.h.copy()
        
        for i in range(n_simulations):  
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Selection
            node = select_node(root)
            
            # Expansion
            if not node.is_terminal() and len(node.children) == 0:
                children = expand_node(node)
            else:
                children = node.children
                
            # Simulation
            if children:
                # Pick one at random
                child = random.choice(children)
                c5 = simulate_and_evaluate(child)
                
                # Update best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = child.h.copy()
                
                # Backpropagate
                backpropagate(child, c5)
                
            # Occasionally reset a portion of nodes to prevent local optima
            if i % 2000 == 0:
                # Simple reset: reset root to initial value
                root.h = h.copy()
                root.visits = 0
                root.value = float('inf')
                root.children = []
        
        return best_h, best_c5
    
    # Start MCTS
    final_h, final_c5 = mcts_search()
    
    # Final check
    final_c5 = evaluate_c5(final_h, dx)
    return (final_h.tolist(), final_c5, len(final_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
