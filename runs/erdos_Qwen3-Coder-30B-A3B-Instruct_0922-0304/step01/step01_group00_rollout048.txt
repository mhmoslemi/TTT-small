import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Find the shift that gives maximum overlap
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient approximating the shift-k overlap
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Find update direction by minimizing quadratic approximation
        # We want to minimize: grad^T * d + 0.5 * d^T * H * d
        # where H is diagonal with (1 - 2*h[i])
        H_diag = 1 - 2 * h
        H_diag = np.clip(H_diag, 1e-6, 1e6)  # Avoid near-zero or negative values
        
        # Solve the quadratic subproblem with box constraints
        # d = - H^{-1} * grad (subject to bounds and sum constraint)
        d = np.zeros(n_points)
        for i in range(n_points):
            if H_diag[i] > 0:
                d[i] = - grad[i] / H_diag[i]
        
        # Adapt step size using line search
        step_size = 0.1
        best_d = d.copy()
        best_h_trial = h + step_size * d
        best_h_trial = np.clip(best_h_trial, 0, 1)
        best_h_trial = best_h_trial * (n_points / 2) / np.sum(best_h_trial)
        new_c5 = evaluate_c5(best_h_trial, dx)
        
        # Reduce step size if not better
        if new_c5 > current_c5:
            step_size = 0.05
            best_h_trial = h + step_size * d
            best_h_trial = np.clip(best_h_trial, 0, 1)
            best_h_trial = best_h_trial * (n_points / 2) / np.sum(best_h_trial)
            new_c5 = evaluate_c5(best_h_trial, dx)
            
        # Update h
        h = best_h_trial
        
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
