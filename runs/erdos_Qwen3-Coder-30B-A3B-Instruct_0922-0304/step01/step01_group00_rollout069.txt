import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Set up tracking
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise Frank-Wolfe style optimization
    max_iter = 10000
    line_search_tol = 1e-4
    adaptive_step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient of L(k_max) w.r.t. h
        # This is a simplified version. A full gradient would be computed via finite differences
        # We manually construct the sparsity pattern based on the structure of correlation
        grad = np.zeros(n_points)
        
        # For the k_max shift, compute gradient
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if i < k_max:
                grad[i] -= h[i]
                
        # Find index that minimizes gradient (most negative)
        i_star = np.argmin(grad)
        
        # Line search along direction e_{i_star}
        h_orig = h[i_star]
        step_size = adaptive_step_size
        
        # Find optimal step
        best_step_h = h.copy()
        best_step_c5 = max_overlap
        
        # Try few discrete steps
        steps = np.linspace(0, 1, 20)
        for s in steps:
            h_temp = h.copy()
            h_temp[i_star] = h_orig + s * (1 - h_orig)
            h_temp = np.clip(h_temp, 0, 1)
            # Renormalize
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            # Compute new overlap
            corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
            new_overlap = np.max(corr_temp) * dx
            if new_overlap < best_step_c5:
                best_step_c5 = new_overlap
                best_step_h = h_temp.copy()
                
        # Apply best step
        h = best_step_h
        
        # Adaptive step size decay
        if iteration % 500 == 0 and iteration > 0:
            adaptive_step_size *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)
