import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    swarm = []
    velocities = []

    for i in range(n_particles - len(swarm)):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
    
    # Track personal bests and global best
    p_best = [s.copy() for s in swarm]
    p_best_fitness = [evaluate_c5(s, 2.0 / n_points) for s in swarm]
    
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]

    stagnation_count = 0

            r1, r2 = np.random.random(2)

                 social * r2 * (g_best - h))
            
            # Clamp velocity
            v = np.clip(v, velocity_clamp[0], velocity_clamp[1])
            
            # Update position
            h = h + v
            
            # Project to feasible set
            h = project_to_feasible(h, n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(h, 2.0 / n_points)
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = h.copy()
                p_best_fitness[i] = fitness
