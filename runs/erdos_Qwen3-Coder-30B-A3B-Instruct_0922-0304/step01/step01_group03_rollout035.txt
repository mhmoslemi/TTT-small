import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 45

        h[:23] = 1.0
        np.random.shuffle(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    # Create initial simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        h_temp = h + simplex_size * np.random.randn(n_points)
        h_temp = np.clip(h_temp, 0, 1)
        scale = (n_points / 2) / np.sum(h_temp)
        h_temp = h_temp * scale
        simplex.append(h_temp)
    
    def evaluate_simplex(simplex):
        scores = []
        for h_point in simplex:
            scores.append(evaluate_c5(h_point, dx))
        return np.array(scores)
    
    def project_and_normalize(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        return h * scale
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate simplex
        scores = evaluate_simplex(simplex)
        order = np.argsort(scores)
        simplex = [simplex[i] for i in order]
        scores = scores[order]
        
        # Check if we have a better solution
        if scores[0] < best_c5:
            best_c5 = scores[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Stop if no improvement for a while
        if iter_since_improvement > max_no_improvement:
            break
            
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflect
        h_reflect = centroid + reflection_factor * (centroid - simplex[-1])
        h_reflect = project_and_normalize(h_reflect)
        score_reflect = evaluate_c5(h_reflect, dx)
        
        # Accept reflect if better than worst but worse than best
        if scores[0] <= score_reflect < scores[-2]:
            simplex[-1] = h_reflect
            continue
            
        # Expand if better than best
        if score_reflect < scores[0]:
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = project_and_normalize(h_expand)
            score_expand = evaluate_c5(h_expand, dx)
            if score_expand < score_reflect:
                simplex[-1] = h_expand
            else:
                simplex[-1] = h_reflect
        else:
            # Contract
            if score_reflect < scores[-1]:
                h_contract = centroid + contraction_factor * (simplex[-1] - centroid)
                h_contract = project_and_normalize(h_contract)
                score_contract = evaluate_c5(h_contract, dx)
                if score_contract < scores[-1]:
                    simplex[-1] = h_contract
                else:
                    # Reduce simplex
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                        simplex[i] = project_and_normalize(simplex[i])
            else:
                # Reduce simplex
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_and_normalize(simplex[i])
                    
        # Perturb every 1000 iterations
        if iteration % 1000 == 0 and iteration > 0:
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            scale = (n_points / 2) / np.sum(h_perturb)
            h_perturb = h_perturb * scale
            # Replace worst with perturbed best
            simplex[-1] = h_perturb
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
