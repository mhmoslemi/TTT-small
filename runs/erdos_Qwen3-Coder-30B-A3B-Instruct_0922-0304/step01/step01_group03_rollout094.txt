import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Solve convex relaxation using CVXPY
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C >= 0
        ]
        
        # Add overlap constraints
        # Only consider a subset of lags for efficiency
        lags = list(range(0, n_points)) + list(range(n_points, 2 * n_points - 1))
        lags = lags[::2]  # Every other lag to reduce constraint set
        
        for k in lags:
            # Compute overlap for lag k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(C >= overlap * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                scale = (n_points / 2) / np.sum(h_new)
                h_new = h_new * scale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * ((n_points / 2) / np.sum(h_new))
                
                new_c5 = evaluate_c5(h_new, dx)
                
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    iter_since_improvement = 0
                    best_h = h.copy()
                else:
                    iter_since_improvement += 1
                    if iter_since_improvement > max_no_improvement:
                        # Perturb to escape local minima
                        perturb_idx = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                        h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
                        h = np.clip(h, 0, 1)
                        h = h * ((n_points / 2) / np.sum(h))
                        iter_since_improvement = 0
        except Exception:
            # Fall back to simple heuristic update if CVXPY fails
            h_new = h + np.random.normal(0, 0.001, n_points)
            h_new = np.clip(h_new, 0, 1)
            scale = (n_points / 2) / np.sum(h_new)
            h_new = h_new * scale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * ((n_points / 2) / np.sum(h_new))
            
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
                
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx.""" 
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
