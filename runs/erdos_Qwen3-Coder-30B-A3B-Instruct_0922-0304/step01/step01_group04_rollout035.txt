import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Use convex optimization for improved solution
    try:
        # Create variables
        h = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
            t >= 0
        ]
        
        # Generate all possible shift constraints for overlap
        # Since we have n_points samples over interval [0,2], shifts range from -n_points to n_points
        # But due to the nature of the problem and the length of h, effective shift range is from -n_points to n_points
        # However, we can precompute the valid shifts for efficient constraint generation
        for i in range(2*n_points - 1):
            # i corresponds to a shift from -n_points+1 to n_points-1
            shift = i - (n_points - 1)
            if shift <= 0:
                # Non-negative shift case
                # sum_{j = i}^min(i+k, n_points-1) h[j] * (1 - h[j-shift])
                # But simpler representation using full convolution-like behavior
                # pair up terms where the correlation pair is defined
                corr_expr = 0
                for j in range(max(0, -shift), min(n_points, n_points - shift)):
                    corr_expr += h[j] * (1 - h[j + shift])
            else:
                # Positive shift case
                corr_expr = 0
                for j in range(max(0, shift), min(n_points, n_points - shift)):
                    corr_expr += h[j] * (1 - h[j - shift])
            constraints.append(corr_expr <= t)
        
        # Define the problem
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        # Solve with MOSEK if available, otherwise default solver
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
        except:
            try:
                # This is applicable even if MOSEK not available
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except:
                prob.solve(verbose=False)
        
        # Extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            current_h = h.value
            current_c5 = t.value
            if current_h is None:
                current_h = h_init.copy()
                current_c5 = evaluate_c5(current_h, dx)
        else:
            current_h = h_init.copy()
            current_c5 = evaluate_c5(current_h, dx)
    except Exception as e:
        # Fallback to initial construction if anything fails
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)

    # Final processing - make sure constraints are exactly met
    try:
        # Ensure h is in [0,1] due to numerical errors
        current_h = np.clip(current_h, 0, 1)
        # Renormalize to sum = target_sum
        current_h = current_h * target_sum / np.sum(current_h)
        # Compute final C5
        current_c5 = evaluate_c5(current_h, dx)
        assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
        assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    except:
        # If post-processing fails, return fallback values
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)

    # Time check - update if needed and return
    elapsed = time.time() - start_time
    if elapsed >= budget_s and elapsed < budget_s + 5:
        # If we're just at the edge, continue anyway
        pass

    return (current_h, current_c5, n_points)
