import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1 to get sum around 22.5
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS-based search
    n_iter = 0
    max_iter = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_freq = 1000
    tree_depth = 5
    node_count = 0
    
    # Store tree nodes as dictionary
    nodes = {}
    
    # MCTS main loop
    while n_iter < max_iter and (time.time() - start_time) < budget_s - 0.1:
        # Periodically reset the tree to prevent local_trap
        if n_iter % reset_freq == 0 and n_iter > 0:
            nodes = {}
        
        # Select node using UCB
        node_id = "root"
        node_path = [node_id]
        for _ in range(tree_depth):
            if node_id not in nodes:
                break
            children = nodes[node_id].get('children', {})
            if not children:
                break
            # Calculate UCB values for selection
            ucb_values = []
            for child_id, child_info in children.items():
                visits = child_info['visits']
                if visits == 0:
                    ucb = float('inf')
                else:
                    # UCB formula with exploration bonus
                    exploit = child_info['c5']
                    ucb = exploit + exploration_weight * np.sqrt(np.log(nodes[node_id]['visits']) / visits)
                ucb_values.append(ucb)
            
            if not ucb_values:
                break
            
            # Select child with highest UCB
            best_child_idx = np.argmax(ucb_values)
            child_ids = list(children.keys())
            node_id = child_ids[best_child_idx]
            node_path.append(node_id)
        
        # Expand if not terminal
        if node_id not in nodes:
            nodes[node_id] = {
                'h': h.copy(),
                'c5': best_c5,
                'visits': 0,
                'children': {}
            }
        
        # Generate new h by mutation
        h_prime = h.copy()
        num_mutate = max(1, int(mutation_rate * len(h_prime)))
        indices = np.random.choice(len(h_prime), num_mutate, replace=False)
        for idx in indices:
            noise = np.random.normal(0, 0.1)
            h_prime[idx] = np.clip(h_prime[idx] + noise, 0, 1)
        
        # Renormalize to satisfy constraint sum(h) = n/2
        h_prime = h_prime * (len(h_prime)/2.0 / np.sum(h_prime))
        h_prime = np.clip(h_prime, 0, 1)
        
        # Evaluate new configuration
        c5_new = evaluate_c5(h_prime, dx)
        
        # Create new node for the mutated h
        new_node_id = f"node_{id(h_prime)}"
        nodes[new_node_id] = {
            'h': h_prime.copy(),
            'c5': c5_new,
            'visits': 0,
            'children': {}
        }
        
        # Update parent with this child
        parent_id = node_path[-1] if len(node_path) > 1 else "root"
        if parent_id not in nodes:
            nodes[parent_id] = {'children': {}}
        nodes[parent_id]['children'][new_node_id] = nodes[new_node_id]
        
        # Backpropagate
        for node_id_hist in reversed(node_path):
            if node_id_hist == "root":
                break
            if node_id_hist not in nodes:
                nodes[node_id_hist] = {'visits': 0, 'c5': float('inf')}
            nodes[node_id_hist]['visits'] += 1
            if 'children' in nodes[node_id_hist]:
                min_c5 = min([nodes[child_id]['c5'] for child_id in nodes[node_id_hist]['children']])
                if min_c5 < nodes[node_id_hist]['c5']:
                    nodes[node_id_hist]['c5'] = min_c5
        
        # Update best if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_prime.copy()
        
        # Go to next iteration
        n_iter += 1
        
        # Occasionally refresh the best solution to avoid stagnation
        if n_iter % 100 == 0:
            h = best_h.copy()
        else:
            h = h_prime.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
