import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, k, n_points):
    """Compute subgradient at dominant lag k"""
    gradient = np.zeros(n_points)
    for i in range(n_points):
        if k >= 0:
            # For positive lag, 1-h[i+k] contributes negatively
            if i + k < n_points:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i + k]
            else:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i + k - n_points]
        else:
            # For negative lag, 1-h[i+k] contributes negatively
            if i + k >= 0:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i + k]
            else:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i + k + n_points]
    return gradient
