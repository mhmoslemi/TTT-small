import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: binary step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated annealing
    T = 0.05
    cooling_rate = 0.9998
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try perturbation
        h_new = h.copy()
        
        # 70% chance of swap, 30% chance of noise
        if np.random.rand() < 0.7:
            # Swap operation
            i1, i2 = np.random.randint(0, n_points, 2)
            h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        else:
            # Noise operation
            i1, i2 = np.random.randint(0, n_points, 2)
            change1 = np.random.normal(0, 0.01)
            change2 = np.random.normal(0, 0.01)
            h_new[i1] += change1
            h_new[i2] += change2
            
            # Adjust to maintain sum
            delta_sum = np.sum(h_new) - (n_points / 2)
            h_new[i1] -= delta_sum / 2
            h_new[i2] -= delta_sum / 2
            
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-scale to correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept with probability
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Cool down
        T *= cooling_rate
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
