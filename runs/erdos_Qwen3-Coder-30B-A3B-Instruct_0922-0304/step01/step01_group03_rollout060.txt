import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    max_no_improvement = 1000
    iter_since_improvement = 0
    
    # Initialize simplex
    simplex = np.zeros((n_points + 1, n_points))
    simplex[0] = h.copy()
    for i in range(1, n_points + 1):
        perturb = np.random.normal(0, simplex_size, n_points)
        h_pert = h + perturb
        h_pert = np.clip(h_pert, 0, 1)
        scale = 22.5 / np.sum(h_pert)
        simplex[i] = h_pert * scale
    
    # Evaluate initial simplex
    values = np.array([evaluate_c5(s, dx) for s in simplex])
    best_index = np.argmin(values)
    best_c5 = values[best_index]
    best_h = simplex[best_index].copy()
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Order simplex by values
        indices = np.argsort(values)
        simplex = simplex[indices]
        values = values[indices]
        
        # Check for improvement
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > max_no_improvement:
            # Perturb best solution to escape local minimum
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_pert = best_h.copy()
            h_pert[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h_pert = np.clip(h_pert, 0, 1)
            scale = 22.5 / np.sum(h_pert)
            h_pert = h_pert * scale
            simplex[-1] = h_pert
            values[-1] = evaluate_c5(h_pert, dx)
            continue
            
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        h_reflect = centroid + reflection_factor * (centroid - simplex[-1])
        h_reflect = np.clip(h_reflect, 0, 1)
        scale = 22.5 / np.sum(h_reflect)
        h_reflect = h_reflect * scale
        c_reflect = evaluate_c5(h_reflect, dx)
        
        if values[0] <= c_reflect < values[-2]:
            simplex[-1] = h_reflect
            values[-1] = c_reflect
        elif c_reflect < values[0]:
            # Expansion
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = np.clip(h_expand, 0, 1)
            scale = 22.5 / np.sum(h_expand)
            h_expand = h_expand * scale
            c_expand = evaluate_c5(h_expand, dx)
            if c_expand < c_reflect:
                simplex[-1] = h_expand
                values[-1] = c_expand
            else:
                simplex[-1] = h_reflect
                values[-1] = c_reflect
        else:
            # Contraction
            h_contract = centroid + contraction_factor * (simplex[-1] - centroid)
            h_contract = np.clip(h_contract, 0, 1)
            scale = 22.5 / np.sum(h_contract)
            h_contract = h_contract * scale
            c_contract = evaluate_c5(h_contract, dx)
            if c_contract < values[-1]:
                simplex[-1] = h_contract
                values[-1] = c_contract
            else:
                # Shrink simplex
                best = simplex[0]
                simplex[1:] = best + 0.5 * (simplex[1:] - best)
                for i in range(1, n_points + 1):
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    scale = 22.5 / np.sum(simplex[i])
                    simplex[i] = simplex[i] * scale
                values[1:] = [evaluate_c5(s, dx) for s in simplex[1:]]
                
        # Periodic perturbation
        if iteration % 1000 == 0 and iteration > 0:
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_pert = simplex[0].copy()
            h_pert[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h_pert = np.clip(h_pert, 0, 1)
            scale = 22.5 / np.sum(h_pert)
            h_pert = h_pert * scale
            simplex[-1] = h_pert
            values[-1] = evaluate_c5(h_pert, dx)
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
