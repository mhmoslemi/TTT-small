import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T = 0.05
    cooling_factor = 0.9998
    max_iter = 100000
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Iteration counter for progress tracking
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Perturb the current solution
        h_new = h.copy()
        
        # Choose perturbation type
        if np.random.random() < 0.3:
            # Add Gaussian noise to two random points
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            noise = np.random.normal(0, 0.01, 2)
            h_new[i1] += noise[0]
            h_new[i2] += noise[1]
            
            # Ensure the sum constraint is maintained
            total_change = np.sum(h_new) - np.sum(h)
            if total_change != 0:
                # Adjust elements to maintain sum(n_points/2)
                adjustments = -total_change / n_points
                h_new += adjustments
        else:
            # Random bit flip (binary-like perturbation)
            i = np.random.randint(0, n_points)
            h_new[i] = 1 - h_new[i]
        
        # Clip to [0, 1] range
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain the integral constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new configuration
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept with simulated annealing probability
        if c5_new < best_c5 or np.random.random() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Update temperature
        T *= cooling_factor
        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
