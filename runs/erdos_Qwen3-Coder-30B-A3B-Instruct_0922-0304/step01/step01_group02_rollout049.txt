import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from seed if provided

        n_points = len(h)
    else:

        n_points = 42
        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    max_iter = 10000
    max_evals = 100000
    eval_count = 0
    
    # Perturbation parameters
    min_perturb = max(1, n_points // 20)
    max_perturb = max(5, n_points // 5)
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Create perturbed solution
        h_new = h.copy()
        
        # Determine number of points to perturb
        num_perturb = np.random.randint(min_perturb, min(max_perturb, n_points) + 1)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Perturb selected points
        for idx in indices:
            # Random change with bounded magnitude
            delta = np.random.normal(0, 0.1 * T)
            h_new[idx] += delta
            # Ensure bounds
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Adjust to satisfy sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        adjustment = (target_sum - current_sum) / n_points
        
        # Uniformly adjust all points to meet sum constraint
        h_new += adjustment
        # Ensure all points remain in valid range
        h_new = np.clip(h_new, 0, 1)
        
        # Accept or reject based on simulated annealing
        C5_new = evaluate_erdos_solution(h_new, dx)
        eval_count += 1
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
            
        # Accept probabilistically if worse (with simulated annealing)
        if C5_new > best_C5 and np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new.copy()
        else:
            h = h_new.copy()
        
        # Cool temperature
        T *= alpha
        
        # Periodically check time limit
        if eval_count >= max_evals:
            break
    
    return (best_h, best_C5, n_points)
