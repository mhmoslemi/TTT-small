import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.linspace(0.9, 0.1, n_points)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Initialize with the best solution
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-5
    alpha = 0.9995
    max_iter = 5000
    iter_count = 0
    
    # Annealing loop
    while iter_count < max_iter and T > T_min:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create perturbed solution
        h_trial = current_h.copy()
        
        # Select indices to modify (5% of points)
        n_modify = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        # Perturb selected points
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_trial[i] += delta
            
        # Project to [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize to maintain sum constraint
        current_sum = np.sum(h_trial)
        if abs(current_sum - target_sum) > 1e-6:
            h_trial = h_trial * target_sum / current_sum
            
        # Evaluate
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject
        delta = c5_trial - current_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h = h_trial
            current_c5 = c5_trial
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Update temperature
        T *= alpha
        iter_count += 1
    
    # Final validation
    final_sum = np.sum(best_h)
    assert np.isclose(final_sum, target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
