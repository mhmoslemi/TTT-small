def tournament_selection(population, fitness, tournament_size):
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return best_idx

def uniform_crossover(parent1, parent2):
    mask = np.random.random(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def gaussian_mutation(individual, rate):
    mutation_mask = np.random.random(len(individual)) < rate
    noise = np.random.normal(0, 0.1, len(individual))
    mutated = individual + noise * mutation_mask
    return mutated
