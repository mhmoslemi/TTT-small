import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Fallback construction
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial guess
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Time tracking
    elapsed = time.time() - start_time
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Simplified optimization with a warm start from initial guess
    # Using a hybrid approach with gradient methods and constraint handling
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try to optimize the solution with direct minimization
    try:
        # First, try a few steps of gradient descent as a warm start
        h_current = h_init.copy()
        
        # Stochastic gradient descent with adaptive step size
        best_h = h_current.copy()
        best_c5 = compute_C5(h_current)
        
        # Run optimization for most of the time budget
        remaining_time = budget_s - elapsed
        steps = 0
        
        # Gradient approximated using finite differences and optimization
        for iteration in range(5000):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try optimizer
            res = minimize(objective, h_current, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 100})
            
            if res.success:
                h_current = res.x
            else:
                # Fallback to projected gradient
                h_current = h_current - 0.001 * (h_current - np.clip(h_current, 0, 1))
                h_current = np.clip(h_current, 0, 1)
            
            # Renormalize
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Evaluate
            c5_val = objective(h_current)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_current.copy()
            
            steps += 1
        
        # Return best solution found
        return (best_h.tolist(), best_c5, n_points)
        
    except Exception as e:
        # Fallback to initial solution 
        return (h_init.tolist(), compute_C5(h_init), n_points)
