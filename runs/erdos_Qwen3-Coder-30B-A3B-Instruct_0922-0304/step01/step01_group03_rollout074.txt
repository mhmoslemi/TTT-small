import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iter = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_frequency = 1000
    
    # MCTS Tree structure
    class Node:
        def __init__(self, h_config, parent=None, action=None):
            self.h = h_config.copy()
            self.parent = parent
            self.action = action
            self.children = []
            self.visits = 0
            self.c5_value = float('inf')
            self.best_c5 = float('inf')
            self.expanded = False
        
        def ucb_value(self, parent_visits):
            if self.visits == 0:
                return float('inf')
            exploitation = self.best_c5
            exploration = exploration_weight * np.sqrt(np.log(parent_visits) / self.visits)
            return exploitation - exploration  # We minimize C5, so subtract exploration
        
        def add_child(self, child_node):
            self.children.append(child_node)
        
        def update(self, c5_value):
            self.visits += 1
            self.c5_value = c5_value
            if c5_value < self.best_c5:
                self.best_c5 = c5_value

    root = Node(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # MCTS selection
        current_node = root
        while current_node.expanded and current_node.children:
            best_ucb = -float('inf')
            best_child = None
            for child in current_node.children:
                ucb_val = child.ucb_value(current_node.visits)
                if ucb_val > best_ucb:
                    best_ucb = ucb_val
                    best_child = child
            if best_child:
                current_node = best_child
            else:
                break
        
        # Expand node if not already expanded
        if not current_node.expanded:
            # Generate new configurations
            new_configs = []
            for _ in range(10):
                h_mutation = current_node.h.copy()
                indices = np.random.choice(len(h_mutation), 
                                          size=int(mutation_rate * len(h_mutation)), 
                                          replace=False)
                for idx in indices:
                    h_mutation[idx] += np.random.normal(0, 0.1)
                h_mutation = np.clip(h_mutation, 0, 1)
                sum_h = np.sum(h_mutation)
                if sum_h > 0:
                    h_mutation = h_mutation * (22.5 / sum_h)
                new_configs.append(h_mutation)
            
            # Evaluate new configurations
            best_new_config = None
            best_new_c5 = float('inf')
            for config in new_configs:
                c5_val = evaluate_c5(config, dx)
                if c5_val < best_new_c5:
                    best_new_c5 = c5_val
                    best_new_config = config
            
            if best_new_config is not None:
                child_node = Node(best_new_config, parent=current_node, action='mutation')
                child_node.update(best_new_c5)
                current_node.add_child(child_node)
                current_node.expanded = True
                
                # Update best if necessary
                if best_new_c5 < best_c5:
                    best_c5 = best_new_c5
                    best_h = best_new_config.copy()
            
            # Update parent
            current_node.update(best_new_c5)
            current_node.visits += 1
        else:
            # Select child based on UCB
            best_child = None
            best_ucb = -float('inf')
            for child in current_node.children:
                ucb_val = child.ucb_value(current_node.visits)
                if ucb_val > best_ucb:
                    best_ucb = ucb_val
                    best_child = child
            
            if best_child:
                current_node = best_child
                # Update best if necessary
                if current_node.best_c5 < best_c5:
                    best_c5 = current_node.best_c5
                    best_h = current_node.h.copy()

        # Periodic reset
        if iteration % reset_frequency == 0 and iteration > 0:
            # Reset some nodes with random configurations
            reset_count = 50
            for _ in range(reset_count):
                # Random node selection from tree
                node = root
                while node.children and np.random.rand() > 0.2:
                    node = np.random.choice(node.children)
                
                if node != root:
                    h_reset = np.random.rand(len(node.h))
                    h_reset = np.clip(h_reset, 0, 1)
                    h_reset = h_reset * (22.5 / np.sum(h_reset))
                    node.h = h_reset
                    node.c5_value = evaluate_c5(h_reset, dx)
                    node.best_c5 = node.c5_value
                    node.visits = 0
                    node.children = []
                    node.expanded = False
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (
        best_h.tolist(),
        final_c5,
        len(best_h)
    )

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
