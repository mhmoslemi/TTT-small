import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper function to optimize
    def objective(h):
        # Ensure h is in [0,1] and sum to target_sum
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint for sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Constraint for bounds
    def constraint_bounds(h):
        return np.minimum(h, 1 - h)
    
    # Differential Evolution for global search
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use DE for global search
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    if de_result.success:
        best_h = de_result.x
    else:
        best_h = h_init.copy()
    
    # Repair to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    current_c5 = objective(best_h)
    
    # Refine with L-BFGS-B
    def objective_for_lbfgsb(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # For L-BFGS-B, we use the variable as-is, but we make sure we meet the constraints.
    # Use a sufficient but not necessarily tight constraint.
    bounds_lbfgsb = [(0, 1) for _ in range(n_points)]
    
    # Optimize with L-BFGS-B using best solution from DE as initial
    try:
        lbfgsb_result = minimize(
            objective_for_lbfgsb,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_lbfgsb,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        
        if lbfgsb_result.success:
            refined_h = lbfgsb_result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * target_sum / np.sum(refined_h)
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < current_c5:
                current_h = refined_h
                current_c5 = refined_c5
            else:
                current_h = best_h
        else:
            current_h = best_h
    except Exception as e:
        current_h = best_h
    
    # Final validation
    if not np.isclose(np.sum(current_h), target_sum, rtol=1e-4):
        current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Final check
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
