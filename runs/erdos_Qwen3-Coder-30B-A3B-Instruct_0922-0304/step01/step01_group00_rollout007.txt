import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated annealing parameters
    temperature = 0.05
    cooling_rate = 0.9998
    max_iter = 100000
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Main simulated annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a perturbed version of h
        h_new = h.copy()
        
        # Choose perturbation type (70% swap, 30% noise)
        if np.random.random() < 0.7:
            # Swap operation: pick two random indices and swap their values
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Noise operation: add noise to two random points and adjust
            # to maintain sum constraint
            i, j = np.random.choice(n_points, 2, replace=False)
            noise_i = np.random.normal(0, 0.01)
            noise_j = np.random.normal(0, 0.01)
            h_new[i] += noise_i
            h_new[j] += noise_j
            
            # Adjust to preserve sum
            delta_sum = np.sum(h_new) - (n_points / 2)
            # Distribute correction evenly
            adjustment = -delta_sum / 2
            h_new[i] -= adjustment
            h_new[j] -= adjustment
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject the new solution
        if c5_new < best_c5 or np.random.random() < np.exp((best_c5 - c5_new) / temperature):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        temperature *= cooling_rate
        
        # Early exit if target is met
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
