import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Structured binary fallback
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        
    # Annealing parameters
    T = 0.05
    cooling_rate = 0.9998
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Try different perturbations
        if np.random.rand() < 0.7:
            # Binary swap: flip one bit
            i = np.random.randint(0, n_points)
            h_new = h.copy()
            h_new[i] = 1 - h_new[i]
        else:
            # Add noise to two points
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            noise = np.random.normal(0, 0.01, 2)
            h_new[i1] += noise[0]
            h_new[i2] += noise[1]
            
            # Adjust to preserve sum
            total_change = np.sum(h_new) - np.sum(h)
            h_new[i1] -= total_change / 2
            h_new[i2] -= total_change / 2
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        # Update temperature
        T *= cooling_rate
        
        iter_count += 1
        
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
