import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Prepare for convex optimization
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]
    
    # Add overlap constraints for all possible shifts
    max_k = n_points - 1
    for k in range(-max_k, max_k + 1):
        # Compute the sum h[i] * (1 - h[i + k]) for valid indices
        expr = 0
        if k >= 0:
            # i from 0 to n_points - k - 1
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i + k])
        else:
            # i from -k to n_points - 1
            for i in range(-k, n_points):
                expr += h[i] * (1 - h[i + k])
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve with MOSEK
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-8, abstol=1e-8)
        current_h = h.value
        if current_h is not None:
            current_c5 = t.value
        else:
            # If problem is infeasible or no solution, fallback to initial
            current_h = h_init.copy()
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
    except Exception as e:
        # Fallback to initial construction
        current_h = h_init.copy()
        current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
    
    # Ensure we still respect bounds and constraints
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full")) * dx
    
    # Validate solution
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
