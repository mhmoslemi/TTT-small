import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iterations = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: binary pattern with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise Frank-Wolfe style optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_val = overlap[k_max] * dx
        
        # Print occasional progress
        if iteration % 1000 == 0:
            print(f"Iteration {iteration}, C5 = {max_val:.6f}")
            
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient of L(k_max) w.r.t. h[i]
        grad = np.zeros(n_points)
        # Gradient computation: for each i, compute how much changing h[i] affects overlap[k_max]
        # L[k_max] = sum h[i] * (1 - h[i+k_max]) 
        # dL/dh[i] = 1 - h[i+k_max] if i+k_max < n_points
        #           -h[i] if i < k_max
        #           0 otherwise

        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = 1 - h[i + k_max]
            if i < k_max:
                grad[i] -= h[i]
                
        # Find coordinate that decreases overlap most
        i_star = np.argmin(grad)
        
        # Perform line search to update h[i_star]
        h_old = h[i_star]
        step_size = 0.01
        
        # Try different step sizes to find best improvement
        best_h_test = h.copy()
        best_change = 0
        best_step = 0
        
        for step in np.linspace(0, 1, 100):
            h_temp = h.copy()
            h_new = h_old + step * (1 - 2 * h_old)  # small step direction: toward 0 or 1
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust other element to maintain sum
            delta_h = h_new - h_old
            h_temp[i_star] = h_new
            
            # Adjust a compensation term to keep sum constant
            # We update only one element, so we must offset by -delta_h to maintain sum
            idx_compensate = (i_star + 1) % n_points
            h_temp[idx_compensate] = np.clip(h_temp[idx_compensate] - delta_h, 0, 1)
            
            # Recompute overlap
            overlap_test = correlate(h_temp, 1 - h_temp, mode="full")
            max_val_test = np.max(overlap_test) * dx
            
            if max_val_test < max_val - best_change:
                best_change = max_val - max_val_test
                best_step = step
                best_h_test = h_temp.copy()
                
        # Apply best step if improvement
        if best_change > 0:
            h = best_h_test.copy()
        else:
            # Try small perturbation to escape local minima
            h_pert = h.copy()
            h_pert[i_star] = np.clip(h[i_star] + np.random.normal(0, 0.01), 0, 1)
            # Compensate
            sum_diff = np.sum(h_pert) - (n_points / 2)
            if abs(sum_diff) > 1e-5:
                idx_compensate = (i_star + 1) % n_points
                h_pert[idx_compensate] = np.clip(h_pert[idx_compensate] - sum_diff, 0, 1)
            h = h_pert
            
        # Ensure feasible sum
        h = h * (n_points / 2) / np.sum(h)
        
    return (best_h.tolist(), best_c5, n_points)
