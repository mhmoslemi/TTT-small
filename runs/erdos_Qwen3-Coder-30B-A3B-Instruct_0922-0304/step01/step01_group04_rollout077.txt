import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    # Constraint dictionary for differential evolution
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize
    def objective(x):
        # Ensure values are within bounds
        x = np.clip(x, 0, 1)
        return evaluate_c5(x, dx)
    
    # Track the best solution
    best_solution = h_init.copy()
    best_c5 = evaluate_c5(best_solution, dx)
    
    # Differential Evolution parameters
    maxiter = 100
    popsize = 15
    mutation = 0.8
    recombination = 0.7
    
    # Perform the optimization
    try:
        result = differential_evolution(
            objective,
            bounds,
            args=(),
            maxiter=maxiter,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            constraints=constraints,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        if result.success:
            best_solution = result.x
            best_c5 = result.fun
            
        # Further optimization with local search if not enough time
        elapsed = time.time() - start_time
        if elapsed < budget_s * 0.8:
            # Simple local search: gradient ascent in feasible region
            current_h = best_solution.copy()
            current_c5 = evaluate_c5(current_h, dx)
            
            for i in range(100):
                # Try small perturbations
                h_trial = current_h.copy()
                idx = np.random.randint(0, n_points)
                pert = np.random.normal(0, 0.01)
                h_trial[idx] = np.clip(h_trial[idx] + pert, 0, 1)
                h_trial = h_trial * target_sum / np.sum(h_trial)
                
                c5_trial = evaluate_c5(h_trial, dx)
                if c5_trial < current_c5:
                    current_h = h_trial
                    current_c5 = c5_trial
                    
            if current_c5 < best_c5:
                best_solution = current_h
                best_c5 = current_c5
            
    except Exception as e:
        # Fallback to initial construction
        pass

    # Final sanity checks
    best_solution = np.clip(best_solution, 0, 1)
    best_solution = best_solution * target_sum / np.sum(best_solution)
    assert np.isclose(np.sum(best_solution), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_solution), "Bounds constraint violated"
    final_c5 = evaluate_c5(best_solution, dx)

    return (best_solution, final_c5, n_points)
