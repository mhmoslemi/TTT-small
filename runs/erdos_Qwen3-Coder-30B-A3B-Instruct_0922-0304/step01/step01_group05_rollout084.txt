import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try to use initial construction if available
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        # Fallback: create a simple construction
        n_points = 42
        initial_h = np.ones(n_points) * 0.5
    
    # Starting point
    h_init = initial_h.copy()
    dx = 2.0 / n_points
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Optimized version with better constraints handling
    def objective(h):
        return compute_C5(h)
    
    # Use a more robust optimization method
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Try multiple initializations to escape local minima
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Try different algorithms and reuse best result
    algorithms = ['SLSQP', 'L-BFGS-B']
    
    for i in range(3):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Slightly perturb initial guess to get new start
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-6})
            
            if res.success:
                h_trial = res.x
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                C5_trial = compute_C5(h_trial)
                if C5_trial < best_C5:
                    best_h = h_trial
                    best_C5 = C5_trial
        except:
            continue
            
    # Post-process to provide smoother curve and ensure bounds
    best_h = np.clip(best_h, 0, 1)
    
    # Final normalization
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Re-check final value
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
