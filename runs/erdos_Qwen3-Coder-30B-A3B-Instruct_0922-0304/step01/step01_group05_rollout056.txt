import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    try:
        initial_h = np.array(initial_h_values)
    except NameError:
        # If no initial construction, create a simple one
        n_points = 42
        initial_h = np.ones(n_points) * 0.5
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define objective function using correlation
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Optimize with multiple restarts
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Use a simple gradient-free method like Nelder-Mead for better exploration
    # but we'll also use SLSQP for our final refinement
    
    restarts = 5
    for i in range(restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbation
        h_perturbed = h_init + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Normalize
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Use SLSQP with bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        try:
            res = minimize(
                compute_C5, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-6}
            )
            
            if res.success:
                candidate_h = res.x
                candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
                candidate_C5 = compute_C5(candidate_h)
                
                if candidate_C5 < best_C5:
                    best_h = candidate_h.copy()
                    best_C5 = candidate_C5
                    
        except:
            pass
    
    # Final refinement with more iterations
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        try:
            res = minimize(
                compute_C5,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8}
            )
            
            if res.success:
                final_h = res.x
                final_h = final_h * (n_points / 2) / np.sum(final_h)
                final_C5 = compute_C5(final_h)
                
                if final_C5 < best_C5:
                    best_h = final_h
                    best_C5 = final_C5
                    
        except:
            pass
            
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
