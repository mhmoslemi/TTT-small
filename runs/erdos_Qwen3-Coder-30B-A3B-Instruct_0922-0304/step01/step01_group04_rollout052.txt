import numpy as np
import time
from scipy.optimize import differential_evolution
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Objective function to minimize
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum
    def sum_constraint(x):
        return np.sum(x) - target_sum
    
    constraints = [
        {'type': 'eq', 'fun': sum_constraint}
    ]
    
    # Use differential evolution for global optimization
    try:
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            seed=seed,
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            atol=1e-6,
            tol=1e-6,
            callback=lambda x, convergence: None,
            workers=1
        )
        
        if result.success:
            current_h = result.x
        else:
            current_h = h_init.copy()
    except Exception:
        # Fallback to initial solution
        current_h = h_init.copy()
    
    # Ensure final constraints are met
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Validate final solution
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
