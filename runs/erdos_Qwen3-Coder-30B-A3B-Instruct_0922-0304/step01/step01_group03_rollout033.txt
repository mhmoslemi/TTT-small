import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values, dtype=float)]
    else:
        population = []
    
    # Fill population
    while len(population) < pop_size:
        h = np.zeros(n_points)
        # Randomly set 22 or 23 elements to 1
        num_ones = np.random.randint(22, 24)
        ones_indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[ones_indices] = 1.0
        # Renormalize
        h = h * (n_points / 2.0) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Add initial h if available
    if 'initial_h_values' in globals():
        population[0] = np.array(initial_h_values, dtype=float)
        # Renormalize if needed
        population[0] = population[0] * (n_points / 2.0) / np.sum(population[0])
        population[0] = np.clip(population[0], 0, 1)
    
    # Function to evaluate C5
    def evaluate_c5(h):
        # Compute correlation
        corr = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    generations = 2000
    elite_size = pop_size // 10
    crossover_rate = 0.8
    mutation_rate = 0.05
    mutation_noise = 0.05
    no_improvement_limit = 50
    no_improvement_count = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep top individuals
        elites = population[:elite_size]
        new_population = elites[:]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents with probability proportional to fitness (inverse)
            probs = 1.0 / (np.array(fitness) + 1e-10)
            probs = probs / np.sum(probs)
            parent_indices = np.random.choice(pop_size, size=2, p=probs)
            parent1, parent2 = population[parent_indices[0]], population[parent_indices[1]]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                offspring = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                offspring = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                num_mutations = max(1, int(mutation_rate * n_points))
                mutation_indices = np.random.choice(n_points, size=num_mutations, replace=False)
                for i in mutation_indices:
                    offspring[i] += np.random.normal(0, mutation_noise)
                offspring = np.clip(offspring, 0, 1)
                # Renormalize
                offspring = offspring * (n_points / 2.0) / np.sum(offspring)
                offspring = np.clip(offspring, 0, 1)
            
            new_population.append(offspring)
        
        # Replace population
        population = new_population[:pop_size]
        fitness = [evaluate_c5(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Diversification
        if no_improvement_count > no_improvement_limit:
            # Reset some individuals
            num_reset = pop_size // 5
            reset_indices = np.random.choice(pop_size, size=num_reset, replace=False)
            for i in reset_indices:
                h = np.zeros(n_points)
                num_ones = np.random.randint(22, 24)
                ones_indices = np.random.choice(n_points, size=num_ones, replace=False)
                h[ones_indices] = 1.0
                h = h * (n_points / 2.0) / np.sum(h)
                h = np.clip(h, 0, 1)
                population[i] = h
                fitness[i] = evaluate_c5(h)
            mutation_noise *= 0.9
            no_improvement_count = 0
    
    return (best_h.tolist(), best_c5, len(best_h))
