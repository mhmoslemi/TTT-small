import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    try:
        h_init = np.array(initial_h_values)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Normalize initial guess to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define the optimization variables
    h = Variable(n_points)
    
    # Create frequency domain variables using Fourier transform
    # We work with the DFT (Discrete Fourier Transform) of h
    # Let H = fft(h) and G = fft(1-h) = 1 - fft(h) (element-wise)
    # Then the overlap C(k) is related to the inverse DFT of H*G
    
    # For better numerical stability, we will use CVXPY variables directly in the
    # spectral domain and translate back
    
    # Create constraints and problem
    constraints = [
        sum(h) == n_points / 2,
        h >= 0,
        h <= 1
    ]
    
    # Objective: minimize maximum overlap
    # We'll do this by minimizing the maximum of the values of C_k
    # The maximum overlap is computed as the convolution of h with (1-h)
    # In frequency domain convolution becomes pointwise multiplication
    
    # Define the overlap expression
    # We use the fact that the overlap is equivalent to convolution h * (1 - h)
    # We want to minimize the maximum of that convolution
    
    # Evaluate using correlation to compute C5
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    # Try to solve using a convex formulation in spectral domain
    # Since it's a frequency domain problem, we use CVXPY to minimize
    # the maximum value of the resulting frequencies
    
    # Create a variable for the maximum value, and use the constraint
    # that each element of h must be between 0 and 1
    # and its sum equals n_points/2
    
    # For optimization, we frame it differently:
    # minimize max_k integral h(x)(1-h(x+k)) dx
    # which we directly evaluate as convolution
    
    # Use iterative approach with trust region or similar
    # But to maintain simplicity and speed, we will first try
    # the simplest convex relaxation
    
    # Solving directly with CVXPY
    max_overlap = Variable()
    
    # Approximation: we approximate the max by a soft penalty
    # or just take a simple direct solve based on constraints
    
    # Use a simple line search in some direction
    # Here we use a differential evolution approach to initialize search
    from scipy.optimize import differential_evolution
    
    def objective(h_vals):
        h_vals = np.array(h_vals)
        if np.sum(h_vals) != n_points / 2:
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    bounds = [(0, 1) for i in range(n_points)]
    
    try:
        # The actual optimization is highly expensive on CPU
        # We therefore try a small-scale optimization with initial guess and simple local search
        res = differential_evolution(objective, bounds, maxiter=100, popsize=5)
        h_opt = res.x
        best_C5 = objective(h_opt)
    except:
        h_opt = h_init.copy()
        best_C5 = compute_C5(h_opt)
    
    # Post-process optimization using simple gradient ascent if time permits
    elapsed = time.time() - start_time
    if elapsed < 0.1 * budget_s:
        # We run a simple gradient descent update on the best solution
        # to potentially improve it
        step_size = 0.001
        for _ in range(1000):
            # finite difference gradient estimation
            h_current = np.array(h_opt)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            corr = np.correlate(h_current, 1 - h_current, mode="full")
            max_corr_idx = np.argmax(corr)
            max_corr = corr[max_corr_idx] * dx
            
            # For max overlap minimization, we want to decrease
            # gradient of the overlapping effect
            
            # Estimate gradient using small step
            step = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Perturb h[i] a bit
                h_pert = h_current.copy()
                h_pert[i] += step
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                corr_plus = np.correlate(h_pert, 1 - h_pert, mode="full")
                grad[i] = (np.max(corr_plus) * dx - max_corr) / step
                
            # Update
            h_opt -= step_size * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            if time.time() - start_time > 0.8 * budget_s:
                break
                
        best_C5 = compute_C5(h_opt)
    
    # Final normalization and validation
    h_opt = np.array(h_opt)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    final_C5 = compute_C5(h_opt)
    
    # Return final results
    return (h_opt.tolist(), final_C5, n_points)
