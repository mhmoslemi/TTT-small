import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Set up CVXPy problem
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Overlap constraints for all lags
    for k in range(-n_points + 1, n_points):
        # Determine valid indices for this lag
        if k >= 0:
            valid_indices = range(n_points - k)
            expr = sum(h[i] * (1 - h[i + k]) for i in valid_indices)
        else:
            valid_indices = range(-k, n_points)
            expr = sum(h[i + k] * (1 - h[i]) for i in valid_indices)
        
        constraints.append(expr <= C * dx)
    
    # Objective
    objective = Minimize(C)
    
    # Create and solve the problem
    prob = Problem(objective, constraints)
    
    # Use SCS solver, warm-start with initial guess
    prob.solve(solver=SCS, verbose=False, warm_start=True, max_iters=10000)
    
    # Extract solution
    best_h = h.value.copy()
    final_C5 = C.value
    
    # Ensure constraints for the result
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Validate the final solution
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    computed_C5 = np.max(corr) * dx
    
    return (best_h.tolist(), computed_C5, n_points)
