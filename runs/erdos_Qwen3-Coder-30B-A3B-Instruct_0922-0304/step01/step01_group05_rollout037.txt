import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initial guess
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier domain optimization
    # Variables in frequency domain
    H = Variable(n_points)
    # Constraint: H[0] = n_points/2 (DC component)
    constraints = [H[0] == n_points / 2]
    
    # Model the system in frequency domain
    # We want to minimize the maximum of overlap integral
    # The overlap is convolution of h and (1-h)
    # In frequency domain: C = H * (1-H)
    
    # For simplicity in convex optimization, we use a relaxed approach:
    # new variables for each frequency component of h
    # The direct optimization in Fourier domain with constraints is complex.
    # Instead, we still optimize in time domain but with smarter initialization.
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # More sophisticated initialization with better heuristic
    # Start with the initial construction but smooth it
    h = h_init.copy()
    
    # Run local optimization with better algorithm settings
    from scipy.optimize import differential_evolution
    
    def objective(h):
        return compute_C5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for better exploration
    res = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=500,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        disp=False
    )
    
    if time.time() - start_time > budget_s - 1:
        best_h = h
        best_c5 = compute_C5(best_h)
    else:
        best_h = res.x
        best_c5 = compute_C5(best_h)
    
    # Final normalization to ensure the sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
