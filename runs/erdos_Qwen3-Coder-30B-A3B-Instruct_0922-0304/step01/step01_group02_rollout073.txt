import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    max_iter = 100000
    T = T0
    iteration = 0
    
    # Main simulated annealing loop
    while iteration < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Create a perturbed version of h
        h_new = h.copy()
        
        # Perturb about 10% of points
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Apply small perturbations
        for idx in indices:
            # Random change with bounded magnitude
            delta = np.random.normal(0, 0.1)
            h_new[idx] += delta
            
        # Project back to feasible region [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        scaling_factor = target_sum / current_sum
        h_new = h_new * scaling_factor
        
        # Check if new solution is better or accepted
        C5_new = evaluate_erdos_solution(h_new, dx)
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elif np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new.copy()
        
        # Cool down temperature
        T = T * alpha
        iteration += 1
    
    # Final fine tuning with optimization
    bounds = [(0, 1) for _ in range(n_points)]
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points/2.0
    
    try:
        result = optimize.minimize(objective, 
                                   best_h, 
                                   method='SLSQP', 
                                   bounds=bounds,
                                   constraints={'type': 'eq', 'fun': constraint_func},
                                   options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6})
        if result.success:
            h_final = result.x
            C5_final = evaluate_erdos_solution(h_final, dx)
            if C5_final < best_C5:
                best_C5 = C5_final
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h, best_C5, n_points)
