```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
    except NameError:
        # Fallback to simple construction if no initial_h_values
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial guess to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Objective function using correlation
    def compute_C5(h):
        h = np.array(h)
        # Ensure h sums to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Gradient-based optimization with improved constraints
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum constraint and bounds
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds constraint (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Optimizer parameters
    maxiter = 500
    ftol = 1e-7
    early_stop = 1e-6
    
    # Initial training phase with medium complexity
    try:
        res = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        best_h = res.x
        curr_C5 = compute_C5(best_h)
    except:
        # Fallback to basic heuristic
        best_h = h_init.copy()
        curr_C5 = compute_C5(best_h)
    
    # Second phase: Randomized search for better solutions
    best_C5 = curr_C5
    best_solution = best_h.copy()
    
    # Runtime tracking
    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed
    
    if remaining_time > 10:
        # Local search with multiple restarts
        try:
            for restart in range(50):
                elapsed = time.time() - start_time
                if elapsed > budget_s - 1:
                    break
                # Random perturbation of solution
                noise = np.random.normal(0, 0.02, n_points)
                h_perturb = best_solution + noise
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                new_C5 = compute_C5(h_perturb)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_solution = h_perturb.copy()
        except:
            pass

    # Final refinement with scipy
    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed
    
    try:
        # Try even better restarts if time remains
        if remaining_time > 10:
            for restart in range(30):
                elapsed = time.time() - start_time
                if elapsed > budget_s - 1:
                    break
                # Generate new starting point with variance and clip
                new_start = np.random.uniform(0, 1, n_points)
                new_start = new_start * (n_points / 2) / np.sum(new_start)
                res = minimize(objective, new_start, method='SLSQP', 
                               bounds=bounds, constraints=constraints, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    new_h = res.x
                    new_C5 = compute_C5(new_h)
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        best_solution = new_h.copy()
    except:
        pass

    # Final normalization & evaluation
    final_h = best_solution * (n_points / 2) / np.sum(best_solution)
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
```
