import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h0 = np.zeros(n_points)
        h0[:23] = 1.0
        np.random.shuffle(h0)
        h0 = h0 * (n_points / 2.0 / np.sum(h0))
    
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=22, replace=False)
        h[indices] = 1.0
        h = h * (n_points / 2.0 / np.sum(h))
        population.append(h)
    
    def evaluate_c5(h):
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * (2.0 / n_points)
    
    # Main GA loop
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    no_improvement_count = 0
    mutation_noise = 0.05
    max_no_improvement = 50
    
    num_generations = 2000
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append((c5, h))
        
        # Sort by fitness (lower is better)
        fitness.sort(key=lambda x: x[0])
        best_in_gen = fitness[0][0]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = fitness[0][1].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection
        top_indices = list(range(pop_size // 2))
        top_pop = [fitness[i][1] for i in top_indices]
        
        # Elitism: keep best 10%
        elite_count = pop_size // 10
        elite = [fitness[i][1] for i in range(elite_count)]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_count:
            # Crossover
            parent1, parent2 = np.random.choice(top_pop, size=2, replace=False)
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_mask = np.random.random(n_points) < 0.05
            if np.any(mutation_mask):
                child[mutation_mask] += np.random.normal(0, mutation_noise, np.sum(mutation_mask))
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2.0 / np.sum(child))
            
            offspring.append(child)
        
        # Combine elite and offspring
        population = elite + offspring
        
        # Diversification
        if no_improvement_count > max_no_improvement:
            for i in range(pop_size // 5):
                h = np.zeros(n_points)
                indices = np.random.choice(n_points, size=22, replace=False)
                h[indices] = 1.0
                h = h * (n_points / 2.0 / np.sum(h))
                population[np.random.randint(0, pop_size)] = h
            mutation_noise *= 0.9
            no_improvement_count = 0

    return (best_h.tolist(), best_c5, len(best_h))
