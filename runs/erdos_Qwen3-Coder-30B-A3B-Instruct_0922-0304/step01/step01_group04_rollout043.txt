import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        h = np.linspace(0.9, 0.1, n_points)
        h = h / h.sum() * 21.0
    
    dx = 2.0 / n_points
    target_sum = 21.0  # n_points / 2
    
    # Ensure initial solution is valid
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.9995
    T_min = 1e-5
    max_iter = 5000
    
    T = T0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Generate neighbor solution
        mask = np.random.rand(n_points) < 0.05  # 5% of points
        delta = np.random.uniform(-0.1, 0.1, size=(n_points)) * mask
        
        h_trial = h + delta
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize to maintain sum constraint
        if abs(np.sum(h_trial) - target_sum) > 1e-6:
            scale = target_sum / np.sum(h_trial)
            h_trial = h_trial * scale
        
        # Evaluate new solution
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject
        delta_f = c5_trial - best_c5
        if delta_f < 0 or np.exp(-delta_f / T) > np.random.rand():
            h = h_trial.copy()
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
        
        # Cool down
        T *= alpha
        iteration += 1
    
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
