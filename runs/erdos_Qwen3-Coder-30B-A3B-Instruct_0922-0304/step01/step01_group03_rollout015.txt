import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to n_points/2
        h = h * (n_points / 2 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Convex optimization approach
        # Setup variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Add overlap constraints
        for k in range(2*n_points - 1):
            # Align k-th lag correlation
            lag_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                lag_sum += h_var[i] * (1 - h_var[j])
            constraints.append(C >= lag_sum * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve with specific solver for stability
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
            new_c5 = C.value
            
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
                
        except Exception:
            pass  # Failed to solve, fall back to previous step
        
        # Periodic perturbation to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            perturb_idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.02, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2 / np.sum(h))

    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
