import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    scale = n_points / 2 / np.sum(h)
    h = h * scale
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    mask = np.random.rand(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Mutate h with Gaussian noise"""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h = h + np.where(mask, noise, 0)
    return h
