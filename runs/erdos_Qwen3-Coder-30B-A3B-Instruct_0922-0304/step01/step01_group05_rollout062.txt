import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
    except NameError:
        # Fallback initialization
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial guess
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier domain approach using CVXPY
    # Variables in time domain
    h = Variable(n_points)
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
    
    # Objective: minimize maximum overlap in frequency domain
    # Define the Fourier transform of h and (1-h)
    # Using the fact that convolution in time becomes multiplication in frequency
    
    # Compute the overlap using convolution theorem - approximate
    # However since we are relaxing to optimize the maximum, 
    # we'll consider the effect of non-negative constraints in a modified form
    
    # Let's solve this via a simplified cone program:
    # We'll use the fact that correlations can be used as constraints in a max problem
    
    # Use a first order method: gradient descent with projected updates
    # Start with a good initialization
    
    # Final optimization - start with the initial solution and refine
    best_h = h_init.copy()
    best_C5 = float('inf')
    
    # Inner loop optimization for refined solution
    for iteration in range(10):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Use local optimization with bounds
        def objective(h_vals):
            h_vals = np.array(h_vals)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
        
        # Simple local search with random perturbations
        current_h = best_h.copy()
        current_C5 = objective(current_h)
        
        for trial in range(20):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Make small perturbations
            perturb = np.random.normal(0, 0.01, n_points)
            new_h = current_h + perturb
            new_h = np.clip(new_h, 0, 1)
            normalization_factor = (n_points / 2) / np.sum(new_h)
            new_h = new_h * normalization_factor
            
            new_C5 = objective(new_h)
            if new_C5 < current_C5:
                current_h = new_h
                current_C5 = new_C5
                
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
    
    # Final validation
    final_C5 = objective(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
