import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Prepare convex optimization variables
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum, t >= 0]
    
    # Add overlap constraints for all possible shifts
    # This is optimization-oriented, with constraint tightening for efficiency
    for k in range(-n_points + 1, n_points):
        # Efficiently compute index range for valid shifts
        valid_indices = []
        if k >= 0:
            # For k >= 0, indices i, i+k valid if 0 <= i+k < n_points
            # => 0 <= i < n_points - k
            for i in range(n_points - k):
                valid_indices.append((i, i + k))
        else:
            # For k < 0, indices i, i+k valid if 0 <= i < n_points and 0 <= i+k < n_points
            # => -k <= i < n_points, and 0 <= i+k < n_points, so 0 <= i < n_points + k
            for i in range(-k, n_points):
                valid_indices.append((i, i + k))
        
        # Add constraint sum(h[i]*(1-h[j])) <= t for shift k
        # For current shift k, sum h[i]*(1-h[j]) for i,j where j=i+k
        constraint_expr = 0
        if len(valid_indices) > 0:
            for i, j in valid_indices:
                constraint_expr += h[i] * (1 - h[j])
            constraints.append(constraint_expr <= t)
    
    # Solve convex optimization program
    prob = cp.Problem(objective, constraints)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 5:
        # Fallback to initial solution
        h_opt = h_init.copy()
        c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    else:
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=5000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.array(h.value).flatten()
                c5_bound = float(t.value)
            else:
                # Fallback if solver didn't converge properly
                h_opt = h_init.copy()
                c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
        except Exception:
            # On solver failure, fallback to initial
            h_opt = h_init.copy()
            c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx

    # Final checks and normalization
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * target_sum / np.sum(h_opt)
    assert abs(np.sum(h_opt) - target_sum) < 1e-4, "Sum constraint violated"
    
    # Recompute C5 to reflect final values
    correlation = np.correlate(h_opt, 1 - h_opt, mode="full")
    final_c5 = np.max(correlation) * dx
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (h_opt, final_c5, n_points)
