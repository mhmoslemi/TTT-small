import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a binary-like initial vector
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        # Renormalize to exactly sum to n_points / 2
        h = h * (n_points / 2.0 / np.sum(h))
    
    # Scaling factor
    dx = 2.0 / len(h)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    max_no_improvement = 1000
    iter_since_improvement = 0
    
    # Initial simplex construction
    simplex = [h.copy()]
    for _ in range(n_points):
        perturbation = np.random.uniform(-simplex_size, simplex_size, n_points)
        h_perturbed = h + perturbation
        # Clip and normalize
        h_perturbed = np.clip(h_perturbed, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_perturbed)
        h_perturbed = h_perturbed * scale
        simplex.append(h_perturbed)
    
    def project_h(new_h):
        """Project h to maintain constraints"""
        new_h = np.clip(new_h, 0, 1)
        scale = (n_points / 2.0) / np.sum(new_h)
        new_h = new_h * scale
        return new_h
    
    def evaluate_c5(h, dx):
        """Evaluate C5 given h and dx"""
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * dx
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.1:
            break
            
        # Evaluate simplex
        values = []
        for point in simplex:
            value = evaluate_c5(point, dx)
            values.append(value)
        
        # Order simplex by function values
        indices = np.argsort(values)
        simplex = [simplex[i] for i in indices]
        values = [values[i] for i in indices]
        
        # Update best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Termination
        if iter_since_improvement > max_no_improvement:
            break
            
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        worst = simplex[-1]
        reflected = centroid + reflection_factor * (centroid - worst)
        reflected = project_h(reflected)
        reflected_value = evaluate_c5(reflected, dx)
        
        # Expansion (if better than best)
        if reflected_value < values[0]:
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = project_h(expanded)
            expanded_value = evaluate_c5(expanded, dx)
            if expanded_value < reflected_value:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        # Contraction (if worse than worst but better than second worst)
        elif reflected_value < values[-2]:
            contracted = centroid + contraction_factor * (worst - centroid)
            contracted = project_h(contracted)
            contracted_value = evaluate_c5(contracted, dx)
            if contracted_value < values[-1]:
                simplex[-1] = contracted
            else:
                # Shrink simplex
                best_point = simplex[0]
                for i in range(1, n_points):
                    simplex[i] = best_point + 0.5 * (simplex[i] - best_point)
                    simplex[i] = project_h(simplex[i])
        else:
            # Shrink simplex
            best_point = simplex[0]
            for i in range(1, n_points):
                simplex[i] = best_point + 0.5 * (simplex[i] - best_point)
                simplex[i] = project_h(simplex[i])
                
        # Periodic perturbation
        if iteration % 1000 == 0 and iteration > 0:
            perturb_idx = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            best_h_noisy = best_h.copy()
            best_h_noisy[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            best_h_noisy = project_h(best_h_noisy)
            # Replace worst point with perturbed best
            simplex[-1] = best_h_noisy
    
    return (best_h.tolist(), best_c5, len(best_h))
