import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    n_points = len(h)
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 500
    
    # MCTS parameters
    exploration_weight = 1.0
    max_nodes_per_iter = 100
    reset_freq = 1000
    
    # Shared state for MCTS
    tree = {}
    
    def get_c5(h_vec):
        return evaluate_c5(h_vec, dx)
    
    def get_node_key(h_vec):
        # Create a hashable key for the tree
        return tuple(np.round(h_vec, decimals=6))
    
    def random_mutation(h_orig, rate=0.05):
        h_new = h_orig.copy()
        n_mutate = max(1, int(rate * len(h_new)))
        indices = np.random.choice(len(h_new), size=n_mutate, replace=False)
        for idx in indices:
            noise = np.random.normal(0, 0.1)
            h_new[idx] += noise
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (22.5 / np.sum(h_new))
        return h_new
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Tree search
        current_node = get_node_key(h)
        node_values = []
        
        # Sample few mutations from current node
        for _ in range(5):
            if time.time() - start_time > budget_s - 0.1:
                break
            h_m = random_mutation(h)
            c5 = get_c5(h_m)
            node_values.append((h_m, c5))
        
        # Find the best in this batch
        best_mutation = min(node_values, key=lambda x: x[1])
        h_best = best_mutation[0]
        c5_best = best_mutation[1]
        
        if c5_best < best_c5:
            h = h_best
            best_c5 = c5_best
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            iter_since_improvement += 1
        
        # Additional mutations to keep exploring
        for _ in range(2):
            if time.time() - start_time > budget_s - 0.1:
                break
            h_new = random_mutation(h)
            new_c5 = get_c5(h_new)
            
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
                
                if time.time() - start_time > budget_s - 0.1:
                    break
        
        # Periodic reset to escape local minima
        if iteration % reset_freq == 0 and iteration > 0:
            # Reset most recent 20% of solutions
            n_reset = max(1, int(0.2 * n_points))
            indices = np.random.choice(n_points, size=n_reset, replace=False)
            h[indices] = np.random.uniform(0, 1, n_reset)
            # Renormalize
            h = h * (22.5 / np.sum(h))
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
