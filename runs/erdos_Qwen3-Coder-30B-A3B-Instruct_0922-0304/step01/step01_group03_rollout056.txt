import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Genetic algorithm parameters
    pop_size = 50
    num_generations = 2000
    mutation_rate = 0.05
    elite_size = pop_size // 10
    diversification_threshold = 50
    noise_level = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(np.array(initial_h_values, dtype=float))
    else:
        population.append(h.copy())
        
    # Create initial population
    for _ in range(pop_size - 1):
        h_candidate = np.zeros(len(h))
        # Randomly place 22-23 ones
        num_ones = np.random.randint(22, 24)
        ones_indices = np.random.choice(len(h), size=num_ones, replace=False)
        h_candidate[ones_indices] = 1.0
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (22.5 / np.sum(h_candidate))
        population.append(h_candidate)
    
    generations_since_improvement = 0
    
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness of population
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual, dx)
            fitness.append(c5)
            
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            generations_since_improvement = 0
        else:
            generations_since_improvement += 1
            
        # Diversification if no improvement
        if generations_since_improvement > diversification_threshold:
            for i in range(pop_size // 5):
                random_idx = np.random.randint(pop_size // 2, pop_size)
                h_random = np.zeros(len(h))
                num_ones = np.random.randint(22, 24)
                ones_indices = np.random.choice(len(h), size=num_ones, replace=False)
                h_random[ones_indices] = 1.0
                h_random = np.clip(h_random, 0, 1)
                h_random = h_random * (22.5 / np.sum(h_random))
                population[random_idx] = h_random
            noise_level *= 0.9
            generations_since_improvement = 0
            
        # Create new population with elitism and crossover
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, len(h) - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(len(child)):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, noise_level)
            child = np.clip(child, 0, 1)
            child = child * (22.5 / np.sum(child))
            
            new_population.append(child)
            
        population = new_population
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
