import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Set up convex optimization problem
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]
    
    # Add the overlap constraints
    # For each shift k, we have the constraint:
    # sum_{i} h[i] * (1 - h[i+k]) <= t
    # which can be written as:
    # sum_{i} h[i] - sum_{i} h[i] * h[i+k] <= t
    # Then we align the correlation
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift + 1):
        if k >= 0:
            corr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
        else:
            corr = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
        constraints.append(corr <= t)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        # Use MOSEK due to better stability for this problem
        prob.solve(solver=cp.MOSEK, verbose=False)
        
        # Retrieve solution
        current_h = h.value
        current_c5 = t.value
    except:
        # Fallback if solver fails
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)
    
    # Final processing
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Validate solution
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"

    return (current_h, current_c5, n_points)
