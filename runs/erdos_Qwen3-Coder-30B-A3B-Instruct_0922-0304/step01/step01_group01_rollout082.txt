def evaluate_c5(h, dx):

    corr = np.correlate(h, 1 - h, mode='full')

    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""

    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Rerun scaling due to clipping
    h = h * (n_points / 2) / np.sum(h)
    return h
