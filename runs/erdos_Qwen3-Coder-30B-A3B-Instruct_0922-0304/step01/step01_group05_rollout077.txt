import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Vars
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i in range(n_points - k)
            expr = sum(h[i] * (1 - h[i + k]) for i in range(n_points - k))
        else:
            # h[i] * (1 - h[i + k]) for i in range(-k, n_points)
            expr = sum(h[i] * (1 - h[i + k]) for i in range(-k, n_points))
        constraints.append(expr <= C * dx)
    
    # Objective
    objective = Minimize(C)
    
    # Problem
    prob = Problem(objective, constraints)
    
    # Try solving
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-6, 
                   acceleration_lookback=10, warm_start=True)
        
        if prob.status in {'optimal', 'optimal_inaccurate'}:
            best_h = h.value
            best_C5 = C.value
        else:
            # Fallback to gradient approach if QP fails
            return fallback_optimization(initial_h, n_points, dx, start_time, budget_s)
    except Exception:
        # Fallback to gradient approach if QP fails
        return fallback_optimization(initial_h, n_points, dx, start_time, budget_s)
    
    # Ensure normalization
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Return results
    return (best_h.tolist(), float(best_C5), n_points)

def fallback_optimization(initial_h, n_points, dx, start_time, budget_s):
    from scipy.optimize import minimize
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try a few different initial guesses
    best_h = None
    best_C5 = float('inf')
    
    for i in range(5):
        np.random.seed(i)
        h_init = initial_h.copy()
        # Add small random noise
        h_init += np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        res = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 500, 'ftol': 1e-6})
        
        if res.success:
            final_C5 = compute_C5(res.x)
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = res.x
    
    if best_h is None:
        # Revert to initial guess
        best_h = initial_h.copy()
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
