import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    T0 = 1e-3
    cooling_rate = 0.995
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 200
    reset_freq = 5000
    temperature = T0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Finite difference gradient evaluation
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Update with simulated annealing step
        h_new = h - alpha * grad
        noise = np.random.normal(0, temperature / 2.0, len(h))
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        scale = np.sum(h_new) / (len(h) / 2.0)
        h_new = h_new / scale
        h_new = np.clip(h_new, 0, 1)
        
        # Apply constraint: sum to exactly 22.5
        h_new = h_new * (len(h) / 2.0) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
            
        # Update temperature
        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:
                alpha *= gamma
            temperature *= cooling_rate
            iter_since_improvement = 0
            
        # Periodic reset
        if iteration % reset_freq == 0 and iteration > 0:
            # Binary configuration
            binary_h = np.zeros_like(h)
            segments = np.random.randint(3, 8, size=5)
            pos = 0
            for segment in segments:
                end = min(pos + segment, len(h))
                binary_h[pos:end] = 1
                pos = end
                if pos >= len(h):
                    break
            # Clip and renormalize
            h = np.clip(binary_h, 0, 1)
            h = h * (len(h) / 2.0) / np.sum(h)
            
        # Periodic small perturbation to escape plateaus
        if iteration % 500 == 0 and iteration > 0:
            perturb_idx = np.random.choice(len(h), size=int(0.1 * len(h)), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (len(h) / 2.0) / np.sum(h)

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
