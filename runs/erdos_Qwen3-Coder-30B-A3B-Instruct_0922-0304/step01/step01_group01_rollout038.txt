import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

    # Initialization
    h = h_init.copy()

    alpha = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha_decay = 0.999
    max_iter = 10000

        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current value and find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        max_corr = corr[k]
        c5_current = max_corr * (2.0 / n_points)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Compute gradient with respect to the dominant lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i - k >= 0:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i - k]
            else:
                # Cross-correlation handles negative indices specially
                # But we're just computing one specific gradient
                pass
        
        # Apply gradient with ADAM
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        
        # Bias correction
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        # Update parameters
        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        alpha *= alpha_decay
        
        # Early stopping on stagnation
        if t > 100 and abs(corr[k] - corr[np.argmax(corr)]) < 1e-5:
            break
            
    return (best_h, best_c5, n_points)
