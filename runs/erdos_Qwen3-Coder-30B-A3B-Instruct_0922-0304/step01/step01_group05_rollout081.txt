import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:

        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize h_init to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Function to compute overlap integral
    def compute_C5(h):
        h = np.array(h)
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max scaled by dx
        return np.max(corr) * dx
    
    # Define optimization problem using CVXPY
    def solve_with_cvxpy(h_start):
        # Create variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Objective - minimize C5 using max correlation
        # We want to minimize max of correlations
        # Fast approximate by directly optimizing using L-BFGS
        
        # For better convergence, try warm-start with rectified gradient directions
        # First compute current solution
        current_C5 = compute_C5(h_start)
        
        return h_start, current_C5
    
    # Optimization with restarts to avoid local minima
    best_h = h_init.copy()
    best_C5 = compute_C5(h_init)
    
    # Try different starting points
    for restart in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbation
        h_perturbed = h_init.copy() + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Try to refine
        def objective(h):
            return compute_C5(h)
        
        # Use L-BFGS with bounds
        res = minimize(objective, h_perturbed, method='L-BFGS-B', 
                       bounds=[(0, 1) for _ in range(n_points)],
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_opt = res.x
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            C5_opt = compute_C5(h_opt)
            if C5_opt < best_C5:
                best_C5 = C5_opt
                best_h = h_opt.copy()
    
    # Final refinement using a simple gradient descent approach with bounds
    h_final = best_h.copy()
    lr = 1e-3
    decay = 0.99
    
    max_iter = 10000
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        current_C5 = compute_C5(h_final)
        
        # Compute gradient approximately using finite differences
        eps = 1e-4
        grad = np.zeros_like(h_final)
        
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h_final.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
        
        # Update with gradient
        h_new = h_final - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_final = h_new
        lr *= decay
        
        if it % 100 == 0:
            new_C5 = compute_C5(h_final)
            if abs(new_C5 - current_C5) < 1e-8:
                break
    
    # Final evaluation
    final_C5 = compute_C5(h_final)
    
    return (h_final.tolist(), final_C5, n_points)
