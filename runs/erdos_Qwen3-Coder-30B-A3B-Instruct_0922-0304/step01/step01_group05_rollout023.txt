def tournament_selection(population, tournament_size):
    selected = np.random.choice(len(population), tournament_size)
    fitness = [compute_C5(population[i]) for i in selected]
    return selected[np.argmin(fitness)]

def uniform_crossover(parent1, parent2):
    crossover_point = np.random.randint(1, len(parent1))
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate_individual(h, mutation_rate):
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, 0.1)
    return h
