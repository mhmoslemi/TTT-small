import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization with CVXPY
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Relax the problem using CVXPY
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Add overlap constraints for a subset of lags to manage complexity
        # We sample lags to avoid excessive constraints
        max_lag = 2 * n_points - 2
        lag_indices = np.linspace(0, max_lag, min(100, max_lag), dtype=int)
        
        for k in lag_indices:
            # Compute overlap for lag k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(C >= overlap_expr * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                scale = (n_points / 2) / np.sum(h_new)
                h_new = h_new * scale
                h_new = np.clip(h_new, 0, 1)
                
                new_c5 = evaluate_c5(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
                    if iter_since_improvement > max_no_improvement:
                        # Perturb slightly to escape local minimum
                        perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                        h_new[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * ((n_points / 2) / np.sum(h_new))
                        new_c5 = evaluate_c5(h_new, dx)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_new.copy()
                            iter_since_improvement = 0
            else:
                # Fallback to previous solution slightly perturbed
                h_new = h.copy() + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * ((n_points / 2) / np.sum(h_new))
                new_c5 = evaluate_c5(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    
        except Exception:
            # Fallback
            h_new = h.copy() + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * ((n_points / 2) / np.sum(h_new))
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                
        h = h_new.copy()
        
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
