import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Create variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    
    # Integral constraint
    constraints.append(sum(h) == n_points / 2)
    
    # Overlap constraints - reuse indices to make it faster
    max_lag = n_points - 1
    for k in range(-max_lag, max_lag + 1):
        if k >= 0:
            # For positive lag: shift h forward
            # Indices: 0 -> k, 1 -> k+1, ..., n-k-1 -> n-1
            indices_i = np.arange(n_points - k)
            indices_j = np.arange(k, n_points)
        else:
            # For negative lag: shift h backward
            # Indices: -k -> 0, -k+1 -> 1, ..., n-1 -> n+k-1
            indices_i = np.arange(-k, n_points)
            indices_j = np.arange(n_points + k)
        if len(indices_i) == 0 or len(indices_j) == 0:
            continue
        
        # Ensure valid indices
        indices_i = indices_i[indices_i < n_points]
        indices_j = indices_j[indices_j < n_points]
        
        # Create overlap constraint
        if len(indices_i) != len(indices_j):
            # Pad the shorter one
            if len(indices_i) > len(indices_j):
                indices_i = indices_i[:len(indices_j)]
            else:
                indices_j = indices_j[:len(indices_i)]
                
        overlap = sum(h[indices_i] * (1 - h[indices_j]))
        constraints.append(overlap <= C * dx)
    
    # Objective
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    # Solve using SCS solver
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-6)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_result = h.value
            c5_bound = C.value
        else:
            # Fallback: Use initial guess and local optimization if QCP fails
            h_result = initial_h.copy()
            h_result = h_result * (n_points / 2) / np.sum(h_result)
            c5_bound = compute_C5(h_result, n_points, dx)
            
    except:
        # Fallback: Use initial guess and local optimization if QCP fails
        h_result = initial_h.copy()
        h_result = h_result * (n_points / 2) / np.sum(h_result)
        c5_bound = compute_C5(h_result, n_points, dx)
    
    # Local refinement with scipy if time allows
    if time.time() - start_time < budget_s - 2:
        def compute_C5(h_vals, n_points, dx):
            h = np.array(h_vals)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        def objective(h_vals):
            return compute_C5(h_vals, n_points, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        # Local optimization
        try:
            res = minimize(objective, h_result, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 500, 'ftol': 1e-8})
            if res.success:
                h_result = res.x
                c5_bound = compute_C5(h_result, n_points, dx)
        except:
            pass
    
    return (h_result.tolist(), float(c5_bound), n_points)
