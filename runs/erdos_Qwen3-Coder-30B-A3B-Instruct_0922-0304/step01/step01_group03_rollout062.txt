import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    
    # Constraint normalization
    h = h * (len(h) / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturb_freq = 1000
    
    # Create initial simplex
    n_points = len(h)
    simplex = [h.copy()]
    for i in range(n_points):
        perturb = np.random.normal(0, simplex_size, n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (len(h_perturb) / 2.0) / np.sum(h_perturb)
        simplex.append(h_perturb)
    
    simplex_eval = [evaluate_c5(v, dx) for v in simplex]
    
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(50000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Order simplex by C5
        order = np.argsort(simplex_eval)
        simplex = [simplex[i] for i in order]
        simplex_eval = [simplex_eval[i] for i in order]
        
        if simplex_eval[0] < best_c5:
            best_h = simplex[0].copy()
            best_c5 = simplex_eval[0]
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > max_no_improvement:
            # Perturb best solution
            idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_perturb = best_h.copy()
            h_perturb[idx] += np.random.normal(0, 0.01, len(idx))
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (len(h_perturb) / 2.0) / np.sum(h_perturb)
            simplex[-1] = h_perturb
            simplex_eval[-1] = evaluate_c5(h_perturb, dx)
            iter_since_improvement = 0
            
        # Nelder-Mead steps
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflect
        h_reflect = centroid + reflection_factor * (centroid - simplex[-1])
        h_reflect = np.clip(h_reflect, 0, 1)
        h_reflect = h_reflect * (len(h_reflect) / 2.0) / np.sum(h_reflect)
        c_reflect = evaluate_c5(h_reflect, dx)
        
        if c_reflect < simplex_eval[0]:
            # Expand
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = np.clip(h_expand, 0, 1)
            h_expand = h_expand * (len(h_expand) / 2.0) / np.sum(h_expand)
            c_expand = evaluate_c5(h_expand, dx)
            if c_expand < c_reflect:
                simplex[-1] = h_expand
                simplex_eval[-1] = c_expand
            else:
                simplex[-1] = h_reflect
                simplex_eval[-1] = c_reflect
        elif c_reflect < simplex_eval[-2]:
            simplex[-1] = h_reflect
            simplex_eval[-1] = c_reflect
        else:
            # Contract
            h_contract = simplex[-1] + contraction_factor * (centroid - simplex[-1])
            h_contract = np.clip(h_contract, 0, 1)
            h_contract = h_contract * (len(h_contract) / 2.0) / np.sum(h_contract)
            c_contract = evaluate_c5(h_contract, dx)
            if c_contract < simplex_eval[-1]:
                simplex[-1] = h_contract
                simplex_eval[-1] = c_contract
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (len(simplex[i]) / 2.0) / np.sum(simplex[i])
                    simplex_eval[i] = evaluate_c5(simplex[i], dx)
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
