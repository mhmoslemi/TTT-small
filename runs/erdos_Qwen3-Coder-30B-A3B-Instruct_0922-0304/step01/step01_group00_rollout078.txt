import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum, norm

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Initialize hyperparameters
    max_iter = 1000
    step_size = 0.01
    iterations = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient-based refinement to reduce maximum overlap
        current_c5 = evaluate_c5(h, dx)
        
        # Compute gradient of C5 with respect to h using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        h_base = h.copy()
        
        # Perturb each dimension to compute finite difference gradient 
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update h in descending direction of gradient
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:

        if iteration % 50 == 0 and iteration > 0:

                step_size *= 0.9
            else:
                step_size *= 1.05  # If improving, increase slightly

        if iteration % 30 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Ensure the final result is feasible
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
