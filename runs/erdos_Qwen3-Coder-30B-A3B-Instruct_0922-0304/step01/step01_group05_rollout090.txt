import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Compute the C5 bound for given h values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute correlation for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Return max correlation (scaled by dx)
    return np.max(corr) * dx

def repair(h, n_points):
    """Repair h to ensure bounds and sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    desired_sum = n_points / 2
    if current_sum != 0:
        h *= desired_sum / current_sum
    else:
        h = np.ones_like(h) * (desired_sum / n_points)
    return h

def compute_overlap(h, dx):
    """Compute the maximum overlap for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
