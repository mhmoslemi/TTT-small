import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h = h * (n_points / 2) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Uniform crossover of two parents"""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate, n_points):
    """Gaussian mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    individual += mask * np.random.normal(0, 0.05, len(individual))
    return project_feasible(individual, n_points)
