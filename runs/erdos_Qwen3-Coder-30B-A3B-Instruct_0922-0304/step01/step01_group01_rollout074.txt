import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    h_init = project_to_feasible(h_init, n_points)
    
    # PSO parameters
    n_particles = 50
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_iterations = 1000
    max_stagnation = 50
    velocity_clamp = (-0.5, 0.5)
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Include the initial solution as one particle
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Initialize other particles
    for _ in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Evaluate initial fitness
    fitness = [evaluate_c5(p, 2.0 / n_points) for p in particles]
    
    # Track personal best
    p_best = particles.copy()
    p_best_fitness = fitness.copy()
    
    # Track global best
    g_best_idx = np.argmin(fitness)
    g_best = particles[g_best_idx].copy()
    g_best_fitness = fitness[g_best_idx]
    
    best_h, best_c5 = g_best.copy(), g_best_fitness
    
    start_time = time.time()
    stagnation_counter = 0
    
    for iteration in range(max_iterations):

            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = (inertia * velocities[i] + 
                           cognitive * r1 * (p_best[i] - particles[i]) + 
                           social * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            new_velocity = np.clip(new_velocity, velocity_clamp[0], velocity_clamp[1])
            
            # Update position
            new_position = particles[i] + new_velocity
            
            # Apply feasibility projection
            new_position = project_to_feasible(new_position, n_points)
            
            # Evaluate new fitness
            new_fitness = evaluate_c5(new_position, 2.0 / n_points)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_position.copy()
                p_best_fitness[i] = new_fitness

            else:
                stagnation_counter += 1
        
        if stagnation_counter >= max_stagnation:

    # Final validation
    best_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_c5(best_h, 2.0 / n_points)
