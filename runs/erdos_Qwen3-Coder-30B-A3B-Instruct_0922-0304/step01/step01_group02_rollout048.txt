import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create a step function with integral 1
        n_points = 42
        h = np.ones(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # normalize to correct integral
        h = h * (n_points / 2.0) / np.sum(h)
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    max_iter = 100000
    current_T = T0
    
    # Keep track of iterations
    iterations = 0
    
    # Main SA loop
    while iterations < max_iter and time.time() - start_time < budget_s * 0.95:
        # Create a perturbed version of h
        h_new = h.copy()
        
        # Select a random subset of points to modify (5-10%)
        num_changes = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, num_changes, replace=False)
        
        # Modify selected points with small random changes
        for idx in indices:
            # Small random step
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
            
        # Ensure h_new is within bounds [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the new solution
        if C5_new < best_C5:
            # Always accept when better
            h = h_new
            best_C5 = C5_new
            best_h = h_new.copy()
        else:
            # Accept with probability based on temperature
            delta_E = C5_new - best_C5
            if np.random.rand() < np.exp(-delta_E / current_T):
                h = h_new
                # Don't update best if not better, but change h anyway for next iteration
        
        # Reduce temperature
        current_T *= alpha
        iterations += 1
        
        # Occasionally reinitialize to escape local optima
        if iterations % 5000 == 0 and current_T < 0.01:
            # Random restart
            h = np.random.rand(n_points)
            h = h * (n_points / 2.0) / np.sum(h)
            h = np.clip(h, 0, 1)
    
    # Final evaluation of best solution
    final_C5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h, final_C5, n_points)
