import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Objective function for optimization
    def objective(h):
        h = np.abs(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Constraint bounds
    def constraint_bounds(h):
        return np.minimum(h, 1 - h)
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)},
    ]
    
    # Initial evaluation
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Local search with simulated annealing approach
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simulated annealing parameters
    T = 1.0
    T_min = 1e-8
    alpha = 0.99
    max_iter = 10000
    iter_count = 0
    
    # Use scipy minimize as a base with custom strategy
    try:
        # Try a few optimization approaches
        
        # Approach 1: Direct optimization with constraints
        options = {'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
        result = minimize(
            objective,
            current_h,
            method='SLSQP',
            constraints=constraints,
            options=options,
        )
        
        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = evaluate_c5(current_h, dx)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Approach 2: Local refinement with perturbations
        for _ in range(500):
            if time.time() - start_time > budget_s * 0.95:
                break
            iter_count += 1
            
            # Make small random perturbations
            perturbation = np.random.normal(0, 0.01, n_points)
            h_perturbed = current_h + perturbation
            
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_c5(h_perturbed, dx)
            
            if c5_perturbed < current_c5:
                current_h = h_perturbed
                current_c5 = c5_perturbed
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
            # Simulated annealing cooling schedule
            if iter_count % 100 == 0:
                T *= alpha
                if T < T_min:
                    T = T_min
                    
    except Exception as e:
        pass  # Fallback to initial construction
    
    # Final check with better optimizer and possible local search
    try:
        # Ensure the final solution is valid
        current_h = np.clip(best_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = evaluate_c5(current_h, dx)
        
        # Ensure correctness even with a final check
        assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
        assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
        
    except Exception as e:
        current_h = best_h.copy()
        current_c5 = evaluate_c5(current_h, dx)

    return (current_h, current_c5, n_points)
