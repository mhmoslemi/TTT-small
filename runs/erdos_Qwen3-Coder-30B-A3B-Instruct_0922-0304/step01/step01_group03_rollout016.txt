import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a starting construction
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1 to make sum = 22.5
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))

    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    # Optimization parameters
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    atol = 1e-6

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Try convex optimization approach
        try:
            # Create variables
            h_var = Variable(len(h), bounds=(0, 1))
            C = Variable()
            
            # Constraint for the sum
            constraints = [sum(h_var) == 22.5]
            
            # Add overlap constraints
            # For each lag k (0 to 2*len(h)-2)
            # C >= sum(h[i] * (1 - h[(i+k)%n])) * dx
            # But we can approximate with a subset of critical lags
            # We include every other lag to reduce the number of constraints
            # but make sure we cover the most important ones
            lags_to_check = list(range(0, 2*len(h)-1, 2))  # Every other lag
            # For better accuracy, also include some lags around 0 and others
            lags_to_check.extend([len(h)-1, len(h), len(h)+1, 2*len(h)-2])  # edge lags
            lags_to_check = list(set(lags_to_check))
            lags_to_check = [l for l in lags_to_check if 0 <= l < 2*len(h)-1]

            for k in lags_to_check:
                # Compute the overlap for current lag k
                # h[i] * (1 - h[(i+k)%n])
                overlap_expr = 0.0
                for i in range(len(h)):
                    j = (i + k) % len(h)
                    overlap_expr += h_var[i] * (1 - h_var[j])
                # Add constraint: C >= overlap_expr * dx
                constraints.append(C >= overlap_expr * dx)
            
            # Objective: minimize C
            objective = Minimize(C)
            prob = Problem(objective, constraints)
            
            # Try to solve with different solvers and settings
            prob.solve(solver=CPPLEX, verbose=False, max_iter=1000)
            # If CPLEX fails, try other solvers
            if prob.status != 'optimal':
                prob.solve(solver=SCS, verbose=False, max_iters=5000)
            if prob.status != 'optimal':
                prob.solve(solver=ECOS, verbose=False, max_iter=1000)
            
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
                # Renormalize
                h_new = h_new * (22.5 / np.sum(h_new))
                h_new = np.clip(h_new, 0, 1)
                
                new_c5 = evaluate_c5(h_new, dx)
                if new_c5 < best_c5 - atol:
                    h = h_new.copy()
                    best_c5 = new_c5
                    best_h = h.copy()
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
            else:
                iter_since_improvement += 1
        except:
            iter_since_improvement += 1

        # Add some randomness occasionally to escape local optima
        if iteration % 100 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, len(h))
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (22.5 / np.sum(h))

        # Early stopping
        if iter_since_improvement > max_no_improvement:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    # Compute the overlap for all lags
    # This replicates the behavior of np.correlate(h, 1-h, mode='full') 
    # but using a more explicit loop for clarity
    n = len(h)
    full_size = 2 * n - 1
    result = np.zeros(full_size)
    
    # Explicitly compute overlaps
    for k in range(full_size):
        overlap = 0.0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        result[k] = overlap
    
    return np.max(result) * dx
