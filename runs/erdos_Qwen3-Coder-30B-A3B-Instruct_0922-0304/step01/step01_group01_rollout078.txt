import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h onto feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def tournament_select(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection"""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Gaussian mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    individual = individual.copy()
    individual[mask] += np.random.normal(0, sigma, np.sum(mask))
    return individual
