import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create structured initial solution: step function with noise
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated annealing parameters
    T = 0.05
    cooling_rate = 0.9998
    min_T = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iteration counter
    iterations = 0
    max_iterations = 100000
    
    while iterations < max_iterations and time.time() - start_time < budget_s - 1:
        # Try different perturbation strategies
        if np.random.random() < 0.7:
            # Swap perturbation: flip two random points
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Add noise to two points and adjust
            h_new = h.copy()
            i, j = np.random.choice(n_points, 2, replace=False)
            noise_i = np.random.normal(0, 0.01)
            noise_j = np.random.normal(0, 0.01)
            h_new[i] += noise_i
            h_new[j] += noise_j
            
            # Adjust to keep sum constraint
            adjustment = -(np.sum(h_new) - n_points / 2) / 2
            h_new[i] += adjustment
            h_new[j] += adjustment
        
        # Clip to feasible range
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure the sum constraint is met
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5 or np.random.random() < np.exp((best_c5 - new_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature
        T = max(min_T, T * cooling_rate)
        
        iterations += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
