import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Initialize best solution
    best_h = initial_h.copy()
    best_C5 = float('inf')
    
    # Early exit check
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_C5, n_points)
    
    # Define problem variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Generate overlap constraints for all shifts
    for k in range(-n_points + 1, n_points):
        if k < 0:
            # For negative k, we shift h to the right
            valid_indices = range(0, n_points + k)
            expr = sum(h[i] * (1 - h[i - k]) for i in valid_indices)
        else:
            # For positive k, we shift h to the left
            valid_indices = range(k, n_points)
            expr = sum(h[i] * (1 - h[i - k]) for i in valid_indices)
        
        constraints.append(expr <= C * dx)
    
    # Define objective
    objective = Minimize(C)
    
    # Construct problem
    prob = Problem(objective, constraints)
    
    # Solve using SCS
    try:
        prob.solve(solver=SCS, verbose=False, eps=1e-6, max_iters=5000, 
                   use_indirect=True, entropy=True)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            if C.value < best_C5:
                best_C5 = C.value
                best_h = h.value
    except Exception:
        pass
    
    # Constrained local refinement using SLSQP if needed
    if time.time() - start_time < budget_s - 1:
        from scipy.optimize import minimize
        
        def compute_C5(h_vals):
            h_vals = np.array(h_vals)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
        
        def objective(h_vals):
            return compute_C5(h_vals)
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
        
        def constraint_bounds(h_vals):
            return np.concatenate([h_vals, 1 - h_vals])
        
        # Local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 500, 'ftol': 1e-6})
        
        if res.success:
            test_h = res.x
            test_h = test_h * (n_points / 2) / np.sum(test_h)
            test_C5 = compute_C5(test_h)
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = test_h
    
    return (best_h.tolist(), best_C5, n_points)
