import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper function to evaluate and repair solution
    def evaluate_and_repair(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h, evaluate_c5(h, dx)
    
    # Differential Evolution phase
    def de_objective(h):
        h, c5 = evaluate_and_repair(h)
        return c5
    
    # Bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        tol=1e-6,
        seed=seed,
        callback=lambda x, convergence: None,
        workers=1
    )
    
    # L-BFGS-B refinement using best from DE
    best_h = de_result.x
    best_h, best_c5 = evaluate_and_repair(best_h)
    
    # Optimization using L-BFGS-B with constraints
    def lbfgs_objective(h):
        h, c5 = evaluate_and_repair(h)
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    # Bounds for L-BFGS-B
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    # Refinement with L-BFGS-B
    try:
        result = minimize(
            lbfgs_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None,
        )
        
        if result.success:
            final_h, final_c5 = evaluate_and_repair(result.x)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    current_h = best_h
    current_c5 = best_c5
    
    # Final validation
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
