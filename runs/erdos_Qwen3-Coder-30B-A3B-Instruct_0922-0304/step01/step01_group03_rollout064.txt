import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable initial guess
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        # Renormalize
        h = h * (n_points / 2.0 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 5000
    max_no_improvement = 1000
    iter_since_improvement = 0
    
    # Initialize simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        h_pert = h + simplex_size * np.random.randn(n_points)
        h_pert = np.clip(h_pert, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_pert)
        h_pert = h_pert * scale
        simplex.append(h_pert)
    
    def evaluate_simplex(simplex_points):
        values = []
        for point in simplex_points:
            c5 = evaluate_c5(point, dx)
            values.append(c5)
        return np.array(values)
    
    def project_and_normalize(h_point):
        h_point = np.clip(h_point, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_point)
        h_point = h_point * scale
        return h_point
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate simplex
        values = evaluate_simplex(simplex)
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = values[sorted_indices]
        
        # Check for improvement
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Stop if no improvement
        if iter_since_improvement > max_no_improvement:
            break
            
        # Create new simplex
        centroid = np.mean(simplex[:-1], axis=0)
        worst = simplex[-1]
        
        # Reflection
        h_reflect = centroid + reflection_factor * (centroid - worst)
        h_reflect = project_and_normalize(h_reflect)
        c5_reflect = evaluate_c5(h_reflect, dx)
        
        # Check if reflection is better than worst and worse than best
        if values[0] <= c5_reflect < values[-2]:
            simplex[-1] = h_reflect
        elif c5_reflect < values[0]:
            # Expansion
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = project_and_normalize(h_expand)
            c5_expand = evaluate_c5(h_expand, dx)
            if c5_expand < c5_reflect:
                simplex[-1] = h_expand
            else:
                simplex[-1] = h_reflect
        else:
            # Contraction
            if c5_reflect < values[-1]:
                h_contract = centroid + contraction_factor * (worst - centroid)
                h_contract = project_and_normalize(h_contract)
                c5_contract = evaluate_c5(h_contract, dx)
                if c5_contract < values[-1]:
                    simplex[-1] = h_contract
                else:
                    # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                        simplex[i] = project_and_normalize(simplex[i])
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_and_normalize(simplex[i])
        
        # Periodic perturbation
        if iteration % 1000 == 0 and iteration > 0:
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            h_perturb = project_and_normalize(h_perturb)
            simplex[-1] = h_perturb
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
