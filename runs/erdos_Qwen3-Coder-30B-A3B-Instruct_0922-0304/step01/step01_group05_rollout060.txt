import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initial guess
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define optimization variables
    h = Variable(n_points)
    
    # Objective: minimize max overlap
    max_overlap = Variable()
    
    # Create constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        max_overlap >= 0
    ]
    
    # Add overlap constraints using Fourier domain approach
    # We use a simplified approach: compute correlations and 
    # constrain maximum value
    for k in range(n_points):
        # Compute the overlap with shift k
        # This is the cross-correlation term
        if k == 0:
            overlap = sum(h[i] * (1 - h[i]) for i in range(n_points))
        else:
            # For shift k, compute sum(h[i] * (1 - h[i+k])) for valid i
            overlap = 0
            for i in range(n_points - k):
                overlap += h[i] * (1 - h[i + k])
        constraints.append(overlap <= max_overlap)
    
    # Formulate and solve the optimization problem
    objective = Minimize(max_overlap)
    prob = Problem(objective, constraints)
    
    # Solve with ECOS for speed (maximum 500 iterations)
    try:
        prob.solve(solver=ECOS, max_iter=500, verbose=False)
    except:
        # Fallback to a simpler approach if the above fails
        try:
            prob.solve(solver=OSQP, max_iter=1000, verbose=False)
        except:
            pass
    
    # Extract solution
    if prob.status == 'optimal':
        h_sol = h.value
    else:
        # If optimization fails, use the initial guess
        h_sol = h_init
    
    # Ensure feasibility
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Compute final C5 bound
    corr = np.correlate(h_sol, 1 - h_sol, mode="full")
    final_C5 = np.max(corr) * dx
    
    return (h_sol.tolist(), final_C5, n_points)
