import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Blend crossover between two parents"""
    alpha = 0.5
    offspring = alpha * parent1 + (1 - alpha) * parent2
    return offspring

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, sigma, len(individual))
    individual = individual + mask * noise
    return individual
