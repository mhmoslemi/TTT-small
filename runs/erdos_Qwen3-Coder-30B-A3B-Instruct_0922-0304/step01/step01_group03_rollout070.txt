import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))  # Renormalize to sum to 22.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iter = 5000
    exploration_weight = np.sqrt(2)
    max_children = 10
    reset_freq = 1000
    
    # Tree structure
    class TreeNode:
        def __init__(self, h_config):
            self.h = h_config.copy()
            self.c5 = evaluate_c5(self.h, dx)
            self.visits = 0
            self.children = {}
            self.parent = None
            self.action = None
    
    # Initialize root
    root = TreeNode(h)
    
    def ucb_score(node, parent_visits):
        if node.visits == 0:
            return float('inf')
        return node.c5 + exploration_weight * np.sqrt(np.log(parent_visits) / node.visits)
    
    def mutate(h_vec):
        h_new = h_vec.copy()
        num_mutate = max(1, int(0.05 * len(h_vec)))
        indices = np.random.choice(len(h_vec), size=num_mutate, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.1)
        h_new = np.clip(h_new, 0, 1)
        scale = 22.5 / np.sum(h_new)
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    def get_best_child(node):
        best_child = None
        best_score = float('inf')
        for child in node.children.values():
            score = ucb_score(child, node.visits)
            if score < best_score:
                best_score = score
                best_child = child
        return best_child
    
    # Main MCTS loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection
        current_node = root
        while current_node.children:
            current_node = get_best_child(current_node)
            if time.time() - start_time > budget_s - 0.1:
                break
                
        # Expansion
        if current_node.visits > 0 and len(current_node.children) < max_children:
            # Generate children
            for _ in range(max_children):
                if time.time() - start_time > budget_s - 0.1:
                    break
                h_new = mutate(current_node.h)
                child = TreeNode(h_new)
                current_node.children[child.h.tobytes()] = child
                child.parent = current_node
                child.action = "mutation"
                
        # Simulation
        if not current_node.children:
            # We reached a leaf node, evaluate it
            current_node.visits += 1
            current_node.c5 = evaluate_c5(current_node.h, dx)
            
            # Backpropagation
            node = current_node
            while node is not None:
                node.visits += 1
                if node.c5 < node.parent.c5 if node.parent else True:
                    node.c5 = evaluate_c5(node.h, dx)
                node = node.parent
                
        else:
            # Select a child for simulation
            child = list(current_node.children.values())[0]
            child.visits += 1
            child.c5 = evaluate_c5(child.h, dx)
            
            # Backpropagation
            node = child
            while node is not None:
                node.visits += 1
                if node.c5 < node.parent.c5 if node.parent else True:
                    node.c5 = evaluate_c5(node.h, dx)
                node = node.parent
                
        # Update best solution
        if root.c5 < best_c5:
            best_c5 = root.c5
            best_h = root.h.copy()
            
        # Periodic reset
        if iteration % reset_freq == 0 and iteration > 0:
            for i in range(10):
                idx = np.random.randint(0, len(root.h))
                root.h[idx] = np.random.random()
            root.h = np.clip(root.h, 0, 1)
            scale = 22.5 / np.sum(root.h)
            root.h *= scale
            root.c5 = evaluate_c5(root.h, dx)
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
