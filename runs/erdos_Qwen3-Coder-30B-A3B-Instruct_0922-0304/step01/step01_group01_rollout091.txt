import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

        gradient = np.zeros(n_points)
        for i in range(n_points):
            # gradient at index i is the partial derivative

            if k >= 0:

                # For simplicity, use discrete derivative:
                gradient[i] = 0

                if j < n_points:

                gradient[i] -= 2 * h[i] # the h[i] term in the overlap integral
            else:

                gradient[i] = 0
                j = (i + k) % n_points
                if 0 <= j < n_points:
                    gradient[i] += 1 - 2 * h[j]
                gradient[i] -= 2 * h[i]  # the h[i] term
        
        # Update h
        old_h = h.copy()
        h = h - alpha * gradient
