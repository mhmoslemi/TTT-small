import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with a simple construction: alternating 1s and 0s with small perturbation
        n_points = 45
        h = np.zeros(n_points)
        h[::2] = 1.0  # Every other element is 1
        # Add some randomness to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        # Renormalize to sum to n_points / 2
        h = h * (n_points / 2.0 / np.sum(h))

    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    n_points = len(h)
    
    # Use CVXPY to solve the convex optimization problem
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2.0,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Constraint for each lag
    max_lag = 2 * n_points - 2
    overlap_constraints = []
    
    # Limit the number of constraints to keep the problem tractable
    constraint_lags = list(range(0, max_lag + 1, 2))  # Every 2nd lag for efficiency
    
    for k in constraint_lags:
        # Compute the overlap expression manually using the convolution sum
        # For lag k, compute sum_i h[i] * (1 - h[(i+k) % n]) * dx
        expr = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            expr += h_var[i] * (1 - h_var[j])
        expr *= dx
        overlap_constraints.append(C >= expr)
        
    constraints.extend(overlap_constraints)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Solve problem in iterations to avoid timeout
    last_improvement = 0
    iteration = 0
    max_iter = 1000
    solver = 'ECOS'  # Fast and suitable for LP

    while iteration < max_iter and time.time() - start_time < budget_s - 0.1:
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                new_h = h_var.value
                # Renormalize and clip
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2.0 / np.sum(new_h))
                new_c5 = evaluate_c5(new_h, dx)
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    last_improvement = iteration
        except:
            # Fall back to heuristic if optimization fails
            pass
        
        iteration += 1
        
        # Periodic perturbation to escape local minima
        if iteration % 100 == 0:
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.05, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0 / np.sum(h))
            h = np.clip(h, 0, 1)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
