import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points, dx):
    # L1-ball projection to maintain sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / current_sum
        h = h * scaling
    return h

def generate_individual(n_points):
    h = np.zeros(n_points)
    half = n_points // 2
    h[:half] = 1.0
    np.random.shuffle(h)
    h += np.random.normal(0, 0.05, n_points)
    return project_feasible(h, n_points, 2.0/n_points)

def segment_crossover(h1, h2):
    n_points = len(h1)
    cut = random.randint(1, n_points-1)
    child1 = np.concatenate([h1[:cut], h2[cut:]])
    child2 = np.concatenate([h2[:cut], h1[cut:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.02):
    if random.random() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(h))
        h = h + noise
    return project_feasible(h, len(h), 2.0/len(h))

def genetic_algorithm(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    # If initial_h_values exists, use it as first individual
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    else:
        population.append(generate_individual(n_points))
    
    # Fill up population
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points))
    
    best_c5 = float('inf')
    best_h = None
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append((c5, h))
        
        # Sort by fitness (lower C5 is better)
        fitness_scores.sort(key=lambda x: x[0])
        best_current = fitness_scores[0][0]
        if best_current < best_c5:
            best_c5 = best_current
            best_h = fitness_scores[0][1].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
        
        # Selection
        survivors = [h for _, h in fitness_scores[:population_size//2]]
        
        # Crossover and mutation
        new_population = survivors[:]
        
        # Add offspring
        while len(new_population) < population_size:
            parent1 = random.choice(survivors)
            parent2 = random.choice(survivors)
            child1, child2 = segment_crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Occasionally run relaxation step
        if generation % 10 == 0:
            # Simple relaxation: gradient descent on the best solution
            best_solution = best_h.copy()
            step_size = 0.01
            iterations = 50
            best_c5_current = best_c5
            for _ in range(iterations):
                corr = correlate(best_solution, 1 - best_solution, mode="full")
                max_k = np.argmax(corr)
                # Compute approximate gradient
                eps = 1e-4
                grad = np.zeros_like(best_solution)
                for i in range(len(best_solution)):
                    h_plus = best_solution.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - best_c5_current) / eps
                
                # Update using gradient
                best_solution -= step_size * grad
                best_solution = project_feasible(best_solution, n_points, dx)
                best_c5_current = evaluate_c5(best_solution, dx)
                
                if best_c5_current < best_c5:
                    best_c5 = best_c5_current
                    best_h = best_solution.copy()
        
        # Ensure we don't exceed budget
        if time.time() - start_time > budget_s - 1:
            break

    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
