import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.zeros(n_points)
        h_init[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h_init)
    
    # Renormalize to have sum = n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    dx = 2.0 / n_points
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Optimization loop
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    # Index of largest overlap constraint to focus
    max_lag = n_points * 2 - 2
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Using CVXPY to solve the convex relaxation of the problem
        h = Variable(n_points, name='h')
        C = Variable(1, name='C')

        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2,
            C >= 0
        ]

        # Add overlap constraints
        for k in range(max_lag):
            # Compute the correct indices for circular correlation
            # sum over i of h[i] * (1 - h[i + k]) where indices wrap
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h[i] * (1 - h[j])
            constraints.append(C >= overlap_expr * dx)

        # Objective
        objective = Minimize(C)

        # Problem
        prob = Problem(objective, constraints)
        
        try:
            # Solve with some solver eco-friendly for this size
            prob.solve(solver=SCS, verbose=False, max_iters=5000)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_val = h.value
                c5_val = C.value
                
                # Normalize and clip
                h_val = np.clip(h_val, 0, 1)
                h_val = h_val * (n_points / 2) / np.sum(h_val)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_val.copy()
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
                    
                if iter_since_improvement > max_no_improvement:
                    break
            else:
                # fallback to a simpler gradient
                break
        except:
            # fallback to iterated gradient method if SCS fails
            break
    
    # If optimization failed or didn't improve, return best_h from earlier iterations
    # or from the initialization if it's better
    return (best_h.tolist(), float(best_c5), n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    # Convolution approach
    correlate_result = np.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
