import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h from the provided construction
    h_init = np.array(initial_h_values)
    
    # Normalize to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Set up CVXPY problem
    h = Variable(n_points)
    
    # Constraint 1: h[i] in [0, 1]
    constraints = [h >= 0, h <= 1]
    
    # Constraint 2: sum(h) = n_points / 2
    constraints.append(sum(h) == n_points / 2)
    
    # Construct the overlap expression using CVXPY
    # This is a bit tricky in CVXPY, so we compute manually
    # Overlap function using linear approximation
    C5_max = Variable()
    
    # We compute the convolution manually for the constraint
    # We know that overlap for shift k is sum of h[i] * (1 - h[i+k])
    # So we formulate it as a constraint
    
    # Instead, we'll minimize the maximum of the convolution
    # The maximum of all C_k will be our objective
    
    # Let's try a direct optimization approach 
    # Use a trust-constr solver to avoid complex spectral formulations
    
    # Define problem using numpy optimization with careful constraint handling
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        # Compute full convolution of h with (1 - h)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Max correlation scaled by dx
        return np.max(corr) * dx
    
    # Set up optimization with trust-constr
    bounds = [(0, 1) for _ in range(n_points)]
    constraints_eq = {'type': 'eq', 'fun': lambda h_vals: np.sum(h_vals) - n_points / 2}
    
    # Use trust-constr for better handling of constraints
    from scipy.optimize import minimize
    
    result = minimize(
        compute_C5,
        h_init,
        method='trust-constr',
        bounds=bounds,
        constraints=constraints_eq,
        options={'maxiter': 1000, 'gtol': 1e-6, 'xtol': 1e-6}
    )
    
    # Early termination check
    if time.time() - start_time > budget_s - 1:
        # Return best solution so far if we're close to time limit
        final_h = result.x
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_C5 = compute_C5(final_h)
        return (final_h.tolist(), final_C5, n_points)
    
    # Refinement with different method
    refined_result = minimize(
        compute_C5,
        result.x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints_eq,
        options={'maxiter': 1000, 'ftol': 1e-7}
    )
    
    final_h = refined_result.x
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
