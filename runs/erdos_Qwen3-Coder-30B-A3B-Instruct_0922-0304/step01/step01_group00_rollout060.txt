import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to feasible region: sum=h, bounds [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2):
    # Segment-wise crossover
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_std=0.02):
    # Mutate by adding Gaussian noise
    if random.random() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(individual))
        individual = individual + noise
    return individual

def genetic_algorithm(seed, budget_s):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1200
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create starting individual
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main GA loop
    best_solution = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep top 50%
        selected = population[:population_size//2]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        # Crossover (elitism + crossover + mutation)
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            
            if random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                child1 = project_to_feasible(child1, n_points)
                child2 = project_to_feasible(child2, n_points)
                
                # Mutate children with probability
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                
                child1 = project_to_feasible(child1, n_points)
                child2 = project_to_feasible(child2, n_points)
                
                new_population.extend([child1, child2])
            else:
                # Direct reproduction with mutation
                child = mutate(parent1.copy(), mutation_rate)
                child = project_to_feasible(child, n_points)
                new_population.append(child)
                
        population = new_population[:population_size]
        
        # Occasionally apply relaxation step
        if gen % relaxation_interval == 0 and gen > 0:
            # Apply a local optimization to improve best individual
            h_best = population[0].copy()
            # Simple gradient ascent with step size control
            step_size = 0.01
            iterations = 50
            for _ in range(iterations):
                corr = correlate(h_best, 1 - h_best, mode="full")
                c5_val = np.max(corr) * dx
                
                # Compute gradient via finite differences (simplified approach)
                grad = np.zeros_like(h_best)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_best.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_val) / eps
                
                # Gradient descent update
                h_best = h_best - step_size * grad
                h_best = project_to_feasible(h_best, n_points)
            
            # Replace worst individual with relaxed solution
            population[-1] = h_best
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
