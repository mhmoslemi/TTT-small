import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select parent using tournament selection"""
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_index)
    return [population[i] for i in selected_indices]

def crossover(parent1, parent2):
    """Average the two parents to create offspring"""
    return (parent1 + parent2) / 2.0

def mutate(individual, mutation_rate=0.1, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual = individual + noise
    return individual
