        h = h * (22.5 / np.sum(h))

    tree = {}

        'visits': 0,
        'total_c5': 0,
        'children': [],
        'best_c5': float('inf')
    }

        current_h = root_h

            # UCB selection
            children = tree[current_h]['children']
            ucb_scores = []
            for child_h in children:
                node = tree[child_h]
                if node['visits'] == 0:
                    ucb_scores.append(float('inf'))
                else:
                    ucb = node['total_c5'] / node['visits'] + \
                          exploration_weight * np.sqrt(np.log(tree[current_h]['visits']) / node['visits'])
                    ucb_scores.append(ucb)
            
            # Select child with highest UCB
            selected_child_idx = np.argmax(ucb_scores)
            current_h = children[selected_child_idx]
            
        # Expansion: add new children if not terminal
        if current_h in tree:
            node = tree[current_h]
            # Generate mutations
            generated_children = []

                    break

                new_h = list(node['h'])

                    new_h[i] = max(0, min(1, new_h[i]))

                        'h': new_h,
                        'visits': 0,
                        'total_c5': 0,
                        'children': [],

                    }
                    generated_children.append(new_h_tuple)

            node['best_c5'] = min(node['best_c5'], c5)
            
            # Backpropagate

                parent_node['total_c5'] += c5

            nodes_to_reset = random.sample(list(tree.keys()), min(len(tree) // 10, 5))

    # Final evaluation
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
