import numpy as np
import time
from scipy.optimize import differential_evolution
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to evaluate and penalize constraint violations
    def objective(h):
        h = np.clip(h, 0, 1)  # Ensure bounds
        h_sum = np.sum(h)
        penalty = 0
        if abs(h_sum - target_sum) > 1e-3:
            penalty = 1e6 * abs(h_sum - target_sum)
        return evaluate_c5(h, dx) + penalty

    # Define bounds [0,1] for all variables
    bounds = [(0, 1) for _ in range(n_points)]

    # Set up DE parameters
    maxiter = 150
    popsize = 40
    
    # Importantly, we want to explore the space around the initial solution
    # so we'll include it in the initial population
    initial_population = []
    if 'initial_h_values' in globals():
        initial_population.append(h_init.copy())
    else:
        initial_population.append(np.ones(n_points) * 0.5)
    
    # Generate additional random samples
    for i in range(popsize - 1):
        candidate = np.random.rand(n_points)
        candidate = candidate * target_sum / np.sum(candidate)
        initial_population.append(candidate)
        
    # Run differential evolution
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            seed=seed,
            disp=False,
            init=initial_population,
            atol=1e-6,
            tol=1e-6
        )
        
        if result.success:
            current_h = result.x
        else:
            # Fallback to initial construction
            current_h = h_init.copy()
            
    except Exception as e:
        # Fallback to initial construction
        current_h = h_init.copy()
    
    # Refine final solution to exactly meet constraints
    current_h = np.clip(current_h, 0, 1)
    current_sum = np.sum(current_h)
    current_h = current_h * target_sum / current_sum
    current_c5 = evaluate_c5(current_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
        
    # Return solution
    return (current_h, current_c5, n_points)
