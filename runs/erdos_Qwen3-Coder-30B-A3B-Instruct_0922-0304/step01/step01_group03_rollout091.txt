import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use a sensible starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to n_points/2
        h = h * (n_points / 2.0 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Iteration counter and tracking
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 1000
    alpha = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Try to solve a relaxed convex optimization problem to improve h
        # Here we model the overlaps using a set of linear constraints with auxiliary CVX variables
        try:
            # Define variable and problem
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints: h in [0,1], sum = n_points/2
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2.0,
                C >= 0
            ]

            # Add overlap constraints for a range of k values
            # In practice, we use more strategic lag choices
            # To avoid computational overload, we consider 100 most relevant lags
            lags = list(range(n_points))
            if n_points > 100:
                lags = np.linspace(0, n_points-1, 100, dtype=int)
            
            for k in lags:
                # Compute effective overlap using the full correlation logic
                # But focus on current lag only for efficiency
                overlap_expr = 0.0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap_expr += h_vars[i] * (1 - h_vars[j])
                # Scale and ensure we're imposing the constraint
                # This is a relaxation: target is maximum over all lags
                constraints.append(C >= overlap_expr * dx)

            # Create and solve the CVX problem
            objective = cp.Minimize(C)
            problem = cp.Problem(objective, constraints)
            
            # Try to solve it (check solver availability)
            try:
                problem.solve(solver=cp.ECOS, verbose=False)
                if problem.status == cp.OPTIMAL or problem.status == cp.OPTIMAL_INACCURATE:
                    h_new = np.array(h_vars.value).flatten()
                    if np.all(h_new >= 0) and np.all(h_new <= 1):
                        # Renormalize to sum to n_points/2
                        h_new = h_new * (n_points / 2.0 / np.sum(h_new))
                        new_c5 = evaluate_c5(h_new, dx)
                        if new_c5 < best_c5:
                            h = h_new.copy()
                            best_c5 = new_c5
                            iter_since_improvement = 0
                            best_h = h.copy()
                        else:
                            iter_since_improvement += 1
                            if iter_since_improvement > max_no_improvement:
                                # Perturb h if no improvement is seen for too long
                                h = h + np.random.normal(0, 0.001, n_points)
                                h = np.clip(h, 0, 1)
                                h = h * (n_points / 2.0 / np.sum(h))
                                iter_since_improvement = 0
                else:
                    # If infeasible or any issue, revert to gradient-based update with small move
                    h = h + np.random.normal(0, 0.001, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2.0 / np.sum(h))
            except Exception:
                # If CVXPY fails, fallback to a simple gradient model
                h = h + np.random.normal(0, 0.001, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2.0 / np.sum(h))
                
        except Exception:
            # Final fallback to just perturbation if anything fails
            h = h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0 / np.sum(h))
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
