import numpy as np
import time
from scipy import signal
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h)
        h = h * (n_points / 2.0) / np.sum(h)  # Renormalize to sum to n_points/2
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, 2.0 / n_points)
    
    # Optimization loop with alternating steps
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Use CVXPY for convex relaxation approach
        try:
            # Create variables for optimization
            h_var = cp.Variable(n_points, bounds=(0, 1))
            C = cp.Variable()
            
            # Set up constraints
            dx = 2.0 / n_points
            constraints = [cp.sum(h_var) == n_points / 2.0]
            
            # Approximate max overlap constraints
            # Use subset of lag values for tractability 
            lags = list(range(0, min(20, 2 * n_points)))  # Use some critical lags
            
            for k in lags:
                # Compute overlap sum for lag k
                overlap_sum = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap_sum += h_var[i] * (1 - h_var[j])
                constraints.append(C >= overlap_sum * dx)
            
            # Objective: minimize C
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            
            # Solve the convex problem
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).reshape(-1)
                # Renormalize to sum exactly to n_points/2
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                
                new_c5 = evaluate_c5(h_new, dx)
                
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    iter_since_improvement = 0
                    best_h = h.copy()
                else:
                    iter_since_improvement += 1
            else:
                # Fallback to simple gradient descent step if convex solve fails
                h_new = h.copy()
                # Simple gradient step with perturbation
                eps = 1e-4
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    overlap_plus = evaluate_c5(h_plus, dx)
                    overlap_minus = evaluate_c5(h_minus, dx)
                    grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
                
                h_new = h - 0.01 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                
                new_c5 = evaluate_c5(h_new, dx)
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    iter_since_improvement = 0
                    best_h = h.copy()
                else:
                    iter_since_improvement += 1
                    
        except Exception as e:
            # Fallback to gradient descent on basic problem
            eps = 1e-5
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                overlap_plus = evaluate_c5(h_plus, dx)
                overlap_minus = evaluate_c5(h_minus, dx)
                grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
            
            h_new = h - 0.01 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
        
        if iter_since_improvement > max_no_improvement:
            # Perturb solution to escape local minima
            perturb_idx = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
            iter_since_improvement = 0

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
