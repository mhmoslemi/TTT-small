import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Parameters for optimization
    max_iter = 1000
    step_size = 0.1
    max_grad_norm = 1000.0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Early exit if target reached
        if current_c5 <= 0.3807:
            break
            
        # Find index of maximum overlap
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient approximation of overlap at k_max
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
            
        # Ensure gradient is reasonable
        grad_norm = np.linalg.norm(grad)
        if grad_norm > max_grad_norm:
            grad = grad / grad_norm * max_grad_norm
            
        # Compute descent step
        step = -step_size * grad
        
        # Apply step
        h_new = h + step
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept step if it improves the best C5
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
        else:
            # Sometimes accept worse steps for escape from local minima
            if np.random.random() < 0.05:
                h = h_new
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive step size
        if iteration % 10 == 0 and iteration > 0:
            step_size *= 0.98
            if step_size < 1e-5:
                step_size = 1e-5

    return (best_h.tolist(), best_c5, n_points)
