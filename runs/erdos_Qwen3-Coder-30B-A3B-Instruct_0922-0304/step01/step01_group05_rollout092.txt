            )

            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            particles[i] += velocities[i]

            particles[i] = repair(particles[i])

            if current_val < p_best_vals[i]:

                particles[i] += noise
                particles[i] = repair(particles[i])
            last_improvement = iteration
    
    # Use final best solution
    final_h = g_best
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
