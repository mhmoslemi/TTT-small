import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to evaluate C5 and enforce constraints
    def objective(h):
        h = np.clip(h, 0, 1)  # Ensure bounds
        h = h * target_sum / np.sum(h)  # Ensure sum constraint
        return evaluate_c5(h, dx)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    popsize = 40
    mutation = 0.8
    crossover = 0.9
    maxiter = 100
    
    # Run Differential Evolution
    de_result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        crossover=crossover,
        seed=seed,
        disp=False,
        recombination=0.9
    )
    
    # Use the best result from DE as starting point for local optimization
    current_h = de_result.x
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Local optimization with L-BFGS-B
    def objective_for_lbfgsb(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Set bounds for L-BFGS-B (necessary for scipy.optimize.minimize)
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    # Try to improve using L-BFGS-B with only the best from DE
    result = minimize(
        objective_for_lbfgsb,
        current_h,
        method='L-BFGS-B',
        bounds=bounds_lbfgs,
        options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8},
        callback=lambda x: None if time.time() - start_time > budget_s * 0.95 else None,
    )
    
    if result.success:
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = evaluate_c5(current_h, dx)
    
    # Validate final solution
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
