import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except:
        # Fallback construction
        n_points = 42
        initial_h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Constraint handling for normalization
    def normalize_h(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Objective function using correlation
    def compute_C5(h):
        h_norm = normalize_h(h)
        # Compute correlation for all lags
        corr = np.correlate(h_norm, 1 - h_norm, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Multi-start optimization with several strategies
    best_C5 = float('inf')
    best_h = None
    
    # Strategy 1: Gradient-based with SLSQP
    h_init = initial_h.copy()
    h_init = normalize_h(h_init)
    
    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    # Try multiple random initializations
    for i in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Mutate initial solution to avoid local optima
        h_mutated = h_init + np.random.normal(0, 0.05, n_points)
        h_mutated = np.clip(h_mutated, 0, 1)
        h_mutated = normalize_h(h_mutated)
        
        try:
            res = minimize(objective, h_mutated, method='SLSQP', 
                          bounds=bounds, constraints=constraints, 
                          options={'maxiter': 500, 'ftol': 1e-6})
            
            if res.success:
                current_h = res.x
                current_h = normalize_h(current_h)
                current_C5 = compute_C5(current_h)
                
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = current_h.copy()
        except:
            continue
    
    # If no good solution found, return the initial solution
    if best_h is None:
        best_h = normalize_h(initial_h.copy())
        best_C5 = compute_C5(best_h)
    
    # Refinement with local search if time permits
    if time.time() - start_time < budget_s - 10:
        # Refinement using a more sophisticated approach
        try:
            # Simple smoothing to reduce oscillations
            smoothed_h = np.convolve(best_h, np.ones(5)/5, mode='same')
            smoothed_h = np.clip(smoothed_h, 0, 1)
            smoothed_h = normalize_h(smoothed_h)
            
            refined_C5 = compute_C5(smoothed_h)
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = smoothed_h
        except:
            pass
    
    return (best_h.tolist(), best_C5, n_points)
