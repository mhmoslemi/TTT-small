import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    n_points = len(h)
    n_lags = 2 * n_points - 1
    
    # Adjust the number of lags to reduce computational complexity
    lags_to_consider = np.arange(0, n_lags, 2)  # Only every 2nd lag
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Create the LP relaxation problem
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints for selected lags
        for k in lags_to_consider:
            # Compute the right hand side of overlap constraint
            h_k = np.concatenate([h_vars, h_vars[:k]])
            h_k = h_k[:n_points]
            # h_k corresponds to h[(i + k) % n_points]
            # overlap = \sum h[i] * (1 - h[(i + k) % n_points])
            overlap = cp.sum(cp.multiply(h_vars, 1 - h_k))
            constraints.append(overlap <= C)
        
        # Objective function
        objective = cp.Minimize(C)
        
        # Build and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_vars.value
                c5_opt = C.value
                
                if c5_opt < best_c5:
                    h = h_opt
                    best_c5 = c5_opt
                    iter_since_improvement = 0
                    best_h = h.copy()
                else:
                    iter_since_improvement += 1
                    if iter_since_improvement > max_no_improvement:
                        # Occasionally perturb the solution if no improvement
                        perturbation_magnitude = 0.01
                        perturbation = np.random.normal(0, perturbation_magnitude, n_points)
                        h = np.clip(h + perturbation, 0, 1)
                        h = h * (n_points / 2 / np.sum(h))
                    
            else:
                # If optimization failed, try again with a slightly different problem
                h = np.random.rand(n_points)
                h = h * (n_points / 2 / np.sum(h))
                h = np.clip(h, 0, 1)
        except:
            # If optimization fails due to solver issues, generate fresh point
            h = np.random.rand(n_points)
            h = h * (n_points / 2 / np.sum(h))
            h = np.clip(h, 0, 1)
            
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2 / np.sum(h))

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
