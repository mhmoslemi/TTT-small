import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    particles = []
    velocities = []
    p_best = []

    h_init = project_to_feasible(h_init, n_points)

    # Initialize other particles randomly
    for i in range(n_particles - 1):

        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        p_best.append(h.copy())
        p_best_fitness.append(evaluate_c5(h, 2.0 / n_points))
    
    # Global best
    best_idx = np.argmin(p_best_fitness)
    g_best = p_best[best_idx].copy()

            velocities[i] = np.clip(velocities[i], velocity_clamp[0], velocity_clamp[1])

            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)

            current_fitness = evaluate_c5(particles[i], 2.0 / n_points)
