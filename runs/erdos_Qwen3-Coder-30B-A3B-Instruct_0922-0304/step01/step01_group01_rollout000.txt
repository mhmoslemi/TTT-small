import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select one individual using tournament selection"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    # Average the two parents
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate=0.1, noise_std=0.05):
    """Mutate individual with Gaussian noise"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(individual))
    individual = individual + mask * noise
    return individual

def genetic_algorithm(seed, n_points, budget_s, initial_h_values=None):
    np.random.seed(seed)
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3
    stagnation_generations = 20
    max_generations = 1000
    
    # Initialize population
    population = []
    if initial_h_values is not None:
        # Add initial solution if available
        h = project_to_feasible(initial_h_values.copy(), n_points)
        population.append(h)
    
    # Fill rest of population randomly
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial fitness
    fitnesses = []
    for h in population:
        fitness = evaluate_c5(h, 2.0 / n_points)
        fitnesses.append(fitness)
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Track stagnation
    last_improvement = 0
    generation = 0
    
    # Run genetic algorithm
    start_time = time.time()
    while time.time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for stagnation
        if generation - last_improvement > stagnation_generations:
            break
            
        # Create new population
        new_population = [population[0].copy()]  # Keep best
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutate(offspring, mutation_rate)
            
            # Project to feasible region
            offspring = project_to_feasible(offspring, n_points)
            
            new_population.append(offspring)
        
        population = new_population
        
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_c5(h, 2.0 / n_points)
            fitnesses.append(fitness)
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_h = population[current_best_idx].copy()
            last_improvement = generation
            
        generation += 1
    
    return best_h, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    # Determine number of points
    try:
        h_init = initial_h_values.copy()
        n_points = len(h_init)
    except:
        # Default construction
        n_points = 42
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(seed, n_points, budget_s, initial_h_values)
    
    # Ensure final solution meets constraints
    best_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    return (best_h, final_c5, n_points)
