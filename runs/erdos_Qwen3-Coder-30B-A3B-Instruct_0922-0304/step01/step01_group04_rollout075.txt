import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Define constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Initialize best solution
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    
    # Differential Evolution parameters
    maxiter = min(1000, int(budget_s / 0.5))
    popsize = 15
    mutation = 0.8
    recombination = 0.7
    
    try:
        # Run optimization
        result = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            tol=1e-6,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False,
        )
        
        if result.success:
            best_h = result.x.copy()
            # Enforce bounds and sum constraint
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)
            best_c5 = evaluate_c5(best_h, dx)
    except Exception as e:
        pass 

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
