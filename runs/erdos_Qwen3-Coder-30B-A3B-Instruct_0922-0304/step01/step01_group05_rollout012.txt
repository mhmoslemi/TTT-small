import numpy as np

    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Use a different optimization approach: a direct convex relaxation
    # We work with the frequency domain as suggested
    def frequency_optimize():
        h = Variable(n_points)
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2
        ]
        
        # Create a CVXPY problem with a placeholder for the objective
        # We minimize a variable that bounds all the correlation values
        max_overlap = Variable()
        # We'll approximate max overlap using some linearized bounds
        # For now, let's do a simpler approach: just optimize with trusts region and coordinate descent
        
        # Helper for correlation
        def corr_max(h_vals): 
            h_vals = np.array(h_vals)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
            
        # Use scipy optimize with bounds and constraints
        # The objective for trust region
        def objective(h):
            return corr_max(h)
        
        # Use different optimization approaches
        bounds = [(0, 1) for _ in range(n_points)]
        h_init = initial_h.copy()
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        res = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 500, 'ftol': 1e-7, 'gtol': 1e-7})
        
        if not res.success:
            # fallback to simpler approach
            h = np.array(initial_h_values)
            h = h * (n_points / 2) / np.sum(h)
            return h
            
        return res.x
    
    # Run optimization
    h_opt = frequency_optimize()
    
    # Re-normalize 
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Final computation of C5
    final_C5 = compute_C5(h_opt)
    
    return (h_opt.tolist(), final_C5, n_points)
