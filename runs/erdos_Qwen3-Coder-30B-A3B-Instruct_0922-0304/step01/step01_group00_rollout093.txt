import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style coordinate-wise updates
    max_iter = 10000
    line_search_precision = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Compute the gradient of L(k_max) with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution to overlap at shift k_max from changing h[i]
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max]
        
        # Find index that decreases overlap the most (minimize gradient)
        i_star = np.argmin(grad)
        
        # Line search along direction e_{i_star}
        alpha = 0.0
        best_h_i_star = h[i_star]
        best_overlap = max_overlap
        
        # Simple line search
        step_size = 0.01
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            h_test = h.copy()
            h_test[i_star] += step_size
            
            # Ensure bounds
            if h_test[i_star] > 1:
                h_test[i_star] = 1
            elif h_test[i_star] < 0:
                h_test[i_star] = 0
                
            # Maintain sum constraint
            diff = h_test[i_star] - h[i_star]
            if i_star + k_max < n_points:
                h_test[i_star + k_max] -= diff
                if h_test[i_star + k_max] < 0:
                    h_test[i_star + k_max] = 0
                if h_test[i_star + k_max] > 1:
                    h_test[i_star + k_max] = 1
                    
            # Ensure overall sum constraint
            sum_diff = np.sum(h_test) - n_points / 2
            if abs(sum_diff) > 1e-10:
                # Distribute the error to other elements
                h_test += sum_diff / n_points
                
            h_test = np.clip(h_test, 0, 1)
            
            # Compute new overlap
            overlap_test = correlate(h_test, 1 - h_test, mode="full")
            max_overlap_test = np.max(overlap_test) * dx
            
            if max_overlap_test <= best_overlap:
                best_overlap = max_overlap_test
                best_h_i_star = h_test[i_star]
                h = h_test.copy()
            else:
                break
                
            step_size *= 0.9
        
        h[i_star] = best_h_i_star
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
