import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        # Create binary-like initial configuration
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1
        np.random.shuffle(h)
        
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1e-3
    gamma = 0.995
    alpha = 0.1
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 500
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current overlaps using FFT for efficiency
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Temperature cooling
        T = T0 * (gamma ** iteration)
        
        # Update with stochastic perturbation
        h_new = h - alpha * grad + T * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to feasible domain (renormalize to sum = n/2)
        h_new = h_new * (len(h) / 2.0) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept with Metropolis criterion
        if new_c5 < best_c5 or np.random.random() < np.exp(-(new_c5 - best_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
        
        # Adaptive learning rate decay
        if iter_since_improvement > max_no_improvement:
            alpha *= 0.95
            iter_since_improvement = 0
            
        # Binary perturbation every 5000 iterations
        if iteration % 5000 == 0 and iteration > 0:
            # Create new binary pattern with same sum
            binary_h = np.zeros(len(h))
            binary_h[:len(h)//2] = 1.0
            np.random.shuffle(binary_h)
            h = binary_h * (len(h) / 2.0) / np.sum(binary_h)
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
