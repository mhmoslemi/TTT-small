import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Initialize parameters
    max_iter = 1000
    step_size = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential Convex Programming
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find maximum
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 <= 0.3807:
            break
            
        # Compute gradient w.r.t. h for the maximum overlap shift
        # Gradient of L(k) = sum h[i] * (1 - h[i+k]) 
        # Gradient[i] = 1 - 2*h[i] for the k-th shift
        grad = np.zeros(n_points)
        if max_k < n_points:
            # For shift k, gradient is 1 - 2*h for first n-k elements
            for i in range(n_points - max_k):
                grad[i] += 1 - 2 * h[i + max_k]
        else:
            # For negative shifts (reverse), the first k elements
            k = max_k - (n_points - 1)
            for i in range(k):
                grad[i + max_k - k] += 1 - 2 * h[i]

        # Update step: move in direction that reduces the largest overlap
        h_new = h - step_size * grad
        
        # Project to feasible region [0,1] and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new

    return (best_h.tolist(), best_c5, n_points)
