import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto feasible region: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

        gradient = np.zeros(n_points)

                gradient[i] = 1 - 2 * h[i] - 2 * h[j]
        
        # Update step
        alpha = alpha_init * (gamma ** t)
        h_new = h - alpha * gradient

        h_new = project_h(h_new, n_points)
