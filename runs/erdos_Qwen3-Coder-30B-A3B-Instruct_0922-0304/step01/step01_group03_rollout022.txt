import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1
        # Random shuffle and renormalize
        np.random.shuffle(h)
        h = h * (n_points / 2.0 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_interval = 1000
    num_children = 10
    
    # Tree structure: node = (h, c5, visits, children)
    tree = {}
    root_key = tuple(h)
    tree[root_key] = [h.copy(), best_c5, 1, {}]
    
    def ucb_value(node_key, parent_visits):
        c5, visits, _ = tree[node_key][1], tree[node_key][2], tree[node_key][3]
        if visits == 0:
            return float('inf')
        return c5 + exploration_weight * np.sqrt(np.log(parent_visits) / visits)
    
    def mutate_h(current_h):
        h_new = current_h.copy()
        num_to_mutate = max(1, int(mutation_rate * len(h_new)))
        indices = np.random.choice(len(h_new), num_to_mutate, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.1)
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points / 2.0 / np.sum(h_new))
        return h_new
    
    def evaluate_c5(h, dx):
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * dx
    
    def get_key(h_array):
        return tuple(np.round(h_array, decimals=10))
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Selection phase: traverse tree to leaf
        current_key = root_key
        path = []
        
        while True:
            node = tree[current_key]
            children = node[3]
            
            if not children or len(children) < num_children:
                break
            
            # UCB selection
            ucb_values = []
            for child_key in children:
                ucb_values.append(ucb_value(child_key, node[2]))
            
            max_ucb_idx = np.argmax(ucb_values)
            selected_child = list(children.keys())[max_ucb_idx]
            path.append(current_key)
            current_key = selected_child
        
        # Expansion phase: generate children
        current_h = tree[current_key][0]
        children_keys = []
        
        for _ in range(num_children):
            child_h = mutate_h(current_h)
            child_key = get_key(child_h)
            if child_key not in tree:
                child_c5 = evaluate_c5(child_h, dx)
                tree[child_key] = [child_h, child_c5, 0, {}]
                children_keys.append(child_key)
            else:
                children_keys.append(child_key)
        
        # Add children to parent node
        tree[current_key][3] = {k: None for k in children_keys}
        
        # Simulation phase: evaluate newly added nodes
        for child_key in children_keys:
            child_node = tree[child_key]
            child_node[2] += 1  # Increment visit count
            if child_node[1] < best_c5:
                best_c5 = child_node[1]
                best_h = child_node[0].copy()
            
            # Update parent visits and weights if not root
            if path:
                parent_key = path[-1]
                parent_node = tree[parent_key]
                parent_node[2] += 1  # Parent visit count increased
        
        # Backpropagation phase: propagate best value back up
        for parent_key in reversed(path):
            parent_node = tree[parent_key]
            parent_c5 = parent_node[1]
            children = parent_node[3]
            
            if children:
                min_child_c5 = float('inf')
                for child_key in children:
                    child_c5 = tree[child_key][1]
                    if child_c5 < min_child_c5:
                        min_child_c5 = child_c5
                if min_child_c5 < parent_c5:
                    parent_node[1] = min_child_c5
        
        # Periodic reset to maintain diversity
        if iteration % reset_interval == 0 and iteration > 0:
            for key in list(tree.keys()):
                if random.random() < 0.1:  # Reset 10% of nodes
                    current_node_h = tree[key][0]
                    reset_h = mutate_h(current_node_h)
                    reset_key = get_key(reset_h)
                    tree[reset_key] = [reset_h, evaluate_c5(reset_h, dx), 0, {}]
                    del tree[key]
                    # Update this node in parents' children list if exists
                    for parent_key in list(tree.keys()):
                        parent_children = tree[parent_key][3]
                        if key in parent_children:
                            del parent_children[key]
                            parent_children[reset_key] = None
    
    return (best_h.tolist(), best_c5, len(best_h))
