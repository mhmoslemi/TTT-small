import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Binary initialization: 22 ones, 23 zeros (adjusting for even split)
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0
        np.random.shuffle(h)
        
    # Normalize to sum to 22.5 (exact value for n_points=45)
    h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated Annealing Parameters
    temp = 1e-3
    temp_min = 1e-8
    gamma = 0.995
    alpha = 0.1
    max_no_improvement = 500
    iter_since_improvement = 0
    
    # Reset frequency
    reset_freq = 5000
    
    # Optimization Loop
    for iteration in range(100000):  # Up to 100k iterations with early break
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute overlaps
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Simulated Annealing Update
        noise = np.random.normal(0, temp, len(h))
        h_new = h - alpha * grad + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (22.5 / np.sum(h_new))  # Renormalize to exact sum
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update based on Metropolis criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
        
        # Update temperature
        if iter_since_improvement > max_no_improvement:
            temp *= gamma
            iter_since_improvement = 0
            if temp < temp_min:
                temp = temp_min
        
        # Periodic reset to binary configuration
        if iteration % reset_freq == 0 and iteration > 0:
            # Create new binary pattern
            binary_h = np.zeros(len(h))
            num_ones = 22
            indices = np.random.choice(len(h), size=num_ones, replace=False)
            binary_h[indices] = 1.0
            
            # Add some noise to avoid plateauing
            binary_h += np.random.normal(0, 0.05, len(h))
            binary_h = np.clip(binary_h, 0, 1)
            binary_h = binary_h * (22.5 / np.sum(binary_h))
            h = binary_h.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
