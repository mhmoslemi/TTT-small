```python
import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum=h_sum, 0<=h[i]<=1"""
    # L1-ball projection to enforce sum constraint
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / h_sum
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def select_parents(population, fitnesses):
    """Tournament selection"""
    selected = []
    n_parents = len(population) // 2
    for _ in range(n_parents):
        i1, i2 = np.random.choice(len(population), 2, replace=False)
        if fitnesses[i1] < fitnesses[i2]:
            selected.append(population[i1])
        else:
            selected.append(population[i2])
    return selected

def crossover(parent1, parent2):
    """Segment-wise crossover"""
    n = len(parent1)
    if n == 1:
        return parent1.copy(), parent2.copy()
    # Random segment crossover
    start = np.random.randint(0, n)
    end = np.random.randint(start + 1, n + 1)
    child1 = np.concatenate([parent1[:start], parent2[start:end], parent1[end:]])
    child2 = np.concatenate([parent2[:start], parent1[start:end], parent2[end:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.02):
    """Gaussian mutation"""
    if np.random.random() < mutation_rate:
        h = h + np.random.normal(0, noise_std, len(h))
        h = np.clip(h, 0, 1)
    return h

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / n_points
    population_size = 50
    max_generations = int(budget_s * 20)  # conservative estimate

    # Initialize population with one individual from initial_h_values if possible
    population = []
    fitnesses = []
    
    # Ensure initial_h_values available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values)
        h_start = project_to_feasible(h_start, n_points)
        population.append(h_start)
        fitnesses.append(evaluate_c5(h_start, dx))
    else:
        # Generate structured initial individual
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        population.append(h_start)
        fitnesses.append(evaluate_c5(h_start, dx))

    # Fill rest of population with random individuals
    for _ in range(population_size - 1):
        h_rand = np.zeros(n_points)
        h_rand[:n_points//2] = 1.0
        h_rand += np.random.normal(0, 0.05, n_points)
        h_rand = project_to_feasible(h_rand, n_points)
        population.append(h_rand)
        fitnesses.append(evaluate_c5(h_rand, dx))
        
    best_idx = np.argmin(fitnesses)
    best_c5 = fitnesses[best_idx]
    best_h = population[best_idx].copy()

    # Generation loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Generate offspring
        offspring = []
        while len(offspring) < population_size - 1:
            if len(parents) < 2:
                break
            p1, p2 = np.random.choice(parents, 2, replace=False)
            if np.random.random() < 0.8:  # crossover rate
                c1, c2 = crossover(p1, p2)
                c1 = project_to_feasible(c1, n_points)
                c2 = project_to_feasible(c2, n_points)
                c1 = mutate(c1)
                c2 = mutate(c2)
                offspring.extend([c1, c2])
            else:
                # Clone with mutation
                p = np.random.choice(parents)
                c = mutate(p.copy())
                c = project_to_feasible(c, n_points)
                offspring.append(c)
                
        # Maintain population size
        offspring = offspring[:population_size - 1]
        
        # Evaluate offspring
        new_fitnesses = [evaluate_c5(h, dx) for h in offspring]
        
        # Combine and select next generation
        population = [population[best_idx]] + offspring  # keep the best
        fitnesses = [best_c5] + new_fitnesses
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
```
