def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy the constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
