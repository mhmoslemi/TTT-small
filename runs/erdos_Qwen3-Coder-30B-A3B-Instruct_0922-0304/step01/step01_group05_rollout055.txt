import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with 42 points as in the previous approach
    n_points = 42
    dx = 2.0 / n_points
    
    # Get initial solution
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx

    # Precompute the correlation matrix for efficiency
    def compute_correlation_matrix(h):
        # Make sure h has the right sum
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Padding to avoid wraparound effects in correlation
        pad_len = n_points - 1
        h_padded = np.pad(h, pad_len, mode='constant', constant_values=0)
        # Correlate h with (1-h) 
        corr = np.correlate(h_padded, 1 - h, mode="full")
        return corr
    
    # Use scipy optimize with a better starting point and single objective
    def objective(h):
        return compute_C5(h)

    # Normalize h to sum to n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Initial guess
    h_guess = h_init.copy()
    h_guess = h_guess * (n_points / 2) / np.sum(h_guess)
    
    # Constraints for optimization
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    # Bounds for each h[i] in [0, 1] 
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization using SLSQP method
    res = minimize(objective, h_guess, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-8}, callback=None)
    
    # Extract solution
    best_h = res.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
