import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Fallback to a reasonable construction
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation scaled by dx
        return np.max(corr) * dx
    
    # Use CVXPY for convex optimization
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Add overlap constraints
    for k in range(-n_points + 1, n_points):
        if k < 0:
            # Forward shift: h[i] * (1 - h[i-k]) for i = 0..n_points-1+|k|
            indices = slice(0, n_points + k)
            shift_indices = slice(-k, n_points)
        else:
            # Backward shift: h[i] * (1 - h[i+k]) for i = 0..n_points-1-k
            indices = slice(0, n_points - k)
            shift_indices = slice(k, n_points)
            
        if k == 0:
            # Just sum of h * (1-h) for all i
            constraint = sum(h[i] * (1 - h[i]) for i in range(n_points)) <= C * dx
        else:
            # More complex correlation term
            if k > 0:
                constraint = sum(h[i] * (1 - h[i + k]) for i in range(n_points - k)) <= C * dx
            else:
                constraint = sum(h[i] * (1 - h[i - k]) for i in range(-k, n_points)) <= C * dx
        constraints.append(constraint)
    
    # Define the problem
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    # Solve using SCS with enhanced settings
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-4, 
                   scale=10, adaptive_scale=False, warm_start=True)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_solution = h.value
            final_C5 = C.value
        else:
            h_solution = h_init
            final_C5 = compute_C5(h_init)
    except Exception as e:
        # If QCP fails, fallback to direct optimization
        print(f"CVXPY error: {e}")
        h_solution = h_init
        final_C5 = compute_C5(h_init)

    # Ensure the result is valid
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
    final_C5 = compute_C5(h_solution)
    
    # Final check and cleanup
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.95:
        return (h_solution.tolist(), final_C5, n_points)
    
    return (h_solution.tolist(), final_C5, n_points)
