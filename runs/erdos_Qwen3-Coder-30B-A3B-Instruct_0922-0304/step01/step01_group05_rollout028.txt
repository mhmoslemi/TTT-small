import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Use SCS solver with entropy correction for better handling of constraints
    # Direct QCP formulation
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, 
                   sum(h) == n_points / 2,
                   C >= 0]
    
    # Add overlap constraints for all lags
    for k in range(-n_points + 1, n_points):
        # indices for valid positions
        if k >= 0:
            valid_range = range(n_points - k)
            corr = sum(h[i] * (1 - h[i + k]) for i in valid_range)
        else:
            valid_range = range(-k, n_points)
            corr = sum(h[i + k] * (1 - h[i]) for i in valid_range)
        constraints.append(corr <= C * dx)
    
    # Objective
    objective = Minimize(C)
    
    prob = Problem(objective, constraints)
    
    # Solve using SCS with entropy=True which is good for handling quadratic constraints
    prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-5, 
               use_indirect=True, max_rank_update=1000, 
               entropy=True, warm_start=True)
    
    # Extract solution
    solution_h = h.value
    if solution_h is None:
        # If convex solver fails, fallback to SLSQP
        from scipy.optimize import minimize
        
        def compute_C5(h_vals):
            h_vals = np.array(h_vals)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
        
        def obj(h_vals):
            return compute_C5(h_vals)
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        res = minimize(obj, initial_h.copy(), method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-6})
        if res.success:
            solution_h = res.x * (n_points / 2) / np.sum(res.x)
        else:
            solution_h = initial_h.copy()
    
    # Final computation of C5 (always compute to ensure correctness)
    h_final = solution_h * (n_points / 2) / np.sum(solution_h)
    corr = np.correlate(h_final, 1 - h_final, mode="full")
    final_C5 = np.max(corr) * dx
    
    # Ensure the final h is scaled correctly to the constraint
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), final_C5, n_points)
