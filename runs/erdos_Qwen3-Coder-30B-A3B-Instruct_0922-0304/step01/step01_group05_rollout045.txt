import time
from scipy.signal import correlate

def evaluate_h(h_values, n_points):
    """Evaluate C5 for given h_values."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all correlations for different lags
    corr = correlate(h, 1 - h, mode="full")
    # Find maximum correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    return h

        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
    return h

    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

    """Perform uniform crossover on two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

    mask = np.random.rand(len(h)) < mutation_rate

    h = h + noise * mask
