import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h to satisfy sum(h) = n_points / 2 and 0 <= h[i] <= 1
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def crossover(parent1, parent2):
    # Segment-wise crossover
    n_points = len(parent1)
    crossover_point = random.randint(1, n_points - 1)
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, noise_scale=0.02):
    # Gaussian mutation
    if random.random() < mutation_rate:
        individual += np.random.normal(0, noise_scale, len(individual))
    return individual

def genetic_algorithm(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Create a simple binary initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Fill remaining population
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_solution = None
    best_c5 = float('inf')
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_solution = population[0].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection (top 50%)
        parents = population[:population_size//2]
        
        # Create new population
        new_population = parents[:2]  # Keep top 2
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parent1, parent2 = random.sample(parents, 2)
            
            # Crossover
            if random.random() < 0.8:
                offspring1, offspring2 = crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1)
            offspring2 = mutate(offspring2)
            
            # Project to feasible
            offspring1 = project_to_feasible(offspring1, n_points)
            offspring2 = project_to_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
            
        population = new_population[:population_size]
        
        # Relaxation step every 10 generations
        if generation % 10 == 0:
            # Solve relaxation using gradient descent on the best individual
            best_h = population[0]
            step = 0.005
            inner_iters = 50
            
            for _ in range(inner_iters):
                corr = correlate(best_h, 1 - best_h, mode="full")
                c5_val = np.max(corr) * dx
                
                grad = np.zeros_like(best_h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = best_h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_val) / eps
                
                best_h -= step * grad
                best_h = project_to_feasible(best_h, n_points)
            
            # Replace worst individual
            population[-1] = best_h.copy()
    
    return (best_solution.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
