import cvxpy as cp

    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h[:23] = 1.0
        np.random.shuffle(h)
        # Renormalize to exactly sum to n_points / 2
        h = h * (n_points / 2.0 / np.sum(h))
    
    # Make sure we have exactly the right sum
    h = h * (n_points / 2.0 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000

        if time.time() - start_time > budget_s - 0.1:
            break

        if iteration % 4 == 0:
            # Try optimization with CVXPY if time allows
            try:
                h_new, c5_new = lp_optimize_h(h.copy(), n_points, dx, start_time, budget_s)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
            except:
                pass  # Fall back to gradient descent approach
                
        # Gradient descent approach
        if iter_since_improvement < max_no_improvement:
            eps = 1e-5
            grad = np.zeros_like(h)
            for i in range(len(h)):
                # Compute gradient via finite differences
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                overlap_plus = evaluate_c5(h_plus, dx)
                overlap_minus = evaluate_c5(h_minus, dx)
                grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
                
            # Update
            alpha = 0.05
            h_temp = h - alpha * grad
            h_temp = np.clip(h_temp, 0, 1)
            scale = (n_points / 2.0) / np.sum(h_temp)
            h_new = h_temp * scale
            
            # Evaluate
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                iter_since_improvement = 0

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx

def lp_optimize_h(h, n_points, dx, start_time, budget_s):
    """Solve a LP relaxation for better h"""
    if time.time() - start_time > budget_s - 1.0:
        return h, evaluate_c5(h, dx)

    # Use CVXPY
    h_var = cp.Variable(n_points, name='h')
    C = cp.Variable(name='C')

    # Constraints
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2.0, C >= 0]
    max_lags = 2 * n_points - 1  # Full correlation length

    for k in range(max_lags):
        # Compute the overlap expression for lag k
        # correlation with h and (1-h) in appropriate alignment
        overlap_expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap_expr += h_var[i] * (1 - h_var[j])
        constraints.append(overlap_expr <= C * dx)
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            if h_opt is not None:
                return h_opt, C.value
    except:
        pass
    
    # If LP fails, just use current h
    return h, evaluate_c5(h, dx)
