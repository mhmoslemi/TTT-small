import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_algorithm(seed, budget_s, n_points=42):
    np.random.seed(seed)
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3
    stagnation_generations = 20
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
        h_init = project_to_feasible(h_init, n_points)
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_feasible(h_init, n_points)
        population.append(h_init)
        
    # Fill rest of population randomly
    for _ in range(pop_size - len(population)):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_c5(h, 2.0 / n_points))

    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    generation = 0
    stagnant_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover: average
            child1 = (parent1 + parent2) / 2.0
            child2 = (parent1 + parent2) / 2.0
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child1 += np.random.normal(0, 0.05, n_points)
            if np.random.rand() < mutation_rate:
                child2 += np.random.normal(0, 0.05, n_points)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            new_fitness.append(evaluate_c5(h, 2.0 / n_points))
        
        # Replace worst with new population
        population = new_population
        fitness = new_fitness
        
        # Update best solution
        cur_best_idx = np.argmin(fitness)
        if fitness[cur_best_idx] < best_c5:
            best_c5 = fitness[cur_best_idx]
            best_h = population[cur_best_idx].copy()
            stagnant_count = 0
        else:
            stagnant_count += 1
            
        if stagnant_count >= stagnation_generations:
            break
            
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5 = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h, c5, n_points)
