import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Check if we have enough time to do convex optimization
    if time.time() - start_time > budget_s - 10:
        # Not enough time, return initial solution
        return (h_init, evaluate_c5(h_init, dx), n_points)
    
    # Prepare the convex optimization problem
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Create constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Bounds constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Maximum overlap constraint
    # For each possible shift k in the full correlation
    # We form the constraint sum(h[i] * (1 - h[i+k])) <= t
    # Since we're checking for maximum overlap, we will construct
    # the full correlation expression and bound t
    h_full = np.concatenate([h, np.zeros(n_points - 1)])
    one_minus_h_full = np.concatenate([1 - h, np.zeros(n_points - 1)])
    
    # Manually check for all shifts and bound by maximum
    # This is a simplified convex relaxation approach
    for k in range(-n_points + 1, n_points):
        # Compute correlation for shift k using linear constraints
        if k >= 0:
            # Take h[:n-k] and (1-h)[k:]
            h_part = h[:n_points-k]
            one_minus_h_part = 1 - h[k:]
        else:
            # Take h[-k:] and (1-h)[:n+k]
            h_part = h[-k:]
            one_minus_h_part = 1 - h[:n_points+k]
        
        # Linearization of h[i] * (1 - h[i+k]) <= t is tricky
        # Instead we enforce max of valid correlations <= t
        corr_expr = cp.sum(cp.multiply(h_part, one_minus_h_part))
        constraints.append(corr_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try with MOSEK or if that fails, fallback to ECOS
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_optimal = h.value
            c5_bound = t.value
        else:
            # Fallback to initial solution or another method
            h_optimal = h_init.copy()
            c5_bound = evaluate_c5(h_optimal, dx)
    except Exception as e:
        # If CVXOPT fails, use the initial solution
        h_optimal = h_init.copy()
        c5_bound = evaluate_c5(h_optimal, dx)
    
    # Ensure final results are valid
    h_optimal = np.clip(h_optimal, 0, 1)
    if np.abs(np.sum(h_optimal) - target_sum) > 1e-6:
        # Renormalize
        h_optimal = h_optimal * target_sum / np.sum(h_optimal)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_optimal, dx)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (h_optimal, final_c5, n_points)
