import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to evaluate objective
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint function for sum
    def sum_constraint(h):
        return np.sum(h) - target_sum
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization
    de_options = {'maxiter': 100, 'popsize': 40, 'mutation': 0.8, 'recombination': 0.9, 'seed': seed}
    de_result = differential_evolution(objective, bounds, **de_options)
    
    # Get DE result
    best_h = de_result.x.copy()
    best_c5 = de_result.fun
    
    # Repair solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    
    # Local refinement with L-BFGS-B
    def obj_func(h):
        return evaluate_c5(h, dx)
    
    def jac_func(h):
        # For now just return a dummy gradient - numerical approx is too slow
        return np.zeros_like(h)
    
    # SCIPY L-BFGS-B
    lbfgs_options = {'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
    try:
        # Use bounds in constraints
        constraints = [{'type': 'eq', 'fun': sum_constraint}]
        # Initial guess is the best DE solution
        result = minimize(obj_func, best_h, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraints, options=lbfgs_options)
        if result.success:
            final_h = result.x.copy()
            final_c5 = evaluate_c5(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Ensure final constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = evaluate_c5(best_h, dx)
    
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    # Ensure we don't run out of time - return at least a valid solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass
    
    return (best_h, best_c5, n_points)
