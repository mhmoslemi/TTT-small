import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Repair function to ensure constraints
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Objective function for optimization
    def objective(h):
        h = np.abs(h)
        h = repair_h(h)
        return evaluate_c5(h, dx)
    
    # Differential Evolution for global search
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    popsize = 40
    maxiter = 100
    mutation = 0.8
    recombination = 0.9
    
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            disp=False,
            callback=lambda x, convergence: None,
        )
        
        if de_result.success:
            best_h = de_result.x
        else:
            best_h = h_init
            
        # Refine with L-BFGS-B
        # Ensure best_h is valid
        best_h = repair_h(best_h)
        
        # Define constraints for local optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        def constraint_bounds(h):
            return np.minimum(h, 1 - h)
        
        # Optimize with L-BFGS-B
        options = {'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_sum},
            options=options,
            callback=lambda x: None,
        )
        
        if result.success:
            final_h = result.x
        else:
            final_h = best_h
            
        final_h = repair_h(final_h)
        final_c5 = evaluate_c5(final_h, dx)
        
    except Exception as e:
        # Fallback to initial construction
        final_h = h_init.copy()
        final_c5 = evaluate_c5(final_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(final_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in final_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (final_h, final_c5, n_points)
