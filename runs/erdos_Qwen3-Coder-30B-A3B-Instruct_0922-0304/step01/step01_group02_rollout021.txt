import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h, dx):
    correlation = signal.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Simple step function with integral 1
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T_min = 1e-6
    max_iter = 10000
    max_evals = 100000
    
    # Control variables
    T = T0
    eval_count = 0
    iter_count = 0
    
    # Main optimization loop
    while T > T_min and eval_count < max_evals and iter_count < max_iter:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Make a perturbation
        h_new = h.copy()
        
        # Perturb 5% of points
        n_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Apply small random changes
        for i in indices:
            # Small random adjustment
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
            
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to preserve integral constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        scale_factor = target_sum / current_sum
        h_new = h_new * scale_factor
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        eval_count += 1
        iter_count += 1
        
        # Accept or reject the new solution
        delta = C5_new - best_C5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
        
        # Cool down
        T *= alpha
    
    return (best_h, best_C5, n_points)
