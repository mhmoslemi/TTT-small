import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 1000
    step_size = 0.01
    max_grad_iter = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 value
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Gradient ascent on the maximum overlap
        h_temp = h.copy()
        
        # Compute overlap
        corr = correlate(h_temp, 1 - h_temp, mode="full")
        c5_val = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Gradient approximation via finite differences
        grad = np.zeros_like(h_temp)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_temp.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_val) / eps
            
        # Update h
        h_temp -= step_size * grad
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        h = h_temp
        
        # Repair feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust step size based on progress
        if iteration % 50 == 0 and iteration > 0:
            step_size *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
