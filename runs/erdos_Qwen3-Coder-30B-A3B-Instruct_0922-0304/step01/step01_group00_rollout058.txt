import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Simulated Annealing parameters
    T = 0.05  # Initial temperature
    cooling_rate = 0.9998
    max_iter = 100000
    
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    iterations = 0
    stagnation_count = 0
    last_improvement = 0
    
    while iterations < max_iter and time.time() - start_time < budget_s - 1:
        # Perturbation: 70% swap, 30% noise
        if np.random.random() < 0.7:
            # Flip operation: randomly pick two indices and swap values
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Noise perturbation: add Gaussian noise to two random points
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            noise = np.random.normal(0, 0.01, 2)
            h_new[i] += noise[0]
            h_new[j] += noise[1]
            # Maintain sum constraint
            sum_diff = np.sum(h_new) - (n_points / 2)
            if abs(sum_diff) > 1e-10:
                h_new[i] -= sum_diff / 2
                h_new[j] -= sum_diff / 2
                
        # Clip to feasible range
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain exact sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject the perturbation
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / T):
            h = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improvement = iterations
        
        # Update temperature and iteration count
        T *= cooling_rate
        iterations += 1
        
        # Early stopping if we've converged or hit target
        if best_c5 <= 0.3807:
            break
            
        # Reset stagnation if we made progress
        if iterations - last_improvement > 5000:
            stagnation_count += 1
            if stagnation_count > 10:
                break
    
    return (best_h.tolist(), best_c5, n_points)
