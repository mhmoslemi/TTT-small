import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to optimize
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Set bounds for each variable [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add a penalty for sum constraint violation
    def penalty_sum(h):
        return max(0, abs(np.sum(h) - target_sum))
    
    def modified_objective(h):
        penalty = 1e6 * penalty_sum(h)
        return objective(h) + penalty
    
    # Use differential evolution for global optimization
    try:
        result = differential_evolution(
            modified_objective,
            bounds,
            maxiter=100,
            popsize=20,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            atol=1e-6,
            rtol=1e-6
        )
        
        if result.success:
            current_h = result.x
        else:
            current_h = h_init.copy()
    except Exception:
        # Fallback to initial construction
        current_h = h_init.copy()
    
    # Final processing
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
