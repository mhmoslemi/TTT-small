import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s with small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated annealing parameters
    temperature = 0.05
    cooling_factor = 0.9998
    max_iter = 100000
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Make a perturbation
        h_new = h.copy()
        
        # Choose perturbation type (70% swap, 30% noise)
        if np.random.random() < 0.7:
            # Binary swap: flip one random bit
            i = np.random.randint(n_points)
            h_new[i] = 1 - h_new[i]
        else:
            # Gaussian noise: add to two random points and adjust to preserve sum
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            noise = np.random.normal(0, 0.01, 2)
            h_new[i1] += noise[0]
            h_new[i2] += noise[1]
            
            # Adjust to maintain sum = n_points/2
            current_sum = np.sum(h_new)
            adjustment = (n_points / 2 - current_sum) / 2
            h_new[i1] -= adjustment
            h_new[i2] -= adjustment
            
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < best_c5 or np.random.random() < np.exp((best_c5 - new_c5) / temperature):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature
        temperature *= cooling_factor
        
        iter_count += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
