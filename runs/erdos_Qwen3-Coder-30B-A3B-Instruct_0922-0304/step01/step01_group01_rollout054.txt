import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(fitnesses, tournament_size=3):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(fitnesses), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return best_idx

def crossover(parent1, parent2):
    """Perform average crossover"""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate=0.1, std=0.05):
    """Apply Gaussian mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    individual += mask * np.random.normal(0, std, len(individual))
    return individual
