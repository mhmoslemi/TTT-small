import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    # Project h to feasible region: [0,1]^n and sum = n_points/2
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2, n_points):
    # Segment-wise crossover
    segment_length = max(1, n_points // 10)
    offspring = parent1.copy()
    for i in range(0, n_points, segment_length):
        end = min(i + segment_length, n_points)
        if np.random.random() > 0.5:
            offspring[i:end] = parent2[i:end]
    return offspring

def mutate(individual, n_points, mutation_rate=0.2):
    # Gaussian noise mutation
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, 0.02, n_points)
        individual += noise
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 0
    max_generations = budget_s * 20  # Rough estimate
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Generate remaining individuals
    for i in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(population):
        c5 = evaluate_c5(h, dx)
        fitness.append(c5)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic Algorithm Loop
    gen = 0
    while gen < max_generations and time.time() - start_time < budget_s - 1:
        gen += 1
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//2]
        parents = [population[i] for i in top_indices]
        
        # Create new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            
            offspring = crossover(parent1, parent2, n_points)
            offspring = mutate(offspring, n_points)
            offspring = project_feasible(offspring, n_points)
            new_population.append(offspring)
        
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = []
        for i, h in enumerate(population):
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Relaxation step every 10 generations
        if gen % 10 == 0:
            # Find the best individual
            best_idx = np.argmin(fitness)
            h_best = population[best_idx]
            
            # Local refinement using gradient descent
            h_refined = h_best.copy()
            step_size = 0.005
            for _ in range(50):
                corr = correlate(h_refined, 1 - h_refined, mode="full")
                c5_val = np.max(corr) * dx
                
                # Simple gradient estimation
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = project_feasible(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - c5_val) / eps
                
                # Update
                h_refined -= step_size * grad
                h_refined = project_feasible(h_refined, n_points)
                
                # Early stopping if improvement is small
                new_c5 = evaluate_c5(h_refined, dx)
                if new_c5 >= c5_val:
                    break
                    
            # Replace worst individual with refined solution
            worst_idx = np.argmax(fitness)
            population[worst_idx] = h_refined.copy()
            fitness[worst_idx] = evaluate_c5(h_refined, dx)
            
            # Update best if improved
            if fitness[worst_idx] < best_c5:
                best_c5 = fitness[worst_idx]
                best_h = h_refined.copy()
                
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
