import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    # Normalize to correct sum
    h = h * (22.5 / np.sum(h))
    
    def evaluate_c5(h_vals):
        """Evaluate C5 given h and dx."""
        dx = 2.0 / len(h_vals)
        correlate_result = signal.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlate_result) * dx
    
    def project_h(h_vals):
        """Project h to feasible set: [0,1] and sum=22.5"""
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (22.5 / np.sum(h_vals))
        return h_vals
    
    def nelder_mead_step_simplex(h_vals, f_vals, alpha=1.0, gamma=2.0, rho=0.5, sigma=0.5):
        """Perform one Nelder-Mead step."""
        n = len(h_vals)
        # Order vertices by function value
        indices = np.argsort(f_vals)
        h_sorted = h_vals[indices]
        f_sorted = f_vals[indices]
        
        # Compute centroid excluding worst point
        centroid = np.mean(h_sorted[:-1], axis=0)
        
        # Reflection
        h_reflect = centroid + alpha * (centroid - h_sorted[-1])
        h_reflect = project_h(h_reflect)
        f_reflect = evaluate_c5(h_reflect)
        
        if f_sorted[0] <= f_reflect < f_sorted[-2]:
            # Accept reflection
            h_vals[indices[-1]] = h_reflect
            f_vals[indices[-1]] = f_reflect
        elif f_reflect < f_sorted[0]:
            # Expansion
            h_expand = centroid + gamma * (h_reflect - centroid)
            h_expand = project_h(h_expand)
            f_expand = evaluate_c5(h_expand)
            if f_expand < f_reflect:
                h_vals[indices[-1]] = h_expand
                f_vals[indices[-1]] = f_expand
            else:
                h_vals[indices[-1]] = h_reflect
                f_vals[indices[-1]] = f_reflect
        else:
            # Contraction
            if f_reflect < f_sorted[-1]:
                # Outside contraction
                h_contract = centroid + rho * (h_reflect - centroid)
            else:
                # Inside contraction
                h_contract = centroid + rho * (h_sorted[-1] - centroid)
            h_contract = project_h(h_contract)
            f_contract = evaluate_c5(h_contract)
            if f_contract < f_sorted[-1]:
                h_vals[indices[-1]] = h_contract
                f_vals[indices[-1]] = f_contract
            else:
                # Shrink
                for i in range(1, n):
                    h_vals[indices[i]] = h_sorted[0] + sigma * (h_vals[indices[i]] - h_sorted[0])
                    h_vals[indices[i]] = project_h(h_vals[indices[i]])
                    f_vals[indices[i]] = evaluate_c5(h_vals[indices[i]])
    
    # Create initial simplex
    n_points = len(h)
    simplex_size = 0.1
    initial_h = h.copy()
    simplex = [initial_h]
    for _ in range(n_points):
        pert = np.random.normal(0, simplex_size, n_points)
        h_pert = initial_h + pert
        h_pert = project_h(h_pert)
        simplex.append(h_pert)
    
    # Initial function values
    f_vals = [evaluate_c5(h) for h in simplex]
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    max_no_improvement = 1000
    iter_since_improvement = 0
    
    iter_count = 0
    while time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        if iter_count % 1000 == 0:
            # Perturb the best solution
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            best_h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            best_h = project_h(best_h)
            # Update simplex with best solution
            simplex[0] = best_h
            f_vals[0] = evaluate_c5(best_h)
        
        # Nelder-Mead step
        nelder_mead_step_simplex(np.array(simplex), np.array(f_vals))
        
        # Update best
        min_idx = np.argmin(f_vals)
        if f_vals[min_idx] < best_c5:
            best_h = simplex[min_idx].copy()
            best_c5 = f_vals[min_idx]
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > max_no_improvement:
            # Introduce random perturbation to escape local minimum
            h = best_h.copy()
            if np.random.rand() > 0.5:
                # Add Gaussian noise to part of the points
                perturb_idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                h[perturb_idx] += np.random.normal(0, 0.05, len(perturb_idx))
            else:
                # Try small perturbations to whole array
                h += np.random.normal(0, 0.02, n_points)
            h = project_h(h)
            best_h = h
            best_c5 = evaluate_c5(best_h)
            iter_since_improvement = 0
            
            # Rebuild simplex
            simplex = [best_h]
            for _ in range(n_points):
                pert = np.random.normal(0, simplex_size, n_points)
                h_pert = best_h + pert
                h_pert = project_h(h_pert)
                simplex.append(h_pert)
            f_vals = [evaluate_c5(h) for h in simplex]

    return (best_h.tolist(), best_c5, len(best_h))
