import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

            if i - k >= 0:
                grad[i] = 1 - 2 * h[i] - 2 * h[i - k]
        return grad
    
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:

        current_c5 = evaluate_c5(h, 2.0 / n_points)

            best_c5 = current_c5
            best_h = h.copy()

        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (n_points - 1)

        gradient = compute_gradient(h, k)

        m = beta1 * m + (1 - beta1) * gradient

        h = np.clip(h, 0, 1)

        # Decay learning rate
        alpha *= decay_rate
