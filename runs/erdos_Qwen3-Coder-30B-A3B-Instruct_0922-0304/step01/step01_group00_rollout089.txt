import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project to L1 ball with sum = n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def segment_crossover(parent1, parent2, n_points):
    # Segment-wise crossover
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_scale=0.02):
    # Gaussian noise mutation
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, noise_scale, len(h))
        h = h + noise
    return h

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    
    # Seed with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        # Fallback initialization: half 1s and half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
    
    population.append(h.copy())
    
    # Generate rest of population
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for i, h in enumerate(population):
            c5 = evaluate_c5(h, dx)
            fitness.append((c5, i))
        
        # Sort by C5
        fitness.sort()
        
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = population[fitness[0][1]].copy()
        
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50% as parents
        num_parents = population_size // 2
        parents = [population[fitness[i][1]] for i in range(num_parents)]
        
        # Generate new population
        new_population = [parents[0].copy()]  # Keep best
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.random() < 0.8 and len(parents) >= 2:
                i1, i2 = np.random.choice(len(parents), 2, replace=False)
                child1, child2 = segment_crossover(parents[i1], parents[i2], n_points)
            else:
                # Random selection
                i1 = np.random.choice(len(parents))
                child1 = parents[i1].copy()
                child2 = parents[i1].copy()
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Projection
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s, n_points=70)
    return (h.tolist(), c5_bound, 70)
