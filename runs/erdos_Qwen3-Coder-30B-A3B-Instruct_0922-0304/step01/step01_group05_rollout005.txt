import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # More sophisticated optimization approach
    # Start with a coarse grid for exploration
    coarse_points = 20
    coarse_dx = 2.0 / coarse_points
    coarse_h = np.interp(np.linspace(0, 2, coarse_points), 
                         np.linspace(0, 2, len(initial_h)), 
                         initial_h)
    
    # Coarse optimization with CVXPY
    def coarse_optimization():
        h_vars = cp.Variable(coarse_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == coarse_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        for k in range(-coarse_points + 1, coarse_points):
            # Compute correlation for this lag
            corr = 0
            if k >= 0:
                corr = cp.sum(cp.multiply(h_vars[:coarse_points-k], 
                                       1 - h_vars[k:]))
            else:
                corr = cp.sum(cp.multiply(h_vars[-k:], 
                                       1 - h_vars[:coarse_points+k]))
            constraints.append(corr <= C * coarse_dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_vars.value
        except:
            pass
        return coarse_h
    
    # Try coarse optimization first
    try:
        coarse_result = coarse_optimization()
        if coarse_result is not None:
            coarse_h = coarse_result
    except:
        pass
    
    # Refine to fine grid
    fine_h = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, coarse_points), 
                       coarse_h)
    
    # Fine optimization using scipy
    def objective_fine(h):
        return compute_C5(h)
    
    # Constraints for fine grid
    def constraint_sum_fine(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum_fine}]
    
    # Use the coarse result as initial point
    h_init = fine_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Optimization with different methods to avoid local minima
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Try multiple starting points
    for i in range(5):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Perturb initial solution
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        res = minimize(objective_fine, h_perturbed, method='SLSQP', 
                       bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-6})
        
        if res.success:
            h_candidate = res.x
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            candidate_C5 = compute_C5(h_candidate)
            
            if candidate_C5 < best_C5:
                best_h = h_candidate
                best_C5 = candidate_C5
    
    # Final refinement with direct optimization
    if time.time() - start_time < budget_s * 0.9:
        res = minimize(objective_fine, best_h, method='SLSQP', 
                       bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_C5 = compute_C5(h_final)
            
            if final_C5 < best_C5:
                best_h = h_final
                best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
