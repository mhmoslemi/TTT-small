import numpy as np
import time
from scipy import signal
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:

        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    T0 = 1e-3
    cooling_rate = 0.995
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    T = T0
    temp_decay = 0.99
    
    def compute_correlation(h):
        # Type: A convolution with 1-h
        # Since correlation(x,y) = conv(x, flip(y)), and flip(1-h) = 1 - flip(h)
        # We compute conv(h, 1-h) using FFT
        h_full = np.concatenate([h, np.zeros(len(h)-1)])
        ones_full = np.concatenate([np.ones(len(h)), np.zeros(len(h)-1)])
        h_flip = np.flip(h)
        ones_flip = np.flip(ones_full)
        corr = np.real(ifft(fft(h_full) * fft(ones_flip)))
        corr = corr[:len(h)]  
        return corr * dx
    
    def find_max_overlap(h):
        corr = compute_correlation(h)
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        return max_overlap, max_k
    
    def evaluate_gradient(h, k_max):
        # Estimate gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            _, k_plus = find_max_overlap(h_plus)
            _, k_minus = find_max_overlap(h_minus)
            grad[i] = (k_plus - k_minus)/(2.0*eps)
        return grad
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        max_overlap, max_k = find_max_overlap(h)
        
        # Compute gradient
        grad = evaluate_gradient(h, max_k)
        
        # Update with simulated annealing
        h_new = h - alpha * grad + T * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        scale = len(h) / 2.0 / np.sum(h_new)
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update using Metropolis criterion
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            iter_since_improvement += 1
            if iter_since_improvement > max_no_improvement:
                if alpha > alpha_min:
                    alpha *= gamma
                iter_since_improvement = 0
        
        T *= temp_decay
        
        # Reset periodically to escape local minima
        if iteration % 5000 == 0 and iteration > 0:
            # Perturb with a new binary configuration
            h_binary = np.zeros(len(h))
            # Select coalesced blocks of 1's and 0's
            block_len = np.random.randint(3, 10)
            num_blocks = len(h) // block_len
            for i in range(num_blocks):
                start = i * block_len
                end = min(start + block_len, len(h))
                if i % 2 == 0:
                    h_binary[start:end] = 1
            h = h_binary.copy()
            h = h * (len(h) / 2.0 / np.sum(h))
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
