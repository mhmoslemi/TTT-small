import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a sawtooth pattern as initial guess
        n_points = 42
        h_init = np.array([1.0 - abs(np.sin(i)) for i in range(n_points)])
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Gradient-based optimization with constraints
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Optimization with multiple restarts
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Try a few random restarts to avoid local minima
    for restart in range(5):
        # Random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = current_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        
        # Optimize from this point
        current_h_opt = h_perturbed.copy()
        try:
            result = minimize(
                objective,
                current_h_opt,
                method='SLSQP',
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-6},
                callback=lambda x: None,
            )
            if result.success:
                optimized_h = result.x
                optimized_h = np.clip(optimized_h, 0, 1)
                optimized_h = optimized_h * target_sum / np.sum(optimized_h)
                optimized_c5 = evaluate_c5(optimized_h, dx)

                if optimized_c5 < best_c5:
                    best_c5 = optimized_c5
                    best_h = optimized_h.copy()
        except:
            pass
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s * 0.9:
            break
    
    # Final check to make sure constraints hold
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    assert np.isclose(np.sum(final_h), target_sum, rtol=1e-4)
    assert all(0 <= h <= 1 for h in final_h)
    
    return (final_h, final_c5, n_points)
