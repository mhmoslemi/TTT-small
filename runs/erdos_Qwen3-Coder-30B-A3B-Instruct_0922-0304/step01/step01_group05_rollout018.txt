import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initial construction
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Objective function
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Repair function to maintain constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (n_points / 2) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # PSO parameters
    num_particles = 50
    max_iter_pso = 500
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initialize particles and velocities
    particles = np.array([h_init + 0.2 * (2 * np.random.rand(n_points) - 1) for _ in range(num_particles)])
    particles = np.array([repair(p) for p in particles])
    velocities = 0.1 * np.random.randn(num_particles, n_points)
    
    # Evaluate initial particles
    fitness = np.array([compute_C5(p) for p in particles])
    personal_best = particles.copy()
    personal_best_fitness = fitness.copy()
    global_best_idx = np.argmin(fitness)
    global_best = particles[global_best_idx].copy()
    global_best_fitness = fitness[global_best_idx]
    
    # PSO iterations
    last_improvement = 0
    for iteration in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / max_iter_pso)
        
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair to maintain constraints
            particles[i] = repair(particles[i])
            
            # Evaluate new position
            new_fitness = compute_C5(particles[i])
            
            # Update personal best
            if new_fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = new_fitness
                    last_improvement = iteration
        
        # Perturbation if no improvement in last 20 iterations
        if iteration - last_improvement > 20:
            for i in range(num_particles):
                if np.random.rand() < 0.3:
                    particles[i] = repair(particles[i] + 0.1 * (2 * np.random.rand(n_points) - 1))
                    fitness[i] = compute_C5(particles[i])
                    if fitness[i] < personal_best_fitness[i]:
                        personal_best[i] = particles[i].copy()
                        personal_best_fitness[i] = fitness[i]
                        if fitness[i] < global_best_fitness:
                            global_best = particles[i].copy()
                            global_best_fitness = fitness[i]
                            last_improvement = iteration
    
    # Final check
    best_h = global_best
    final_C5 = compute_C5(best_h)
    
    # Refine with local optimization
    res = minimize(lambda x: compute_C5(x), best_h, method='SLSQP',
                   bounds=[(0, 1) for _ in range(n_points)],
                   options={'maxiter': 200, 'ftol': 1e-8})
    
    if res.success:
        final_h = res.x
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_C5 = compute_C5(final_h)
    else:
        final_h = best_h
        final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
