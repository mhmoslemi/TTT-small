import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()

    # Coordinate-wise Frank-Wolfe like optimization
    max_iter = 10000
    line_search_precision = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap_val = overlap[k_max] * dx
        
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Calculate gradient of overlap w.r.t. h
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max]
        
        # Find index that maximizes negative gradient
        i_star = np.argmin(grad)
        
        # Line search to find optimal step along gradient direction
        h_old = h.copy()
        current_c5 = max_overlap_val
        
        # Simple line search along armijo rule
        alpha = 1.0
        step_size = 0.01
        
        # Find alpha that minimizes overlap loss
        min_grad = grad[i_star]
        if min_grad >= 0:
            # Current direction doesn't improve outcome, so just adjust in small increments
            alpha = 0.005
        else:
            # Try to reduce overlap by adjusting the selected index
            alpha = 1.0
            found = False
            while not found and alpha > 1e-6:
                h_new = h.copy()
                h_new[i_star] += alpha * step_size * np.sign(min_grad)
                h_new = np.clip(h_new, 0, 1)
                # Adjust other indices to satisfy constraint sum(h) = n_points/2
                total_diff = np.sum(h_new) - n_points / 2
                # Simple adjustment: redistribute error
                if abs(total_diff) > 1e-10:
                    # Distribute to indices that might help fix discrepancy
                    indices = list(range(n_points))
                    if i_star < n_points - 1:
                        indices.remove(i_star)
                    if i_star > 0:
                        indices.remove(i_star - 1)  # may not be valid
                    if len(indices) > 0:
                        # Add to a few random indices to balance sum
                        for idx in indices[:min(5, len(indices))]:
                            if abs(total_diff) < 1e-10:
                                break
                            delta = min(abs(total_diff), 0.01)
                            if total_diff > 0:
                                h_new[idx] = max(0, h_new[idx] - delta)
                                total_diff -= delta
                            else:
                                h_new[idx] = min(1, h_new[idx] + delta)
                                total_diff += delta
                h_new = np.clip(h_new, 0, 1)
                
                # Recalculate overlaps
                overlap_new = correlate(h_new, 1 - h_new, mode="full")
                max_overlap_new = np.max(overlap_new) * dx
                
                if max_overlap_new < current_c5:
                    h = h_new
                    current_c5 = max_overlap_new
                    found = True
                else:
                    alpha *= 0.5  # Reduce step size
                    
        if not found:
            # Fallback to small step in negative gradient direction
            h[i_star] = max(0, min(1, h[i_star] - step_size * (-min_grad)))
            # Normalize sum
            h = h * (n_points / 2) / np.sum(h)
            
    # Ensure final feasibility
    if np.abs(np.sum(h) - n_points / 2) > 1e-5:
        h = h * (n_points / 2) / np.sum(h)
        
    # Evaluate final result
    final_c5 = evaluate_c5(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
