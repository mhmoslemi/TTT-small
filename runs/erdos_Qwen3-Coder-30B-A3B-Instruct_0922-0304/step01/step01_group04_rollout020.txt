import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():

        n_points = len(h_init)
    else:

        n_points = 42
        h_init = np.ones(n_points) * 0.5
        # Modify to a more structured pattern
        for i in range(n_points):
            h_init[i] = 0.5 + 0.3 * np.sin(np.pi * i / (n_points - 1))
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Prepare for optimization
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    max_iter = 500
    iter_count = 0
    
    # Gradient-based optimization with trust region
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Mutate current h slightly and evaluate
        epsilon = 0.01
        h_trial = current_h + np.random.normal(0, epsilon, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        # Normalize to target sum
        h_trial = h_trial * target_sum / np.sum(h_trial)
        
        trial_c5 = evaluate_c5(h_trial, dx)
        
        if trial_c5 < current_c5:
            current_h = h_trial
            current_c5 = trial_c5
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        iter_count += 1
        
        # Early exit if improvement is very small
        if iter_count > 10 and abs(current_c5 - best_c5) < 1e-6:

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, final_c5, n_points)
