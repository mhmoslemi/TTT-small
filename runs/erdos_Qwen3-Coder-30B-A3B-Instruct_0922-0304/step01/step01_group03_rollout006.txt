import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 45
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable initial h
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
    
    # Renormalize to ensure sum = n_points/2
    h = h * (n_points / 2.0 / np.sum(h))
    h = np.clip(h, 0, 1)
    population.append(h.copy())
    
    # Generate additional random individuals
    for i in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0 / np.sum(h))
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    best_h = population[0].copy()
    best_c5 = evaluate_c5(population[0], 2.0 / n_points)
    
    # Evolution parameters
    generations = 2000
    elite_size = population_size // 10
    crossover_rate = 0.7
    mutation_rate = 0.05
    noise_std = 0.05
    stagnation_limit = 50
    stagnation_count = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Elitism: Keep top individuals
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = population[np.random.randint(0, population_size//2)]
            parent2 = population[np.random.randint(0, population_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            mutate_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
            for i in mutate_indices:
                child[i] += np.random.normal(0, noise_std)
            
            # Clamp and renormalize
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2.0 / np.sum(child))
            
            new_population.append(child)
        
        population = new_population
        
        # Diversification if no improvement
        if stagnation_count > stagnation_limit:
            for i in range(int(population_size * 0.2)):
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0 / np.sum(h))
                h = np.clip(h, 0, 1)
                population[np.random.randint(0, population_size)] = h
            noise_std *= 0.9  # Reduce mutation noise
            stagnation_count = 0

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
