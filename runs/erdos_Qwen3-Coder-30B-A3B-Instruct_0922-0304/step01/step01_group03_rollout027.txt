import numpy as np
import time
from scipy import signal
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    max_no_improvement = 100
    iter_since_improvement = 0
    
    # Optimization params
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    T0 = 1e-3
    cooling_rate = 0.995
    max_iter = 10000
    
    # Simulated annealing params
    temp = T0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute correlation using FFT for efficiency
        h_padded = np.pad(h, (0, len(h)-1), mode='constant')
        ones_padded = np.pad(1 - h, (0, len(h)-1), mode='constant')
        correlation_fft = ifft(fft(h_padded) * np.conj(fft(ones_padded)))
        overlaps = np.real(correlation_fft[:len(h)]) * dx
        
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            # Compute with padding
            h_plus_padded = np.pad(h_plus, (0, len(h_plus)-1), mode='constant')
            h_minus_padded = np.pad(h_minus, (0, len(h_minus)-1), mode='constant')
            ones_plus = np.pad(1 - h_plus, (0, len(h_plus)-1), mode='constant')
            ones_minus = np.pad(1 - h_minus, (0, len(h_minus)-1), mode='constant')
            
            corr_plus = ifft(fft(h_plus_padded) * np.conj(fft(ones_plus)))
            corr_minus = ifft(fft(h_minus_padded) * np.conj(fft(ones_minus)))
            
            overlap_plus = np.real(corr_plus[:len(h_plus)]) * dx
            overlap_minus = np.real(corr_minus[:len(h_minus)]) * dx
            grad[i] = (overlap_plus[max_k] - overlap_minus[max_k]) / (2.0 * eps)
        
        # Update using gradient step with stochastic perturbation
        noise = np.random.normal(0, temp, len(h))
        h_new = h - alpha * grad + noise
        
        # Enforce bounds and renormalize
        h_new = np.clip(h_new, 0, 1)
        scale = 22.5 / np.sum(h_new)
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)
        
        # Final renormalization 
        h_new = h_new * (22.5 / np.sum(h_new))
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Acceptance probability
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
            
        # Adaptively decrease temperature
        if iter_since_improvement > 100:
            temp *= cooling_rate
            iter_since_improvement = 0
            
        # Adaptive learning rate decay
        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:
                alpha *= gamma
            iter_since_improvement = 0
            
        # Periodic full reset to binary candidate
        if iteration % 5000 == 0 and iteration > 0:
            new_h = np.zeros(len(h))
            # Small binary improvement
            ones_count = 22
            new_h[:ones_count] = 1.0
            np.random.shuffle(new_h)
            # Renormalize
            new_h = new_h * (22.5 / np.sum(new_h))
            h = new_h
            
        # Occasionally reset to binary to avoid local optima
        if iteration % 200 == 0:
            new_h = np.zeros(len(h))
            ones_count = 22
            new_h[:ones_count] = 1.0
            np.random.shuffle(new_h)
            # Renormalize
            new_h = new_h * (22.5 / np.sum(new_h))
            h = new_h
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
