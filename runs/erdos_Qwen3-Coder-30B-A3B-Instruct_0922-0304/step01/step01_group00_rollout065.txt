import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral C5."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_jacobian(h, k_max):
    """Compute the gradient of L(k_max) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_max < n:
            grad[i] += 1 - h[i + k_max]
        if i >= k_max:
            grad[i] -= h[i]
    return grad

def line_search(h, i_star, grad, k_max, dx):
    """Perform line search to minimize L(k_max) along the gradient direction."""
    n = len(h)
    # Original value
    corr = correlate(h, 1 - h, mode="full")
    original_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = original_c5
    best_alpha = 0.0
    
    # Try different alpha values
    alphas = np.linspace(0, 1, 100)
    for alpha in alphas:
        h_new = h.copy()
        h_new[i_star] = np.clip(h[i_star] - alpha * grad[i_star], 0, 1)
        # Adjust other variable to keep sum constraint
        delta = h_new[i_star] - h[i_star]
        # If we are reducing i_star, we must distribute to other indices
        if delta < 0:
            # Try to increase h[i_star + k_max] if possible
            if i_star + k_max < n:
                h_new[i_star + k_max] = np.clip(h_new[i_star + k_max] - delta, 0, 1)
                delta = h_new[i_star + k_max] - h[i_star + k_max]
        # Re-normalize
        scale = (n / 2) / np.sum(h_new)
        h_new = h_new * scale
        # Recompute overlap
        corr_new = correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_alpha = alpha
    
    return best_h, best_c5, best_alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute Jacobian of L(k_max)
        grad = compute_jacobian(h, k_max)
        
        # Select coordinate to update
        i_star = np.argmin(grad)
        
        # Perform line search
        h, new_c5, _ = line_search(h, i_star, grad, k_max, dx)
        
        # Update h to be feasible
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
