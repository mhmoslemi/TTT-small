import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T = 0.05
    cooling_factor = 0.9998
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    iter_count = 0
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        iter_count += 1
        # Acceptance probability threshold
        accept_prob = np.exp((best_c5 - evaluate_c5(h, dx)) / T)
        
        # Perturb h
        h_new = h.copy()
        if np.random.rand() < 0.7:
            # Binary swap: flip two random bits
            i, j = np.random.randint(0, n_points, 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Gaussian noise: add to two random points
            i, j = np.random.randint(0, n_points, 2)
            noise_i = np.random.normal(0, 0.01)
            noise_j = np.random.normal(0, 0.01)
            h_new[i] += noise_i
            h_new[j] += noise_j
            
            # Adjust to maintain sum constraint
            h_new[i] = np.clip(h_new[i], 0, 1)
            h_new[j] = np.clip(h_new[j], 0, 1)
            diff = (h_new[i] + h_new[j]) - (h[i] + h[j])
            if diff > 0:
                # Adjust other points to compensate
                indices = list(range(n_points))
                indices.remove(i)
                indices.remove(j)
                for k in indices[:min(2, len(indices))]:
                    h_new[k] -= diff / 2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing conditions
        if c5_new < best_c5 or np.random.rand() < accept_prob:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        T *= cooling_factor
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
