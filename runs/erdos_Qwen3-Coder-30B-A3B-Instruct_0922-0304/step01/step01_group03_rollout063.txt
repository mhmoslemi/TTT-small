import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))
    
    # Normalize to exactly sum to 22.5
    h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    max_no_improvement = 1000
    iter_since_improvement = 0
    
    # Create initial simplex
    simplex = []
    simplex.append(h.copy())
    
    # Add perturbed points to the simplex
    for i in range(n_points):
        h_perturb = h + simplex_size * np.random.randn(n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        scale = 22.5 / np.sum(h_perturb)
        h_perturb = h_perturb * scale
        simplex.append(h_perturb.copy())
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        scale = 22.5 / np.sum(h_vec)
        h_vec = h_vec * scale
        return h_vec
    
    def evaluate_c5(h_vec, dx):
        correlate_result = signal.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlate_result) * dx
    
    def get_simplex_values(simplex):
        return [evaluate_c5(h, dx) for h in simplex]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate simplex
        simplex_values = get_simplex_values(simplex)
        indices = np.argsort(simplex_values)
        simplex = [simplex[i] for i in indices]
        simplex_values = [simplex_values[i] for i in indices]
        
        if simplex_values[0] < best_c5:
            best_c5 = simplex_values[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
        
        if iter_since_improvement > max_no_improvement:
            # Reinitialize simplex with better point
            h_new = project_to_feasible(best_h + 0.01 * np.random.randn(n_points))
            simplex[0] = h_new.copy()
            h_new = project_to_feasible(best_h + 0.1 * np.random.randn(n_points))
            simplex[1] = h_new.copy()
            iter_since_improvement = 0
        
        # Select best and worst vertices
        best_simplex = simplex[:-1]
        worst = simplex[-1]
        worst_value = simplex_values[-1]
        
        # Compute centroid (excluding worst)
        centroid = np.mean(best_simplex, axis=0)
        
        # Reflect
        h_reflect = centroid + reflection_factor * (centroid - worst)
        h_reflect = project_to_feasible(h_reflect)
        reflect_value = evaluate_c5(h_reflect, dx)
        
        if reflect_value < simplex_values[0]:
            # Expand
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = project_to_feasible(h_expand)
            expand_value = evaluate_c5(h_expand, dx)
            if expand_value < reflect_value:
                simplex[-1] = h_expand.copy()
                simplex_values[-1] = expand_value
            else:
                simplex[-1] = h_reflect.copy()
                simplex_values[-1] = reflect_value
        else:
            if reflect_value < worst_value:
                simplex[-1] = h_reflect.copy()
                simplex_values[-1] = reflect_value
            else:
                # Contract
                h_contract = centroid + contraction_factor * (worst - centroid)
                h_contract = project_to_feasible(h_contract)
                contract_value = evaluate_c5(h_contract, dx)
                if contract_value < worst_value:
                    simplex[-1] = h_contract.copy()
                    simplex_values[-1] = contract_value
                else:
                    # Shrink simplex
                    best = simplex[0]
                    for i in range(1, len(simplex)):
                        simplex[i] = best + 0.5 * (simplex[i] - best)
                        simplex[i] = project_to_feasible(simplex[i])
        
        # Periodic perturbation
        if iteration % 1000 == 0 and iteration > 0:
            perturb_idx = np.random.choice(len(h), size=int(0.05 * len(h)), replace=False)
            perturbed = best_h.copy()
            perturbed[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            perturbed = project_to_feasible(perturbed)
            simplex[0] = perturbed.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))
