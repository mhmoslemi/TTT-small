import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        # Renormalize to sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Use fewer lags for efficiency while still capturing the maximum
    # Lags to consider for constraint tightening
    max_lag = 2 * len(h) - 2  # full correlation length
    lags_to_consider = np.arange(max_lag + 1)
    
    # Optimization parameters
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Use a convex relaxation approach with CVXPY
        n_points = len(h)
        h_var = cp.Variable(n_points, name='h')
        C = cp.Variable(1, name='C')

        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(cp.sum(h_var) == 22.5)  # sum = n_points/2 = 22.5
        constraints.append(C >= 0)
        
        # For each lag, create the constraint representing integral of h(x)(1-h(x+k))
        # This is a discrete approximation of the overlap
        for k in lags_to_consider[::2]:  # to reduce constraint count
            # Dynamic index computation
            if k == 0:
                # overlap at k=0: sum h[i] * (1 - h[i]) = sum h[i] - sum h[i]^2
                overlap = cp.sum(cp.multiply(h_var, 1 - h_var))
            else:
                # For k > 0, aggregate: sum h[i] * (1 - h[(i+k)%n_points])
                overlap = 0
                for i in range(n_points):
                    idx = (i + k) % n_points
                    overlap += h_var[i] * (1 - h_var[idx])
            constraints.append(C >= overlap * dx)

        # Objective: minimize maximum overlap C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)

        # Attempt solve with CVXPY using ECOS solver
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Ensure it meets exact constraints
                h_new = np.clip(h_new, 0, 1)
                scale = 22.5 / np.sum(h_new)
                h_new = h_new * scale

                # Re-evaluate the C5 bound for this new solution
                new_c5 = evaluate_c5(h_new, dx)
                
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    iter_since_improvement = 0
                    best_h = h.copy()
                else:
                    iter_since_improvement += 1
                    if iter_since_improvement > max_no_improvement:
                        # Reduce exploration and shuffle
                        h = np.random.permutation(h)
                        h = np.clip(h, 0, 1)
                        h = h * (22.5 / np.sum(h))
                        iter_since_improvement = 0
            else:
                # If infeasible, try with a different approach
                h = np.random.permutation(h)
                h = np.clip(h, 0, 1)
                h = h * (22.5 / np.sum(h))
        except Exception:
            # Fallback to finite-differences based gradient descent if CVXPY fails
            new_h = h.copy()
            last_best = evaluate_c5(h, dx)
            for i in range(len(h)):
                eps = 1e-5
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                overlap_plus = evaluate_c5(h_plus, dx)
                overlap_minus = evaluate_c5(h_minus, dx)
                # Approximate gradient
                grad = (overlap_plus - overlap_minus) / (2.0 * eps)
                new_h[i] = h[i] - 0.001 * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (22.5 / np.sum(new_h))
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
                if iter_since_improvement > max_no_improvement:
                    h = np.random.permutation(h)
                    h = np.clip(h, 0, 1)
                    h = h * (22.5 / np.sum(h))
                    iter_since_improvement = 0

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
