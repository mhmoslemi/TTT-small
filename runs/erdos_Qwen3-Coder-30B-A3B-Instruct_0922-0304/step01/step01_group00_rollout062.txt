import numpy as np
from scipy.signal import correlate
import time
from math import sqrt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project to sum = n_points / 2 and bounds [0, 1]
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / sum_h
    return h

def crossover(parent1, parent2):
    # Segment-wise crossover
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_std=0.02):
    n = len(individual)
    mask = np.random.random(n) < mutation_rate
    individual = individual.copy()
    individual[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return project_to_feasible(individual, n)

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Fallback to structured construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(pop_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Main loop
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness_scores = [evaluate_c5(h, dx) for h in population]
        
        # Sort by fitness (C5 values)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Selection: top 50%
        num_parents = pop_size // 2
        parents = population[:num_parents]
        
        # Create new population
        new_population = parents.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.random() < crossover_rate and len(parents) >= 2:
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                new_population.extend([child1, child2])
            else:
                # Just mutate a random individual
                individual = np.random.choice(parents)
                mutated = mutate(individual, mutation_rate)
                new_population.append(mutated)
        
        # Trim to exact population size
        population = new_population[:pop_size]
        generation += 1
        
        # Relaxation step every 10 generations
        if generation % relaxation_interval == 0 and len(population) > 0:
            # Improve the best individual via gradient ascent
            h = population[0].copy()
            step = 0.01
            for _ in range(20):
                h_ = h.copy()
                corr = correlate(h_, 1 - h_, mode="full")
                c5_val = np.max(corr) * dx
                grad = np.zeros_like(h_)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx 
                    grad[i] = (c5_plus - c5_val) / eps
                
                h_ -= step * grad
                h_ = np.clip(h_, 0, 1)
                h_ = h_ * (n_points / 2) / np.sum(h_)
                
                # Ensure it doesn't get worse
                corr_new = correlate(h_, 1 - h_, mode="full")
                new_c5 = np.max(corr_new) * dx
                if new_c5 < c5_val:
                    h = h_
                else:
                    break
                
            # Replace worst individual with improved version
            if len(population) > 0:
                population[-1] = h
            
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
