import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided

        n_points = len(h)

        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated annealing parameters
    T0 = 1.0
    alpha = 0.99
    min_T = 1e-8
    max_iter = 10000
    
    # Current solution
    current_h = h.copy()
    current_C5 = best_C5
    
    T = T0
    iter_count = 0
    
    while T > min_T and time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        
        # Make a perturbation
        new_h = current_h.copy()
        
        # Perturb 5-10% of the points
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # For each index, perturb it with a small random change
        for i in indices:
            # Adjust by a small random amount
            delta = np.random.normal(0, 0.1 * T)
            new_h[i] += delta
            
        # Ensure new_h stays in [0,1] 
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain the integral constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2.0
        
        if abs(current_sum - target_sum) > 1e-10:
            # Scale to maintain sum
            scale = target_sum / current_sum
            new_h = new_h * scale
            
        # Compute new energy
        new_C5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject the new solution
        delta_E = new_C5 - current_C5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h = new_h
            current_C5 = new_C5
            
            # Update best if better
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cool down temperature
        T *= alpha
        
        # Occasionally reset T if stuck
        if iter_count % 500 == 0:
            T = max(T * 0.95, min_T)
    
    return (best_h, best_C5, n_points)
