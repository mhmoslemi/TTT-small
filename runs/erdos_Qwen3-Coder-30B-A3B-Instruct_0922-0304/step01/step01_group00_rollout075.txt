import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, multiply
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 500
    learning_rate = 0.01
    max_gradient_iter = 100
    # Regularization to enforce constraints
    reg_strength = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Stop if we achieved target
        if best_c5 <= 0.3807:
            break
            
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        base_overlap = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            plus_overlap = evaluate_c5(h_plus, dx)
            grad[i] = (plus_overlap - base_overlap) / eps
        
        # Update h using gradient descent (with a projected step)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally use a more sophisticated optimization approach
        if iteration % 10 == 0:
            # Try to solve a constrained optimization problem to reduce overlap
            try:
                h_new = Variable(n_points)
                # Objective: minimize max overlap (approximated)
                # We minimize the maximum correlation term
                # For convexity, we use a quadratic approximation
                overlaps = []
                h_corr = correlate(h, 1-h, mode='full')
                max_corr_idx = np.argmax(h_corr)
                
                # Create a simple convex constraints set
                constraints = [
                    cp_sum(h_new) == n_points / 2,
                    h_new >= 0,
                    h_new <= 1
                ]
                
                prob = Problem(Minimize(cp_sum(multiply(h_new, 1 - h_new))), constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h = np.array(h_new.value).flatten()
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except:
                pass  # Fallback to gradient step
    
    # Final check and return
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
