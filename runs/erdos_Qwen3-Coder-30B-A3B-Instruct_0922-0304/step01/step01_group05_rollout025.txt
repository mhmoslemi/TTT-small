import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    best_C5 = float('inf')
    best_h = None
    
    # Convex optimization approach
    # Use SCS solver with entropy=true for better handling of quadratic constraints
    
    # Set up problem variables
    h = Variable(n_points)
    C = Variable()
    
    # Objective: minimize C
    objective = Minimize(C)
    
    # Constraints
    constraints = []
    
    # Integral constraint
    constraints.append(sum(h) == n_points / 2)
    
    # Bounds
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # All overlap constraints
    # For each possible shift k, we add a constraint for the correlation
    # We compute the overlap integral for each shift
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # For positive k: h[i] * (1 - h[i+k]) for i = 0..(n_points-1-k)
            indices_i = np.arange(n_points - k)
            indices_j = np.arange(k, n_points)
            if len(indices_i) > 0:
                overlap_expr = sum(h[indices_i] * (1 - h[indices_j]))
            else:
                overlap_expr = 0
        else:
            # For negative k: h[i+k] * (1 - h[i]) for i = -k..(n_points-1)
            indices_i = np.arange(-k, n_points)
            indices_j = np.arange(n_points + k)
            if len(indices_i) > 0:
                overlap_expr = sum(h[indices_j] * (1 - h[indices_i]))
            else:
                overlap_expr = 0
        
        # Add constraint: overlap_expr <= C * dx
        constraints.append(overlap_expr <= C * dx)
    
    # Formulate and solve the problem
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-6, 
                   acceleration_lookback=10, entropy=True)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            sol_h = h.value
            sol_C = C.value
            # Re-normalize h to meet constraint
            sol_h = sol_h * (n_points / 2) / np.sum(sol_h)
            # Compute final C5
            final_C5 = max(np.correlate(sol_h, 1 - sol_h, mode="full")) * dx
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = sol_h
    except:
        pass
    
    # Fall back to gradient based method if needed
    if best_h is None:
        # Use trust-constr with expensive bounds check 
        from scipy.optimize import minimize
        
        def compute_C5(h_vals):
            h = np.array(h_vals)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        h_init = initial_h.copy()
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        def objective(h):
            return compute_C5(h)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        res = minimize(objective, h_init, method='trust-constr', 
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 1000, 'gtol': 1e-6, 'ftol': 1e-6})
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_C5 = compute_C5(final_h)
            best_C5 = final_C5
            best_h = final_h
    
    if best_h is None:
        # Fallback to initial guess
        best_h = initial_h.copy()
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_C5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_C5, n_points)
