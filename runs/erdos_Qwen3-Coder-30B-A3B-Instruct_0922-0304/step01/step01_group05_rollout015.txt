import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 5000
    num_particles = 50
    w_max, w_min = 0.9, 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initial h values
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles
    particles = []
    velocities = []
    personal_bests = []
    personal_best_scores = []
    
    for _ in range(num_particles):
        h = h_init + 0.2 * (2 * np.random.rand(n_points) - 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(0.1 * np.random.randn(n_points))
        personal_bests.append(h.copy())
        personal_best_scores.append(compute_C5(h, dx))
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO loop
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        iteration += 1
        
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_bests[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Compute overlap
            current_score = compute_C5(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_bests[i] = particles[i].copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = particles[i].copy()
                    last_improvement = iteration
                    
        # Perturb if no improvement for a long time
        if iteration - last_improvement > 20:
            for i in range(num_particles):
                noise = 0.01 * np.random.randn(n_points)
                particles[i] = np.clip(particles[i] + noise, 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                personal_bests[i] = particles[i].copy()
                personal_best_scores[i] = compute_C5(particles[i], dx)
            last_improvement = iteration
    
    return (global_best.tolist(), global_best_score, n_points)

def compute_C5(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
