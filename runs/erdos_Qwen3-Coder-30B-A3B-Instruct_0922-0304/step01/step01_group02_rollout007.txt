import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Define constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]

        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_func(h_vec):

    ]
