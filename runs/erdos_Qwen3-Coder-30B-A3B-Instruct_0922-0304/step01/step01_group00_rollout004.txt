import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective: minimize maximum overlap
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with adaptive learning rate
    max_iter = 1000
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and identify worst case shift
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient with respect to h
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Finite difference approximation of gradient
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)

            grad[i] = (c5_plus - current_c5) / eps
        
        # Update h using gradient descent
