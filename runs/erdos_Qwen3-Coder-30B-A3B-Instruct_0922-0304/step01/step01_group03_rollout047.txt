import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    
    # Normalize to sum to 22.5 (n_points/2)
    h = h * (22.5 / np.sum(h))
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    max_iter = 5000
    no_improvement_iter = 0
    max_no_improvement = 100
    alpha = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Randomly perturb
        h_new = h.copy()
        num_perturbed = max(1, int(0.08 * len(h)))
        perturb_idx = np.random.choice(len(h), size=num_perturbed, replace=False)
        perturb_values = np.random.normal(0, 0.05, num_perturbed)
        h_new[perturb_idx] += perturb_values
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (22.5 / np.sum(h_new))
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_iter = 0
            best_h = h.copy()
        else:
            no_improvement_iter += 1
            if no_improvement_iter > max_no_improvement:
                if alpha > 1e-6:
                    alpha *= 0.95
                no_improvement_iter = 0
        
        # Periodic restart
        if iteration % 500 == 0 and iteration > 0:
            h = best_h.copy()
            h += np.random.normal(0, 0.02, len(h))
            h = np.clip(h, 0, 1)
            h = h * (22.5 / np.sum(h))
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
