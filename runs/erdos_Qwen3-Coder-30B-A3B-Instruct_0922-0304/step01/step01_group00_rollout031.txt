import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure it's feasible
        h = np.clip(h, 0, 1)
        # Scale to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create structured binary fallback
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.05
    cooling_rate = 0.9998
    max_iter = int(1e5)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a perturbation
        h_new = h.copy()
        
        # 70% chance of bit flip, 30% chance of noise
        if np.random.random() < 0.7:
            # Flip a random point between 0 and 1
            idx = np.random.randint(n_points)
            h_new[idx] = 1 - h_new[idx]
        else:
            # Gaussian noise on two points
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            noise = np.random.normal(0, 0.01, 2)
            diff = noise[0] - noise[1]
            h_new[idx1] += noise[0]
            h_new[idx2] += noise[1]
            # Adjust to maintain sum
            h_new[idx1] -= diff * (n_points / 2 - np.sum(h_new)) / (n_points * 2)
            h_new[idx2] += diff * (n_points / 2 - np.sum(h_new)) / (n_points * 2)
        
        # Clip values
        h_new = np.clip(h_new, 0, 1)
        
        # Force sum constraint by rescaling
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject the change
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / T):
            h = h_new
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Update temperature
        T *= cooling_rate
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
