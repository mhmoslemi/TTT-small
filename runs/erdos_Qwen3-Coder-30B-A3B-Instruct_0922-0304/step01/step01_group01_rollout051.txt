import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, n_points):
    """Project h onto feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection"""
    tournament_indices = random.sample(range(len(population)), k)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2):
    """Average two parents to create offspring"""
    offspring = (parent1 + parent2) / 2.0
    return offspring

def mutate(individual, mutation_rate, n_points):
    """Mutate individual with Gaussian noise"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project_feasible(mutated, n_points)
