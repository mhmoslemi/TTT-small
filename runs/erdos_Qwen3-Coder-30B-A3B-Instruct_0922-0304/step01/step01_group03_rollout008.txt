import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a random vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # GA parameters
    pop_size = 50
    generations = 2000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_rate = 0.1
    max_no_improvement = 50
    mutation_noise = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h.copy())
    else:
        population.append(np.random.rand(len(h)))
    
    for _ in range(pop_size - 1):
        h_new = np.random.rand(len(h))
        h_new = h_new * (22.5 / np.sum(h_new))
        population.append(h_new)
    
    no_improvement_count = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h_ind in population:
            c5 = evaluate_c5(h_ind, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Elitism
        elite_size = int(elite_rate * pop_size)
        elite = population[:elite_size]
        
        # Create new population by crossover and mutation
        new_population = elite[:]
        
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                # Select parents
                parent1_idx = np.random.randint(0, pop_size // 2)
                parent2_idx = np.random.randint(0, pop_size // 2)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Single point crossover
                crossover_point = np.random.randint(1, len(h) - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for child in [child1, child2]:
                    for i in range(len(child)):
                        if np.random.rand() < mutation_rate:
                            child[i] += np.random.normal(0, mutation_noise)
                    child = np.clip(child, 0, 1)
                    scale = 22.5 / np.sum(child)
                    child = child * scale
                    new_population.append(child)
            else:
                # Just mutate a random individual
                idx = np.random.randint(0, pop_size)
                child = population[idx].copy()
                for i in range(len(child)):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, mutation_noise)
                child = np.clip(child, 0, 1)
                scale = 22.5 / np.sum(child)
                child = child * scale
                new_population.append(child)
        
        # Keep only pop_size individuals
        population = new_population[:pop_size]
        
        # Diversification check
        if no_improvement_count > max_no_improvement:
            # Introduce diversity by randomizing part of the population
            for i in range(int(0.2 * pop_size)):
                idx = np.random.randint(0, pop_size)
                new_h = np.random.rand(len(h))
                new_h = new_h * (22.5 / np.sum(new_h))
                population[idx] = new_h
            mutation_noise *= 0.9
            no_improvement_count = 0
    
    return (best_h.tolist(), best_c5, len(best_h))
