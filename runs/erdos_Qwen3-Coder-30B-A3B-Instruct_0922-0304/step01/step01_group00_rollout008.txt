import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T = 0.05  # Initial temperature
    cooling_rate = 0.9998
    max_iter = 100000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Perturbation probabilities
    prob_flip = 0.3  # Probability of flipping a single point

    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Save current state
        current_h = h.copy()
        current_c5 = evaluate_c5(h, dx)
        
        # Choose perturbation type
        if np.random.random() < prob_flip:
            # Flip a random point (if it's not at boundary)
            idx = np.random.randint(n_points)
            h[idx] = 1 - h[idx]
        else:
            # Add noise to two random points
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            noise = np.random.normal(0, 0.01, 2)
            h[idx1] += noise[0]

            # Correct to maintain sum constraint
            diff = (h[idx1] + h[idx2]) - (current_h[idx1] + current_h[idx2])

            else:

        new_c5 = evaluate_c5(h, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / T):
            # Accept the new solution
            pass
        else:
            # Reject the new solution; revert to previous
            h = current_h
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Adjust temperature
        T *= cooling_rate
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
