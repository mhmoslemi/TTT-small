import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Initialize with sawtooth pattern
        h_init = np.array([0.5 * (1.0 - np.sin(i * np.pi / n_points)) for i in range(n_points)])
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to optimize
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def constraint_bounds(h):
        return np.minimum(h, 1 - h)  # Ensures h[i] ∈ [0,1]
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)},
    ]
    
    # Perform optimization using scipy minimize with multiple starting points
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Try different optimization methods to escape local minima
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    
    for method in methods:
        try:
            if time.time() - start_time >= budget_s * 0.95:
                break
            result = minimize(
                objective,
                current_h,
                method=method,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-7, 'gtol': 1e-7},
                callback=lambda x: None,
            )
            if result.success:
                temp_h = result.x
                temp_h = np.clip(temp_h, 0, 1)
                temp_h = temp_h * target_sum / np.sum(temp_h)
                temp_c5 = evaluate_c5(temp_h, dx)
                if temp_c5 < current_c5:
                    current_h = temp_h
                    current_c5 = temp_c5
        except Exception:
            continue
    
    # Final refinement with additional local search
    if time.time() - start_time < budget_s * 0.9:
        # Add some random perturbations for diversification
        for _ in range(10):
            if time.time() - start_time >= budget_s * 0.95:
                break
            pert_h = current_h + np.random.normal(0, 0.02, len(current_h))
            pert_h = np.clip(pert_h, 0, 1)
            pert_h = pert_h * target_sum / np.sum(pert_h)
            pert_c5 = evaluate_c5(pert_h, dx)
            if pert_c5 < current_c5:
                current_h = pert_h
                current_c5 = pert_c5
    
    # Ensure correctness
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
