import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    n_points = 45
    dx = 2.0 / n_points
    
    population = []
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        population.append(h.copy())
    else:
        # Create a simple initial construction
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))  # Renormalize
        population.append(h.copy())

    # Fill the rest of the population with random constructions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (22.5 / np.sum(h))
        population.append(h.copy())

    best_c5 = float('inf')
    best_h = None

    # Evolution parameters
    generations = 2000
    mutation_rate = 0.05
    mutation_noise = 0.05
    crossover_rate = 0.8
    elitism_rate = 0.1
    diversification_threshold = 50
    no_improvement_count = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = signal.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            fitness.append((c5, h))
            
        # Sort by fitness (lower C5 first)
        fitness.sort(key=lambda x: x[0])
        current_best_c5 = fitness[0][0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = fitness[0][1].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Diversification check
        if no_improvement_count > diversification_threshold:
            # Reset 20% of population
            reset_count = int(0.2 * population_size)
            for _ in range(reset_count):
                idx = np.random.randint(0, population_size)
                h = np.random.rand(n_points)
                h = h * (22.5 / np.sum(h))
                population[idx] = h.copy()
            # Reduce mutation noise
            mutation_noise *= 0.9
            no_improvement_count = 0

        # Selection
        elite_count = int(elitism_rate * population_size)
        elites = [h for _, h in fitness[:elite_count]]
        
        # Create new population
        new_population = elites[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                # Select two parents
                parent1_idx = np.random.randint(0, len(fitness))
                parent2_idx = np.random.randint(0, len(fitness))
                parent1 = fitness[parent1_idx][1].copy()
                parent2 = fitness[parent2_idx][1].copy()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for h in [offspring1, offspring2]:
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            h[i] += np.random.normal(0, mutation_noise)
                    # Ensure constraints
                    h = np.clip(h, 0, 1)
                    h = h * (22.5 / np.sum(h))
                    new_population.append(h.copy())
            else:
                # Mutation only
                parent_idx = np.random.randint(0, len(fitness))
                parent = fitness[parent_idx][1].copy()
                h = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        h[i] += np.random.normal(0, mutation_noise)
                # Ensure constraints
                h = np.clip(h, 0, 1)
                h = h * (22.5 / np.sum(h))
                new_population.append(h.copy())
                
        # Keep population size constant
        population = new_population[:population_size]

    return (best_h.tolist(), best_c5, len(best_h))
