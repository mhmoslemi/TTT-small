import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # L1-ball projection to enforce sum = n_points / 2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    # Segment-wise crossover
    n = len(parent1)
    if n < 2:
        return parent1.copy()
    cut_point = random.randint(1, n - 1)
    child = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
    return child

def mutate(individual, mutation_rate=0.2, noise_scale=0.02):
    child = individual.copy()
    for i in range(len(child)):
        if random.random() < mutation_rate:
            child[i] += np.random.normal(0, noise_scale)
    return child

def genetic_algorithm(seed, budget_s, n_points=70):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Population initialization
    population_size = 50
    population = []
    
    # If we have a parent solution, start with that
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = project_to_feasible(parent, n_points)
        population.append(parent)
    else:
        # Create structured initial solution
        parent = np.ones(n_points)
        parent[n_points//2:] = 0
        parent = project_to_feasible(parent, n_points)
        population.append(parent)
    
    # Fill rest of population with random individuals
    while len(population) < population_size:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual, 2.0 / n_points)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = individual.copy()
        
        # Early termination if good enough
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament selection)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = random.sample(range(population_size), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_index].copy())
            
        # Crossover and Mutation
        new_population = [best_individual.copy()]  # Keep the best
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate=0.2, noise_scale=0.02)
            child = project_to_feasible(child, n_points)
            new_population.append(child)
        
        population = new_population
        
        # Occasionally apply a relaxation step
        if generation % 10 == 0:
            # Try to improve the best individual using gradient descent
            best_ind = best_individual.copy()
            step_size = 0.01
            for _ in range(10):
                c5_val = evaluate_c5(best_ind, 2.0 / n_points)
                
                # Simple gradient approximation
                grad = np.zeros_like(best_ind)
                eps = 1e-4
                for i in range(len(best_ind)):
                    temp_h = best_ind.copy()
                    temp_h[i] += eps
                    temp_h = project_to_feasible(temp_h, n_points)
                    c5_plus = evaluate_c5(temp_h, 2.0 / n_points)
                    grad[i] = (c5_plus - c5_val) / eps
                
                # Update  
                best_ind -= step_size * grad
                best_ind = project_to_feasible(best_ind, n_points)
            
            # Replace worst with improved version
            worst_idx = np.argmax(fitness)
            population[worst_idx] = best_ind
            
    return best_individual, best_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    h, c5_bound = genetic_algorithm(seed, budget_s, n_points)
    return (h.tolist(), c5_bound, n_points)
