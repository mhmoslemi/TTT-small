import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0

    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    max_iter = 10000
    max_evals = 100000
    eval_count = 0
    T = T0
    
    # Simulated Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Create a neighbor solution by perturbing a random subset
        h_new = h.copy()
        
        # Select a random subset of points to perturb (5-10%)
        num_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Perturb each selected point
        for idx in indices:
            # Create small change
            delta = np.random.normal(0, 0.1 * T)
            new_value = h[idx] + delta
            
            # Keep within bounds and adjust other points to maintain integral
            h_new[idx] = np.clip(new_value, 0, 1)
            
        # Adjust the rest of the vector to preserve the sum constraint
        current_sum = np.sum(h_new)
        adjustment = (n_points / 2.0 - current_sum) / (n_points - num_perturb)
        
        # Apply adjustment to non-perturbed points
        non_perturbed = [i for i in range(n_points) if i not in indices]
        for i in non_perturbed:
            if i not in indices:
                h_new[i] = np.clip(h_new[i] + adjustment, 0, 1)
                
        # Ensure sum constraint is maintained exactly
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2.0) / current_sum

        # Evaluate the new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        eval_count += 1
        
        # Accept or reject the new solution
        delta = C5_new - best_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                
        # Reduce temperature
        T *= alpha
        
        if eval_count >= max_evals:
            break
    
    # Final check and return
    return (best_h, best_C5, n_points)
