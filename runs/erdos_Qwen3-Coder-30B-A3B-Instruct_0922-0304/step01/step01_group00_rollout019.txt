import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise Frank-Wolfe like approach
    max_iter = 10000
    line_search_precision = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

        # Compute gradient of overlap with respect to each h[i]
        # Use finite difference to estimate gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        
        # Compute overlap for each position to get gradient
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        # Choose coordinate to update that decreases overlap most
        # (minimize negative gradient)
        i_update = np.argmin(grad)
        
        # Line search for best step in direction of decreasing gradient
        step_size = 0.01
        best_step = 0.0
        best_h_trial = h.copy()
        best_trial_c5 = max_overlap
        
        # Try a few step sizes
        for s in [0.001, 0.005, 0.01, 0.02, 0.05]:
            h_trial = h.copy()
            h_trial[i_update] += s
            h_trial = np.clip(h_trial, 0, 1)
            
            # Ensure the sum constraint
            sum_diff = np.sum(h_trial) - (n_points / 2)
            if abs(sum_diff) > 1e-10:
                h_trial -= sum_diff / n_points
                
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            trial_c5 = evaluate_c5(h_trial, dx)
            if trial_c5 < best_trial_c5:
                best_trial_c5 = trial_c5
                best_h_trial = h_trial.copy()
                best_step = s
                
        # Update h
        h = best_h_trial.copy()
        
    return (best_h.tolist(), best_c5, n_points)
