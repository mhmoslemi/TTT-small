import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Use an adaptive simulated annealing approach
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters for optimization
    max_iter = 10000
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    step_size = 0.05
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Create new candidate by perturbing current solution
        h_new = current_h + np.random.normal(0, step_size, n_points)
        
        # Ensure bounds [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate new candidate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept with probability based on simulated annealing
        if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new
            current_c5 = c5_new
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4)
    assert all(0 <= h <= 1 for h in best_h)
    
    return (best_h, best_c5, n_points)
