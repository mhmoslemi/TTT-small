import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    h_init = project_to_feasible(h_init, n_points)

    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))

    g_best_idx = np.argmin(p_best_fitness)

            swarm[i] = swarm[i] + velocities[i]
