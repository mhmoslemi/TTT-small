import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a simple starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Genetic Algorithm Parameters
    pop_size = 50
    generations = 2000
    elite_size = pop_size // 10
    crossover_rate = 0.8
    mutation_rate = 0.05
    mutation_noise = 0.05
    no_improvement_limit = 50
    no_improvement_count = 0
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h.copy())
        pop_size -= 1
    else:
        # Create a diverse starting population
        for _ in range(pop_size):
            h_temp = np.zeros(n_points)
            # Set ~50% of values to 1, randomly
            indices = np.random.choice(n_points, size=n_points//2, replace=False)
            h_temp[indices] = 1.0
            population.append(h_temp)
    
    # Create additional random individuals to fill population
    for _ in range(pop_size - len(population)):
        h_temp = np.random.rand(n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2 / np.sum(h_temp))
        population.append(h_temp)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for stagnation and diversify
        if no_improvement_count > no_improvement_limit:
            # Introduce diversity
            for i in range(pop_size // 5):
                idx = np.random.randint(0, pop_size)
                population[idx] = np.random.rand(n_points)
                population[idx] = np.clip(population[idx], 0, 1)
                population[idx] = population[idx] * (n_points / 2 / np.sum(population[idx]))
            no_improvement_count = 0
            
        # Selection: keep elite
        selected = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_pop = selected[:]
        
        while len(new_pop) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, mutation_noise)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, mutation_noise)
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2 / np.sum(child1))
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2 / np.sum(child2))
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        population = new_pop[:pop_size]
        # Reduce mutation rate over time
        if gen % 100 == 0 and mutation_noise > 0.001:
            mutation_noise *= 0.9
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
