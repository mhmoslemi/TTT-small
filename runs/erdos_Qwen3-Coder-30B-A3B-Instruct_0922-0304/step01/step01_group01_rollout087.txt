import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

        h = best_h.copy()
        
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient with respect to the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i - max_lag < n_points:
                grad[i] = 1 - 2 * h[i] - 2 * h[i - max_lag]
        
        # Update ADAM moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        
        # Update h
        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        # Update best solution if improved
        c5_new = evaluate_c5(h, 2.0 / n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Update learning rate
        alpha *= alpha_decay
        
        iteration += 1
