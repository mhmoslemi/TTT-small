import numpy as np
import time
from scipy.optimize import differential_evolution
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the objective function
    def objective(h):
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Apply sum constraint by scaling
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return evaluate_c5(h, dx)
    
    # Helper function to evaluate the constraint
    def sum_constraint(h):
        return np.sum(h) - target_sum
    
    # Record best solution
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    best_sum = np.sum(best_h)
    
    # Track evaluation time
    elapsed = time.time() - start_time
    remaining_time = budget_s - elapsed
    
    # Use differential evolution on a subset of the domain if we have time
    if remaining_time > 10:
        # Then we run DE optimization
        try:
            # Use a different optimization approach that allows global search
            # Starting with a random initial population
            options = {
                'maxiter': int(remaining_time * 5),  # adaptive number of iterations
                'popsize': 15,
                'tol': 1e-4,
                'recombination': 0.7,
                'mutation': (0.5, 1.0)
            }
            
            # Run optimization
            result = differential_evolution(
                objective,
                bounds,
                seed=seed,
                maxiter=options['maxiter'],
                popsize=options['popsize'],
                tol=options['tol'],

            if result.success:

                h_opt = np.clip(h_opt, 0, 1)

                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    best_sum = np.sum(h_opt)
            
        except Exception:
            # fallback to starting solution regardless
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Ensure constraints are met
    assert abs(np.sum(final_h) - target_sum) < 1e-3, "Sum constraint violated"
    assert all(0 <= h <= 1 for h in final_h), "bounds constraint violated"
    
    return (final_h, final_c5, n_points)
