import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    best_C5 = float('inf')
    best_h = None
    
    # Construct the optimization problem
    def solve_qcp(h_init):
        h = Variable(n_points)
        C = Variable()
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2,
            C >= 0
        ]
        
        # Overlap constraints
        # For all valid shifts
        max_k = n_points - 1
        for k in range(-max_k, max_k + 1):
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in range(n_points - k)
                expr = sum(h[i] * (1 - h[i + k]) for i in range(n_points - k))
            else:
                # h[i+k] * (1 - h[i]) for i in range(-k, n_points)
                expr = sum(h[i + k] * (1 - h[i]) for i in range(-k, n_points))
            constraints.append(expr <= C * dx)
        
        # Formulate and solve
        objective = Minimize(C)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-6)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                C_val = C.value
                h_vals = h.value
                return C_val, h_vals
        except:
            pass
        return None, None
    
    # Initial solution
    C5_val, h_vals = solve_qcp(initial_h)
    if C5_val is not None and C5_val < best_C5:
        best_C5 = C5_val
        best_h = h_vals
    
    # Local refinement with gradient-based method using the best result so far
    if best_h is not None:
        def compute_C5(h):
            h = np.array(h)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        def objective(h):
            return compute_C5(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        initial_h_refine = best_h.copy()
        initial_h_refine = initial_h_refine * (n_points / 2) / np.sum(initial_h_refine)
        
        res = minimize(objective, initial_h_refine, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-6})
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_C5 = compute_C5(final_h)
            
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = final_h

    # Return best solution found
    if best_h is None:
        # Fallback to initial construction
        best_h = initial_h.copy()
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        # Recompute C5 properly to get correct value
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_C5 = np.max(corr) * dx

    return (best_h.tolist(), best_C5, n_points)
