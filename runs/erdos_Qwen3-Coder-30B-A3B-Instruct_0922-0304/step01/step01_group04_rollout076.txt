import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Objective function with constraints
    def objective(h):
        # Clip h to [0,1] and rescale to meet sum constraint
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        return evaluate_c5(h_scaled, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create a wrapper that penalizes violations of the sum constraint
    def penalized_objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        c5_val = evaluate_c5(h_scaled, dx)
        # Add penalty if sum is far from target
        sum_penalty = 1000 * abs(np.sum(h_clipped) - target_sum)
        return c5_val + sum_penalty
    
    # Time tracking
    remaining_time = budget_s - (time.time() - start_time)
    
    # Use differential evolution with limited iterations to avoid timeout
    maxiter = min(100, int(remaining_time / 2))
    
    try:
        result = differential_evolution(
            penalized_objective,
            bounds,
            maxiter=maxiter,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False
        )
        
        if result.success:
            best_h = result.x
        else:
            # Fallback to initial if optimization fails
            best_h = h_init.copy()
    except Exception:
        # Fallback to initial construction
        best_h = h_init.copy()

    # Ensure constraints are satisfied
    try:
        best_h = np.clip(best_h, 0, 1)
        final_sum = np.sum(best_h)
        if abs(final_sum - target_sum) > 1e-6:
            best_h = best_h * target_sum / final_sum
        final_c5 = evaluate_c5(best_h, dx)
    except:
        # If there's an error, return initial
        best_h = h_init.copy()
        final_c5 = evaluate_c5(best_h, dx)
    
    # Make sure it's feasible
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-6:
        best_h = best_h * target_sum / final_sum
    
    return (best_h, final_c5, n_points)
