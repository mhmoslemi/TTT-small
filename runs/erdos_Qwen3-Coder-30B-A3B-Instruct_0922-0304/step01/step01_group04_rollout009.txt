import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Repair function to maintain constraints
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Objective function for DE
    def de_objective(h):
        h = repair_h(h)
        return evaluate_c5(h, dx)
    
    # Objective function for local optimization
    def local_objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for local optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum

    # DE optimization
    bounds = [(0, 1) for _ in range(n_points)]
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    best_h = repair_h(de_result.x)
    best_c5 = de_result.fun
    
    # Local optimization refinement with L-BFGS-B
    try:
        result = minimize(
            local_objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        
        if result.success:
            refined_h = repair_h(result.x)
            refined_c5 = local_objective(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass

    # Final validation
    best_h = repair_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (best_h, final_c5, n_points)
