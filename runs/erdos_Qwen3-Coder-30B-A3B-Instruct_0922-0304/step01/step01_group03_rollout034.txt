import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable starting point
        n_points = 45
        h = np.zeros(n_points)
        # Set first ~23 elements to 1 to get sum ~22.5
        h[:23] = 1.0
        # Renormalize
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # GA parameters
    population_size = 50
    generations = 2000
    elite_size = population_size // 10
    mutation_rate = 0.05
    mutation_noise = 0.05
    diversification_threshold = 50
    no_improvement_count = 0
    current_gen = 0
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        # Create random individuals
        new_h = np.random.rand(len(h))
        # Set some values to 1 to potentially get a near-optimal sum
        num_ones = np.random.randint(10, 25)
        new_h[:num_ones] = 1.0
        new_h = np.clip(new_h, 0, 1)
        # Renormalize
        new_h = new_h * (22.5 / np.sum(new_h))
        population.append(new_h)
    
    # Main GA loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for individual in population:
            c5_val = evaluate_c5(individual, dx)
            fitness_scores.append((c5_val, individual))
        
        # Sort by fitness
        fitness_scores.sort(key=lambda x: x[0])
        
        # Update best solution
        current_best_c5 = fitness_scores[0][0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = fitness_scores[0][1].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for diversification
        if no_improvement_count >= diversification_threshold:
            # Reset some individuals
            for i in range(int(0.2 * population_size)):
                idx = np.random.randint(len(population))
                new_h = np.random.rand(len(h))
                num_ones = np.random.randint(10, 25)
                new_h[:num_ones] = 1.0
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (22.5 / np.sum(new_h))
                population[idx] = new_h
            no_improvement_count = 0
            # Decrease mutation noise
            mutation_noise = max(0.005, mutation_noise * 0.9)
        
        # Selection (tournament selection)
        selected = []
        for _ in range(population_size):
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(len(fitness_scores), tournament_size)
            tournament_scores = [fitness_scores[i][0] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_scores)]
            selected.append(fitness_scores[winner_idx][1])
        
        # Elitism
        elites = [ind for _, ind in fitness_scores[:elite_size]]
        
        # Create offspring through crossover and mutation
        offspring = []
        
        # Crossover: single point
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Single-point crossover
            point = np.random.randint(1, len(h) - 1)
            child1 = np.concatenate([parent1[:point], parent2[point:]])
            child2 = np.concatenate([parent2[:point], parent1[point:]])
            
            # Mutation
            for child in [child1, child2]:
                # Randomly flip positions
                indices = np.random.choice(len(h), size=int(mutation_rate * len(h)), replace=False)
                for idx in indices:
                    child[idx] += np.random.normal(0, mutation_noise)
                child = np.clip(child, 0, 1)
                # Renormalize
                child = child * (22.5 / np.sum(child))
                offspring.append(child)
        
        # Replace population
        population = elites + offspring[:population_size - elite_size]
        
        current_gen += 1
    
    # Final evaluation
    final_best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
