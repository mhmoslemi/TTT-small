import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a starting vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2.0) / np.sum(h)  # Renormalize to sum to n_points/2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    iteration = 0
    max_iter = 5000
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 0.1:
        iteration += 1
        
        # --- Convex Optimization Approach ---
        # Create variables
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2.0,
            C >= 0
        ]
        
        # Add overlap constraints
        # For each lag k (0 to 2*n_points-2)
        # We only consider a subset of lags that are likely to produce high overlap
        # We'll use k = 0 to 2*n_points-2 but apply a reasonable subset for efficiency  
        lag_subset = list(range(0, min(2*n_points, 50)))
        for k in lag_subset:
            # Compute the overlap for this lag
            # overlap = sum h[i] * (1 - h[(i+k) % n_points]) * dx
            if k < n_points:
                # For k in [0, n_points), we do standard correlation
                corr = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:
                # For k >= n_points, we have wraparound
                corr = cp.sum(cp.multiply(h_vars[n_points-k:], 1 - h_vars[:k-n_points]))
            
            constraints.append(C >= corr * dx)
        
        # Objective
        objective = cp.Minimize(C)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve the problem
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = h_vars.value
                c5_opt = C.value
                
                # Ensure h_opt meets constraints
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
                
                # Evaluate the result
                new_c5 = evaluate_c5(h_opt, dx)
                
                if new_c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = new_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                
        except:
            no_improvement_count += 1
            
        # Periodic perturbation
        if iteration % 100 == 0:
            perturb_idx = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.02, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
        
        # Random restarts
        if no_improvement_count > max_no_improvement:
            h = np.zeros(n_points)
            h[:23] = 1.0
            np.random.shuffle(h)
            h = h * (n_points / 2.0) / np.sum(h)
            no_improvement_count = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
