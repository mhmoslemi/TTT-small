import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    T = 1e-3  # Initial temperature
    T_min = 1e-8
    cooling_rate = 0.995
    
    # Precompute all lags
    lags = np.arange(-len(h)+1, len(h))
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps 
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Apply gradient and noise
        h_new = h - alpha * grad
        noise = np.random.normal(0, T, len(h))
        h_new = h_new + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (22.5 / np.sum(h_new))
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update based on Metropolis criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
            
        # Adaptively decrease temperature
        if iter_since_improvement > max_no_improvement // 2:
            T = max(T_min, T * cooling_rate)
            iter_since_improvement = 0
            if alpha > alpha_min:
                alpha *= gamma
                
        # Periodic binary reset to escape local minima
        if iteration % 5000 == 0 and iteration > 0:
            h = np.zeros(len(h))
            # Create a new binary pattern
            num_ones = 22
            indices = np.random.choice(len(h), num_ones, replace=False)
            h[indices] = 1.0
            # Renormalize 
            h = h * (22.5 / np.sum(h))

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
