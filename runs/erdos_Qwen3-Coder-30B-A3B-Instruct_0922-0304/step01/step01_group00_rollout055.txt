import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    temperature = 0.05
    cooling_rate = 0.9998
    max_iter = 100000
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Iteration counter
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Generate a perturbation
        if np.random.random() < 0.7:
            # Binary swap: flip two random positions to 0 and 1
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            h_new[i] = 1 - h_new[i]
            h_new[j] = 1 - h_new[j]
        else:
            # Gaussian noise perturbation
            h_new = h.copy()
            indices = np.random.choice(n_points, 2, replace=False)
            # Add perturbation and ensure sum constraint
            perturbation = np.random.normal(0, 0.01, 2)
            h_new[indices[0]] += perturbation[0]
            h_new[indices[1]] += perturbation[1]
            
            # Adjust to maintain sum constraint
            total_diff = np.sum(h_new) - (n_points / 2)
            if abs(total_diff) > 1e-10:
                # Bias adjustment to maintain sum constraint
                h_new[0] -= total_diff / 2.0
                h_new[1] -= total_diff / 2.0
            
        # Clip values to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to meet constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject the new solution
        if new_c5 < best_c5 or np.random.random() < np.exp((best_c5 - new_c5) / temperature):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature
        temperature *= cooling_rate
        
        iteration += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
