import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iter = 10000
    exploration_weight = 1.0
    max_children_per_node = 10
    reset_frequency = 1000
    
    # MCTS Tree Node class
    class Node:
        def __init__(self, h_config, parent=None, action=None):
            self.h = h_config.copy()
            self.parent = parent
            self.action = action
            self.children = []
            self.visits = 0
            self.total_c5 = float('inf')
            self.best_c5 = float('inf')
            
        def ucb_score(self):
            if self.visits == 0:
                return float('inf')
            exploit = self.total_c5 / self.visits
            explore = exploration_weight * np.sqrt(np.log(self.parent.visits) / self.visits)
            return exploit - explore
            
        def is_fully_expanded(self):
            return len(self.children) >= max_children_per_node
            
        def is_terminal(self):
            return self.visits > 100  # Simple stop condition
            
    # MCTS implementation
    root = Node(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection: traverse from root using UCB
        current = root
        while not current.is_terminal() and current.is_fully_expanded():
            best_child = max(current.children, key=lambda x: x.ucb_score())
            current = best_child
            
        # Expansion: create children if not terminal
        if not current.is_terminal() and not current.is_fully_expanded():
            # Generate children
            children_h = []
            for _ in range(max_children_per_node):
                h_copy = current.h.copy()
                # Perturb about 5% of elements
                num_mutate = max(1, int(0.05 * len(h_copy)))
                indices = np.random.choice(len(h_copy), size=num_mutate, replace=False)
                for i in indices:
                    h_copy[i] += np.random.normal(0, 0.1)
                    h_copy[i] = np.clip(h_copy[i], 0, 1)
                
                # Renormalize
                scale_factor = 22.5 / np.sum(h_copy)
                h_copy = h_copy * scale_factor
                h_copy = np.clip(h_copy, 0, 1)
                children_h.append(h_copy)
                
            # Create nodes for children
            for i, h_new in enumerate(children_h):
                child_node = Node(h_new, parent=current, action=f"mutation_{i}")
                current.children.append(child_node)
            
            # Pick a child at random to simulate
            child_node = random.choice(current.children)
            current = child_node
        else:
            # If expanded or terminal, simulate from this node
            pass
            
        # Simulation: compute C5 for this node
        c5_value = evaluate_c5(current.h, dx)
        current.total_c5 = min(current.total_c5, c5_value)
        current.visits += 1
        current.best_c5 = min(current.best_c5, c5_value)
        
        # Backpropagation: update all ancestors
        node = current
        while node.parent is not None:
            node = node.parent
            node.total_c5 = min(node.total_c5, c5_value)
            node.visits += 1
            node.best_c5 = min(node.best_c5, c5_value)
            
        # Periodic reset of some nodes
        if iteration % reset_frequency == 0 and iteration > 0:
            # Reset some nodes to random configurations
            def reset_random_nodes(node, prob=0.1):
                if np.random.random() < prob:
                    # Generate new random node with similar properties
                    new_h = np.random.rand(len(node.h))
                    scale_factor = 22.5 / np.sum(new_h)
                    new_h = new_h * scale_factor
                    new_h = np.clip(new_h, 0, 1)
                    node.h = new_h
                    node.total_c5 = float('inf')
                    node.best_c5 = float('inf')
                    node.visits = 0
                    node.children = []
                for child in node.children:
                    reset_random_nodes(child, prob=prob/2)
            
            reset_random_nodes(root)
            
        # Update best solution found so far
        if current.best_c5 < best_c5:
            best_c5 = current.best_c5
            best_h = current.h.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
