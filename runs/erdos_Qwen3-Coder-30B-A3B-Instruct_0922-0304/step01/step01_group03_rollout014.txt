import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead simplex parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturbation_frequency = 1000
    max_no_improvement = 1000
    no_improvement_count = 0
    
    # Create initial simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        perturbed = h + np.random.normal(0, simplex_size, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        scale = 22.5 / np.sum(perturbed)
        perturbed = perturbed * scale
        simplex.append(perturbed.copy())
    
    def evaluate_simplex(simplex_points):
        """Evaluate C5 for all points in simplex."""
        evaluations = []
        for point in simplex_points:
            c5 = evaluate_c5(point, dx)
            evaluations.append(c5)
        return np.array(evaluations)
    
    def project_to_feasible(h_vec):
        """Project h to feasible region: 0 <= h <= 1 and sum(h) = n_points/2."""
        h_vec = np.clip(h_vec, 0, 1)
        scale = (n_points / 2) / np.sum(h_vec)
        h_vec = h_vec * scale
        return h_vec
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Evaluate simplex
        values = evaluate_simplex(simplex)
        indices = np.argsort(values)
        simplex = [simplex[i] for i in indices]
        values = values[indices]
        
        # Update best if needed
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Compute centroid (excluding the worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflect
        worst = simplex[-1]
        reflected = centroid + reflection_factor * (centroid - worst)
        reflected = project_to_feasible(reflected)
        reflected_c5 = evaluate_c5(reflected, dx)
        
        if values[0] <= reflected_c5 < values[-2]:
            # Accept reflected point
            simplex[-1] = reflected
        elif reflected_c5 < values[0]:
            # Expand
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = project_to_feasible(expanded)
            expanded_c5 = evaluate_c5(expanded, dx)
            if expanded_c5 < reflected_c5:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contract
            if reflected_c5 < values[-1]:
                # Outside contraction
                contracted = centroid + contraction_factor * (reflected - centroid)
            else:
                # Inside contraction
                contracted = centroid + contraction_factor * (worst - centroid)
            contracted = project_to_feasible(contracted)
            contracted_c5 = evaluate_c5(contracted, dx)
            if contracted_c5 < values[-1]:
                simplex[-1] = contracted
            else:
                # Shrink simplex
                best = simplex[0]
                for i in range(1, len(simplex)):
                    simplex[i] = best + 0.5 * (simplex[i] - best)
        
        # Periodic perturbations
        if iteration % perturbation_frequency == 0:
            perturb_idx = np.random.choice(len(h), size=int(0.05 * len(h)), replace=False)
            best_h_pert = best_h.copy()
            best_h_pert[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            best_h_pert = project_to_feasible(best_h_pert)
            # Replace worst in simplex with perturbed best
            simplex[-1] = best_h_pert

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
