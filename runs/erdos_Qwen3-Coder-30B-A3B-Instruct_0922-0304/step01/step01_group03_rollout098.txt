import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():

    else:

        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_frequency = 1000
    
    # MCTS node class
    class MCTSNode:
        def __init__(self, h_config, parent=None, action=None):
            self.h = h_config.copy()
            self.parent = parent
            self.action = action
            self.children = {}
            self.visits = 0
            self.total_c5 = float('inf')
            self.best_c5 = float('inf')
            self.cached_c5 = None
            
        def calculate_ucb(self, parent_visits):
            if self.visits == 0:
                return float('inf')
            return self.best_c5 + exploration_weight * np.sqrt(2 * np.log(parent_visits) / self.visits)
        
        def evaluate_c5_cached(self):
            if self.cached_c5 is None:
                self.cached_c5 = evaluate_c5(self.h, dx)
            return self.cached_c5
            
        def expand(self):
            # Generate 5 mutations of the current state
            mutations = 5
            children = []
            for _ in range(mutations):
                child_h = self.h.copy()
                # Select random indices to mutate
                n_mutation = int(len(child_h) * mutation_rate)
                indices = np.random.choice(len(child_h), size=n_mutation, replace=False)
                # Add noise
                noise = np.random.normal(0, 0.1, n_mutation)
                child_h[indices] += noise
                # Clip to bounds
                child_h = np.clip(child_h, 0, 1)
                # Renormalize
                child_h = child_h * (22.5 / np.sum(child_h))
                # Create node
                new_node = MCTSNode(child_h, self, indices)
                children.append(new_node)
            self.children = {tuple(child.action): child for child in children}
            return children
    
    # Create initial root node
    root = MCTSNode(h)
    root.evaluate_c5_cached()
    
    # Main MCTS loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection
        current = root
        while current.children:
            best_child = max(current.children.values(), key=lambda node: node.calculate_ucb(current.visits))
            current = best_child
            
        # Expansion
        if current.visits == 0:
            # First visit, evaluate
            c5 = current.evaluate_c5_cached()
            current.best_c5 = min(current.best_c5, c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.h.copy()
            current.visits += 1
        else:
            # Expand
            candidates = current.expand()
            # Evaluate all children
            for child in candidates:
                c5 = child.evaluate_c5_cached()
                child.best_c5 = c5
                child.total_c5 = c5
                child.visits = 1
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = child.h.copy()
            # Backpropagate
            current.visits += 1
            
        # Reset some nodes periodically
        if iteration % reset_frequency == 0 and iteration > 0:

            pass
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
