    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    max_no_improvement = 1000
    iter_since_improvement = 0
    
    # Create initial simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        perturb = np.random.normal(0, simplex_size, n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_perturb)
        h_perturb = h_perturb * scale
        simplex.append(h_perturb)
    
    def evaluate_simplex(simplex):
        values = []
        for h_point in simplex:
            c5 = evaluate_c5(h_point, dx)
            values.append(c5)
        return np.array(values)
    
    def normalize_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_vec)
        return h_vec * scale

            break

        values = evaluate_simplex(simplex)

            iter_since_improvement = 0

            simplex[-1] = reflected

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
