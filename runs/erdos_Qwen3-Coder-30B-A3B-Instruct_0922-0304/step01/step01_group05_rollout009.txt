import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Gradient-based optimization with improved initialization
    def objective(h):
        return compute_C5(h)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Constraints for optimization
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform optimization with multiple restarts
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Try multiple random restarts to escape local minima
    for _ in range(5):
        # Add some noise to initial guess
        h_random = h_init + np.random.normal(0, 0.01, n_points)
        h_random = np.clip(h_random, 0, 1)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        
        res = minimize(objective, h_random, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 500, 'ftol': 1e-6})
        
        if res.success:
            h_candidate = res.x
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            C5_candidate = compute_C5(h_candidate)
            if C5_candidate < best_C5:
                best_h = h_candidate
                best_C5 = C5_candidate
    
    # Final polishing using different optimization method
    res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 1000, 'ftol': 1e-8})
    
    if res.success:
        best_h = res.x
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_C5 = compute_C5(best_h)
    
    # Ensure final solution is normalized and within bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
