import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

    swarm = []

        p_best_fitness.append(evaluate_c5(h_init, 2.0 / n_points))

    # Initialize remaining particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)

            velocities[i] = np.clip(velocities[i], velocity_clamp[0], velocity_clamp[1])
