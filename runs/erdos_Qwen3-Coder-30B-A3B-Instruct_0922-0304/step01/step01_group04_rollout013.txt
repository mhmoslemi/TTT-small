import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Set up convex optimization using CVXPY
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]
    
    # For each possible shift k, add the constraint corresponding to the correlation
    # This creates many constraints, but CVXPY handles them efficiently in the convex setting
    for k in range(-n_points + 1, n_points):
        # Get indices where we compute correlation
        left_indices = np.arange(max(0, -k), min(n_points, n_points - k))
        right_indices = np.arange(max(0, k), min(n_points, n_points + k))
        
        if len(left_indices) > 0 and len(right_indices) > 0:
            # Compute the sum over matching indices: sum(h[i] * (1 - h[i + k]))
            # which is sum(h[i] - h[i] * h[i + k]) = sum(h[i]) - sum(h[i] * h[i + k])
            # For consistency, compute:
            #   sum_{i: i+k >= 0 and i+k < n} h[i] * (1 - h[i+k]) 
            # This is equivalent to 
            #   sum(h[i]) - sum(h[i]*h[i+k]) for valid indices where i+k within domain.
            
            # For each k, compute correlation term using cvxpy vector operations
            if k >= 0:
                # h[i] * (1 - h[i + k]) for i in [0, n_points-k-1]
                # sum(h[i]) - sum(h[i]*h[i+k]) over i
                # i=0 to n_points-k-1
                # Constraint: sum(h_i * (1-h_{i+k})) <= t
                h_left = h[:n_points-k]
                h_right = h[k:]
                constraints.append(cp.sum(cp.multiply(h_left, 1 - h_right)) <= t)
            else:
                # h[i] * (1 - h[i+k]) for i in [n_points+k, n_points-1] -> index goes from 0 to n_points+k-1
                # here the valid i is 0 to n_points+k-1, i+k is k to n_points-1
                # so i should run from -k to n_points-1 and we end in i+k <= n_points-1 => i <= n_points-1-k
                # and i >= 0
                # i in [0, n_points+k-1]
                # but wrap that correctly
                h_left = h[-k:]
                h_right = h[:-k]
                constraints.append(cp.sum(cp.multiply(h_left, 1 - h_right)) <= t)
    
    # Objective: minimize t
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        # Solve the convex optimization problem using MOSEK solver (fast and robust)
        problem.solve(solver=cp.MOSEK, verbose=False, feastol=1e-8, relgap=1e-8)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = t.value
        else:
            # Fallback to initial solution
            h_opt = h_init.copy()
            c5_bound = evaluate_c5(h_opt, dx)
    except:
        # If MOSEK fails or is not available, fallback to the previous approach
        h_opt = h_init.copy()
        c5_bound = evaluate_c5(h_opt, dx)
        
    # Final validation and cleanup
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * target_sum / np.sum(h_opt)
    final_c5 = evaluate_c5(h_opt, dx)
    
    # Ensure it is within tolerances
    assert np.isclose(np.sum(h_opt), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_opt), "Bounds constraint violated"
    
    return (h_opt, final_c5, n_points)
