import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        # Fallback to a simple construction
        n_points = 42
        initial_h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Objective function
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Custom optimization approach
    best_C5 = float('inf')
    best_h = initial_h.copy()
    
    # Multistart with different initializations
    for i in range(10):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Random perturbation of initial solution
        if i == 0:
            h_current = initial_h.copy()
        else:
            h_current = initial_h.copy() + np.random.normal(0, 0.05, n_points)
            h_current = np.clip(h_current, 0, 1)
            
        # Normalize
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Optimization with different methods
        def objective(h):
            return compute_C5(h)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try different optimization approaches
        methods = ['SLSQP', 'TNC', 'L-BFGS-B']
        
        for method in methods:
            if time.time() - start_time > budget_s - 10:
                break
                
            try:
                res = minimize(objective, h_current, method=method, 
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 500, 'ftol': 1e-6})
                if res.success:
                    h_opt = res.x
                    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                    c5_val = compute_C5(h_opt)
                    if c5_val < best_C5:
                        best_C5 = c5_val
                        best_h = h_opt.copy()
            except:
                continue
                
        # Local refinement with trust-constr
        try:
            res = minimize(objective, h_current, method='trust-constr', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-7})
            if res.success:
                h_opt = res.x
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_val = compute_C5(h_opt)
                if c5_val < best_C5:
                    best_C5 = c5_val
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final refinement if time allows
    if time.time() - start_time < budget_s - 5:
        h_current = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = compute_C5(h_final)
            if c5_final < best_C5:
                best_C5 = c5_final
                best_h = h_final.copy()
    
    return (best_h.tolist(), best_C5, n_points)
