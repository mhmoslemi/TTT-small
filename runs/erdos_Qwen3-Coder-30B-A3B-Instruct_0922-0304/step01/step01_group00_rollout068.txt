import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    line_search_precision = 1e-4
    
    # Initialize h 
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Binary approach: first 35 values 1, rest 0
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add small noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient of overlap at k_max
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += 1 - h[i + k_max]
            if i >= k_max:
                gradient[i] -= h[i - k_max]
                
        # Select coordinate to update (minimize gradient)
        i_update = np.argmin(gradient)
        
        # Line search for optimal step size
        step_size = 1.0
        h_orig = h[i_update]
        h_new = h_orig
        min_overlap = max_overlap
        best_step = 0.0
        
        # Simple backtracking line search
        for _ in range(100):
            h[i_update] = h_orig + step_size * (1 if i_update < n_points // 2 else -1)
            h[i_update] = np.clip(h[i_update], 0, 1)
            
            # Adjust corresponding term in overlap if exists
            if i_update + k_max < n_points:
                h[i_update + k_max] = np.clip(h[i_update + k_max], 0, 1)
            
            # Rebalance to maintain constraint sum = n_points/2
            current_sum = np.sum(h)
            diff = (n_points / 2) - current_sum
            if abs(diff) > 1e-6:
                if np.sum(h) < n_points / 2:
                    # Need to increase values
                    if i_update + k_max < n_points:
                        h[i_update + k_max] = np.clip(h[i_update + k_max] + diff, 0, 1)
                else:
                    # Need to decrease values
                    if i_update + k_max < n_points:
                        h[i_update + k_max] = np.clip(h[i_update + k_max] - diff, 0, 1)
                # Ensure all values remain in [0,1]
                h = np.clip(h, 0, 1)
                # Normalize to get correct sum
                h = h * (n_points / 2) / np.sum(h)
            
            # Check overlap
            corr_new = np.correlate(h, 1 - h, mode="full")
            overlap_new = np.max(corr_new) * dx
            if overlap_new < min_overlap:
                min_overlap = overlap_new
                best_step = step_size
            
            step_size *= 0.5
            if step_size < line_search_precision:
                break
                
        # Apply best step
        h[i_update] = h_orig + best_step * (1 if i_update < n_points // 2 else -1)
        h[i_update] = np.clip(h[i_update], 0, 1)
        
        # Adjust k_max corresponding term
        if i_update + k_max < n_points:
            h[i_update + k_max] = np.clip(h[i_update + k_max], 0, 1)
        
        # Re-normalize to maintain constraint
        h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
