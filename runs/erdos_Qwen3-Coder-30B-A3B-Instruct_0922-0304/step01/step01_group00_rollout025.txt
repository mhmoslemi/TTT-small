import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute overlap integral C5"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Helper to project and normalize h
    def project_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        if c5_val <= target_bound:
            break
            
        # Convex relaxation approach
        try:
            # Create variable for new h
            h_new = Variable(n_points)
            
            # Optimization constraint: max overlap is bounded
            # For simplicity, we minimize the peak overlap directly
            objective = cp.Minimize(cp_sum(h_new))  # Minimize sum to encourage sparsity
            
            # Constraint: sum must equal n_points / 2
            constraints = [cp_sum(h_new) == n_points / 2]
            
            # Bounds on h
            for i in range(n_points):
                constraints.append(h_new[i] >= 0)
                constraints.append(h_new[i] <= 1)
            
            # Form and solve problem
            prob = Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new_sol = np.array([float(h_new[i].value) for i in range(n_points)])
                h = project_and_normalize(h_new_sol)
                
        except:
            # Fallback: take a gradient step
            pass
        
        # Gradient-based refinement for k_max shift
        h_temp = h.copy()
        grad_step = 0.01
        grad_iter = 100
        
        for _ in range(grad_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            c5_val_new = np.max(corr) * dx
            k_max_new = np.argmax(corr)
            
            # Compute gradient approximation using finite differences
            eps = 1e-4
            grad = np.zeros_like(h_temp)
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_val_new) / eps
            
            # Gradient step
            h_temp -= grad_step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Return result
    return (best_h.tolist(), best_c5, n_points)
