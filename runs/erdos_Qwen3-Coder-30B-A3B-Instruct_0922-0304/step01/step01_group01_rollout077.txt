import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    def evaluate_population(pop):
        fitness = []

            fitness.append(fit)

    def tournament_select(pop, fitness, k=tournament_size):
        indices = np.random.choice(len(pop), k)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        return pop[best_idx]
    
    # Crossover
    def crossover(parent1, parent2):
        # Simple arithmetic crossover
        alpha = np.random.random()
        child = alpha * parent1 + (1 - alpha) * parent2
        return child
    
    # Mutation
    def mutate(h):
        for i in range(len(h)):

                h[i] += np.random.normal(0, 0.05)

    start_time = time.time()

        fitness = evaluate_population(population)

        current_best_idx = np.argmin(fitness)

        new_population = [best_individual]
        while len(new_population) < pop_size:
            parent1 = tournament_select(population, fitness)
