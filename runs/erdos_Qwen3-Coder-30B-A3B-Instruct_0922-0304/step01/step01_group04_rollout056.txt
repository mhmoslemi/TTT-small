import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to repair h to be in [0,1] and sum to target_sum
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Objective function for optimization
    def objective(h):
        h = np.abs(h)
        h = repair_h(h)
        return evaluate_c5(h, dx)
    
    # Bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None,
    )
    
    current_h = repair_h(de_result.x)
    current_c5 = objective(current_h)
    
    # Local refinement with L-BFGS-B
    # Prepare constraints
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Optimize with L-BFGS-B
    try:
        result = minimize(
            objective,
            current_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
        )
        
        if result.success:
            current_h = result.x
            current_h = repair_h(current_h)
            current_c5 = objective(current_h)
    except Exception:
        pass
    
    # Final validation
    current_h = repair_h(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Ensure correctness
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    # Ensure we don't run out of time - return at least a valid solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
