import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Set up convex optimization problem
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints for each shift k
    # For k = -n_points+1 to n_points-1
    for k in range(-n_points+1, n_points):
        # This is a double sum over i,j such that j-i = k
        # which is equivalent to sum_{i} h[i] * (1 - h[i+k]) 
        # i runs from max(0, -k) to min(n_points-1, n_points-1-k)
        # This is equivalent to computing correlation for shift k
        
        # Will form the sum over valid i, j pairs where j-i = k
        corr_expr = 0
        for i in range(max(0, -k), min(n_points, n_points-k)):
            j = i + k
            corr_expr += h[i] * (1 - h[j])
        
        if k >= 0:
            # When k >= 0, j = i+k, and i ranges from 0 to n_points-1-k
            # How many pairs?
            # i from 0 to n_points-1-k, so that j = i+k is from k to n_points-1
            # The total is (n_points-k) terms
            corr_expr = cp.sum(cp.multiply(h[:n_points-k], 1-h[k:]))
        else:
            # When k < 0, j = i+k, so i = j-k, so we need j from 0 to n_points-1+k
            # This is the same as summing over i from -k to n_points-1
            corr_expr = cp.sum(cp.multiply(h[-k:], 1-h[:n_points+k]))
            
        constraints.append(corr_expr <= t)
        
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use MOSEK solver if available, otherwise try CVXOPT
    solver = None
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-8, reltol=1e-8)
        solver = 'MOSEK'
    except:
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-8, reltol=1e-8)
            solver = 'CVXOPT'
        except Exception as e:
            # Fallback to initial solution
            return (h_init, np.max(np.correlate(h_init, 1-h_init, mode="full")) * dx, n_points)
    
    if prob.status != cp.OPTIMAL:
        # Fallback to initial solution
        return (h_init, np.max(np.correlate(h_init, 1-h_init, mode="full")) * dx, n_points)
        
    # Extract solution and post-process
    h_sol = h.value
    if h_sol is None:
        h_sol = h_init
        
    # Ensure h is in [0,1] and sum equals target_sum
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * target_sum / np.sum(h_sol)
    
    # Final C5 computation
    c5_value = np.max(np.correlate(h_sol, 1-h_sol, mode="full")) * dx
    
    # Return result
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (h_sol, c5_value, n_points)
