import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Ensure we have enough time for convex optimization
    if time.time() - start_time > budget_s - 30:
        return (h_init, evaluate_c5(h_init, dx), n_points)
        
    try:
        # Define CVXPY variables
        h = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
            t >= 0
        ]
        
        # Add the overlap constraints
        # For each shift k, we have: sum over i of h[i] * (1 - h[i+k]) <= t
        # This is equivalent to: sum of h[i] - h[i] * h[i+k] <= t
        
        # Compute valid indices for all possible shifts
        for k in range(-n_points + 1, n_points):  # range of valid shifts
            # Create the sum expression for shift k
            expr = 0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    expr += h[i] * (1 - h[j])
            
            constraints.append(expr <= t)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with MOSEK for higher precision
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-8, reltol=1e-8)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            current_c5 = t_opt
            current_h = h_opt
        else:
            # If convex solve fails, fallback to initial point
            current_h = h_init.copy()
            current_c5 = evaluate_c5(current_h, dx)
            
    except Exception as e:
        # Fallback solution
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)
    
    # Ensure the solution is within bounds
    current_h = np.clip(current_h, 0, 1)
    
    # Ensure the sum is exactly correct
    current_h = current_h * target_sum / np.sum(current_h)
    
    # Final evaluation
    current_c5 = evaluate_c5(current_h, dx)
    
    # Validate constraints
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
