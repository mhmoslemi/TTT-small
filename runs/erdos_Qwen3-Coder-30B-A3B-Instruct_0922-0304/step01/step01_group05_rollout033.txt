        corr = np.correlate(h, 1 - h, mode="full")

        return np.max(corr) * dx

        # Within a single constraint row, we evaluate for valid combinations

        num_constraints = 2 * n_points - 1

        for k in range(-n_points + 1, n_points):

            if k >= 0:
                i_start, i_end = 0, n_points - k
            else:
                i_start, i_end = -k, n_points

            if i_start < i_end:
                # Build constraint matrix for this overlap
                # We want: sum(h[i] * (1 - h[i+k])) <= C * dx
                A_row = np.zeros(n_points)
                for i in range(i_start, i_end):
                    a_val = 1 - h[i + k] if i + k < n_points else 0
                    A_row[i] += a_val
                A[num_constraints // 2 + k, :] = A_row
                # For rhs, we'll put a placeholder, but solve via optimization
        
        return A

    best_h = h_init.copy()
