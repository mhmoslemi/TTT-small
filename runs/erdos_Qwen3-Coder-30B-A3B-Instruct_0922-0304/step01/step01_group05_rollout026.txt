import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Best solution tracking
    best_C5 = float('inf')
    best_h = None
    
    # Use CVXPY to solve QCP
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Compute all overlap constraints
    max_lag = n_points - 1
    for k in range(-max_lag, max_lag + 1):
        if k >= 0:
            # For k >= 0, h[i] * (1 - h[i + k]) where valid
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i + k])
            constraints.append(expr <= C * dx)
        else:
            # For k < 0, h[i + k] * (1 - h[i]) where valid
            expr = 0
            for i in range(-k, n_points):
                expr += h[i + k] * (1 - h[i])
            constraints.append(expr <= C * dx)
    
    # Objective
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    # Solve with SCS
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=10000, eps=1e-5)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            C5_bound = C.value
            if C5_bound < best_C5:
                best_C5 = C5_bound
                best_h = h_sol.copy()
    except Exception:
        pass
    
    # Local refinement with scipy if needed
    if best_h is None:
        best_h = initial_h.copy()
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        # Normalize best_h
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    # Use scipy for final local refinement
    from scipy.optimize import minimize
    
    def objective(h_vals):
        h = np.array(h_vals)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-8})
    if res.success:
        final_h = res.x
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_C5 = objective(final_h)
        if final_C5 < best_C5:
            best_h = final_h
            best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
