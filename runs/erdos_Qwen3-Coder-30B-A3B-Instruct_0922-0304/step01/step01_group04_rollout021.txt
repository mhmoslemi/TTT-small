import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Create a sawtooth-like pattern
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = 1.0 - abs(np.sin(i))
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to optimize
    def objective(h):
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints
    def sum_constraint(h):
        return np.sum(h) - target_sum
    
    # Initial evaluation
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Perform optimization using scipy minimize with different methods
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for method in methods:
        if time.time() - start_time > budget_s * 0.95:
            break
        try:
            # Use a more sophisticated approach with bounds constraint
            bounds = [(0, 1) for _ in range(n_points)]
            options = {'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
            
            result = minimize(
                objective,
                current_h,
                method=method,
                bounds=bounds,
                options=options,
                callback=lambda x: None,
            )
            
            if result.success:
                optimized_h = result.x
                optimized_h = np.clip(optimized_h, 0, 1)
                optimized_h = optimized_h * target_sum / np.sum(optimized_h)
                optimized_c5 = evaluate_c5(optimized_h, dx)
                
                if optimized_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = optimized_c5
                    
        except Exception as e:
            continue
    
    # Final refinement with local search
    if time.time() - start_time < budget_s * 0.95:
        # Try a local perturbation approach
        for _ in range(20):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Small random perturbations
            perturbed_h = best_h + np.random.normal(0, 0.01, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            perturbed_c5 = evaluate_c5(perturbed_h, dx)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
    
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
