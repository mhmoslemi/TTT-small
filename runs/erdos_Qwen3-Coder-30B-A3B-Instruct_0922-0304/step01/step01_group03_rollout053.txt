import time
from scipy import signal
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    best_c5 = evaluate_c5(h, dx)
    
    # Constants
    alpha = 0.1
    gamma = 0.95
    alpha_min = 1e-6
    max_iter = 10000
    max_no_improvement = 500
    T0 = 1e-3
    cooling_rate = 0.995
    reset_freq = 5000
    iter_since_improvement = 0
    T = T0

    def evaluate_gradient(h, k_max):
        """Numerical gradient of overlap w.r.t h at the max overlap index."""
        eps = 1e-5
        grad = np.zeros_like(h)
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(len(h)):
            h_plus[i] += eps
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        return grad

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute overlaps
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]

        # Compute gradient at max overlap lag
        grad = evaluate_gradient(h, max_k)

        # Simulated Annealing step
        h_new = h - alpha * grad + T * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2.0 / np.sum(h_new))  # Renormalize to exact sum = len(h)/2

        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Metropolis acceptance probability
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                iter_since_improvement = 0
                best_h = h.copy()
        else:
            iter_since_improvement += 1

        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:
                alpha *= gamma
            T *= cooling_rate
            iter_since_improvement = 0

        # Periodic perturbation
        if iteration % reset_freq == 0 and iteration > 0:
            # Reinitialize h to a new binary configuration
            h = np.zeros(len(h))
            num_ones = len(h) // 2
            indices = np.random.choice(len(h), num_ones, replace=False)
            h[indices] = 1.0
            h = h * (len(h) / 2.0 / np.sum(h))  # Renormalize

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
