import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Best solution tracking
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise Frank-Wolfe style updates
    max_iter = 10000
    precision = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # If we're already at target, break early
        if max_overlap <= 0.3807:
            break
            
        # Compute gradient of overlap at k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = 1 - h[i + k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max]
        
        # Find index that most reduces the overlap
        i_star = np.argmin(grad)
        
        # Line search along the gradient direction
        # We want to move h[i_star] to minimize overlap at k_max
        # without violating constraints
        original_h_i = h[i_star]
        step_size = 1.0
        best_step = 0.0
        best_overlap = max_overlap
        
        # Simple binary search/line search
        for _ in range(20):
            # Try to move h[i_star] by step_size
            new_h = h.copy()
            new_h[i_star] += step_size
            # Adjust i_star+k_max term if it exists
            if i_star + k_max < n_points:
                new_h[i_star + k_max] -= step_size
            
            # Clip to [0,1]
            new_h = np.clip(new_h, 0, 1)
            
            # Re-normalize to maintain sum constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Compute new overlap
            new_overlap = evaluate_c5(new_h, dx)
            
            # Check if we improved
            if new_overlap < best_overlap:
                best_overlap = new_overlap
                best_step = step_size
                
            # Reduce step size
            step_size *= 0.5
            
        # Apply the best step
        if best_step > 0:
            h[i_star] += best_step
            if i_star + k_max < n_points:
                h[i_star + k_max] -= best_step
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
