import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def repair_h(h, target_sum):
    # Clamp to [0,1] and normalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_h = repair_h(h_init, target_sum)
    
    # DE Parameters
    de_popsize = 40
    de_maxiter = 100
    de_tol = 1e-6
    de_mutation = 0.8
    de_cross_prob = 0.9
    
    # L-BFGS-B Parameters
    lbfgs_maxiter = 500
    
    # Best solution tracking
    best_h = current_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # DE optimization
    def de_objective(h):
        h = repair_h(h, target_sum)
        return evaluate_c5(h, dx)
    
    # Bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=de_maxiter,
        popsize=de_popsize,
        mutation=de_mutation,
        recombination=de_cross_prob,
        tol=de_tol,
        seed=seed,
        polish=False
    )
    
    if de_result.success:
        h_de = repair_h(de_result.x, target_sum)
        c5_de = evaluate_c5(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with L-BFGS-B
    def lbfgs_objective(h):
        h = repair_h(h, target_sum)
        return evaluate_c5(h, dx)
    
    def lbfgs_constraint_sum(h):
        return np.sum(h) - target_sum
    
    # L-BFGS-B constraints
    constraints = {'type': 'eq', 'fun': lbfgs_constraint_sum}
    
    # Bounds for L-BFGS-B
    bnds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            lbfgs_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bnds,
            constraints=constraints,
            options={'maxiter': lbfgs_maxiter},
            callback=lambda x: None,
        )
        
        if result.success:
            h_refined = repair_h(result.x, target_sum)
            c5_refined = evaluate_c5(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Final check
    best_h = repair_h(best_h, target_sum)
    best_c5 = evaluate_c5(best_h, dx)
    
    # Final assertion
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
