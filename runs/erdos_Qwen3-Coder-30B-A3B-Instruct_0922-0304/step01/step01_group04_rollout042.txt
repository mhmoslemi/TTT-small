import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given step function h."""
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h = h / h.sum() * (n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure initial solution is valid
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.9995
    T_min = 1e-5
    max_iter = 5000
    
    T = T0
    iteration = 0
    
    # Annealing loop
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Generate neighbor solution
        h_trial = h.copy()
        
        # Randomly select about 5% of indices to modify
        mask = np.random.rand(n_points) < 0.05
        delta = np.random.uniform(-0.1, 0.1, size=(n_points)) * mask
        
        h_trial += delta
        
        # Project to valid range [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize sum to target
        if abs(np.sum(h_trial) - target_sum) > 1e-6:
            scale = target_sum / np.sum(h_trial)
            h_trial *= scale
        
        # Evaluate
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject
        delta_c5 = c5_trial - best_c5
