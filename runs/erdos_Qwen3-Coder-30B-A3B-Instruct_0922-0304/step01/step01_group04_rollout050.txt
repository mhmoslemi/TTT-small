import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Define bounds for each variable (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum
    def sum_constraint(x):
        return np.sum(x) - target_sum
    
    # Objective function for optimization
    def objective(x):
        # Ensure it's within bounds
        x = np.clip(x, 0, 1)
        # Normalize to satisfy sum constraint exactly
        x = x * target_sum / np.sum(x)
        return evaluate_c5(x, dx)
    
    # Add constraint to the bounds
    constraints = {'type': 'eq', 'fun': sum_constraint}
    
    # Allow time to run DE for a while
    # Return a basic solution if we're running low on time
    try:
        options = {'maxiter': 200, 'popsize': 15}
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            seed=seed,
            maxiter=100,
            popsize=15,
            tol=1e-6,
            recombination=0.9,
            mutation=0.8,
            callback=lambda x, convergence: None,
            disp=False,
        )
        if result.success:
            current_h = result.x
        else:
            # Fallback
            current_h = h_init.copy()
    except Exception:
        # Just use the initial h if anything goes wrong
        current_h = h_init.copy()
    
    # Ensure current_h meets all constraints
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Make sure the sum is exactly correct (within numerical precision)
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    # Ensure we don't run out of time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
