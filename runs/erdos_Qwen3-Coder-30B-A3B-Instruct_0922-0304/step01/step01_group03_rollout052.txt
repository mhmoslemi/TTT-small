import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Initialize h from initial_h_values if available

        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1e-3
    gamma = 0.995
    alpha = 0.1
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 500
    temperature = T0
    
    # Periodic perturbation
    reset_freq = 5000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Gradient computation using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Update h using Simulated Annealing
        noise = np.random.normal(0, temperature / 2.0, len(h))
        h_new = h - alpha * grad + noise
        h_new = np.clip(h_new, 0, 1)
        scale = 22.5 / np.sum(h_new)
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)

        new_c5 = evaluate_c5(h_new, dx)
        
        # Metropolis acceptance criterion
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            iter_since_improvement += 1
            # Accept with probability based on temperature
            delta_c5 = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta_c5 / temperature):
                h = h_new
            if iter_since_improvement > max_no_improvement:
                temperature *= gamma
                iter_since_improvement = 0

            temp_h = np.zeros_like(h)

                    temp_h[i:end_idx] = 1.0
            h = temp_h

            h = h * (22.5 / np.sum(h))
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
