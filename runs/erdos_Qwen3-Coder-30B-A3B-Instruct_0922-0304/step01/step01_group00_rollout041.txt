import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style coordinate-wise updates
    max_iter = 10000
    line_search_precision = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute local gradient of the max overlap with respect to h
        # We just work directly with the structure of correlation
        # For a fixed k_max that achieves the maximum, compute partial derivatives
        # This will be straightforward since it's only a linear function of nearby h[i]
        lk = max_k
        # The partial derivative of overlap at shift lk with respect to h[i]
        grad = np.zeros(n_points)
        # Just compute the extent of influence of h[i] on this overlap
        for i in range(n_points):
            if i + lk < n_points:
                grad[i] += (1 - h[i+lk])  # h[i] contributes to L(lk) through h[i+k] term
            if i >= lk: 
                grad[i] -= h[i-lk]  # h[i] contributes to itself in L(lk)
        
        # Choose the coordinate to update
        # We move in direction that decreases max overlap
        update_coordinate = np.argmin(grad)
        
        # Perform line search in the direction of update
        # try to minimize correlation at k_max by adjusting h[update_coordinate]
        old_val = h[update_coordinate]
        best_h_update = old_val
        best_overlap = max_overlap
        
        # Simple line search to find better h[update_coordinate]
        step_size = 0.01
        if old_val < 1 and old_val > 0:  # can move up or down
            for direction in [-1, 1]:  # try decreasing then increasing
                h_test = h.copy()
                test_val = old_val + direction * step_size
                test_val = np.clip(test_val, 0, 1)
                
                if test_val != old_val:
                    h_test[update_coordinate] = test_val
                    # Now we need to adjust the other components to keep sum constraint
                    # This is tricky with non-linear summation, so we do exact conservative update
                    try:
                        new_corr = correlate(h_test, 1 - h_test, mode="full")
                        new_max = new_corr[max_k] * dx
                        
                        if new_max < best_overlap:
                            best_overlap = new_max
                            best_h_update = test_val
                    except:
                        pass
                else:
                    continue  # no change in this case
        else: 
            if old_val >= 1:
                best_h_update = 1.0
            else:
                best_h_update = 0.0
        
        # Do the actual update
        h[update_coordinate] = best_h_update
        
        # Renormalize sum
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            # spread adjustment to keep sum equal to n_points/2
            adjustment = (n_points/2 - h_sum)/n_points
            h = h + adjustment
            # Ensure it stays within bounds
            h = np.clip(h, 0, 1)
            
            if abs(np.sum(h) - n_points / 2) > 1e-5:
                # Fix it to strictly satisfy sum constraint
                h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
