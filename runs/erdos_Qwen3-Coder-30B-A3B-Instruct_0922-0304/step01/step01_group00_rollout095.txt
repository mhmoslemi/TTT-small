import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    step_size = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential Convex Programming
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts to find k_max
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient at k_max
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
        
        # Hessian approximation: diagonal
        hess_diag = 1 - 2 * h
        
        # Solve convex subproblem: min_d (grad^T d + 0.5 * d^T H d) subject to 
        # h+d in [0,1]^n, sum(d) = 0
        # Projected gradient step
        d = -step_size * grad
        # Apply Hessian
        d = d - step_size * hess_diag * d
        
        # Maintain sum constraint by projecting
        d_sum = np.sum(d)
        d = d - d_sum / n_points
        
        # Apply update
        h_new = h + d
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
