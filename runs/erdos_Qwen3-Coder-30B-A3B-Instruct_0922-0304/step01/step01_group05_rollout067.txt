import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values"""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
