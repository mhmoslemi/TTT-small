import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Initialize h
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Constraints for the CVX problem
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum, t >= 0]
    
    # For each shift k, add the constraint
    # sum_{i} h[i] * (1 - h[i+k]) <= t
    # This is equivalent to sum_{i} h[i] - h[i] * h[i+k] <= t
    # i.e., sum_{i} h[i] - sum_{i} h[i] * h[i+k] <= t
    # Since sum(h) is fixed, this is just sum_{i} h[i] * h[i+k] >= sum(h) - t
    
    # Generate all correlations for different shifts k in [-n_points+1, n_points-1]
    # but we only care about k in [0, n_points-1] for the correlation computation
    for k in range(n_points):  # This is the shift
        # Indices for h[i] and h[i+k] where i+k < n_points
        # sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k]) <= t
        # i.e., sum h[i] - sum h[i] * h[i+k] <= t
        # and sum h[i] = target_sum for valid i, so:
        # sum h[i] * h[i+k] >= target_sum - t
        if k == 0:
            # Special case: h[i] * (1 - h[i]) which means sum(h) - sum(h^2) <= t
            # So sum(h^2) >= sum(h) - t, i.e., sum(h^2) >= target_sum - t
            constraints.append(cp.sum(cp.multiply(h, h)) >= target_sum - t)
        else:
            # sum h[i] * (1 - h[i+k]) for i = 0, ..., n_points-k-1
            # that is sum h[i] - sum h[i] * h[i+k] <= t (what we want)
            # So sum h[i] * h[i+k] >= sum h[i] - t
            # But we're summing over i from 0 to (n_points-k-1)
            # And the range here is [0, n_points-k-1] so there are n_points-k elements
            # h[i] * h[i+k] for i in range(n_points-k)
            corr = cp.sum(cp.multiply(h[:n_points-k], h[k:]))
            # sum h[i] for index i in 0 to n_points-k-1 = (n_points-k) * target_sum / n_points
            # but in our constraint, we use direct sum
            # In fact, better to correct this to ensure we correctly compute the actual sum
            # for each k, the valid indices are 0..(n_points-k-1)
            # Since we expect using linalg for summing over all possible indices
            # The sum for the first k terms we compute in poly quadratic extent
            # The actual constraint should be sum_{i} h[i] * (1 - h[i+k]) <= t
            
            # Simple evaluation approach, probably better to do it correctly
            # But we'll directly handle the constraint correctly in a loop
            # i: from 0 to n_points - k - 1 
            # this is from n_points -k  elements
            # Summation is sum_{i = 0}^{n_points-k-1} h[i] - h[i] * h[i+k] <= t
            # The sum of h[i] for i in range(n_points-k) is not fixed, so we compute directly:
            # Split h into two arrays for careful indexing
            left = cp.sum(h[:n_points-k])
            right = cp.sum(cp.multiply(h[:n_points-k], h[k:]))
            # h[i] - h[i]*h[i+k] for i = 0 to ... etc
            # Sum of difference = left - right
            # We want left - right <= t
            # So right >= left - t
            # But forтехнического consistency, we prefer left - right <= t  
            constraints.append( left - right <= t )
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with MOSEK if available
    # If not available, fallback to ECOS or another
    solver = cp.MOSEK
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                # Fallback to a simpler one: just not optimizing, use the initial
                h_solution = h_init.copy()
                # Ensure it's within bounds
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * target_sum / np.sum(h_solution)
                c5_bound = evaluate_c5(h_solution, dx)
                return (h_solution, c5_bound, n_points)
    
    # Extract solution
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_solution = h.value.copy()
    else:
        h_solution = h_init
        h_solution = np.clip(h_solution, 0, 1)
        h_solution = h_solution * target_sum / np.sum(h_solution)
    
    # Postprocess and validate
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * target_sum / np.sum(h_solution)
    c5_bound = evaluate_c5(h_solution, dx)
    
    # Final validation checks
    assert np.isclose(np.sum(h_solution), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_solution), "Bounds constraint violated"
    
    return (h_solution, c5_bound, n_points)
