import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Scale to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Rerun scaling due to clipping
    h = h * (n_points / 2) / np.sum(h)
    return h

def prox_ψ(z, alpha):
    """Proximal operator for negative entropy mirror map"""
    u = z.copy()
    for i in range(len(u)):
        if u[i] <= 0:
            u[i] = 0
        elif u[i] >= 1:
            u[i] = 1
        else:
            u[i] = u[i] / (1 + (1 - u[i]) * (np.exp(alpha) - 1) / (u[i] * (1 - u[i])))
    return u

def compute_gradient(h, dx, max_lag_idx):

    n = len(h)
    grad = np.zeros(n)

    for i in range(n):
        j = (i - max_lag_idx) % n
        grad[i] = 2 * (h[i] - 1) * (1 - h[j])
    
    return grad * dx
