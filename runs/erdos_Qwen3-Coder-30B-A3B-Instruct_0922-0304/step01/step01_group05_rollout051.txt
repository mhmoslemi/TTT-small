import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_time = start_time + budget_s - 1  # Leave 1s for finalization
    
    # Initial guess from provided construction
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Try convex optimization approach
    # Variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C >= 0]
    
    # Generate all overlap constraints
    # For each possible shift k, compute the overlap integral
    # We loop over k from -(n_points-1) to n_points-1
    for k in range(-(n_points - 1), n_points):
        if k < 0:
            # Right overlap: h[i] * (1 - h[i+k]) for i = 0, ..., n_points-1+|k|
            expr = 0
            for i in range(n_points + k):
                expr += h[i] * (1 - h[i - k])
        elif k == 0:
            continue  # This is just a sum of h[i]*(1-h[i]), not a shift
        else:
            # Left overlap: h[i] * (1 - h[i+k]) for i = 0, ..., n_points-1-k
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i + k])
        
        constraints.append(expr <= C * dx)
    
    # Note: This way of coding the constraints can be quite lengthy.
    # In practice, we can be more efficient. But for clarity we keep the
    # constraint forming logic visible.
    
    # Objective
    objective = Minimize(C)
    
    # Problem
    prob = Problem(objective, constraints)
    
    # Solve with SCS
    try:
        prob.solve(solver=SCS, max_it=5000, verbose=False, eps=1e-6, 
                   acceleration_lookback=10, normalize=True, 
                   use_indirect=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_sol = h.value
            C_sol = C.value
        else:
            # Fallback to simpler optimization
            h_sol = h_init
            C_sol = float('inf')
    except:
        # If convex optimization fails for any reason, fall back to
        # local optimization with a linearized approach
        h_sol = h_init
        C_sol = float('inf')
    
    best_h = h_sol
    best_C = C_sol
    
    # Final validation and coarse optimization refinement
    if best_C == float('inf'):
        # Match the old approach but refine it with a hybrid strategy
        from scipy.optimize import minimize
        
        def compute_C5(h):
            h = np.array(h)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        def objective(h):
            return compute_C5(h)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        h_init = h_init.copy()
        res = minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 1000, 'ftol': 1e-7},
            callback=lambda x, _ : None  # Avoid callback logging
        )
        
        if res.success:
            h_final = res.x * (n_points / 2) / np.sum(res.x)
            final_C5 = compute_C5(h_final)
            best_h = h_final
            best_C = final_C5
    
    # Ensure proper final normalization and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_C, n_points)
