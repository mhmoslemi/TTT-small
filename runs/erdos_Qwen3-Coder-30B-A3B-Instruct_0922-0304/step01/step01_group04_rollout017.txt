import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a good starting point
        n_points = 42
        h_init = np.linspace(0.9, 0.1, n_points)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure initial h is feasible
    current_h = np.clip(h_init, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h, dx)
    
    # Simulated Annealing parameters
    temperature = 1.0
    cooling_rate = 0.9995
    min_temperature = 1e-5
    max_iterations = 5000
    
    iteration = 0
    while iteration < max_iterations and temperature > min_temperature:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor solution
        h_trial = current_h.copy()
        
        # Randomly perturb 5% of the points
        indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_trial[i] += delta
            
        # Project back to [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize to maintain sum constraint
        h_trial = h_trial * target_sum / np.sum(h_trial)
        
        # Evaluate the trial solution
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject based on simulated annealing probability
        delta_c5 = c5_trial - best_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / temperature) > np.random.rand():
            current_h = h_trial
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
        
        # Cool down
        temperature *= cooling_rate
        iteration += 1
    
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h, final_c5, n_points)
