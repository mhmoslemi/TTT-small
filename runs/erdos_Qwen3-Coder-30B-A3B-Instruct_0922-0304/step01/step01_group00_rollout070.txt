import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential Convex Programming
    max_iter = 1000
    step_size = 0.1
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Gradient approximation for the worst-case overlap
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
            
        # Hessian approximation - diagonal entries
        hess_diag = 1 - 2 * h
        
        # Solve trust region subproblem
        # min_d (grad^T * d + 0.5 * d^T * H * d) subject to h+d in [0,1], sum(d) = 0
        # Projected gradient descent step
        d = -step_size * grad
        # Project to box constraints [0,1]
        d_proj = d.copy()
        for i in range(n_points):
            if h[i] + d[i] < 0:
                d_proj[i] = -h[i]
            elif h[i] + d[i] > 1:
                d_proj[i] = 1 - h[i]
        
        # Adjust to satisfy sum constraint
        d_proj = d_proj - np.sum(d_proj) / n_points
        
        # Update h
        h = h + d_proj
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Reduce step size gradually
        step_size *= 0.995
        
    return (best_h.tolist(), best_c5, n_points)
