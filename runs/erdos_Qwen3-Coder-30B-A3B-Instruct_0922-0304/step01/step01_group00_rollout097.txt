import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization loop
    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 1000
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient approximations
        # For each position i, compute the gradient of overlap with respect to h[i]
        grad = np.zeros(n_points)
        eps = 1e-4
        
        # Compute gradient for each variable
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive step sizing
        step_size *= 0.999

    return (best_h.tolist(), best_c5, n_points)
