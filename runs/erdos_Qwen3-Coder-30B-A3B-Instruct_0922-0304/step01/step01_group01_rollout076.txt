import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clamp to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm(seed, budget_s, n_points):
    np.random.seed(seed)
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3
    stagnation_generations = 20
    max_generations = 1000
    
    # Initial population
    population = []
    if 'initial_h_values' in globals():
        h_init = project_to_feasible(initial_h_values.copy(), n_points)
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_feasible(h_init, n_points)
        population.append(h_init)
    
    # Fill population
    for _ in range(pop_size - len(population)):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    last_best_c5 = best_c5
    
    while time.time() - start_time < budget_s * 0.95 and generation < max_generations:
        generation += 1
        
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, 2.0 / n_points) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count >= stagnation_generations:
            # Restart with new random population
            population = []
            if 'initial_h_values' in globals():
                h_init = project_to_feasible(initial_h_values.copy(), n_points)
                population.append(h_init)
            else:
                h_init = np.ones(n_points) * 0.5
                h_init = project_to_feasible(h_init, n_points)
                population.append(h_init)
            for _ in range(pop_size - len(population)):
                h = np.random.uniform(0, 1, n_points)
                h = project_to_feasible(h, n_points)
                population.append(h)
            stagnation_count = 0
            last_best_c5 = best_c5
            continue
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(range(pop_size), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size]
            
            # Crossover: average
            child1 = (parent1 + parent2) / 2.0
            child2 = (parent1 + parent2) / 2.0
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
                child1 = project_to_feasible(child1, n_points)
            
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
                child2 = project_to_feasible(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 42
        
    return genetic_algorithm(seed, budget_s, n_points)
