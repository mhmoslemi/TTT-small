import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initial guess from provided construction
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_C5 = float('inf')
    best_h = h_init.copy()
    
    # We'll try convex optimization approach with QCP
    # Set up CVXPY variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Generate all the overlap constraints
    # For each lag k, we need to compute sum_{i} h[i] * (1 - h[i+k])
    # We only go up to the valid range
    
    # Precompute indices for all lags
    for k in range(-(n_points-1), n_points):
        if k >= 0:
            # lag k: h[i] * (1 - h[i+k]) for i in [0, n_points-1-k]
            valid_indices = list(range(n_points - k))
            expr = sum(h[i] * (1 - h[i+k]) for i in valid_indices)
        else:
            # lag k: h[i] * (1 - h[i+k]) for i in [0, n_points+ k -1] 
            valid_indices = list(range(-k, n_points))
            expr = sum(h[i] * (1 - h[i+k]) for i in valid_indices)
        
        # Add constraint that this expression is at most C*dx
        constraints.append(expr <= C * dx)
    
    # Create problem
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    # Solve with SCS
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=5000, 
                   eps=1e-5, use_indirect=True)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            computed_C = C.value
            if computed_C < best_C5:
                best_C5 = computed_C
                best_h = h.value
    except:
        # Fall back to heuristic
        pass
    
    # Final evaluation of best solution found
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    if best_C5 == float('inf'):
        # Fallback to SLSQP if no convex solution found
        from scipy.optimize import minimize
        
        def objective(h_vals):
            return compute_C5(h_vals)
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': constraint_sum}
        ]
        
        res = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 500, 'ftol': 1e-6})
        
        if res.success:
            best_h = res.x
            best_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
