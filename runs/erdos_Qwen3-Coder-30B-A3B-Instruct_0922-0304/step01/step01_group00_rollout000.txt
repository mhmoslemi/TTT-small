import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum, norm
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a structured pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing schedule
    max_iter = 3000
    temp_start = 1.0
    temp_decay = 0.995
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Local gradient-based update
        step_size = 0.005
        iterations = 100

        h_temp = h.copy()
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute overlap and gradient
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_k = np.argmax(corr)
            c5_val = corr[max_k] * dx

            # Estimate gradient
            grad = np.zeros_like(h_temp)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = corr_plus[max_k] * dx
                grad[i] = (c5_plus - c5_val) / eps

            # Update with gradient descent
            h_temp -= step_size * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)

        h = h_temp
        
        # Add noise to escape local minima
        if iteration % 10 == 0:
            noise_scale = temp_start * (temp_decay ** iteration)
            h += np.random.normal(0, noise_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Periodic feasibility correction
        if iteration % 50 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
