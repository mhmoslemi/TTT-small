import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # LP-based optimization
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Solve a relaxed LP
        n_points = len(h)
        h_var = cp.Variable(n_points, name="h")
        C = cp.Variable(name="C")
        
        # Define constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == 22.5
        ]
        
        # Add overlap constraints
        # Consider a subset of critical lags to reduce problem size
        lags_to_consider = min(2 * n_points - 1, 100)  # Limit to 100 lags for efficiency
        for k in range(lags_to_consider):
            # Create the correlation term for lag k
            corr_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_expr += h_var[i] * (1.0 - h_var[j])
            constraints.append(C >= corr_expr * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = np.array(h_var.value).reshape(-1)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (22.5 / np.sum(h_new))
                new_c5 = evaluate_c5(h_new, dx)
                
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    iter_since_improvement = 0
                    best_h = h.copy()
                else:
                    iter_since_improvement += 1
                    if iter_since_improvement > max_no_improvement:
                        # Perturb to escape local minima
                        perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                        h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
                        h = np.clip(h, 0, 1)
                        h = h * (22.5 / np.sum(h))
                        iter_since_improvement = 0
            else:
                # Fallback to simple random perturbation
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (22.5 / np.sum(h))
        except Exception:
            # Fallback to simple random perturbation
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (22.5 / np.sum(h))
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
