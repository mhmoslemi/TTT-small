import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    n_points = 45  # As used in previous solution
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        population.append(h)
    else:
        # Create a random initial solution
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h)
        
    # Fill population with diverse random solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h)
        
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 50
    mutation_noise = 0.05
    generation = 0
    max_generations = 2000
    
    while generation < max_generations and time.time() - start_time < budget_s - 0.1:
        # Evaluate fitness of all individuals
        fitness = []
        for h in population:
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep top 10%
        elite_count = max(1, pop_size // 10)
        elite = population[:elite_count]
        
        # Create new population via crossover and mutation
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two parents (tournament selection)
            parent1 = population[np.random.randint(0, pop_size // 2)]
            parent2 = population[np.random.randint(0, pop_size // 2)]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutate_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            for i in mutate_indices:
                child[i] += np.random.normal(0, mutation_noise)
                
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            scale = (n_points / 2.0) / np.sum(child)
            child = child * scale
            
            new_population.append(child)
            
        # Replace population
        population = new_population[:pop_size]
        
        # Adaptive mutation noise
        if no_improvement_count >= 20:
            mutation_noise *= 0.9
            no_improvement_count = 0
            
        generation += 1
        no_improvement_count += 1
        
        # Periodic reset
        if no_improvement_count > max_no_improvement:
            for i in range(pop_size // 5):
                idx = np.random.randint(0, pop_size)
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0) / np.sum(h)
                population[idx] = h
            no_improvement_count = 0
          
    return (best_h.tolist(), best_c5, n_points)
