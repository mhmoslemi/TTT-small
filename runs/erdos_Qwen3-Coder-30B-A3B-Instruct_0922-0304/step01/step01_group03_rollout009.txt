import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    n_points = 45
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a random binary-like vector
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
    
    # Normalize to exact sum
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Add the initial solution to population
    population.append(h.copy())
    
    # Fill remaining population with diverse solutions
    for _ in range(pop_size - 1):
        h_new = np.zeros(n_points)
        # Randomly assign 22 values as 1, rest 0
        ones_indices = np.random.choice(n_points, size=22, replace=False)
        h_new[ones_indices] = 1.0
        # Add some noise
        noise = np.random.normal(0, 0.1, n_points)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        population.append(h_new)
        
    best_h = population[0].copy()
    best_c5 = evaluate_c5(population[0], 2.0 / n_points)
    
    # Parameters
    generations = 2000
    crossover_rate = 0.7
    mutation_rate = 0.05
    elitism_rate = 0.1
    diversification_threshold = 50
    no_improvement_count = 0
    mutation_noise = 0.05
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Diversification
        if no_improvement_count > diversification_threshold:
            for i in range(int(0.2 * pop_size)):
                idx = np.random.randint(0, pop_size)
                h_new = np.zeros(n_points)
                ones_indices = np.random.choice(n_points, size=22, replace=False)
                h_new[ones_indices] = 1.0
                noise = np.random.normal(0, mutation_noise, n_points)
                h_new += noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                population[idx] = h_new
            mutation_noise *= 0.9
            no_improvement_count = 0
            
        # Selection (top 50%)
        n_parents = pop_size // 2
        parents = population[:n_parents]
        
        # Elitism: keep top 10%
        new_population = population[:int(elitism_rate * pop_size)]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two parents
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            else:
                child = p1.copy()
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, mutation_noise)
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2.0) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:pop_size]
        
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
