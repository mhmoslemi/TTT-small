import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper to project solution back to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    # Objective function to minimize (C5)
    def objective(h):
        h = project(h)
        return evaluate_c5(h, dx)
    
    # Use differential evolution for global search
    bounds = [(0, 1) for _ in range(n_points)]
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
        tol=1e-6
    )
    
    # Refine with L-BFGS-B
    h_best = project(de_result.x)
    result = minimize(
        objective,
        h_best,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    if result.success:
        h_best = project(result.x)
    else:
        h_best = project(h_best)

    # Final evaluation
    current_c5 = evaluate_c5(h_best, dx)
    current_h = h_best.copy()
    
    # Ensure all constraints are met
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
