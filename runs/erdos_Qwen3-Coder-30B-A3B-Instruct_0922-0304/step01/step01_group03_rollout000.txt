import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():

    else:

        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 to get sum = 22
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    
    # Simulated Annealing parameters
    T0 = 1e-3
    T = T0
    cooling_rate = 0.995
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps using FFT for efficiency
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Gradient computation using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Simulated Annealing update
        h_new = h - alpha * grad + T * np.random.normal(0, 1, len(h))
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (22.5 / np.sum(h_new))  # Renormalize to exact sum
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update using Metropolis criterion
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
            iter_since_improvement += 1
        
        # Update temperature
        if iter_since_improvement > max_no_improvement:
            if alpha > alpha_min:
                alpha *= gamma
            iter_since_improvement = 0
            T *= cooling_rate  # Cool down temperature
        
        # Periodic binary reset to explore new areas
        if iteration % 5000 == 0 and iteration > 0:
            # Create new binary configuration
            new_h = np.zeros(len(h))
            # Split into random blocks
            block_size = np.random.randint(3, 10)
            blocks = np.random.choice([0, 1], size=len(h) // block_size)
            for i, block_val in enumerate(blocks):
                start_idx = i * block_size
                end_idx = min((i + 1) * block_size, len(h))
                new_h[start_idx:end_idx] = block_val
            h = new_h.copy()
            h = np.clip(h, 0, 1)
            h = h * (22.5 / np.sum(h))

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
