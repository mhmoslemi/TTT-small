import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Get initial guess
    h_init = np.array(initial_h_values)
    
    # Initialize with normalized values
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define optimization variables
    h = Variable(n_points)
    
    # Objective: minimize max overlap
    # Using the fact that the overlap is a convolution, we can introduce auxiliary variables
    # to make the problem convex
    C_max = Variable()
    
    # Create the constraint using CVXPY's built-in support for convolutions
    # We'll use the fact that the problem can be formulated using a transformation
    # to a manageable convex form
    
    # For details, this uses the spectral approach:
    # 1. We formulate a linear program using the Fourier domain definition
    # 2. Using the constraint that sum = n_points / 2
    # 3. Only minimizing over the maximum overlap
    
    # We use a relaxation that approximates the maximum over all lags via a linear constraint
    # This is a close approximation for minimum overlap maximization
    
    # Constraint for the integral
    integral_constraint = cp.sum(h) == n_points / 2
    
    # Bound constraints
    bound_constraints = [h >= 0, h <= 1]
    
    # Approach: heuristic optimization with gradient descent
    def compute_C5_manual(h_vals):
        h_vals = np.array(h_vals)
        # Ensure the constraint sum = n_points/2
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        # Compute full correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    # Gradient-based multistart optimization
    best_h = h_init.copy()
    best_C5 = compute_C5_manual(best_h)
    
    # Run for at most budget_s seconds
    while time.time() - start_time < budget_s - 1:
        # Try many random starting points with random perturbations
        for _ in range(10):
            if time.time() - start_time >= budget_s - 1:
                break
                
            h_start = h_init + np.random.normal(0, 0.01, n_points)
            h_start = np.clip(h_start, 0, 1)
            # Normalize to integral constraint
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
            # Optimization step
            try:
                res = minimize(
                    lambda x: compute_C5_manual(x), 
                    h_start,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(n_points)],
                    options={'maxiter': 100}
                )
                if res.success:
                    temp_h = res.x
                    temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
                    temp_C5 = compute_C5_manual(temp_h)
                    if temp_C5 < best_C5:
                        best_C5 = temp_C5
                        best_h = temp_h.copy()
            except:
                continue
                
    # Return final result
    return (best_h.tolist(), best_C5, n_points)
