import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    n_points = 45
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create binary-like initial vector
        h = np.zeros(n_points)
        h[:23] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2.0 / np.sum(h))
    
    # Fill population
    population.append(h.copy())
    for _ in range(population_size - 1):
        h_new = np.zeros(n_points)
        indices = np.random.choice(n_points, size=23, replace=False)
        h_new[indices] = 1.0
        h_new = h_new * (n_points / 2.0 / np.sum(h_new))
        population.append(h_new)
    
    dx = 2.0 / n_points
    best_h = population[0].copy()
    best_c5 = evaluate_c5(population[0], dx)
    
    # Evolution parameters
    generations = 2000
    elite_size = population_size // 10
    mutation_rate = 0.05
    noise_std = 0.05
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Crossover and mutation
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < population_size:
            # Selection (tournament)
            parent1_idx = np.random.choice(elite_size)
            parent2_idx = np.random.choice(elite_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_mask = np.random.random(n_points) < mutation_rate
            if np.any(mutation_mask):
                noise = np.random.normal(0, noise_std, n_points)
                child = child + noise * mutation_mask
                child = np.clip(child, 0, 1)
                scale_factor = (n_points / 2.0) / np.sum(child)
                child = child * scale_factor
                
            new_population.append(child)
        
        population = new_population
        
        # Diversification
        if no_improvement_count > max_no_improvement:
            # Introduce random individuals
            for i in range(population_size // 5):
                h_new = np.zeros(n_points)
                indices = np.random.choice(n_points, size=23, replace=False)
                h_new[indices] = 1.0
                h_new = h_new * (n_points / 2.0 / np.sum(h_new))
                population[i] = h_new
            noise_std *= 0.9
            no_improvement_count = 0
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
