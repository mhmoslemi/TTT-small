import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select one individual via tournament selection"""
    indices = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Uniform crossover with averaging"""
    child = (parent1 + parent2) / 2.0
    return child

def mutate(individual, mutation_rate=0.1, noise_scale=0.1):
    """Gaussian mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, noise_scale, len(individual))
    individual = individual + mask * noise
    return individual
