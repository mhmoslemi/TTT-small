import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    max_no_improvement = 1000
    iter_since_improvement = 0

    def project_and_normalize(h):
        """Project h to [0,1] and normalize to sum to n_points/2"""
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        return h * scale

    def evaluate_c5(h, dx):
        """Evaluate C5 given h and dx."""
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * dx

    # Initialize simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        h_pert = h + simplex_size * np.random.normal(0, 1, n_points)
        h_pert = project_and_normalize(h_pert)
        simplex.append(h_pert)

    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Evaluate all simplex points
        evaluations = [evaluate_c5(s, dx) for s in simplex]
        sorted_indices = np.argsort(evaluations)
        best_idx = sorted_indices[0]
        worst_idx = sorted_indices[-1]

        # Update best solution
        if evaluations[best_idx] < best_c5:
            best_h = simplex[best_idx].copy()
            best_c5 = evaluations[best_idx]
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1

        if iter_since_improvement > max_no_improvement:
            if iteration % 1000 == 0:
                # Perturb best solution to escape local minima
                perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                best_h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
                best_h = project_and_normalize(best_h)
                # Replace worst with perturbed best
                simplex[worst_idx] = best_h.copy()
                iter_since_improvement = 0
            else:
                break

        # Build new simplex
        best_s = simplex[best_idx]
        worst_s = simplex[worst_idx]
        centroid = np.mean([s for i, s in enumerate(simplex) if i != worst_idx], axis=0)
        
        # Reflection
        h_reflect = centroid + reflection_factor * (centroid - worst_s)
        h_reflect = project_and_normalize(h_reflect)
        c_reflect = evaluate_c5(h_reflect, dx)

        if c_reflect < evaluations[best_idx]:
            # Expansion
            h_expand = centroid + expansion_factor * (h_reflect - centroid)
            h_expand = project_and_normalize(h_expand)
            c_expand = evaluate_c5(h_expand, dx)
            if c_expand < c_reflect:
                simplex[worst_idx] = h_expand
            else:
                simplex[worst_idx] = h_reflect
        else:
            if c_reflect < evaluations[worst_idx]:
                simplex[worst_idx] = h_reflect
            else:
                # Contraction
                h_contract = centroid + contraction_factor * (worst_s - centroid)
                h_contract = project_and_normalize(h_contract)
                c_contract = evaluate_c5(h_contract, dx)
                if c_contract < evaluations[worst_idx]:
                    simplex[worst_idx] = h_contract
                else:
                    # Shrink
                    for i in range(len(simplex)):
                        if i != best_idx:
                            simplex[i] = best_s + 0.5 * (simplex[i] - best_s)
                            simplex[i] = project_and_normalize(simplex[i])

    return (best_h.tolist(), best_c5, len(best_h))
