import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T = 0.05
    cooling_factor = 0.9998
    max_iter = int(1e5)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Alternative perturbation types
    def perturb_swap(h, n_points):
        # Randomly swap two values
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        return h_new
    
    def perturb_noise(h, n_points, dx):
        # Add noise to two random points while maintaining sum constraint
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        perturbation = np.random.normal(0, 0.01)
        h_new[i] += perturbation
        h_new[j] -= perturbation
        # Keep values within bounds
        h_new = np.clip(h_new, 0, 1)
        # Normalize to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Choose perturbation type
        if np.random.random() < 0.7:
            # Swap perturbation
            h_new = perturb_swap(h, n_points)
        else:
            # Noise perturbation
            h_new = perturb_noise(h, n_points, dx)

        # Compute new C5
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.random() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        T *= cooling_factor
        
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
