import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.zeros(n_points)
        h_init[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h_init)
    
    # Normalize h to satisfy sum condition
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    dx = 2.0 / n_points
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Optimization loop
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 1000
    alpha = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Use CVXPY to solve the convex relaxation
        h = Variable(n_points)
        C = Variable()
        
        # Objective
        objective = Minimize(C)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(sum(h) == n_points / 2)
        
        # Generate critical lags (full convolution)
        lags = range(2 * n_points - 1)
        for k in lags:
            # Compute overlap term for lag k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            constraints.append(C >= expr * dx)
        
        # Create and solve problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=CPLEX, verbose=False, max_iters=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h.value).flatten()
                new_c5 = prob.value
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
            else:
                iter_since_improvement += 1
        except:
            iter_since_improvement += 1
        
        # Apply gradient kind of update to escape local minima
        if iter_since_improvement > 0 and iter_since_improvement % 100 == 0:
            h_perturb = best_h + alpha * (np.random.rand(n_points) - 0.5)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            new_c5 = evaluate_c5(h_perturb, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_perturb.copy()
                iter_since_improvement = 0
        
        if iter_since_improvement > max_no_improvement:
            break

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = np.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
