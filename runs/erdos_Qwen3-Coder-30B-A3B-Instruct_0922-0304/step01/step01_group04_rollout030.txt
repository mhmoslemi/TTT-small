import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Helper function to ensure constraints and compute objective
    def objective(h):
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        h = h * target_sum / np.sum(h)  # Ensure sum constraint
        return evaluate_c5(h, dx)

    # Constraints
    def constraint_sum(h):
        return np.sum(h) - target_sum

    # DE optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    maxiter = 100
    popsize = 40
    mutation = (0.5, 1)
    recombination = 0.9

    # DE optimization
    result_de = differential_evolution(
        objective,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        seed=seed,
        disp=False
    )

    # L-BFGS-B refinement using the best DE solution
    h_lbfgs = result_de.x.copy()
    h_lbfgs = np.clip(h_lbfgs, 0, 1)
    h_lbfgs = h_lbfgs * target_sum / np.sum(h_lbfgs)

    # Local optimization
    options = {'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9}
    constraints = {'type': 'eq', 'fun': constraint_sum}

    try:
        result_lbfgs = minimize(
            objective,
            h_lbfgs,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=constraints,
            options=options,
            callback=lambda x: None
        )
        
        if result_lbfgs.success:
            h_final = result_lbfgs.x.copy()
        else:
            h_final = h_lbfgs.copy()
    except:
        h_final = h_lbfgs.copy()

    # Final validation and repair
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    current_c5 = evaluate_c5(h_final, dx)

    # Ensure correctness
    assert np.isclose(np.sum(h_final), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in h_final), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (h_final, current_c5, n_points)
