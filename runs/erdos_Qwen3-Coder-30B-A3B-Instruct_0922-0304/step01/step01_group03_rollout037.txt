import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturbation_freq = 1000
    
    # Initialize simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        h_pert = h + simplex_size * np.random.normal(0, 1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        scale = 22.5 / np.sum(h_pert)
        h_pert = h_pert * scale
        simplex.append(h_pert)
    
    # Helper functions
    def project_and_normalize(h):
        h = np.clip(h, 0, 1)
        scale = 22.5 / np.sum(h)
        h = h * scale
        return h
    
    def evaluate_c5(h, dx):
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlate_result) * dx
    
    # Main optimization loop
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(50000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate all points in simplex
        vals = []
        for point in simplex:
            val = evaluate_c5(point, dx)
            vals.append(val)
        
        # Sort simplex points by function value
        sorted_indices = np.argsort(vals)
        simplex = [simplex[i] for i in sorted_indices]
        vals = [vals[i] for i in sorted_indices]
        
        # Check if we've improved
        if vals[0] < best_c5:
            best_c5 = vals[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Stop if no improvement for a while
        if iter_since_improvement > max_no_improvement:
            break
        
        # Compute centroid (excluding worst point)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflect
        reflected = centroid + reflection_factor * (centroid - simplex[-1])
        reflected = project_and_normalize(reflected)
        reflected_val = evaluate_c5(reflected, dx)
        
        # If reflection improves, try to expand
        if vals[0] > reflected_val > vals[-2]:
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = project_and_normalize(expanded)
            expanded_val = evaluate_c5(expanded, dx)
            if expanded_val < reflected_val:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        # If reflected is better than all but worst, replace worst
        elif reflected_val < vals[0]:
            simplex[-1] = reflected
        # Otherwise, do contraction
        else:
            if reflected_val < vals[-1]:
                contracted = centroid + contraction_factor * (reflected - centroid)
            else:
                contracted = centroid + contraction_factor * (simplex[-1] - centroid)
            contracted = project_and_normalize(contracted)
            contracted_val = evaluate_c5(contracted, dx)
            if contracted_val < vals[-1]:
                simplex[-1] = contracted
            else:
                # Shrink simplex
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_and_normalize(simplex[i])
        
        # Periodic perturbation
        if iteration % perturbation_freq == 0 and iteration > 0:
            h_pert = best_h + 0.01 * np.random.normal(0, 1, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            scale = 22.5 / np.sum(h_pert)
            h_pert = h_pert * scale
            simplex[-1] = h_pert
    
    return (best_h.tolist(), best_c5, len(best_h))
