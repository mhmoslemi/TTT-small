import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential convex programming approach
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Compute gradient of the worst-case overlap
        # For shift k_max that gives max overlap
        # L(k) = sum h[i] * (1 - h[i+k])
        # gradient w.r.t. h[i] is:
        #   for i < n_points - k_max: 1 - h[i+k_max]  
        #   for i >= n_points - k_max: 0
        # Plus gradient from shift k_max of the 1-h term
        # The relevant part of gradient is:
        # sum of 1 - h[i+k_max] over i where i+k_max < n_points
        
        grad = np.zeros(n_points)
        # Based on derivative of sum_{i} h[i] * (1 - h[i+k])
        # Gradient in i direction:
        # For i in [0, n_points - k_max - 1]
        # Gradient = (1 - h[i + k_max])
        # But there are other terms too (the 1-h part)
        # In standard correlation notation:
        # grad[i] ~   (1 - h[i+k_max]) - (1 - h[i]) 
        # Where the first term is derived from correlation and the second from the derivative of 1-h
        # This is an approximation
        
        # Better gradient approximation using finite difference
        # Take a small step in direction of gradient estimate
        step_size = 0.01
        h_new = h.copy()
        eps = 1e-5
        
        # Use finite differences to estimate subgradient
        for i in range(n_points):
            h_temp = h.copy()
            h_temp[i] += eps
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            overlap_large = correlate(h_temp, 1 - h_temp, mode="full")
            c5_large = np.max(overlap_large) * dx
            
            h_temp = h.copy()
            h_temp[i] -= eps
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            overlap_small = correlate(h_temp, 1 - h_temp, mode="full")
            c5_small = np.max(overlap_small) * dx
            
            grad[i] = (c5_large - c5_small) / (2 * eps)
            
        # Project gradient into feasible set (i.e., direction that decreases C5)
        # Update h in gradient direction
        h_new = h - step_size * grad
        
        # Project to feasible set [0,1] and maintain sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
