import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def genetic_algorithm(seed, budget_s, n_points):
    np.random.seed(seed)
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3
    stagnation_generations = 20
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
        h_init = project_to_feasible(h_init, n_points)
        population.append(h_init)
    
    # Fill the rest with random solutions
    while len(population) < pop_size:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for stagnation
        if generation > 0 and fitness[0] >= best_c5:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count >= stagnation_generations:
            break
            
        # Create new population
        new_population = [population[0]]  # Elitism
        
        while len(new_population) < pop_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(population[winner_idx])
            
            # Crossover (average)
            child = (parents[0] + parents[1]) / 2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
            
            # Project to feasible region
            child = project_to_feasible(child, n_points)
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    return best_h, best_c5
