import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available, otherwise construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Simulated Annealing parameters
    T = 0.05
    cooling_rate = 0.9998
    max_iter = int(budget_s * 1000)  # Roughly 1000 iterations per second
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Choose perturbation type
        if np.random.random() < 0.7:
            # Binary swap perturbation
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Gaussian noise perturbation
            h_new = h.copy()
            # Pick two random indices
            i, j = np.random.choice(n_points, 2, replace=False)
            noise_i = np.random.normal(0, 0.01)
            noise_j = np.random.normal(0, 0.01)
            # Apply noise but ensure constraint
            h_new[i] += noise_i
            h_new[j] -= noise_i  # Ensure sum constraint
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            # Adjust to maintain sum = n_points / 2
            current_sum = np.sum(h_new)
            adjustment = (n_points / 2 - current_sum) / n_points
            h_new += adjustment
        
        # Ensure sum constraint is maintained
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5 or np.random.random() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        T *= cooling_rate
        
        # Check if target bound is achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
