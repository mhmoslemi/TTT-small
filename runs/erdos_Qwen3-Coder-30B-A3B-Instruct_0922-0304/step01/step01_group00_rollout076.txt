import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a simple pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check if we've reached our target
        if best_c5 <= 0.3807:
            break

        # Local gradient-based update
        step_size = 0.01
        grad_iterations = 50
        
        h_temp = h.copy()
        for _ in range(grad_iterations):
            # Compute current overlap
            overlap = correlate(h_temp, 1 - h_temp, mode="full")
            max_overlap_idx = np.argmax(overlap)
            max_overlap_val = overlap[max_overlap_idx] * dx
            
            # Compute gradient via finite differences
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] = min(1.0, h_plus[i] + eps)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap_val) / eps
            
            # Update h
            h_temp -= step_size * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp

    # Final verification
    final_corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
