import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    swarm = []
    velocities = []

    h_init_feasible = project_to_feasible(h_init, n_points)

    # Initialize remaining particles randomly
    for i in range(n_particles - 1):

        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best.append(h_feasible.copy())

    best_idx = np.argmin(p_best_fitness)

            r1, r2 = np.random.rand(2)

            velocities[i] = np.clip(velocities[i], velocity_clamp[0], velocity_clamp[1])

            swarm[i] = swarm[i] + velocities[i]

            swarm[i] = project_to_feasible(swarm[i], n_points)
