import numpy as np
import time
from scipy import signal
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    
    # Normalize to sum to 22.5
    h = h * (22.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iter = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_freq = 1000
    num_children = 8
    
    # Tree storage
    tree = {}
    root = tuple(h)
    tree[root] = {
        'h': h.copy(),
        'c5': best_c5,
        'visits': 0,
        'children': [],
        'parent': None
    }
    node_list = [root]
    
    # Quick optimization of initial solution
    for _ in range(200):
        if time.time() - start_time > budget_s - 0.1:
            break
        h = best_h.copy()
        # Small random perturbations
        perturb_idx = np.random.choice(len(h), size=max(1, int(mutation_rate * len(h))), replace=False)
        h[perturb_idx] += np.random.normal(0, 0.02, len(perturb_idx))
        h = np.clip(h, 0, 1)
        h = h * (22.5 / np.sum(h))
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection with UCB
        current = root
        while tree[current]['children']:
            children = tree[current]['children']
            ucb_scores = []
            for child in children:
                if tree[child]['visits'] == 0:
                    ucb_scores.append(float('inf'))
                else:
                    ucb = tree[child]['c5'] + exploration_weight * np.sqrt(np.log(tree[current]['visits']) / tree[child]['visits'])
                    ucb_scores.append(ucb)
            best_child_idx = np.argmax(ucb_scores)
            current = children[best_child_idx]
        
        # Expansion
        new_children = []
        current_h = tree[current]['h']
        for _ in range(num_children):
            h_new = current_h.copy()
            # Mutation: change a small portion of elements
            perturb_idx = np.random.choice(len(h_new), size=max(1, int(mutation_rate * len(h_new))), replace=False)
            h_new[perturb_idx] += np.random.normal(0, 0.05, len(perturb_idx))
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (22.5 / np.sum(h_new))
            
            h_tuple = tuple(h_new)
            if h_tuple not in tree:
                new_c5 = evaluate_c5(h_new, dx)
                tree[h_tuple] = {
                    'h': h_new,
                    'c5': new_c5,
                    'visits': 0,
                    'children': [],
                    'parent': current
                }
                tree[current]['children'].append(h_tuple)
                node_list.append(h_tuple)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
            new_children.append(h_tuple)
        
        # Simulation & Backprop
        # Simulate a few steps with greedy improvements
        current_node = current
        for child in new_children:
            tree[current_node]['visits'] += 1
            tree[child]['visits'] += 1
            if tree[child]['c5'] < best_c5:
                best_h = tree[child]['h'].copy()
                best_c5 = tree[child]['c5']
            # Update parents
            parent = tree[child]['parent']
            while parent is not None:
                tree[parent]['visits'] += 1
                parent = tree[parent]['parent']
        
        # Periodic reset
        if iteration % reset_freq == 0 and iteration > 0:
            # Reset a few percent of nodes randomly
            nodes_to_reset = random.sample(node_list, max(1, int(0.1 * len(node_list))))
            for node in nodes_to_reset:
                node_h = tree[node]['h']
                # Quick improvement
                for _ in range(5):
                    perturb_idx = np.random.choice(len(node_h), size=max(1, int(mutation_rate * len(node_h))), replace=False)
                    node_h[perturb_idx] += np.random.normal(0, 0.02, len(perturb_idx))
                    node_h = np.clip(node_h, 0, 1)
                    node_h = node_h * (22.5 / np.sum(node_h))
                new_c5 = evaluate_c5(node_h, dx)
                tree[node] = {
                    'h': node_h,
                    'c5': new_c5,
                    'visits': 0,
                    'children': [],
                    'parent': tree[node]['parent']
                }
                if new_c5 < best_c5:
                    best_h = node_h.copy()
                    best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
