import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2, n_points):
    # Segment-wise crossover
    n_segments = 5
    segment_len = n_points // n_segments
    child = np.zeros(n_points)
    for i in range(n_segments):
        start = i * segment_len
        end = min((i+1) * segment_len, n_points)
        if i % 2 == 0:
            child[start:end] = parent1[start:end]
        else:
            child[start:end] = parent2[start:end]
    return child

def mutate(individual, mutation_rate, n_points):
    # Gaussian noise mutation
    if random.random() < mutation_rate:
        noise = np.random.normal(0, 0.02, n_points)
        individual += noise
        individual = project_to_feasible(individual, n_points)
    return individual

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    
    # Population initialization
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Fallback initial construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Generate additional random individuals
    for _ in range(population_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Main loop
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        fitness = [fitness[i] for i in sorted_indices]
        population = [population[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection: top 50%
        new_population = population[:population_size//2]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(new_population)
            parent2 = random.choice(new_population)
            
            # Crossover
            child = crossover(parent1, parent2, n_points)
            
            # Mutation
            child = mutate(child, 0.2, n_points)
            
            # Add to population
            new_population.append(child)
        
        population = new_population
        generation += 1
        # Occasionally apply relaxation (simplified)
        if generation % 10 == 0:
            # Simple local refinement
            ref = population[0]  # Most fit
            # Make small random changes to keep diversity
            for i in range(len(population)):
                if np.random.random() < 0.1:
                    population[i] = project_to_feasible(ref + np.random.normal(0, 0.01, n_points), n_points)
    
    return (best_individual.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
