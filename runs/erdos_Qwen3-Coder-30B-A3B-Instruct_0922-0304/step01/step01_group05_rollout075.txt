import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    best_C5 = float('inf')
    best_h = None
    
    # Try multiple random restarts to avoid local minima
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize with scaled initial guess
        h_init = initial_h.copy()
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Add small random perturbation
        h_init += np.random.uniform(-0.05, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Define CVXPY variables
        h = Variable(n_points)
        C = Variable()
        
        # Constraints for integral and bounds
        constraints = [sum(h) == n_points / 2]
        constraints += [h >= 0]
        constraints += [h <= 1]
        
        # Add overlap constraints for all possible shifts
        max_shift = n_points - 1
        for k in range(-max_shift, max_shift + 1):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute valid indices for this shift
            if k >= 0:
                i_indices = np.arange(n_points - k)
                k_indices = np.arange(k, n_points)
            else:
                i_indices = np.arange(-k, n_points)
                k_indices = np.arange(n_points + k)
            
            # Build linear constraint for this shift
            expr = 0
            valid_pairs = min(len(i_indices), len(k_indices))
            for i in range(valid_pairs):
                expr += h[i_indices[i]] * (1 - h[k_indices[i]])
            
            constraints.append(expr <= C * dx)
        
        # Objective
        objective = Minimize(C)
        prob = Problem(objective, constraints)
        
        try:
            # Solve with SCS solver
            prob.solve(solver=SCS, verbose=False, max_iters=5000, eps=1e-5)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                current_C5 = C.value
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.value.copy()
                
                # Early termination if we reach target
                if best_C5 <= 0.3807:
                    break
                    
        except Exception:
            pass
    
    # If no good solution found, fallback to simplest approach
    if best_h is None:
        # Use fixed initial guess and run simple optimization
        h = initial_h.copy()
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        
        def compute_C5(h_vec):
            h_vec = np.array(h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(corr) * dx
        
        best_C5 = compute_C5(best_h)
    
    # Final normalization to ensure integral is exactly correct
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final computation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)

def compute_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
