import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    # Optimization parameters
    max_iter = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # First stage: convex optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we hit target
        if best_c5 <= 0.3807:
            break
            
        # Reduce overlap k_max using convex optimization
        h_var = Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp_sum(h_var) == n_points / 2]
        
        # The objective is to minimize the maximum overlap
        # We relax the constraint by adding a penalty to the maximum overlap
        # This is approximated by maximizing (1 - h) * h over all possible shifts
        # Use the existing overlap as a proxy for constraint relaxation
        
        # Set up problem for soft constraint
        try:
            # Use explicit optimization for h update
            # Simpler: perform gradient descent on the overlap
            step_size = 0.005
            
            # Compute gradient using finite differences
            h_new = h.copy()
            for _ in range(50):
                temp_corr = correlate(h_new, 1 - h_new, mode="full")
                max_k = np.argmax(temp_corr)
                
                # Compute gradient for h_new[i] w.r.t. overlap at max_k 
                gradient = np.zeros(n_points)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_new.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    gradient[i] = (corr_plus[max_k] - temp_corr[max_k]) / eps
                
                # Update h_new
                h_new -= step_size * gradient
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            h = h_new
        except:
            # Fallback to slight random perturbation
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
