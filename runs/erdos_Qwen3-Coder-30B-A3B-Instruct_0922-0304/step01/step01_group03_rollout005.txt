import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 45
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(n_points)
        h0[:23] = 1.0
        np.random.shuffle(h0)
    h0 = h0 * (n_points / 2.0 / np.sum(h0))
    population.append(h0)
    
    # Generate diverse initial individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0 / np.sum(h))
        population.append(h)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Parameters
    elite_size = population_size // 10
    mutation_rate = 0.05
    mutation_noise = 0.05
    no_improvement_limit = 50
    no_improvement_count = 0
    
    generation = 0
    max_generations = 2000
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 0.1:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Crossover and Mutation
        new_population = population[:elite_size]  # Keep elites
        
        while len(new_population) < population_size:
            # Selection
            parents = np.random.choice(population_size // 2, 2, replace=False)
            parent1, parent2 = population[parents[0]], population[parents[1]]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutate = np.random.rand(n_points) < mutation_rate
            noise = np.random.normal(0, mutation_noise, n_points)
            child[mutate] += noise[mutate]
            child = np.clip(child, 0, 1)
            
            # Renormalize
            child = child * (n_points / 2.0 / np.sum(child))
            
            new_population.append(child)
        
        population = new_population
        
        # Diversity maintenance
        if no_improvement_count > no_improvement_limit:
            # Introduce randomness
            for i in range(int(0.2 * population_size)):
                idx = np.random.randint(elite_size, population_size)
                population[idx] = np.random.rand(n_points)
                population[idx] = population[idx] * (n_points / 2.0 / np.sum(population[idx]))
            # Reduce mutation noise
            mutation_noise *= 0.9
            
        generation += 1
    
    if best_h is None:
        # Fallback to last evaluated
        best_h = population[0]
        corr = signal.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
