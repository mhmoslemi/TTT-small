import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_iterations = 1000
    max_stagnation = 50
    velocity_clamp = (-0.5, 0.5)
    
    # Initialize swarm
    swarm = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    # Include initial solution
    h_init = project_to_feasible(h_init, n_points)
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    fitness = evaluate_c5(h_init, 2.0 / n_points)
    p_best.append(h_init.copy())
    p_best_fitness.append(fitness)
    
    # Initialize other particles
    for i in range(n_particles - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        fitness = evaluate_c5(h, 2.0 / n_points)
        p_best.append(h.copy())
        p_best_fitness.append(fitness)
    
    # Global best
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_fitness

    start_time = time.time()
    
    stagnation_count = 0
    iteration = 0

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (p_best[i] - swarm[i]) + 
                            social * r2 * (g_best - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], velocity_clamp[0], velocity_clamp[1])
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(swarm[i], 2.0 / n_points)

    # Sanity checks

    assert abs(np.sum(best_h) - n_points / 2) < 1e-10
