import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1
        np.random.shuffle(h)
        
    dx = 2.0 / len(h)
    
    # Normalize to sum to 22.5 (n_points/2)
    h = h * (22.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    no_improvement_limit = 1000
    
    # Initialize simplex
    def create_simplex(start_h, size, n):
        simplex = [start_h.copy()]
        for i in range(n):
            perturbed = start_h + size * np.random.normal(0, 1, n)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (22.5 / np.sum(perturbed))
            simplex.append(perturbed)
        return simplex
    
    simplex = create_simplex(h, simplex_size, n_points)
    simplex_values = [evaluate_c5(s, dx) for s in simplex]
    
    # Main optimization loop
    iter_since_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort simplex by function values
        indices = np.argsort(simplex_values)
        sorted_simplex = [simplex[i] for i in indices]
        sorted_values = [simplex_values[i] for i in indices]
        
        # Check for improvement
        if sorted_values[0] < best_c5:
            best_c5 = sorted_values[0]
            best_h = sorted_simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Stop if no improvement for too long
        if iter_since_improvement > no_improvement_limit:
            break
            
        # Compute centroid of best points (excluding worst)
        centroid = np.mean(sorted_simplex[:-1], axis=0)
        
        # Reflect
        worst = sorted_simplex[-1]
        reflected = centroid + reflection_factor * (centroid - worst)
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (22.5 / np.sum(reflected))
        reflected_value = evaluate_c5(reflected, dx)
        
        # Accept or reject reflect
        if sorted_values[0] <= reflected_value < sorted_values[-2]:
            sorted_simplex[-1] = reflected
            sorted_values[-1] = reflected_value
        elif reflected_value < sorted_values[0]:
            # Expand
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (22.5 / np.sum(expanded))
            expanded_value = evaluate_c5(expanded, dx)
            if expanded_value < reflected_value:
                sorted_simplex[-1] = expanded
                sorted_values[-1] = expanded_value
            else:
                sorted_simplex[-1] = reflected
                sorted_values[-1] = reflected_value
        else:
            # Contract
            if reflected_value < sorted_values[-1]:
                contracted = centroid + contraction_factor * (reflected - centroid)
            else:
                contracted = centroid + contraction_factor * (worst - centroid)
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (22.5 / np.sum(contracted))
            contracted_value = evaluate_c5(contracted, dx)
            if contracted_value < sorted_values[-1]:
                sorted_simplex[-1] = contracted
                sorted_values[-1] = contracted_value
            else:
                # Shrink
                for i in range(1, len(sorted_simplex)):
                    sorted_simplex[i] = sorted_simplex[0] + 0.5 * (sorted_simplex[i] - sorted_simplex[0])
                    sorted_simplex[i] = np.clip(sorted_simplex[i], 0, 1)
                    sorted_simplex[i] = sorted_simplex[i] * (22.5 / np.sum(sorted_simplex[i]))
                    sorted_values[i] = evaluate_c5(sorted_simplex[i], dx)
        
        simplex = sorted_simplex
        simplex_values = sorted_values
        
        # Periodic perturbation
        if iteration % 1000 == 0 and iteration > 0:
            # Perturb best solution
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            best_h_perturbed = best_h.copy()
            best_h_perturbed[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            best_h_perturbed = np.clip(best_h_perturbed, 0, 1)
            best_h_perturbed = best_h_perturbed * (22.5 / np.sum(best_h_perturbed))
            
            # Replace worst vertex with perturbed best solution
            max_idx = np.argmax(simplex_values)
            simplex[max_idx] = best_h_perturbed
            simplex_values[max_idx] = evaluate_c5(best_h_perturbed, dx)
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
