import numpy as np
import time
from scipy import signal
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iter = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_freq = 1000
    
    # MCTS tree structure
    class Node:
        def __init__(self, h_config, parent=None, action=None):
            self.h = h_config
            self.parent = parent
            self.action = action  # mutation that led here
            self.children = []
            self.visits = 0
            self.value = float('inf')
            self.c5 = evaluate_c5(h_config, dx)
            self.is_terminal = False
            
        def ucb_value(self, parent_visits, exploration_weight=1.0):
            if self.visits == 0:
                return float('inf')
            return self.c5 + exploration_weight * np.sqrt(np.log(parent_visits) / self.visits)
    
    # Create root node
    root = Node(h.copy())
    
    # Perform MCTS
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection phase
        node = root
        path = [node]
        while node.children and not node.is_terminal:
            # Select child using UCB
            best_child = max(node.children, key=lambda child: child.ucb_value(node.visits, exploration_weight))
            node = best_child
            path.append(node)
            
        # Expansion phase
        if not node.is_terminal and len(node.children) == 0:
            # Generate mutations
            children = []
            for _ in range(5):
                h_new = node.h.copy()
                # Apply mutation to 5% of indices
                n_mutation = max(1, int(mutation_rate * len(h_new)))
                indices = np.random.choice(len(h_new), n_mutation, replace=False)
                for i in indices:
                    h_new[i] += np.random.normal(0, 0.1)
                # Clamp to [0,1]
                h_new = np.clip(h_new, 0, 1)
                # Renormalize
                h_new = h_new * (22.5 / np.sum(h_new))
                child = Node(h_new, parent=node, action=(indices, h_new[indices]))
                children.append(child)
            
            node.children = children
            if children:
                # Select best child to continue simulation
                node = min(children, key=lambda child: child.c5)
                path.append(node)
            else:
                node.is_terminal = True
                
        # Simulation/evaluation phase
        if not node.is_terminal:
            # Use the current node's evaluation for backprop
            current_c5 = node.c5
        else:
            # Rerun evaluation if terminal (should not happen if children just created)
            current_c5 = evaluate_c5(node.h, dx)
        
        # Backpropagation
        for n in path:
            n.visits += 1
            if n.c5 > current_c5:
                n.c5 = current_c5
                
        # Update global best
        if root.c5 < best_c5:
            best_c5 = root.c5
            best_h = root.h.copy()
        
        # Periodic reset
        if iteration % reset_freq == 0 and iteration > 0:
            # Reset some nodes to random
            def reset_random(node):
                if random.random() < 0.3 and len(node.children) > 0:
                    for child in node.children:
                        reset_random(child)
                elif not node.is_terminal:
                    # Random reset
                    temp_h = np.zeros_like(node.h)
                    temp_h[:22] = 1.0
                    np.random.shuffle(temp_h)
                    temp_h = temp_h * (22.5 / np.sum(temp_h))
                    node.h = temp_h
                    node.c5 = evaluate_c5(temp_h, dx)  
            reset_random(root)
    
    # Return final best
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
