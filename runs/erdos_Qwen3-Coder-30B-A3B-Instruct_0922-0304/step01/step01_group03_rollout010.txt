import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Nelder-Mead parameters
    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    perturb_freq = 1000
    max_no_improvement = 1000
    improvement_count = 0

    # Create initial simplex
    simplex = [h.copy()]
    for i in range(n_points):
        h_perturbed = h + simplex_size * np.random.normal(0, 1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (22.5 / np.sum(h_perturbed))
        simplex.append(h_perturbed)
    
    def project_point(point):
        point = np.clip(point, 0, 1)
        point = point * (22.5 / np.sum(point))
        return point

    def evaluate_simplex(simplex):
        values = []
        for point in simplex:
            c5 = evaluate_c5(point, dx)
            values.append(c5)
        return np.array(values)

    # Main loop
    iter_count = 0
    while time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        
        # Evaluate all points in simplex
        values = evaluate_simplex(simplex)
        sorted_indices = np.argsort(values)
        
        # Best, worst, second worst
        best_idx = sorted_indices[0]
        worst_idx = sorted_indices[-1]
        second_worst_idx = sorted_indices[-2]
        
        # Update best solution
        if values[best_idx] < best_c5:
            best_c5 = values[best_idx]
            best_h = simplex[best_idx].copy()
            improvement_count = 0
        else:
            improvement_count += 1
            if improvement_count > max_no_improvement:
                break
                
        # Compute centroid (excluding worst)
        centroid = np.mean([simplex[i] for i in sorted_indices[:-1]], axis=0)
        
        # Reflect
        reflected = centroid + reflection_factor * (centroid - simplex[worst_idx])
        reflected = project_point(reflected)
        reflected_c5 = evaluate_c5(reflected, dx)
        
        # Check if reflected point is better than worst but not best
        if values[best_idx] <= reflected_c5 < values[second_worst_idx]:
            simplex[worst_idx] = reflected
            continue
        
        # Expand if better than best
        if reflected_c5 < values[best_idx]:
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = project_point(expanded)
            expanded_c5 = evaluate_c5(expanded, dx)
            if expanded_c5 < reflected_c5:
                simplex[worst_idx] = expanded
            else:
                simplex[worst_idx] = reflected
            continue
        
        # Contract
        if reflected_c5 >= values[second_worst_idx]: 
            contracted = centroid + contraction_factor * (simplex[worst_idx] - centroid)
            contracted = project_point(contracted)
            contracted_c5 = evaluate_c5(contracted, dx)
            if contracted_c5 < values[worst_idx]:
                simplex[worst_idx] = contracted
            else:
                # Shrink simplex toward best point
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[best_idx] + 0.5 * (simplex[i] - simplex[best_idx])
                    simplex[i] = project_point(simplex[i])
                continue
        
        # Accept reflection
        simplex[worst_idx] = reflected
        
        # Perturbation
        if iter_count % perturb_freq == 0:
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            best_h_noisy = best_h.copy()
            best_h_noisy[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            best_h_noisy = project_point(best_h_noisy)
            # Replace worst vertex with the perturbed best solution
            worst_idx = np.argmax([evaluate_c5(p, dx) for p in simplex])
            simplex[worst_idx] = best_h_noisy
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
