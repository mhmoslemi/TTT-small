import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)

    # Perform differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
    )
    
    if de_result.success:
        current_h = de_result.x
    else:
        current_h = h_init.copy()

    # Local refinement with L-BFGS-B
    def constraint_sum(h):
        return np.sum(h) - target_sum

    # Define bounds for L-BFGS
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    # Optimize with L-BFGS-B using the DE result as starting point
    try:
        result_local = minimize(
            objective,
            current_h,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            callback=lambda x: None,
        )
        
        if result_local.success:
            current_h = result_local.x
    except:
        pass

    # Final processing
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * target_sum / np.sum(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Validate final solution
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    return (current_h, current_c5, n_points)
