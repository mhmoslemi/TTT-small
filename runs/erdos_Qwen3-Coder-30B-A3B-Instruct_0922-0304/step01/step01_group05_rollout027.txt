import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
    except NameError:
        # Fallback initialization
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial guess
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convex optimization approach
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Build constraints for all possible shifts
    # For each lag k from -(n_points-1) to (n_points-1)
    for k in range(-(n_points-1), n_points):
        # Compute the indices for the correlation
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i = 0 to n_points-1-k
            indices = list(range(n_points - k))
            left = h[indices]
            right = 1 - h[indices + k]
        else:
            # h[i+k] * (1 - h[i]) for i = -k to n_points-1
            indices = list(range(-k, n_points))
            left = h[indices + k]
            right = 1 - h[indices]
        
        # Constraint: sum(left * right) <= C * dx
        # But make sure we have valid indices
        if len(indices) > 0:  
            # Avoid unused variable usage in constraint list
            factor = sum(left * right)
            constraints.append(factor <= C * dx)
    
    # Objective: minimize C
    objective = Minimize(C)
    
    # Solve problem
    prob = Problem(objective, constraints)
    
    # Try different solvers
    solver_list = [SCS, ECOS, OSQP]
    best_solution = None
    best_c = float('inf')
    
    for solver in solver_list:
        if time.time() - start_time > budget_s * 0.9:
            break
        
        try:
            prob.solve(solver=solver, verbose=False, max_iters=5000, 
                      eps=1e-6, warm_start=True)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                current_c = C.value
                if current_c is not None and current_c < best_c:
                    best_c = current_c
                    best_solution = h.value.copy()
                    
        except Exception:
            continue
    
    # If no solution found, use the initial guess with refinement
    if best_solution is None:
        best_solution = h_init.copy()
        best_c = np.max(np.correlate(best_solution, 1-best_solution, mode="full")) * dx
    
    # Refinement with scipy if within time
    if time.time() - start_time < budget_s * 0.8:
        from scipy.optimize import minimize
        
        def compute_C5(h_vals):
            h = np.array(h_vals)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        def objective(h_vals):
            return compute_C5(h_vals)
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        res = minimize(objective, best_solution, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-6},
                       callback=lambda x: None)
        
        if res.success:
            best_solution = res.x.copy()
            best_c = compute_C5(best_solution)
    
    # Final normalization
    best_solution = best_solution * (n_points / 2) / np.sum(best_solution)
    
    return (best_solution.tolist(), float(best_c), n_points)
