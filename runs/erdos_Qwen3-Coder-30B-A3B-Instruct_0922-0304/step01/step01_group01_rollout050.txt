import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to have correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Average two parents to create offspring"""
    return (parent1 + parent2) / 2.0

def mutate(individual, mutation_rate=0.1, std=0.1):
    """Add Gaussian noise to individual with given mutation rate"""
    mask = np.random.random(len(individual)) < mutation_rate
    individual = individual.copy()
    individual[mask] += np.random.normal(0, std, np.sum(mask))
    return individual

    stagnation_generations = 20
    max_generations = 1000

    population = []

    population.append(h_init.copy())

    best_c5 = fitnesses[best_idx]
    
    start_time = time.time()
    generations_no_improve = 0
    current_generation = 0
    
    while time.time() - start_time < budget_s * 0.95 and current_generation < max_generations:
        # Evaluate fitness for entire population 
        fitnesses = [evaluate_c5(h, 2.0 / n_points) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            generations_no_improve = 0
        else:
            generations_no_improve += 1
            
        if generations_no_improve > stagnation_generations:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(population[min_fitness_idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutate(offspring, mutation_rate)
            
            # Project to feasible region
            offspring = project_to_feasible(offspring, n_points)
            
            new_population.append(offspring)
            
        population = new_population
        current_generation += 1
    
    return (best_h, best_c5, n_points)
