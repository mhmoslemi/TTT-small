import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: create a structured binary-like pattern with some noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    current_c5 = evaluate_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Simulated annealing parameters
    temp = 0.05
    cooling_rate = 0.9998
    max_iter = int(budget_s * 1000)  # Rough estimate of iterations
    
    # Perturbation types
    SWAP_TYPE = 0
    NOISE_TYPE = 1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Randomly choose perturbation type
        pert_type = np.random.choice([SWAP_TYPE, NOISE_TYPE], p=[0.7, 0.3])
        
        h_new = h.copy()
        if pert_type == SWAP_TYPE:
            # Flip a single bit
            idx = np.random.randint(n_points)
            h_new[idx] = 1 - h_new[idx]
        else:
            # Add noise to two random points and adjust to maintain sum
            idx1, idx2 = np.random.choice(n_points, size=2, replace=False)
            noise1 = np.random.normal(0, 0.01)
            noise2 = np.random.normal(0, 0.01)
            
            # Adjust to preserve sum constraint
            scale_factor = (n_points / 2 - np.sum(h_new)) / (np.sum(h_new) - n_points / 2)
            h_new[idx1] += noise1
            h_new[idx2] += noise2 * scale_factor
            
        # Ensure values are within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temp):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cool down temperature
        temp *= cooling_rate
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
