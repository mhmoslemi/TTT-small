import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Prepare initial guess and normalize it
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    def compute_C5(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Create a coarse version for initial exploration
    coarse_n = 20
    coarse_dx = 2.0 / coarse_n
    coarse_h_init = np.interp(np.linspace(0, 2, coarse_n), 
                              np.linspace(0, 2, n_points), 
                              h_init)
    coarse_h_init = coarse_h_init * (coarse_n / 2) / np.sum(coarse_h_init)
    
    # Optimize on coarse grid
    def coarse_objective(h):
        h = h * (coarse_n / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * coarse_dx
    
    bounds = [(0, 1) for _ in range(coarse_n)]
    
    # Refine using the golden section search-like start
    best_coarse_h = coarse_h_init.copy()
    best_C5_coarse = coarse_objective(best_coarse_h)
    
    # Perform several random restarts to avoid local optima
    n_restarts = 10
    for i in range(n_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
        h_trial = np.random.rand(coarse_n) * 0.5
        h_trial = h_trial * (coarse_n / 2) / np.sum(h_trial)
        
        def coarse_obj(h):
            return coarse_objective(h)
        
        res = minimize(coarse_obj, h_trial, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 5000})
        if res.success and res.fun < best_C5_coarse:
            best_C5_coarse = res.fun
            best_coarse_h = res.x.copy()
    
    # Interpolate to fine grid
    fine_h = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, coarse_n), 
                       best_coarse_h)
    fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
    
    # Fine-tune with local optimization
    bounds_fine = [(0, 1) for _ in range(n_points)]
    
    def fine_objective(h):
        return compute_C5(h) 
    
    # Try gradient-based local optimization
    res = minimize(fine_objective, fine_h, method='L-BFGS-B', bounds=bounds_fine,
                   options={'maxiter': 2000, 'ftol': 1e-8})
    
    final_h = res.x if res.success else fine_h
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_C5 = compute_C5(final_h)
    
    # Final iterations for refinement with random restarts
    best_h = final_h.copy()
    best_C5 = final_C5
    
    for i in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_trial = np.random.rand(n_points) * 0.5
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        res = minimize(fine_objective, h_trial, method='L-BFGS-B', bounds=bounds_fine,
                       options={'maxiter': 1000, 'ftol': 1e-8})
        if res.success and res.fun < best_C5:
            best_C5 = res.fun
            best_h = res.x.copy()
    
    return (best_h.tolist(), best_C5, n_points)
