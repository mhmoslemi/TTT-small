import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a structured starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    temp = 0.05
    cooling_rate = 0.9998
    max_iter = int(1e5)
    
    best_h = h.copy()
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Choose perturbation type
        if np.random.random() < 0.3:
            # Gaussian noise perturbation on two points
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = np.random.normal(0, 0.01, 2)
            h_new = h.copy()
            h_new[i1] += delta[0]
            h_new[i2] += delta[1]
            
            # Maintain sum constraint
            diff = (h_new[i1] + h_new[i2]) - (h[i1] + h[i2])
            if abs(diff) > 1e-10:
                # Adjust another point to maintain sum
                i3 = np.random.choice([i for i in range(n_points) if i not in [i1, i2]])
                h_new[i3] -= diff
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        else:
            # Binary swap: set one element to 1, another to 0
            h_new = h.copy()
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            # Ensure we're swapping values that are within bounds for this operation
            if (h_new[i1] > 0.5 and h_new[i2] < 0.5) or (h_new[i1] < 0.5 and h_new[i2] > 0.5):
                h_new[i1] = 1 - h_new[i1]
                h_new[i2] = 1 - h_new[i2]
            elif h_new[i1] > 0.5 and h_new[i2] > 0.5:
                h_new[i1] = 0
                h_new[i2] = 1
            elif h_new[i1] < 0.5 and h_new[i2] < 0.5:
                h_new[i1] = 1
                h_new[i2] = 0
            else:
                # Do nothing if they are already at their limits
                pass
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / temp):
            h = h_new
            current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Update temperature
        temp *= cooling_rate
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
