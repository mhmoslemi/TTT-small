import numpy as np
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.linspace(0.9, 0.1, n_points)
        h_init = h_init / h_init.sum() * 21.0
        
    dx = 2.0 / n_points
    target_sum = 21.0  # n_points / 2
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.9995
    T_min = 1e-5
    max_iter = 5000
    
    T = T0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Generate neighbor solution
        h_trial = current_h.copy()
        
        # Modify ~5% of points
        mask = np.random.rand(n_points) < 0.05
        delta = np.random.uniform(-0.1, 0.1, size=(n_points)) * mask
        h_trial += delta
        
        # Project to [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize sum to target_sum
        if abs(h_trial.sum() - target_sum) > 1e-6:
            scale = target_sum / h_trial.sum()
            h_trial *= scale
            
        # Evaluate trial solution
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject
        delta_c5 = c5_trial - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            current_h = h_trial
            current_c5 = c5_trial
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Decrease temperature
        T *= alpha
        iteration += 1
        
    # Final validation
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    return (best_h, best_c5, n_points)
