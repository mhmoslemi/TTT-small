import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(signal.correlate(h, 1 - h, mode='full')) * dx
    
    # Optimization parameters
    max_iter = 5000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Use convex optimization to solve
        # Variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Add overlap constraints for a subset of lags
        # Only consider lags that are likely to be critical based on initial heuristic
        max_lags = min(100, 2 * n_points - 1)
        lags_to_consider = np.linspace(0, 2 * n_points - 2, max_lags, dtype=int)
        
        for k in lags_to_consider:
            # Compute the overlap for lag k
            # This uses the full convolution result's k-th element 
            # when we slice appropriately
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(C >= overlap_expr * dx)
        
        # Objective
        objective = cp.Minimize(C)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                new_c5 = C.value
            else:
                # Fallback to a gradient step if optimization fails
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                scale = (n_points / 2) / np.sum(h_new)
                h_new = h_new * scale
                new_c5 = np.max(signal.correlate(h_new, 1 - h_new, mode='full')) * dx
        except:
            # Fallback to gradient-based method
            # Simple finite difference gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            h_plus = h.copy()
            h_minus = h.copy()
            
            for i in range(n_points):
                h_plus[i] += eps
                h_minus[i] -= eps
                
                o_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full')) * dx
                o_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full')) * dx
                
                grad[i] = (o_plus - o_minus) / (2 * eps)
                h_plus[i] -= eps
                h_minus[i] += eps
                
            h_new = h - 0.1 * grad
            h_new = np.clip(h_new, 0, 1)
            scale = (n_points / 2) / np.sum(h_new)
            h_new = h_new * scale
            new_c5 = np.max(signal.correlate(h_new, 1 - h_new, mode='full')) * dx

        # Accept or reject update
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            iter_since_improvement += 1
            if iter_since_improvement > max_no_improvement:
                # Restart with random perturbation
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                iter_since_improvement = 0
                
    return (best_h.tolist(), best_c5, n_points)
