    personal_best = np.zeros((num_particles, n_points))
    personal_best_scores = np.full(num_particles, np.inf)
    global_best = None
    global_best_score = np.inf
    
    # Initialize particles
    for i in range(num_particles):

        # Rescale to maintain sum constraint
        current_sum = np.sum(particles[i])
        particles[i] *= (n_points / 2) / current_sum
        
        # Store initial personal best
        personal_best[i] = particles[i].copy()
        personal_best_scores[i] = compute_C5(particles[i], dx)
        
        # Update global best
        if personal_best_scores[i] < global_best_score:
            global_best = personal_best[i].copy()
            global_best_score = personal_best_scores[i]
    
    # PSO main loop
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter:
        if time.time() - start_time > budget_s * 0.98:
            break
            
        current_time = time.time() - start_time
        inertia = inertia_start - (inertia_start - inertia_end) * (current_time / budget_s)

            r1, r2 = np.random.rand(2)

            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)

            particles[i] += velocities[i]

            particles[i] *= (n_points / 2) / current_sum

    final_C5 = compute_C5(best_h, dx)

def compute_C5(h, dx):
    """Compute the maximum overlap for a given h"""
    # Ensure h is properly normalized
    h = np.array(h)
    # Compute correlations for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Return max correlation (scaled by dx)
    return np.max(corr) * dx
