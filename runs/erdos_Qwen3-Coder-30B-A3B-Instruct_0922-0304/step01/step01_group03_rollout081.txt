import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple binary-like starting vector with 23 ones and 22 zeros
        n_points = 45
        initial_h = np.zeros(n_points)
        initial_h[:23] = 1.0
        np.random.shuffle(initial_h)
        initial_h = initial_h * (22.5 / np.sum(initial_h))
    
    # GA parameters
    pop_size = 50
    num_generations = 2000
    num_parents = pop_size // 2
    num_elite = pop_size // 10
    mutation_rate = 0.05
    crossover_point = pop_size // 2
    noise_std = 0.05
    noise_reduction_factor = 0.9
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.zeros_like(initial_h)
        # Set a few elements to 1.0
        num_ones = np.random.randint(10, 35)
        ones_indices = np.random.choice(len(h), num_ones, replace=False)
        h[ones_indices] = 1.0
        # Normalize and randomize
        h = h * (22.5 / np.sum(h))
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(initial_h)
    
    gen_since_improvement = 0
    best_c5_history = []
    
    # Main GA loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append((c5, h))
        
        # Sort by fitness (lower is better)
        fitness.sort(key=lambda x: x[0])
        best_c5_history.append(fitness[0][0])
        
        # Check for improvement
        if fitness[0][0] < best_c5:
            best_h = fitness[0][1].copy()
            best_c5 = fitness[0][0]
            gen_since_improvement = 0
        else:
            gen_since_improvement += 1
        
        # Elitism: keep top individuals
        elites = [item[1] for item in fitness[:num_elite]]

        # Selection and reproduction
        parents = [item[1] for item in fitness[:num_parents]]
        new_population = elites[:]
        
        # Generate offspring using crossover & mutation
        while len(new_population) < pop_size:
            parent1 = parents[np.random.randint(0, len(parents))]
            parent2 = parents[np.random.randint(0, len(parents))]
            
            # Single-point crossover
            point = np.random.randint(1, len(parent1) - 1)
            child = np.concatenate([parent1[:point], parent2[point:]])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, noise_std, len(child))
                child = child + noise
                child = np.clip(child, 0, 1)
                # Renormalize
                child = child * (22.5 / np.sum(child))
            
            new_population.append(child)
            
        # Replace old population
        population = new_population[:pop_size]
        
        # Reset some if no improvement for a while
        if gen_since_improvement > 50:
            for i in range(int(pop_size * 0.2)):
                h = np.zeros_like(initial_h)
                num_ones = np.random.randint(10, 35)
                ones_indices = np.random.choice(len(h), num_ones, replace=False)
                h[ones_indices] = 1.0
                h = h * (22.5 / np.sum(h))
                h = np.clip(h, 0, 1)
                population[np.random.randint(0, len(population))] = h.copy()
            noise_std *= noise_reduction_factor
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h):
    """Evaluate C5 given h."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * (2.0 / len(h))
