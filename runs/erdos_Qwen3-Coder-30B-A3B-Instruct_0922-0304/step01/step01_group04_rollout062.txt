import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Convex optimization approach
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints for h
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Compute all possible shifts for the correlation
    max_shift = n_points - 1
    shift_range = range(-max_shift, max_shift + 1)
    
    # Add constraints of the form: sum(h[i] * (1 - h[i+k])) <= t for all valid k
    # This translates to: sum(h[i] + h[i+k] - h[i]*h[i+k]) <= t
    # But since we want sum(h[i] * (1 - h[i+k])) <= t: sum(h[i] - h[i]*h[i+k]) <= t
    # Which is: sum(h[i]) - sum(h[i]*h[i+k]) <= t
    for k in shift_range:
        if k == 0:  # Skip zero shift to ensure correlation with shifted version like in full mode
            continue
        # We'll compute all terms h[i] * (1 - h[i+k]) for i, i+k valid indices
        sum_expr = 0
        for i in range(n_points):
            if 0 <= i + k < n_points:
                sum_expr += h[i] * (1 - h[i + k])
        constraints.append(sum_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        # Use MOSEK solver for better performance
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-8, reltol=1e-8, abstol=1e-8)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = t.value
        else:
            h_opt = h_init.copy()
            c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    except Exception as e:
        # Fallback to a modified approach
        h_opt = h_init.copy()
        c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx

    # Ensure h_opt is within bounds
    h_opt = np.clip(h_opt, 0, 1)
    
    # Renormalize h_opt to sum to target_sum
    if np.sum(h_opt) > 0:
        h_opt = h_opt * target_sum / np.sum(h_opt)
    
    # Final check
    final_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
    
    # Return the result
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (h_opt, final_c5, n_points)
