import numpy as np
from scipy.signal import correlate
import time
from math import floor

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum = n_points/2, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_crossover(parent1, parent2, n_points):
    """Segment-wise crossover"""
    # Randomly choose crossover point
    crossover_point = np.random.randint(1, n_points)
    offspring = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return offspring

def genetic_mutation(individual, mutation_rate=0.2, noise_std=0.02):
    """Apply Gaussian noise to individuals with given probability"""
    if np.random.random() < mutation_rate:
        individual = individual + np.random.normal(0, noise_std, len(individual))
    return individual

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    generations = 1200  # Budget allows ~1200 generations
    mutation_rate = 0.2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
        population.append(h)
    else:
        # Default starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add random individuals
    for _ in range(pop_size - 1):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_c5(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection: top half
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Crossover and mutation
        new_population = parents[:]
        for _ in range(pop_size - len(parents)):
            # Select two parents
            p1, p2 = np.random.choice(parents, 2, replace=False)
            offspring = genetic_crossover(p1, p2, n_points)
            offspring = genetic_mutation(offspring, mutation_rate)
            offspring = project_to_feasible(offspring, n_points)
            new_population.append(offspring)
        
        # Evaluate new generation
        population = new_population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Every 10 generations, relax the best individual using gradient descent
        if gen % 10 == 0 and best_h is not None:
            # Simple gradient step for refinement
            step_size = 0.005
            h_refined = best_h.copy()
            for _ in range(20):
                corr = correlate(h_refined, 1 - h_refined, mode="full")
                c5_val = np.max(corr) * dx
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_val) / eps
                h_refined -= step_size * grad
                h_refined = project_to_feasible(h_refined, n_points)
            best_h = h_refined
            best_c5 = evaluate_c5(best_h, dx)
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed=seed, budget_s=budget_s)
