import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper to ensure constraints are satisfied
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Objective function to minimize
    def objective(h):
        h = repair_h(h)
        return evaluate_c5(h, dx)
    
    # DE optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE with custom bounds and repair
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=100,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None
    )
    
    # Use DE result to initialize L-BFGS-B
    if de_result.success:
        best_h = repair_h(de_result.x)
    else:
        best_h = repair_h(h_init)
    
    # L-BFGS-B refinement
    def lbfgsb_objective(h):
        h = repair_h(h)
        return evaluate_c5(h, dx)
    
    # Constraints for L-BFGS-B
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    try:
        lbfgsb_result = minimize(
            lbfgsb_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=lambda x: None
        )
        
        if lbfgsb_result.success:
            best_h = repair_h(lbfgsb_result.x)
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure constraints
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (best_h, final_c5, n_points)
