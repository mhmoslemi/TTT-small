import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    min_T = 1e-8
    max_iter = 100000
    
    # Start with initial temperature
    T = T0
    
    # Main optimization loop
    iter_count = 0
    while T > min_T and iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Create a candidate solution by perturbing the current solution
        h_new = h.copy()
        
        # Choose random subset to modify
        num_modify = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, size=num_modify, replace=False)
        
        # Modify the selected points
        for i in indices:
            # Random change with bounds
            delta = np.random.uniform(-0.2, 0.2)
            h_new[i] += delta
            
        # Ensure h_new stays within [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain the integral constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust to maintain sum = target_sum
            adjustment = (target_sum - current_sum) / len(indices)
            for i in indices:
                if current_sum < target_sum:
                    h_new[i] = min(h_new[i] + abs(adjustment), 1.0)
                else:
                    h_new[i] = max(h_new[i] - abs(adjustment), 0.0)
            
            # Final normalization to ensure the exact sum
            h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the new solution
        if C5_new < best_C5:
            h = h_new.copy()
            best_C5 = C5_new
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta = C5_new - best_C5
            if delta > 0 and np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
        # Cool down
        T *= alpha
        iter_count += 1
        
    return (best_h, best_C5, n_points)
