import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, n_points):
    """Project h onto feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_algorithm(seed, budget_s, n_points):
    np.random.seed(seed)
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 3
    stagnation_generations = 20
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = initial_h_values.copy()
        h_init = project_feasible(h_init, n_points)
        population.append(h_init)
    
    # Fill population with random individuals
    while len(population) < pop_size:
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Initialize best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, 2.0 / n_points)
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Update best solution
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= stagnation_generations:
            break
            
        # Tournament selection
        new_population = []
        for _ in range(pop_size):
            # Tournament
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent = population[winner_idx]
            
            # Crossover (simple average)
            if np.random.rand() < 0.5 and len(population) > 1:
                other_idx = np.random.choice([i for i in range(pop_size) if i != winner_idx])
                offspring = (parent + population[other_idx]) / 2
            else:
                offspring = parent.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                offspring += noise
                offspring = project_feasible(offspring, n_points)
            
            new_population.append(offspring)
        
        population = new_population
        generation += 1
    
    return best_h, best_c5
