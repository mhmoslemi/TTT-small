import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h

def generate_individual(n_points):
    h = np.random.rand(n_points)
    h = project_to_feasible(h, n_points)
    return h

def crossover(parent1, parent2):
    n_points = len(parent1)
    split_point = random.randint(1, n_points - 1)
    child = np.concatenate([parent1[:split_point], parent2[split_point:]])
    return child

def mutate(individual, mutation_rate=0.2, noise_std=0.02):
    n_points = len(individual)
    mask = np.random.rand(n_points) < mutation_rate
    individual = individual.copy()
    individual[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return project_to_feasible(individual, n_points)

def initial_individual(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    return h

def genetic_algorithm(n_points, budget_s):
    population_size = 50
    generations = 0
    start_time = time.time()
    
    # Initialize population
    population = []
    population.append(initial_individual(n_points))
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points))
    
    # Evaluate initial population
    fitness = [evaluate_c5(ind, 2.0 / n_points) for ind in population]
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Main loop
    while time.time() - start_time < budget_s - 1:
        generations += 1
        
        # Select parents (tournament selection)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            if random.random() < 0.8:  # Crossover rate
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            child = mutate(child)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_c5(ind, 2.0 / n_points) for ind in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_fitness:
            best_fitness = fitness[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Occasionally run relaxation step
        if generations % 10 == 0:
            # Try local optimization on best solution
            h_original = best_individual.copy()
            h_opt = h_original.copy()
            dx = 2.0 / n_points
            
            # Simple gradient descent for local refinement
            learning_rate = 0.01
            for _ in range(20):
                corr = correlate(h_opt, 1 - h_opt, mode="full")
                max_k = np.argmax(corr)
                max_val = corr[max_k]
                
                # Estimate gradient using finite differences
                grad = np.zeros(n_points)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_opt.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    grad[i] = (corr_plus[max_k] - max_val) / eps
                
                h_opt -= learning_rate * grad
                h_opt = project_to_feasible(h_opt, n_points)
            
            # Check if we improved
            c5_new = evaluate_c5(h_opt, dx)
            if c5_new < best_fitness:
                best_fitness = c5_new
                best_individual = h_opt.copy()
        
        # Early stopping if good enough
        if best_fitness <= 0.3807:
            break
    
    return best_individual, best_fitness, n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 70
    h_values, c5_bound, n_points = genetic_algorithm(n_points, budget_s)
    return (h_values.tolist(), c5_bound, n_points)
