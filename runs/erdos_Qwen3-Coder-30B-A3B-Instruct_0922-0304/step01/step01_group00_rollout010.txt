import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Ensure sum = n_points / 2 and h[i] in [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def segment_crossover(parent1, parent2, n_points):
    # Segment-wise crossover with random cut points
    cut1 = np.random.randint(1, n_points-1)
    cut2 = np.random.randint(cut1+1, n_points)
    
    offspring = np.copy(parent1)
    offspring[cut1:cut2] = parent2[cut1:cut2]
    return offspring

def mutate(individual, sigma=0.02, n_points=70):
    # Add Gaussian noise and project
    mutated = individual + np.random.normal(0, sigma, n_points)
    return project_to_feasible(mutated, n_points)

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        h = project_to_feasible(h, n_points)
    else:
        # Create a structured initial solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, n_points)
    
    population.append(h)
    
    # Fill with diverse random individuals
    for _ in range(population_size - 1):
        # Start with binary structure
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(ind, dx) for ind in population]
    
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    generation = 0
    relaxation_interval = 10
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        parents = [population[i] for i in top_indices]
        parent_fitness = [fitness[i] for i in top_indices]
        
        # Create new population
        new_population = deepcopy(parents)
        
        # Crossover
        crossover_rate = 0.8
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                offspring = segment_crossover(p1, p2, n_points)
                # Mutation
                if np.random.rand() < 0.2:
                    offspring = mutate(offspring, 0.02, n_points)
                else:
                    offspring = project_to_feasible(offspring, n_points)
                new_population.append(offspring)
            else:
                # Clone parent
                parent = np.random.choice(parents)
                offspring = mutate(parent, 0.02, n_points)
                new_population.append(offspring)
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Optional relaxation step
        if generation % relaxation_interval == 0 and len(parents) > 0:
            # Solve a convex relaxation for the best individual
            # This uses a simplified regularized approach with gradient descent
            best_parent = parents[0]
            best_relaxed = best_parent.copy()
            lr = 0.001  # learning rate
            max_iter = 50
            
            for _ in range(max_iter):
                # Compute current overlap
                corr = correlate(best_relaxed, 1 - best_relaxed, mode="full")
                current_c5 = np.max(corr) * dx
                
                # Approximate gradient using finite differences
                grad = np.zeros_like(best_relaxed)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = best_relaxed.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - current_c5) / eps
                
                # Update
                best_relaxed -= lr * grad
                best_relaxed = project_to_feasible(best_relaxed, n_points)
            
            # Replace worst individual with relaxed solution
            worst_idx = np.argmax([evaluate_c5(ind, dx) for ind in new_population])
            new_population[worst_idx] = best_relaxed
        
        # Evaluate new generation
        fitness = [evaluate_c5(ind, dx) for ind in new_population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = new_population[current_best_idx].copy()
        
        # Update population
        population = new_population
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
