import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        # Use a sawtooth pattern for better initial guess
        h_init = np.abs(np.sin(np.arange(n_points) * np.pi / n_points)) * 0.5 + 0.25
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h to satisfy sum constraint
    current_sum = np.sum(h_init)
    h_current = h_init * target_sum / current_sum
    
    # Final evaluation
    current_c5 = evaluate_c5(h_current, dx)
    
    # Optimization routine
    max_iter = 1000
    epsilon = 1e-6
    best_c5 = current_c5
    best_h = h_current.copy()
    
    # More aggressive optimization using L-BFGS-B with bounds
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts to avoid local minima
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with a slightly randomized version
        h_start = h_current + np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * target_sum / np.sum(h_start)
        
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': epsilon, 'gtol': epsilon}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = np.clip(result.x, 0, 1)
                best_h = best_h * target_sum / np.sum(best_h)
                
        except Exception:
            pass
    
    # Final projection step to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    # Validate final result
    assert np.isclose(np.sum(best_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in best_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass
    
    return (best_h, final_c5, n_points)
