import numpy as np
import time
import random
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1
        np.random.shuffle(h)
    
    dx = 2.0 / len(h)
    
    # Normalize to sum to 22.5 (since n_points=45, dx=2/45, so sum*h*dx = 1 means sum=22.5)
    h = h * (22.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # MCTS parameters
    max_iterations = 10000
    exploration_weight = 1.0
    mutation_rate = 0.05
    reset_interval = 1000
    
    # Tree structure to store node information
    class Node:
        def __init__(self, h, parent=None, action=None):
            self.h = h.copy()
            self.parent = parent
            self.action = action
            self.children = {}
            self.visits = 0
            self.value = float('inf')
            self.cached_c5 = None
            
        def get_c5(self):
            if self.cached_c5 is None:
                self.cached_c5 = evaluate_c5(self.h, dx)
            return self.cached_c5
            
        def ucb_score(self, exploration_weight):
            if self.visits == 0:
                return float('inf')
            return self.get_c5() + exploration_weight * np.sqrt(np.log(self.parent.visits) / self.visits)
    
    # Initialize root node
    root = Node(h)
    best_node = root
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Selection phase
        node = root
        while node.children and not node.cached_c5:
            # Find child with max UCB score
            best_child = max(node.children.values(), key=lambda child: child.ucb_score(exploration_weight))
            node = best_child
            
        # Expansion phase
        if not node.children and node.get_c5() < best_c5:
            # Generate new children by mutating the node
            num_mutations = max(1, int(len(node.h) * mutation_rate))
            for _ in range(5):
                h_copy = node.h.copy()
                # Select indices to mutate
                indices = np.random.choice(len(h_copy), size=num_mutations, replace=False)
                # Mutate with noise
                noise = np.random.normal(0, 0.1, num_mutations)
                h_copy[indices] += noise
                h_copy = np.clip(h_copy, 0, 1)
                # Renormalize to correct sum
                h_copy = h_copy * (22.5 / np.sum(h_copy))
                new_node = Node(h_copy, parent=node, action=indices)
                node.children[new_node] = new_node
        
        # Simulation phase (evaluate each child)
        if node.children:
            best_child_c5 = float('inf')
            best_child = None
            for child in node.children.values():
                c5_val = child.get_c5()
                if c5_val < best_child_c5:
                    best_child_c5 = c5_val
                    best_child = child
                    
            if best_child_c5 < best_c5:
                best_c5 = best_child_c5
                best_h = best_child.h.copy()
                best_node = best_child
                
            # Backpropagation
            current = best_child
            while current is not None:
                current.visits += 1
                # Update value
                if current.get_c5() < current.value:
                    current.value = current.get_c5()
                current = current.parent
    
        # Periodic reset
        if iteration % reset_interval == 0 and iteration > 0:
            # Reset 10% of nodes randomly
            for _ in range(len(root.children) // 10):
                node_to_reset = random.choice(list(root.children.values()))
                # Generate random starting point for this node
                new_h = np.zeros_like(node_to_reset.h)
                new_h[:22] = 1.0
                np.random.shuffle(new_h)
                new_h = new_h * (22.5 / np.sum(new_h))
                node_to_reset.h = new_h
                node_to_reset.cached_c5 = None
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
