import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to optimize, handles constraints
    def objective(h):
        # Ensure values are within bounds
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Set bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial evaluation
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Use differential evolution for global search
    try:
        # Collect best solution so far
        best_h = current_h.copy()
        best_c5 = current_c5
        
        # Run differential evolution with time constraint
        de_result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1),
            recombine=0.7,
            atol=1e-6,
            tol=1e-6,
            callback=lambda x, convergence: None,
            workers=1,
        )
        
        if de_result.success:
            final_h = np.clip(de_result.x, 0, 1)
            final_h = final_h * target_sum / np.sum(final_h)
            final_c5 = evaluate_c5(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
        
    except Exception as e:
        # Fallback in case of optimization error
        pass
    
    # Ensure final solution meets constraints
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Validate solution
    assert np.isclose(np.sum(final_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in final_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.95:
        # Run a quick local refinement if time allows
        try:
            from scipy.optimize import minimize
            def constrained_objective(h):
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
                return evaluate_c5(h, dx)
            
            # Simple local search around best
            method = 'L-BFGS-B'
            result = minimize(
                constrained_objective,
                final_h,
                method=method,
                bounds=[(0,1) for _ in range(n_points)],
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8},
            )
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_h = refined_h * target_sum / np.sum(refined_h)
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < final_c5:
                    final_h = refined_h
                    final_c5 = refined_c5
        except:
            pass

    return (final_h, final_c5, n_points)
