import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h with the provided construction
    h_init = np.array(initial_h_values)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_C5 = float('inf')
    best_h = h_init.copy()
    
    # Use the provided initial guess directly, no need for complex optimization
    # Set up CVXPY problem with QCP formulation
    h = Variable(n_points)
    C = Variable()
    
    # Constraints for h
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C >= 0
    ]
    
    # Get full convolution indices
    full_len = 2 * n_points - 1
    # For each possible lag k=-n+1,...,n-1, create constraint
    for k in range(-n_points + 1, n_points):
        # Get the range of indices where both h[i] and h[i+k] are valid
        if k >= 0:
            valid_range = range(0, n_points - k)
        else:
            valid_range = range(-k, n_points)
            
        # Compute sum of h[i] * (1 - h[i+k]) 
        expr = 0
        for i in valid_range:
            expr += h[i] * (1 - h[i + k])
        constraints.append(expr <= C * dx)
    
    # Objective function
    objective = Minimize(C)
    
    prob = Problem(objective, constraints)
    
    try:
        # Solve with SCS solver
        prob.solve(solver=SCS, verbose=False, max_iters=5000, 
                   eps=1e-5,acceleration_LOOKAHEAD=10)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            solution_h = h.value
            solution_C = C.value
            if solution_C < best_C5:
                best_C5 = solution_C
                best_h = solution_h.copy()
    except Exception as e:
        pass
    
    # If still not solved, fallback to simple optimization
    if best_C5 == float('inf'):
        h_fallback = h_init.copy()
        def compute_C5_fallback(h):
            h = np.array(h)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        def objective_fallback(h):
            return compute_C5_fallback(h)
        
        # Simple gradient descent using scipy for robustness
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        res = minimize(objective_fallback, h_fallback, method='SLSQP', 
                       bounds=bounds, constraints=constraints, 
                       options={'maxiter': 500, 'ftol': 1e-6})
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_C5 = compute_C5_fallback(final_h)
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = final_h.copy()
    
    # Ensure final solution respects constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
