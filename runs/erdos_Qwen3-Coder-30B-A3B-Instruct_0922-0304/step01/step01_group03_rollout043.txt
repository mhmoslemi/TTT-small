import numpy as np
import time
import cvxpy as cp
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Define optimization problem using CVXPY
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Add overlap constraints for a subset of lags
        # It's expensive to add all constraints, so sample a few
        max_lags = 2 * n_points - 2
        # Sample some lags to use for constraints
        lags_to_use = list(range(0, max_lags, max(1, max_lags // 20)))  # Sample about 20 lags
        
        # For each lag k (non-negative indices)
        for k in lags_to_use:
            left_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                left_sum += h_var[i] * (1 - h_var[j])
            # Add constraint: C >= left_sum * dx  
            constraints.append(C >= left_sum * dx)
        
        # Objective: minimize C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Solve the problem
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    # Renormalize
                    h_new = h_new * (22.5 / np.sum(h_new))
                    # Evaluate
                    new_c5 = evaluate_c5(h_new, dx)
                    if new_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = new_c5
                        iter_since_improvement = 0
                    else:
                        iter_since_improvement += 1
                else:
                    # Failed, fallback to random perturbation
                    iter_since_improvement += 1
            else:
                # Failed to solve, fallback to random perturbation
                iter_since_improvement += 1
        except Exception:
            # Any failure, fallback to random perturbation
            iter_since_improvement += 1
            
        # Create new h by slight random perturbation if no good solution
        if iter_since_improvement > 10:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (22.5 / np.sum(h))
            iter_since_improvement = 0
        
        h = h * (22.5 / np.sum(h))
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
