import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from seed if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    min_T = 1e-6
    max_iter = 10000
    max_evals = 100000
    evals = 0
    
    # Cooling schedule
    T = T0
    iteration = 0
    
    # Main optimization loop
    while evals < max_evals and time.time() - start_time < budget_s * 0.95:
        if T <= min_T:
            break
            
        # Make a perturbation to current solution
        h_new = h.copy()
        
        # Perturb a random subset of points (5-10%)
        subset_size = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, subset_size, replace=False)
        
        # Adjust values with bounded random changes
        for i in indices:
            delta = np.random.normal(0, 0.1)
            h_new[i] += delta
            
        # Maintain feasibility
        h_new = np.clip(h_new, 0, 1)
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        if current_sum > 0:
            scale_factor = target_sum / current_sum
            h_new = h_new * scale_factor
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new, dx)
        evals += 1
        
        # Accept or reject based on Metropolis criterion
        delta = C5_new - best_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        
        # Cooling schedule
        T *= alpha
        iteration += 1
    
    # Final evaluation with best solution
    final_C5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_C5, n_points)
