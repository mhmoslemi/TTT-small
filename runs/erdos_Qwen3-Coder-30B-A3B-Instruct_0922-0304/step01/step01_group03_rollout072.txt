        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))

    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    perturb_rate = 0.1
    mutation_strength = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Create new candidate by perturbing current h
        h_new = h.copy()
        
        # Select random subset to modify
        n_modify = max(1, int(len(h) * perturb_rate))
        indices = np.random.choice(len(h), size=n_modify, replace=False)
        
        # Apply perturbation
        for i in indices:
            h_new[i] += np.random.normal(0, mutation_strength)
        
        # Ensure values remain in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to sum = 22.5
        h_new = h_new * (22.5 / np.sum(h_new))
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        
        # Periodic random restart to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            restart_h = np.zeros(len(h))

                best_c5 = restart_c5

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
