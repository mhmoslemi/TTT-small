import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Initialize h with the provided values, normalize to sum to n_points/2
    h_init = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Optimization using CVXPY with ECOS solver
    h = Variable(n_points)
    max_overlap = Variable()
    
    # Convolution constraints in frequency domain
    # We want to minimize max C(k) where C(k) = sum h[i] * (1 - h[i+k])
    
    # Objective: minimize max overlap
    objective = Minimize(max_overlap)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        max_overlap >= 0
    ]
    
    # Add overlap constraints for all shifts via spectral method
    # For a step function optimization, a practical approach is to use a relaxation
    # We model this via the fact that the correlation is upper bounded by a linear program
    
    # Heuristic grid search to find good starting point
    best_h = h_init.copy()
    best_C5 = np.inf
    
    # Iterative refinement with warm starts from different starting points
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a new random perturbation while maintaining sum constraint
        perturbation = np.random.normal(0, 0.05, n_points)
        # Normalize perturbation to maintain sum constraint
        perturbation = perturbation - np.mean(perturbation)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute C5 for trial solution
        corr = np.correlate(h_trial, 1 - h_trial, mode="full")
        trial_C5 = np.max(corr) * dx
        
        if trial_C5 < best_C5:
            best_C5 = trial_C5
            best_h = h_trial.copy()
    
    # Use an optimization approach for final refinement
    # Use copy of best solution as starting point
    h_refine = best_h.copy()
    
    # Direct gradient-free optimization using L-BFGS-B
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    from scipy.optimize import minimize
    
    # We use a slightly modified version to avoid numerical issues
    def objective(h):
        return compute_C5(h)
        
    # Additional refinement with bounds + sum constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    # We do a final optimization step
    h_final = h_refine.copy()
    
    # Using the bounds and sum constraint, we try to refine one more time
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    res = minimize(
        objective, 
        h_final, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    if res.success:
        final_h = res.x
    else:
        final_h = h_final
    
    # Ensure the final solution is valid
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Compute final C5 value
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
