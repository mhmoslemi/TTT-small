import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to feasible region: clip to [0,1] and scale to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient of C5 with respect to h at lag k"""
    gradient = np.zeros(n_points)
    for i in range(n_points):
        if k >= 0:
            # For positive lag k: h[i] contributes to overlap with h[i-k]
            if i >= k:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i - k]
            else:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i - k + n_points]
        else:
            # For negative lag k: h[i] contributes to overlap with h[i+k]
            if i < n_points + k:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i - k]
            else:
                gradient[i] = 1 - 2 * h[i] - 2 * h[i - k - n_points]
    return gradient
