import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    # Ensure sum equals n_points/2 and values in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    generations = 0
    max_generations = budget_s * 10  # Rough estimate
    
    # Initialize population
    population = []
    fitness = []
    
    # Use provided initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
        population.append(h.copy())
        fitness.append(evaluate_c5(h, dx))
    
    # Generate random individuals
    for _ in range(pop_size - len(population)):
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, n_points)
        population.append(h.copy())
        fitness.append(evaluate_c5(h, dx))
    
    best_individual = population[np.argmin(fitness)].copy()
    best_fitness = min(fitness)
    
    # Genetic parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    relaxation_interval = 10
    
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        # Selection based on fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:pop_size//2]]
        parent_fitness = [fitness[i] for i in sorted_indices[:pop_size//2]]
        
        # Create new population
        new_population = parents.copy()
        new_fitness = parent_fitness.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(parents) >= 2:
                # Choose two random parents
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                
                # Segment-wise crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Apply feasibility projection
                child1 = project_h(child1, n_points)
                child2 = project_h(child2, n_points)
                
                new_population.extend([child1, child2])
                new_fitness.extend([evaluate_c5(child1, dx), evaluate_c5(child2, dx)])
            else:
                # Mutation: add Gaussian noise
                parent = np.random.choice(parents)
                child = parent + np.random.normal(0, 0.02, n_points)
                child = project_h(child, n_points)
                new_population.append(child)
                new_fitness.append(evaluate_c5(child, dx))
        
        # Trim to exact population size
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Periodic relaxation step
        if generations % relaxation_interval == 0 and len(parents) > 0:
            # Use best individual to perform relaxation
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            
            # Try gradient descent on the best individual
            try:
                h_temp = best_h.copy()
                step = 0.01
                iterations = 100
                
                for _ in range(iterations):
                    # Compute overlap
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    c5_val = np.max(corr) * dx
                    
                    # Gradient via finite differences
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - c5_val) / eps
                    
                    # Update h
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                # Update if better
                new_c5 = evaluate_c5(h_temp, dx)
                if new_c5 < best_fitness:
                    best_individual = h_temp.copy()
                    best_fitness = new_c5
            except:
                pass
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        generations += 1
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    return (best_individual.tolist(), best_fitness, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
