import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # CVXPY approach for convex relaxation
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        t >= 0
    ]
    
    # Add overlap constraints for all possible shifts
    # For shift k, we have sum(h[i] * (1 - h[i+k])) <= t
    # This corresponds to sum(h[i] - h[i] * h[i+k]) <= t
    # So sum(h[i]) - sum(h[i] * h[i+k]) <= t
    # Since we know sum(h[i]) = target_sum:
    # target_sum - sum(h[i] * h[i+k]) <= t
    # Thus: sum(h[i] * h[i+k]) >= target_sum - t
    
    # In the full correlation case, we consider all correlations:
    # For shift k, the correlation is sum_{i} h[i] * (1 - h[i+k])
    # = sum_{i} h[i] - sum_{i} h[i] * h[i+k] 
    # = target_sum - sum_{i} h[i] * h[i+k]
    # So constraint is: target_sum - sum_{i} h[i] * h[i+k] <= t
    # => sum_{i} h[i] * h[i+k] >= target_sum - t
    
    # Use dual formulation: it's easier to explicitly include the correlation constraints
    # but we'll add them w.r.t. the correlation of h with (1-h) for all shifts
    
    # Better approach: Sample the correlation function and bound it
    # For discrete case with 42 points, we have correlation of size 2*n_points - 1 = 83
    
    # Create correlation matrix terms
    # But for efficiency, we directly code the constraint using variable substitution
    
    # We will build a constraint per shift in [-n+1, n-1]
    # where we consider correlation = sum(h[i] * (1 - h[i+k])) for shift k
    
    # Add all overlap constraints via the correlation calculation
    # We compute it directly in numpy for reference
    max_shift = n_points - 1  # Full support of correlation
    
    # We consider the full convolution but in constraint form we use:
    # sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k]) <= t  for all k in [0, n-1]  
    # For negative shifts we get differently placed terms but the same idea.
    # Compute all valid positions for each shift k:
    for k in range(-max_shift, max_shift + 1):
        if k >= 0:
            i_indices = np.arange(0, n_points - k)
            j_indices = np.arange(k, n_points)
        else:
            i_indices = np.arange(-k, n_points)
            j_indices = np.arange(0, n_points + k)
        
        # Term is: h[i] * (1 - h[j]) = h[i] - h[i] * h[j]
        # So sum over indices of h[i] * (1 - h[j]) = sum(h[i]) - sum(h[i]*h[j])
        # Constraint: sum(h[i]) - sum(h[i]*h[j]) <= t
        # => sum(h[i]*h[j]) >= (sum(h) - t)
        
        # But to ensure the constraint is correctly setup,
        # we reformulate as: sum(h[i]*h[j]) <= sum(h) - t for the case when t > 0
        
        # Simpler constraint: total correlation <= t for any shift
        # But better: sum(h[i] * (1-h[i+k])) <= t for all shifts k
        # That means sum(h[i]) - sum(h[i]*h[i+k]) <= t
        # So sum(h[i]*h[i+k]) >= sum(h[i]) - t = target_sum - t
            
        # We use a linear constraint
        # sum_{i=0 to n-k-1} h[i] * (1 - h[i+k]) <= t
        # That's sum_{i=0 to n-k-1} [h[i] - h[i]*h[i+k]] <= t
        # The constraint in CVXPY will be expressed in a matrix-vector form.
        
        # Avoid complicated indexing now. Use loop for dynamics accuracy.
        # Let's encode just what we care about in this approach:
        # For k=-n+1 to n-1, we define:
        # corr = sum_{i} h[i] * (1 - h[i+k]) <= t
        
        # Reconstruct one by one
        if k >= 0:
            # sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            # Total contribution to correlation sum is the same as before
            corr_expr = 0
            for i in range(n_points - k):
                corr_expr += h[i] * (1 - h[i + k])
        else:
            # k < 0 
            # sum_{i=-k}^{n-1} h[i] * (1 - h[i+k])
            corr_expr = 0
            for i in range(-k, n_points):
                corr_expr += h[i] * (1 - h[i + k])
                
        constraints.append(corr_expr <= t)

    # Create problem
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, 
                   mosek_params={'MSK_DPAR_OPTIMIZER_MAX_TIME': budget_s - 10})
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            final_c5 = t_opt
        else:
            # We fallback to initial point if no success
            h_opt = h_init.copy()
            final_c5 = evaluate_c5(h_opt, dx)
    except Exception as e:
        h_opt = h_init.copy()
        final_c5 = evaluate_c5(h_opt, dx)
    
    # Final normalization
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * target_sum / np.sum(h_opt)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass
    
    return (h_opt, final_c5, n_points)
