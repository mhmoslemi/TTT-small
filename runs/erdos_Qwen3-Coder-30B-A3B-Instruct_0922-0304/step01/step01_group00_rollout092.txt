import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Find maximum overlap index
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient of overlap with respect to each h[i]
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute partial derivative of L(k_max) w.r.t h[i]
            if i + k_max >= 0 and i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i - k_max >= 0 and i - k_max < n_points:
                grad[i] -= h[i - k_max]
        
        # Find the coordinate that decreases the overlap most
        i_star = np.argmin(grad)
        
        # Perform line search to update h[i_star]
        # Keep h[i_star] in bounds [0,1]
        # keep sum(h) = n_points/2 constant
        
        # Initial estimate: change h[i_star] by small amount
        step = 0.01
        best_new_h = h.copy()
        best_new_c5 = current_c5
        
        new_h = h.copy()
        new_h[i_star] += step
        # Adjust other elements to maintain sum
        if i_star + k_max < n_points:
            new_h[i_star + k_max] -= step * (1 - h[i_star]) / (1 - h[i_star + k_max]) if h[i_star + k_max] < 1.0 else 0.0
        elif i_star - k_max >= 0:
            new_h[i_star - k_max] -= step * h[i_star] / h[i_star - k_max] if h[i_star - k_max] > 0.0 else 0.0
            
        # Clip new_h to [0,1] and normalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = evaluate_c5(new_h, dx)
        if new_c5 < best_new_c5:
            best_new_c5 = new_c5
            best_new_h = new_h.copy()
            
        if best_new_c5 < current_c5:
            h = best_new_h.copy()
        else:
            # Sometimes stochastic jumps help
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
