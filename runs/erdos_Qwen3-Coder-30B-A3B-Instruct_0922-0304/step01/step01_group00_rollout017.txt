import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization: half 1s, half 0s, plus small noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient of overlap at k_max
        grad = np.zeros(n_points)
        k = k_max
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += 1 - h[i + k]
            if i >= k:
                grad[i] -= h[i - k]
                
        # Select coordinate to improve
        i_update = np.argmin(grad)
        
        # Compute optimal step size
        # We want to minimize L(k_max) = sum h[i] * (1 - h[i+k])
        # Which is equivalent to minimizing sum h[i]*h[i+k] - sum h[i]
        # Gradient is: 1 - h[i+k] if i + k < n, -h[i] if i >= k
        # So we compute the impact of moving h[i_update]
        
        # Simple line search for best update
        step_size = 0.01
        old_val = h[i_update]
        
        # Try a few candidates
        best_h_candidate = h.copy()
        min_c5_val = max_overlap
        
        # Try various values for the coordinate
        for delta in [0.0001, 0.001, 0.01, 0.1]:
            h_test = h.copy()
            h_test[i_update] = max(0, min(1, h[i_update] + delta))
            # Recalculate overlap tensor with new h to compute exact value
            new_overlap = correlate(h_test, 1 - h_test, mode="full")
            max_overlap_test = np.max(new_overlap) * dx
            if max_overlap_test < min_c5_val:
                min_c5_val = max_overlap_test
                best_h_candidate = h_test.copy()
        
        # Try a small step in opposite direction too
        if i_update + k < n_points:
            # This is a complex adjustment; use simpler approach:
            # Just make a small change to the coordinate
            h_update = h.copy()
            h_update[i_update] = max(0, min(1, h[i_update] + step_size * (grad[i_update] < 0) - step_size * (grad[i_update] > 0)))
            # Normalize the step to keep sum constraint
            h_update = h_update * (n_points / 2) / np.sum(h_update)
            new_overlap = correlate(h_update, 1 - h_update, mode="full")
            max_overlap_test = np.max(new_overlap) * dx
            if max_overlap_test < min_c5_val:
                min_c5_val = max_overlap_test
                best_h_candidate = h_update.copy()

        h = best_h_candidate.copy()

    # Ensure final solution meets constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
