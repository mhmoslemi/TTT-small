import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_and_normalize(h, n_points, dx):
    # Project h to [0,1] and normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points

        # Simple initialization: half 1s, half 0s
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_and_normalize(h, n_points, dx)

    # Optimization parameters
    max_iter = 1000
    lr = 0.01
    max_lr = 0.1
    min_lr = 1e-5
    patience = 10
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps and C5
        corr = correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        if patience_counter > patience:
            lr = max(min_lr, lr * 0.5)
            patience_counter = 0
            
        # Local gradient-based update
        gradient = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()

            h_plus = project_and_normalize(h_plus, n_points, dx)

            gradient[i] = (c5_plus - c5_val) / eps

        # Update h using gradient
        h -= lr * gradient
        h = project_and_normalize(h, n_points, dx)
