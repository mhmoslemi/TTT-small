import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper to project solution back to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Objective function for optimization
    def objective(h):
        h = np.abs(h)
        h = project(h)
        return evaluate_c5(h, dx)
    
    # DE bounds: (0, 1) for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    de_options = {
        'strategy': 'best1bin',
        'popsize': 40,
        'mut': 0.8,
        'recomb': 0.9,
        'maxiter': 100,
        'seed': seed,
        'workers': 1,
        'updating': 'func'
    }
    
    # Run DE for global search
    de_result = differential_evolution(
        objective,
        bounds,
        **de_options,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    # Initialize with DE solution
    current_h = project(de_result.x)
    current_c5 = evaluate_c5(current_h, dx)
    
    # Local refinement with L-BFGS-B
    # Define bounds for L-BFGS-B
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    # Objective function for L-BFGS-B (with sum constraint)
    def objective_lbfgs(h):
        h = np.clip(h, 0, 1)
        # Maintain sum constraint
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint to maintain sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    try:
        # Optimize with L-BFGS-B using best DE solution
        result = minimize(
            objective_lbfgs,
            current_h,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            options={'maxiter': 500, 'eps': 1e-6},
            callback=lambda x: None,
        )
        
        # Check if optimization successful and update if better
        if result.success:
            refined_h = project(result.x)
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < current_c5:
                current_h = refined_h
                current_c5 = refined_c5
    except:
        pass
    
    # Final validation
    current_h = project(current_h)
    current_c5 = evaluate_c5(current_h, dx)
    
    assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
    assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s * 0.95:  # Avoid time explosion
        pass

    return (current_h, current_c5, n_points)
