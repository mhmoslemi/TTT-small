import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # This is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)

        # Make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.99
    min_T = 1e-8
    max_iter = 10000
    max_evals = 100000
    
    # Evaluate initial solution
    current_h = h.copy()
    current_C5 = best_C5
    T = T0
    eval_count = 0
    
    # Main Simulated Annealing loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the current solution
        new_h = current_h.copy()
        
        # Choose a random subset of points to modify (5-10%)
        num_modify = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_modify, replace=False)
        
        # Modify values with random perturbations
        for i in indices:
            # Small random change
            delta = np.random.uniform(-0.1, 0.1)
            new_h[i] += delta
            
        # Ensure bounds [0, 1]
        new_h = np.clip(new_h, 0, 1)
        
        # Adjust to maintain integral constraint
        old_sum = np.sum(current_h)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            # Rescale if necessary
            factor = old_sum / new_sum
            new_h = new_h * factor
            
        # Exactly enforce sum constraint
        new_h = new_h * (n_points / 2.0) / np.sum(new_h)
        
        # Evaluate the new solution
        new_C5 = evaluate_erdos_solution(new_h, dx)
        eval_count += 1
        
        # Accept or reject the new solution
        delta = new_C5 - current_C5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
        
        # Decrease temperature
        T = max(min_T, T * alpha)
        
        # Early stopping if we're not making progress
        if iter_num > 100 and eval_count > max_evals:
            break
    
    return (best_h, best_C5, n_points)
