import numpy as np
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Scale to sum constraint
    h = h * (n_points / 2) / np.sum(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-scale to make sum correct
    h = h * (n_points / 2) / np.sum(h)
    return h
