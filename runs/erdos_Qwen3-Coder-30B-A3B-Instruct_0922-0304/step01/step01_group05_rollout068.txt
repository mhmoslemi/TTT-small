import numpy as np
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 500
    num_particles = 50
    w_max, w_min = 0.9, 0.5
    c1, c2 = 2.0, 2.0
    v_max = 0.5
    
    # Initialize particles around the initial solution
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= (n_points / 2) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # PSO initialization
    particles = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    initial_h = np.array(initial_h_values)
    for _ in range(num_particles):
        h = initial_h + 0.2 * (2 * np.random.rand(n_points) - 1)
        h = repair(h)
        particles.append(h.copy())
        velocities.append(0.1 * np.random.randn(n_points))
        personal_best.append(h.copy())
        personal_best_values.append(compute_overlap(h))
    
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO loop
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        w = w_max - (w_max - w_min) * (iteration / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair and evaluate
            particles[i] = repair(particles[i])
            current_value = compute_overlap(particles[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_value
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
                    last_improvement = iteration
        
        # Perturbation if no improvement in last 20 iterations
        if iteration - last_improvement > 20:
            for i in range(num_particles):
                if np.random.rand() < 0.3:  # 30% chance to perturb
                    particles[i] = repair(particles[i] + 0.1 * (2 * np.random.rand(n_points) - 1))
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = compute_overlap(particles[i])
            
            # Update global best
            new_global_idx = np.argmin(personal_best_values)
            if personal_best_values[new_global_idx] < global_best_value:
                global_best = personal_best[new_global_idx].copy()
                global_best_value = personal_best_values[new_global_idx]
                last_improvement = iteration
    
    # Final evaluation
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
