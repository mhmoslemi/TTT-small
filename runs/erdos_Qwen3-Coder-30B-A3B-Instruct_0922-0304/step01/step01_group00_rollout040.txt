import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: half 1s, half 0s + noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Hyperparameters
    max_iter = 10000
    line_search_precision = 1e-4
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Approximate gradient w.r.t. each h[i] for the maximum overlap
        # The gradient is complex and has special structure dependent on k_max
        # We compute it by finite difference with small perturbations
        
        # Component-wise update - find coordinate that contributes most to max overlap
        # and reduce it 
        
        # Identify k_max and positions
        k_max = max_k - (n_points - 1)
        # k_max = k_max if positive, else negative shift
        
        # Identify which h[i] has the most impact on this overlap
        # For element i: contributes to overlap when k_max is such that i + k_max is valid
        grad_approx = np.zeros(n_points)
        
        # I'll approximate this using a smoothed approach
        # For overlap involving h[i] and 1-h[i+k_max], the contribution to the overlap is
        # h[i] * (1 - h[i + k_max])
        
        # Define contribution of each element to the maximum overlap
        # The maximum overlap corresponds to the all-overlapping case
        # For each i, compute how a change affects the total overlap
        for i in range(n_points):
            if i + k_max < n_points and i + k_max >= 0:
                if k_max >= 0:
                    grad_approx[i] = -h[i + k_max]  # See paper
                else:
                    grad_approx[i] = h[i]  # See paper
            else:
                grad_approx[i] = 0
                
        # Find the index to reduce
        # We want minimum gradient (i.e., largest negative gradient to reduce overlap)
        i_star = np.argmin(grad_approx)
        
        # Perform line search to adjust h[i_star] along the direction that reduces overlap
        # The line search step is: h_new = h - alpha * e_{i_star}
        # We must ensure the sum is preserved
        
        # Simple 1D line search along direction (1, -1, 0, ..., 0) that keeps sum constraint
        # Two parameters need adjustment: h[i_star] and another related for the sum constraint
        
        # We will be more systematic and do projected line search
        # Let's consider small step in negative direction of the gradient of max overlap w.r.t. h[i_star]
        
        step_size = 0.01
        old_value = h[i_star]
        
        # We assume the impact on the overlap function is linear to h[i_star] for small changes
        # with the effect delta = (h[i_star] - (1-h[i_star+k_max])) * d(h[i_star])/d(h[i_star])
        # Approximate as: dL/dh = -h[i_star+k_max] if continuous and it depends on h[i_star+k_max]
        
        # Compute delta_k with limits for safe updating
        safe_h = h.copy()
        min_val = 0.0
        max_val = 1.0
        if i_star + k_max < n_points and i_star + k_max >= 0:
            # Moving down reduces overlap
            to_move_down = np.min([1.0, safe_h[i_star], safe_h[i_star + k_max]])
            if i_star + k_max >= 0 and k_max > 0:
                # Let's conservatively decrease h[i_star] and increase or decrease h[i_star+k_max]
                # by same amount to preserve |sum| constraints, but just adjust one for direct effect
                
                step = min(step_size, to_move_down / 2.0)  # Ensure it's within bounds
                safe_h[i_star] -= step
                # If h[i_star+k_max] is at 1, it won't help, so it's better not to adjust it
                
                # Early exit if not beneficial
                test_c5 = evaluate_c5(safe_h, dx)
                if test_c5 < max_overlap:
                    h = safe_h.copy()
                else:
                    h[i_star] = old_value  # rollback
            else:
                # Adjusting h[i] directly
                step = min(step_size, to_move_down)
                safe_h[i_star] -= step
                test_c5 = evaluate_c5(safe_h, dx)
                if test_c5 < max_overlap:
                    h = safe_h.copy()
                else:
                    h[i_star] = old_value  # rollback
                    
        # Normalize sum to maintain constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Let's add an alternative search pattern that also tries multiple paths
        # Idea: add stochasticity from intermittent large updates
        
        if iteration % 50 == 0 and np.random.rand() > 0.7:
            # Add entropy to avoid getting stuck
            tiny_pert = np.random.normal(0, 0.005, n_points)
            h += tiny_pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
