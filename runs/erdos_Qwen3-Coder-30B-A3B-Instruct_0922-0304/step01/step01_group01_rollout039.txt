import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

        # Compute gradient for the dominant lag
        grad = np.zeros(n_points)

            if 0 <= i - max_lag < n_points:
                grad[i] = 1 - 2*h[i] - 2*h[i-max_lag]
        
        # Update ADAM moments
        m = beta1 * m + (1 - beta1) * grad

        # Decay learning rate
        alpha *= decay_rate
        
        # Early stopping if no improvement
        if abs(prev_c5 - c5_val) < 1e-5:
