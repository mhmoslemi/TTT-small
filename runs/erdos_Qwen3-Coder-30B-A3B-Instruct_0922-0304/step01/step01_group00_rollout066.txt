import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe style optimization with coordinate-wise updates
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(corr)
        max_overlap_val = corr[max_overlap_idx] * dx
        
        # Update best solution
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Find which coordinate contributes most to the max overlap
        # Compute gradient of overlap at k_max
        k_max = max_overlap_idx - (n_points - 1)
        
        # Compute gradient of L(k_max) with respect to h[i]
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max]
                
        # Choose coordinate to update (minimize gradient = most negative)
        update_idx = np.argmin(grad)
        
        # Compute step size that maintains feasibility
        # New value h_new = h[update_idx] + alpha * direction
        # We want h[update_idx] + alpha * direction to be in [0,1]

        # Find maximum feasible step
        alpha_max = 1.0
        direction = 1.0  # direction of change for update_idx
        
        # Check bounds
        if h[update_idx] + direction > 1.0:
            alpha_max = min(alpha_max, (1.0 - h[update_idx]) / direction)
        if h[update_idx] + direction < 0.0:
            alpha_max = min(alpha_max, -h[update_idx] / direction)
        
        # Find delta sum to maintain constraint

        # Actual update
        h[update_idx] += direction * alpha_max
        h[update_idx] = np.clip(h[update_idx], 0, 1)
        
        # Adjust other coordinates to maintain sum (simplified approx)
        # Find index that maximizes contribution to max_overlap
        h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
