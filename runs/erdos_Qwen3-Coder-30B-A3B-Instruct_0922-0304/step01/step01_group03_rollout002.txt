import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:22] = 1.0  # First 22 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 50
    temp = 1e-3
    temp_decay = 0.995
    noise_scale = temp / 2.0
    
    # Precompute correlation indices for faster evaluation
    def compute_overlaps(h_vec):
        correlate_result = signal.correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = correlate_result * dx
        return overlaps
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        overlaps = compute_overlaps(h)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient (finite difference approximation)
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(compute_overlaps(h_plus))
            overlap_minus = np.max(compute_overlaps(h_minus))
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
            
        # Apply gradient update with noise
        h_temp = h - alpha * grad + np.random.normal(0, noise_scale, len(h))
        h_temp = np.clip(h_temp, 0, 1)
        
        # Renormalize
        scale = (len(h) / 2) / np.sum(h_temp)
        h_new = h_temp * scale
        h_new = np.clip(h_new, 0, 1)
        
        # Re-renormalize to exactly satisfy constraint
        h_new = h_new * (len(h) / 2) / np.sum(h_new)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update using Metropolis criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1
        else:
            iter_since_improvement += 1
            
        # Temperature decay
        temp *= temp_decay
        
        # Reduce learning rate if no improvement
        if iter_since_improvement > max_no_improvement:
            alpha = max(alpha * gamma, alpha_min)
            iter_since_improvement = 0
            
        # Periodic reset to binary configuration
        if iteration % 2000 == 0 and iteration > 0:
            h_binary = np.zeros(len(h))
            # Create a binary pattern with varying blocks
            block_size = np.random.randint(3, 8)
            num_blocks = len(h) // block_size + 1
            for i in range(min(num_blocks, len(h) // block_size + 1)):
                start = i * block_size
                end = min(start + block_size, len(h))
                if i % 2 == 0:
                    h_binary[start:end] = 1.0
            h = np.clip(h_binary, 0, 1)
            h = h * (len(h) / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
