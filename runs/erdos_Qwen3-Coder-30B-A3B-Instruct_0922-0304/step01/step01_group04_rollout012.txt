import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum

    # Prepare convex optimization
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(name="t")
    
    # Constraints for h in [0,1]
    constraints = [h >= 0, h <= 1]
    
    # Constraint: sum(h) = target_sum
    constraints.append(cp.sum(h) == target_sum)
    
    # Constraint: t >= max_k integral
    # We'll construct the full correlation matrix implicitly
    
    # Due to memory and time constraints, we'll add constraints for a few key shifts
    # specifically the most critical ones of small |k|
    max_shift = min(20, n_points - 1)
    for k in range(-max_shift, max_shift + 1):
        if k == 0:
            continue
        # Create the constraint for lag k
        # integral = sum(h[i] * (1 - h[i+k])) 
        # That means we loop over i and compute i+k, ensuring bounds
        # So we need to build shifting constraint efficiently, by concatenating
        # For now just do it explicitly for small k.values
        
        # i indices: 0 to n_points-1-k (if k>0), or 0 to n_points-1+k (if k<0)
        # Use correlation-like expression
        if k > 0:
            # i from 0 to n_points-1-k, j=i+k
            # h[i] * (1 - h[j])
            # Sum over all such i: h[i] * (1 - h[i+k])
            indices_i = np.arange(0, n_points - k)
            indices_j = np.arange(k, n_points)
            expr = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))
        else:
            # k < 0, so j = i+k => i = j-k, from 0 to n_points+k
            # but indices_j = 0 to n_points+k
            # and indices_i = 0+k to n_points, i.e. k to n_points
            indices_j = np.arange(0, n_points + k)
            indices_i = np.arange(-k, n_points)
            expr = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))
            
        constraints.append(expr <= t)
        
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem with MOSEK
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - (time.time() - start_time) - 1)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            t_opt = t.value
            current_h = np.clip(h_opt, 0, 1)
            # Enforce sum = target_sum exactly
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = evaluate_c5(current_h, dx)
        else:
            # Fallback to initial solution
            current_h = h_init.copy()
            current_c5 = evaluate_c5(current_h, dx)
            
    except Exception as e:
        # Fallback to initial solution
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)
        
    # Ensure we don't run out of time - return at least a valid solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
