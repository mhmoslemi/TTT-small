import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():

    else:

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    n_points = len(h)
    simplex_size = 0.1
    reflection_factor = 1.3
    expansion_factor = 1.5
    contraction_factor = 0.8
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 1000
    
    # Initialize simplex
    simplex = [h.copy()]
    for _ in range(n_points):
        h_pert = h + simplex_size * np.random.normal(0, 1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        scale = (n_points / 2) / np.sum(h_pert)
        h_pert = h_pert * scale
        simplex.append(h_pert)
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate simplex
        evals = []
        for point in simplex:
            evals.append(evaluate_c5(point, dx))
        
        # Order by fitness
        order = np.argsort(evals)
        simplex = [simplex[i] for i in order]
        evals = [evals[i] for i in order]
        
        if evals[0] < best_c5:
            best_c5 = evals[0]
            best_h = simplex[0].copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > max_no_improvement:
            # Perturb best solution and restart simplex
            perturb_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            scale = (n_points / 2) / np.sum(h_pert)
            h_pert = h_pert * scale
            
            # Replace worst simplex member
            simplex[-1] = h_pert
            iter_since_improvement = 0
        
        # Compute centroid (excluding worst point)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        reflected = centroid + reflection_factor * (centroid - simplex[-1])
        reflected = np.clip(reflected, 0, 1)
        scale = (n_points / 2) / np.sum(reflected)
        reflected = reflected * scale
        
        reflected_c5 = evaluate_c5(reflected, dx)
        
        if evals[0] <= reflected_c5 < evals[-2]:
            # Accept reflection
            simplex[-1] = reflected
        elif reflected_c5 < evals[0]:
            # Expansion
            expanded = centroid + expansion_factor * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            scale = (n_points / 2) / np.sum(expanded)
            expanded = expanded * scale
            expanded_c5 = evaluate_c5(expanded, dx)
            if expanded_c5 < reflected_c5:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            if reflected_c5 < evals[-1]:
                # Outside contraction
                contracted = centroid + contraction_factor * (reflected - centroid)
            else:
                # Inside contraction
                contracted = centroid - contraction_factor * (centroid - simplex[-1])
            contracted = np.clip(contracted, 0, 1)
            scale = (n_points / 2) / np.sum(contracted)
            contracted = contracted * scale
            contracted_c5 = evaluate_c5(contracted, dx)
            if contracted_c5 < evals[-1]:
                simplex[-1] = contracted
            else:
                # Shrink towards best
                for i in range(1, n_points):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    scale = (n_points / 2) / np.sum(simplex[i])
                    simplex[i] = simplex[i] * scale
    
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
