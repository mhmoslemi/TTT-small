import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 45
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        population.append(h)
    else:
        # Create a simple initial configuration
        h = np.zeros(n_points)
        h[:23] = 1.0
        h = h * (n_points / 2.0 / np.sum(h))
        population.append(h)
    
    # Fill the rest with random configurations
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0 / np.sum(h))
        population.append(h)

    # Renormalization helper
    def renormalize(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2.0) / np.sum(h)
        return h * scale

    # Evaluation function
    def evaluate(h):
        if time.time() - start_time > budget_s - 0.1:
            return float('inf')
        overlaps = signal.correlate(h, 1 - h, mode='full') * (2.0/n_points)
        return np.max(overlaps)

    # Selection operator - tournament
    def tournament_select(pop, k=3):
        candidates = np.random.choice(len(pop), size=k, replace=False)
        scores = [evaluate(pop[i]) for i in candidates]
        return pop[candidates[np.argmin(scores)]]

    # Crossover operator
    def crossover(p1, p2):
        point = np.random.randint(1, n_points)
        child = np.concatenate([p1[:point], p2[point:]])
        return renormalize(child)

    # Mutation operator
    def mutate(h, noise_level=0.05):
        mutated = h.copy()
        indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        mutated[indices] += np.random.normal(0, noise_level, len(indices))
        return renormalize(mutated)

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    generation = 0
    no_improvement = 0
    noise_level = 0.05
    
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        if generation % 100 == 0:
            no_improvement = 0
            if noise_level > 0.005:
                noise_level *= 0.95
        
        # Evaluate entire population
        scores = [evaluate(h) for h in population]
        sorted_indices = np.argsort(scores)
        best_idx = sorted_indices[0]
        
        if scores[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = scores[best_idx]
            no_improvement = 0
        else:
            no_improvement += 1

        # Retain top performers
        elite_count = population_size // 10
        top_performers = [population[i] for i in sorted_indices[:elite_count]]
        
        # Generate new population
        new_population = top_performers[:]
        
        # Create offspring through crossover and mutation
        while len(new_population) < population_size:
            parent1 = tournament_select(population)
            parent2 = tournament_select(population)
            child = crossover(parent1, parent2)
            child = mutate(child, noise_level)
            new_population.append(child)
            
        population = new_population[:population_size]
        
        # Check if diversification needed
        if no_improvement > 50:
            for i in range(int(0.2 * population_size)):
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0 / np.sum(h))
                population[np.random.randint(0, population_size)] = h
            noise_level *= 0.9
            no_improvement = 0
    
    return (best_h.tolist(), best_c5, n_points)
