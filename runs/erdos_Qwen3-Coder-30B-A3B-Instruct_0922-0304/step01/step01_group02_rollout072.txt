import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

        half = n_points // 2
        h[:half] = 1.0
        h[half:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.995
    T_min = 1e-6
    max_iter = 100000
    evals = 0
    
    # Run simulated annealing
    T = T0
    current_h = h.copy()
    current_C5 = best_C5
    
    it = 0
    while T > T_min and evals < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Generate a perturbation
        new_h = current_h.copy()
        
        # Randomly decide if we make small or large changes
        if np.random.random() < 0.7:  # small changes
            # Modify a few points with small perturbations
            n_changes = max(1, n_points // 20)
            indices = np.random.choice(n_points, n_changes, replace=False)
            for i in indices:
                delta = np.random.normal(0, 0.05)
                new_h[i] += delta
        else:  # larger changes
            # Modify a larger portion of the function
            n_changes = max(1, n_points // 4)
            indices = np.random.choice(n_points, n_changes, replace=False)
            for i in indices:
                new_h[i] = np.random.random()
        
        # Enforce sum constraint
        old_sum = np.sum(current_h)

            new_h = new_h * (old_sum / new_sum)
        
        # Enforce bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(new_h, dx)

        delta = new_C5 - current_C5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_h = new_h
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cool down
        T *= alpha
        it += 1
        
        if it % 1000 == 0:
            # Safety check on time
            if (time.time() - start_time) > budget_s * 0.95:
