import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Fallback initialization - triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
    
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current maximum overlap and corresponding shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Create quadratic program for this shift
        # Objective: minimize sum over i of h[i] * (1 - h[i + shift])
        # which is sum of h[i] - h[i]*h[i + shift]
        # = sum h[i] - sum h[i]*h[i + shift]
        # We'll formulate the QP from scratch for simplicity
        
        # Variable: h as a cvxpy variable
        h_var = cp.Variable(n_points)
        
        # Constraint 1: integral constraint
        integral_constraint = cp.Equality(cp.sum(h_var), n_points / 2)
        
        # Constraint 2: bounds
        bound_constraints = [h_var[i] >= 0 for i in range(n_points)] + \
                            [h_var[i] <= 1 for i in range(n_points)]
        
        # For shift k, we want to minimize sum_{i} h[i] * (1 - h[i + k])
        # = sum(h[i]) - sum(h[i] * h[i + k]) 
        # Let's consider the maximally overlapping shift k = max_shift
        
        # Build the quadratic term
        quad_term = 0.0
        linear_term = 0.0
        
        # Compute based on shift
        k = max_shift
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i = 0 to n_points - k - 1
            for i in range(n_points - k):
                quad_term += h_var[i] * h_var[i + k]
                linear_term += h_var[i]
        else:
            # h[i + k] * (1 - h[i]) for i = -k to n_points - 1
            for i in range(-k, n_points):
                quad_term += h_var[i] * h_var[i + k]
                linear_term += h_var[i]
                
        # Objective: minimize (linear_term - quad_term) 
        # As we minimize negative, we maximize overlapping shift
        # So to minimize overlap, we minimize the function
        objective = cp.Minimize(linear_term - quad_term)
        
        # Note: linear_term is constant with respect to h
        # Therefore, we minimize -quad_term, i.e., sum of h[i]*h[i+k] for shift k
        
        constraints = [integral_constraint] + bound_constraints
        
        # Solve the QP
        try:
            prob = cp.Problem(objective, constraints)
            # Use MOSEK as solver
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=0.1)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    # Renormalize
                    h = h * (n_points / 2) / np.sum(h)
        except:
            pass
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
