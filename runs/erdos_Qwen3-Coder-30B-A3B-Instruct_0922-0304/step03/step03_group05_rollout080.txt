import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with basin hopping
    n_starts = 5
    starts = [h.copy()]
    for _ in range(n_starts - 1):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        starts.append(perturbed)
    
    # Local optimization with refinement
    max_iter = 1000
    iter_count = 0
    
    for s_id, start_h in enumerate(starts):
        h_current = start_h.copy()
        for iteration in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            iter_count += 1
            
            # Evaluate current solution
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break

            # Optimize using SLSQP
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                h_current += np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Basin hopping with adaptive perturbations
    if time.time() - start_time < budget_s - 1:
        h_current = best_h.copy()
        sigma0 = 0.05
        T = 1000
        
        for iter_bh in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
				
            # Adaptive perturbation
            sigma = sigma0 * np.exp(-iter_bh / T)
            perturbation = np.random.normal(0, sigma, n_points)
            h_perturbed = h_current + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Local optimization
            result = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_candidate = result.x
            else:
                h_candidate = h_perturbed
            
            # Ensure feasibility
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate and update
            candidate_c5 = compute_c5(h_candidate)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h_candidate.copy()
                
            if best_c5 <= 0.3807:
                break
                
            h_current = h_candidate
            
    # Final refinement
    if time.time() - start_time < budget_s - 1:
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            final_h = result.x
        else:
            final_h = best_h
            
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = compute_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    return (best_h.tolist(), best_c5, n_points)
