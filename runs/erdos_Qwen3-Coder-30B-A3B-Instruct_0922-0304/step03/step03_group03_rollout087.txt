import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a more sophisticated initialization
        h = np.zeros(n_points)
        # Place two windows with different shapes
        # First window: triangular
        mid1 = n_points // 3
        width1 = n_points // 6
        for i in range(max(0, mid1 - width1), min(n_points, mid1 + width1)):
            h[i] = 1 - abs(i - mid1) / width1
        # Second window: Gaussian-like shape
        mid2 = 2 * n_points // 3
        width2 = n_points // 4
        for i in range(max(0, mid2 - width2), min(n_points, mid2 + width2)):
            h[i] = h[i] + np.exp(-0.5 * ((i - mid2) / (width2 / 2))**2) * 0.5
        # Normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with adaptive learning rate
    max_iter = 1000
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            patience = 0
        else:
            patience += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        if patience > 10:
            learning_rate *= 0.9
            patience = 0
            
        # Local search with gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Add small random perturbations occasionally to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
