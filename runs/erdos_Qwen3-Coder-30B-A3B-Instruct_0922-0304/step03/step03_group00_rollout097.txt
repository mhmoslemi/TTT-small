import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, k_max, dx, n_points):
    """Compute gradient of overlap corresponding to max overlap shift k_max"""
    grad = np.zeros(n_points)
    # For each position i, compute partial derivative of overlap w.r.t. h[i]
    # The overlap sum includes terms with h[i] and h[i+k_max]
    for i in range(n_points):
        if i + k_max < n_points:
            # h[i] appears in the overlap with h[i+k_max] 
            grad[i] += 1.0 * dx  # d/dh[i] h[i](1-h[i+k_max]) = 1*dx
        if i - k_max >= 0:
            # h[i] appears in the overlap with h[i-k_max] 
            grad[i] += 1.0 * dx  # d/dh[i] h[i](1-h[i-k_max]) = 1*dx
    
    # Negative because we want to minimize
    return -grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 2.0 * i / (mid - 1)
            h[n_points - 1 - i] = 2.0 * i / (mid - 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Accelerated gradient descent with restart
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Learning rate
    step_size = 0.01
    
    # Main optimization loop
    max_iter = 50000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

        # Find shift k that gives maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        # Convert from full correlation index to actual shift k
        n = n_points
        if max_idx < n - 1:
            k = max_idx
        else:
            k = max_idx - (n - 1)
        k = int(k)  # Ensure it's integer
        k = min(k, n - 1)

        # Compute gradient for that shift
        grad = compute_gradient(h, k, dx, n_points)
        
        # Update h with projected gradient descent
        h_new = h + step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Line search to find a suitable step size
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_candidate = h + alpha * grad
            h_candidate = project_h(h_candidate, n_points)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < current_c5:
                h_new = h_candidate
                step_size = alpha
                break
        else:
            # If no improvement, reduce step size
            step_size *= 0.9
            
        h = h_new

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
