import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    return h

        mid = n_points // 2
        half_width = n_points // 4
        for i in range(mid - half_width, mid + half_width):
            if 0 <= i < n_points:
                h[i] = 1.0 - abs(i - mid) / half_width
        h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000

            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            current_c5 = evaluate_c5(h, dx)
            
            # For each position, compute gradient
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - current_c5) / eps

            h_trial = h - step_size * grad
