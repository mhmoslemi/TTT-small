import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple initial structure
        h = np.zeros(n_points)
        h[20:50] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()

    # Optimization loop using coordinate descent-like approach
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Apply perturbation techniques
        # Try small random changes
        h_new = h + np.random.normal(0, 0.01, n_points)
        h_new = project_to_feasible(h_new)
        
        # Try another variant with periodic modulation
        h_mod = h.copy()
        for i in range(0, n_points, 10):
            if i + 5 < n_points:
                # Slight decrease in some regions
                h_mod[i:i+5] *= 0.95
                # Increase in others
                h_mod[i+5:i+10] *= 1.05
                
        h_mod = project_to_feasible(h_mod)
        
        # Evaluate candidates
        c5_new = compute_c5(h_new)
        c5_mod = compute_c5(h_mod)
        
        # Select the best variant
        if c5_new < c5_mod and c5_new < current_c5:
            h = h_new
        elif c5_mod < current_c5:
            h = h_mod
        else:
            # If no improvement, add more aggressive noise
            h += np.random.normal(0, 0.03, n_points)
            h = project_to_feasible(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
