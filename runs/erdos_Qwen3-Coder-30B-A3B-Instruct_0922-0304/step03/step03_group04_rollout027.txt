import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for multi-resolution search
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    # Loop over different resolutions
    for i, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / res
        n_points = res
        
        # Initialize h
        if i == 0:
            # Start with initial_h_values if available, otherwise a simple construction
            if 'initial_h_values' in globals() and len(initial_h_values) == res:
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (res / 2) / np.sum(h)
            else:
                # Simple initial construction
                h = np.ones(res)
                h[res//2:] = 0
                # Add small noise
                h += np.random.normal(0, 0.05, res)
                h = np.clip(h, 0, 1)
                h = h * (res / 2) / np.sum(h)
        else:
            # Up-sample previous solution
            prev_h = best_h
            h = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, len(prev_h)), prev_h)
            # Add noise
            h += np.random.normal(0, 0.02, res)
            h = np.clip(h, 0, 1)
            h = h * (res / 2) / np.sum(h)
        
        # Define bounds and constraint
        bounds = [(0, 1) for _ in range(res)]
        target_sum = res / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
            
        # Differential Evolution for global search
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=20,
            seed=seed+i,
            tol=1e-6
        )
        
        if de_result.success:
            h = de_result.x
            h = np.clip(h, 0, 1)
            h = h * (res / 2) / np.sum(h)
            
            # Local refinement with SLSQP
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (res / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break

    # Final refinement
    if best_h is not None:
        dx_final = 2.0 / n_points
        bounds_final = [(0, 1) for _ in range(n_points)]
        target_sum_final = n_points / 2

        def constraint_sum_final(h):
            return np.sum(h) - target_sum_final

        cons_final = {'type': 'eq', 'fun': constraint_sum_final}

        def objective_final(h):
            return evaluate_c5(h, dx_final)

        final_result = minimize(
            objective_final,
            best_h,
            method='SLSQP',
            bounds=bounds_final,
            constraints=cons_final,
            options={'maxiter': 200}
        )
        
        if final_result.success:
            best_h = final_result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = evaluate_c5(best_h, dx_final)
    
    if best_h is None:
        # Fallback if nothing was found
        n_points = 70
        dx_fallback = 2.0 / n_points
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        else:
            best_h = np.ones(n_points)
            best_h[n_points//2:] = 0
            best_h += np.random.normal(0, 0.05, n_points)
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, dx_fallback)
    
    return (best_h.tolist(), best_c5, n_points)
