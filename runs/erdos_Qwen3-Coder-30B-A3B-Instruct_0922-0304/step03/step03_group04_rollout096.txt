import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import pywt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Wavelet representations
    wavelet = 'haar'
    max_level = min(6, int(np.log2(n_points)))
    if max_level < 2:
        max_level = 2
        
    # Initialize wavelet coefficients
    coeffs = pywt.wavedec(h, wavelet, level=max_level)
    # Flattened coefficients array
    theta = np.concatenate(coeffs)
    
    best_c5 = float('inf')
    best_h = h.copy()
    best_theta = theta.copy()
    
    # Sparsity and step controls
    lambda_thresh = 0.1
    step_size = 0.1
    max_wavelet_iters = 20
    max_spatial_refinements = 2
    
    def soft_threshold(x, threshold):
        return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

    def update_h_from_theta(theta_flat):
        # Resize coefficients to match the original size
        coeffs_list = []
        start_idx = 0
        for i in range(max_level + 1):
            if i == 0:
                length = len(coeffs[0])
            else:
                length = len(coeffs[i])
            coeffs_list.append(theta_flat[start_idx:start_idx+length])
            start_idx += length
        # Reconstruct
        reconstructed = pywt.waverec(coeffs_list, wavelet)
        if len(reconstructed) > n_points:
            reconstructed = reconstructed[:n_points]
        elif len(reconstructed) < n_points:
            reconstructed = np.pad(reconstructed, (0, n_points - len(reconstructed)), mode='constant')
        return reconstructed
    
    def compute_gradient(h, dx):
        # Compute correlation matrix
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        # For gradient, let's use finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        return grad
    
    # Main optimization loop
    iteration = 0
    last_improvement = 0
    
    while iteration < max_wavelet_iters and time.time() - start_time < budget_s - 1:
        # Wavelet domain optimization
        for i in range(5):
            # Gradient
            grad_h = compute_gradient(h, dx)
            # Transform to wavelet domain
            grad_coeffs = pywt.wavedec(grad_h, wavelet, level=max_level)
            grad_theta = np.concatenate(grad_coeffs)
            
            # Update with thresholding
            new_theta = theta - step_size * grad_theta
            new_theta = soft_threshold(new_theta, lambda_thresh)
            
            # Update h
            h_new = update_h_from_theta(new_theta)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            current_c5 = evaluate_c5(h_new, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
                best_theta = new_theta.copy()
                last_improvement = iteration
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        iteration += 1
        if iteration - last_improvement > 10:
            lambda_thresh *= 1.5
        if best_c5 <= 0.3807:
            break
            
        # Spatial refinement
        if iteration < max_spatial_refinements * 2:
            # Select top 20% by wavelet magnitude
            top_features = np.argsort(np.abs(theta))[-int(n_points * 0.2):]
            # Gradient descent refinement
            h_refined = h.copy()
            # Small gradient step to avoid large jumps
            step_ref = 0.01
            for _ in range(30):
                grad = compute_gradient(h_refined, dx)
                h_refined = h_refined - step_ref * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
            current_c5 = evaluate_c5(h_refined, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined.copy()
                
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
