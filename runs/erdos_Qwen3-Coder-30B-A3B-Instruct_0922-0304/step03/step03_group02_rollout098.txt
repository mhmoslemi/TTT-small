import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    noise_sigma = 0.05
    elite_size = population_size // 2
    local_search_fraction = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    else:
        # Create random initialization
        h = np.ones(n_points)
        h[n_points//3:2*n_points//3] = 0
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fill population with random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    # Main genetic algorithm loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Selection using tournament
        new_population = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Select parent
            parent = population[winner_idx].copy()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Select second parent
                tournament_indices2 = np.random.choice(population_size, tournament_size, replace=False)
                tournament_fitness2 = [fitness[i] for i in tournament_indices2]
                parent2_idx = tournament_indices2[np.argmin(tournament_fitness2)]
                parent2 = population[parent2_idx].copy()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, noise_sigma, n_points)
                child += noise
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # Local search on top individuals (gradient-based)
        num_local = int(local_search_fraction * population_size)
        top_individuals = sorted_population[:num_local]
        
        for i in range(num_local):
            h = top_individuals[i].copy()
            
            # Simple gradient descent-like local search
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Get current value
                current_c5 = evaluate_c5(h, dx)
                
                # Compute gradient approximation (simple finite difference)
                grad = np.zeros(n_points)
                eps = 1e-6
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate_c5(h_plus, dx)
                    
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Gradient step
                step_size = 0.01
                h_new = h - step_size * grad
                
                # Clamp and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Accept the new solution if it improves
                new_c5 = evaluate_c5(h_new, dx)
                if new_c5 < current_c5:
                    h = h_new
                else:
                    break
            
            # Replace in population
            population[np.argmin([evaluate_c5(p, dx) for p in population])] = h

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
