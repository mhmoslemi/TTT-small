import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Default initialization
        h_init = np.ones(70)
        h_init[35:] = 0
        
    # Try different discretizations
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Tries various interval counts to balance resolution and speed
    for n_points in [20, 30, 40, 50, 60, 70]:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        h = h_init[:n_points].copy()
        h = np.clip(h, 0, 1)
        
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # For very small sizes, no optimization needed - just evaluate
        if n_points < 20:
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points
            continue
        
        # Use convex optimization approach for medium sizes
        try:
            # Create interval-based approximation
            M = n_points  # same as n_points for simplicity
            w = 2.0 / M
            
            # Variable for interval heights
            h_vars = cp.Variable(M)
            
            # Objective variable
            C = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(h_vars * w) == 1,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Compute overlap matrix for all possible shifts
            # For large M, we'll sample a few critical shifts instead of all
            max_shifts = min(2 * M - 1, 100)
            shift_indices = np.linspace(0, 2 * M - 2, max_shifts, dtype=int)
            
            for k in shift_indices:
                # Compute overlap for this shift
                total_overlap = 0
                for i in range(M):
                    shifted_i = i + k - M + 1
                    if 0 <= shifted_i < M:
                        # Compute overlap length from interval i to interval shifted_i
                        overlap_length = min(w, w)  # Simplified const overlap for now
                        total_overlap += h_vars[i] * (1 - h_vars[shifted_i]) * overlap_length
                constraints.append(total_overlap <= C)
            
            # Formulate and solve the problem
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-5, reltol=1e-5)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.array(h_vars.value).reshape(-1)
                h_opt = np.clip(h_opt, 0, 1)
                # Re-normalize
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                current_c5 = evaluate_c5(h_opt, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                    best_n_points = n_points
        except Exception as e:
            # Fall back to initial solution for this size
            pass
            
        # For very large points, check original initialization too
        if n_points >= 60:
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points

    # Final check with optimization on best found size
    if best_h is None:
        n_points = 70
        dx = 2.0 / n_points
        best_h = h_init[:n_points].copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, dx)
        best_n_points = n_points

    return (best_h.tolist(), best_c5, best_n_points)
