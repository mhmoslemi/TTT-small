import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant for given h values"""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    start_time = time.time()

    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization
    max_iter = 1000
    patience = 0
    max_patience = 50
    
    # For line search
    alpha = 0.01
    beta = 0.5
    max_backtrack = 20
    
    # Gradient computation with finite differences
    def compute_gradient(h, eps=1e-4):
        base_c5 = evaluate_c5(h, dx)
        grad = np.zeros_like(h)
        h_plus = h.copy()
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (evaluate_c5(h_plus, dx) - base_c5) / eps
            h_plus[i] = h[i]  # Reset
        return grad
    
    # Backtracking line search
    def line_search(h, grad, alpha=0.01, beta=0.5, max_iter=20):
        c5_current = evaluate_c5(h, dx)
        for i in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new <= c5_current - 0.5 * alpha * np.dot(grad, grad):
                return h_new, c5_new, alpha
            alpha *= beta
        return h, c5_current, alpha
    
    # Iterative optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Line search
        h_new, c5_new, step_size = line_search(h, grad, alpha=alpha)
        h = h_new
        
        # Projection to feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Perturbation
        if patience >= max_patience:
            noise_magnitude = 1e-3 / (1 + iteration ** 0.5)
            h += np.random.normal(0, noise_magnitude, n_points)

            h = h * (n_points / 2) / np.sum(h)

        # Update alpha
        alpha = max(alpha * 0.999, 1e-6)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h.tolist())
    
    return (best_h.tolist(), final_c5, n_points)
