import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Iterative improvement
    max_iter = 1000
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Reduce step size if no improvement
                max_move *= 0.95
                no_improvement_count = 0
        
        if best_c5 <= target_c5:
            break
            
        # Identify the shift that causes maximum overlap
        shift_size = max_overlap_idx - (n_points - 1)
        
        # Find indices contributing most to this overlap
        top_contributors = []
        for i in range(n_points):
            j = (i + shift_size) % n_points
            contrib = h[i] * (1 - h[j])
            top_contributors.append((contrib, i, j))
        
        top_contributors.sort(reverse=True)
        top_indices = top_contributors[:max(1, n_points // 20)]
        
        # Adjust the h values to reduce contributions
        for contrib, i, j in top_indices:
            if h[i] <= 0 or h[j] >= 1:
                continue
            # Compute maximum feasible adjustment
            max_decrease = min(h[i], max_move)
            max_increase = min(1 - h[j], max_move)
            epsilon = min(max_decrease, max_increase)
            if epsilon <= 0:
                continue
            # Apply adjustment
            h[i] -= epsilon
            h[j] += epsilon
            
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
