import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.1
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # Normalize positions to satisfy constraint sum(h) = n_points / 2
    for i in range(swarm_size):
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
        positions[i] = np.clip(positions[i], 0, 1)
    
    # PSO iterations
    for pso_iter in range(max_iter_pso):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate swarm
        for i in range(swarm_size):
            pos = positions[i]
            score = evaluate_c5(pos, dx)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = pos.copy()
            if score < global_best_score:
                global_best_score = score
                global_best_position = pos.copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                            social * r2 * (global_best_position - positions[i]))
            # Clip velocity to prevent large jumps
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            positions[i] += velocities[i]
            # Ensure within bounds and feasible sum
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
        # Early termination if target achieved
        if global_best_score <= 0.3807:
            break

    # SLSQP refinement using the best from PSO
    if global_best_position is not None:
        h = global_best_position.copy()
    else:
        h = best_h.copy()
    
    # Optimize with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                     options={'maxiter': 500, 'ftol': 1e-9})
    
    if result.success:
        h = result.x
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    else:
        # Use previous best if SLSQP fails
        h = best_h.copy()
    
    # Final check and refinement
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = evaluate_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
