import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        # Solve QP for this shift
        try:
            # Create variables
            h_vars = cp.Variable(n_points)
            
            # Objective function: minimize overlap at shift k
            # Sum h[i] * (1 - h[i+k]) for valid i
            objective = 0
            for i in range(n_points - abs(k)):
                if k >= 0:
                    objective += h_vars[i] * (1 - h_vars[i + k])
                else:
                    objective += h_vars[i - k] * (1 - h_vars[i])
            
            objective *= dx
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Create and solve problem
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = np.array(h_vars.value).reshape(-1)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback if QP fails
            h = h + 0.001 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
