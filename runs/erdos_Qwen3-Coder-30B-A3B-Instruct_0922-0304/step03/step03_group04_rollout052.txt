import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Ensure h is in [0,1] and sums to target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if res_idx == 0:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = project_h(h, n_points / 2)
            else:
                # Simple initial guess
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h = project_h(h, n_points / 2)
        else:
            # Up-sample previous solution
            h_prev = best_h
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_prev)), h_prev)
            h = project_h(h, n_points / 2)
            
        # Differential Evolution for global search (only on first few resolutions)
        if res_idx < 3:
            bounds = [(0, 1) for _ in range(n_points)]
            def objective(h):
                return evaluate_c5(h, dx)
            
            # Use DE for global search
            result = differential_evolution(
                objective,
                bounds,
                maxiter=50,
                popsize=20,
                seed=seed,
                tol=1e-6
            )
            if result.success:
                h = result.x
                h = project_h(h, n_points / 2)
        
        # Local optimization with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Try a few iterations of local search
        for _ in range(10):
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if result.success:
                h = result.x
                h = project_h(h, n_points / 2)
            else:
                break
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit if we achieve the target
            if best_c5 <= 0.3807:
                break
        
        # Perturb and try again if stuck (very basic restart)
        if res_idx >= 2:
            perturb_amount = 0.01
            h += np.random.normal(0, perturb_amount, n_points)
            h = project_h(h, n_points / 2)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final validation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, 2.0 / len(best_h))
        return (best_h.tolist(), final_c5, len(best_h))
    else:
        # Fallback to the last resolution with default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, n_points / 2)
        final_c5 = evaluate_c5(h, 2.0 / n_points)
        return (h.tolist(), final_c5, n_points)
