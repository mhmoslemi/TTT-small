import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fft import fft, ifft, fftfreq

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()

    def generate_latin_hypercube_samples(n_samples, n_freq):
        samples = np.zeros((n_samples, n_freq))
        for i in range(n_freq):
            # Generate samples in [0, 1] and shuffle
            samples[:, i] = np.random.permutation(np.linspace(0, 1, n_samples))
        return samples
    
    def apply_frequency_mask(h, mask_indices, noise_level=0.05):
        # Transform to frequency domain
        H = fft(h)
        # Apply mask (set masked frequencies to zero)
        H_masked = H.copy()
        H_masked[mask_indices] = 0
        # Add noise in frequency domain
        noise = np.random.normal(0, noise_level, len(H_masked)) 
        H_masked += noise * 1j
        # Transform back
        h_new = np.real(ifft(H_masked))
        # Bound to [0, 1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Spectral initialization
    H_base = fft(h)
    # Compute frequency content
    freqs = fftfreq(n_points, dx)
    # Select the frequency bands that contribute the most to large overlaps
    # For simplicity, just pick high frequencies that have large impact on correlation
    magnitude_freq = np.abs(H_base)

    peak_indices = np.where(magnitude_freq > 0.1 * np.max(magnitude_freq))[0]

        peak_indices = [0, 1, n_points//2-1, n_points//2, n_points//2+1]
    
    # Start with initial solution
    current_h = h.copy()
    current_c5 = best_c5

    exp_time = start_time + 300
    num_mutations = 0

        num_to_generate = min(50, 1000 // (num_mutations + 1) + 1)
        mutations = []

            mutations.append(h_new)

            mutation_costs.append(cost)

        sorted_indices = np.argsort(mutation_costs)

            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            simplex.append(h_perturb)
        
        # Apply scipy's Nelder-Mead with custom settings
        # Evaluate simplex
        simplex_values = [obj_fun(s) for s in simplex]
        
        while num_refinements < max_refinements and time.time() < start_time + budget_s - 10:
            # Find best and worst
            best_idx = np.argmin(simplex_values)
            worst_idx = np.argmax(simplex_values)
            # Calculate centroid (excluding worst)
            centroid = np.mean(np.array(simplex[:worst_idx] + simplex[worst_idx+1:]), axis=0)

            r = 1.5 * (centroid - simplex[worst_idx])

            if f_r < simplex_values[best_idx]:

                if f_e < f_r:
                    simplex[worst_idx] = e
                    simplex_values[worst_idx] = f_e
                else:
                    simplex[worst_idx] = x_r
                    simplex_values[worst_idx] = f_r
            else:

                c = 0.5 * (simplex[worst_idx] - centroid)
                x_c = centroid + c
                x_c = np.clip(x_c, 0, 1)
                x_c = x_c * (n_points / 2) / np.sum(x_c)
                f_c = obj_fun(x_c)
                if f_c < simplex_values[worst_idx]:
                    simplex[worst_idx] = x_c
                    simplex_values[worst_idx] = f_c
                else:
                    # Shrink toward best
                    best_point = simplex[best_idx]
                    for i in range(len(simplex)):
                        if i != best_idx:
                            simplex[i] = best_point + 0.5 * (simplex[i] - best_point)
                            # Renormalize
                            simplex[i] = np.clip(simplex[i], 0, 1)
                            simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
                            simplex_values[i] = obj_fun(simplex[i])
                            
            # Update best if available
            new_best_idx = np.argmin(simplex_values)
            if simplex_values[new_best_idx] < best_c5:
                best_c5 = simplex_values[new_best_idx]
                best_h = simplex[new_best_idx].copy()

    # Final verification
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure proper normalization and bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
