import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        # Add some noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints and bounds
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start optimization
    n_initials = 5
    initial_h_list = [h.copy()]
    for _ in range(n_initials - 1):
        h_pert = h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_h_list.append(h_pert)

    # Run optimizations from all initial points
    for i, h_init in enumerate(initial_h_list):
        h_current = h_init.copy()
        
        for iter in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                # Fallback to perturbation
                h_current += np.random.normal(0, 0.01, n_points)
            
            # Project back to feasible region
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Evaluate and update best
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            if best_c5 <= 0.3807:
                break
                
            if time.time() - start_time > budget_s - 1:
                break

    # Basin hopping: perturb best solution and optimize
    current_h = best_h.copy()
    step_size = 0.1
    cooling_rate = 1000
    max_basin_hopping = 1000
    
    for iter_bh in range(max_basin_hopping):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution
        perturbation = np.random.normal(0, step_size, n_points)
        h_perturbed = current_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local optimization
        result = minimize(
            objective, 
            h_perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = h_perturbed
            
        # Project back
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Evaluate
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Cooling
        step_size *= 0.995
        
        if time.time() - start_time > budget_s - 1:
            break

    # Final refinement at highest resolution if time allows
    if n_points == 70 and time.time() - start_time < budget_s - 5:
        # Interpolate to higher resolution
        n_high_res = 140
        dx_high = 2.0 / n_high_res
        h_high = np.interp(np.linspace(0, 2, n_high_res), np.linspace(0, 2, n_points), best_h)
        h_high = h_high * (n_high_res / 2) / np.sum(h_high)
        
        # Optimize high resolution
        def compute_c5_high(h):
            overlap = correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx_high

        def objective_high(h):
            return compute_c5_high(h)
            
        bounds_high = [(0, 1) for _ in range(n_high_res)]
        constraints_high = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_high_res / 2}]
        
        result = minimize(
            objective_high, 
            h_high, 
            method='SLSQP', 
            bounds=bounds_high, 
            constraints=constraints_high,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = h_high
            
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_high_res / 2) / np.sum(h_final)
        
        c5_final = compute_c5_high(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            n_points = n_high_res

    return (best_h.tolist(), best_c5, n_points)
