import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, dx):
    """Compute gradient of overlap integral w.r.t. h for shift k"""
    n = len(h)
    grad = np.zeros(n)
    # h[i] appears in term h[i](1-h[i+k]) for i = 0..n-1-k
    for i in range(n - abs(k)):
        if k >= 0:
            grad[i] += 1 - h[i + k]
        else:
            grad[i + (-k)] += 1 - h[i]
    return grad * dx

def line_search(h, grad, max_iter=10):
    """Perform line search to find a good step size"""
    step = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    
    for _ in range(max_iter):
        h_new = h - step * grad
        h_new = project_h(h_new, len(h))
        c5 = evaluate_c5(h_new, 1.0/len(h))  # dx = 1/n in this function for simplicity
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        step *= 0.5
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h

    else:
        # Simple initialization
        h = np.zeros(n_points)
        h[20:50] = 1.0
    
    h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with adaptive step
    max_iter = 50000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find shift that maximizes overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (len(h) - 1)  # Convert index to shift
        
        # Compute gradient
        grad = compute_gradient(h, k, dx)
        
        # Line search
        h_new, new_c5 = line_search(h, grad)
        
        h = h_new
        
        # Project to feasible region
        h = project_h(h, n_points)
        
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
