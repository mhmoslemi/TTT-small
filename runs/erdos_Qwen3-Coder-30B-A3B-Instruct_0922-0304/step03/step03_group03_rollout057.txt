import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum
from scipy.linalg import eigh

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try SDP relaxation approach
        try:
            # Create variable matrix to represent h
            h_vars = Variable(n_points)
            t = Variable()
            
            # Constraints
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp_sum(h_vars) == n_points / 2,
                t >= 0
            ]
            
            # Generate overlap constraints for all lags
            max_lag = n_points - 1
            for k in range(-max_lag, max_lag + 1):
                # Positive lags
                if k >= 0:
                    # Create constraint on overlap for lag k
                    if k == 0:
                        # This is not a lag but the integral directly
                        overlap = cp_sum(h_vars)
                        constraints.append(overlap <= t)
                    else:
                        # Align h[i] and h[i+k] to compute h[i]*(1 - h[i+k])
                        total_overlap = 0
                        for i in range(n_points - k):
                            product = h_vars[i] * (1 - h_vars[i + k])
                            total_overlap += product
                        constraints.append(total_overlap <= t)
                
                # Negative lags (symmetric)
                elif k < 0:
                    kabs = -k
                    # Align h[i] and h[i-kabs]
                    total_overlap = 0
                    for i in range(n_points - kabs):
                        product = h_vars[i] * (1 - h_vars[i + kabs])
                        total_overlap += product
                    constraints.append(total_overlap <= t)
            
            # Solve the problem
            prob = Problem(Minimize(t), constraints)
            prob.solve(solver=cp.GLOP, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to local refinement
                _local_refinement(h, dx, start_time, budget_s)
        except:
            # Fallback to local refinement
            _local_refinement(h, dx, start_time, budget_s)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def _local_refinement(h, dx, start_time, budget_s):
    """Local refinement with projected gradient."""
    step = 0.01
    
    for iter_step in range(100):
        if time.time() - start_time > budget_s - 1:
            return
            
        # Evaluate overlap
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        max_idx = np.argmax(corr)
        
        # Approximate gradient via finite difference
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (len(h) / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full") 
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
            
        # Gradient descent step
        h -= step * grad
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
        
        # Reduce step size
        step *= 0.999
