import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        swarm.append(h0.copy())
        velocities.append(np.zeros(n_points))
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        swarm.append(h0.copy())
        velocities.append(np.zeros(n_points))
    
    # Add additional particles with triangular initialization
    for i in range(n_particles - 1):
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))

    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track best solutions
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_c5 = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_c5 = gbest_c5
    best_h = gbest.copy()
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        # Update each particle
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Compute fitness
            current_c5 = evaluate_c5(h, dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = current_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            
            # Update global best
            if current_c5 < gbest_c5:
                gbest = h.copy()
                gbest_c5 = current_c5
            
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            v = (w * v + 
                 c_p * r_p * (pbest[i] - h) + 
                 c_g * r_g * (gbest - h))
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            
            # Maintain sum constraint
            h = h * (n_points / 2) / np.sum(h)
            
            # Update swarm
            swarm[i] = h.copy()
            velocities[i] = v.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
