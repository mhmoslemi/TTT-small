import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a triangular-like initial function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 2 * i / (mid - 1)
            h[n_points - 1 - i] = 2 * i / (mid - 1)
        h[mid] = 1.0
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute shift indices
    shift_indices = list(range(-n_points + 1, n_points))
    
    # Cutting plane approach with LP
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        worst_shift = max_idx - (n_points - 1)
        
        # Create optimization problem using CVXPY
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Add the current worst shift constraint
        if worst_shift >= 0:
            # h[i] * (1 - h[i + shift]) term:
            # Note: this uses a linear approximation
            x = np.zeros(n_points)
            for i in range(n_points - worst_shift):
                x[i] += h_vars[i] * (1 - h_vars[i + worst_shift])
            constraints.append(cp.sum(x) <= c5_var * dx)
        else:
            # Shift is negative, so we need to handle it opposite direction
            shift_abs = -worst_shift
            x = np.zeros(n_points)
            for i in range(n_points - shift_abs):
                x[i + shift_abs] += h_vars[i] * (1 - h_vars[i + shift_abs])
            constraints.append(cp.sum(x) <= c5_var * dx)
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_vars.value
                # Clip and normalize
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # Fallback to gradient descent
                step = 0.005
                iterations = 50
                h_temp = h.copy()
                
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    
                h = h_temp
        except Exception:
            # Fallback to simple gradient descent
            step = 0.005
            iterations = 50
            h_temp = h.copy()
            
            for _ in range(iterations):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
