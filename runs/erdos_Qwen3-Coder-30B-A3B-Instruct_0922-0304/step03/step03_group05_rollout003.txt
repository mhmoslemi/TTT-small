import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initializations
    initial_hs = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        # Interpolate to new resolution
        if len(parent_h) != n_points:
            x_orig = np.linspace(0, 2, len(parent_h))
            x_new = np.linspace(0, 2, n_points)
            parent_h = np.interp(x_new, x_orig, parent_h)
        # Ensure feasibility
        parent_h = np.clip(parent_h, 0, 1)
        parent_h = parent_h * (n_points / 2) / np.sum(parent_h)
        initial_hs.append(parent_h.copy())
    else:
        # Fallback: triangular construction
        h = np.zeros(n_points)
        center = n_points // 2
        width = n_points // 4
        for i in range(max(0, center - width), min(n_points, center + width)):
            h[i] = 1.0 - abs(i - center) / width
        h = h * (n_points / 2) / np.sum(h)
        initial_hs.append(h.copy())
    
    # Random perturbations
    for _ in range(4):
        h = initial_hs[0].copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_hs.append(h.copy())
    
    def compute_c5(h):
        # Compute all overlaps using correlation
        overlap = correlate(h, 1 - h, mode="full")
        # The full correlation has 2*n_points - 1 elements
        # We only need the overlapping parts (excluding the zero-padding)
        # The maximum overlap is at the middle (n_points - 1)
        # The relevant overlaps are from index n_points - 1 to 2 * n_points - 2
        relevant = overlap[n_points-1:]
        return np.max(relevant) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    best_c5 = float('inf')
    best_h = None

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Multi-start optimization
    max_iter = 1000
    iter_count = 0
    no_improvement_count = 0
    
    # Evaluate all initial candidates
    for h in initial_hs:
        h = project_to_feasible(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Continue optimization loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Select a random candidate for perturbation
        h_idx = np.random.randint(0, len(initial_hs))
        h = initial_hs[h_idx].copy()
        
        # Add some noise
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h)
        
        # Local optimization
        result = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = h
            
        # Ensure feasibility after optimization
        h_opt = project_to_feasible(h_opt)
        c5_opt = compute_c5(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # If no improvement in a while, make a larger perturbation
        if no_improvement_count > 50:
            h = best_h.copy()
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = project_to_feasible(h)
            no_improvement_count = 0
            
        iter_count += 1
        
    # Final evaluation
    if best_h is None:
        # Just in case, if no valid best_h found, use a good fallback
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h)
        best_c5 = compute_c5(h)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
