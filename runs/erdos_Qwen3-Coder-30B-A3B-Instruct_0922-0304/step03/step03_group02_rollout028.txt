import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # number of particles
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize swarm
    particles = np.array([h0 + np.random.normal(0, 0.01, n_points) for _ in range(m)])
    particles = np.clip(particles, 0, 1)
    for i in range(m):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = np.zeros_like(particles)
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Track personal bests
    pbest = particles.copy()
    pbest_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Track global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_c5 = gbest_c5
    best_h = gbest.copy()
    
    # PSO iterations
    max_iter = int(budget_s * 20)  # ~20 iters/sec
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints: clip to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to maintain sum(h) = n_points/2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest_c5[i] = current_c5
                pbest[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest_c5 = current_c5
                    gbest = particles[i].copy()
        
        # Early stopping
        if gbest_c5 <= 0.3807:
            break
            
        # Update best overall solution
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
