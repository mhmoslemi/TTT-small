import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import pywt
from functools import partial

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    """Project h to satisfy the sum constraint."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet-based optimization
    # Use Haar wavelets (sufficient for our resolution)
    wavelet = 'haar'
    level = min(6, int(np.log2(n_points)) - 1)
    
    # Create wavelet basis
    coeffs = pywt.wavedec(h, wavelet, level=level)
    wavelet_coeffs = pywt.vector_to_array(coeffs)
    basis_size = len(wavelet_coeffs)
    
    # Thresholding parameters
    threshold = 0.1
    learning_rate = 0.1
    
    wavelet_iter = 0
    max_wavelet_iter = 30
    spatial_iter = 0
    max_spatial_iter = 2
    
    # Precompute filtered versions for gradient calculation
    def compute_gradient(h_vec):
        # Compute correlation gradient with finite differences
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Approximate gradient with finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        return grad
    
    # Main loop
    while time.time() - start_time < budget_s - 1 and (wavelet_iter < max_wavelet_iter or spatial_iter < max_spatial_iter):
        # Wavelet domain optimization
        if wavelet_iter < max_wavelet_iter:
            coeffs = pywt.wavedec(h, wavelet, level=level)
            wavelet_coeffs = pywt.vector_to_array(coeffs)
            
            # Estimate gradient in wavelet domain
            grad = np.zeros_like(wavelet_coeffs)
            eps = 1e-5
            for i in range(len(wavelet_coeffs)):
                coeffs_plus = wavelet_coeffs.copy()
                coeffs_plus[i] += eps
                h_plus = pywt.array_to_vector(coeffs_plus, coeffs)
                h_plus = project_to_feasible(h_plus, target_sum)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - evaluate_c5(h, dx)) / eps
            
            # Soft thresholding
            wavelet_coeffs = wavelet_coeffs - learning_rate * grad
            wavelet_coeffs[np.abs(wavelet_coeffs) < threshold] = 0
            
            # Convert back to spatial domain
            h = pywt.array_to_vector(wavelet_coeffs, coeffs)
            h = project_to_feasible(h, target_sum)
            
            wavelet_iter += 1
            
        # Spatial refinement (every few wavelet iterations)
        if spatial_iter < max_spatial_iter and wavelet_iter % 5 == 0:
            # Use gradient descent on the spatial domain
            def objective(h_vec):
                return evaluate_c5(h_vec, dx)
                
            # Optimize using L-BFGS
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            
            result = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
            
            spatial_iter += 1

        # Optimization check
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive threshold change
        if wavelet_iter % 10 == 0 and wavelet_iter > 0 and threshold > 0.01:
            threshold *= 1.1  # Increase threshold to encourage sparsity 
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
