import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # Constraint: sum(h) = n_points / 2
    def project_to_constraint(h):
        return h * (n_points / 2) / np.sum(h)
    
    # PSO main loop
    pso_budget = budget_s * 0.8
    for iter_num in range(max_iter):
        if time.time() - start_time > pso_budget:
            break
            
        for i in range(swarm_size):
            # Apply constraints
            h = positions[i]
            h = np.clip(h, 0, 1)
            h = project_to_constraint(h)
            positions[i] = h
            
            # Evaluate
            score = evaluate_c5(h, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = h.copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (personal_best_positions[i] - positions[i]) + 
                            social * r2 * (global_best_position - positions[i]))
            
            positions[i] += velocities[i]
    
    # Refine with SLSQP
    if global_best_position is not None:
        h_refine = global_best_position.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        try:
            result = minimize(objective, h_refine, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 500, 'ftol': 1e-9})
            if result.success:
                h_refine = result.x
                final_c5 = evaluate_c5(h_refine, dx)
                if final_c5 < global_best_score:
                    global_best_score = final_c5
                    global_best_position = h_refine.copy()
        except:
            pass
    
    # Final check
    if global_best_position is not None:
        final_h = global_best_position.copy()
        final_c5 = global_best_score
        return (final_h.tolist(), final_c5, n_points)
    
    # Fallback to initial if nothing better
    return (h.tolist(), evaluate_c5(h, dx), n_points)
