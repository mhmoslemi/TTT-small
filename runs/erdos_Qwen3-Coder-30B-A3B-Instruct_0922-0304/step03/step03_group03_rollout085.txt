import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp


def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Start from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Simple gradient descent approach:
        # Approximate gradient using finite differences
        gradient = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus) * dx
            
            # Compute gradient estimate
            gradient[i] = (c5_plus - current_c5) / eps
        
        # Update with gradient descent
        learning_rate = 0.01
        h = h - learning_rate * gradient
        
        # Project to feasible set 
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally add small random perturbations
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation 
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
