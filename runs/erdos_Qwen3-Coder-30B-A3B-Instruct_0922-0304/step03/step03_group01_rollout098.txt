import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_erdos_solution(h):
    """Evaluate solution using the provided helper"""
    # Compute correlation between h and (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value of correlation times dx gives the C5 bound
    c5 = np.max(corr) * (2.0 / len(h))
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two pulses
        h = np.zeros(n_points)
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        width = n_points // 10
        for i in range(max(0, mid1 - width), min(n_points, mid1 + width)):
            h[i] = 0.5
        for i in range(max(0, mid2 - width), min(n_points, mid2 + width)):
            h[i] = 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h sums to exactly n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation using finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Backtracking line search for step size
    def line_search(h, grad, max_iter=20):
        alpha = 0.01
        c1 = 1e-4
        rho = 0.5
        current_c5 = evaluate_c5(h, dx)
        
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 <= current_c5 - c1 * alpha * np.dot(grad, grad):
                return alpha, h_new
            alpha *= rho
        return alpha, h_new
    
    iter_no_improvement = 0
    max_no_improvement = 50
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        if best_c5 <= 0.3807:
            break
            
        grad = compute_gradient(h)
        alpha, h_new = line_search(h, grad)
        h = h_new
        
        # Perturbation to escape local minima
        if iter_no_improvement > max_no_improvement:
            noise_magnitude = 1e-3 * (1.0 / (1.0 + iteration**0.5))
            noise = np.random.normal(0, noise_magnitude, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            iter_no_improvement = 0
            
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
