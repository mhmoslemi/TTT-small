import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[20:50] = 0.5  # Two pulses
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation parameters
    eps = 1e-4
    
    # Line search parameters
    eta_initial = 0.01
    backtracking_factor = 0.5
    max_line_search = 20
    
    # Perturbation parameters
    perturbation_freq = 50
    perturbation_magnitude = 1e-3
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        h_base = h.copy()
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h_base.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Line search for optimal step size
        eta = eta_initial
        h_trial = h - eta * grad
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        c5_trial = evaluate_c5(h_trial, dx)
        
        line_search_iter = 0
        while c5_trial >= current_c5 and line_search_iter < max_line_search:
            eta *= backtracking_factor
            h_trial = h - eta * grad
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_c5(h_trial, dx)
            line_search_iter += 1
            
        h = h_trial
        
        # Periodic perturbations
        if no_improvement_count >= perturbation_freq:
            noise_magnitude = perturbation_magnitude / (1.0 + iteration ** 0.5)
            h += np.random.normal(0, noise_magnitude, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
