import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(swarm_size):
        # Initialize particle position
        pos = h.copy()
        # Add small random perturbations to improve diversity
        pos += np.random.normal(0, 0.02, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(pos.copy())
        pbest_values.append(evaluate_c5(pos, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest_positions[i] - particles[i]) +
                           social * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest_position = particles[i].copy()
                
        # Early exit if target achieved
        if gbest_value <= 0.3807:
            break
            
    # Store best from PSO
    best_h = gbest_position.copy()
    best_c5 = gbest_value
    
    # Local refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Refine with SLSQP
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                     options={'maxiter': 500, 'ftol': 1e-9})
    
    if result.success:
        h_final = result.x
        c5_final = objective(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final constraint enforcement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation with the accurate method
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
