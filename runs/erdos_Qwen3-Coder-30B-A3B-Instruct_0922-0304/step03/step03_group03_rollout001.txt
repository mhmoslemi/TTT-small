import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Dual approach: build and solve LP for the constraint matrix
        # Try to solve a better LP problem with constraints
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp_sum(h_vars) == n_points / 2,
            t >= 0
        ]
        
        # Add a small regularizer to make problem more stable
        # Try to minimize a relaxed order of the objective
        # Introduce dual relaxation for the overlap
        # Simplified version: use LP relaxation of overlap constraints
        # Based on max_k ∫ h(x)(1-h(x+k)) dx, we can bound it with linear program
        # But instead, use a projected gradient descent with alternating steps
        
        # Do not use the full LP formulation as it's computationally expensive
        # Instead, do efficient local search
        
        # Local refinement step using projected gradient with adaptive learning rate
        step = 0.01
        for _ in range(20):
            curr_c5 = evaluate_c5(h, dx)
            grad = np.zeros_like(h)
            eps = 1e-5
            
            # Approximate gradient with finite differences
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - curr_c5) / eps
            
            # Apply gradient step
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Adaptive step size update
            if (iteration + iteration % 10) > 0:
                step = step * 0.995
                
        # Re-project to satisfy constraints, even after gradient step
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
