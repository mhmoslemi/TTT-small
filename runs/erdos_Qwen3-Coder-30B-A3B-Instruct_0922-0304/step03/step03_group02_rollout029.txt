import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = int(budget_s * 20)  # Roughly 20 iterations per second
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Initialize particles and velocities
    particles = np.tile(h_init, (n_particles, 1))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Apply constraints to initial particles
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Initialize global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
                    
            # Early exit if target achieved
            if gbest_score <= 0.3807:
                break
    
    return (gbest.tolist(), gbest_score, n_points)
