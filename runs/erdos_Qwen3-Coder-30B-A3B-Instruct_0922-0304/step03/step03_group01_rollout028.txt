import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO Parameters
    swarm_size = 50
    max_iter_pso = 200
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) - 0.5
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # Normalize initial positions to satisfy constraint
    for i in range(swarm_size):
        positions[i] *= (n_points / 2) / np.sum(positions[i])
        
    # Set initial personal best
    for i in range(swarm_size):
        score = evaluate_c5(positions[i], dx)
        if score < personal_best_scores[i]:
            personal_best_scores[i] = score
            personal_best_positions[i] = positions[i].copy()
            if score < global_best_score:
                global_best_score = score
                global_best_position = positions[i].copy()
    
    # PSO iterations
    for iteration in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocities
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best_positions[i] - positions[i]) + 
                           social * r2 * (global_best_position - positions[i]))
            
            # Update positions
            positions[i] += velocities[i]
            
            # Apply bounds and constraint
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate fitness
            score = evaluate_c5(positions[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = positions[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = positions[i].copy()
                    
            # Early exit if target achieved
            if global_best_score <= 0.3807:
                break
                
        # Explicit early exit if we've found acceptable solution
        if global_best_score <= 0.3807:
            break
    
    # SLSQP Refinement
    max_iter_slsqp = 1000
    # Use the best PSO solution as starting point
    initial_pos = global_best_position.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimization with SLSQP
    try:
        result = minimize(objective, initial_pos, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            h = result.x
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Fallback to the best from PSO
            best_c5 = global_best_score
            best_h = global_best_position.copy()
    except:
        # If SLSQP fails, use PSO solution
        best_c5 = global_best_score
        best_h = global_best_position.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Make sure we return the best solution
    best_c5 = min(final_c5, best_c5)
    
    return (best_h.tolist(), best_c5, n_points)
