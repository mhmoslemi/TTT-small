import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Fallback initialization
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize swarm with diverse starting points
    swarm = [h0.copy()]
    velocities = [np.zeros(n_points)]
    
    # Add more particles with triangular functions
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        center = n_points // 2
        for j in range(n_points):
            h[j] = max(0, 1 - 2 * abs(j - center) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal and global bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Main loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * it / max_iter
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c_p * r1 * (pbest[i] - swarm[i]) +
                c_g * r2 * (gbest - swarm[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Apply sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate and update personal best
            score = evaluate_c5(swarm[i], dx)
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = swarm[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = swarm[i].copy()
        
        # Early termination
        if gbest_score <= 0.3807:
            break
    
    # Return best solution
    return (gbest.tolist(), gbest_score, n_points)
