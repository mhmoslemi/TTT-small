import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: clip to [0,1] and normalize to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient for the shift k that maximizes overlap"""
    n = len(h)
    grad = np.zeros(n)
    # The overlap contribution from shift k is sum_{i=0}^{n-|k|-1} h[i](1-h[i+k])
    # So grad[i] = (1 - h[i+k]) for i < n-k if k > 0, or (1 - h[i+k]) for i+k < n if k < 0
    # Also, for each i, h[i] appears in overlap terms where it's in a position with another h term
    
    if k_max >= 0:
        for i in range(n - k_max):
            grad[i] += (1 - h[i + k_max])
            grad[i + k_max] -= h[i]
    else:
        for i in range(-k_max, n):
            grad[i] += (1 - h[i + k_max])
            grad[i + k_max] -= h[i]
            
    return grad * dx

def line_search(h, grad, dx, max_iter=10):
    """Perform line search to find good step size"""
    step = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    
    for _ in range(max_iter):
        h_new = h - step * grad
        h_new = project_h(h_new, len(h))
        overlap = correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(overlap) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        step *= 0.5
    return best_h, best_c5

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = i / mid
            else:
                h[i] = 1 - (i - mid) / (n_points - mid)
        h = project_h(h, n_points)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the shift k causing maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        # Convert index to shift value
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, dx, k_max)
        
        # Line search to find good step
        h_new, c5_new = line_search(h, grad, dx, max_iter=5)
        
        h = h_new
        current_c5 = c5_new
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
