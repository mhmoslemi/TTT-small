import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach with decreasing budget allocation
    resolutions = [10, 20, 40, 70, 100, 140, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    # Budget allocation: each resolution gets ~10% of total budget
    # but we start from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(10) * 0.5
    
    # Initialize with coarse resolution
    for i, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # For resolution 10, start with uniform distribution
        if i == 0:
            h = np.ones(n_points) * 0.5
        else:
            # Interpolate previous solution to new resolution
            h_old = h.copy()
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_old)), h_old)
            
        # Now refine with a hybrid approach
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Use differential evolution for global search on first few steps
        if i < 3:
            de_result = differential_evolution(
                objective,
                bounds,
                seed=seed+i,
                maxiter=50,
                popsize=20
            )
            if de_result.success:
                h = de_result.x.copy()
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Apply SLSQP refinement
        max_iter = 100
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
            # Early stop
            if best_c5 <= 0.3807:
                break
        
        if n_points >= 70:
            # Add a small perturbation to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, 2.0 / len(best_h))
        return (best_h.tolist(), final_c5, len(best_h))
    else:
        # Fallback
        return (np.ones(70) * 0.5 * 2/70, 0.5, 70)
