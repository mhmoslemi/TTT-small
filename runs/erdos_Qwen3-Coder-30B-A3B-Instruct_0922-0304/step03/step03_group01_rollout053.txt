import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: Smooth step function
        h = np.ones(n_points)
        h[:n_points//3] = 0.5
        h[n_points//3:2*n_points//3] = 1.0
        h[2*n_points//3:] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 200
    max_iter_slsqp = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # PSO initialization
    positions = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        pos = h.copy() + np.random.normal(0, 0.05, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        positions.append(pos)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best_positions.append(pos.copy())
        personal_best_scores.append(float('inf'))

    global_best_position = h.copy()
    global_best_score = float('inf')
    
    # PSO optimization phase
    for pso_iter in range(max_iter_pso):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        for i in range(swarm_size):
            pos = positions[i]
            vel = velocities[i]
            
            # Evaluate current position
            score = evaluate_c5(pos, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = pos.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = pos.copy()
                if global_best_score < best_c5:
                    best_c5 = global_best_score
                    best_h = pos.copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            vel = (inertia * vel + 
                   cognitive * r1 * (personal_best_positions[i] - pos) +
                   social * r2 * (global_best_position - pos))
            pos += vel
            
            # Enforce constraints
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
            
            positions[i] = pos
            velocities[i] = vel
            
        # Early exit
        if global_best_score <= 0.3807:
            break
    
    # SLSQP refinement phase
    if time.time() - start_time < 0.2 * budget_s:
        # Use best from PSO as starting point for SLSQP
        h = global_best_position.copy()
        
        # Define bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': max_iter_slsqp, 'ftol': 1e-9})
            
            if result.success:
                h = result.x
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            pass
    
    # Final evaluation with exact C5
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
