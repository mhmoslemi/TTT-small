import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 100
    max_iter_slsqp = 200
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    
    # Clip and normalize positions to satisfy constraints
    for i in range(swarm_size):
        positions[i] = np.clip(positions[i], 0, 1)
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    personal_best_positions = positions.copy()
    personal_best_fitness = np.array([evaluate_c5(p, dx) for p in positions])
    
    # Find global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    pso_iter = 0
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        pso_iter += 1
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                             cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                             social * r2 * (global_best_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply bounds
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate fitness
            current_fitness = evaluate_c5(positions[i], dx)
            
            # Update personal best
            if current_fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = current_fitness
                personal_best_positions[i] = positions[i].copy()
                
                # Update global best
                if current_fitness < global_best_fitness:
                    global_best_fitness = current_fitness
                    global_best_position = positions[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        # Early stopping condition
        if stagnation_count >= max_stagnation:
            break
            
    # Refinement with SLSQP
    refine_h = global_best_position.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    for iteration in range(max_iter_slsqp):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, refine_h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 50, 'ftol': 1e-10}, callback=None)
        
        if result.success:
            refine_h = result.x
            current_c5 = objective(refine_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = refine_h.copy()
                
            if best_c5 <= 0.3807:
                break
        else:
            break
            
        # Reinitialize with current best for next iteration
        refine_h = best_h.copy()
        
    # Final refinement if needed
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
