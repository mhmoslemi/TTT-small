import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is within bounds and sum constraint is met"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    h_mut = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Either Gaussian perturbation or random reset
            if np.random.random() < 0.5:
                h_mut[i] += np.random.normal(0, 0.05)
            else:
                h_mut[i] = np.random.random()
    return repair_h(h_mut, n_points)

def crossover_individuals(h1, h2, crossover_rate):
    """Perform crossover between two individuals"""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    point = np.random.randint(1, len(h1)-1)
    h1_child = np.concatenate([h1[:point], h2[point:]])
    h2_child = np.concatenate([h2[:point], h1[point:]])
    return h1_child, h2_child

def genetic_algorithm(seed, budget_s, n_points):
    np.random.seed(seed)
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
    else:
        h_base = np.ones(n_points)
        h_base[n_points//2:] = 0
        h_base += np.random.normal(0, 0.05, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    for _ in range(pop_size):
        h = mutate_individual(h_base, 0.1, n_points)
        population.append(h)
    
    # GA parameters
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_size = int(pop_size * 0.2)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s * 0.8 and no_improvement < 50:
        # Evaluate fitness (negative because we minimize)
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual, dx)
            fitness.append(-c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        current_best = -fitness[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.randint(0, pop_size//2)
            idx2 = np.random.randint(0, pop_size//2)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            child1, child2 = crossover_individuals(parent1, parent2, crossover_rate)
            
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    start_time = time.time()
    
    # Run GA
    best_h, best_c5 = genetic_algorithm(seed, budget_s, n_points)
    
    # Local refinement
    dx = 2.0 / n_points
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Refine with gradient descent
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        grad = np.zeros_like(best_h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            if abs(c5_plus - max_overlap) > 1e-10:
                grad[i] = (c5_plus - max_overlap) / eps
            else:
                grad[i] = 0
        
        step = 0.01
        best_h -= step * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            break
            
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
