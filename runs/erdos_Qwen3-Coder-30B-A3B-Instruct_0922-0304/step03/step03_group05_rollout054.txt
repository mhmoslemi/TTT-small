from scipy.signal import correlate
from scipy.optimize import minimize
import time

    start_time = time.time()

    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

        h_parent = np.clip(h_parent, 0, 1)

        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multiple initializations
    initial_hs = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        # Interpolate
        h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        initial_hs.append(h_interp.copy())
    
    # Random perturbations
    for _ in range(5):
        h_pert = h.copy()
        h_pert += np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_hs.append(h_pert.copy())
    
    # Fallback if no parent available
    if not initial_hs:
        h_fallback = np.ones(n_points)
        h_fallback[n_points//2:] = 0
        h_fallback += np.random.normal(0, 0.05, n_points)
        h_fallback = np.clip(h_fallback, 0, 1)
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        initial_hs.append(h_fallback.copy())

    # Optimization loop
    max_iter = 1000
    no_improve_count = 0
    last_best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Test all initializations
        for h_init in initial_hs:
            h_current = h_init.copy()
            
            # Refine with optimization
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
                # Ensure feasibility
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                current_c5 = compute_c5(h_opt)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                    no_improve_count = 0
            else:
                # If optimization fails, keep initial
                current_c5 = compute_c5(h_current)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
                    no_improve_count = 0

                break

        if abs(last_best_c5 - best_c5) < 1e-7:
            no_improve_count += 1
            if no_improve_count > 100:
                break
        else:
            no_improve_count = 0
        last_best_c5 = best_c5
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
