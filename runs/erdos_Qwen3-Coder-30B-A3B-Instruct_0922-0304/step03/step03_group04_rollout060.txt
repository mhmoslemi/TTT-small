import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 30  # number of intervals
    w = 2.0 / M  # width of each interval
    dx = w
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Average initial_h_values into intervals
        h_init = h_init.reshape(M, -1).mean(axis=1)
        h = np.clip(h_init, 0, 1)
        h = h * (M / 2) / np.sum(h)
    else:
        h = np.ones(M) * 0.5
        h = h * (M / 2) / np.sum(h)
    
    # Precompute overlap matrix
    # For each shift, compute overlap lengths
    A = np.zeros((2*M - 1, M))  # shift index, interval index
    for shift_idx in range(2*M - 1):
        for i in range(M):
            # Compute overlap with shifted interval
            shift = shift_idx - (M - 1)
            start_i = i * w
            end_i = start_i + w
            start_shifted = (i + shift) * w
            end_shifted = start_shifted + w
            
            if start_shifted >= end_i or end_shifted <= start_i:
                overlap = 0
            else:
                overlap = min(end_i, end_shifted) - max(start_i, start_shifted)
            A[shift_idx, i] = overlap

    # Prepare variables
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    # Constraint for integral = 1
    integral_constraint = cp.Equality(cp.sum(cp.multiply(h_vars, w)), 1.0)
    
    # Constraints for C >= sum of overlaps for all k
    constraints = [cp.GreaterThan(C)]
    for shift_idx in range(2*M - 1):
        # For each shift k, compute sum(h_i * (1 - h_{i+k}) * overlap)
        # This is a bit tricky due to indexing, let's precompute
        # Construct the mask for valid interval shifts
        left_indices = []
        right_indices = []
        for i in range(M):
            shifted_i = i + shift_idx - (M - 1)
            if 0 <= shifted_i < M:
                left_indices.append(i)
                right_indices.append(shifted_i)
            elif shifted_i < 0:
                # This gets handled via zero padding for now
                left_indices.append(i)
                right_indices.append(-1)  # Indicates zero padding
        
        # The overlap constraints
        if len(left_indices) > 0:
            # Accumulate the product
            expression = 0
            for l, r in zip(left_indices, right_indices):
                if r >= 0:
                    expression += A[shift_idx, l] * h_vars[l] * (1 - h_vars[r])
                else:
                    expression += A[shift_idx, l] * h_vars[l]  # implicitly h[r] = 0
            # Add constraint: expression <= C
            constraints.append(expression <= C)
    
    # Bounds on h
    bounds = [cp.GreaterThan(h_vars), cp.LessThan(h_vars)]
    
    # Combine constraints
    all_constraints = constraints + bounds
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, all_constraints)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try to solve using different solvers
    solvers = ['ECOS', 'MOSEK']
    found_feasible = False
    
    for solver_name in solvers:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            prob.solve(solver=getattr(cp, solver_name), verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                found_feasible = True
                h_opt = h_vars.value
                if h_opt is not None:
                    h_opt = np.clip(h_opt, 0, 1)
                    h_opt = h_opt * (M / 2) / np.sum(h_opt)
                    
                    # Compute C5 using original evaluation
                    overlap = correlate(h_opt, 1 - h_opt, mode="full")
                    max_overlap = np.max(overlap) * dx
                    if max_overlap < best_c5:
                        best_c5 = max_overlap
                        best_h = h_opt.copy()
                        
                    if best_c5 <= 0.3807:
                        break
        except Exception as e:
            continue  # If solver fails, try next one
    
    # Fallback to numerical optimization if LP fails
    if not found_feasible:
        h = best_h
        bounds = [(0, 1) for _ in range(M)]
        target_sum = M / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        max_iter = 100
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                # Ensure constraints are met
                h = np.clip(h, 0, 1)
                h = h * (M / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
            else:
                break
    
    return (best_h.tolist(), best_c5, M)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
