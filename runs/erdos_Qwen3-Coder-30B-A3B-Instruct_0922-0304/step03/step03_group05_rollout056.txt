import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start with Basin Hopping
    n_starts = 5
    starts = [h.copy()]
    for _ in range(n_starts - 1):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        starts.append(perturbed)

    max_iter = 1000
    iter_count = 0
    
    for start in starts:
        h_current = start.copy()
        current_c5 = compute_c5(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()

        # Basin Hopping
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            iter_count += 1
            
            # Local optimization
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                # Fallback to small perturbation
                h_current += np.random.normal(0, 0.01, n_points)
                
            # Ensure feasibility
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Evaluate and update best
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
            # Basin hopping perturbation with exponential decay
            if iter_count % 100 == 0:
                sigma = 0.05 * np.exp(-iter_count / 1000)
                h_current += np.random.normal(0, sigma, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Try to refine with higher resolution if time allows
    if time.time() - start_time < budget_s - 5 and n_points == 70:
        # Interpolate to higher resolution
        n_high = 140
        dx_high = 2.0 / n_high
        h_high = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, 70), best_h)
        h_high = h_high * (n_high / 2) / np.sum(h_high)
        
        # Local optimization on high resolution
        def compute_c5_high(h):
            overlap = correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx_high

        def objective_high(h):
            return compute_c5_high(h)

        def constraint_sum_high(h):
            return np.sum(h) - n_high / 2

        constraints_high = [{'type': 'eq', 'fun': constraint_sum_high}]
        bounds_high = [(0, 1) for _ in range(n_high)]

        result = minimize(
            objective_high,
            h_high,
            method='SLSQP',
            bounds=bounds_high,
            constraints=constraints_high,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_high = result.x
        h_high = np.clip(h_high, 0, 1)
        h_high = h_high * (n_high / 2) / np.sum(h_high)
        
        c5_high = compute_c5_high(h_high)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high
            n_points = n_high

    # Final evaluation with tighter tolerance
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
