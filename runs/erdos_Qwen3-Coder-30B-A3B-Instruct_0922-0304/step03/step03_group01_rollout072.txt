import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_erdos_solution(h_values):
    # This is the helper function provided by the harness
    h = np.array(h_values)
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - two pulses
        h = np.zeros(n_points)
        # First pulse
        idx1 = int(n_points * 0.25) 
        width1 = int(n_points * 0.15)
        h[idx1:idx1+width1] = 0.5
        # Second pulse  
        idx2 = int(n_points * 0.75)
        width2 = int(n_points * 0.15)
        h[idx2:idx2+width2] = 0.5
        # Normalize
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation with finite differences
    def compute_gradient(h, eps=1e-5):
        base_c5 = evaluate_c5(h, dx)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Line search for optimal step size
    def line_search(h, grad, max_iter=20):
        alpha = 0.01
        eps = 1e-5
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, dx)
            # Armijo condition
            if new_c5 < evaluate_c5(h, dx) - 0.5 * alpha * np.sum(grad**2):
                return alpha, h_new
            alpha *= 0.5
        return alpha, h
    
    # Optimization loop
    iterations = 0
    no_improvement_since = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Line search
        alpha, h_new = line_search(h, grad)
        
        # Update
        h = h_new
        
        # Evaluate
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_since = 0
        else:
            no_improvement_since += 1
            
        # Perturb if stuck
        if no_improvement_since > max_no_improvement:
            noise_magnitude = 1e-3 * (1.0 / (1.0 + iterations**0.5))
            h += np.random.normal(0, noise_magnitude, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_since = 0
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
