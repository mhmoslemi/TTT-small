import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Compute a good starting point with better optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    # We'll use only direct local refinement with good initialization
    # targets used: convergence, speed and avoiding simple local minima
    
    bounds = [(0, 1) for _ in range(n_points)]
    target_sum = n_points / 2

    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimization with various starting points and local search methods
    max_iter = 100
    local_search_methods = ['Nelder-Mead', 'Powell', 'CG']
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Slight perturbation for additional diversity
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        current_h = h_perturbed
        
        for method in local_search_methods:
            if time.time() - start_time > budget_s - 1:
                break
            
            result = minimize(
                objective,
                current_h.copy(),
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 20, 'ftol': 1e-8} 
            )
            
            if result.success:
                h_temp = result.x
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                current_c5 = evaluate_c5(h_temp, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_temp.copy()
                    
                if best_c5 <= 0.3807:
                    break
                    
        # Adaptive refinement
        if iteration % 10 == 0:
            # Add random small noise to escape local minima
            h += np.random.uniform(-0.01, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        if best_c5 <= 0.3807:
            break
            
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure the final solution is valid
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-4
    assert all(0 <= x <= 1 for x in best_h)
    
    return (best_h.tolist(), final_c5, n_points)
