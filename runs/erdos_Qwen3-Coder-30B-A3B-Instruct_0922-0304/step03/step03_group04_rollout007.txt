import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    # Direct optimization using scipy with multiple restarts
    n_restarts = 20
    n_iter_per_restart = 100
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random restart
        if restart == 0:
            h_restart = h.copy()
        else:
            h_restart = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        # Optimize with L-BFGS-B and then SLSQP for better convergence
        for method in ['L-BFGS-B', 'SLSQP']:
            if time.time() - start_time > budget_s - 1:
                break
                
            try:
                result = minimize(
                    objective,
                    h_restart.copy(),
                    method=method,
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': n_iter_per_restart // 2, 'ftol': 1e-8}
                )
                
                if result.success:
                    h_temp = result.x
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    
                    current_c5 = evaluate_c5(h_temp, dx)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_temp.copy()
                        
                    if best_c5 <= 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
                        
                h_restart = h_temp.copy()
            except Exception:
                continue

    # Final refinement with more expensive local search
    if time.time() - start_time < budget_s - 1:
        # Try gradient-based refinement using finite differences
        step = 0.005
        max_iter = 1000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient via finite differences
            corr = correlate(best_h, 1 - best_h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(best_h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - max_overlap) / eps
            
            # Update with gradient descent
            best_h -= step * grad
            # Ensure constraints
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            
            current_c5 = evaluate_c5(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
