import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available, otherwise create a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    h = h * (n_points / 2) / np.sum(h)
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add more particles with random variations
    for _ in range(n_particles - 1):
        h_temp = h + np.random.normal(0, 0.05, n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        particles.append(h_temp.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal best and global best
    pbest = [p.copy() for p in particles]
    pbest_vals = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight linearly
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                            c_p * r_p * (pbest[i] - particles[i]) + 
                            c_g * r_g * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update particle
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to satisfy constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new solution
            current_val = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best if needed
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    
        # Early exit if target achieved
        if gbest_val <= 0.3807:
            break
    
    # Return the best solution found
    return (gbest.tolist(), gbest_val, n_points)
