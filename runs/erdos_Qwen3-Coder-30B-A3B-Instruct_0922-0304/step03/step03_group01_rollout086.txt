import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    elitism_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    def initialize_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            # Fallback initialization
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        return h
    
    # Initialize population
    population = [initialize_individual() for _ in range(pop_size)]
    for ind in population:
        ind *= (n_points / 2) / np.sum(ind)
    
    # Evaluate initial population
    def evaluate_individual(h):
        return evaluate_c5(h, dx)
    
    fitness_scores = [evaluate_individual(ind) for ind in population]
    best_idx = np.argmin(fitness_scores)
    best_h = population[best_idx].copy()
    best_c5 = fitness_scores[best_idx]
    
    # Evolution loop
    no_improvement_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s * 0.8 and generation < max_generations:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Elitism
        elite_size = max(1, int(elitism_rate * pop_size))
        new_population = sorted_population[:elite_size]
        
        # Selection (tournament)
        def tournament_selection(tournament_size=3):
            indices = np.random.choice(pop_size, tournament_size)
            best_idx = indices[np.argmin([fitness_scores[i] for i in indices])]
            return population[best_idx]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.random() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                parent = tournament_selection()
                child = parent.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                
            # Repair to satisfy constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Replace population
        population = new_population[:pop_size]
        fitness_scores = [evaluate_individual(ind) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_c5 = fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 50:
            break
            
        if best_c5 <= 0.3807:
            break
    
    # Local refinement with gradient descent
    def project_to_constraint(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Gradient-based refinement
    def gradient_descent_refinement(h_init, max_iter=500):
        h = h_init.copy()
        h = project_to_constraint(h)
        learning_rate = 0.001
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient numerically
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = project_to_constraint(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Gradient step
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = project_to_constraint(h)
            
            # Early stopping if close to convergence
            if np.linalg.norm(grad) < 1e-6:
                break
        
        return h
    
    if time.time() - start_time < budget_s * 0.95:
        refined_h = gradient_descent_refinement(best_h)
        refined_c5 = evaluate_c5(refined_h, dx)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return (best_h.tolist(), best_c5, n_points)
