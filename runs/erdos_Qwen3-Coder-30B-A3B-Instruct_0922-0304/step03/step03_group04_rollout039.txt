import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_overlap_matrix(M):
    # Compute overlap matrix for piecewise constant functions
    # Each interval has width 2/M
    w = 2.0 / M
    # Generate all possible shifts
    shifts = []
    for i in range(M):
        for j in range(M):
            # Overlap between interval i and interval j shifted by (j-i) positions
            shift = j - i
            shifts.append(shift)
    shifts = sorted(list(set(shifts)))
    
    A = np.zeros((len(shifts), M))
    for k_idx, k in enumerate(shifts):
        for i in range(M):
            # Compute overlap with shifted interval
            if i + k >= 0 and i + k < M:
                start1 = i * w
                end1 = (i + 1) * w
                start2 = (i + k) * w
                end2 = (i + k + 1) * w
                overlap_start = max(start1, start2)
                overlap_end = min(end1, end2)
                if overlap_end > overlap_start:
                    overlap_length = overlap_end - overlap_start
                    A[k_idx, i] = overlap_length
                else:
                    A[k_idx, i] = 0.0
            else:
                A[k_idx, i] = 0.0
    return A, shifts

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    M = 30  # Number of intervals
    w = 2.0 / M
    dx = w
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (M / 2) / np.sum(h_init)
        # Average initial values into intervals
        h_avg = np.zeros(M)
        for i in range(M):
            start_idx = int(i * len(h_init) / M)
            end_idx = int((i + 1) * len(h_init) / M)
            if end_idx > start_idx:
                h_avg[i] = np.mean(h_init[start_idx:end_idx])
            else:
                h_avg[i] = 0
        h_init = h_avg
    else:
        h_init = np.ones(M) * 0.5  # fallback
        
    # Compute overlap matrix
    A, shifts = compute_overlap_matrix(M)
    
    # Solve optimization problem
    h = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == M / 2,  # Integral constraint
        h >= 0,              # Non-negativity
        h <= 1               # Boundedness
    ]
    
    # Add overlap constraints
    for k_idx, shift in enumerate(shifts):
        # Skip impossible shifts
        if shift >= M or shift < -M:
            continue
        rhs = 0.0
        for i in range(M):
            if 0 <= i + shift < M:
                rhs += A[k_idx, i] * (1.0 - h[i + shift])
            else:
                rhs += A[k_idx, i] * 0.0  # contribution is 0
        constraints.append(rhs <= C)
    
    # Objective function
    objective = cp.Minimize(C)
    
    # Construct the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with a reasonable solver
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, feastol=1e-5, reltol=1e-5)
    except:
        # Fallback to SLSQP if ECOS fails
        pass
    
    if prob.status != cp.OPTIMAL:
        # Fallback to gradient-based optimization
        h = h_init.copy()
        # Keep using previous solution if now infeasible
        
    # Retrieve solution
    if prob.status == cp.OPTIMAL:
        h_sol = h.value
    else:
        # Try mild optimization on original values
        h = h_init.copy()
        h_sol = h
        
    # Normalize so that sum(h) * w = 1
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * (M / 2) / np.sum(h_sol)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_sol, dx)
    
    # Possibly estimate improvement by cycling a few times
    best_c5 = final_c5
    best_h = h_sol.copy()
    
    for _ in range(30):
        if time.time() - start_time > budget_s - 1:
            break
        # Local improvement using SLSQP
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        bounds = [(0, 1) for _ in range(M)]
        def constraint_sum(h):
            return np.sum(h) - M / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        try:
            result = minimize(
                objective,
                h_sol.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 5, 'ftol': 1e-8}
            )
            if result.success:
                h_sol = result.x
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (M / 2) / np.sum(h_sol)
                current_c5 = evaluate_c5(h_sol, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_sol.copy()
                if best_c5 <= 0.3807:
                    break
        except:
            pass
            
        # Try perturbations
        h_pert = h_sol.copy() + np.random.normal(0, 0.01, M)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (M / 2) / np.sum(h_pert)
        current_c5 = evaluate_c5(h_pert, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_pert.copy()
    
    return (best_h.tolist(), best_c5, M)
