import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill population with random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    
    # Genetic algorithm parameters
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    elite_size = population_size // 2
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = population[:elite_size]
        
        # Tournament selection and crossover/mutation
        while len(new_population) < population_size:
            # Selection
            tournament_indices = np.random.choice(elite_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # Second parent
            tournament_indices = np.random.choice(elite_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent2 = population[winner_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    # Rescale to meet constraint
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
        
        population = new_population[:population_size]
    
    # Local refinement of best individual
    if best_individual is not None:
        h_refined = best_individual.copy()
        best_c5_refined = best_c5
        
        # Simple gradient-based refinement
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Gradient approximated by finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                c5_plus = evaluate_c5(h_plus, dx)
                c5_minus = evaluate_c5(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update using gradient descent
            step_size = 0.1
            h_new = h_refined - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5_refined:
                h_refined = h_new
                best_c5_refined = c5_new
            else:
                break
                
        if best_c5_refined < best_c5:
            best_c5 = best_c5_refined
            best_individual = h_refined
    
    final_c5 = evaluate_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
