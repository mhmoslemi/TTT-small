import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    max_iter = 1000
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(overlap)
        max_overlap = overlap[max_k_idx] * dx

        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            max_move *= 0.95
            no_improvement_count = 0
            if max_move < 1e-6:
                break

        # Find indices contributing most to maximum overlap
        # This is for the shift that achieves max overlap
        # We analyze the overlap array in terms of its shift index
        # The full correlation has length 2*n_points - 1
        # For shift k, overlap is at index k (or k - n_points if k >= n_points)
        if max_k_idx < n_points:
            shift = max_k_idx
        else:
            shift = max_k_idx - n_points
        
        # For this shift, we look at overlap contribution indices
        # h[i] * (1 - h[i + shift]) contributes to overlap at shift
        # Find top contributing h[i] and h[i+shift]
        top_contributors = []
        for i in range(n_points):
            j = (i + shift) % n_points
            contrib = h[i] * (1 - h[j])
            top_contributors.append((contrib, i, j))
        
        # Sort by contribution and take top 10%
        top_contributors.sort(reverse=True)
        top_count = max(1, n_points // 10)
        top_contributors = top_contributors[:top_count]

        # Update h to reduce overlap for top contributors
        for _, i, j in top_contributors:
            # Try to move mass from i to j
            # Decrease h[i], increase h[j]
            move_amount = min(max_move, h[i], 1 - h[j])
            if move_amount > 1e-10:
                h[i] -= move_amount
                h[j] += move_amount

        # Ensure constraints
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h *= scale_factor

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
