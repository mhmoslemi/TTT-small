from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Main cutting-plane algorithm
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        current_c5 = evaluate_c5(h, dx)

            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find the shift k that maximizes overlap
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Create a new constraint to help with optimization
        # h[i] * (1 - h[i+k]) <= z for all i
        # Then we minimize max_z (linearized max-overlap constraint)
        # but for now we'll do refinement using cvxpy and possibly gradient update
        
        # Linearized optimization using CVXPY

        # For now we directly compute this via a local optimization
        # without solver, so our approach is projective gradient
        step = 0.005
        
        # Keep track of slowness of LP solver (if using large number of constraints)
        h_temp = h.copy()
        grad_norm = 0.0
        
        # Gradient estimates based on finite differences
        eps = 1e-4
        grad = np.zeros_like(h_temp)
        current_overlap = correlate(h_temp, 1 - h_temp, mode="full")

        for i in range(n_points):
            h_plus = h_temp.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - reference_val) / eps
            
        # Apply gradient update
        h_temp -= step * grad
        h_temp = np.clip(h_temp, 0, 1)

        h = h_temp
        
        # Verify improvement
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation of the best solution
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
