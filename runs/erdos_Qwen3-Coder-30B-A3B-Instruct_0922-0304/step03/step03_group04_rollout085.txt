import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 30  # Number of intervals
    w = 2.0 / M  # Width of each interval
    dx = w
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Convert to interval representation
        h_interval = np.zeros(M)
        for i in range(M):
            start_idx = int(i * (len(h) / M))
            end_idx = int((i + 1) * (len(h) / M))
            h_interval[i] = np.mean(h[start_idx:end_idx])
        h = h_interval
    else:
        # Fallback initialization
        h = np.ones(M) * 0.5
    
    # Normalize to ensure integral = 1
    h = h * (M / 2) / np.sum(h)
    
    # Optimized interval-based approach using convex optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Build overlap matrix for all shifts
    # For M intervals of width w, there are 2*M-1 possible shifts
    A = np.zeros((2*M-1, M))
    
    # Compute overlap matrix
    for shift in range(-M+1, M):
        for i in range(M):
            if shift >= 0:
                # Shift is positive (future intervals)
                j = i + shift
                if j < M:
                    # Overlap area
                    overlap = min(w, w)  # Always full overlap of same intervals
                    A[shift+M-1, i] = overlap
            else:
                # Shift is negative (past intervals)
                j = i + shift
                if j >= 0:
                    # Overlap area (slightly different computation)
                    overlap = min(w, w)
                    A[shift+M-1, i] = overlap
    
    # Solve the LP using CVXPY
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(cp.multiply(h_vars, w)) == 1,  # Integral constraint
        0 <= h_vars, h_vars <= 1,  # Bounds
        C >= 0  # C >= 0
    ]
    
    # Add overlap constraints for each shift
    for shift_idx in range(2*M-1):
        # For each shift, compute constraint:
        # sum_{i} h_i * (1 - h_{i+shift}) * overlap_coefficient <= C
        # We handle this by computing sums over valid combinations
        # Actually compute the overlap sum directly:
        expr = 0
        shift_val = shift_idx - (M - 1)
        for i in range(M):
            j = i + shift_val
            if 0 <= j < M:
                # Need to know the overlap of interval i and interval j when shifted
                # We'll simplify here and use average overlap (this is an approximation)
                # In practice we're making a simpler optimization model
                expr += h_vars[i] * (1 - h_vars[j]) * A[shift_idx, i]
        constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, feastol=1e-5, reltol=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_vars.value
            c5_opt = C.value
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # If not successful or better solution is found, try alternating optimization
    if best_c5 == float('inf') or best_c5 > 0.381:
        # Re-optimization approach using gradient descent or simpler scheme
        # Faster heuristic that might be good enough for the current budget
        
        # Create step function from h
        h = np.clip(best_h, 0, 1)
        
        # Ensure integral constraint
        h = h * (M / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * (M / 2) / np.sum(h)
        
        # Refine by optimizing C5 with simulated annealing

            # Simple perturb

            new_h = h + delta
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (M / 2) / np.sum(new_h)
            return new_h
        
        current_h = h.copy()

            new_h = update_h(current_h)
            new_c5 = evaluate_c5(new_h, dx)

                current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            # Early termination if target reached
            if best_c5 <= 0.3807:
                break
    
    # Final check of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, M)
