import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def initialize_population(n_points, pop_size):
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
        population.append(h)
    
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=3):
    selected = []
    for _ in range(len(population) // 2):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2):
    if np.random.rand() < 0.7:
        crossover_point = np.random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    return parent1.copy(), parent2.copy()

def mutate(h, mutation_rate=0.2, noise_sigma=0.05):
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, noise_sigma)
    return repair_h(h, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    max_generations = 1000
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            child1, child2 = crossover(parent1, parent2)
            new_population.append(mutate(child1))
            new_population.append(mutate(child2))
            
        population = new_population[:pop_size]
    
    # Ensure the best solution meets constraints
    if best_h is not None:
        best_h = repair_h(best_h, n_points)
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # Fallback to initial if nothing was found
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = repair_h(h, n_points)
            final_c5 = evaluate_c5(h, dx)
            best_h = h
        else:
            # Create a simple valid solution
            h = np.ones(n_points)
            h = h * (n_points / 2) / np.sum(h)
            final_c5 = evaluate_c5(h, dx)
            best_h = h
    
    return (best_h.tolist(), final_c5, n_points)
