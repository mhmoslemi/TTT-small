import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Instead of SLSQP, let's use a combination of gradient descent
    # and local search to escape local minima
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def evaluate_gradient(h, k):
        """Compute gradient of overlap term with respect to h[k]"""
        # This is an approximate gradient computation
        eps = 1e-4
        h_plus = h.copy()
        h_plus[k] += eps
        h_plus = project_feasible(h_plus)
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h.copy()
        h_minus[k] -= eps
        h_minus = project_feasible(h_minus)
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        c5_minus = np.max(corr_minus) * dx
        
        return (c5_plus - c5_minus) / (2*eps)

    # Gradient descent with adaptive step size
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive gradient descent step
        step_size = 0.005
        h_new = h.copy()
        
        # Sample some indices to update
        indices = np.random.choice(n_points, size=min(20, n_points), replace=False)
        for i in indices:
            grad = evaluate_gradient(h_new, i)
            h_new[i] -= step_size * grad
            # Project to feasible region
            h_new = project_feasible(h_new)
        
        h = h_new
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final optimization using local search around best solution
    h = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try small perturbations and accept better values
        h_trial = h.copy()
        i = np.random.randint(n_points)
        delta = np.random.normal(0, 0.02)
        h_trial[i] += delta
        h_trial = project_feasible(h_trial)
        
        current_c5 = evaluate_c5(h, dx)
        trial_c5 = evaluate_c5(h_trial, dx)
        if trial_c5 < current_c5:
            h = h_trial
            
        # Also try swapping two segments
        if n_points >= 4:
            i1 = np.random.randint(n_points - 2)
            i2 = np.random.randint(i1 + 2, n_points)
            h_trial = h.copy()
            h_trial[i1:i2] = h[i2-1:i1-1:-1] if i1 < i2-1 else h[i2:i1]
            h_trial = project_feasible(h_trial)
            trial_c5 = evaluate_c5(h_trial, dx)
            if trial_c5 < current_c5:
                h = h_trial
    
    final_c5 = evaluate_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
