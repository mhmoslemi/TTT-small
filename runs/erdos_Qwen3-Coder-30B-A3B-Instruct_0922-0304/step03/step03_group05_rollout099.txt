import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_c5_grad(h, k_max, i_max):
        # Compute gradient at i_max, k_max
        # We only care about the change in the max overlap term
        n = len(h)
        grad = np.zeros(n)
        # Efficient gradient computation for that specific term
        grad[i_max] = 1 - h[(i_max + k_max) % n]
        grad[(i_max + k_max) % n] = -h[i_max]
        return grad * dx

    def find_max_overlap_indices(h):
        # Find all indices that would contribute to max overlap
        n = len(h)
        overlaps = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlaps)
        # correspond to full convolution where overlap[i] = sum(h[j] * (1 - h[i+j]))
        # Find contributing indices for the max
        # This is the direct way
        k = max_k - (n - 1)  # shift identification
        max_val = overlaps[max_k]
        
        # Build a list of contributing indices for top overlap
        contrib_indices = []
        for i in range(n):
            if h[i] > 0 and (1 - h[(i + k) % n]) > 0:
                contrib_indices.append(i)
        contrib_indices.sort(key=lambda i: h[i] * (1 - h[(i + k) % n]), reverse=True)
        return k, contrib_indices[:max(1, len(contrib_indices) // 4)]

    def adjust_h(h, indices, k, max_move=0.01):
        # Adjust values to decrease the maximum overlap contribution
        # By moving values around to reduce product values
        h_new = h.copy()
        total_adjusted = 0
        
        for i in indices:
            # Determine max adjustment to reduce overlap at i,k
            # current product: h[i] * (1 - h[i+k])
            # try to reduce this
    
            # decrease h[i] and increase h[i+k] in appropriate fashion
            # We want to move mass from i to i+k to reduce product
            if h_new[i] > 0 and h_new[(i + k) % n] < 1:
                # max movement: move h[i] * (1-h[i+k]) restriction
                max_delta = min(h_new[i], 1 - h_new[(i + k) % n])
                if max_delta < 1e-8:
                    continue
                delta = min(max_delta, max_move)
                h_new[i] -= delta
                h_new[(i + k) % n] += delta
                total_adjusted += delta
    
        # re-normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    max_move = 0.01
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                # Reduce tolerance to search for finer adjustments
                max_move *= 0.95
                no_improvement_count = 0
                if max_move < 1e-6:
                    break
                    
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find key overlap term and indices
        max_k, top_indices = find_max_overlap_indices(h)

        if len(top_indices) > 0:
            h = adjust_h(h, top_indices, max_k, max_move)
            
        # Ensure constraints (feasibility)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
