import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop using quadratic programming
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find the shift with maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Build Quadratic Program for this shift
        # Objective: minimize sum_i h[i] * (1 - h[i + shift])
        # Constraint: sum(h) = n_points / 2, 0 <= h[i] <= 1
        
        # Define the quadratic objective
        # Let's construct a quadratic form of the overlap
        # h[i] * (1 - h[i + shift]) = h[i] - h[i] * h[i + shift]
        # For a fixed shift k, the objective is to minimize sum over valid indices
        # Let's consider a smaller subset to avoid boundary issues
        
        # For shift k, construct an objective that sums over valid pairs
        # This is handled via matrix representation for QP
        h_var = cp.Variable(n_points)
        
        # Handle indices properly for the shift
        upper_bound = n_points - abs(max_shift)
        if max_shift >= 0:
            # h[i] * (1 - h[i + shift])
            # Sum_{i=0}^{upper_bound-1} h[i] * (1 - h[i+shift])
            objective = 0
            for i in range(upper_bound):
                objective += h_var[i] * (1 - h_var[i + max_shift])
        else:
            # h[i + shift] * (1 - h[i])
            # Sum_{i=-shift}^{upper_bound-1} (1-h[i]) * h[i-shift]
            objective = 0
            shift = -max_shift
            for i in range(shift, upper_bound + shift):
                objective += (1 - h_var[i]) * h_var[i - shift]
                
        # Quadratic programming
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.maximum(0, np.minimum(1, h_new))
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
                    continue
        except:
            pass
        
        # Fallback gradient descent with backtracking
        step = 0.005
        iterations = 50
        h_temp = h.copy()
        
        for _ in range(iterations):
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update h
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
