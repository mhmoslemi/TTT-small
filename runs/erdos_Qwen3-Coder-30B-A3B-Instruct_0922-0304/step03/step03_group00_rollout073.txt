import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_target"""
    h = np.clip(h, 0, 1)
    h_target = n_points / 2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * h_target / h_sum
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient of overlap integral for shift k"""
    grad = np.zeros(n_points)
    # For shift k, h[i] contributes to overlap with h[i+k]
    # gradient w.r.t. h[i] is (1 - h[i+k]) if i+k < n_points
    # and -h[i-k] if i-k >= 0
    
    # Forward shift: h[i] contributes to term h[i+k] (where i+k < n_points)
    for i in range(n_points):
        if i + k < n_points:
            grad[i] += (1 - h[i+k]) * dx
        if i - k >= 0:
            grad[i] -= h[i-k] * dx
    return grad

def compute_c5(h, dx):
    """Compute C5 = max overlap"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = project_h(h, n_points)
    else:
        # Fallback initialization - triangular
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points-1-i] = i / mid
        h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps to find best shift
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_k = max_overlap_idx - (n_points - 1)  # Convert index to shift
        
        # Compute gradient for this shift
        grad = compute_gradient(h, max_k, n_points, dx)
        
        # Adaptive step size
        step = 0.01
        # Try a few iterations to find a better step
        for _ in range(5):
            h_new = h - step * grad
            h_new = project_h(h_new, n_points)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                h = h_new
                break
            else:
                step *= 0.5  # Reduce step size if no improvement
        else:
            # If no improvement with our current step, just make a small update
            h = h - step * grad
            h = project_h(h, n_points)
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
