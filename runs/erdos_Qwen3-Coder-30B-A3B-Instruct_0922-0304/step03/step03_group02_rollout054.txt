import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # number of particles
    max_iter = int(budget_s * 20)  # rough estimate of iterations
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
    else:
        h_base = np.ones(n_points)
        h_base[n_points//2:] = 0
        h_base += np.random.normal(0, 0.05, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(m):
        h = h_base.copy()
        # Add small random perturbations to diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h.copy())
        pbest_values.append(evaluate_c5(h, dx))
    
    # Global best
    global_best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[global_best_idx].copy()
    gbest_value = pbest_values[global_best_idx]
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = w * v + c1 * r1 * (pbest_positions[i] - h) + c2 * r2 * (gbest_position - h)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            current_c5 = evaluate_c5(h_new, dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_positions[i] = h_new.copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_position = h_new.copy()
                    gbest_value = current_c5
            
            # Keep updated
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
            
            # Early termination
            if gbest_value <= 0.3807:
                break
                
        # Early termination check
        if gbest_value <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(gbest_position, dx)
    
    return (gbest_position.tolist(), final_c5, n_points)
