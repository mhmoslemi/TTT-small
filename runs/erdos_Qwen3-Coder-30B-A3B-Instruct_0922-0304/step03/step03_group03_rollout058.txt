import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 300
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Improve using trust region method
        def objective(h_vec):
            h_plot = h_vec.copy()
            h_plot = np.clip(h_plot, 0, 1)
            h_plot = h_plot * (n_points / 2) / np.sum(h_plot)
            return evaluate_c5(h_plot, dx)
        
        def grad(h_vec):
            h_plot = h_vec.copy()
            h_plot = np.clip(h_plot, 0, 1)
            h_plot = h_plot * (n_points / 2) / np.sum(h_plot)
            
            corr = correlate(h_plot, 1 - h_plot, mode="full")
            max_overlap = np.max(corr) * dx
            max_idx = np.argmax(corr)
            
            # Gradient approximation using finite differences
            grad_vals = np.zeros_like(h_plot)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_plot.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full") 
                c5_plus = np.max(corr_plus) * dx
                grad_vals[i] = (c5_plus - max_overlap) / eps
                
            return grad_vals
            
        # Adaptive trust region step
        try:
            # Use scipy minimize with trust-constr
            result = minimize(objective, h, method='trust-constr', jac=grad,
                              options={'maxiter': 10, 'disp': False})
            if result.success:
                h = result.x.copy()
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to basic gradient descent
            step = 0.001
            grad_vals = grad(h)
            h -= step * grad_vals
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
