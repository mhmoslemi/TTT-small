import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - create a more structured pattern
        h = np.zeros(n_points)
        # Create a periodic pattern with some randomization
        periods = 4
        for i in range(n_points):
            phase = (i * periods) / n_points
            h[i] = max(0, 0.5 + 0.3 * np.sin(2 * np.pi * phase) * (1 - 0.1 * np.random.random()))
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Trust-region optimization approach
    max_iter = 5000
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with a more robust method
        try:
            result = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=cons,
                            options={'maxiter': 50, 'gtol': 1e-8, 'xtol': 1e-8})
            
            if result.success:
                h = result.x
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
                    
                # Perturb if no improvement for some iterations
                if no_improvement_count > max_no_improvement:
                    h = best_h.copy()
                    perturbation_magnitude = 0.02
                    noise = np.random.normal(0, perturbation_magnitude, n_points)
                    h += noise
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    no_improvement_count = 0
            else:
                # Fallback to gradient descent if trust-constr fails
                h = update_h_gradient_descent(h, dx, 5, 0.005)
                
        except Exception:
            # Fallback to gradient descent if any optimization method fails
            h = update_h_gradient_descent(h, dx, 5, 0.005)
            
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def update_h_gradient_descent(h, dx, iterations, step_size):
    """Local gradient descent update"""
    n_points = len(h)
    for _ in range(iterations):
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx

        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    return h
