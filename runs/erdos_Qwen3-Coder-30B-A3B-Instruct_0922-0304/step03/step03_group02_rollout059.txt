import numpy as np
from scipy.signal import correlate
import time
from sklearn.preprocessing import StandardScaler

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for a given step function h."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to ensure it satisfies constraints: 0≤h≤1 and sum(h)=n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h
