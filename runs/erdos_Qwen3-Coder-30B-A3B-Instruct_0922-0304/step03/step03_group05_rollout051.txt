import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a triangular initialization
        h = np.zeros(n_points)
        half = n_points // 2
        for i in range(half):
            h[i] = i / half
            h[n_points - 1 - i] = i / half
        h[half] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    # Generate 5 random perturbations
    perturbations = []
    for _ in range(5):
        h_pert = h + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        perturbations.append(h_pert)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Evaluate all initial solutions
    initial_solutions = [h] + perturbations
    for sol in initial_solutions:
        c5 = compute_c5(sol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = sol.copy()

    # Main optimization loop
    max_iter = 500
    iter_count = 0
    no_improvement_count = 0
    last_best = best_c5

    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Pick a random initial solution
        h_current = np.random.choice(initial_solutions)
        
        # Optimize using SLSQP
        result = minimize(
            objective, 
            h_current, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        # If optimization fails, use a small perturbation instead
        if not result.success:
            h_pert = h_current + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            current_c5 = compute_c5(h_pert)
        else:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            current_c5 = compute_c5(h_opt)
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current if not result.success else h_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Early exit if no improvement for a while
        if no_improvement_count > 100:
            break
            
        iter_count += 1

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
