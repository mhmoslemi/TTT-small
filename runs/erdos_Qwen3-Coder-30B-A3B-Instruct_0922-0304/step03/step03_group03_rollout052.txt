import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 <= 0.3807:

        def objective(x):

            x = x * (n_points / 2) / np.sum(x)
            return evaluate_c5(x, dx)

            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2
            
        # Use SLSQP optimizer with bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds,
