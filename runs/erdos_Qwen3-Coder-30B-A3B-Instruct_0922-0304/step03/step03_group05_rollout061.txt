import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_autocorr(h):
        # Compute autocorrelation for all lags
        autocorr = correlate(h, h, mode="full")
        return autocorr

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    max_iter = 300
    patience = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # For every 10 iterations, perturb solution to escape local optima
        if iteration % 10 == 0 and iteration > 0:
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            continue
            
        # If no improvement for 20 iterations, break
        if patience > 20:
            break
            
        # Find worst lags (those with lowest autocorrelation)
        autocorr = compute_autocorr(h)
        # We only look at the middle part (the actual cross-correlation values)
        # For a full autocorrelation of length 2*n_points-1, the valid cross-correlation is in the center
        middle = len(autocorr) // 2
        valid_autocorr = autocorr[middle:middle+n_points-1]
        
        # Target lags with weakest autocorrelation
        num_target_lags = min(5, len(valid_autocorr))
        worst_lags = np.argsort(valid_autocorr)[:num_target_lags]
        
        # Formulate quadratic optimization problem to increase minimum autocorrelation
        # Here we solve an unconstrained quadratic (least-squares style) on differences
        def obj_delta(h_new):
            new_autocorr = correlate(h_new, h_new, mode="full")
            new_valid = new_autocorr[middle:middle+n_points-1]
            # Minimize the negative of the sum of the worst auto-correlations
            return -np.sum(new_valid[worst_lags])
            
        # Use SLSQP to minimize this objective
        try:
            result = minimize(
                obj_delta,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback: small perturbation
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback in case of exception
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
