import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop using projected gradient method
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        worst_shift = max_idx - (n_points - 1)
        
        # Formulate and solve QP for this shift
        # Minimize sum h[i] * (1 - h[i + shift]) * dx
        # subject to sum(h) = n_points/2 and 0 <= h[i] <= 1
        
        # Build quadratic objective
        Q = np.zeros((n_points, n_points))
        c = np.zeros(n_points)
        
        # For the specific shift, compute the linear and quadratic terms
        # Q[i,j] represents contribution of h[i]*h[j]  
        for i in range(n_points):
            j = i + worst_shift
            if 0 <= j < n_points:
                Q[i, j] = dx
                c[i] = -dx  # Linear term from h[i] * 1
        
        # Add identity to Q for quadratic term
        for i in range(n_points):
            Q[i, i] += dx
            
        # Solve QP: Minimize (1/2) * h^T * Q * h + c^T * h with constraints
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(0.5 * cp.quad_form(h_var, Q) + c.T @ h_var)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h = h_new
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient step if QP fails
            step = 0.001
            h_temp = h.copy()
            
            # Evaluate the gradient
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Numerical gradient (only for the shift we are optimizing)
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update and normalize
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
