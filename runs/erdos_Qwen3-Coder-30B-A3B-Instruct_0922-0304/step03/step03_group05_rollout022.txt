import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # Find the shift k that maximizes overlap
        k_max = np.argmax(overlap)
        # Compute overlap contribution for all indices
        overlap_contrib = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            overlap_contrib[i] = h[i] * (1 - h[j])
        
        # Get indices with highest contribution
        top_indices = np.argsort(overlap_contrib)[-int(n_points * 0.05):][::-1]
        
        # Adjust these indices
        for i in top_indices:
            j = (i + k_max) % n_points
            # Compute maximum adjustment
            decrease = min(h[i], max_move)
            increase = min(1 - h[j], max_move)
            epsilon = min(decrease, increase)
            if epsilon > 1e-10:
                h[i] -= epsilon
                h[j] += epsilon
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust step size
        if no_improvement_count > 10:
            max_move *= 0.95
        
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
