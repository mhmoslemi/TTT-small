import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = repair_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h):
        h = repair_h(h, n_points)
        return evaluate_c5(h, dx)
    
    # CMA-ES setup
    es = CMAEvolutionStrategy(h, 0.1, {'popsize': 100, 'seed': seed})
    
    iteration = 0
    while not es.stop() and iteration < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get candidate solutions
        X = es.ask()
        fitness = []
        for x in X:
            f = objective(x)
            fitness.append(f)
        
        # Tell evaluation results
        es.tell(X, fitness)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        current_h = repair_h(X[best_idx], n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
