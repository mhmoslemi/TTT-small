import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def wavelet_project(h, wavelet='haar', level=6):
    coeffs = pywt.wavedec(h, wavelet, level=level)
    return coeffs

def wavelet_reconstruct(coeffs, wavelet='haar'):
    return pywt.waverec(coeffs, wavelet)

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

    wavelet_name = 'haar'

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet coefficients
    coeffs = wavelet_project(h, wavelet_name, level)
    coeffs = [c.copy() for c in coeffs]
    
    # Optimization parameters
    max_wavelet_iter = 30
    max_spatial_refinements = 2
    lambda_thresh = 0.1
    learning_rate = 0.01
    
    iteration = 0
    stagnation_count = 0
    last_best_c5 = float('inf')
    
    while iteration < max_wavelet_iter and time.time() - start_time < budget_s - 10:
        # Wavelet domain optimization
        # Build the wavelet basis
        wavelet_coeffs = coeffs.copy()
        
        # Compute gradient in wavelet domain (simplified)
        # Use finite differences to approximate gradient
        eps = 1e-4
        grad_all = [np.zeros_like(c) for c in wavelet_coeffs]
        
        # Compute original value
        h_original = wavelet_reconstruct(wavelet_coeffs, wavelet_name)
        h_original = np.clip(h_original, 0, 1) 
        h_original = h_original * (n_points / 2) / np.sum(h_original)
        current_c5 = evaluate_c5(h_original, dx)
        
        # Compute gradients using finite differences in coefficients
        for i in range(len(wavelet_coeffs)):
            c_temp = [c.copy() for c in wavelet_coeffs]
            c_temp[i] += eps
            h_temp = wavelet_reconstruct(c_temp, wavelet_name)
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            c5_temp = evaluate_c5(h_temp, dx)
            grad_all[i] = (c5_temp - current_c5) / eps
            
        # Apply gradient update with soft-thresholding
        for i in range(len(wavelet_coeffs)):
            wavelet_coeffs[i] -= 0.1 * grad_all[i]
            wavelet_coeffs[i] = soft_threshold(wavelet_coeffs[i], lambda_thresh)
            
        # Reconstruct h
        h_wavelet = wavelet_reconstruct(wavelet_coeffs, wavelet_name)
        h_wavelet = np.clip(h_wavelet, 0, 1)
        h_wavelet = h_wavelet * (n_points / 2) / np.sum(h_wavelet)

    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
