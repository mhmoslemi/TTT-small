import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a different approach: convex optimization with piecewise constant functions
    M = 30  # Number of intervals
    w = 2.0 / M  # Width of each interval
    dx = w  # Step size for discretization
    
    # Initialize h based on initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Average initial values into intervals
        h_init = h_init.reshape(M, -1).mean(axis=1)
        h_init = np.clip(h_init, 0, 1)
    else:
        h_init = np.ones(M) * 0.5
    
    # Normalize to satisfy integral constraint
    h_init = h_init * (M / 2) / np.sum(h_init)
    
    # Compute all overlap lengths between intervals
    # Construct matrix A where A[k][i] = overlap length between interval i and i+k
    A = np.zeros((2*M-1, M))
    # k ranges from -(M-1) to (M-1)
    # But we only care about valid overlaps
    for k in range(-(M-1), M):  # k = shift
        for i in range(M):
            if 0 <= i+k < M:
                # Overlap between interval i and interval i+k
                # The overlap length is min(w, w) = w if they overlap
                # Actually, compute precise overlap length
                int1_start = i * w
                int1_end = (i+1) * w
                int2_start = (i+k) * w
                int2_end = (i+k+1) * w
                
                overlap_start = max(int1_start, int2_start)
                overlap_end = min(int1_end, int2_end)
                if overlap_end > overlap_start:
                    overlap_len = overlap_end - overlap_start
                else:
                    overlap_len = 0
                
                # Store in row k+(M-1) (for indexing)
                A[k + (M-1), i] = overlap_len
    
    # Define variables for convex optimization
    h = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints
    # Integral constraint: sum(h * w) = 1
    integral_constraint = cp.sum(cp.multiply(h, w)) == 1
    
    # Overlap constraints: for each possible shift k
    constraints = []
    
    # For each possible shift k (from -M+1 to M-1)
    for k_idx in range(2*M-1):
        # Compute sum_{i} h_i * (1 - h_{i+k}) * A[k_idx][i]
        shift_sum = 0
        for i in range(M):
            if 0 <= i + (k_idx - (M-1)) < M:
                shift_sum += h[i] * (1 - h[i + (k_idx - (M-1))]) * A[k_idx, i]
        constraints.append(shift_sum <= C)
    
    # Variable bounds
    bounds = [h[i] >= 0 for i in range(M)] + [h[i] <= 1 for i in range(M)]
    
    # Combine all constraints
    all_constraints = [integral_constraint] + constraints + bounds
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Create and solve problem
    prob = cp.Problem(objective, all_constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, feastol=1e-6, reltol=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = C.value
        else:
            # Fallback optimization
            h_opt = h_init
            c5_bound = evaluate_c5(h_opt, dx)
    except Exception:
        # If convex optimization fails, use gradient descent
        h_opt = h_init
        c5_bound = evaluate_c5(h_opt, dx)
    
    # Use gradient descent for refinement
    best_h = h_opt.copy()
    best_c5 = c5_bound
    
    h_refined = best_h.copy()
    learning_rate = 0.01
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient: finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        
        for i in range(M):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update with gradient descent
        h_refined -= learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (M / 2) / np.sum(h_refined)
        
        # Evaluate
        current_c5 = evaluate_c5(h_refined, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
            
        # Early exit if target is reached
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, M)
