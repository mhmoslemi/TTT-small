import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters  
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initial construction: 5 pulses
        h = np.zeros(n_points)
        n_pulses = 5
        pulse_width = 0.4
        for i in range(n_pulses):
            pos = 0.2 + i * 0.4
            start = int(max(0, (pos - pulse_width/2) / dx))
            end = int(min(n_points, (pos + pulse_width/2) / dx))
            h[start:end] = 1.0
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Ensure h ∈ [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return h

    def sample_pulse_positions(h):
        """Sample positions from the pulse distribution"""
        # Construct pulse representation from current h
        total_mass = np.sum(h) * dx
        # Approximate as intervals of width ~dx
        pulses = []
        for i in range(n_points):
            if h[i] > 1e-6:
                # Approximate this as a pulse
                pulses.append((i * dx + dx/2, h[i]))
        return pulses
    
    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Step 1: Optimization using gradient-based method for a small subset
        # Alternate between different optimization approaches
        if iteration % 3 == 0:
            # Perturb and optimize to avoid local minima
            h_pert = h + np.random.normal(0, 0.01, n_points)
            h = project_h(h_pert)
        elif iteration % 3 == 1:
            # Simple random search with projection
            h_new = h + np.random.normal(0, 0.02, n_points) * np.random.choice([0, 1], size=n_points, p=[0.8, 0.2])
            h = project_h(h_new)
        else:
            # Gradient-like update based on gradient of overlap
            # For robustness, we use a simple update
            h_new = h * 0.95 + np.random.normal(0, 0.02, n_points) * 0.05 
            h = project_h(h_new)
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Re-evaluate to ensure correctness
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
