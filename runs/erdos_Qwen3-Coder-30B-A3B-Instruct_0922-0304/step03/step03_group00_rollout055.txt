import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0.copy())
        velocities.append(np.zeros(n_points))
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill remaining particles with triangular functions
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate_c5(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_value
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            
            velocities[i] = (w * velocities[i] + 
                           c_p * r_p * (pbest[i] - particles[i]) +
                           c_g * r_g * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Enforce constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_value = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best if needed
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    best_h = gbest.copy()
                    best_c5 = gbest_value
            
            # Early termination
            if best_c5 <= 0.3807:
                break
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
