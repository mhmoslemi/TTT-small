import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 30  # Using fewer points for faster optimization
    dx = 2.0 / n_points
    M = n_points  # Using M intervals for convex optimization
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        # Discretize into intervals
        h = np.zeros(M)
        for i in range(M):
            start_idx = int(i * len(h_init) / M)
            end_idx = int((i + 1) * len(h_init) / M)
            if start_idx < end_idx:
                h[i] = np.mean(h_init[start_idx:end_idx])
            else:
                h[i] = 0
        h = np.clip(h, 0, 1)
        h = h * (M / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(M)
        h[M//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (M / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization approach
    # Variables
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    # Compute overlap matrix
    w = 2.0 / M
    A = np.zeros((2 * M - 1, M))
    
    # Construct overlap matrix
    for k in range(-M + 1, M):
        # For each shift, compute overlaps between intervals
        for i in range(M):
            if k >= 0:
                j = i + k
                if j < M:
                    # Compute overlap between interval i and interval j
                    overlap_length = min(w, w)  # Both intervals have the same width
                    if k == 0:
                        # Special handling for zero-shift case
                        overlap_length = w
                    A[k + M - 1, i] = overlap_length
            else:
                # Negative k
                j = i + k
                if j >= 0:
                    overlap_length = min(w, w)
                    A[k + M - 1, i] = overlap_length
    
    # Optimization with CVXPY
    constraints = [
        cp.sum(h_vars * w) == 1,
        0 <= h_vars,
        h_vars <= 1
    ]
    
    # Constraints from overlaps
    for k in range(-M + 1, M):
        if k >= 0:
            # For index matching
            shift_idx = k + M - 1
            # Compute overlap expression
            # For overlap between interval i and (i+k) with the function 1-h_{i+k}
            # The expression is sum over i of h_i * (1 - h_{i+k}) * A[k,M] for k>0
            # But we must be more careful about indexing
            
            # This version of overlap is: integral of h(x) * (1 - h(x+k)) dx
            # When discretized: sum of h_i * (1 - h_{i+k}) * w^2 for valid indices
            # Which means for shift k: sum_{i=0}^{M-k-1} h_i * (1 - h_{i+k}) * w^2
            
            # Simplified: sum of h[i] * (1 - h[i+k]) * w for non-zero overlaps
            # Place weights in the correct positions and build constraint
            expr = 0
            for i in range(M - abs(k)):
                if k > 0:
                    expr += h_vars[i] * (1 - h_vars[i + k]) * w
                else:
                    expr += h_vars[i] * (1 - h_vars[i - k]) * w
            constraints.append(expr <= C)
        else:
            # For negative shifts, case is similar but reversed
            # When k < 0, we shift by -k to the left, so we look at h_{i-k}
            # But we ensure i - k < M
            expr = 0
            shift_idx = k + M - 1
            for i in range(-k, M):
                if k < 0:
                    expr += h_vars[i] * (1 - h_vars[i - k]) * w
                else:
                    expr += h_vars[i] * (1 - h_vars[i + k]) * w
            constraints.append(expr <= C)

    
    # Objective to minimize
    objective = cp.Minimize(C)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use a reliable solver
    while time.time() - start_time < budget_s - 5:  # Leave buffer
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_vars.value
                c5_opt =(prob.value)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                # If we're close to target, use a specialized approach
                if c5_opt <= 0.3807:
                    break
            break
        except:
            # Fallback to basic gradient descent
            step = 0.002
            h_temp = h.copy()
            for _ in range(200):
                if time.time() - start_time > budget_s - 1:
                    break
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                # Update h
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                current_c5 = evaluate_c5(h_temp, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_temp.copy()
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
