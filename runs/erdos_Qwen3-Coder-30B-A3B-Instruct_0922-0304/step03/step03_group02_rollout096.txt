import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective for scipy.optimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Define constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Local optimization step
    def local_optimize(h):
        # Try to refine with gradient-based method
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-9}
        )
        if result.success:
            h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Start with a local optimization of the initial solution
    h = local_optimize(h)
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Main optimization loop with local search
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Mutation: slightly perturb the current h
        h_perturbed = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
        for i in indices:
            h_perturbed[i] += np.random.normal(0, 0.05)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local refinement
        h_refined = local_optimize(h_perturbed)
        current_c5 = evaluate_c5(h_refined, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Replace with good solution and continue with new iteration
        h = h_refined
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
