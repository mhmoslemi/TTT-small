import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Trust-region optimization variables
    trust_radius = 0.1
    max_trust_radius = 1.0
    min_trust_radius = 1e-6
    expansion_factor = 2.0
    contraction_factor = 0.5
    bfgs_history = []
    bfgs_update_freq = 5
    line_search_alpha = 0.5
    line_search_max_iter = 20
    perturb_freq = 50
    perturb_magnitude = 0.01
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Compute gradient using finite differences
    def compute_gradient(h, epsilon=1e-4):
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / epsilon
        return grad

    # Optimization with trust-region method
    max_iter = 5000
    no_improvement_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Compute Hessian using BFGS if possible
        if len(bfgs_history) > 0 and iteration % bfgs_update_freq == 0:
            try:
                # Simple BFGS approximation for direction
                H_inv = np.eye(n_points)
                for i in range(len(bfgs_history) - 1, max(-1, len(bfgs_history) - 5), -1):
                    y = grad - bfgs_history[i][1]
                    s = h - bfgs_history[i][0]
                    if np.dot(s, y) > 0:
                        rho = 1.0 / np.dot(s, y)
                        I = np.eye(n_points)
                        H_inv = (I - rho * np.outer(s, y)) @ H_inv @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
            except:
                H_inv = np.eye(n_points)
        else:
            H_inv = np.eye(n_points)
            
        # Compute search direction
        try:
            search_dir = - H_inv @ grad
        except:
            search_dir = - grad
            
        # Line search with backtracking
        step_size = 1.0
        for _ in range(line_search_max_iter):
            h_candidate = h + step_size * search_dir
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            candidate_c5 = evaluate_c5(h_candidate, dx)
            
            if candidate_c5 < current_c5:
                h = h_candidate
                break
            else:
                step_size *= line_search_alpha
                
        # Update trust region
        if step_size < 1e-6:
            trust_radius *= contraction_factor
        else:
            trust_radius = min(trust_radius * expansion_factor, max_trust_radius)
            
        # Store for BFGS if needed
        if iteration % bfgs_update_freq == 0:
            bfgs_history.append((h.copy(), grad.copy()))
            
        # Perturbation if no improvement
        no_improvement_count += 1
        if no_improvement_count > perturb_freq:
            h += np.random.normal(0, perturb_magnitude, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
        # Maintain feasibility and constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
