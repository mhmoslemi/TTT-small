import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Pulse-based optimization approach
    # Represent h as a sum of M pulses with parameters (heights, positions, widths)
    # We will use 5 pulses for good balance of complexity and performance
    
    M = 5  # number of pulses 
    
    # Initialize pulse parameters
    # We define parameters as [h1, p1, w1, h2, p2, w2, ..., hM, pM, wM]
    h_params = np.ones(3 * M)
    h_params[::3] = 0.5  # heights
    h_params[1::3] = np.linspace(0.2, 1.8, M)  # positions
    h_params[2::3] = 0.2  # widths
    
    # Optimization parameters
    max_iter = 1000
    best_c5 = float('inf')
    best_params = h_params.copy()
    
    def pulse_to_h(params):
        h_vals = np.zeros(n_points)
        p = np.linspace(0, 2, n_points)
        
        for i in range(M):
            h = params[3*i]
            pos = params[3*i + 1]
            width = params[3*i + 2]
            
            # Ensure valid bounds
            h = max(0, min(1, h))
            pos = max(0, min(2, pos))
            width = max(0.01, min(2, width))
            
            # Compute pulse interval
            left = pos - width/2
            right = pos + width/2
            
            # Fill values
            indices = np.where((p >= left) & (p <= right))[0]
            h_vals[indices] += h
            
        # Clip and normalize
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        
        return h_vals
    
    def objective(params):
        h_vals = pulse_to_h(params)
        return compute_c5(h_vals)
    
    # Gradient-free optimization (Nelder-Mead) due to discrete nature
    from scipy.optimize import differential_evolution
    
    def optimize_pulses_with_de():
        bounds = []
        # Height bounds: [0, 1]
        for i in range(M):
            bounds.append((0, 1))
        # Position bounds: [0, 2]
        for i in range(M):
            bounds.append((0, 2))
        # Width bounds: [0.01, 2]
        for i in range(M):
            bounds.append((0.01, 2))
            
        def de_objective(params):
            return objective(params)
            
        result = differential_evolution(
            de_objective,
            bounds,
            maxiter=100,
            popsize=15,
            seed=seed,
            disp=False
        )
        
        return result.x
    
    # Alternating optimization loop
    current_params = h_params.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use optimization to improve parameters
        new_params = optimize_pulses_with_de()
        current_params = new_params
        
        # Compute C5 with current configuration
        current_h = pulse_to_h(current_params)
        current_c5 = compute_c5(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_params = current_params.copy()
            
            # Early exit if we've achieved target
            if best_c5 <= 0.3807:
                break
                
    # Final evaluation and return
    final_h = pulse_to_h(best_params)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
