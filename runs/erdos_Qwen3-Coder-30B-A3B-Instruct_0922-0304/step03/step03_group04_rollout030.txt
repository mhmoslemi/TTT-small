import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with different strategies
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different optimization approaches
        current_h = h.copy()
        
        # Approach 1: SLSQP optimization
        try:
            from scipy.optimize import minimize
            bounds = [(0, 1) for _ in range(n_points)]
            target_sum = n_points / 2
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                overlap = correlate(h, 1 - h, mode="full")
                return np.max(overlap) * dx
            
            result = minimize(
                objective,
                current_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
        except:
            pass
        
        # Approach 2: Gradient descent with adaptive step size
        if time.time() - start_time > budget_s - 2:
            break
            
        try:
            # Random perturbation if previous optimization failed
            step = 0.01 * (1.0 - (time.time() - start_time) / budget_s)
            iterations = 20
            
            h_temp = current_h.copy()
            for _ in range(iterations):
                if time.time() - start_time > budget_s - 2:
                    break
                    
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient using finite differences for a small set of indices
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                indices = np.random.choice(n_points, min(5, n_points), replace=False)
                
                for i in indices:
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Update h
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            current_h = h_temp
        except:
            pass
            
        # Evaluate and update best solution
        current_c5 = np.max(correlate(current_h, 1 - current_h, mode="full")) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Simple local search - perturb current solution to escape local optima
        if iteration % 50 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            current_h += perturbation
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            h = current_h.copy()
        else:
            h = current_h.copy()

    # Final evaluation
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
