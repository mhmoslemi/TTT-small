import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start optimization
    initial_candidates = []
    
    # Parent initialization
    initial_candidates.append(h.copy())
    
    # Random perturbations
    for _ in range(5):
        perturbed = h + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_candidates.append(perturbed)
        
    # Fallback initialization if needed
    if len(initial_candidates) == 1:  # Only parent was used
        fallback = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                fallback[i] = i / mid
            else:
                fallback[i] = (n_points - i) / (n_points - mid)
        fallback = np.clip(fallback, 0, 1)
        fallback = fallback * (n_points / 2) / np.sum(fallback)
        initial_candidates.append(fallback)

    # Evaluate all initial candidates
    for candidate in initial_candidates:
        current_c5 = compute_c5(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate.copy()
    
    # Early exit if target achieved
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    # Optimization loop with multiple restarts
    max_iter = 200  # Reduced iterations for faster inner optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select best candidate from initial set
        current_h = best_h.copy()
        
        # Local refinement with SLSQP
        result = minimize(
            objective, 
            current_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_refined = result.x
        else:
            h_refined = current_h
            
        # Ensure feasibility
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Evaluate refined solution
        refined_c5 = compute_c5(h_refined)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = h_refined.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
