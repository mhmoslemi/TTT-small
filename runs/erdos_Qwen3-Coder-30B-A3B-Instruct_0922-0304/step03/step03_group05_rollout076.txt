import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        # Interpolate to new resolution
        h_parent = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h = np.clip(h_parent, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multiple initialization strategies
    initial_hs = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_hs.append(h_parent.copy())
    
    # Random perturbations
    for _ in range(5):
        h_rand = h.copy()
        noise = np.random.normal(0, 0.1 * np.mean(h), n_points)
        h_rand += noise
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_hs.append(h_rand.copy())
    
    # Run optimization on all initializations
    for init_h in initial_hs:
        h_current = init_h.copy()
        
        # Local optimization with SLSQP
        for iter_num in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Optimize using SLSQP
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
                # Ensure feasibility
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
            else:
                # If optimization fails, use a small perturbation
                h_current += np.random.normal(0, 0.001, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
