import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Line search parameters
    max_line_search_iter = 20
    eta_init = 0.01
    backtracking_factor = 0.5
    
    # Perturbation parameters
    perturb_freq = 50
    perturb_magnitude = 1e-3
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        h_base = h.copy()
        base_c5 = current_c5
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
            
        # Line search for optimal step size
        eta = eta_init
        h_trial = h_base - eta * grad
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        trial_c5 = evaluate_c5(h_trial, dx)
        
        line_search_iter = 0
        while trial_c5 >= current_c5 and line_search_iter < max_line_search_iter:
            eta *= backtracking_factor
            h_trial = h_base - eta * grad
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            trial_c5 = evaluate_c5(h_trial, dx)
            line_search_iter += 1
            
        h = h_trial
        
        # Perturbation to escape local minima
        if no_improvement_count % perturb_freq == 0 and no_improvement_count > 0:
            magnitude = perturb_magnitude / (1 + iteration ** 0.5)
            noise = np.random.normal(0, magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
