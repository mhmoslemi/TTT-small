import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two overlapping pulses
        h = np.zeros(n_points)
        h[:n_points//4] = 1
        h[n_points//4:n_points//2] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    step_size = 0.01
    decay_factor = 0.5
    decay_interval = 200
    
    # Optimization loop
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and its maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        current_c5 = overlap[max_idx] * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= target_c5:
            break
            
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_plus = overlap_plus[np.argmax(overlap_plus)]
            
            grad[i] = (max_plus * dx - current_c5) / eps
            
        # Descent step
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adapt step size
        if iteration % decay_interval == 0 and iteration > 0:
            step_size *= decay_factor
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
