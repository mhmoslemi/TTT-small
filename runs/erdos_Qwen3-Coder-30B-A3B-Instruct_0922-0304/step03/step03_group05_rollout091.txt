import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Alternative strategy: Pulse-based representation
    # Represent h as a set of M pulses
    M = 5  # Number of pulses
    # Parameters: [height1, pos1, width1, height2, pos2, width2, ...]
    # We'll start with a simpler approach: optimize on a subset of variables
    
    best_c5 = float('inf')
    best_h = h.copy()

    # Simple gradient-based optimization with restarts
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Small random perturbation
        perturbation = np.random.normal(0, 0.005, n_points)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept better or randomly accept worse solutions
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5 or np.random.random() < 0.1:
            h = h_new
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
