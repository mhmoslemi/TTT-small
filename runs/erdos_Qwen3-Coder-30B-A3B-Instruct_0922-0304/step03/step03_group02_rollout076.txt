import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # Number of particles
    omega = 0.7  # Inertia weight
    phi_p = 1.5  # Cognitive parameter
    phi_g = 1.5  # Social parameter
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(m):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_c5(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
    
    # Initialize global best
    best_idx = np.argmin(pbest_c5)
    gbest = pbest[best_idx].copy()
    gbest_c5 = pbest_c5[best_idx]
    
    # PSO iterations
    max_iter = budget_s * 20  # Rough estimate of iterations we can do
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            velocities[i] = (omega * velocities[i] + 
                           phi_p * r1 * (pbest[i] - particles[i]) + 
                           phi_g * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to meet constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
                    
                    # Early exit
                    if gbest_c5 <= 0.3807:
                        break
                        
        iter_count += 1
    
    # Final evaluation for best solution
    final_c5 = evaluate_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
