import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # number of particles
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        best_h = np.random.uniform(0, 1, n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    best_c5 = evaluate_c5(best_h, dx)
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(m):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_c5.append(evaluate_c5(h, dx))
    
    gbest = best_h.copy()
    gbest_c5 = best_c5
    
    # PSO main loop
    max_iter = int(budget_s * 20)  # ~20 iterations per second
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (pbest[i] - particles[i]) + 
                            social * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Keep within bounds [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to satisfy sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
                    
                    if gbest_c5 <= 0.3807:
                        break
                        
        # Early termination check
        if gbest_c5 <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_c5, n_points)
