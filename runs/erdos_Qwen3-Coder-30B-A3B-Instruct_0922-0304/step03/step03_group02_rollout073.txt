import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    noise_std = 0.05
    
    # Initialize population
    population = []
    
    # If initial_h_values is available, use it as one of the starting individuals
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill the rest of the population with random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    def selection_tournament(population, fitnesses):
        selected = []
        for _ in range(len(population)):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual):
        if np.random.rand() > mutation_rate:
            return individual.copy()
        mutated = individual.copy()
        idx = np.random.randint(0, n_points)
        mutated[idx] += np.random.normal(0, noise_std)
        # Clip and rescale to meet constraints
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Local optimization
    def local_optimize(h):
        def objective(h_vec):
            return evaluate_c5(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h,method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 50, 'ftol': 1e-9})
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                return h_opt
        except:
            pass
        return h.copy()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_c5(individual, dx) for individual in population]
        
        # Find best individual in current generation
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = selection_tournament(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Keep top 50% of the best individuals
        sorted_indices = np.argsort(fitnesses)
        for i in range(population_size // 2):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Perform local optimization on top 10%
        top_count = max(1, population_size // 10)
        for i in range(top_count):
            if np.random.rand() < 0.5:  # Apply local search with some probability
                idx = sorted_indices[i]
                optimized = local_optimize(population[idx])
                fitness_opt = evaluate_c5(optimized, dx)
                if fitness_opt < fitnesses[idx]:
                    population[idx] = optimized.copy()
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx) if best_h is not None else float('inf')
    
    return (best_h.tolist() if best_h is not None else [0]*n_points, final_c5, n_points)
