import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constrain_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb_h(h, scale=0.01):
        h += np.random.normal(0, scale, n_points)
        return constrain_h(h)

    # Early exit if already good
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()

    # Use a hybrid approach: direct optimization + local search
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Try random perturbations to escape local minima
        if iteration % 20 == 0:
            h = perturb_h(h, 0.05)
        else:
            h = perturb_h(h, 0.01)

        # Evaluate and track best
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Early exit
        if best_c5 <= 0.3807:
            break

    # Final fine-tune with optimization
    if time.time() - start_time < budget_s - 2:
        from scipy.optimize import minimize

        def objective(h):
            return compute_c5(h)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]

        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            best_h = constrain_h(result.x)
            best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
