import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, sigma=0.05):
    for i in range(len(individual)):
        if np.random.rand() < mutation_rate:
            individual[i] += np.random.normal(0, sigma)
    return individual

def tournament_selection(population, fitnesses, tournament_size=3):
    selected_indices = []
    for _ in range(len(population) // 2):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_index)
    return [population[i] for i in selected_indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        h = repair_h(h, n_points)

    else:
        # Fallback: Create random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    
    # Fill rest of population
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    max_generations = 1000
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            fitness = evaluate_c5(h, dx)
            fitnesses.append(fitness)
            
        # Track best solution
        min_fitness_index = np.argmin(fitnesses)
        if fitnesses[min_fitness_index] < best_c5:
            best_c5 = fitnesses[min_fitness_index]
            best_h = population[min_fitness_index].copy()
            
        # Check for early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected_parents = tournament_selection(population, fitnesses)
        
        # Generate new population through crossover and mutation
        new_population = []
        
        # Elitism: add best individual
        new_population.append(best_h.copy())
        
        # Crossover and mutation
        for i in range(0, len(selected_parents), 2):
            if i + 1 < len(selected_parents):
                child1, child2 = crossover(selected_parents[i], selected_parents[i+1])
            else:
                child1, child2 = selected_parents[i].copy(), selected_parents[i].copy()
                
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.extend([child1, child2])
            
        # Maintain population size
        population = new_population[:population_size]
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
