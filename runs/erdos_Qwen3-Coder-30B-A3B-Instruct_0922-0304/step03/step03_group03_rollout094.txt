import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a basic initialization
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative update parameters
    epsilon = 0.1
    damping = 0.99
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Initialize update factors
        update_factors = np.ones(n_points)
        
        # Compute update factors based on overlap contributions
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # For each position, compute total update influence
        for i in range(n_points):
            contribution = 0.0
            # Consider all lags
            for k in range(-n_points + 1, n_points):
                if i + k >= 0 and i + k < n_points:
                    # Get the appropriate index in the full correlation array
                    idx = k + n_points - 1
                    contribution += exp_neg_S[idx] / (1.0 + exp_pos_S[idx])
            update_factors[i] = contribution
        
        # Apply multiplicative updates
        h *= update_factors * damping
        
        # Maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
