import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet decomposition
    wavelet = 'haar'
    max_level = int(np.log2(n_points)) - 1
    if max_level < 3:
        max_level = 3
    max_level = min(max_level, 6)
    
    # Compute wavelet coefficients
    coeffs = pywt.wavedec(h, wavelet, level=max_level)
    threshold = 0.1
    
    iteration = 0
    wavelet_iter = 0
    spatial_refine = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Wavelet domain optimization
        if wavelet_iter < 3:  # 3 wavelet iterations
            # Compute C5 gradient in wavelet domain
            h_temp = h.copy()
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Simple gradient approximation in spatial domain
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Convert to wavelet domain
            grad_coeffs = pywt.wavedec(grad, wavelet, level=max_level)
            
            # Update wavelet coefficients
            new_coeffs = []
            for i, c in enumerate(coeffs):
                if i == 0:  # Approximation coefficients
                    new_coeffs.append(c - 0.1 * grad_coeffs[i])
                else:  # Detail coefficients
                    new_coeffs.append(soft_threshold(c - 0.1 * grad_coeffs[i], threshold))
            
            # Reconstruct
            h_new = pywt.waverec(new_coeffs, wavelet)
            if len(h_new) > n_points:
                h_new = h_new[:n_points]
            elif len(h_new) < n_points:
                pad = n_points - len(h_new)
                h_new = np.pad(h_new, (0, pad), mode='constant')
            
            # Project back to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new.copy()
            wavelet_iter += 1
            
        # Spatial refinement every few iterations
        elif spatial_refine < 2:  # 2 spatial refinements
            # Get top features in wavelet domain
            wavelet_mags = []
            for i, c in enumerate(coeffs):
                if i > 0:  # Detailed coefficients only
                    wavelet_mags.extend(np.abs(c.flatten()))
            threshold_idx = int(0.8 * len(wavelet_mags))
            threshold_val = np.partition(wavelet_mags, threshold_idx)[threshold_idx]
            
            # Refine in spatial domain with selected points
            refinement_indices = []
            for i, c in enumerate(coeffs):
                if i > 0:
                    for j, val in enumerate(c.flatten()):
                        if np.abs(val) > threshold_val:
                            idx = j  # approximate
                            refinement_indices.append(idx)
            
            if len(refinement_indices) > 0:
                # Gradient descent on selected points
                h_refine = h.copy()
                for i in range(100):  # iterations per refinement
                    if time.time() - start_time > budget_s - 1:
                        break
                    corr = correlate(h_refine, 1 - h_refine, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    # Gradient approximation
                    grad_refine = np.zeros_like(h_refine)
                    eps = 1e-4
                    for idx in refinement_indices:
                        h_plus = h_refine.copy()
                        h_plus[idx] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad_refine[idx] = (c5_plus - max_overlap) / eps
                    
                    # Update
                    h_refine -= 0.01 * grad_refine
                    h_refine = np.clip(h_refine, 0, 1)
                    h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                
                h = h_refine.copy()
                spatial_refine += 1
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
