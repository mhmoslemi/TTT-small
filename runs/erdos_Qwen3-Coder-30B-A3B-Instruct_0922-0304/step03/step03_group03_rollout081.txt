import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, trace
from cvxpy import semidefinite

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Semi-definite programming approach
        try:
            # Variables
            h_vars = Variable(n_points)
            t = Variable()
            
            # Constraints
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp_sum(h_vars) == n_points / 2,
                t >= 0
            ]
            
            # Construct constraints using rotated second-order cone relaxation
            # For each lag k, compute overlap bounds
            max_lag = n_points - 1
            for k in range(-max_lag, max_lag + 1):
                if k >= 0:
                    # Positive lag case
                    for i in range(n_points - k):
                        # h[i] * (1 - h[i + k]) <--> 1 - h[i + k] + h[i] - h[i] * h[i + k]
                        # Using the constraint that h[i] * (1 - h[i + k]) <= h[i] + h[i + k] - 2 * h[i] * h[i + k]
                        constraints.append(h_vars[i] + h_vars[i + k] - 2 * h_vars[i] * h_vars[i + k] >= h_vars[i] * (1 - h_vars[i + k]))
                        
                        # For each k, we also add t >= sum of overlaps at this lag
                        # This is a conservative approach - we will minimize t subject to all overlaps <= t
                        if i == 0:
                            # Do simple approach for just 't' constraint
                            pass
                        
                else:
                    # Negative lag case - symmetric
                    k_abs = -k
                    for i in range(n_points - k_abs):
                        # Similar terms
                        constraints.append(h_vars[i] + h_vars[i + k_abs] - 2 * h_vars[i] * h_vars[i + k_abs] >= h_vars[i] * (1 - h_vars[i + k_abs]))
            
            # Optimization strategy: If SDP fails, fall back to direct gradient descent
            prob = Problem(Minimize(t), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    # Repair the sum
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to local search if SDP doesn't work
                step = 0.001
                for _ in range(20):
                    corr = correlate(h, 1 - h, mode="full")
                    max_overlap = np.max(corr) * dx
                    max_idx = np.argmax(corr)
                    
                    # Gradient by finite difference
                    grad = np.zeros_like(h)
                    eps = 1e-5
                    for i in range(n_points):
                        h_plus = h.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full") 
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                        
                    h -= step * grad
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient-based method
            step = 0.001

                max_idx = np.argmax(corr)

                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                h -= step * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
