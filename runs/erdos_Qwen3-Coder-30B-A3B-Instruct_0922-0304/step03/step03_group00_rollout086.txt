import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use SDP approach for tighter bounds
    max_iter = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create SDP formulation
        h_var = cp.Variable(n_points, name="h")
        C5_var = cp.Variable(name="C5")
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C5_var >= 0
        ]
        
        # Add overlap constraints
        for k in range(-n_points+1, n_points):
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in range(n_points-k)
                overlap_expr = cp.sum(cp.multiply(h_var[:n_points-k], 
                                                 1 - h_var[k:]))
            else:
                # h[i] * (1 - h[i+k]) for i in range(-k, n_points)
                overlap_expr = cp.sum(cp.multiply(h_var[-k:], 
                                                 1 - h_var[:n_points+k]))
            constraints.append(C5_var >= overlap_expr * dx)
        
        # Objective
        objective = cp.Minimize(C5_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                new_h = h_var.value
                new_c5 = C5_var.value
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
        except:
            # Fallback to gradient descent
            step = 0.001
            iterations = 100
            h_temp = h.copy()
            
            for _ in range(iterations):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute overlaps for gradient approximation
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_overlap = np.max(corr) * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Update h
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            current_c5 = np.max(correlate(h_temp, 1 - h_temp, mode="full")) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_temp.copy()
        
        # Perturb h slightly for diversity
        h = best_h.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation with original method
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
