import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective for scipy.optimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Define constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Memetic algorithm
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    
    # Create initial population
    population = []
    if 'initial_h_values' in globals():
        population.append(np.array(initial_h_values))
        population[-1] = np.clip(population[-1], 0, 1)
        population[-1] = population[-1] * (n_points / 2) / np.sum(population[-1])
    else:
        population.append(h)
    
    while len(population) < population_size:
        new_h = np.ones(n_points)
        new_h[n_points//2:] = 0
        new_h += np.random.normal(0, 0.05, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(individual, dx) for individual in population]
        
        # Find best in current population
        best_in_gen = min(fitness)
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[fitness.index(best_in_gen)].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection using tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.random() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
                    
            # Repair constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Local search on top 10% (with at least one individual)
        top_k = max(1, population_size // 10)
        top_indices = np.argsort(fitness)[:top_k]
        for idx in top_indices:
            h = population[idx].copy()
            
            # Local optimization with SLSQP
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                population[idx] = h
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
