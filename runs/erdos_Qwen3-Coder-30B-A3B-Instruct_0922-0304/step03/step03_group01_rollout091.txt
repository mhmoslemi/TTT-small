import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - structured pattern
        h = np.zeros(n_points)
        # Create a balanced step function with some randomness
        h[:n_points//3] = 1.0
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.0
        # Add noise to avoid perfect symmetry
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Gradient approximation using finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = objective(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad

    # Trust-region optimization with adaptive radius
    max_iter = 10000
    tr_radius = 0.1
    patience = 0
    max_patience = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        # Compute gradient
        grad = compute_gradient(h)
        
        # Determine step using trust region
        step = -tr_radius * grad / (np.linalg.norm(grad) + 1e-12)
        
        # Ensure step respects bounds
        h_candidate = h + step
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        candidate_c5 = objective(h_candidate)
        
        # Accept if improvement
        if candidate_c5 < current_c5:
            h = h_candidate
            tr_radius = min(tr_radius * 1.2, 0.5)  # Increase radius
            patience = 0
        else:
            tr_radius *= 0.5  # Decrease radius
            patience += 1
            
        # Perturbation if no improvement for too long
        if patience > max_patience:
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
