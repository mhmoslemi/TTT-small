import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

    max_iter = 5000

    # Best solution tracking
    best_c5 = float('inf')
    best_h = h.copy()

    # Using a population size λ = 100 and initial step size σ
    λ = 100

    # Create the CMA-ES optimizer
    es = CMAEvolutionStrategy(initial_mean, σ, {'popsize': λ, 'seed': seed})
    
    # Optimization loop
    for iteration in range(max_iter):
