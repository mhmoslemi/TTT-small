```python
import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - triangular
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def get_constraint_for_shift(h, k):
        # Compute the overlap for shift k
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        return corr[max_idx] * dx
    
    # Cutting-plane approach
    C5_var = cp.Variable()
    h_vars = cp.Variable(n_points)
    
    # Linearization approach
    constraints = []
    # Initial constraints
    for k in range(n_points):
        if k == 0:
            continue
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        # Approximate the nonlinear term with linearization
        agg = 0
        for i in range(n_points):
            if i + k < n_points:
                agg += h_vars[i] * (1 - h_vars[i + k])
        # Add constraint that this should not exceed current C5
        constraints.append(agg <= C5_var * dx)
    
    # Bounds
    constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
    constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
    
    # Integral constraint
    constraints.append(cp.sum(h_vars) == n_points / 2)
    
    # Objective: minimize C5
    objective = cp.Minimize(C5_var)
    
    # Initialize with current solution
    current_c5 = evaluate_c5(h, dx)
    
    # Use cutting plane with iterative LPs
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 1:
            break

        # Create problem with current constraints
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=CPLEX, time_limit=0.1, tol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h_sol = h_vars.value
                new_h_sol = np.clip(new_h_sol, 0, 1)
                new_h_sol = new_h_sol * (n_points / 2) / np.sum(new_h_sol)
                new_c5 = evaluate_c5(new_h_sol, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h_sol.copy()
                    
            # Add a new constraint for worst overlap
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            worst_k = max_idx - (n_points - 1)
            
            # Add a hard constraint on worst shift
            agg = 0
            for i in range(n_points):
                if i + worst_k >= 0 and i + worst_k < n_points:
                    agg += h_vars[i] * (1 - h_vars[i + worst_k])
            if worst_k > 0:
                constraints.append(agg <= C5_var * dx)
            else:
                # For negative shift --> reindex
                # easier to just use the full corr indexing
                pass
                
        except:
            # Fall-back to gradient descent: continue
            pass

        # Try some gradient descent refinement
        step = 0.01
        iters = 20
        h_temp = h.copy()
        for _ in range(iters):
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Compute gradient
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)

        h = h_temp
        
        # Evaluate new solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
