import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Two-phase optimization:
    # Phase 1: Particle Swarm Optimization (PSO) 
    # Phase 2: Local refinement with SLSQP
    
    # PSO Parameters
    swarm_size = 50
    max_iter_pso = 200
    inertia = 0.7
    social = 1.5
    cognitive = 1.5
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) - 0.5
    personal_best_positions = positions.copy()
    personal_best_values = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_value = float('inf')
    
    # Normalize initial positions
    for i in range(swarm_size):
        positions[i] *= (n_points / 2) / np.sum(positions[i])
        positions[i] = np.clip(positions[i], 0, 1)
    
    # PSO main loop
    for pso_iter in range(max_iter_pso):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        for i in range(swarm_size):
            # Evaluate fitness
            pos = positions[i]
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
            fitness = objective(pos)
            
            # Update personal best
            if fitness < personal_best_values[i]:
                personal_best_values[i] = fitness
                personal_best_positions[i] = pos.copy()
            
            # Update global best
            if fitness < global_best_value:
                global_best_value = fitness
                global_best_position = pos.copy()
                
        # Update swarm
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                social * r2 * (global_best_position - positions[i])
            )
            positions[i] += velocities[i]
            
            # Ensure bounds and normalization
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # If PSO did not find good solution, use fallback
    if global_best_value < best_c5:
        best_c5 = global_best_value
        best_h = global_best_position.copy()
    
    # Phase 2: Local refinement with SLSQP
    n_refinement_iters = 100
    for i in range(n_refinement_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use best solution from PSO as starting point
        h = best_h.copy()
        
        # Refine using SLSQP
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 50, 'ftol': 1e-9})
        
        if result.success:
            h = result.x
            
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        else:
            # Gradient-based refinement fallback
            step = 0.001
            steps = 50
            h_temp = h.copy()
            
            for _ in range(steps):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
            
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h.copy()
        best_c5 = objective(h)
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
