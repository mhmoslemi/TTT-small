import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def wavelet_optimize(h, dx, n_points, wavelet='haar'):
    # Initialize wavelet coefficients
    coeffs = pywt.wavedec(h, wavelet, level=6)
    coeffs_flat = np.concatenate(coeffs)
    
    # Optimizer parameters
    alpha = 0.1
    lambda_threshold = 0.1
    max_iter = 30
    best_c5 = float('inf')
    best_coeffs = coeffs_flat.copy()
    
    # Simple gradient descent in wavelet domain
    for i in range(max_iter):
        if i % 10 == 0 and i > 0:
            lambda_threshold *= 1.5
            
        # Compute current C5
        h_current = pywt.waverec([best_coeffs[:len(coeffs[0])], 
                                  best_coeffs[len(coeffs[0]):2*len(coeffs[0])], 
                                  best_coeffs[2*len(coeffs[0]):3*len(coeffs[0])], 
                                  best_coeffs[3*len(coeffs[0]):4*len(coeffs[0])], 
                                  best_coeffs[4*len(coeffs[0]):5*len(coeffs[0])], 
                                  best_coeffs[5*len(coeffs[0]):6*len(coeffs[0])], 
                                  best_coeffs[6*len(coeffs[0]):]], wavelet)
        
        h_current = h_current[:n_points]
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        c5_current = evaluate_c5(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_coeffs = coeffs_flat.copy()
            
        # Compute gradient in wavelet domain (simplified)
        grad = np.zeros_like(coeffs_flat)
        eps = 1e-5
        for j in range(len(coeffs_flat)):
            coeffs_perturb = coeffs_flat.copy()
            coeffs_perturb[j] += eps
            h_perturb = pywt.waverec([coeffs_perturb[:len(coeffs[0])], 
                                      coeffs_perturb[len(coeffs[0]):2*len(coeffs[0])], 
                                      coeffs_perturb[2*len(coeffs[0]):3*len(coeffs[0])], 
                                      coeffs_perturb[3*len(coeffs[0]):4*len(coeffs[0])], 
                                      coeffs_perturb[4*len(coeffs[0]):5*len(coeffs[0])], 
                                      coeffs_perturb[5*len(coeffs[0]):6*len(coeffs[0])], 
                                      coeffs_perturb[6*len(coeffs[0]):]], wavelet)
            h_perturb = h_perturb[:n_points]
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            c5_perturb = evaluate_c5(h_perturb, dx)
            grad[j] = (c5_perturb - c5_current) / eps
        
        # Update coefficients
        coeffs_flat -= alpha * grad
        coeffs_flat = soft_threshold(coeffs_flat, lambda_threshold)
        
    # Final reconstruction
    h_final = pywt.waverec([coeffs_flat[:len(coeffs[0])], 
                            coeffs_flat[len(coeffs[0]):2*len(coeffs[0])], 
                            coeffs_flat[2*len(coeffs[0]):3*len(coeffs[0])], 
                            coeffs_flat[3*len(coeffs[0]):4*len(coeffs[0])], 
                            coeffs_flat[4*len(coeffs[0]):5*len(coeffs[0])], 
                            coeffs_flat[5*len(coeffs[0]):6*len(coeffs[0])], 
                            coeffs_flat[6*len(coeffs[0]):]], wavelet)
    h_final = h_final[:n_points]
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_c5(h_final, dx)
    
    return h_final, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Wavelet-based optimization
    iterations = 3
    for i in range(iterations):
        if time.time() - start_time > budget_s - 5:
            break
        h_opt, c5_opt = wavelet_optimize(h, dx, n_points)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h = h_opt.copy()
    
    # Final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    target_sum = n_points / 2
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Refine with local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 10, 'ftol': 1e-6}
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
