import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def solve_qp_for_shift(h, shift_k):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize overlap at shift k
        # This is sum_{i=0}^{n-|k|-1} h[i] * (1 - h[i+k]) * dx
        # We reformulate this as: sum_{i=0}^{n-|k|-1} h[i] * (1 - h[i+k]) * dx 
        # Which is h^T * Q * h + c^T * h + constant, but we're solving for a specific shift k
        
        # For now, we'll directly create the constraints
        # We want to minimize the contribution of h[i] * (1 - h[i+shift_k]) 
        # for i from 0 to n_points - shift_k - 1

        # Let's build a quick quadratic objective
        # For shift k, index i in 0 to n_points-shift_k-1, we minimize h[i] * (1 - h[i+k]) * dx
        
        # Objective is sum of h[i] * (1 - h[i+shift_k]) * dx
        # This is a bilinear expression
        
        # Convert to QP: h[i] * (1 - h[i+shift_k]) = h[i] - h[i] * h[i+shift_k]
        # But since our variables are h[i], and we're minimizing h[i] * (1 - h[i+shift_k])
        # We don't directly solve this with QP. Instead, we use a constrained optimization over the worst shift
        
        # Reuse the current h values, run optimization pattern based on it
        
        # Let's do this by returning a new candidate solution 
        # which could be used in a line search momentum-like system
        return h_vars.value if h_vars.value is not None else h
    
    # Main optimization loop using QP approach
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find current maximum overlap shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)  # Convert index to shift
        
        # If the shift is within valid range
        if abs(max_k) < n_points:
            # Calls to solver can be expensive, so instead we use heuristics:
            # We'll try several small, adaptive steps towards minimizing that specific shift's contribution
            # This is a practical workaround
            
            # Simple heuristic: modify both h[i] and h[i+shift_k] to reduce their product
            h_temp = h.copy()
            
            # If shift is positive, try to decrease h[i] and h[i+shift_k] pairs
            # If shift is negative, adjust patterns in reverse

            # A pattern from optimization literature is to solve subproblems
            # We solve a simple optimization on the "worst" shift
            # Build a small modification 
            # But we'll go with a gradient-based smoothing method:
            
            # 1. Compute overlap contribution for shift k
            if abs(max_k) < n_points:
                # Try to minimize the contribution at shift k 
                # Identify the worst i indices
                indices = range(n_points - abs(max_k))
                if max_k < 0:
                    # h[i]*1-h[i-k] 
                    overlap_contributions = [h_temp[i] * (1 - h_temp[i - max_k]) for i in indices]
                else:
                    # h[i]*1-h[i+k]
                    overlap_contributions = [h_temp[i] * (1 - h_temp[i + max_k]) for i in indices]
                
                # Maximize the overlap at this index
                max_overlap_at_shift = max(overlap_contributions)
                max_idx_at_shift = max(overlap_contributions, key=lambda x: x) if overlap_contributions else 0
                
                if overlap_contributions:
                    # Find its index
                    max_contrib_idx = overlap_contributions.index(max_overlap_at_shift)
                    
                    # Take a gradient descent step, but only in areas where significant contribution occurs
                    step_size = 1e-4
                    h_temp[max_contrib_idx] *= (1 - step_size)
                    h_temp[max_contrib_idx] = np.clip(h_temp[max_contrib_idx], 0, 1)
                    
                    if max_k < 0:
                        h_temp[max_contrib_idx - max_k] *= (1 - step_size)
                        h_temp[max_contrib_idx - max_k] = np.clip(h_temp[max_contrib_idx - max_k], 0, 1)
                    else:
                        h_temp[max_contrib_idx + max_k] *= (1 - step_size)
                        h_temp[max_contrib_idx + max_k] = np.clip(h_temp[max_contrib_idx + max_k], 0, 1)
                    
                    # Renormalize the support
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
