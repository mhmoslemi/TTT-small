import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        
    # Repair initial h to satisfy sum constraint
    h = repair_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define objective function for CMA-ES
    def objective(h):
        h_repaired = repair_h(h, n_points)
        return evaluate_c5(h_repaired, dx)
    
    # Set up CMA-ES
    # Use initial_h_values as initial mean if available
    initial_mean = h.copy()
    es = CMAEvolutionStrategy(initial_mean, 0.1, {'popsize': 100, 'seed': seed})
    
    # Optimization loop
    max_iter = 10000  # CMA-ES will handle iterations efficiently
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Ask for new solutions
        solutions = es.ask()
        
        # Evaluate each solution
        fitnesses = []
        for sol in solutions:
            fitness = objective(sol)
            fitnesses.append(fitness)
        
        # Tell CMA-ES about fitness values
        es.tell(solutions, fitnesses)
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        current_best_h = solutions[current_best_idx]
        
        if current_best_fitness < best_c5:

            best_h = current_best_h.copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final repair and evaluation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
