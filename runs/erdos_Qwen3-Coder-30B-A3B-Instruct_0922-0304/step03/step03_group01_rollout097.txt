import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two pulses
        h = np.zeros(n_points)
        width = int(0.3 / dx)
        center1 = int(0.5 / dx)
        center2 = int(1.5 / dx)
        h[max(0, center1 - width//2):min(n_points, center1 + width//2)] = 0.5
        h[max(0, center2 - width//2):min(n_points, center2 + width//2)] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation with finite differences
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Line search for step size
    def line_search(h, grad, max_iter=20):
        alpha = 0.01
        min_alpha = 1e-6
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < evaluate_c5(h, dx):
                return alpha, h_new
            alpha *= 0.5
            if alpha < min_alpha:
                break
        return alpha, h
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Line search to find update step
        alpha, h_new = line_search(h, grad)
        
        h = h_new
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Periodic perturbations
        if no_improvement_count > 50:
            pert_mag = 1e-3 / (1 + iter_count**0.5)
            h += np.random.normal(0, pert_mag, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
