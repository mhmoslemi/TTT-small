import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy the constraint sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    h_new = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Either a small perturbation or reset
            if np.random.random() < 0.5:
                h_new[i] += np.random.normal(0, 0.05)
            else:
                h_new[i] = np.random.random()
    return repair_h(h_new, n_points)

def crossover_individuals(h1, h2, crossover_rate):
    """Perform crossover between two individuals"""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    point = np.random.randint(1, len(h1)-1)
    h1_new = np.concatenate([h1[:point], h2[point:]])
    h2_new = np.concatenate([h2[:point], h1[point:]])
    return h1_new, h2_new

def genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, 
                      max_generations, dx, start_time, budget_s):
    """Use genetic algorithm to search for better solutions"""
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.random(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_solution = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(-c5)  # Negative because we minimize
            
        # Selection (tournament selection)
        selected = []
        for _ in range(pop_size):
            competitor_indices = np.random.choice(pop_size, 3, replace=False)
            best_competitor = max(competitor_indices, key=lambda i: fitness_scores[i])
            selected.append(population[best_competitor])
        
        # Create offspring
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, pop_size-1)]
            child1, child2 = crossover_individuals(parent1, parent2, crossover_rate)
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            offspring.append(child1)
            if len(offspring) < pop_size:
                offspring.append(child2)
        
        # Replace population
        population = offspring
        
        # Track best
        for h in population:
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_solution = h.copy()
                
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_generations = 200
    
    # Run GA
    ga_h, ga_c5 = genetic_algorithm(n_points, pop_size, mutation_rate, 
                                    crossover_rate, max_generations, dx, 
                                    start_time, budget_s)
    
    if ga_h is not None and ga_c5 < evaluate_c5(h, dx):
        h = ga_h
        c5_bound = ga_c5
    else:
        c5_bound = evaluate_c5(h, dx)
    
    # Local refinement
    if time.time() - start_time < budget_s - 1:
        # Gradient-based refinement
        step = 0.005
        max_iter = 100
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < c5_bound:
                c5_bound = current_c5
    
    return (h.tolist(), c5_bound, n_points)
