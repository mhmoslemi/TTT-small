import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two pulses
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0
        h[3*n_points//4:] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 5000
    step_size = 0.01
    decay_factor = 0.5
    decay_interval = 200
    
    # Gradient computation and update
    def compute_gradient(h, dx):
        # Compute correlation between h and (1-h) 
        corr = correlate(h, 1 - h, mode="full")
        # Find maximum overlap lag
        max_idx = np.argmax(corr)
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - np.max(corr) * dx) / eps
        return grad
    
    # Descent step with feasibility repair
    def descent_step(h, grad, step_size, dx):
        # Gradient descent update
        h_new = h - step_size * grad
        # Project to feasible set [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current objective and gradient
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Step size decay
        if iteration % decay_interval == 0 and iteration > 0:
            step_size *= decay_factor
            
        # Update h
        h = descent_step(h, grad, step_size, dx)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
