import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # For the cutting plane method, we'll iterate and heuristically add constraints
    # We start with the standard LP solving approach but with added constraints
    try:
        # Attempt to use a solver like GLPK or similar for LPs
        solver = cp.GLPK
    except:
        solver = cp.CLARABEL  # Fallback to something else if GLPK is not available

    max_iter = 5000
    c5_bound = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current maximum overlap and corresponding shift k
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        k = max_idx - (n_points - 1)  # map index to actual shift

        # Update best solution if we're improving
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # We'll sample multiple shifts over time to increase diversity
        if iteration % 50 == 0:
            # Partially randomize shifts to avoid local minima
            shifts = []
            for _ in range(5):
                if np.random.rand() > 0.5:
                    shifts.append(np.random.randint(-n_points+1, n_points))
                else:
                    shifts.append(k)
            # Add some newly computed worst shifts from samples
            samples = np.random.choice(n_points, 10, replace=False)
            for s in samples:
                h_shift = np.roll(h, s)
                overlap = np.sum(h * (1 - h_shift)) * dx
                shifts.append(s - n_points + 1)
            shifts = list(set(shifts))  # Remove duplicates
        else:
            shifts = [k]  # Always include current worst
    
        # Try solving a relaxed problem with the added constraints
        # Construct shifted vectors and articulate constraints
        
        # Rebuild optimization model with additional constraints
        h_var = cp.Variable(n_points, name='h')
        
        # Equality constraint
        constraints = [cp.sum(h_var) == n_points / 2]
        # Bound constraints
        constraints.extend([h_var[i] >= 0 for i in range(n_points)])
        constraints.extend([h_var[i] <= 1 for i in range(n_points)])
        
        # Add the constraint from the most problematic shifts
        for shift in shifts[:3]:  # Limit to 3 added constraints per iteration
            
            # Vectorize the constraint manually
            # sum(h[i] * (1 - h[i + shift])) <= C5*dx 
            # For shift=k, we can formulate it carefully
            if shift >= 0:
                left_slice = h_var[:-shift] if shift != 0 else h_var[:]
                right_slice = h_var[shift:] if shift != 0 else h_var[:]
            else:
                left_slice = h_var[-shift:] if shift != 0 else h_var[:]
                right_slice = h_var[:shift] if shift != 0 else h_var[:]
                
                # In case of negative shift, make sure we don't index incorrectly
                if shift < 0:
                    left_slice = h_var[:shift]
                    right_slice = h_var[-shift:]
                    
            # Clip both slices to the same size to avoid mismatch
            min_len = min(len(left_slice), len(right_slice))
            left_slice = left_slice[:min_len]
            right_slice = right_slice[:min_len]
            
            # This gets tricky as we're adding a quadratic constraint approximate linearization

            # We solve a simpler problem now for new vector h
            
            # So instead, directly optimize in a reduced and sorted way
            # Construct constraint based on h shifted by k
            if shift < 0:
                shifted_h = np.concatenate([h[shift:], np.zeros(-shift)])
                corr_val = np.sum(h * (1 - shifted_h)) * dx
            else:
                shifted_h = np.concatenate([np.zeros(shift), h[:-shift]])
                corr_val = np.sum(h * (1 - shifted_h)) * dx
            
            # We want to lower the value of this constraint, so just add one constraint
            if shift >= 0:
                left_term = h_var[:-shift] if shift != 0 else h_var[:]

                left_term = left_term[:len(right_term)]

                right_term = right_term[:len(left_term)]

        corr = correlate(h, 1 - h, mode="full")

        # Update h using gradient descent with constraints on SLSQP as capability
        try:
            # Update h with a simplex or modified approach  
            # Try a gradient method with step size
            step = 0.005
            
            # Gradient direction: finite difference approximation of the constraint on max 
            h_new = h.copy()
            eps = 1e-4
            
            # This is a very aggressive descent step for now - in practice you'd do LP or better optimization
            # But forcing sinusoidal perturbations might work as well
            
            # Perform more advanced descent if feasible (skip in full iteration)
            # Only do detailed steps on small number of iterations to speed up
            if iteration % 20 == 0:         
                grad = np.zeros(n_points)

                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Descend along gradient
                h_new = h - step * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new[:]
                
            else:
                # Regular gradient
                h = np.clip(h + 0.005 * (np.random.rand(n_points) - 0.5), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            
        except:
            # Fallback to basic update if required
            h = np.clip(h + 0.005 * (np.random.rand(n_points) - 0.5), 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
