import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with two pulses
        h = np.zeros(n_points)
        center1, center2 = n_points // 4, 3 * n_points // 4
        width = n_points // 10
        for i in range(max(0, center1 - width//2), min(n_points, center1 + width//2)):
            h[i] = 0.5
        for i in range(max(0, center2 - width//2), min(n_points, center2 + width//2)):
            h[i] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization settings
    max_iter = 2000
    epsilon = 1e-5
    initial_eta = 0.01
    backtracking_factor = 0.5
    max_backtracking = 20
    
    # Track improvements
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        h_backup = h.copy()
        for i in range(n_points):
            h_plus = h_backup.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / epsilon
            
        # Line search
        eta = initial_eta
        h_candidate = h - eta * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        for _ in range(max_backtracking):
            candidate_c5 = evaluate_c5(h_candidate, dx)
            if candidate_c5 < current_c5:
                h = h_candidate.copy()
                break
            else:
                eta *= backtracking_factor
                h_candidate = h - eta * grad
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        else:
            # If line search failed, use full step anyway
            h = h_candidate.copy()
            
        # Projection to maintain sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbation if no improvement in last 50 steps
        if iteration - last_improvement > 50:
            noise_magnitude = 1e-3 / np.sqrt(iteration + 1)
            noise = np.random.normal(0, noise_magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
