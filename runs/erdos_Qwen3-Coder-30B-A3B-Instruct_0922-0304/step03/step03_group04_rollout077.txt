import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for resolution levels
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    max_iter_per_res = 100
    population_size = 20
    
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    # Multi-resolution optimization
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if res_idx == 0:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.ones(n_points) * 0.5
        else:
            # Up-sample previous solution
            h_prev = best_h
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_prev)), h_prev)
            # Add some noise for exploration
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Define constraints
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Differential Evolution for global search
        de_result = differential_evolution(
            objective,
            bounds,
            seed=seed + res_idx,
            popsize=population_size,
            maxiter=max_iter_per_res // 5,
            disp=False
        )
        
        if de_result.success:
            h = de_result.x
        else:
            # Fallback to SLSQP if DE fails
            try:
                result = minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': max_iter_per_res // 5}
                )
                if result.success:
                    h = result.x
            except:
                pass
        
        # Refinement with SLSQP
        for _ in range(max_iter_per_res // 2):
            if time.time() - start_time > budget_s - 1:
                break
                
            try:
                result = minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 5}
                )
                if result.success:
                    h = result.x
                else:
                    break
            except:
                break
                
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit
            if best_c5 <= 0.3807:
                break
                
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Perturb slightly if needed
        if res_idx > 0:
            if np.random.rand() > 0.7:
                perturbation = np.random.normal(0, 0.01, n_points)
                h = h + perturbation
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, 2.0 / len(best_h))
        return (best_h.tolist(), final_c5, len(best_h))
    else:
        # Fallback
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.ones(70) * 0.5
        dx = 2.0 / 70
        final_c5 = evaluate_c5(h, dx)
        return (h.tolist(), final_c5, 70)
