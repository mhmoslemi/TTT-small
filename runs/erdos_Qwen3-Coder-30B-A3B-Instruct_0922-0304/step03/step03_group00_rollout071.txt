import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: clip and normalize"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient of overlap w.r.t. h for shift k"""
    grad = np.zeros(n_points)
    # For shift k, we have terms h[i](1-h[i+k]) and h[i+k](1-h[i])
    # gradient of h[i](1-h[i+k]) w.r.t. h[i] = 1-h[i+k]
    # gradient of h[i](1-h[i+k]) w.r.t. h[i+k] = -h[i]
    for i in range(n_points):
        if i + k >= 0 and i + k < n_points:
            grad[i] += 1 - h[i + k]
        if i - k >= 0 and i - k < n_points:
            grad[i] -= h[i - k]
    return grad * dx

def adaptive_step_size(current_c5, new_c5, step_size, improvement_threshold=0.0001):
    """Adaptively adjust step size based on improvement"""
    if new_c5 < current_c5 - improvement_threshold:
        return min(step_size * 1.2, 0.1)
    elif new_c5 > current_c5:
        return max(step_size * 0.5, 1e-6)
    else:
        return max(step_size * 0.9, 1e-6)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = (i + 1) / (mid + 1)
            h[n_points - 1 - i] = (i + 1) / (mid + 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent
    max_iter = 10000
    step_size = 0.01
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find shift that gives maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        # Convert index to shift k
        k = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, k, n_points, dx)
        
        # Update h
        h_new = h - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Adapt step size
        step_size = adaptive_step_size(best_c5, new_c5, step_size)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Sometimes accept worse steps to escape local minima
            if np.random.rand() < 0.01:
                h = h_new.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
