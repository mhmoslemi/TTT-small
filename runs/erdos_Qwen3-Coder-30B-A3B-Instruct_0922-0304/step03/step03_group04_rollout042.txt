import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Spectral initialization
    def get_spectral_features(h):
        H = fft(h)
        # Find dominant frequencies (ignore DC)
        mag_H = np.abs(H)
        # Only consider frequencies up to Nyquist
        freq_indices = np.argsort(mag_H[1:n_points//2])[::-1][:3]
        return freq_indices + 1  # Add 1 to skip DC component
    
    def mutate_h(h, freq_indices, strength=0.1):
        # Create a copy and make some changes in frequency domain
        h_mut = h.copy()
        H = fft(h_mut)
        
        # Randomly perturb the specified frequency bins
        for idx in freq_indices:
            noise = np.random.normal(0, strength)
            H[idx] += noise
            # Also modify the conjugate symmetric tuple
            if idx < n_points - idx:
                H[n_points - idx] += noise
        
        # Inverse FFT
        h_new = np.real(ifft(H))
        # Ensure bounds and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Initialize with 50 candidates using spectral mutation
    candidates = []
    best_indices = get_spectral_features(h)
    
    # Create initial diverse set of candidates
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        h_new = mutate_h(h, best_indices, strength=0.05)
        if np.all(h_new >= 0) and np.all(h_new <= 1) and abs(np.sum(h_new) * dx - 1.0) < 1e-3:
            candidates.append(h_new)
    
    # Evaluate all initial candidates
    for candidate in candidates:
        c5 = evaluate_c5(candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate.copy()
        if best_c5 <= 0.3807:
            break
    
    # Now perform local refinement on the best solutions
    n_refinement = min(20, len(candidates)) if len(candidates) > 0 else 5
    
    # Select the top candidates for refinement
    candidate_scores = []
    for candidate in candidates:
        c5 = evaluate_c5(candidate, dx)
        candidate_scores.append((c5, candidate))
    candidate_scores.sort(key=lambda x: x[0])
    
    top_candidates = [cand for _, cand in candidate_scores[:n_refinement]]
    
    # Use NLopt for local search (If available, else fallback to minimize)
    # For this solution, we will use scipy's minimize with Nelder-Mead
    for _ in range(500):  # Max iterations for local refinement
        if time.time() - start_time > budget_s - 10:
            break
            
        # Pick top 5 candidates and refine them
        for i in range(min(5, len(top_candidates))):
            candidate = top_candidates[i]
            # Define objective function
            def obj(h):
                return evaluate_c5(h, dx)
            
            # Try several local optimizations
            for _ in range(10):
                if time.time() - start_time > budget_s - 10:
                    break
                
                # Perturb the candidate slightly
                h_perturb = np.clip(candidate + np.random.normal(0, 0.02, n_points), 0, 1)
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                
                # Nelder-Mead local search
                result = minimize(obj, h_perturb, method='Nelder-Mead', 
                                  options={'maxiter': 50, 'adaptive': True})
                
                if result.success:
                    h = result.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    c5 = evaluate_c5(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        if best_c5 <= 0.3807:
                            return (best_h.tolist(), best_c5, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
