import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape + noise
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, np.linspace(1, 0, n_points//2)])
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project(h):
        # Ensure h is in [0,1] and sum equals n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def objective(h):
        return compute_c5(h)

    # Initial population of diverse solutions
    population = [h.copy()]
    for _ in range(4):
        h_pert = h + np.random.normal(0, 0.05, n_points)
        population.append(project(h_pert))

    best_c5 = float('inf')
    best_h = h.copy()

    # Local optimization with multi-start approach
    for i, h_start in enumerate(population):
        h_current = h_start.copy()
        for iteration in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = project(result.x)
            else:
                h_current += np.random.normal(0, 0.01, n_points)
                h_current = project(h_current)
                
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            if best_c5 <= 0.3807:
                break

    # Basin hopping with adaptive perturbation
    h_current = best_h.copy()
    step_size = 0.1
    cooling_rate = 0.995
    max_basin_iter = 1000
    basin_iter = 0
    
    while basin_iter < max_basin_iter and time.time() - start_time < budget_s - 1:
        # Random perturbation
        h_pert = h_current + np.random.normal(0, step_size, n_points)
        h_pert = project(h_pert)
        
        # Local optimization
        result = minimize(
            objective, 
            h_pert, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            h_candidate = project(result.x)
        else:
            h_candidate = h_pert
            
        # Accept or reject based on C5
        current_c5 = compute_c5(h_current)
        candidate_c5 = compute_c5(h_candidate)
        
        if candidate_c5 < current_c5:
            h_current = h_candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h_current.copy()
                
        # Adaptive step size decay
        step_size *= cooling_rate
        basin_iter += 1

    # Refine final solution with stringent tolerance
    final_result = minimize(
        objective, 
        best_h,
        method='SLSQP', 
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if final_result.success:
        final_h = project(final_result.x)
        final_c5 = compute_c5(final_h)
    else:
        final_h = best_h.copy()
        final_c5 = compute_c5(final_h)

    return (final_h.tolist(), final_c5, n_points)
