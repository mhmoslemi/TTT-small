import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - create a balanced pattern
        h = np.ones(n_points)
        # Alternate high/low values to create a more balanced start
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 0.6
            else:
                h[i] = 0.4
        # Normalize
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO Parameters
    swarm_size = 50
    max_iter = 1000
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Initialize PSO swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    for i in range(swarm_size):
        # Random initialization with constraint
        h_temp = np.random.rand(n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        swarm.append(h_temp.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        eval_val = evaluate_c5(h_temp, dx)
        pbest.append(h_temp.copy())
        pbest_vals.append(eval_val)
        if eval_val < gbest_val:
            gbest = h_temp.copy()
            gbest_val = eval_val
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply boundary constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            # Apply sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_val = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                
            # Update global best
            if current_val < gbest_val:
                gbest = swarm[i].copy()
                gbest_val = current_val

    # Update best solution found by PSO
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()
    
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Use the best PSO solution as starting point
        h = best_h.copy()
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 1000, 'ftol': 1e-9})
        
        if result.success:
            h = result.x
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Ensure feasibility
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, dx)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
