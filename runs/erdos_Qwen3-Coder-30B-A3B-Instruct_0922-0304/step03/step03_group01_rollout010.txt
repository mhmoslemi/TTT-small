import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_size = pop_size // 5
    
    # Create initial population
    def create_population(size):
        population = []
        for _ in range(size):
            individual = h.copy() + np.random.normal(0, 0.02, n_points)
            individual = np.clip(individual, 0, 1)
            individual = individual * (n_points / 2) / np.sum(individual)
            population.append(individual)
        return population
    
    def repair_individual(individual):
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    # Fitness function
    def fitness(individual):
        return -evaluate_c5(individual, dx)
    
    # Selection using tournament
    def tournament_selection(population, fitnesses, k=3):
        selected = []
        for _ in range(len(population)):
            idxs = np.random.choice(len(population), k)
            best_idx = idxs[np.argmax([fitnesses[i] for i in idxs])]
            selected.append(population[best_idx].copy())
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Mutation
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.normal(0, 0.03)
        return repair_individual(individual)
    
    # Main GA loop
    population = create_population(pop_size)
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < (budget_s * 0.8) and no_improvement_count < 50:
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Track best individual in this generation
        best_in_gen_idx = np.argmax(fitnesses)
        if -fitnesses[best_in_gen_idx] < best_c5:
            best_c5 = -fitnesses[best_in_gen_idx]
            best_h = population[best_in_gen_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create offspring
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[-elite_size:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Truncate to exact population size
        population = new_population[:pop_size]
        generation += 1
    
    # Local refinement with gradient-based method
    def gradient_descent_refinement(start_h, max_iter=100):
        h = start_h.copy()
        step = 0.005
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - max_overlap) / eps
            
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        return h
    
    # Run local refinement on best solution from GA
    refined_h = gradient_descent_refinement(best_h)
    final_c5 = evaluate_c5(refined_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = refined_h
    
    return (best_h.tolist(), best_c5, n_points)
