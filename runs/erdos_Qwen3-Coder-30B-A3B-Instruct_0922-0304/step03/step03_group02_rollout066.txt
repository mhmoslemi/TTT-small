import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Higher resolution for better accuracy
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - smooth step function
        h = np.ones(n_points)
        half = n_points // 2
        h[half:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 1000
    
    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    # Define bounds for each h[i] 
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Main optimization loop with mixed strategies
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different optimization strategies
        if iteration % 3 == 0:  # SLSQP with gradient
            result = minimize(
                evaluate_c5,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-10}
            )
        elif iteration % 3 == 1:  # L-BFGS without constraints
            result = minimize(
                evaluate_c5,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 30, 'ftol': 1e-10}
            )
        else:  # Random perturbation + projection
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            result = type('obj', (object,), {'x': h_perturbed, 'success': True})
            
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate current solution
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        else:
            # Fallback to previous best if optimization fails
            h = best_h.copy()
    
    # Final refinement with more precise method
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
