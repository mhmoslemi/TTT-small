import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_algorithm(seed, budget_s, n_points):
    np.random.seed(seed)
    dx = 2.0 / n_points
    pop_size = 50
    generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.7
    elitism_rate = 0.2
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population[0] = np.clip(population[0], 0, 1)
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
    else:
        population = [np.ones(n_points) * 0.5]
    
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (lower overlap is better)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if -fitness[0] < best_c5:
            best_c5 = -fitness[0]
            best_h = population[0].copy()
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
        
        # Selection
        elite_count = int(elitism_rate * pop_size)
        elite = population[:elite_count]
        selected = population[:pop_size//2]
        
        # Generate new population
        new_population = elite[:]
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_select(selected, fitness)
            parent2 = tournament_select(selected, fitness)
            
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child1 = mutate(child1)
            if np.random.rand() < mutation_rate:
                child2 = mutate(child2)
            
            # Repair
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    return best_h, best_c5

def tournament_select(population, fitness, k=3):
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmax([fitness[i] for i in indices])]
    return population[best_idx]

def crossover(h1, h2):
    # Uniform crossover
    crossover_point = np.random.randint(1, len(h1))
    child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return child1, child2

def mutate(h):
    # Gaussian mutation
    mutation_strength = 0.1
    noise = np.random.normal(0, mutation_strength, len(h))
    h = h + noise
    return h

def local_refinement(h, n_points, dx):
    """Use gradient-based method to refine"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simple gradient ascent on a few iterations
    step = 0.001
    for _ in range(100):
        # Compute gradient (approximated for discrete case) 
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(corr)
        grad = np.zeros_like(best_h)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            eps = 1e-4
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update
        best_h -= step * grad
        best_h = repair_h(best_h, n_points)
        current_c5 = evaluate_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    
    # Use GA to explore solution space
    h, c5_bound = genetic_algorithm(seed, budget_s * 0.8, n_points)
    
    # Local refinement
    h, c5_bound = local_refinement(h, n_points, dx)
    
    # Final check
    h = repair_h(h, n_points)
    final_c5 = evaluate_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
