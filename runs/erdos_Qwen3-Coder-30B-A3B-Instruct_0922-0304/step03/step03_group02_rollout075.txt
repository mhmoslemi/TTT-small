import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = budget_s * 20  # Rough estimate for iterations
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    else:
        h_ref = np.ones(n_points) * 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        h = h_ref + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_c5(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
    
    # PSO parameters
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate current particle
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
            
            # Early exit if target achieved
            if gbest_c5 <= 0.3807:
                break
                
        if gbest_c5 <= 0.3807:
            break
    
    # Ensure final values meet constraints
    final_h = gbest.copy()
    final_c5 = gbest_c5
    
    return (final_h.tolist(), final_c5, n_points)
