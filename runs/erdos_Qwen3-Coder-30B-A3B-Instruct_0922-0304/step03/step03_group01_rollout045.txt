import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: two pulses
        h = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 6
        h[mid-width:mid+width] = 0.5
        h[mid+2*width:mid+4*width] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution  
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 1000
    step_size = 0.01
    decay_rate = 0.995
    last_improvement = 0
    patience = 50
    perturbation_magnitude = 1e-3

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps

        # Line search for step size
        eta = step_size
        for _ in range(20):
            h_new = h - eta * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)  # Rescale
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < current_c5:
                h = h_new
                break
            else:
                eta *= 0.5
                
        # Perturb periodically if no improvement
        if iteration - last_improvement > patience:
            noise = np.random.normal(0, perturbation_magnitude / (1 + iteration**0.5), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Decay step size
        step_size *= decay_rate

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
