import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # SDP-based optimization
    # Variables
    h_var = Variable(n_points)
    C5 = Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        sum(h_var) == n_points / 2,
        C5 >= 0
    ]
    
    # Add quadratic constraints using SDP relaxation
    # For each k, we need to enforce:
    # C5 >= sum h[i] * (1 - h[i+k]) over all valid i
    
    # Create a matrix for the quadratic terms using SDP relaxation
    # This introduces auxiliary variables to handle product terms
    # We'll reformulate overlapping terms into second-order cone constraints
    
    # Simpler approach: Direct optimization on a small set of shifts
    # Use the initial solution as a starting point
    best_c5 = float('inf')
    best_h = h.copy()
    best_h_arr = h.copy()
    
    def objective(h_vals):
        return evaluate_c5(h_vals, dx)
    
    # Various fine-tuning approaches:
    # 1. Local gradient descent with manual gradient
    # 2. Modified approach for better smoothness
    
    # First, try to improve upon initialization
    h_current = best_h.copy()
    
    # Small gradient steps with constraint projection
    max_steps = 10000
    step_size = 0.01
    for step in range(max_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Estimate gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
            
        # Gradient descent update
        h_current -= step_size * grad
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Evaluation
        current_c5 = evaluate_c5(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_arr = h_current.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h_arr, dx)
    
    return (best_h_arr.tolist(), final_c5, n_points)
