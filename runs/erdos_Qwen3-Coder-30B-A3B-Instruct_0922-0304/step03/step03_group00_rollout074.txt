import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_shift_idx, n_points):
    grad = np.zeros_like(h)
    # Compute the contribution to gradient from the maximum overlap
    # For each position i, contributions are from h[i] and h[i + k] terms
    # In the full correlation, only the relevant entries contribute to the gradient
    # So we compute for the specific shift k that gives maximum overlap
    
    # The gradient is complex to derive directly, we use finite differences to approximate
    # But to be more precise, we should compute the gradient for the identified shift:
    
    k = max_shift_idx - (n_points - 1)
    # k can be negative, positive, or zero (but normalized)
    # We compute: grad[i] = d/dh[i] (sum over all j of h[j](1-h[j+k]))
    # This is non-zero when i = j or i = j + k
    # Simplified: grad[i] = (1 - h[i + k]) + (1 - h[i - k]) 
    # But care needed with bounds
    
    # For simplicity, let's compute this using a dual approach:
    # compute manually the contribution for the identified shift
    
    # Grad[i] = (1 - h[i + shift]) * indicator
    # but only for valid indices, and we will accumulate the appropriate terms
    # Actually, let me make a different and efficient approach:
    # When we have overlap = Σ h[i](1 - h[i + k]), the gradient of this w.r.t. h[i] is 1 - h[i + k] (when i+k is valid)
    # and also depend on h[i] itself when computing the analogous derivative
    
    # We derive analytically for relevant positions:
    # ∂/∂h[i] ( h[j](1-h[j+k]) ) = δ[i,j] (1 - h[j+k]) + h[j] δ[i,j+k] (-1)
    # So total per h[i]:
    # grad[i] = (1 - h[i + k]) if j = i, and also -h[i - k] if j + k = i
    # hence when shift k is positive: ...
    # Let's compute from first principles:
    
    # For a given k in (-n+1, n-1), we consider i in [0, n-1], and we want the derivative of:
    # Σ h[i](1 - h[i+k]) (possible khi i+k < n)
    
    # hence d/dh[i] of this integral term:
    # for the term h[i](1-h[i+k]):
    # grad[i] = (1 - h[i + k]) if i+k < n (i+k >= 0 also)
    # grad[i] = - h[i - k] if i - k >= 0 and i-k < n (note if k is -v, then -k is v)
    # need handling depending on k:
    
    # simplify and compute incrementally once
    if k >= 0:
        # contribution from term h[i](1 - h[i + k])
        grad[:-k] += 1 - h[k:]
        grad[k:] -= h[:-k]
    else:
        # k < 0: contribution from h[i](1 - h[i + k])
        grad[-k:] += 1 - h[:k]
        grad[:k] -= h[-k:]
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # because sum(h) * dx = 1
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        # Initialize with a triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid + 1):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h = project_h(h, target_sum)
    
    # Early prunes
    best_c5 = float('inf')
    best_h = h.copy()

    adapt_lr = 0.01
    lr_decay = 0.9
    
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Note: evaluate exact maximum overlap by computing cross-correlation
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        max_shift_idx = np.argmax(overlap)
        current_c5 = max_overlap * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Compute gradient
        grad = compute_gradient(h, dx, max_shift_idx, n_points)
        
        # Update step
        h = h - adapt_lr * grad
        
        # Project back
        h = project_h(h, target_sum)
        
        # Adjust learning rate
        adapt_lr *= lr_decay
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
