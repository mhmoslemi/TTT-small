import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_erdos_solution(h, dx):
    return evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        h[20:30] = 0.5
        h[40:50] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation with finite differences and line search
    def compute_gradient(h, eps=1e-4):
        current_c5 = evaluate_c5(h, dx)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        return grad
    
    # Line search to find optimal step size
    def line_search(h, grad, max_iter=20, alpha=0.01, beta=0.5):
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)

        # Perform line search
        h_new = line_search(h, grad)
        
        # Update h
        h = h_new

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5

            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
