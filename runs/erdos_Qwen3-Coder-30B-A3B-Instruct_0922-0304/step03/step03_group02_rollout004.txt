import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # PSO parameters
    n_points = 100
    n_particles = 50
    max_iter = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    dx = 2.0 / n_points
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.random.rand(n_points)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        # Start with h0 or random
        h = h0.copy()
        if _ > 0:
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    pbest_positions = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Track global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest_positions[i] - particles[i]) +
                social * r2 * (gbest_position - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i].copy()
                    
        # Early termination
        if gbest_score <= 0.3807:
            break
    
    # Return best solution
    final_c5 = evaluate_c5(gbest_position, dx)
    
    return (gbest_position.tolist(), final_c5, n_points)
