import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_erdos_solution(h):
    # Using predefined evaluation function
    return evaluate_c5(h, 2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[20:30] = 0.5
        h[40:50] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 10000
    epsilon = 1e-6
    initial_eta = 0.01
    backtrack_factor = 0.5
    max_backtrack = 20
    perturb_freq = 50
    perturb_magnitude = 1e-3
    
    def compute_gradient(h, eps):
        # Compute gradient via finite difference
        grad = np.zeros_like(h)
        base_val = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Clip to maintain feasibility
            h_plus = np.clip(h_plus, 0, 1)
            # Rescale
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            val_plus = evaluate_c5(h_plus, dx)
            grad[i] = (val_plus - base_val) / eps
            
        return grad

    def line_search(h, grad, eta):
        # Backtracking line search for step size
        for _ in range(max_backtrack):
            h_new = h - eta * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_val = evaluate_c5(h_new, dx)
            if new_val < evaluate_c5(h, dx):
                return h_new, new_val
            eta *= backtrack_factor
        return h, evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute max overlap for this h
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradient(h, epsilon)
        
        # Line search for optimal step size
        eta = initial_eta
        h_new, new_c5 = line_search(h, grad, eta)
        
        # Update h
        h = h_new
        
        # Periodic perturbation to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            noise = np.random.normal(0, perturb_magnitude / (1 + iteration**0.5), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation with full function
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
