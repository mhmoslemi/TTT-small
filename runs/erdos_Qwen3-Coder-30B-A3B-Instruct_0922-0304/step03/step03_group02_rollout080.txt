import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES parameters
    lambda_ = 100  # Population size
    mu = 50        # Number of parents
    sigma = 0.1    # Initial step size
    
    # Initialize mean and covariance matrix
    mean = h.copy()
    C = np.eye(n_points)
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Main CMA-ES loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sample offspring
        samples = []
        for _ in range(lambda_):
            z = np.random.randn(n_points)
            x = mean + sigma * np.dot(C, z)
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)  # Ensure sum constraint
            samples.append(x)
        
        # Evaluate fitness
        fitness = []
        for sample in samples:
            f = evaluate_c5(sample, dx)
            fitness.append(f)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_indices = sorted_indices[:mu]
        best_samples = [samples[i] for i in best_indices]
        best_fitness = [fitness[i] for i in best_indices]
        
        # Update mean
        mean = np.zeros(n_points)
        for i in range(mu):
            mean += weights[i] * best_samples[i]
        
        # Update covariance matrix
        C = np.zeros((n_points, n_points))
        for i in range(mu):
            diff = best_samples[i] - mean
            C += weights[i] * np.outer(diff, diff)
        
        # Update step size
        sigma *= np.exp(0.3 * (np.mean(best_fitness) - best_c5) / np.std(best_fitness) / 100)
        
        # Update best solution
        current_best_idx = np.argmin(best_fitness)
        current_c5 = best_fitness[current_best_idx]
        current_h = best_samples[current_best_idx]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
