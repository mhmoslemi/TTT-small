import numpy as np
from scipy.signal import correlate
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def wavelet_optimize(h, dx, n_points, max_iter=30):
    """Optimize using wavelet domain representation"""
    # Use Haar wavelets (ensure at least 6 levels)
    levels = int(np.log2(n_points))
    if levels < 6:
        levels = 6
    try:
        # Fast wavelet transform
        coeffs = pywt.wavedec(h, 'haar', level=levels)
        # Flatten coefficients
        flat_coeffs = pywt.vector_to_array(coeffs)
    except:
        # Fallback to basic approach if wavelet fails
        return h
        
    # Initialize
    wavelet_coeffs = flat_coeffs.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    learning_rate = 0.1
    threshold = 0.1
    for iteration in range(max_iter):
        # Reconstruct h from wavelet coefficients
        try:
            coeffs_recon = pywt.array_to_vector(wavelet_coeffs, coeffs)
            h_recon = pywt.waverec(coeffs_recon, 'haar')
            if len(h_recon) > n_points:
                h_recon = h_recon[:n_points]
            elif len(h_recon) < n_points:
                pad_len = n_points - len(h_recon)
                h_recon = np.pad(h_recon, (0, pad_len), mode='constant')
            h_temp = h_recon
        except:
            h_temp = h.copy()
        
        # Clip and normalize
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        # Compute current C5
        current_c5 = evaluate_c5(h_temp, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_temp.copy()
        
        # Compute gradient estimation in wavelet domain
        gradient = np.zeros_like(wavelet_coeffs)
        eps = 1e-5
        for i in range(len(wavelet_coeffs)):
            perturbed_coeffs = wavelet_coeffs.copy()
            perturbed_coeffs[i] += eps
            try:
                coeffs_pert = pywt.array_to_vector(perturbed_coeffs, coeffs)
                h_pert = pywt.waverec(coeffs_pert, 'haar')
                if len(h_pert) > n_points:
                    h_pert = h_pert[:n_points]
                elif len(h_pert) < n_points:
                    pad_len = n_points - len(h_pert)
                    h_pert = np.pad(h_pert, (0, pad_len), mode='constant')
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                c5_pert = evaluate_c5(h_pert, dx)
                gradient[i] = (c5_pert - current_c5) / eps
            except:
                pass
        
        # Update wavelet coefficients with gradient
        wavelet_coeffs = wavelet_coeffs - learning_rate * gradient
        # Apply soft thresholding
        wavelet_coeffs = soft_threshold(wavelet_coeffs, threshold)
        
        # Decrease threshold if stagnating
        if iteration > 5 and iteration % 5 == 0:
            threshold *= 1.2
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 64
    dx = 2.0 / n_points
    
    # Initialization from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Perform optimization
    iteration = 0
    wavelet_iterations = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 200:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Alternate wavelet and local optimizations
        if iteration % 2 == 0:
            # Due to time constraints, do wavelet optimization only (shorter)
            h_opt = wavelet_optimize(h, dx, n_points, max_iter=3)
            h = h_opt
        else:
            # Simple local search
            h_temp = h.copy()
            step = 0.01
            eps = 1e-4
            grad = np.zeros_like(h_temp)
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_overlap = np.max(corr) * dx
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        # Renewal
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
            last_improvement = iteration
        else:
            stagnation_count += 1
            
        iteration += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
