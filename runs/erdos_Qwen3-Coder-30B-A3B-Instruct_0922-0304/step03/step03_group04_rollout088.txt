import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use convex optimization approach with interval-based discretization
    # Variables: M intervals, h_j = height of interval j
    M = 30  # Number of intervals
    w = 2.0 / M  # width of each interval
    
    # Initialize h in interval form
    if 'initial_h_values' in globals():
        h_interval = np.zeros(M)
        for i in range(M):
            start_idx = int(i * n_points / M)
            end_idx = int((i + 1) * n_points / M)
            h_interval[i] = np.mean(initial_h_values[start_idx:end_idx])
        h_interval = np.clip(h_interval, 0, 1)
        h_interval = h_interval * (M / 2) / np.sum(h_interval)
    else:
        h_interval = np.ones(M) * 0.5
        h_interval[M//2:] = 0.1
        h_interval = np.clip(h_interval, 0, 1)
        h_interval = h_interval * (M / 2) / np.sum(h_interval)
    
    # Precompute overlap information
    # For each shift index k, compute overlap between intervals
    A = np.zeros((2 * M - 1, M))
    shift_indices = []
    
    # Precompute overlaps for all shifts
    for k in range(-M + 1, M):  # k ranges from -(M-1) to M-1
        shift_indices.append(k)
        # For shift k, compute overlap between each interval i and interval i+k
        for i in range(M):
            if i + k < 0 or i + k >= M:
                overlap = 0
            else:
                # Compute length of overlap between interval i and i+k
                # Both intervals have width w
                # Overlap = min(w, w + min(0, w - abs(i+k-i)*w)) = min(w, w - abs(k)*w)
                # But we need to compute actual overlap considering their positions
                # Position of interval i is [i*w, (i+1)*w)
                # Position of interval i+k is [(i+k)*w, (i+k+1)*w)
                # Overlap = max(0, min((i+1)*w, (i+k+1)*w) - max(i*w, (i+k)*w))
                pos1_start = i * w
                pos1_end = (i + 1) * w
                pos2_start = (i + k) * w
                pos2_end = (i + k + 1) * w
                overlap = max(0, min(pos1_end, pos2_end) - max(pos1_start, pos2_start))
                if overlap < 0:
                    overlap = 0
            A[k + M - 1, i] = overlap
    
    # For simplicity and to save time in search, we'll go iterative approach
    # but using sharper optimization method than previous.
    
    # Optimization using CVXPY
    # Define variables
    h_vars = cp.Variable(M, name='h')
    C = cp.Variable(1, name='C')
    
    # Define constraints
    objective = cp.Minimize(C)
    
    # Constraint: sum(h) * w = 1
    constraints = [
        cp.sum(h_vars) == M / 2,
        h_vars >= 0,
        h_vars <= 1,
        C >= 0
    ]
    
    # For all shifts k, we add a constraint on max overlap
    # There are 2*M-1 potential shift patterns
    # But the constraint is: sum_{i} h_i * (1 - h_{i+k}) * overlap_{ik} <= C
    # This accounts for properly indexed shifts.
    
    for k_idx in range(2 * M - 1):
        k = k_idx - M + 1
        # Compute sum over all intervals i of h_i * (1 - h_{i+k}*) * overlap_{ik}
        # where h_{i+k}* is 0 if i+k out of bounds
        # Compute constraint:
        expr = 0
        for i in range(M):
            if i + k < 0 or i + k >= M:
                # No contribution
                pass
            else:
                expr += h_vars[i] * (1 - h_vars[i + k]) * A[k_idx, i]
        constraints.append(expr <= C)
    
    # Create and solve problem
    problem = cp.Problem(objective, constraints)
    solver_name = "ECOS"  # Most suitable for this sparse LP
    try:
        problem.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, reltol=1e-5, abstol=1e-6)
        if problem.status in ['optimal', 'optimal_inaccurate']:
            h_opt = h_vars.value
            c5_opt = C.value[0]
            if c5_opt < best_c5:
                best_c5 = c5_opt
                # Convert back to step function
                best_h = np.repeat(h_opt, n_points // M)
                if len(best_h) > n_points:
                    best_h = best_h[:n_points]
                elif len(best_h) < n_points:
                    # Pad with last element
                    best_h = np.pad(best_h, (0, n_points - len(best_h)), mode='constant', constant_values=best_h[-1])
                best_h = np.clip(best_h, 0, 1)
                # Normalize
                best_h = best_h * (n_points / 2) / np.sum(best_h)
            
            # Return immediately if target reached
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
    except:
        pass

    # Fallback gradient method:
    # Refine using a simple gradient descent with finite differences
    h_temp = h_interval.copy()
    if len(h_temp) != M:
        h_temp = np.resize(h_temp, M)
    h_final = h_temp.copy()
    current_c5 = evaluate_c5(h_final, dx)
    best_c5 = current_c5
    best_h = h_final.copy()
    
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Simple gradient descent with finite differences
        step = 0.02
        iterations = 100
        h_iter = h_final.copy()
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # Evaluate current C5
            C5_current = evaluate_c5(h_iter, dx)
            # Approximate gradient
            grad = np.zeros_like(h_iter)
            eps = 1e-4
            for i in range(M):
                h_plus = h_iter.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                C5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (C5_plus - C5_current) / eps
            
            # Update h
            h_iter = h_iter - step * grad
            h_iter = np.clip(h_iter, 0, 1)
            # Rescale to meet constraint
            h_iter = h_iter * (M / 2) / np.sum(h_iter)
            
            # Check improvement
            new_c5 = evaluate_c5(h_iter, dx)
            if new_c5 < best_c5:
                best_h = h_iter.copy()
                best_c5 = new_c5
                
                if best_c5 <= 0.3807:
                    break
                    
        h_final = h_iter.copy()
        
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
