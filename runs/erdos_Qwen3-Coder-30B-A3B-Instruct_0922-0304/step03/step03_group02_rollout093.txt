import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fftpack import fft, ifft
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

        h[n_points//2:] = 0

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    max_iter = 1000

        if time.time() - start_time > budget_s - 1:
            break
            
        # Try multiple optimizers to avoid local minima
        optimizers = ['SLSQP', 'L-BFGS-B', 'TNC']
        for opt_method in optimizers:
            try:
                result = minimize(
                    objective,
                    h,
                    method=opt_method,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 30, 'ftol': 1e-9}
                )
                if result.success:
                    h_new = result.x
                    # Ensure feasibility
                    h_new = np.clip(h_new, 0, 1)

            h += np.random.normal(0, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
