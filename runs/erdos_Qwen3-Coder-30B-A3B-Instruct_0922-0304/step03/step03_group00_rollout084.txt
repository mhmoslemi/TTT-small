import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize swarm
    n_particles = 15
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.zeros(n_points))
    
    # Generate other particles with triangular patterns
    for _ in range(n_particles - 1):
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - mid) / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Initialize personal bests
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    max_iter = 3000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            v = w * v + c_p * r_p * (pbest[i] - h) + c_g * r_g * (gbest - h)
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Store new position
            particles[i] = h_new
            
            # Evaluate fitness
            score = evaluate_c5(h_new, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = score
            
            # Update global best
            if score < gbest_score:
                gbest = h_new.copy()
                gbest_score = score
                
                # Early exit if target achieved
                if gbest_score <= 0.3807:
                    break
    
    return (gbest.tolist(), gbest_score, n_points)
