import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start approach
    initial_h_list = [best_h.copy()]
    
    # Add random perturbations
    for i in range(5):
        h_pert = h + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_h_list.append(h_pert)
    
    # Optimization loop
    for i, h_start in enumerate(initial_h_list):
        h = h_start.copy()
        last_improvement = 0
        for iteration in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate current solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                if last_improvement > 10:
                    break
                    
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
            # Optimize using SLSQP
            result = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
            else:
                # If optimization fails, use a small perturbation
                h += np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final refinement
    h = best_h.copy()
    for i in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-8}
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            break
    
    final_c5 = compute_c5(h)
    
    return (h.tolist(), final_c5, n_points)
