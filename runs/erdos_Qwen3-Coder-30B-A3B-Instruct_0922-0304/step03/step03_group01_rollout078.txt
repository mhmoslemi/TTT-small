import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 200
    max_iter_slsqp = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    # Initialize swarm
    for _ in range(swarm_size):
        particle = h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        particle += noise
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        val = evaluate_c5(particle, dx)
        pbest.append(particle.copy())
        pbest_vals.append(val)
        if val < gbest_val:
            gbest = particle.copy()
            gbest_val = val
    
    # PSO iterations
    for pso_iter in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (pbest[i] - particles[i]) + 
                            social * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            val = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = val
                
                # Update global best
                if val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = val
                    
    # SLSQP refinement
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()
    
    # SLSQP optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Optimize with SLSQP
    max_iter = max_iter_slsqp
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 10, 'ftol': 1e-9})
        
        if result.success:
            best_h = result.x
            current_c5 = objective(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                
            if current_c5 <= 0.3807:
                break
        else:
            # Local gradient descent fallback
            step = 0.001
            iterations = 10
            h_temp = best_h.copy()
            
            for _ in range(iterations):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            best_h = h_temp
            
        # Feasibility repair
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
