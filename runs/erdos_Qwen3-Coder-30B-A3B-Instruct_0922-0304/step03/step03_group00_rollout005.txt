import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # PSO Parameters
    n_points = 70
    n_particles = 15
    dx = 2.0 / n_points
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Fallback initialization
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Add initial solution to swarm
    swarm.append(h0.copy())
    velocities.append(np.zeros(n_points))
    
    # Add other particles with triangular functions
    for _ in range(n_particles - 1):
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1 - 2 * abs(i - mid) / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Add noise
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Track best
    pbest = [h.copy() for h in swarm]
    pbest_scores = [evaluate_c5(h, dx) for h in swarm]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    best_c5 = gbest_score
    best_h = gbest.copy()
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c1 * r1 * (pbest[i] - h) + 
                 c2 * r2 * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            score = evaluate_c5(h_new, dx)
            swarm[i] = h_new.copy()
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = h_new.copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = h_new.copy()
                    if gbest_score < best_c5:
                        best_c5 = gbest_score
                        best_h = gbest.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
