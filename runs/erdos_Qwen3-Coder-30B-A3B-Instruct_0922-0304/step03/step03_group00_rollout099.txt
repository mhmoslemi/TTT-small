import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_overlap_gradient(h, k, n_points):
    """Compute gradient of overlap w.r.t. h for shift k"""
    grad = np.zeros(n_points)
    # For shift k, h[i] contributes to overlap with h[i+k]
    # gradient w.r.t. h[i]: 1 - h[i+k] (when i+k < n_points)
    # gradient w.r.t. h[i+k]: -h[i] (when i < n_points)
    
    # Compute overlap for shift k using correlation
    # Don't compute full correlation, just get the contribution
    # to gradient for this specific shift
    if k >= 0:
        # h[0:n-k] contributes to overlap with h[k:n]
        # grad[i] += 1 - h[i+k] for i < n-k
        # grad[i+k] += -h[i] for i < n-k
        for i in range(n_points - k):
            grad[i] += 1 - h[i + k]
        for i in range(n_points - k):
            if i + k < n_points:
                grad[i + k] -= h[i]
    else:
        # k < 0, h[-k:n] contributes to overlap with h[0:n+k]
        # grad[i] += 1 - h[i-k] for i >= -k and i < n
        # grad[i-k] += -h[i] for i >= -k and i < n
        k_abs = -k
        for i in range(k_abs, n_points):
            grad[i] += 1 - h[i - k_abs]
        for i in range(k_abs, n_points):
            if i - k_abs < n_points:
                grad[i - k_abs] -= h[i]
    return grad

def compute_max_overlap_and_shift(h, n_points, dx):
    """Compute maximum overlap and corresponding shift"""
    # Compute all overlaps
    corr = correlate(h, 1 - h, mode="full")
    # The full correlation has length 2*n_points - 1
    # The valid shifts are from -(n_points-1) to (n_points-1)
    # But we only look at the part where overlap is at least n_points long
    # The correlation is symmetric, and maximum will be at the center
    # The shift k corresponds to the index k + (n_points-1) in the correlation array
    max_idx = np.argmax(corr)
    max_overlap = corr[max_idx] * dx
    shift = max_idx - (n_points - 1)
    return max_overlap, shift

        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop using projected gradient descent
    max_iter = 50000
    learning_rate = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute maximum overlap and its corresponding shift
        current_c5, shift = compute_max_overlap_and_shift(h, n_points, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_overlap_gradient(h, shift, n_points)
        
        # Update h in gradient direction
        h_new = h - learning_rate * grad
        
        # Project h_new back to feasible region
        h_new = project_h(h_new, n_points)
        
        # Simple line search: try a few different learning rates
        best_h_trial = h_new
        best_trial_c5 = float('inf')
