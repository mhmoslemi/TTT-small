import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: triangular shape with peak at center
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Multi-start optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    candidates = []

    # Add parent solution
    candidates.append(h.copy())
    
    # Add random perturbations
    for i in range(5):
        h_pert = h + np.random.normal(0, 0.1, n_points)
        h_pert = project_feasible(h_pert)
        candidates.append(h_pert)

    # Evaluate all candidates
    for h_candidate in candidates:
        h_opt = h_candidate.copy()
        # Local optimization
        max_iter = 100
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break            

            def objective(h):
                return compute_c5(h)

            # Define bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]

            result = minimize(
                objective,
                h_opt,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 20, 'ftol': 1e-8}
            )

            if result.success:
                h_opt = result.x
            else:
                h_opt += np.random.normal(0, 0.005, n_points)
            
            h_opt = project_feasible(h_opt)

    # Final evaluation
    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), final_c5, n_points)
