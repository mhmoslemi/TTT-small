import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start with Basin Hopping
    num_starts = 5
    perturbation_scale = 0.05
    
    # Create initial set of candidates
    candidates = [h.copy()]
    for _ in range(num_starts - 1):
        perturbed = h + np.random.normal(0, perturbation_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        candidates.append(perturbed)
    
    # Local optimization of initial candidates
    for i, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(
            objective, 
            candidate, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            candidate = result.x
        # Ensure feasibility
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        candidates[i] = candidate
        
        current_c5 = compute_c5(candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = candidate.copy()
    
    # Basin hopping
    current_h = best_h.copy()
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb current solution
        perturbation = np.random.normal(0, perturbation_scale, n_points)
        new_h = current_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Local optimization
        result = minimize(
            objective, 
            new_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            new_h = result.x
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept or reject based on improvement
        current_c5 = compute_c5(current_h)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h.copy()
            
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptively reduce perturbation scale
        if iteration % 50 == 0 and perturbation_scale > 1e-5:
            perturbation_scale *= 0.9
            
    # Final refinement with tighter tolerance
    result = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if result.success:
        best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
