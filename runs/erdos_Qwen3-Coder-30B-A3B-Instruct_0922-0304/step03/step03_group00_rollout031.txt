import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    swarm.append(h0)
    velocities.append(np.zeros(n_points))
    
    # Add triangular initializations
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1.0, 2.0 * abs(j - n_points/2) / n_points)
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests
    pbest = [h.copy() for h in swarm]
    pbest_scores = [evaluate_c5(h, dx) for h in swarm]
    
    # Track global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c_p * r1 * (pbest[i] - h) + c_g * r2 * (gbest - h)
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to satisfy sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new position
            score = evaluate_c5(h_new, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = h_new.copy()
                    gbest_score = score
                    
            swarm[i] = h_new
    
    # Return best solution found
    return (gbest.tolist(), gbest_score, n_points)
