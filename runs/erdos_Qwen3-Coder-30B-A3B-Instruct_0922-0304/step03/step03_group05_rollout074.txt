import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            max_move *= 0.95
            no_improvement_count = 0
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        
        # Determine shift k and indices that contribute most
        # Full convolution: indices [0, 2*n_points-2] 
        # The actual shift k is in range [-n_points+1, n_points-1]
        k = max_idx - (n_points - 1)
        
        # Find top contributing indices
        top_contributors = []
        for i in range(n_points):
            j = (i + k) % n_points
            contribution = h[i] * (1 - h[j])
            top_contributors.append((contribution, i, j))
        top_contributors.sort(reverse=True)
        top_contributors = top_contributors[:max(1, n_points // 20)]
        
        # Adjust h values
        for contrib, i, j in top_contributors:
            # Compute max adjustment
            max_decrease = min(h[i], max_move)
            max_increase = min(1 - h[j], max_move)
            
            # Choose epsilon to reduce overlap
            epsilon = min(max_decrease, max_increase)
            if epsilon <= 0:
                continue
                
            # Update h
            h[i] -= epsilon
            h[j] += epsilon
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
