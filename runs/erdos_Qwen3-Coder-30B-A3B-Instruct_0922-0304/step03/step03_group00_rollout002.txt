import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

    else:

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst shift k
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        # Convert index to shift k
        k = max_idx - (n_points - 1)
        
        # Create QP for this shift k
        # Objective: minimize sum_i h[i] * (1 - h[i+k]) 
        # which is sum_i h[i] - h[i] * h[i+k]
        
        # Construct quadratic program
        h_var = cp.Variable(n_points)
        
        # Objective: minimize - sum_i h[i] * h[i+k] + sum_i h[i]
        # We want to minimize sum_{i=0}^{n-k-1} h[i](1 - h[i+k])
        # = sum_i h[i] - sum_i h[i]*h[i+k]
        
        # Use a compact way to build the quadratic term
        
        # Manually create sparse quadratic matrix
        Q = np.zeros((n_points, n_points))
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                Q[i, i + k] = -1.0
                Q[i + k, i] = -1.0
                
        # Linear term: 1.0 for all h[i]
        c = np.ones(n_points)
        
        objective = cp.Minimize(cp.quad_form(h_var, -Q/2) + c.T @ h_var)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Form and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h = project_to_feasible(h_new)
            else:
                # Fallback to a simple gradient descent approach
                step = 0.001
                iterations = 200
                h_temp = h.copy()
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    k = max_idx - (n_points - 1)
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - np.max(corr) * dx) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                h = h_temp
        except:

            step = 0.001
            iterations = 200
            h_temp = h.copy()

                max_idx = np.argmax(corr)
                k = max_idx - (n_points - 1)
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - np.max(corr) * dx) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            h = h_temp
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
