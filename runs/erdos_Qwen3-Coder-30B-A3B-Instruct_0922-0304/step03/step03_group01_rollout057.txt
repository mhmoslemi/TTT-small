import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two balanced rectangular pulses
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with projected descent
    max_iter = 500
    eta = 0.01
    patience = 0
    patience_limit = 10
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current objective and gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_overlap_plus = np.max(overlap_plus) * dx
            grad[i] = (max_overlap_plus - max_overlap) / eps
        
        # Descent update with Lagrange multiplier
        h -= eta * grad
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            last_improvement = iteration
        else:
            patience += 1
            
        # Adaptive step size
        if iteration > 0 and iteration % 200 == 0:
            eta *= 0.5
            
        # Early exit if target achieved or no improvement
        if best_c5 <= 0.3807 or patience > patience_limit:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
