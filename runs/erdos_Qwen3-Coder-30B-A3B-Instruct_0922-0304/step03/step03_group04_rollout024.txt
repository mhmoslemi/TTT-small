import numpy as np
from scipy.signal import correlate
from scipy import signal
import pywt
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for a given step function h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64  # Use 64 points to ensure at least 6 levels of decomposition
    dx = 2.0 / n_points
    max_wavelet_iter = 3
    max_spatial_refine = 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Wavelet decomposition
    coeffs = pywt.wavedec(h, 'haar', level=6)
    theta = np.concatenate(coeffs)  # Flatten coefficients
    n_theta = len(theta)
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = h.copy()
    best_theta = theta.copy()
    
    # Sparse optimization parameters
    lambda_reg = 0.1
    alpha = 0.1
    patience = 0
    best_patience = 10
    
    # Wavelet domain optimization
    for wave_iter in range(max_wavelet_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient for wavelet coefficients
        # This computes gradient via finite differences in the wavelet domain
        # Since full-wavelet gradient is expensive, we'll use sampled gradient computation
        
        # Apply iterative thresholding
        # Take gradient approximation with respect to theta (wavelet coefficients)
        # Needed: ∇[max_k ∫ h(x)(1 - h(x+k)) dx]
        
        # For efficiency, we compute gradient numerically for a subset of coefficients
        grad = np.zeros_like(theta)
        eps = 1e-4
        
        # Sample about few coefficients for gradient estimation
        sample_indices = np.random.choice(range(n_theta), min(20, n_theta//2), replace=False)
        
        for idx in sample_indices:
            theta_plus = theta.copy()
            theta_plus[idx] += eps
            h_plus = pywt.waverec([theta_plus[:len(coeffs[0])], 
                                 theta_plus[len(coeffs[0]):len(coeffs[0])+len(coeffs[1])],
                                 theta_plus[len(coeffs[0])+len(coeffs[1]):len(coeffs[0])+len(coeffs[1])+len(coeffs[2])],
                                 theta_plus[len(coeffs[0])+len(coeffs[1])+len(coeffs[2]):]], 'haar')
            if len(h_plus) > n_points:
                h_plus = h_plus[:n_points]
            elif len(h_plus) < n_points:
                h_plus = np.pad(h_plus, (0, n_points - len(h_plus)), 'constant')
                
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            # Compute correlation
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            
            corr = correlate(h, 1 - h, mode="full")
            c5_current = np.max(corr) * dx
            
            grad[idx] = (c5_plus - c5_current) / eps
        
        # Update theta with gradient
        theta_new = theta - alpha * grad
        
        # Apply soft thresholding
        theta_new = np.sign(theta_new) * np.maximum(np.abs(theta_new) - lambda_reg, 0)
        
        # Update h representation
        try:
            rec_coeffs = [theta_new[:len(coeffs[0])], 
                         theta_new[len(coeffs[0]):len(coeffs[0])+len(coeffs[1])],
                         theta_new[len(coeffs[0])+len(coeffs[1]):len(coeffs[0])+len(coeffs[1])+len(coeffs[2])],
                         theta_new[len(coeffs[0])+len(coeffs[1])+len(coeffs[2]):]]
            h_new = pywt.waverec(rec_coeffs, 'haar')
            if len(h_new) > n_points:
                h_new = h_new[:n_points]
            elif len(h_new) < n_points:
                h_new = np.pad(h_new, (0, n_points - len(h_new)), 'constant')
        except:
            h_new = h.copy()  # fallback
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        current_c5 = evaluate_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            best_theta = theta_new.copy()
            patience = 0
        else:
            patience += 1
            
        if patience >= best_patience:
            lambda_reg *= 1.5  # Increase sparsity
            patience = 0
            
        # Update theta for next iteration
        theta = theta_new.copy()
        
        # Spatial refinement every few wavelet iterations
        if wave_iter % 2 == 0 and wave_iter > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
            # Run spatial refinement steps
            try:
                # Get sample points for refining specific peaks
                top_indices = np.argsort(h_new)[::-1][:max(10, n_points//8)]
                # Apply gradient descent on selected locations
                h_refined = h_new.copy()
                learning_rate = 0.01
                for _ in range(20):
                    if time.time() - start_time > budget_s - 5:
                        break
                        
                    # Compute current correlations
                    corr = correlate(h_refined, 1 - h_refined, mode="full")
                    max_overlap_idx = np.argmax(corr)
                    max_overlap = corr[max_overlap_idx] * dx
                    
                    grad_spatial = np.zeros_like(h_refined)
                    for i in top_indices:
                        h_plus = h_refined.copy()
                        eps_s = 1e-4
                        h_plus[i] += eps_s
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad_spatial[i] = (c5_plus - max_overlap) / eps_s
                        
                    h_refined -= learning_rate * grad_spatial
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    
                # Evaluate refined solution
                c5_refined = evaluate_c5(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
            except:
                pass
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
