import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # SDP approach
    def sdp_optimize(h_init, dx):
        n = len(h_init)
        h = cp.Variable(n)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h) == n / 2,
            h >= 0,
            h <= 1,
            C5 >= 0
        ]
        
        # Overlap constraints
        for k in range(-n+1, n):
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in range(n-k)
                expr = 0
                for i in range(n-k):
                    expr += h[i] * (1 - h[i+k])
                constraints.append(C5 >= expr * dx)
            else:
                # h[i] * (1 - h[i+k]) for i in range(-k, n)
                expr = 0
                for i in range(-k, n):
                    expr += h[i] * (1 - h[i+k])
                constraints.append(C5 >= expr * dx)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value, prob.value
        except:
            pass
        return h_init, evaluate_c5(h_init, dx)
    
    # Optimizer with SDP refinement
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try SDP refinement
        h_sdp, c5_sdp = sdp_optimize(h, dx)
        if c5_sdp < best_c5:
            best_c5 = c5_sdp
            best_h = h_sdp.copy()
            
        # Gradient descent steps
        step = 0.001
        iterations = 50
        h_temp = h.copy()
        
        for _ in range(iterations):
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update h
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
        h = h_temp
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
