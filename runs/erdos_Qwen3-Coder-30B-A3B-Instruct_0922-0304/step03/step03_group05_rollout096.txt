import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_overlap_at_shift(h, shift):
        """Compute overlap for a specific shift."""
        n = len(h)
        if shift >= n:
            return 0.0
        result = 0.0
        for i in range(n):
            j = (i + shift) % n
            result += h[i] * (1 - h[j])
        return result * dx

    def get_top_contributors(h, max_shift):
        """Identify where most overlap occurs."""
        n = len(h)
        max_overlap = -1.0
        best_k = 0
        best_indices = []
        
        for k in range(n):
            overlap = compute_overlap_at_shift(h, k)
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
                # Find top contributing indices
                contributions = []
                for i in range(n):
                    j = (i + k) % n
                    contrib = h[i] * (1 - h[j])
                    contributions.append((contrib, i))
                contributions.sort(reverse=True)
                best_indices = [idx for _, idx in contributions[:max(n//20, 1)]]
                
        return best_k, best_indices

    def adjust_h(h, k, indices, max_move=0.01):
        """Adjust h to reduce overlap at shift k."""
        new_h = h.copy()
        sum_diff = 0.0
        for i in indices:
            j = (i + k) % len(h)
            # Compute maximum feasible adjustment
            decrease = min(new_h[i], max_move)
            increase = min(1 - new_h[j], max_move)
            adjust = min(decrease, increase)
            if adjust > 0:
                new_h[i] -= adjust
                new_h[j] += adjust
                sum_diff += adjust
                
        # Ensure conservation of mass
        if abs(sum_diff) > 1e-10:
            # Redistribute difference
            delta = sum_diff
            # Adjust some elements to compensate
            total_mass = np.sum(new_h)
            if abs(total_mass - n_points/2) > 1e-10:
                # Find elements to adjust
                adjust_count = min(5, len(new_h))
                # Adjust evenly
                for idx in range(adjust_count):
                    if abs(np.sum(new_h) - n_points/2) < 1e-10:
                        break
                    new_h[idx] = np.clip(new_h[idx] - delta/adjust_count, 0, 1)
                new_h = new_h * (n_points/2) / np.sum(new_h)
        else:
            # Just normalize to ensure sum is correct
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
        return new_h

    best_c5 = float('inf')
    best_h = h.copy()
    max_move = 0.01
    no_improvement_count = 0
    
    # Main optimization loop
    max_iter = 3000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current solution
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                max_move *= 0.95  # Reduce move size if stuck
                no_improvement_count = 0
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Get contributing indices for maximum overlap
        k, indices = get_top_contributors(best_h, 30)
        
        # Make adjustment
        new_h = adjust_h(best_h, k, indices, max_move)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            if abs(new_c5 - best_c5) < 1e-8:
                # Small improvement: reduce learning rate
                max_move *= 0.98
        else:
            # Accept with small probability if not too far
            # (This is to escape local minima)
            if np.random.rand() < 0.05:
                best_h = new_h

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), min(final_c5, best_c5), n_points)
