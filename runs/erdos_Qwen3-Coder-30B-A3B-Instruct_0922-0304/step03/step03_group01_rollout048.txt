import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    orig_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n / 2) / np.sum(h_plus)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - orig_c5) / eps
    return grad

def line_search(h, grad, dx, max_iter=20, alpha=0.01, beta=0.5):
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2) / np.sum(h_new)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < evaluate_c5(h, dx):
            return h_new
        alpha *= beta
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s - 1:
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Line search
        h = line_search(h, grad, dx)
        
        # Projection
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturbation
        if no_improve_count > 50:
            pert_mag = 1e-3 * (1.0 / np.sqrt(iter_count + 1))
            noise = np.random.normal(0, pert_mag, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improve_count = 0
            
        iter_count += 1
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
