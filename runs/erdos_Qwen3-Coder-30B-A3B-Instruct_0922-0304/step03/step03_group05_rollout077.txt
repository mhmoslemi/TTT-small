from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution if needed
        if len(h) != n_points:
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start optimization with several initialization strategies
    initial_hs = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_old = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_old, h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_hs.append(h_parent.copy())
    else:
        initial_hs.append(h.copy())

    # Random perturbations
    for _ in range(4):
        h_rand = h.copy() + np.random.normal(0, 0.1, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_hs.append(h_rand.copy())

    # Fallback triangular shape
    h_tri = np.zeros(n_points)
    mid = n_points // 2
    width = n_points // 4
    for i in range(width):
        h_tri[mid - width//2 + i] = i / (width//2)
        h_tri[mid + width//2 - i] = i / (width//2)
    h_tri = np.clip(h_tri, 0, 1)
    h_tri = h_tri * (n_points / 2) / np.sum(h_tri)
    initial_hs.append(h_tri.copy())

    # Optimization loop
    max_iter = 1000
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different initial points
        for h_start in initial_hs:
            h_current = h_start.copy()
            # Run optimization
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
            else:
                h_opt = h_current
            
            # Ensure feasibility
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Evaluate current solution
            current_c5 = compute_c5(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > 100:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
