import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        worst_k = max_idx - (n_points - 1)
        
        # Solve QP for this shift
        h_new = h.copy()
        try:
            # Set up QP
            h_var = cp.Variable(n_points)
            
            # Objective: minimize overlap at the worst shift
            objective = 0
            for i in range(n_points):
                j = i + worst_k
                if 0 <= j < n_points:
                    objective += h_var[i] * (1 - h_var[j])
                    
            objective *= dx
            
            # Constraints
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Solve
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        except:
            pass  # Failed, fall back to previous
        
        h = h_new
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
