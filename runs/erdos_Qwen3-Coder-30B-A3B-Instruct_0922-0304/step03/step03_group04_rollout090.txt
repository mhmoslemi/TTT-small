import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    n_points = 70

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')

    freq_domain = fft(h)

        # Apply small random phase change
        mod_freq = freq_domain * perturbation
        # Inverse FFT to get back to spatial domain
        new_h = np.real(ifft(mod_freq))
        # Project back to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        return new_h
    
    # Initial candidates from spectral mutation
    candidates = []
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        mutated_h = mutate_spectrum(h, freq_domain)
        candidates.append(mutated_h)
    
    # Evaluate initial candidates
    candidate_scores = []
    for candidate in candidates:
        if time.time() - start_time > budget_s - 1:
            break
        score = evaluate_c5(candidate, dx)

    candidate_scores.sort()

    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 1:

        # Use Nelder-Mead on this candidate
        def obj_func(h_vals):
            h_new = h_vals.copy()
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return evaluate_c5(h_new, dx)

        simplex_start = []

            pert = pert * (n_points / 2) / np.sum(pert)
            simplex_start.append(pert)

        # Simplex optimization
        # Use a basic gradient-free approach due to time constraints
        current_h = candidate.copy()
        for iter in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            current_score = obj_func(current_h)
            
            best_score = current_score
            best_candidate = current_h.copy()

                current_h = best_candidate.copy()

    # Final check

    # Early exit if target achieved

    return (best_h.tolist(), best_c5, n_points)
