import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Helper to get all overlaps efficiently
    def compute_all_overlaps(h):
        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        shifts = np.arange(-len(h)+1, len(h))
        return corr, shifts
    
    # Parameters for iterative adjustment
    max_iter = 1000
    max_move = 0.01
    patience = 100
    no_improvement_count = 0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= target_c5:
            break
            
        # If no improvement for a while, reduce learning rate
        if no_improvement_count > patience:
            max_move *= 0.95
            no_improvement_count = 0
            
        # Evaluate overlaps to find the worst case
        corr, shifts = compute_all_overlaps(h)
        max_index = np.argmax(corr)
        max_shift = shifts[max_index]
        
        # Find the indices in the problematic correlation
        # We'll analyze the worst contributing indices
        # For shift k, we look at integral over indices i
        # h[i](1 - h[i+k]) 
        # This involves fixing the one affecting the most
        
        # Simplification: we loop through all pairs contributing significantly
        # and decrease the h[i] that contributes greatly to overlap
        # This is heuristic but has shown to work well for such problems
        
        # Use a more targeted adjustment approach
        # Find top contributors to the current maximum overlap
        real_corr = np.zeros(len(h)*2 - 1)
        for i in range(len(h)):
            for j in range(len(h)):
                real_corr[i-j+len(h)-1] += h[i] * (1 - h[j])
                
        # Adjust the best candidate to reduce max overlap
        # For simplicity, we consider the shift with highest overlap
        # and adjust individuals which contribute the most
        this_corr = np.zeros(len(h))
        for i in range(len(h)):
            j = (i + max_shift) % len(h)
            this_corr[i] = h[i] * (1 - h[j])
            
        # Find top 5% contributing pairs
        top_indices = np.argsort(this_corr)[-int(len(h)*0.05):][::-1]
        
        # Do the adjustment in a way that respects constraints
        for i in top_indices:
            j = (i + max_shift) % len(h)
            # Compute how much we can adjust
            # Can decrease h[i] by at most min(h[i], max_move)
            # Can increase h[j] by at most min(1-h[j], max_move)
            move = min(h[i], 1 - h[j], max_move)
            if move > 1e-8:
                h[i] -= move
                h[j] += move
     
        # Ensure the final sum is maintained
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Final evaluation 
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
