```python
import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy
import math

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h values"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to required sum
    h = h * (n_points / 2) / np.sum(h)
    return h

def objective(h, n_points, dx):
    """Objective function for CMA-ES"""
    repaired_h = repair_h(h, n_points)
    return evaluate_c5(repaired_h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create an initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Set up CMA-ES
    es = CMAEvolutionStrategy(h, 0.1, {
        'seed': seed,
        'popsize': 100,
        'verbose': -9
    })
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = int(budget_s * 10)  # Roughly 10 iterations per second
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Ask for new candidate solutions
        solutions = es.ask()
        fitness_values = []
        
        # Evaluate all solutions
        for sol in solutions:
            current_c5 = objective(sol, n_points, dx)
            fitness_values.append(current_c5)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = sol.copy()
                
            # Early termination
            if best_c5 <= 0.3807:
                break
                
        # Tell the optimizer about the fitness values
        es.tell(solutions, fitness_values)
        
        # Check if we've achieved the target
        if best_c5 <= 0.3807:
            break
    
    # Final repair and evaluation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
