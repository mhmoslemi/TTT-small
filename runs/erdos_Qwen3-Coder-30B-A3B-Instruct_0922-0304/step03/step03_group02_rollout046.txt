import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    noise_sigma = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Generate random initial candidates
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    # Genetic algorithm loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Track best solution
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Selection with tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep best 10%
        elite_indices = np.argsort(fitness)[:population_size // 10]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1_idx = np.random.randint(0, population_size)
            parent2_idx = np.random.randint(0, population_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([selected[parent1_idx][:crossover_point], 
                                       selected[parent2_idx][crossover_point:]])
                child2 = np.concatenate([selected[parent2_idx][:crossover_point], 
                                       selected[parent1_idx][crossover_point:]])
            else:
                child1 = selected[parent1_idx].copy()
                child2 = selected[parent2_idx].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, noise_sigma)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, noise_sigma)
            
            # Repair
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
