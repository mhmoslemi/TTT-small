import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Trust region optimization with adaptive step sizes
    max_iter = 5000
    patience = 0
    patience_limit = 50
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with trust region approach
        try:
            result = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=cons, 
                             options={'maxiter': 50, 'xtol': 1e-9, 'gtol': 1e-9})
            
            if result.success:
                h = result.x
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except:
            patience += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Occasionally perturb solution to escape local minima
        if patience > patience_limit:
            h = best_h.copy()
            noise = np.random.normal(0, 0.005, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            
        # Apply gradient-based local search in case of stagnation
        if patience > patience_limit // 2:
            try:
                # Compute finite difference gradient
                eps = 1e-4
                grad = np.zeros_like(h)
                base_val = objective(h)
                
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    grad[i] = (objective(h_plus) - base_val) / eps
                    
                h -= step_size * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                patience = 0
            except:
                pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
