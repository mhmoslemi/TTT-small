import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    selected = []
    pop_size = len(population)
    for _ in range(pop_size // 2):
        idxs = np.random.choice(pop_size, tournament_size)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_std=0.05):
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, noise_std)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = repair_h(initial_h, n_points)
        population.append(initial_h)
    else:
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h = repair_h(initial_h, n_points)
        population.append(initial_h)
    
    # Fill population with random individuals
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    max_generations = 1000
    tournament_size = 3
    crossover_rate = 0.7
    mutation_rate = 0.2
    noise_std = 0.05
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = tournament_selection(population, fitnesses, tournament_size)
        
        # Create new population
        new_population = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            child1 = mutate(child1, mutation_rate, noise_std)
            child2 = mutate(child2, mutation_rate, noise_std)
            new_population.append(repair_h(child1, n_points))
            new_population.append(repair_h(child2, n_points))
        
        population = new_population
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        final_c5 = float('inf')
        
    return (best_h.tolist() if best_h is not None else [0]*n_points, final_c5, n_points)
