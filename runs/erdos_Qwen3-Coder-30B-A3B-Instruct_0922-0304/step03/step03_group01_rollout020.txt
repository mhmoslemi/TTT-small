import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[20:35] = 0.5
        h[35:50] = 0.5

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation helper
    eps = 1e-5
    
    # Optimization loop
    iteration = 0
    no_improve_count = 0
    max_no_improve = 100
    
    while time.time() - start_time < budget_s - 1 and no_improve_count < max_no_improve:
        iteration += 1
        
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Line search for step size
        step_size = 0.01
        max_line_search = 20
        for _ in range(max_line_search):
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < current_c5:
                h = h_new
                break
            step_size *= 0.5
            
        # Perturbation to escape local minima
        if iteration % 50 == 0 and no_improve_count > 20:
            pert_magnitude = 1e-3 / np.sqrt(iteration)
            h += np.random.normal(0, pert_magnitude, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
