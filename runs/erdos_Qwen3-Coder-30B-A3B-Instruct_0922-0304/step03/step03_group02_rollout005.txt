import numpy as np
from scipy.signal import correlate
import time
import cma

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def objective_function(h, n_points, dx):
    """Objective function for optimization"""
    h = repair_h(h, n_points)
    return evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES optimization
    es = cma.CMAEvolutionStrategy(h, 0.1, {'popsize': 100, 'seed': seed})
    
    max_iter = int(budget_s * 10)  # Adjusted for expected runtime
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Ask for new solutions
        X = es.ask()
        fitness = []
        
        for x in X:
            repaired_x = repair_h(x, n_points)
            fit = evaluate_c5(repaired_x, dx)
            fitness.append(fit)
            
        # Tell the optimizer
        es.tell(X, fitness)
        
        # Check best solution
        current_h = es.best.x
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_c5(current_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
