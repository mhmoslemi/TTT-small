import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: Create a piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try to improve the solution using SDP relaxation
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try to solve a relaxed version using SDP
        try:
            # Create cvxpy variables
            h_vars = cp.Variable(n_points)
            C5_var = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Define the overlap constraints using the full correlation
            # We will add constraints that C5 >= max_k overlap_k
            # For the full overlap integral, we compute for all shifts
            h_full = np.concatenate([h_vars, np.zeros(n_points-1)])
            one_minus_h_full = np.concatenate([1 - h_vars, np.zeros(n_points-1)])
            
            # Compute all possible overlaps (this is expensive but we do it approximately)
            # We just take a few key ones for now to build the constraint quickly
            
            # Record current h values
            h_current = h_vars.value if h_vars.value is not None else h
            
            # Compute overlaps for some indices
            # We sample to reduce complexity
            overlap_indices = list(range(-n_points+1, n_points))
            overlap_indices = overlap_indices[::max(1, len(overlap_indices)//10)]
            
            for k in overlap_indices:
                # For cross terms to be valid, we need to take care of indexing
                if k >= 0:
                    # h[i] * (1 - h[i+k])
                    # Correlation cue: when k=0, we have h[i] * (1-h[i]) = h[i] - h[i]^2
                    # When k>0, h[i] * (1 - h[i+k])
                    # For simplicity: we approximate by sampling points
                    if k < n_points:
                        h_k = cp.hstack([h_vars[:n_points-k], np.zeros(k)])
                        prod_term = cp.multiply(h_vars, 1-h_k)
                        constraints.append(C5_var >= cp.sum(prod_term) * dx)
                else:
                    # k < 0, so we have h[i+|k|] * (1 - h[i])
                    k_abs = -k
                    h_k = cp.hstack([np.zeros(k_abs), h_vars[:n_points-k_abs]])
                    prod_term = cp.multiply(h_vars, 1-h_k)
                    constraints.append(C5_var >= cp.sum(prod_term) * dx)
            
            # Objective
            objective = cp.Minimize(C5_var)
            
            # Create and solve the problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = np.array(h_vars.value)
                new_h = np.clip(new_h, 0, 1)
                if np.abs(np.sum(new_h) - n_points / 2) < 1e-5:
                    # Evaluate this solution
                    current_c5 = max(np.correlate(new_h, 1-new_h, mode="full")) * dx
                    if current_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = current_c5
                        
                    h = new_h.copy()
                    
        except Exception as e:
            pass
        
        # Local improvement using gradient ascent
        try:
            step = 0.005
            iterations = 100
            h_temp = h.copy()
            
            for _ in range(iterations):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_overlap = np.max(corr) * dx
                
                # Compute gradient using finite differences for each variable
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Use gradient ascent to reduce overlap (because we want to minimize C5)
                h_temp += step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp

            # Evaluate current solution
            current_c5 = max(np.correlate(h, 1 - h, mode="full")) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
        except Exception as e:
            pass
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final check with actual evaluation
    final_c5 = max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
