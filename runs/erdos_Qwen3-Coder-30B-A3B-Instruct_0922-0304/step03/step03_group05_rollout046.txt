import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    max_iter = 5000
    max_move = 0.02
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= target_c5:
            break
            
        if no_improvement_count > max_no_improvement:
            # Reduce learning rate
            max_move *= 0.95
            no_improvement_count = 0

        # Find the maximum overlap and corresponding shift
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        k_shift = k_max - (n_points - 1)
        
        # Get indices that contribute most to this overlap
        # For shift k_shift, we look at positions i where h[i] * (1 - h[i+k_shift]) is large
        overlap_contributions = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_shift) % n_points
            overlap_contributions[i] = h[i] * (1 - h[j])
            
        # Select indices with highest contributions
        top_indices = np.argsort(overlap_contributions)[-max(1, n_points//20):]  # Top 5% or at least 1
        
        # Adjust those indices
        for i in top_indices:
            j = (i + k_shift) % n_points
            
            # Compute maximum adjustment
            max_decrease = min(h[i], max_move)
            max_increase = min(1 - h[j], max_move)
            
            if max_decrease > 1e-8 and max_increase > 1e-8:
                epsilon = min(max_decrease, max_increase)
                h[i] -= epsilon
                h[j] += epsilon

        # Ensure constraints
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h *= scale_factor
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
