import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    max_iter = 1000
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # If no improvement for too long, reduce step size
        if no_improvement_count > max_no_improvement:
            max_move *= 0.95
            no_improvement_count = 0

        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_indices = np.arange(len(overlap))
        # Compute actual shifts (include negative shifts)
        shifts = k_indices - (len(h) - 1)  # shift values from -(n-1) to (n-1)
        
        # Get maximum overlap and corresponding shift
        max_overlap_idx = np.argmax(overlap)
        max_shift = shifts[max_overlap_idx]
        
        # Find indices where the contribution to this maximum overlap is highest
        # For that shift, the term is h[i] * (1 - h[i + shift])
        # Differentiate with respect to h[i] to find the direction to decrease
        contribs = np.zeros(len(h))
        for i in range(len(h)):
            j = (i + max_shift) % len(h)
            contribs[i] = h[i] * (1 - h[j])
        
        # Select top contributors
        top_indices = np.argsort(contribs)[-int(len(h) * 0.05):]  # Top 5%
        
        # Adjust selected indices
        for i in top_indices:
            j = (i + max_shift) % len(h)
            
            # Compute max feasible adjustment
            delta_h_i = min(h[i], max_move)
            delta_h_j = min(1 - h[j], max_move)
            epsilon = min(delta_h_i, delta_h_j, max_move)
            
            # Apply adjustment
            if epsilon > 0:
                h[i] -= epsilon
                h[j] += epsilon
            
        # Enforce constraint: sum(h) = n_points / 2
        h_sum = np.sum(h)
        scale_factor = (n_points / 2) / h_sum
        h = h * scale_factor
        
        # Ensure bounds are preserved
        h = np.clip(h, 0, 1)
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
