import numpy as np
import time

def evaluate_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: a smooth step function
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative update with temperature and damping
    epsilon = 0.1
    damping = 0.995
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Compute update factors for multiplicative updates
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # Update each h[i] using multiplicative update rule
        h_new = h.copy()
        max_lag = n_points - 1
        
        for i in range(n_points):
            # Compute contribution from all lags
            total_factor = 0.0
            count = 0
            for k in range(-max_lag, max_lag + 1):
                if 0 <= i + k < n_points:
                    idx = k + n_points
                    total_factor += exp_neg_S[idx] / (1.0 + exp_pos_S[idx])
                    count += 1
            if count > 0:
                h_new[i] *= total_factor * damping
            
        # Ensure feasibility
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
