from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fftpack import fft, ifft

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    def spectral_init(n_samples=50):
        h_pool = []

            base_h = np.array(initial_h_values)
        else:
            base_h = np.ones(n_samples)
            base_h[n_samples//2:] = 0

            noise = np.random.normal(0.0, 0.1, n_points)
            perturbed = base_h + noise
            perturbed = np.clip(perturbed, 0, 1)
            # Normalize to sum = target_sum
            perturbed = perturbed * (target_sum / np.sum(perturbed))
            h_pool.append(perturbed.copy())
        return h_pool

    # Early stopping threshold
    target_c5 = 0.3807
    
    # 1. Initial spectral initialization
    candidates = spectral_init(50)
    evaluated = []
    for i, candidate in enumerate(candidates):
        c5 = evaluate_c5(candidate, dx)
        evaluated.append((candidate, c5))
    evaluated.sort(key=lambda x: x[1])
    
    # Keep top 10 for refinement
    top_candidates = [x[0] for x in evaluated[:10]]

    refined = []

        if time.time() - start_time > budget_s - 5:
            break
            
        # Nelder-Mead local search
        def obj(h):
            return evaluate_c5(h, dx)

        def feasible_h(x):
            x = np.clip(x, 0, 1)
            x = x * (target_sum / np.sum(x))
            return x
            
        # Warm start from the existing solution
        h_init = feasible_h(candidate)
        
        # Use minimize with Nelder-Mead
        try:
            result = minimize(obj, h_init, method='Nelder-Mead', 
                              options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-6})
            refined_h = feasible_h(result.x)
            refined_c5 = evaluate_c5(refined_h, dx)
            refined.append((refined_h, refined_c5))

            refined.append((h_infeasible, refined_c5))
    
    # Final refinement

    return (best_h.tolist(), final_c5, n_points)
