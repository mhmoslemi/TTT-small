import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Use a convex optimization approach with piecewise constant functions
    # For better resolution, let's construct with fewer points, say 30 intervals
    M = 30
    N = 2 * M  # Number of discrete points for representation
    dx_new = 2.0 / N
    
    # Create intervals
    interval_width = 2.0 / M
    interval_starts = np.arange(0, 2, interval_width)
    interval_ends = np.arange(interval_width, 2 + interval_width, interval_width)
    
    # Initialize h_j values for intervals
    h_j = np.ones(M) * 0.5  # start with midpoint value
    if 'initial_h_values' in globals():
        # Interpolate initial_h_values to intervals
        h_interp = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        for i in range(M):
            start_idx = int(i * interval_width / dx_new)
            end_idx = int((i+1) * interval_width / dx_new)
            if end_idx > start_idx:
                h_j[i] = np.mean(h_interp[start_idx:end_idx])
            else:
                h_j[i] = h_interp[start_idx]
        
    # Ensure it sums to 1 with proper scaling
    h_j = h_j * (N / 2) / np.sum(h_j * interval_width)
    h_j = np.clip(h_j, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_j.copy()
    
    # Formulate convex optimization problem
    # We'll do this in a simplified fashion to stay within time limits
    
    # Objective: minimize max overlap
    # We'll optimize with gradient descent using grid search
    
    # The problem complexity requires more efficient solving - we try a restricted version
    
    # Sweep over different interval sizes for optimization
    max_iterations = 1000
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use gradient descent on interval values
        h_current = h_j.copy()
        step_size = 0.005
        
        # Compute current overlaps
        # Create step function from h_j
        h_full = np.zeros(N)
        for i in range(M):
            start_idx = int(i * interval_width / dx_new)
            end_idx = int((i+1) * interval_width / dx_new)
            if end_idx > start_idx:
                h_full[start_idx:end_idx] = h_current[i]
        
        # Ensure proper normalization (sum = 1)
        h_full = h_full * (N / 2) / np.sum(h_full)
        h_full = np.clip(h_full, 0, 1)
        
        # Calculate overlap
        overlap = correlate(h_full, 1 - h_full, mode="full")
        c5_current = np.max(overlap) * dx_new
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Early exit if good enough
        if best_c5 <= 0.3807:
            break
            
        # Gradient approximation using finite differences
        eps = 1e-4
        grad = np.zeros(M)
        for i in range(M):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            
            # Recreate step function
            h_full_plus = np.zeros(N)
            for j in range(M):
                start_idx = int(j * interval_width / dx_new)
                end_idx = int((j+1) * interval_width / dx_new)
                if end_idx > start_idx:
                    h_full_plus[start_idx:end_idx] = h_plus[j]
            
            h_full_plus = h_full_plus * (N / 2) / np.sum(h_full_plus)
            h_full_plus = np.clip(h_full_plus, 0, 1)
            
            overlap_plus = correlate(h_full_plus, 1 - h_full_plus, mode="full")
            c5_plus = np.max(overlap_plus) * dx_new
            
            grad[i] = (c5_plus - c5_current) / eps
            
        # Update h_j using gradient descent
        h_j -= step_size * grad
        h_j = np.clip(h_j, 0, 1)
        
        # Ensure sum constraint
        h_j = h_j * (N / 2) / np.sum(h_j * interval_width)
        
    # Final computation using corrected step function
    h_final = np.zeros(N)
    for i in range(M):
        start_idx = int(i * interval_width / dx_new)
        end_idx = int((i+1) * interval_width / dx_new)
        if end_idx > start_idx:
            h_final[start_idx:end_idx] = best_h[i]
        else:
            h_final[start_idx] = best_h[i] if start_idx < N else 0
        
    h_final = h_final * (N / 2) / np.sum(h_final)  
    h_final = np.clip(h_final, 0, 1)
    
    # Evaluate final result
    overlap_final = correlate(h_final, 1 - h_final, mode="full")
    final_c5 = np.max(overlap_final) * dx_new
    
    # Convert back to n_points format
    h_return = np.zeros(n_points)
    for i in range(n_points):
        h_return[i] = h_final[i * 2]  # Take every other point for reduction
    
    return (h_return.tolist(), final_c5, n_points)
