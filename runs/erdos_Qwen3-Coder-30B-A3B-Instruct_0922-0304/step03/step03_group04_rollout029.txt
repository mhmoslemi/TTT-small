import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with coarse resolution
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = None
    
    # Start from initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (70 / 2) / np.sum(initial_h)
        h = initial_h
        current_n = 70
    else:
        h = np.ones(10)
        h = h * (10 / 2) / np.sum(h)
        current_n = 10

    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Up-sample if needed
        if n_points > current_n:
            h = np.interp(np.linspace(0, 1, n_points), 
                         np.linspace(0, 1, current_n), h)
        elif n_points < current_n:
            h = h[::current_n//n_points] if current_n//n_points > 0 else h
            
        dx = 2.0 / n_points
        target_sum = n_points / 2

        # Constraint handling
        def constraint_sum(h):
            return np.sum(h) - target_sum

        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)

        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}

        # Initial phase: Differential Evolution
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=20,
            seed=seed,
            tol=1e-6
        )
        
        if de_result.success:
            h = de_result.x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 <= 0.3807:
                    break

        # Refinement with SLSQP
        result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )

        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 <= 0.3807:
                    break
        
        current_n = n_points

    # Return best solution
    if best_h is None:
        return (h.tolist(), evaluate_c5(h, dx), n_points)
    
    return (best_h.tolist(), best_c5, n_points)
