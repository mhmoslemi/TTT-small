import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        if len(h) != n_points:
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape with peak in center
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = (n_points - i) / (n_points - mid)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multiple initializations: parent + random perturbations
    initial_h_list = []
    
    # Add original h
    initial_h_list.append(h.copy())
    
    # Add 5 random perturbations
    for _ in range(5):
        perturbed = h + np.random.normal(0, 0.1 * np.mean(h), n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_h_list.append(perturbed)
    
    # Optimization loop
    max_iter = 1000
    iter_count = 0
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all initial points
        current_best_c5 = float('inf')
        current_best_h = None
        
        for h_init in initial_h_list:
            h_current = h_init.copy()
            
            # Run optimization
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
            else:
                h_opt = h_current
            
            # Ensure feasibility
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Evaluate
            c5 = compute_c5(h_opt)
            
            if c5 < current_best_c5:
                current_best_c5 = c5
                current_best_h = h_opt.copy()
        
        # Update global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update initial points for next iteration
        initial_h_list = []
        for _ in range(5):
            perturbed = best_h + np.random.normal(0, 0.05 * np.mean(best_h), n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            initial_h_list.append(perturbed)
        initial_h_list.append(best_h.copy())
        
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
