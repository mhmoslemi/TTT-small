import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sawtooth pattern
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    def objective(h_vec):
        h = np.array(h_vec)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Optimize using scipy
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Trust region optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Precompute for faster iteration
    iteration = 0
    max_iter = 300
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if iteration % 10 == 0:
            # Evaluate current 
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
        # Anneal bounds to prevent getting stuck
        temp_bounds = []
        for i in range(n_points):
            lb = max(0, h[i] - 0.1)
            ub = min(1, h[i] + 0.1)
            temp_bounds.append((lb, ub))
            
        # Local search using trust-constr
        try:
            res = minimize(
                objective,
                h.copy(),
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 5}
            )
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
            
        # Alternative: gradient descent with projection
        try:
            # Compute gradient approximation
            eps = 1e-4
            grad = np.zeros(n_points)
            h_base = h.copy()
            c_base = evaluate_c5(h_base)
            
            for i in range(n_points):
                h_plus = h_base.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c_plus = evaluate_c5(h_plus)
                grad[i] = (c_plus - c_base) / eps
                
            # Update with projection
            step_size = 0.01
            h = h - step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        except:
            pass
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
