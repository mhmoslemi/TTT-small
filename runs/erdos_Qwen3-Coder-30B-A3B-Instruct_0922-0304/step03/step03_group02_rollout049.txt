import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def tournament_select(population, fitnesses, tournament_size=3):
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.05):
    mask = np.random.rand(len(h)) < mutation_rate
    h = h.copy()
    h[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return h

def local_optimize(h, n_points, dx, max_iter=50):
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-9}
        )
        if result.success:
            h = result.x
    except:
        pass
    
    return repair_h(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Main evolutionary loop
    generations = 1000
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_select(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Keep top 50% (elitism)  
        elite_count = population_size // 2
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            # Local optimization for top 10%
            if len(new_population) < population_size // 10:
                child1 = local_optimize(child1, n_points, dx)
            if len(new_population) < population_size // 10:
                child2 = local_optimize(child2, n_points, dx)
                
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # Fallback to last population member
        final_h = population[-1]
        final_c5 = evaluate_c5(final_h, dx)
        best_h = final_h
    
    return (best_h.tolist(), final_c5, n_points)
