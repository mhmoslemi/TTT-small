import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Helper for pulse representation
    def pulses_to_h(pulses, n_points, dx):
        h = np.zeros(n_points)
        centers = pulses[:, 0]
        heights = pulses[:, 1]
        widths = pulses[:, 2]
        
        for i in range(len(pulses)):
            c = centers[i]
            h_ = heights[i]
            w = widths[i]
            start_idx = int(max(0, (c - w/2) / dx))
            end_idx = int(min(n_points, (c + w/2) / dx))
            h[start_idx:end_idx] = h_
        return h
    
    # Objective for pulses
    def objective_pulses(pulses_flat):
        # Reshape pulses
        pulses = pulses_flat.reshape(-1, 3)  # [center, height, width]
        h = pulses_to_h(pulses, n_points, dx)
        return compute_c5(h)
    
    # Constraint for pulses
    def constraint_sum_pulses(pulses_flat):
        pulses = pulses_flat.reshape(-1, 3)
        heights = pulses[:, 1]
        widths = pulses[:, 2]
        return np.sum(heights * widths) - 1.0

    # Main loop with pulse-based optimization
    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 200
    
    # Initialize with 5 pulses
    num_pulses = 5
    pulses = np.zeros((num_pulses, 3))
    pulse_widths = 0.4
    
    # Initial pulse placement along [0,2]
    for i in range(num_pulses):
        pulses[i, 0] = 0.2 + i * 0.4  # centers
        pulses[i, 1] = 0.5  # heights
        pulses[i, 2] = pulse_widths  # widths
    
    # Flatten pulses for optimization
    pulses_flat = pulses.flatten()
    
    # Parameters
    bounds = []
    for i in range(num_pulses):
        bounds.append((0, 2))      # center
        bounds.append((0, 1))      # height
        bounds.append((0.01, 0.8))  # width
    
    # Ensure initial sum constraint
    h = pulses_to_h(pulses, n_points, dx)
    h = h * (n_points / 2) / np.sum(h)
    pulses = pulses.flatten()
    
    # Optimization components
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Optimize with fixed pulse positions and widths, optimize heights
        previous_pulses = pulses.copy()
        pulses = previous_pulses.reshape(-1, 3)
        
        # Optimize heights
        heights = pulses[:, 1]
        # Use iterative projection on constraints
        old_sum = np.sum(heights * pulses[:, 2])
        heights = heights * (1.0 / old_sum)
        pulses[:, 1] = heights
        
        # Reshape back to flat
        pulses_flat = pulses.flatten()
        h = pulses_to_h(pulses, n_points, dx)
        h = h * (n_points / 2) / np.sum(h)
        
        # Re-resolve pulse positions and widths if needed
        # Simple perturbations
        pulses = pulses_flat.reshape(-1, 3)
        pulses[:, 0] += np.random.normal(0, 0.02, num_pulses)  # center perturb
        pulses[:, 0] = np.clip(pulses[:, 0], 0, 2)
        
        pulses_flat = pulses.flatten()
        
        # Reconstruction
        h = pulses_to_h(pulses, n_points, dx)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
