import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: clip to [0,1] and normalize to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Symmetric triangular initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
    
    h = project_h(h, n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with adaptive step size
    max_iter = 10000
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the shift that creates maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Create gradient for the maximal overlap
        grad = np.zeros_like(h)
        # For each position i, compute how much it affects the overlap
        # The gradient for h[i] is: 1 - h[i+k] for the relevant k
        for i in range(n_points):
            k = max_idx - (n_points - 1)  # The shift which gives max overlap
            if 0 <= i + k < n_points:
                grad[i] += 1 - h[i + k]  # h[i] contributes 1-h[i+k] to the overlap
            if 0 <= i - k < n_points and k != 0:
                grad[i] += -h[i - k]  # h[i-k] would contribute to the overlap term
                # If h[i-k] appears in the overlap with h[i],
                # then d/dh[i] of h[i-k] * (1-h[i]) = -h[i-k]
        
        # Use finite differences for a more robust gradient estimate
        eps = 1e-4
        grad_fd = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad_fd[i] = (c5_plus - max_overlap) / eps
        
        # Adaptive step size
        h_new = h - step_size * grad_fd
        h_new = project_h(h_new, n_points)
        
        # Check if improvement
        current_c5 = evaluate_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Adjust learning rate
        if current_c5 < max_overlap:
            step_size *= 1.1
        else:
            step_size *= 0.5
            
        if step_size < 1e-8:
            step_size = 1e-6
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
