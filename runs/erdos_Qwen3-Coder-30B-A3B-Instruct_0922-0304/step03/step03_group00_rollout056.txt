import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
    
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Add initial solution as first particle
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Add other particles with triangular functions
    for _ in range(n_particles - 1):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 - 2 * abs(i - (n_points//2)) / (n_points//2)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Track global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_score
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            
            v = w * v + c_p * r_p * (pbest[i] - h) + c_g * r_g * (gbest - h)
            
            # Apply velocity limits
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            
            # Apply boundary conditions
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to satisfy constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            new_score = evaluate_c5(h_new, dx)
            
            # Update particle best
            if new_score < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = new_score
                
                # Update global best
                if new_score < gbest_score:
                    gbest = h_new.copy()
                    gbest_score = new_score
                    if gbest_score <= 0.3807:
                        break
            
            # Update swarm
            swarm[i] = h_new
            velocities[i] = v
            
            # Update global best tracking if needed
            if new_score < best_c5:
                best_c5 = new_score
                best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
