import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: alternating pattern
        h = np.ones(n_points)
        h[::2] = 0.8
        h[1::2] = 0.2
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 200
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.1
    
    # Initialize PSO swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        # Initialize particle
        particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        swarm.append(particle.copy())
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        score = evaluate_c5(particle, dx)
        pbest.append(particle.copy())
        pbest_scores.append(score)
        
        if score < best_c5:
            best_c5 = score
            best_h = particle.copy()
    
    # PSO main loop
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest[i] - swarm[i]) +
                social * r2 * (best_h - swarm[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            score = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = score
                
                if score < best_c5:
                    best_c5 = score
                    best_h = swarm[i].copy()
    
    # SLSQP refinement
    from scipy.optimize import minimize
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Refine using SLSQP
    max_iter_slsqp = 100
    
    for _ in range(max_iter_slsqp):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 50, 'ftol': 1e-9})
        
        if result.success:
            best_h = result.x
            current_c5 = objective(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
            else:
                break
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        else:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
