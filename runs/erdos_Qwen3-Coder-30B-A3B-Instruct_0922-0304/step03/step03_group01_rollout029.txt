import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Phase 1: PSO approach for global exploration
    # Parameters
    swarm_size = 50
    max_iter_pso = 1000
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    for _ in range(swarm_size):
        particle = h.copy() + np.random.normal(0, 0.02, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        swarm.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(objective(particle))
        if pbest_vals[-1] < gbest_val:
            gbest = particle.copy()
            gbest_val = pbest_vals[-1]
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # PSO iterations
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = w * velocity + c1 * r1 * (pbest[i] - particle) + c2 * r2 * (gbest - particle)
            
            # Update position
            particle += velocity
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate
            fitness = objective(particle)
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = fitness
                if fitness < gbest_val:
                    gbest = particle.copy()
                    gbest_val = fitness
            
            # Update swarm
            swarm[i] = particle
            velocities[i] = velocity
            
        # Early exit if target achieved
        if gbest_val <= 0.3807:
            break
    
    # Initialize with better solution found by PSO
    h = gbest.copy()
    
    # Phase 2: SLSQP refinement
    max_iter_slsqp = 500
    for iteration in range(max_iter_slsqp):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Optimize with SLSQP
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 100, 'ftol': 1e-9})
            
            if result.success:
                h = result.x
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
            else:
                # If optimization fails, try local search
                step = 0.001
                iterations = 30
                h_temp = h.copy()
                
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    
                h = h_temp
                
        except:
            # Fallback to local search
            step = 0.001
            iterations = 30
            h_temp = h.copy()
            
            for _ in range(iterations):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
            
        # Feasibility repair
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
