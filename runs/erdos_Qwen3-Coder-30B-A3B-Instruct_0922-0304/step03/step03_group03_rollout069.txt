import numpy as np
import time

def evaluate_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: smooth step function
        h = np.zeros(n_points)
        center = n_points // 2
        half_width = n_points // 4
        h[center - half_width:center + half_width] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative update with exponential weighting
    epsilon = 0.1
    damping = 0.995
    max_iter = 2000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Compute exponential weights
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # Update each h[i] using proportional multiplicative updates
        new_h = np.zeros_like(h)
        for i in range(n_points):
            # Compute contribution from all lags where i is valid
            total_weight = 0.0
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    # For the specific overlap term h[i](1-h[i+k])
                    # We're computing the influence of this overlap
                    # As a relative weight to update h[i]
                    idx = k + n_points
                    # Add to total weight scaled by the overlap contribution
                    # This is a simplified approximation to motivate the update
                    total_weight += exp_neg_S[idx] / (1 + exp_pos_S[idx])
            
            # Update h[i] using multiplicative increase/decrease
            if total_weight > 0:
                update_factor = total_weight
                h[i] *= (1 + damping * (update_factor - 1))
            else:
                h[i] *= damping
                
        # Apply feasibility constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
