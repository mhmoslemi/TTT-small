import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    elite_size = population_size // 2
    tournament_size = 3
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Generate rest of population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    # Genetic algorithm
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = population[:elite_size].copy()
        
        # Tournament selection and crossover/mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(elite_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent1 = population[winner_idx]
            
            tournament_indices = np.random.choice(elite_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent2 = population[winner_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Keep only population_size
        population = new_population[:population_size]
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
