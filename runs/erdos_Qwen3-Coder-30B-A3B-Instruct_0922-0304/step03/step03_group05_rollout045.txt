import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Iterative local search with adjustment
    max_iter = 10000
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            max_move *= 0.95
            no_improvement_count = 0
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(overlap)
        max_overlap = overlap[max_k_idx]
        
        # Index i that contributes most to this overlap
        # In full correlation: overlap[k] = sum(h[i] * (1 - h[i+k]))
        # so we want to look at when overlap is maximized
        k = max_k_idx - (n_points - 1)
        
        # Find the indices i that give highest contribution to overlap
        # We want i such that h[i] * (1-h[i+k]) is highest
        contributions = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            contributions[i] = h[i] * (1 - h[j])
            
        # Sort indices by contribution
        sorted_indices = np.argsort(contributions)[::-1]
        
        # Adjust top 5% indices
        top_count = max(1, n_points // 20)
        adjusted = set()
        
        for i in sorted_indices[:top_count]:
            if i in adjusted or (i + k) % n_points in adjusted:
                continue
                
            j = (i + k) % n_points
            # Compute max adjustment
            deincrease = min(h[i], max_move)
            increase = min(1 - h[j], max_move)
            epsilon = min(deincrease, increase)
            
            if epsilon > 1e-8:
                h[i] -= epsilon
                h[j] += epsilon
                adjusted.add(i)
                adjusted.add(j)
            
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
