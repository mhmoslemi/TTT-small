import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with two pulses
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid//2] = 1.0
        h[mid//2:3*mid//4] = 0.5
        h[3*mid//4:] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 1000
    step_size = 0.01
    decay_factor = 0.5
    decay_interval = 200
    
    def compute_gradient(h, dx):
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - np.max(corr) * dx) / eps
        return grad
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Update h using projected gradient descent
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        current_c5 = evaluate_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        h = h_new
        
        # Adjust step size
        if iteration > 0 and iteration % decay_interval == 0:
            step_size *= decay_factor
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
