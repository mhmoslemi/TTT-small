import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a reasonable initialization
        h = np.ones(n_points) * 0.5
        # Start with a piecewise constant function
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Local optimization using scipy minimize with constraints
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_c5(h, dx)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h_flat):
        h = h_flat.reshape(n_points)
        return np.concatenate([h, 1 - h])
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop with different strategies
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random perturbations to escape local minima
        if iteration % 20 == 0:
            h_perturbed = h.copy()
            # Add small random noise and project back to feasible set
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed += noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            h = h_perturbed
        
        # Run optimization for a few steps with a simple method
        result = minimize(
            objective,
            h.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'disp': False}
        )
        
        if result.success:
            h_new = result.x.reshape(n_points)
            # Project back to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        else:
            # Fallback to gradient-descent-style update
            corr = correlate(h, 1 - h, mode="full")
            idx = np.argmax(corr)
            k = idx - (n_points - 1)
            if k < 0:
                grad = 1 - 2*h
            else:
                # Approximate gradient in the direction that reduces overlap
                h_shifted = np.roll(h, k)
                grad = -h_shifted + h
            step = 0.001
            h += step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate and update best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
