import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def objective_c5(h, n_points, dx):
    h = repair_h(h, n_points)
    return evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
    else:
        # Fallback initialization - step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)
    
    # Set up CMA-ES
    opts = {
        'verbose': -1,
        'seed': seed,
        'tolfun': 1e-12,
        'stopfitness': 0.3807,
        'maxiter': 6000
    }
    
    # CMA-ES parameters
    lambda_ = 100  # Population size
    initial_sigma = 0.1
    
    # Create ES
    es = CMAEvolutionStrategy(h, initial_sigma, opts)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Ask for new population
        population = es.ask(lambda_)
        fitnesses = []
        
        # Evaluate each individual in population
        for ind in population:
            fitness = objective_c5(ind, n_points, dx)
            fitnesses.append(fitness)
            
            if fitness < best_c5:
                best_c5 = fitness
                best_h = repair_h(ind, n_points).copy()
                
            # Early termination check
            if best_c5 <= 0.3807:
                break
                
        # Tell ES about fitness values
        es.tell(population, fitnesses)
        
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
