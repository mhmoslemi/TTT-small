import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is within bounds and sums to n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.05):
    """Add Gaussian noise to some genes"""
    mask = np.random.random(len(h)) < mutation_rate
    h += mask * np.random.normal(0, noise_std, len(h))
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    generations = 1000
    crossover_rate = 0.7
    mutation_rate = 0.2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    # Fill population with random individuals
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Update best
        min_fit_idx = np.argmin(fitnesses)
        if fitnesses[min_fit_idx] < best_c5:
            best_c5 = fitnesses[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best
        new_population.append(population[min_fit_idx])
        
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.extend([child1, child2])
            
        # Trim to exact pop_size
        population = new_population[:pop_size]
    
    if best_h is None:
        # Fallback: use last generation best
        fitnesses = [evaluate_c5(h, dx) for h in population]
        min_fit_idx = np.argmin(fitnesses)
        best_h = population[min_fit_idx].copy()
        best_c5 = fitnesses[min_fit_idx]
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
