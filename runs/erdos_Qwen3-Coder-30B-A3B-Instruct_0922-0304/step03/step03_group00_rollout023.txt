import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: clip to [0,1] and normalize to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient of overlap at shift k"""
    grad = np.zeros_like(h)
    # For shift k, we have terms h[i](1-h[i+k]) for valid i
    # For i in [0, n_points-1-k]: grad[i] += (1-h[i+k])
    # For i in [k, n_points-1]: grad[i] += (-h[i-k])
    
    if k >= 0:
        # i from 0 to n_points-1-k
        grad[:n_points-k] += 1 - h[k:]
        # i from k to n_points-1
        if k > 0:
            grad[k:] += -h[:n_points-k]
    else:
        # k < 0, so h[i](1-h[i+k]) means i+k from 0 to n_points-1
        k = -k
        grad[k:] += 1 - h[:n_points-k]
        # i from 0 to k-1: no contribution to gradient
        
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular
        h = np.zeros(n_points)
        for i in range(n_points//2 + 1):
            h[i] = i / (n_points//2)
            h[n_points-1-i] = i / (n_points//2)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find shift that maximizes overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        # Convert index to shift: 0 corresponds to shift -n_points+1, ..., n_points-1 to shift 0, ..., 2*n_points-2 to shift n_points-1  
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient at this shift
        grad = compute_gradient(h, max_k, n_points, dx)
        
        # Adaptive step size (simple line search)
        step = 0.01
        min_c5 = current_c5
        best_step = step
        
        for step_try in [step * 0.5, step, step * 2]:
            h_try = h - step_try * grad
            h_try = project_h(h_try, n_points)
            c5_try = evaluate_c5(h_try, dx)
            if c5_try < min_c5:
                min_c5 = c5_try
                best_step = step_try
                
        # Update with best step
        h = h - best_step * grad
        h = project_h(h, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
