import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    # Adaptive mutation parameters
    max_mutations = 500
    n_initial_samples = 50
    n_refinement_samples = 10

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (target_sum) / np.sum(h_vec)
        return h_vec

    def get_freq_mask(h):
        H = fft(h)
        # Find dominant frequencies that contribute to overlap
        abs_H = np.abs(H)
        # Mask out frequencies that are close to zero for stability
        mask = abs_H >= np.percentile(abs_H, 30)
        return mask

    def mutate_solution(h_in, mask, noise_scale=0.02):
        H = fft(h_in)
        # Apply mask to preserve certain frequency components
        H_masked = H * mask
        # Add noise to unmasked components
        H_noise = H_masked.copy()
        unmasked_indices = ~mask
        noise = np.random.normal(0, noise_scale, len(H)) * 1j
        H_noise[unmasked_indices] += noise[unmasked_indices]
        h_new = np.real(ifft(H_noise))
        return project_h(h_new)

    # Spectral initialization
    initial_mask = get_freq_mask(h)
    
    # Generate initial population
    population = [h.copy()]
    for i in range(n_initial_samples):
        if time.time() - start_time > budget_s - 10:
            break
        h_new = mutate_solution(h, initial_mask, noise_scale=0.05)
        population.append(h_new)
    
    # Evaluate initial population
    candidate_scores = []
    for h_pop in population:
        c5 = evaluate_c5(h_pop, dx)
        candidate_scores.append((c5, h_pop))

    candidate_scores.sort()  # Low to high
    # Select top candidates to refine
    top_candidates = [h_pop for _, h_pop in candidate_scores[:n_refinement_samples]]

    # Local refinement using Nelder-Mead
    refinement_budget = 30
    refine_start = time.time()
    for h_refine in top_candidates:
        if time.time() - refine_start > refinement_budget:
            break
            
        def obj(h_in):
            return evaluate_c5(project_h(h_in), dx)
        
        # Nelder-Mead optimization
        try:
            result = minimize(obj, h_refine, method='Nelder-Mead', 
                              options={'maxiter': 50, 'adaptive': True, 'xatol': 1e-6})
            if result.success:
                h_opt = project_h(result.x)
                c5_opt = evaluate_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
        # Improve the current solution with gradient descent if needed
        steps = 100
        step_size = 0.005
        h_temp = h_refine.copy()
        
        for s in range(steps):
            if time.time() - refine_start > refinement_budget:
                break
            # Compute gradient via finite differences
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - evaluate_c5(h_temp, dx)) / eps
                
            # Update with gradient
            h_temp -= step_size * grad
            h_temp = project_h(h_temp)
            
            # Update best if improvement
            c5_temp = evaluate_c5(h_temp, dx)
            if c5_temp < best_c5:
                best_c5 = c5_temp
                best_h = h_temp.copy()
                
            if best_c5 <= 0.3807:
                break

    # Final evaluation and return
    final_c5 = evaluate_c5(best_h, dx)
    final_h = best_h.tolist()
    return (final_h, final_c5, n_points)
