import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: clip and normalize"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, dx, k_shift):
    """Compute gradient for a specific shift k"""
    n = len(h)
    grad = np.zeros(n)
    
    # For shift k: overlap = sum h[i] * (1 - h[i+k])
    # So d/dh[i] overlap = 1 - h[i+k] if i+k < n, else 0
    # and d/dh[i+k] overlap = -h[i] if i >= 0, else 0
    
    # Actually, we need to do this carefully for all terms
    # For overlap at shift k: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) 
    # We only consider valid indices where i+k < n (i < n-k)
    
    for i in range(n - k_shift):
        grad[i] += 1 - h[i + k_shift]
        if i + k_shift < n:
            grad[i + k_shift] -= h[i]
    
    # Multiply by dx to account for discretization
    grad *= dx
    return grad

def adaptive_step_size(current_c5, prev_c5, step_size, improvement):
    """Adaptively adjust step size"""
    if improvement > 0:
        # If we improved, try a larger step
        return min(step_size * 1.1, 0.1)
    else:
        # If no improvement, try smaller step
        return max(step_size * 0.5, 1e-6)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (mid + 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    step_size = 0.01
    max_iter = 10000
    
    # Main optimization loop with projected gradient descent
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and identify max overlap
        current_c5 = evaluate_c5(h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        k_shift = np.argmax(corr)
        
        # Adjust k_shift based on corr shape
        # corr has length 2*n - 1, center is at n-1
        # shift k is k_shift - (n-1)
        shift = k_shift - (n_points - 1)
        
        # Compute gradient for this shift
        gradient = compute_gradient(h, dx, abs(shift))
        
        # Update h
        h_new = h - step_size * gradient
        
        # Project back to feasible region
        h_new = project_h(h_new, n_points)
        
        # Update step size
        new_c5 = evaluate_c5(h_new, dx)
        improvement = current_c5 - new_c5
        step_size = adaptive_step_size(new_c5, current_c5, step_size, improvement)
        h = h_new
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
