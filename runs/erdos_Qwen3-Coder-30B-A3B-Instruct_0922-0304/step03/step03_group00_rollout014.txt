import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # SDP approach
    # We'll use a hacky but effective SDP relaxation
    def solve_sdp(h_guess, dx):
        # Create variables
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # We'll add empirical constraints based on correlation values
        # This is a heuristic approach - approximate the overlaps using the guess
        corr = correlate(h_guess, 1 - h_guess, mode="full")
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx] * dx
        
        # Add the main constraint C5 >= max_overlap
        # We use a simplified model here rather than full SDP because it's more computationally tractable
        # However, we still want the convex relaxation to be "tight enough"
        
        # Let's create an encoded representation
        # Try a simple gradient-free optimization:
        # Use a projected gradient descent with fixed stepsize to lower overlap
        h_cur = h_guess.copy()
        for _ in range(1000):
            corr_result = correlate(h_cur, 1 - h_cur, mode="full")
            max_corr_idx = np.argmax(corr_result)
            max_overlap = corr_result[max_corr_idx] * dx
            
            if max_overlap < 0.3807:  # Early stopping
                break
                
            # Compute gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h_cur)
            h_plus = h_cur.copy()
            for i in range(n_points):
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                h_plus[i] -= eps  # Revert
            
            # Update
            h_cur -= 0.005 * grad  # Step-size
            h_cur = np.clip(h_cur, 0, 1)
            h_cur = h_cur * (n_points / 2) / np.sum(h_cur)
            
            # Update best 
            current_c5 = evaluate_c5(h_cur, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_cur.copy()
                
        return h_cur, best_c5
    
    # Try multiple random restarts to avoid local minima
    for _ in range(20):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random perturbation
        h_perturb = h.copy()
        h_perturb += np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        
        # Apply optimization
        h_opt, c5_val = solve_sdp(h_perturb, dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
