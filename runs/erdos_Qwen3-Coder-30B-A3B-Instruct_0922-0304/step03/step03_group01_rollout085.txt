import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_size = pop_size // 5
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def repair_individual(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Create initial population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        population.append(create_individual())
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Evolution loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s * 0.8 and no_improvement_count < max_no_improvement:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)  # Negative because we minimize
        
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best_c5 = -fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
        
        # Selection: top elite_size individuals
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Crossover and mutation for the rest
        while len(new_population) < pop_size:
            # Tournament selection
            if np.random.rand() < crossover_rate:
                parent1 = population[np.random.randint(0, elite_size)]
                parent2 = population[np.random.randint(0, elite_size)]
                
                # Crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    mutation_idx = np.random.randint(0, n_points)
                    child1[mutation_idx] = np.random.rand()
                if np.random.rand() < mutation_rate:
                    mutation_idx = np.random.randint(0, n_points)
                    child2[mutation_idx] = np.random.rand()
                
                # Repair and add to new population
                new_population.append(repair_individual(child1))
                if len(new_population) < pop_size:
                    new_population.append(repair_individual(child2))
            else:
                # Just mutation
                parent = population[np.random.randint(0, elite_size)]
                child = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.random.rand()
                new_population.append(repair_individual(child))
        
        # Update population
        population = new_population
        generation += 1
    
    # Local refinement with gradient descent from best solution
    if time.time() - start_time < budget_s * 0.95:
        h_refined = best_h.copy()
        step = 0.001
        max_iter = 1000
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Gradient estimation
            eps = 1e-5
            grad = np.zeros_like(h_refined)
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update step
            h_refined -= step * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            # Check for convergence
            if np.allclose(h_refined, best_h, atol=1e-6):
                break
        
        # Final check
        final_c5 = evaluate_c5(h_refined, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
