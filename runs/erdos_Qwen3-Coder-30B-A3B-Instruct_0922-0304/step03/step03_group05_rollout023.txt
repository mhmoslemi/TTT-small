import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent style adjustment
    adjustment_magnitude = 0.01
    patience = 0
    max_patience = 100
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Reduce adjustment magnitude if stuck
                adjustment_magnitude *= 0.95
                patience = 0
                if adjustment_magnitude < 1e-6:
                    break
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # Find the maximum overlap and its location
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Find the shifts that contribute the most
        # correlate returns full convolution, we want the middle region
        middle_start = n_points - 1
        middle_end = 2 * n_points - 2
        # For maximum overlap at max_idx, find corresponding k
        if max_idx >= middle_start:
            k = max_idx - middle_start
        else:
            # This shouldn't happen with proper indexing
            k = max_idx
            
        # Identify indices that contribute most to the overlap at shift k
        # Look for i such that h[i] * (1 - h[(i+k)%n_points]) is maximal
        contributions = []
        for i in range(n_points):
            idx_k = (i + k) % n_points
            contribution = h[i] * (1 - h[idx_k])
            contributions.append((contribution, i, idx_k))
        
        # Sort by contribution
        contributions.sort(reverse=True)
        top_contributors = contributions[:max(1, n_points // 10)]
        
        # Adjust h values
        for contrib, i, idx_k in top_contributors:
            # Try to reduce h[i] and increase h[idx_k]
            delta = min(adjustment_magnitude, h[i], 1 - h[idx_k])
            if delta > 0:
                h[i] -= delta
                h[idx_k] += delta
                
        # Normalize to maintain sum = n_points / 2
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
