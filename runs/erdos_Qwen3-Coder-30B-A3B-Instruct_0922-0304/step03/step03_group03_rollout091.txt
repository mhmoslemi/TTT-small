import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative update approach
    epsilon = 0.1
    damping = 0.99
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        S = corr * dx
        
        # Compute multiplicative factors
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # Update each h[i] proportionally
        h_new = np.zeros_like(h)
        for i in range(n_points):
            contrib = 0.0
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    idx = k + n_points
                    contrib += exp_neg_S[idx] / (1 + exp_pos_S[idx])
            h_new[i] = h[i] * contrib * damping
            
        # Apply updates with feasibility repair
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
