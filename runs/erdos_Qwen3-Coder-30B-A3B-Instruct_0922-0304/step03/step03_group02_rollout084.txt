import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create simple initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define objective function for CMA-ES
    def objective(h_vec):
        h = repair_h(h_vec, n_points)
        return evaluate_c5(h, dx)
    
    # CMA-ES setup
    mean = h.copy()
    sigma = 0.1
    es = CMAEvolutionStrategy(mean, sigma, {'popsize': 100, 'seed': seed})
    
    # Optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate new candidate solutions
        X = es.ask()
        fitness = []
        
        # Evaluate each candidate
        for h_candidate in X:
            f = objective(h_candidate)
            fitness.append(f)
        
        # Tell the optimizer about fitness values
        es.tell(X, fitness)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        current_h = repair_h(X[best_idx], n_points)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
