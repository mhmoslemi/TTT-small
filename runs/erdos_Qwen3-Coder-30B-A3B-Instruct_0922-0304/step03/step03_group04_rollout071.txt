import numpy as np
from scipy.signal import correlate
from pywt import wavedec, waverec
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    max_wavelet_iter = 3
    max_spatial_iter = 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet domain optimization
    for wavelet_iter in range(max_wavelet_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Wavelet transform
        try:
            coeffs = wavedec(h, 'haar', level=6)
        except:
            coeffs = [h]
            
        # Soft thresholding in wavelet domain
        threshold = 0.1 * (1 + wavelet_iter * 0.1)
        new_coeffs = [soft_threshold(c, threshold) for c in coeffs]
        
        # Reconstruct in spatial domain
        try:
            h_wavelet = waverec(new_coeffs, 'haar')
        except:
            h_wavelet = h
            
        h = h_wavelet[:n_points]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Spatial refinement
    for spatial_iter in range(max_spatial_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Select significant wavelet coefficients for spatial refinement
        try:
            coeffs = wavedec(best_h, 'haar', level=6)
        except:
            coeffs = [best_h]
            
        # Sort by absolute values and select top 20%
        all_coeffs = np.concatenate(coeffs)
        sorted_indices = np.argsort(np.abs(all_coeffs))[::-1]
        top_20_percent = sorted_indices[:max(1, len(sorted_indices) // 5)]
        
        # Optimize spatially with these coefficients
        indices_to_refine = []
        for i, c in enumerate(coeffs):
            if i == 0:  # approximation coefficients
                indices_to_refine.extend(list(range(len(c))))
            else:
                # Refine detail coefficients
                start = sum(len(coeffs[j]) for j in range(i))
                end = start + len(c)
                indices_to_refine.extend(list(range(start, end)))
                break
                
        # Simple gradient descent on selected coefficients
        for _ in range(15):
            if time.time() - start_time > budget_s - 2:
                break
            # Compute correlation gradient estimation
            corr = correlate(best_h, 1 - best_h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Gradient via finite differences
            grad = np.zeros_like(best_h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update
            step = 0.01
            best_h -= step * grad
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            
        # Evaluate after refinement
        c5 = evaluate_c5(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Ensure final solution is valid
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
