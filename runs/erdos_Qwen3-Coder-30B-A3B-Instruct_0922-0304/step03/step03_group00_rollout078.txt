import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the shift with maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Formulate and solve QP for this shift
        h_var = cp.Variable(n_points)
        
        # Construct quadratic objective matrix for shift max_shift
        Q = np.zeros((n_points, n_points))
        shift = max_shift
        if shift > 0:
            # h[i] * (1 - h[i + shift]) for i = 0 .. n_points - shift - 1
            for i in range(n_points - shift):
                Q[i, i + shift] = 1.0
                Q[i + shift, i] = 1.0
        elif shift < 0:
            # h[i] * (1 - h[i + shift]) for i = -shift .. n_points - 1
            for i in range(-shift, n_points):
                Q[i, i + shift] = 1.0
                Q[i + shift, i] = 1.0
        else:
            # Special case: shift = 0
            Q = np.eye(n_points)
        
        # Objective: minimize h^T * Q * h * dx
        obj = cp.Minimize(cp.quad_form(h_var, Q) * dx)
        
        # Constraints
        cons = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve the QP
        prob = cp.Problem(obj, cons)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=0.2)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Check feasibility and numerical stability
                if h_new is not None and np.all(0 <= h_new) and np.all(h_new <= 1):
                    h = h_new.copy()
                else:
                    # Fallback to simple update if QP fails or gives infeasible solution
                    step = 0.01
                    h += step * (1 - 2 * h)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Use simple descent if QP does not solve optimally
                step = 0.01
                h += step * (1 - 2 * h)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback to simple gradient descent if QP fails
            step = 0.01
            h += step * (1 - 2 * h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
