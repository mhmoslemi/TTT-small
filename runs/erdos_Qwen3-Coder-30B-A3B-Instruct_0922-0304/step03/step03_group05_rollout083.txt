import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape
        h = np.ones(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = (i + 1) / (mid + 1)
        for i in range(mid, n_points):
            h[i] = (n_points - i) / (n_points - mid + 1)
        h = np.clip(h, 0, 1)
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start with 5 random perturbations
    initial_candidates = [h.copy()]
    for _ in range(4):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_candidates.append(perturbed)

    # Local optimization on all candidates
    for candidate in initial_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(
            objective, 
            candidate, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = candidate
            
        # Ensure feasibility
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Basin Hopping with decreasing perturbation
    current_h = best_h.copy()
    perturbation_sigma = 0.1
    cooling_factor = 0.995
    max_basin_hopping = 1000
    
    for i in range(max_basin_hopping):
        if time.time() - start_time > budget_s - 1:
            break

        # Random perturbation
        perturbed = current_h + np.random.normal(0, perturbation_sigma, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        
        # Local optimization
        result = minimize(
            objective, 
            perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = perturbed
            
        # Ensure feasibility
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            current_h = h_opt.copy()
        else:
            current_h = perturbed
        
        # Decrease perturbation
        perturbation_sigma *= cooling_factor
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Adaptive resolution
    if time.time() - start_time < budget_s - 10 and n_points == 70:
        # Interpolate to higher resolution
        high_res_h = np.interp(np.linspace(0, 2, 140), np.linspace(0, 2, 70), best_h)
        # Normalize
        high_res_h = high_res_h * (140 / 2) / np.sum(high_res_h)
        # Optimize at higher resolution
        n_points = 140
        dx = 2.0 / n_points

        def compute_c5_high_res(h):
            overlap = correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx

        def objective_high_res(h):
            return compute_c5_high_res(h)

        def constraint_sum_high_res(h):
            return np.sum(h) - n_points / 2

        constraints = [{'type': 'eq', 'fun': constraint_sum_high_res}]
        bounds = [(0, 1) for _ in range(n_points)]

        result = minimize(
            objective_high_res, 
            high_res_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            refined_h = result.x
        else:
            refined_h = high_res_h
            
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        
        final_c5 = compute_c5_high_res(refined_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = refined_h.copy()

    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
