import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[10:30] = 0.5
        h[40:60] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation parameters
    eps = 1e-4
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration >= 10000:  # Prevent infinite loops
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Line search
        alpha = 0.01
        max_iter = 20
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < current_c5:
                h = h_new
                break
            alpha *= 0.5
            
        # Periodic perturbation
        if no_improvement_count > 50:
            noise_magnitude = 1e-3 / (1 + iteration ** 0.5)
            h += np.random.normal(0, noise_magnitude, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
