import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: clip and normalize"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient of overlap w.r.t. h for shift k"""
    grad = np.zeros_like(h)
    # For shift k, the derivative of h[i]*(1-h[i+k]) is:
    # (1-h[i+k]) for i-th component
    # -h[i-k] for (i+k)-th component (if i+k < n)
    
    # Compute for positive shifts
    if k >= 0:
        grad[:-k] += (1 - h[k:])
        grad[k:] -= h[:-k]
    else:  # negative shift
        grad[-k:] += (1 - h[:k])
        grad[:k] -= h[-k:]
        
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to symmetric triangular
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main projected gradient descent
    max_iter = 10000
    step_size = 0.01
    patience = 0
    min_patience = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Find the shift k that maximizes overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, k, n_points, dx)
        
        # Update h
        h_new = h - step_size * grad
        
        # Project
        h = project_h(h_new, n_points)
        
        # Adaptive step size
        if patience > min_patience:
            step_size *= 0.99
            patience = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
