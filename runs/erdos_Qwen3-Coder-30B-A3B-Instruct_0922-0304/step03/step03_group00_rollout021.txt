import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, dx):
    """Compute gradient of overlap at shift k"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Compute overlap for shift k
    # For the shift k, we have h[i] * (1 - h[i+k]) 
    # Partials w.r.t h[i]: 1 - h[i+k] if i+k < n
    # Partials w.r.t h[i+k]: -h[i] if i >= 0
    
    # Handle shift k properly
    if k >= 0:
        # h[i] * (1 - h[i+k]) for i in [0, n-k-1]
        for i in range(n - k):
            grad[i] += 1 - h[i + k]
        for i in range(k, n):
            grad[i] -= h[i - k]
    else:
        # k is negative: h[i] * (1 - h[i+k]) for i in [0, n+k-1]
        for i in range(n + k):
            grad[i] += 1 - h[i - k]
        for i in range(-k, n):
            grad[i] -= h[i + k]
            
    return grad * dx

def find_max_overlap_shift(h):
    """Find the shift k that causes maximum overlap"""
    n = len(h)
    overlaps = np.zeros(2 * n - 1)
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    return max_idx - (n - 1)

def adaptive_step_size(current_c5, prev_c5, step, improvement_factor):
    """Adapt step size based on previous improvement"""
    if prev_c5 - current_c5 > 1e-6:
        return step * improvement_factor
    elif current_c5 > prev_c5:
        return step / improvement_factor
    return step

def evaluate_c5(h, dx):
    """Evaluate C5 for current h"""
    correlation = correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / (mid + 1))
    
    # Normalize
    h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters for gradient descent
    step_size = 0.01
    improvement_factor = 1.2
    max_iter = 10000
    prev_c5 = float('inf')
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Find shift causing max overlap
        k = find_max_overlap_shift(h)
        
        # Compute gradient
        grad = compute_gradient(h, k, dx)
        
        # Update h
        h = h - step_size * grad
        
        # Project back to feasible region
        h = project_h(h, n_points)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adapt step size
        step_size = adaptive_step_size(current_c5, prev_c5, step_size, improvement_factor)
        prev_c5 = current_c5
        
        # Early termination
        if best_c5 <= 0.3807:

    # Final check
    final_c5 = evaluate_c5(best_h, dx)
