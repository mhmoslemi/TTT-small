import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
    else:
        # Fallback triangular shape
        h_parent = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_parent[i] = max(0, 1 - abs(i - mid) / (n_points / 4))
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h_parent.copy()

    # Multiple initial points for diversity
    initial_hs = [h_parent.copy()]
    
    # Add random perturbations
    for _ in range(4):
        h_pert = h_parent + np.random.normal(0, 0.05, n_points)
        h_pert = project(h_pert)
        initial_hs.append(h_pert)

    # Optimization loop over all initial points
    for h_start in initial_hs:
        h = h_start.copy()
        last_improvement = 0
        for iteration in range(100):  # Max iterations per start
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
            # Early stopping
            if best_c5 <= 0.3807 or last_improvement > 50:
                break

            # Local optimization
            result = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
            else:
                # Perturb if optimization fails
                h += np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            h = project(h)
            
            # Prevent stagnation
            if iteration % 20 == 0 and iteration > 0:
                h += np.random.normal(0, 0.01, n_points)
                h = project(h)

    # Final computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
