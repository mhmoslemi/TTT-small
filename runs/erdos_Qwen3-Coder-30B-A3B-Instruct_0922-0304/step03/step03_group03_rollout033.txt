import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, trace, diag
from cvxpy import Semidef

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 200
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # SDP-based optimization approach
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp_sum(h_vars) == n_points / 2,
            t >= 0
        ]
        
        # Use SDP relaxation to get tighter bounds
        # We build a more sophisticated relaxation for the quadratic terms
        # For each i,j pair, we introduce constraint h[i]*(1-h[j]) <= constraint
        # But we also consider pairing h[i]*(1-h[i]) terms with sum constraints
        
        # Add constraints for each k lag
        max_lag = n_points - 1
        # For each lag, construct the overlap sum
        for k in range(-max_lag, max_lag + 1):
            # Compute overlap terms for lag k (positive and negative)
            if k >= 0:
                # Positive lag: h[i] * (1 - h[i + k]) for valid i
                terms = []
                for i in range(n_points - k):
                    terms.append(h_vars[i] * (1 - h_vars[i + k]))
                if terms:
                    # Add no more than one quadratic constraint per lag
                    constraints.append(cp_sum(terms) <= t)
            else:
                # Negative lag: correspond to positive lag case
                k = -k
                terms = []
                for i in range(n_points - k):
                    terms.append(h_vars[i] * (1 - h_vars[i + k]))
                if terms:
                    constraints.append(cp_sum(terms) <= t)
        
        # Objective: minimize t (upper bound on overlap)
        prob = Problem(Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except:
            # If SDP fails, use hybrid local search
            pass
            
        # Local gradient-based refinement
        if iteration % 5 == 0:
            # Compute current overlap
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            
            # Finite difference approach for gradient
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - np.max(corr) * dx) / eps
            
            # Update with gradient and step size
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Adjust step size
            if iteration % 50 == 0:
                step_size *= 0.9
                
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
