import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization constraints handling
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Active constraints for optimization
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization with trust-region method
    max_iter = 2000
    patience = 0
    patience_limit = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using trust-region method
        result = minimize(
            fun=evaluate_c5,
            x0=h,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'disp': False},
            callback=None
        )
        
        if result.success:
            h = result.x
            current_c5 = evaluate_c5(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        else:
            # Perturb if optimization fails
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Feasibility repair and periodic perturbations
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally perturb to escape local minima
        if iteration % 20 == 0 and patience > 0:
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Reinitialize if no progress
        if patience > patience_limit:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
