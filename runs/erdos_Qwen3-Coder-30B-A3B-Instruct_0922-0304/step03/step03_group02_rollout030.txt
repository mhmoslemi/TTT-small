import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to ensure it's in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000  # Limit iterations to fit within time budget
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = repair_h(h_init, n_points)
    else:
        # Fallback initialization: step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = repair_h(h_init, n_points)
    
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Define objective function
    def objective(h):
        if time.time() - start_time > budget_s - 1:
            return float('inf')  # Signal to stop
        h = repair_h(h, n_points)
        return evaluate_c5(h, dx)
    
    # CMA-ES parameters
    es = CMAEvolutionStrategy(h_init, 0.1, {'popsize': 100, 'seed': seed})
    
    # Optimization loop
    iteration = 0
    while not es.stop() and iteration < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate new population
        solutions = es.ask()
        
        # Evaluate solutions
        fitnesses = []
        for sol in solutions:
            fitness = objective(sol)
            fitnesses.append(fitness)
            
        # Tell CMA-ES
        es.tell(solutions, fitnesses)
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = repair_h(solutions[best_idx], n_points)
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
