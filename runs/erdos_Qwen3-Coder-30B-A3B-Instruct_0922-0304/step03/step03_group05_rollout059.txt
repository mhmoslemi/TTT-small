import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints dictionary
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start with basin hopping
    n_initials = 5
    initial_solutions = []
    
    # Generate initial solutions
    for _ in range(n_initials):
        # Initialize from the main one with noise
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_solutions.append(perturbed.copy())
    
    # Add the original solution too
    initial_solutions.append(h.copy())
    
    # Optimization loop
    max_iter = 2000
    cooling_rate = 1000
    sigma_start = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate each initialization
        for init_sol in initial_solutions:
            current_h = init_sol.copy()
            
            # Local optimization with SLSQP
            result = minimize(
                lambda x: compute_c5(x),
                current_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                current_h = result.x
            else:
                current_h = result.x
            
            # Ensure feasibility
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
            # Evaluate and track best
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        # Basin hopping: random perturbations with decreasing step size
        if iteration % 100 == 0 and iteration > 0:
            # Decrease step size
            sigma = sigma_start * np.exp(-iteration / cooling_rate)
            for i in range(len(initial_solutions)):
                # Perturb the best solution
                if iteration % 200 == 0:
                    # Randomize the initialization
                    new_h = np.random.rand(n_points)
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                else:
                    # Perturb current best or one of the initial solutions
                    idx = np.random.randint(len(initial_solutions))
                    perturbed = initial_solutions[idx].copy() + np.random.normal(0, sigma, n_points)
                    perturbed = np.clip(perturbed, 0, 1)
                    perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
                    new_h = perturbed
                
                # Local optimization
                result = minimize(
                    lambda x: compute_c5(x),
                    new_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                
                if result.success:
                    new_h = result.x
                else:
                    new_h = result.x
                    
                # Ensure feasibility
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Evaluate and possibly update the best
                current_c5 = compute_c5(new_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = new_h.copy()
                
                # Update the list of initial solutions (keep diversity)
                if i < len(initial_solutions) - 1:
                    initial_solutions[i] = new_h.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Final refinement with higher tolerance
    if time.time() - start_time < budget_s - 2:
        final_result = minimize(
            lambda x: compute_c5(x),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if final_result.success:
            final_h = final_result.x
        else:
            final_h = best_h.copy()
            
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = compute_c5(final_h)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
