import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO Parameters
    swarm_size = 50
    max_iter_pso = 1000
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Initialize PSO swarm
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    
    # Ensure sum constraint for initial swarm
    for i in range(swarm_size):
        positions[i] = np.clip(positions[i], 0, 1)
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # PSO optimization loop
    for pso_iter in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Evaluate current position
            current_position = positions[i]
            current_score = evaluate_c5(current_position, dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_positions[i] = current_position.copy()
            
            # Update global best
            if current_score < global_best_score:
                global_best_score = current_score
                global_best_position = current_position.copy()
                
        # Update best solution
        if global_best_score < best_c5:
            best_c5 = global_best_score
            best_h = global_best_position.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best_positions[i] - positions[i]) + 
                            c2 * r2 * (global_best_position - positions[i]))
            
            positions[i] += velocities[i]
            
            # Clip and normalize
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # SLSQP refinement phase
    if global_best_position is not None:
        # Use the best from PSO as starting point
        h = global_best_position.copy()
        best_c5 = global_best_score
        
        # Optimization constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint: sum(h) = n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # SLSQP optimization
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 500, 'ftol': 1e-9})
            
            if result.success:
                h_opt = result.x
                current_c5 = objective(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                    
                    # Ensure feasibility
                    best_h = np.clip(best_h, 0, 1)
                    best_h = best_h * (n_points / 2) / np.sum(best_h)
        except:
            pass  # Fall back to previous best if SLSQP fails
            
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
