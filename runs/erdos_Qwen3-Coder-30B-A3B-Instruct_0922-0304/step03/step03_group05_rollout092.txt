import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Custom initialization
        h = np.zeros(n_points)
        # Place 5 pulses of equal width
        pulse_width = 0.4
        centers = np.linspace(0.2, 1.8, 5)
        heights = np.ones(5) * 0.5
        
        for i, (center, height) in enumerate(zip(centers, heights)):
            start_idx = max(0, int((center - pulse_width/2)/dx))
            end_idx = min(n_points, int((center + pulse_width/2)/dx))
            h[start_idx:end_idx] = height
            
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hyperparameters
    n_pulses = 5
    max_iter = 1000
    
    # Initialization for pulses
    pulse_centers = np.linspace(0.2, 1.8, n_pulses)
    pulse_heights = np.ones(n_pulses) * 0.5
    pulse_widths = np.ones(n_pulses) * 0.4

    def pulses_to_h(centers, heights, widths):
        h = np.zeros(n_points)
        for i in range(n_pulses):
            start_idx = max(0, int((centers[i] - widths[i]/2)/dx))
            end_idx = min(n_points, int((centers[i] + widths[i]/2)/dx))
            h[start_idx:end_idx] = heights[i]
        return np.clip(h, 0, 1)

    def compute_pulse_h(pulse_centers, pulse_heights, pulse_widths):
        # Normalize so that sum(h) * dx = 1
        total_mass = np.sum(pulse_heights * pulse_widths)
        h = pulses_to_h(pulse_centers, pulse_heights, pulse_widths)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update h from pulse params
        h = compute_pulse_h(pulse_centers, pulse_heights, pulse_widths)
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Optimize heights for fixed centers/widths
        total_mass = np.sum(pulse_heights * pulse_widths)
        pulse_heights = np.ones(n_pulses) * (1 / total_mass)
        h = compute_pulse_h(pulse_centers, pulse_heights, pulse_widths)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Optimize centers for fixed heights/widths
        def obj_centers(centers_flat):
            centers = centers_flat.reshape(-1)
            h_test = pulses_to_h(centers, pulse_heights, pulse_widths)
            norm_factor = (n_points / 2) / np.sum(h_test)
            h_test = h_test * norm_factor
            return compute_c5(h_test)
        
        # Slightly adjust center positions
        for i in range(n_pulses):
            best_shift = 0
            best_val = current_c5
            for shift in np.linspace(-0.1, 0.1, 10):
                test_centers = pulse_centers.copy()
                test_centers[i] += shift
                test_centers = np.clip(test_centers, 0, 2)
                val = obj_centers(test_centers)
                if val < best_val:
                    best_val = val
                    best_shift = shift
            pulse_centers[i] += best_shift
            if pulse_centers[i] < 0:
                pulse_centers[i] = 0
            if pulse_centers[i] > 2:
                pulse_centers[i] = 2
                
        # Clamp pulse heights
        pulse_heights = np.clip(pulse_heights, 0, 1)
        
        # Renormalize
        h = compute_pulse_h(pulse_centers, pulse_heights, pulse_widths)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
