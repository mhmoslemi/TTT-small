import numpy as np
from scipy.signal import correlate
from pywt import wavedec, waverec
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet-based optimization
    max_iterations = 30
    spatial_refinements = 2
    wavelet_level = 6
    threshold_lambda = 0.1
    learning_rate = 0.01
    patience = 10
    
    # Initialize wavelet coefficients
    coeffs = wavedec(h, 'haar', level=wavelet_level)
    theta = np.concatenate(coeffs)
    
    # Optimization iterations
    wavelet_iter = 0
    spatial_iter = 0
    patience_counter = 0
    best_wavelet_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        # Wavelet domain optimization step
        if wavelet_iter < max_iterations:
            # Compute gradient using finite differences
            grad = np.zeros_like(theta)
            eps = 1e-5
            for i in range(len(theta)):
                theta_plus = theta.copy()
                theta_plus[i] += eps
                # Reconstruct h from wavelet coefficients
                reconstructed_coeffs = [theta_plus[:len(coeffs[0])], 
                                      theta_plus[len(coeffs[0]):len(coeffs[0]) + len(coeffs[1])],
                                      theta_plus[len(coeffs[0]) + len(coeffs[1]):]]
                while len(reconstructed_coeffs) < wavelet_level + 1:
                    reconstructed_coeffs.append(np.zeros_like(reconstructed_coeffs[-1]))
                h_plus = waverec(reconstructed_coeffs, 'haar')
                if len(h_plus) > n_points:
                    h_plus = h_plus[:n_points]
                elif len(h_plus) < n_points:
                    h_plus = np.pad(h_plus, (0, n_points - len(h_plus)))
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                current_c5 = evaluate_c5(h_plus, dx)
                grad[i] = (current_c5 - evaluate_c5(h, dx)) / eps
            
            # Update theta
            theta -= 0.1 * grad
            theta = np.sign(theta) * np.maximum(np.abs(theta) - threshold_lambda, 0)
            
            # Reconstruct h from theta
            reconstructed_coeffs = [theta[:len(coeffs[0])], 
                                  theta[len(coeffs[0]):len(coeffs[0]) + len(coeffs[1])],
                                  theta[len(coeffs[0]) + len(coeffs[1]):]]
            while len(reconstructed_coeffs) < wavelet_level + 1:
                reconstructed_coeffs.append(np.zeros_like(reconstructed_coeffs[-1]))
            h = waverec(reconstructed_coeffs, 'haar')
            if len(h) > n_points:
                h = h[:n_points]
            elif len(h) < n_points:
                h = np.pad(h, (0, n_points - len(h)))
            
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate C5
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                patience_counter = 0
                best_wavelet_c5 = current_c5
            else:
                patience_counter += 1
            
            wavelet_iter += 1
            
            # Adapt threshold_lambda if no improvement
            if patience_counter > patience:
                threshold_lambda *= 1.5
                patience_counter = 0
        else:
            # Spatial refinement with Adam optimization
            if spatial_iter < spatial_refinements:
                # Use the top 20% wavelet coefficients to identify features
                wavelet_magnitude = np.abs(theta)
                top_indices = np.argsort(wavelet_magnitude)[-int(len(theta)*0.2):]
                
                # Create a function to optimize the spatial representation
                def adam_optimizer(h_inputs, steps=50):
                    # Fast little Adam optimizer
                    m = np.zeros_like(h_inputs)
                    v = np.zeros_like(h_inputs)
                    beta1, beta2 = 0.9, 0.999
                    epsilon = 1e-8
                    alpha_t = learning_rate
                    
                    for i in range(steps):
                        if time.time() - start_time > budget_s - 1:
                            break
                            
                        # Compute gradient
                        grad = np.zeros_like(h_inputs)
                        eps = 1e-5
                        for j in range(len(h_inputs)):
                            h_plus = h_inputs.copy()
                            h_plus[j] += eps
                            current_c5 = evaluate_c5(h_plus, dx)
                            grad[j] = (current_c5 - evaluate_c5(h_inputs, dx)) / eps
                        # Update Adam
                        m = beta1 * m + (1 - beta1) * grad
                        v = beta2 * v + (1 - beta2) * grad**2
                        m_hat = m / (1 - beta1**(i+1))
                        v_hat = v / (1 - beta2**(i+1))
                        h_inputs -= alpha_t * m_hat / (np.sqrt(v_hat) + epsilon)
                        
                    h_inputs = np.clip(h_inputs, 0, 1)
                    h_inputs = h_inputs * (n_points / 2) / np.sum(h_inputs)
                    return h_inputs
                
                # Update only the top features
                h_top = h.copy()
                h_top = adam_optimizer(h_top, steps=20)
                
                current_c5 = evaluate_c5(h_top, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_top.copy()
                
                spatial_iter += 1
            else:
                break

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
