import numpy as np
from scipy.signal import correlate
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    # Precompute wavelet decomposition
    wavelet = 'haar'
    max_level = int(np.log2(n_points)) - 1
    if max_level < 2:
        max_level = 2
        
    # Wavelet transform
    def to_wavelet_domain(h):
        coeffs = pywt.wavedec(h, wavelet, level=max_level)
        return np.concatenate(coeffs)
    
    def from_wavelet_domain(theta):
        # Reconstruct from wavelet coefficients
        coeffs = []
        start = 0
        for i in range(max_level, 0, -1):
            length = n_points // (2**i)
            coeffs.append(theta[start:start + length])
            start += length
        coeffs.append(theta[start:])
        return pywt.waverec(coeffs, wavelet)[:n_points]
    
    # Thresholding function
    def soft_threshold(x, threshold):
        return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)
    
    # Optimization loop
    wavelet_iter = 0
    max_wavelet_iter = 30
    spatial_refinements = 0
    max_spatial_refinements = 2
    stagnation_count = 0
    best_c5_stagnation = float('inf')
    lambda_thresh = 0.1
    
    while time.time() - start_time < budget_s - 1 and wavelet_iter < max_wavelet_iter:
        # Wavelet domain optimization
        theta = to_wavelet_domain(h)
        
        # Gradient approximation
        eps = 1e-4
        grad = np.zeros_like(theta)
        for i in range(len(theta)):
            theta_plus = theta.copy()
            theta_plus[i] += eps
            h_plus = from_wavelet_domain(theta_plus)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - evaluate_c5(h, dx)) / eps
        
        # Update coefficients with soft thresholding
        alpha = 0.1  # step size
        new_theta = theta - alpha * grad
        new_theta = soft_threshold(new_theta, lambda_thresh)
        
        # Update h via wavelet basis
        h_temp = from_wavelet_domain(new_theta)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        h = h_temp
        
        # Spatial refinement
        if spatial_refinements < max_spatial_refinements and (wavelet_iter % 5 == 0):
            # Refine spatially using gradient descent on top features
            # Identify significant wavelet coefficients
            wavelet_magnitude = np.abs(theta)
            top_indices = np.where(wavelet_magnitude > np.percentile(wavelet_magnitude, 80))[0]
            
            # Iterate optimization on top features
            for _ in range(100):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute overlap metric
                overlap = correlate(h, 1 - h, mode="full")
                max_overlap = np.max(overlap) * dx
                
                # Compute gradient
                grad_spatial = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(overlap_plus) * dx
                    grad_spatial[i] = (c5_plus - max_overlap) / eps
                
                # Update with adaptive learning rate
                h -= 0.005 * grad_spatial
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
            best_c5_stagnation = best_c5
        else:
            stagnation_count += 1
            if stagnation_count > 10:
                lambda_thresh *= 1.5  # Increase sparsity
                stagnation_count = 0
                
        wavelet_iter += 1
        spatial_refinements += 1

    # Final cleanup and verification
    final_c5 = evaluate_c5(best_h, dx)
    
    if final_c5 > best_c5:
        final_c5 = best_c5
        h = best_h.copy()
    
    return (h.tolist(), final_c5, n_points)
