import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Enhanced local search with projected gradient
        step = 0.01
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * dx
            max_idx = np.argmax(corr)
            
            # Approximate gradient via finite difference
            grad = np.zeros_like(h)
            eps = 1e-4
            h_eval = h.copy()
            for i in range(n_points):
                h_plus = h_eval.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full") 
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Apply gradient step with constraint handling
            h_new = h - step * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check if improvement
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
            else:
                step *= 0.95
            
    # Final refinement with convex optimization
    h_vars = Variable(n_points)
    t = Variable()
    
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp_sum(h_vars) == n_points / 2,
        t >= 0
    ]
    
    # Simplified overlap constraints
    max_lag = n_points - 1
    for k in range(max_lag + 1):
        if k == 0:
            continue  # Skip zero lag for concept simplicity, as it's redundant
        # Use convex relaxation
        overlap_expr = 0
        for i in range(n_points - k):
            # Simple linearization for quadratic term
            overlap_expr += h_vars[i] + h_vars[i + k] - h_vars[i] * h_vars[i + k]
        if overlap_expr > 0:
            constraints.append(overlap_expr <= t)
    
    # Objective: minimize t
    prob = Problem(Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_vars.value
            if h_new is not None:
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
