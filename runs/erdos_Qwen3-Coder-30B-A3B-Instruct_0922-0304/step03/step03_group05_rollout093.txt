import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Ensure h is in [0,1] and sum(h) * dx = 1"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def sample_h_from_pulses(pulse_params):
        """Convert pulse parameters to h"""
        h = np.zeros(n_points)
        for i in range(len(pulse_params)//3):
            height = max(0, min(1, pulse_params[i*3]))
            center = pulse_params[i*3 + 1]
            width = max(0.01, pulse_params[i*3 + 2])  # Minimum width
            start = max(0, center - width/2)
            end = min(2, center + width/2)
            start_idx = int(start / dx)
            end_idx = int(end / dx) + 1
            for j in range(start_idx, min(end_idx, n_points)):
                h[j] += height * min(1, (end - start) / width)  # Simple linear interpolation
        return project_h(h)

    def get_pulses_from_h(h):
        """Naive conversion from h to pulse params - used for initial sampling"""
        # This is geometrically complex, so we'll use a heuristic approach
        return None  # Let's optimize directly in pulse space
        
    def evaluate_pulse_params(pulse_params):
        """Evaluate objective for given pulse parameters"""
        h = sample_h_from_pulses(pulse_params)
        return compute_c5(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Pulse configuration: [height1, center1, width1, height2, center2, width2, ...]
    n_pulses = 5
    current_pulse_params = np.ones(n_pulses * 3)
    current_pulse_params[0::3] = 0.5  # Heights
    current_pulse_params[1::3] = np.linspace(0.1, 1.9, n_pulses)  # Centers
    current_pulse_params[2::3] = 0.5  # Widths

    # Gradient step size for each parameter
    step_size = 0.01
    max_iter = 1000

    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_pulse_params(current_pulse_params)
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Convert best parameters back to h for return
            best_h = sample_h_from_pulses(current_pulse_params)
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Optimization step: perturb each parameter slightly
        for param_idx in range(len(current_pulse_params)):
            param_name = ['height', 'center', 'width'][param_idx % 3]
            # Try both + and - step
            param_old = current_pulse_params[param_idx]
            for sign in [-1, 1]:
                current_pulse_params[param_idx] = param_old + sign * step_size
                if param_name == 'height':
                    # Ensure height is valid
                    if current_pulse_params[param_idx] < 0:
                        current_pulse_params[param_idx] = 0
                    elif current_pulse_params[param_idx] > 1:
                        current_pulse_params[param_idx] = 1
                elif param_name == 'center':
                    # Keep center in [0,2]
                    if current_pulse_params[param_idx] < 0:
                        current_pulse_params[param_idx] = 0
                    elif current_pulse_params[param_idx] > 2:
                        current_pulse_params[param_idx] = 2
                elif param_name == 'width':
                    # Keep width positive, not too big
                    if current_pulse_params[param_idx] < 0.01:
                        current_pulse_params[param_idx] = 0.01
                    elif current_pulse_params[param_idx] > 1:
                        current_pulse_params[param_idx] = 1
                        
                new_c5 = evaluate_pulse_params(current_pulse_params)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    # Accept the step
                else:
                    # Revert step
                    current_pulse_params[param_idx] = param_old

    # Final evaluation and return
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
