import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Multi-start optimization with basin hopping
    num_starts = 5
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start with initial solution and perturbed copies
    candidates = [h.copy()]
    for _ in range(num_starts - 1):
        perturbed_h = h + np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        candidates.append(perturbed_h)
    
    # Optimization loop
    for candidate in candidates:
        # Local optimization with SLSQP
        result = minimize(
            objective, 
            candidate, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            optimized_h = result.x
        else:
            optimized_h = candidate
            
        # Enforce constraints
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h = optimized_h * (n_points / 2) / np.sum(optimized_h)
        
        current_c5 = compute_c5(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h.copy()
            
        if time.time() - start_time > budget_s - 1:
            break
            
    # Basin hopping
    current_h = best_h.copy()
    max_basin_hopping = 1000
    for i in range(max_basin_hopping):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        perturbation_sigma = 0.05 * np.exp(-i / 500.0)  # Decreasing noise
        perturbed_h = current_h + np.random.normal(0, perturbation_sigma, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Local optimization
        result = minimize(
            objective, 
            perturbed_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            optimized_h = result.x
        else:
            optimized_h = perturbed_h
            
        # Enforce constraints
        optimized_h = np.clip(optimized_h, 0, 1)
        optimized_h = optimized_h * (n_points / 2) / np.sum(optimized_h)
        
        current_c5 = compute_c5(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h.copy()
            current_h = optimized_h.copy()
        else:
            current_h = optimized_h.copy()
    
    # Try refinement at higher resolution if time allows
    if time.time() - start_time < budget_s - 10 and n_points == 70:
        # Interpolate to higher resolution
        higher_res = 140
        dx_high = 2.0 / higher_res
        h_high = np.interp(np.linspace(0, 2, higher_res), np.linspace(0, 2, n_points), best_h)
        h_high = np.clip(h_high, 0, 1)
        h_high = h_high * (higher_res / 2) / np.sum(h_high)
        
        # Local optimization at this resolution
        def compute_c5_high(h):
            overlap = correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx_high
            
        def objective_high(h):
            return compute_c5_high(h)

        constraints_high = [{'type': 'eq', 'fun': lambda h: np.sum(h) - higher_res / 2}]
        bounds_high = [(0, 1) for _ in range(higher_res)]

        result = minimize(
            objective_high, 
            h_high, 
            method='SLSQP', 
            bounds=bounds_high, 
            constraints=constraints_high,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            h_final = result.x
        else:
            h_final = h_high
            
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (higher_res / 2) / np.sum(h_final)
        
        final_c5 = compute_c5_high(h_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final.copy()
            n_points = higher_res

    return (best_h.tolist(), best_c5, n_points)
