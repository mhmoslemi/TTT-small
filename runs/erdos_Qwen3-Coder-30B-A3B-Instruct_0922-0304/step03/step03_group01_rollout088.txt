import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    if initial_h_values is not None and len(initial_h_values) == n_points:
        population = [np.array(initial_h_values)]
        population = [repair_h(h, n_points) for h in population]
    else:
        population = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    return population

def mutate_individual(h, mutation_rate, n_points):
    if np.random.rand() < mutation_rate:
        idx = np.random.randint(0, n_points)
        h[idx] = np.random.rand()
    return repair_h(h, n_points)

def crossover(parent1, parent2, crossover_rate):
    if np.random.rand() < crossover_rate:
        crossover_point = np.random.randint(1, len(parent1)-1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    return parent1.copy(), parent2.copy()

def genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_generations, dx, initial_h_values):
    population = initialize_population(n_points, pop_size, initial_h_values)
    best_fitness = float('inf')
    best_individual = None
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_c5(h, dx)
            fitnesses.append(fitness)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Selection (tournament)
        sorted_indices = np.argsort(fitnesses)
        selected = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(selected), 2):
            p1 = selected[i]
            p2 = selected[(i+1) % len(selected)]
            child1, child2 = crossover(p1, p2, crossover_rate)
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            offspring.extend([child1, child2])
        
        # Elitism
        elite_size = pop_size // 5
        elite_indices = sorted_indices[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Replace population
        population = elite + offspring[:pop_size - elite_size]
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_generations = 100
    
    # Run genetic algorithm
    ga_time = budget_s * 0.8
    ga_start = time.time()
    
    best_ga_h, best_ga_c5 = genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_generations, dx, initial_h_values if 'initial_h_values' in globals() else None)
    
    if time.time() - ga_start > ga_time:
        return (best_ga_h.tolist(), best_ga_c5, n_points)
    
    # Local refinement with gradient descent
    def gradient_descent_step(h, dx, learning_rate=0.01, max_iter=50):
        h = h.copy()
        for _ in range(max_iter):
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Use GA result as starting point for local refinement
    refined_h = gradient_descent_step(best_ga_h, dx)
    refined_c5 = evaluate_c5(refined_h, dx)
    
    # Return best solution found
    if refined_c5 < best_ga_c5:
        return (refined_h.tolist(), refined_c5, n_points)
    else:
        return (best_ga_h.tolist(), best_ga_c5, n_points)
