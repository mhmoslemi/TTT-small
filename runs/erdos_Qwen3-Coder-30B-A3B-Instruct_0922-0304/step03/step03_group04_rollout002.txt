import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points_list = [10, 20, 40, 70, 100, 140, 200]
    dx_list = [2.0 / n for n in n_points_list]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for idx, (n_points, dx) in enumerate(zip(n_points_list, dx_list)):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize h
        if idx == 0:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Up-sample from previous resolution
            prev_h = best_h
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(prev_h)), prev_h)
            # Add noise for diversity
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Define constraints and bounds
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Global search with Differential Evolution
        de_result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=50,
            popsize=20,
            disp=False
        )
        
        if de_result.success:
            h = de_result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n = n_points
                
        # Local refinement with SLSQP
        refinement_result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        
        if refinement_result.success:
            h = refinement_result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n = n_points
                
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx_list[n_points_list.index(best_n)])
    else:
        final_c5 = float('inf')
        
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = np.clip(best_h, 0, 1)
        else:
            best_h = np.ones(70)
            best_h[35:] = 0
        best_h = best_h * (70 / 2) / np.sum(best_h)
        final_c5 = evaluate_c5(best_h, 2.0/70)
        best_n = 70
        
    return (best_h.tolist(), final_c5, best_n)
