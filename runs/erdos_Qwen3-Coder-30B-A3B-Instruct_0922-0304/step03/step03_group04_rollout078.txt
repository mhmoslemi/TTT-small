import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points_list = [10, 20, 40, 70, 100, 150, 200]
    dx_list = [2.0 / n for n in n_points_list]
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Use initial_h_values if available as starting point
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Ensure it sums to n/2 where n is the length
        n_points = len(initial_h)
        dx = 2.0 / n_points
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h, dx)
        best_n_points = n_points
        
    for idx, n_points in enumerate(n_points_list):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = dx_list[idx]
        
        # Initialize h
        if idx == 0:
            # Start with a simple step function
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Use previous solution as starting point
            prev_h = best_h
            # Interpolate to new resolution
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(prev_h)), prev_h)
            # Add noise for diversity
            noise = np.random.normal(0, 0.03, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - (n_points / 2)
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Phase 1: Global search with Differential Evolution
        if n_points <= 100:
            de_result = differential_evolution(
                objective,
                bounds,
                seed=seed,
                maxiter=100,
                popsize=20,
                disp=False
            )
            if de_result.success:
                h = de_result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Phase 2: Refinement with SLSQP
        refinement_iter = 200
        for _ in range(refinement_iter):
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': 1e-10}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    best_n_points = n_points
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
                    
        # If neither global nor local search improved current solution
        # Perturb slightly and try again
        if n_points <= 100 and idx > 0:
            # Try locally perturbing more aggressively in early stages
            h_perturbed = h.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed += noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_c5(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                best_n_points = n_points
                
    # Final evaluation to ensure best solution is valid
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
        return (best_h.tolist(), best_c5, best_n_points)
    
    # Fallback: return last known good solution
    # Construct an initial solution if none found
    h = np.ones(70)
    h[35:] = 0
    h = h * (70 / 2) / np.sum(h)
    c5 = evaluate_c5(h, 2.0/70)
    return (h.tolist(), c5, 70)
