import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    lambda_pop = 100  # Population size for CMA-ES
    sigma = 0.1  # Initial step size
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES parameters
    mu = n_points // 4  # Number of parents
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)  # Normalize weights
    
    # Covariance matrix and evolution path
    C = np.eye(n_points)
    p_c = np.zeros(n_points)
    p_s = np.zeros(n_points)
    
    # Main CMA-ES loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sample new population
        population = []
        for _ in range(lambda_pop):
            # Sample from multivariate normal distribution
            z = np.random.randn(n_points)
            x = h + sigma * (C @ z)
            # Repair feasibility
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            population.append(x)
        
        # Evaluate population
        fitness = []
        for x in population:
            f = evaluate_c5(x, dx)
            fitness.append(f)
        
        # Sort by fitness
        indices = np.argsort(fitness)
        fitness = [fitness[i] for i in indices]
        population = [population[i] for i in indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            if best_c5 <= 0.3807:
                break
        
        # Update mean
        h_new = np.zeros(n_points)
        for i in range(mu):
            h_new += weights[i] * population[i]
        h = h_new
        
        # Update evolution paths
        p_s = 0.3 * p_s + np.sqrt(0.3 * (2 - 0.3)) * (h - h) / sigma
        p_c = 0.3 * p_c + np.sqrt(0.3 * (2 - 0.3)) * (h - h) / sigma

        # Update covariance matrix
        # This is a simplified implementation - in practice, more complex updates needed
        # For performance, we'll just rescale the diagonal
        C = np.diag(np.diag(C) * 0.99 + 0.01 * (np.eye(n_points) * 0.1))

        # Update step size
        sigma = sigma * np.exp(0.2 * (np.linalg.norm(p_s) / np.sqrt(n_points) - 0.5))
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
