import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    # Clip to [0,1] then normalize
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, dx):
    # Compute gradient of max overlap w.r.t h for shift k
    n = len(h)
    grad = np.zeros(n)
    # For each position i, this contributes to overlap at shifts k where i+k is valid
    for i in range(n):
        if i + k < n:
            grad[i] += (1 - h[i + k]) * dx
        if i - k >= 0:
            grad[i] += h[i - k] * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps to find the max shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (len(h) - 1)
        
        # Compute gradient for the max shift
        grad = compute_gradient(h, max_shift, dx)
        
        # Adaptive step size using line search
        step_size = 0.01
        current_c5 = evaluate_c5(h, dx)
        
        # Try a few step sizes to find a good one
        best_h_trial = h.copy()
        best_c5_trial = current_c5
        for _ in range(5):
            h_trial = h - step_size * grad
            h_trial = project_h(h_trial, n_points)
            c5_trial = evaluate_c5(h_trial, dx)
            if c5_trial < best_c5_trial:
                best_c5_trial = c5_trial
                best_h_trial = h_trial
            step_size *= 0.5
            
        h = best_h_trial
        current_c5 = best_c5_trial
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
