import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
from scipy.optimize import differential_evolution as de

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    start_time = time.time()

    dx = 2.0 / n_points

    h = repair_h(h, n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Wrap constraints for differential evolution
    constraints = {'type': 'eq', 'fun': constraint_sum}

        # Try multiple strategies to avoid local optima
        strategy = strategies[iteration % len(strategies)]
        
        # Use differential evolution for global search with constraints
        result = de(
            objective,
            bounds,
            constraints=constraints,
            strategy=strategy,
            maxiter=30,
