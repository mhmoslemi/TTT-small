import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    n_particles = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        global_best = np.array(initial_h_values)
        global_best = np.clip(global_best, 0, 1)
        global_best = global_best * (n_points / 2) / np.sum(global_best)
    else:
        global_best = np.random.uniform(0, 1, n_points)
        global_best = global_best * (n_points / 2) / np.sum(global_best)
    
    global_best_c5 = evaluate_c5(global_best, dx)
    particles = []
    velocities = []
    particle_best = []
    particle_best_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_c5(h, dx)
        particle_best.append(h.copy())
        particle_best_c5.append(c5)
        if c5 < global_best_c5:
            global_best = h.copy()
            global_best_c5 = c5
    
    # PSO iterations
    max_iter = int(budget_s * 20)  # Roughly 20 iterations per second
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (particle_best[i] - particles[i]) + 
                           social * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < particle_best_c5[i]:
                particle_best[i] = particles[i].copy()
                particle_best_c5[i] = current_c5
                
                # Update global best
                if current_c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = current_c5
            
            # Early exit
            if global_best_c5 <= 0.3807:
                break
                
        if global_best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
