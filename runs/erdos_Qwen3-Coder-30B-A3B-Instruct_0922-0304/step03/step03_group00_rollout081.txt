import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    swarm.append(h0)
    velocities.append(np.zeros_like(h0))
    
    # Add particles with triangular initialization
    for _ in range(n_particles - 1):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 - 2 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track best
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_c5 = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand()
            r_g = np.random.rand()
            vel = (w * velocities[i] + 
                   c_p * r_p * (pbest[i] - swarm[i]) + 
                   c_g * r_g * (gbest - swarm[i]))
            # Clip velocity
            vel = np.clip(vel, -v_max, v_max)
            velocities[i] = vel
            
            # Update position
            new_h = swarm[i] + vel
            new_h = np.clip(new_h, 0, 1)
            # Normalize to sum constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Evaluate
            new_c5 = evaluate_c5(new_h, dx)
            
            # Update personal best
            if new_c5 < pbest_c5[i]:
                pbest[i] = new_h.copy()
                pbest_c5[i] = new_c5
                
                # Update global best
                if new_c5 < gbest_c5:
                    gbest = new_h.copy()
                    gbest_c5 = new_c5
                    
            swarm[i] = new_h.copy()
            
        # Early stopping
        if gbest_c5 <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_c5, n_points)
