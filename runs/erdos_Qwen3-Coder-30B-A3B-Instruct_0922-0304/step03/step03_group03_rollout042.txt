import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: pulse-like function
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 2000
    epsilon = 0.05
    damping = 0.99
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps for current h
        corr = correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Update each h[i] using multiplicative updates
        h_new = np.zeros_like(h)
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # For each position, compute weighted update
        for i in range(n_points):
            numerator = 0.0
            denominator = 0.0
            for k in range(-n_points+1, n_points):
                if 0 <= i+k < n_points:
                    idx = k + n_points - 1
                    if idx < len(exp_neg_S):
                        numerator += exp_neg_S[idx]
                        denominator += exp_pos_S[idx]
            if denominator > 0:
                update_factor = numerator / (1 + denominator)
            else:
                update_factor = 1.0
            h_new[i] = h[i] * update_factor * damping
        
        # Apply updates with feasibility repair
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
