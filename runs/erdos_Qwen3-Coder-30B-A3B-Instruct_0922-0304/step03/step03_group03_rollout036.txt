import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate a good initial guess using a windowing approach
        h = np.zeros(n_points)
        # Create a piecewise constant function with 3 segments
        segment_len = n_points // 3
        h[:segment_len] = 0.5
        h[segment_len:2*segment_len] = 1.0
        h[2*segment_len:] = 0.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing parameters
    temp = 1.0
    temp_decay = 0.995
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Local perturbation with simulated annealing
        h_new = h.copy()
        
        # Randomly modify one or two segments
        num_modifications = np.random.choice([1, 2])
        for _ in range(num_modifications):
            idx = np.random.randint(n_points)
            # Modify nearby values
            window_size = max(1, n_points // 20)
            start_idx = max(0, idx - window_size)
            end_idx = min(n_points, idx + window_size)
            
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.03)
            h_new[start_idx:end_idx] += perturbation
            
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept or reject based on temperature
        if current_c5 > 0:
            delta = evaluate_c5(h_new, dx) - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_new
        else:
            h = h_new
            
        # Cool down
        temp *= temp_decay
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
