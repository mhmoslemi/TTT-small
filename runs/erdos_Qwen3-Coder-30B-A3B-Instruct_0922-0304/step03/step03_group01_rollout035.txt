import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is feasible: in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_individual(h, mutation_rate=0.1, n_points=None):
    """Apply mutation to an individual"""
    h_new = h.copy()
    for i in range(len(h_new)):
        if np.random.random() < mutation_rate:
            # Either Gaussian perturbation or reset
            if np.random.random() < 0.5:
                h_new[i] += np.random.normal(0, 0.05)
            else:
                h_new[i] = np.random.random()
    return repair_h(h_new, n_points)

def crossover_individuals(h1, h2, crossover_rate=0.7):
    """Crossover two individuals"""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1)-1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return h1_new, h2_new

def genetic_algorithm(n_points, pop_size=50, max_generations=200, 
                      mutation_rate=0.1, crossover_rate=0.7, 
                      elitism_rate=0.2, dx=2.0/n_points):
    """Genetic Algorithm for optimizing h"""
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.random(n_points)
            h = repair_h(h, n_points)
        population.append(h)
    
    best_fitness = float('inf')
    best_individual = None
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_c5(h, dx)
            fitnesses.append(fitness)
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism
        elite_count = int(elitism_rate * pop_size)
        new_population = population[:elite_count]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            child1, child2 = crossover_individuals(parent1, parent2, crossover_rate)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        
        # Check if target reached
        if best_fitness <= 0.3807:
            break
            
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
    
    # Time allocation
    ga_time = budget_s * 0.8
    ref_time = budget_s * 0.2
    ga_start = time.time()
    
    # Run GA
    ga_time_left = ga_time - (time.time() - ga_start)
    if ga_time_left > 1.0:
        best_h, best_c5 = genetic_algorithm(
            n_points=n_points,
            pop_size=50,
            max_generations=100,
            mutation_rate=0.1,
            crossover_rate=0.7,
            elitism_rate=0.2,
            dx=dx
        )
    else:
        best_h, best_c5 = h, evaluate_c5(h, dx)
    
    # Refine with local search
    refine_start = time.time()
    if time.time() - refine_start < ref_time:
        # Use gradient ascent to refine the best from GA
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Simple gradient descent
        max_iter = 500
        step_size = 0.001
        for iteration in range(max_iter):
            if time.time() - refine_start > ref_time - 0.1:
                break
                
            # Compute gradient estimator
            grad = np.zeros_like(best_h)
            eps = 1e-4
            obj_base = objective(best_h)
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                obj_plus = objective(h_plus)
                grad[i] = (obj_plus - obj_base) / eps
            
            # Update
            best_h = best_h - step_size * grad
            best_h = repair_h(best_h, n_points)
        
        best_c5 = objective(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
