import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Piecewise constant optimization approach
    M = 30  # number of intervals
    interval_width = 2.0 / M
    interval_centers = np.linspace(interval_width/2, 2 - interval_width/2, M)
    
    # Initialize piecewise constant approximation
    if 'initial_h_values' in globals():
        # Average initial samples into intervals
        h_piecewise = np.zeros(M)
        for i in range(M):
            start_idx = int(i * n_points / M)
            end_idx = int((i+1) * n_points / M)
            h_piecewise[i] = np.mean(h[start_idx:end_idx])
        h_piecewise = np.clip(h_piecewise, 0, 1)
    else:
        h_piecewise = np.ones(M) * 0.5
        h_piecewise[:M//2] = 1.0
        h_piecewise[M//2:] = 0.0

    # Define CVXPY problem
    h_vars = Variable(M)
    C = Variable()
    constraints = [0 <= h_vars, h_vars <= 1, 
                   sum(h_vars * interval_width) == 1, 
                   C >= 0]
    
    # Compute overlaps for each k
    for k in range(2*M):
        if k == 0:
            continue
        # Approximate correlation coefficients for each k
        # This is a simplified approximation for efficiency
        overlap_expr = 0
        for i in range(M):
            j = i + k - M  # wrap around
            if 0 <= j < M:
                overlap_expr += h_vars[i] * (1 - h_vars[j]) * interval_width
        constraints.append(overlap_expr <= C)
    
    # Objective
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    # Solve
    prob.solve(solver=ECOS, max_iter=1000, verbose=False)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        h_opt = h_vars.value
    else:
        h_opt = h_piecewise
    
    # Convert back to original resolution
    h_final = np.repeat(h_opt, n_points // M)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Evaluate for final result
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
