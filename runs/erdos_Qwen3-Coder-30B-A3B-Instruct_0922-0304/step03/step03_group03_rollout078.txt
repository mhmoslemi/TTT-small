import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Construct dual variables with heuristic weights
        # Simplified approach: use importance sampling
        lag_weights = np.zeros(2 * n_points - 1)
        corr = correlate(h, 1 - h, mode="full")
        # Some weight computation
        for k in range(-n_points + 1, n_points):
            idx = k + n_points - 1
            if 0 <= idx < len(lag_weights):
                lag_weights[idx] = corr[idx]
        
        # Normalize weights
        lag_weights = np.maximum(lag_weights, 0)
        if np.sum(lag_weights) > 0:
            lag_weights = lag_weights / np.sum(lag_weights)
            
        # Build problem to compute constraint weights (dual approach)
        # Directly solve for improved solution using convex relaxation
        h_vars = Variable(n_points)
        t = Variable()
        
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp_sum(h_vars) == n_points / 2,
            t >= 0
        ]
        
        # Approximate through known inequalities 
        # For each k, we bound sum h[i] * (1 - h[i+k]) with valid constraints
        max_lag = n_points - 1
        for k in range(-max_lag, max_lag + 1):
            # Convert k to positive index for full convolution result
            idx = k + n_points - 1
            if 0 <= idx < 2 * n_points - 1:
                # Create constraint incorporating the lag weighting
                # This rule augmented: use Lagrange multipliers interpretation
                if k >= 0:
                    # Positive lag
                    expr = 0
                    for i in range(n_points - k):
                        expr += h_vars[i] * (1 - h_vars[i + k])
                    if idx < len(lag_weights) and lag_weights[idx] > 0:
                        constraints.append(expr <= t)  # Simplified: full max 
                else:
                    # Negative lag - symmetric
                    expr = 0
                    for i in range(n_points + k):
                        expr += h_vars[i] * (1 - h_vars[i - k])
                    if idx < len(lag_weights) and lag_weights[idx] > 0:
                        constraints.append(expr <= t)
        
        # Objective: minimize t (potential metric)
        prob = Problem(Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except:
            pass
            
        # Local improvement step using simple gradient descent on max overlap
        step = 0.002
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h, 1 - h, mode="full") * dx
            max_overlap = np.max(corr)
            max_idx = np.argmax(corr)
            
            grad = np.zeros_like(h)
            eps = 1e-4
            h_eval = h.copy()
            
            for i in range(n_points):
                h_plus = h_eval.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
                c5_plus = np.max(corr_plus)
                grad[i] = (c5_plus - max_overlap) / eps
                
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
