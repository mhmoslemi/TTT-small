import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        # Interpolate to new resolution if needed
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        h = h_parent.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Generate multiple initial points
    initial_candidates = []
    
    # Parent initialization
    initial_candidates.append(h.copy())
    
    # Random perturbations
    for _ in range(5):
        h_pert = h + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_candidates.append(h_pert)
        
    # Fallback initialization if needed
    if not initial_candidates:
        h_fallback = np.zeros(n_points)
        # Create a simple triangular shape
        middle = n_points // 2
        for i in range(middle):
            h_fallback[i] = i / middle
            h_fallback[n_points - 1 - i] = i / middle
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        initial_candidates.append(h_fallback)
        
    # Evaluate all initial candidates and choose best
    for h_init in initial_candidates:
        h_current = h_init.copy()
        for _ in range(20):  # Run a few iterations for each
            if time.time() - start_time > budget_s - 1:
                break
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break            
                
            # Optimize using SLSQP
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                h_current += np.random.normal(0, 0.01, n_points)
            # Ensure feasibility
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Check timeout
            if time.time() - start_time > budget_s - 1:
                break
            
            # Terminate early if no improvement in last 10 iterations
            if current_c5 <= best_c5:
                pass  # Proceed to next iteration
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
