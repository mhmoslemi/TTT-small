import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    
    # Generate initial random population
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    # Normalize the population to sum constraint
    for h in population:
        h *= (n_points / 2) / np.sum(h)
    
    # Genetic algorithm parameters
    generations = 1000
    tournament_size = 3
    crossover_rate = 0.8
    mutation_rate = 0.2
    elite_size = pop_size // 2
    local_search_rate = 0.1
    
    def tournament_selection(pop, fits, k=tournament_size):
        idxs = np.random.randint(0, len(pop), k)
        best_idx = idxs[np.argmin([fits[i] for i in idxs])]
        return pop[best_idx].copy()
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        return child1, child2
    
    def mutate(h, noise_std=0.05):
        mask = np.random.rand(n_points) < mutation_rate
        if np.any(mask):
            noise = np.random.normal(0, noise_std, n_points)
            h = h + noise * mask
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        return h
    
    def local_search(h, max_iter=50):
        # Simple gradient descent approach to refine
        def objective(h):
            return evaluate_c5(h, dx)
            
        for _ in range(max_iter):
            current_c5 = objective(h)
            # Estimate gradient using finite differences
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_eps = h.copy()
                h_eps[i] += eps
                h_eps = np.clip(h_eps, 0, 1)
                h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
                grad[i] = (objective(h_eps) - current_c5) / eps

            # Gradient descent step
            step_size = 1e-4
            h_new = h - step_size * grad
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Accept if it improves
            if objective(h_new) < current_c5:
                h = h_new
                
        return h
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fit = evaluate_c5(h, dx)
            fitnesses.append(fit)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Create new generation
        new_pop = []
        
        # Elite individuals
        elites = population[:elite_size]
        new_pop.extend(elites)
        
        # Generate offspring
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
                
        # Local search on top 10%
        num_local = int(pop_size * local_search_rate)
        for i in range(num_local):
            if np.random.rand() < 0.5:
                new_pop[i] = local_search(new_pop[i])
                
        population = new_pop
    
    # Final validation and evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
