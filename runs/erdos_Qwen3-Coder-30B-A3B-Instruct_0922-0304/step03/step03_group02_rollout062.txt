import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_solution(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    selected = []
    for _ in range(len(population) // 2):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(child, mutation_rate=0.2, noise_std=0.05):
    for i in range(len(child)):
        if np.random.random() < mutation_rate:
            child[i] += np.random.normal(0, noise_std)
    return child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    num_generations = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_solution(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_solution(h, n_points)
        population.append(h)
    
    # Add random initial individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair_solution(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_c5(h, dx)
            fitnesses.append(fitness)
            
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i + 1] if i + 1 < len(selected) else selected[0]
            
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = repair_solution(child1, n_points)
            child2 = repair_solution(child2, n_points)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
    
    # Final evaluation
    if best_h is None:
        # Fallback in case no improvement was found
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_solution(h, n_points)
        best_h = h
        best_c5 = evaluate_c5(best_h, dx)
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
