import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, k):
    """Compute gradient for shift k"""
    n = len(h)
    grad = np.zeros(n)
    # For shift k, compute partial derivatives
    # h[i] appears in terms where h[i] and h[i+k] are both involved
    # Both h[i] and h[i+k] contribute to the overlap if both are active
    # Omit boundary terms for simplicity
    for i in range(n):
        if i + k < n and i + k >= 0:
            grad[i] += 1 - h[i + k]
            grad[i + k] += -h[i]
    return grad * dx

def evaluate_c5(h, dx):
    """Evaluate max overlap for current h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Fallback initialization: triangular
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h = project_h(h, n_points)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5 and find the maximum overlap shift
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Find the shift that gives maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        shift = max_idx - (len(h) - 1)
        
        # Compute gradient for that shift
        grad = compute_gradient(h, dx, shift)
        
        # Adaptive step size
        step_size = 0.01 * (1.0 - (iteration / max_iter))
        
        # Update h
        h_new = h - step_size * grad
        
        # Project to feasible region
        h = project_h(h_new, n_points)
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
