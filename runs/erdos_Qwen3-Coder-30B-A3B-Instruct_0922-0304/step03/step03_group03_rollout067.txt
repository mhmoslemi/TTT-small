import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Parameters for multiplicative updates
    epsilon = 0.1
    damping = 0.99
    max_iter = 1000
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        corr = correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Compute multiplicative factors for each position
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # Update each h[i] proportionally
        h_new = np.zeros_like(h)
        for i in range(n_points):
            contribution = 0.0
            # Consider influence from all valid lag patterns
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    # Weight based on exponential term
                    contribution += exp_neg_S[k + n_points] / (1 + exp_pos_S[k + n_points])
            h_new[i] = h[i] * contribution * damping
            
        # Feasibility repair
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    # Final checks
    assert np.all(0 <= best_h <= 1)
    assert np.abs(np.sum(best_h) - n_points / 2) < 1e-6
    
    return (best_h.tolist(), best_c5, n_points)
