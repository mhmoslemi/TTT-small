def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()

    def objective(h_var):

    # Using SDP relaxation approach

        # Use gradient descent with projection onto feasible set
        h_current = h.copy()
        current_c5 = evaluate_c5(h_current, dx)
        
        # Simple gradient estimation via finite differences
        eps = 1e-5
        gradient = np.zeros_like(h_current)

            h_plus = h_current.copy()

        # Perform gradient descent step

            current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
