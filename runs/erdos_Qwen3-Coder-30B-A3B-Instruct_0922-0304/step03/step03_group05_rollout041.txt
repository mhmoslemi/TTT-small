import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to correct integral
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters for pulse-based representation
    M = 5  # number of pulses
    max_iter = 1000
    
    # Initialize pulse parameters (heights, positions, widths)
    # Heights: [0,1], Positions: [0,2], Widths: [0.01, 0.5] 
    pulse_heights = np.ones(M) * 0.5
    pulse_positions = np.linspace(0.2, 1.8, M)
    pulse_widths = np.ones(M) * 0.2
    
    def compute_h_from_pulses():
        """Convert pulse parameters to h array"""
        h_new = np.zeros(n_points)
        for i in range(M):
            pos = pulse_positions[i]
            width = pulse_widths[i]
            height = pulse_heights[i]
            # Calculate indices for pulse support
            left_idx = max(0, int((pos - width/2) / dx))
            right_idx = min(n_points, int((pos + width/2) / dx) + 1)
            # Set range
            h_new[left_idx:right_idx] = height
        return h_new
    
    def compute_C5(h_vec):
        """Compute maximum overlap with given h"""
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(overlap) * dx
    
    def project_pulses():
        """Enforce constraints on pulses"""
        # Clamp heights to [0,1]
        pulse_heights[:] = np.clip(pulse_heights, 0, 1)
        # Keep positions in [0,2]
        pulse_positions[:] = np.clip(pulse_positions, 0, 2)
        # Clamp widths to reasonable range
        pulse_widths[:] = np.clip(pulse_widths, 0.01, 0.5)
        
        # Renormalize
        total_area = np.sum(pulse_heights * pulse_widths)
        if total_area > 0:
            scale_factor = (n_points / 2) / total_area
            pulse_heights[:] *= scale_factor
    
    def optimize_heights():
        """Optimize heights while keeping positions and widths fixed"""
        # For each k, compute overlaps explicitly
        h_step = compute_h_from_pulses()
        overlap_values = []
        
        # Compute overlap for all lags
        for k in range(-n_points+1, n_points):
            if k >= 0:
                lagged = np.pad(h_step[k:], (0, k), mode='constant')
            else:
                lagged = np.pad(h_step[:k], (-k, 0), mode='constant')
                
            prod = h_step * (1 - lagged)
            overlap_values.append(np.sum(prod) * dx)
        
        # Use simple gradient ascent on current best
        # Simply adjust to reduce overlap where possible
        h_trial = h_step.copy()
        old_c5 = compute_C5(h_step)
        
        # Try perturbing one height at a time
        for m in range(M):
            h_trial = compute_h_from_pulses()
            # Add small perturbation to height m
            orig_h = pulse_heights[m]
            avg_inc = 0.005
            
            if orig_h not in [0, 1]:  
                pulse_heights[m] = min(1, orig_h + avg_inc)
                h_test = compute_h_from_pulses()
                test_c5 = compute_C5(h_test)
                if test_c5 < old_c5:
                    old_c5 = test_c5
                else:
                    pulse_heights[m] = orig_h  # revert
                    
        return
        
    def optimize_positions():
        """Optimize positions while keeping heights and widths fixed"""
        h_step = compute_h_from_pulses()
        old_c5 = compute_C5(h_step)
        
        # Numerical gradient approx - small shifts
        step_size = 0.01
        for m in range(M):
            pos_orig = pulse_positions[m]
            # Try small shift left
            pulse_positions[m] = pos_orig - step_size
            h_trial = compute_h_from_pulses()
            c5_left = compute_C5(h_trial)
            # Try small shift right
            pulse_positions[m] = pos_orig + step_size
            h_trial = compute_h_from_pulses()
            c5_right = compute_C5(h_trial)
            # Restore original if no improvement
            pulse_positions[m] = pos_orig
            if c5_left < old_c5 or c5_right < old_c5:
                # Choose the better one (move towards smaller C5)
                if c5_left < c5_right:
                    if c5_left < old_c5:
                        pulse_positions[m] = pos_orig - step_size
                else:
                    if c5_right < old_c5:
                        pulse_positions[m] = pos_orig + step_size
            else:
                # No improvement: random exploration some steps
                rand_shift = np.random.uniform(-step_size, step_size)
                pulse_positions[m] = pos_orig + rand_shift
                pulse_positions[m] = np.clip(pulse_positions[m], 0, 2)
        
        return

    def optimize_widths():
        """Optimize widths while keeping positions and heights fixed"""
        h_step = compute_h_from_pulses()
        old_c5 = compute_C5(h_step)
        
        base_width = 0.1
        step_size = 0.01
        
        # For each pulse, try adjusting width
        for m in range(M):
            w_orig = pulse_widths[m]
            # Try small changes
            candidates = [w_orig - step_size, w_orig + step_size]
            candidates = [w for w in candidates if 0.01 <= w <= 0.5]
            
            best_c5 = old_c5
            best_width = w_orig
            
            for w_trial in candidates:
                pulse_widths[m] = w_trial
                h_trial = compute_h_from_pulses()
                c5 = compute_C5(h_trial)
                if c5 < best_c5:
                    best_c5 = c5
                    best_width = w_trial
                    
            pulse_widths[m] = best_width
        
        return

    # Main optimization loop
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convert pulses to h
        h = compute_h_from_pulses()
        current_c5 = compute_C5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Project pulses to feasible region
        project_pulses()
        
        # Alternate optimization
        optimize_heights()
        project_pulses()
        
        optimize_positions()
        project_pulses()
        
        optimize_widths()
        project_pulses()
    
    # Final normalization and evaluation
    project_pulses()
    final_h = compute_h_from_pulses()
    final_c5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
