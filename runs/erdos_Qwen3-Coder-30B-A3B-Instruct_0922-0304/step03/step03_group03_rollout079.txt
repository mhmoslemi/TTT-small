import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Target C5
    target_c5 = 0.3807
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Early exit if target achieved
        if current_c5 <= target_c5:
            break
            
        # Conjugate gradient approach with adaptive step size
        step_size = 0.01
        cg_iter = 50
        
        for _ in range(cg_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient approximations via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            h_current = h.copy()
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update h using conjugate gradient direction
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Recompute current_c5 after update
            corr = correlate(h, 1 - h, mode="full")
            current_c5 = np.max(corr) * dx
            
            # Adapt stepsize based on improvement
            if current_c5 < target_c5:
                break
                
            step_size *= 0.95  # decrease step size slowly

    # Final evaluation
    corr = correlate(h, 1 - h, mode="full")
    final_c5 = np.max(corr) * dx
    
    return (h.tolist(), final_c5, n_points)
