import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_population(n_points, population_size):
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
        population.append(h)
    
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def selection(population, fitnesses, tournament_size=3):
    selected = []
    pop_size = len(population)
    for _ in range(pop_size // 2):
        tournament_indices = np.random.choice(pop_size, tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.05):
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, noise_std)
    return h

def genetic_algorithm(n_points, population_size=100, generations=1000, 
                      budget_s=600, seed=42):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population
        new_population = selected.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(repair_h(child1, n_points))
            if len(new_population) < population_size:
                new_population.append(repair_h(child2, n_points))
        
        population = new_population[:population_size]
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    best_h, best_c5 = genetic_algorithm(n_points, budget_s=budget_s, seed=seed)
    return (best_h.tolist(), best_c5, n_points)
