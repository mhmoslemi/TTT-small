import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i * 2.0 / mid
        for i in range(mid, n_points):
            h[i] = 2.0 - i * 2.0 / mid
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start with basin hopping
    num_starts = 5
    initial_candidates = [h.copy()]
    
    # Generate perturbed initial solutions
    for _ in range(num_starts - 1):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_candidates.append(perturbed)
    
    # Optimization loop with basin hopping
    max_iter = 1000
    iter_count = 0
    
    for init_candidate in initial_candidates:
        h_current = init_candidate.copy()
        current_c5 = compute_c5(h_current)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Local optimization
        for iteration in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            iter_count += 1
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 1, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                h_current += np.random.normal(0, 0.01, n_points)
                
            # Ensure feasibility
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            current_c5 = compute_c5(h_current)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            if best_c5 <= 0.3807:
                break
                
    # Basin hopping with adaptive perturbations
    h_current = best_h.copy()
    temp_sigma = 0.1
    cooling_factor = 0.995
    update_interval = 20
    max_basin_iter = 1000
    
    for iteration in range(max_basin_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        noise = np.random.normal(0, temp_sigma, n_points)
        h_perturbed = h_current + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local optimization
        result = minimize(
            objective, 
            h_perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            h_perturbed = result.x
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        else:
            # If optimization fails, keep original
            h_perturbed = h_perturbed
            
        current_c5 = compute_c5(h_perturbed)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Cooling schedule
        temp_sigma *= cooling_factor
        
        if iteration % update_interval == 0:
            temp_sigma = max(temp_sigma, 0.001)

    # Adaptive resolution if time allows
    if n_points == 70 and time.time() - start_time < budget_s - 5:
        # Interpolate to higher resolution
        n_points_high = 140
        dx_high = 2.0 / n_points_high
        h_high = np.interp(np.linspace(0, 2, n_points_high), 
                          np.linspace(0, 2, n_points), best_h)
        
        # Scale to sum constraint
        h_high = h_high * (n_points_high / 2) / np.sum(h_high)
        
        best_c5 = float('inf')
        best_h = h_high.copy()
        
        # Local optimization at higher resolution
        bounds_high = [(0, 1) for _ in range(n_points_high)]
        constraints_high = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_high / 2}]
        
        result = minimize(
            lambda x: np.max(correlate(x, 1 - x, mode="full")) * dx_high,
            best_h, 
            method='SLSQP', 
            bounds=bounds_high, 
            constraints=constraints_high,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points_high / 2) / np.sum(best_h)
            
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            n_points = n_points_high

    return (best_h.tolist(), best_c5, n_points)
