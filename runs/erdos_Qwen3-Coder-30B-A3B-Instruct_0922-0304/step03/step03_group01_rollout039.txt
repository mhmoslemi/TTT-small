import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def mutate_individual(h, mutation_rate=0.1, mutation_strength=0.1):
    """Apply mutation to an individual"""
    h_new = h.copy()
    mask = np.random.random(len(h_new)) < mutation_rate
    h_new[mask] += np.random.normal(0, mutation_strength, np.sum(mask))
    return repair_h(h_new, len(h_new))

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two individuals"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def genetic_algorithm(seed, n_points, pop_size, budget_s, initial_h=None):
    """Genetic algorithm optimization"""
    np.random.seed(seed)
    start_time = time.time()
    
    dx = 2.0 / n_points
    
    # Initialize population
    if initial_h is not None:
        population = [repair_h(initial_h.copy(), n_points)]
        while len(population) < pop_size:
            h = initial_h + np.random.normal(0, 0.05, n_points)
            population.append(repair_h(h, n_points))
    else:
        population = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s * 0.8 and no_improvement_count < 50:
        # Evaluate fitness (negative for minimization)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitness[idx1] > fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep top 20%
        elite_count = pop_size // 5
        elite_indices = np.argsort(fitness)[-elite_count:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(selected, 2, replace=False)
            c1, c2 = crossover(p1, p2)
            
            c1 = mutate_individual(c1)
            c2 = mutate_individual(c2)
            
            new_population.extend([c1, c2])
        
        # Cap population size
        population = new_population[:pop_size]
        
        generation += 1
        if generation > 1 and not (generation % 10):
            no_improvement_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70

    # Local refinement if needed
    if time.time() - time.time() < budget_s * 0.2:
        # Gradient-based refinement
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return evaluate_c5(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Start with the best from GA
        h_refine = best_h.copy()
        current_c5 = objective(h_refine)
        
        # Simple gradient descent with projection
        for _ in range(200):
            if time.time() - time.time() > budget_s - 1:
                break
            corr = correlate(h_refine, 1-h_refine, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h_refine)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            h_refine -= 0.01 * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            
            new_c5 = objective(h_refine)
            if new_c5 < current_c5:
                current_c5 = new_c5
                best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
