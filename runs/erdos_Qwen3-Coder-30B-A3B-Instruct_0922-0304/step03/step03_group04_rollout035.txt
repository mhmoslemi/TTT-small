import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 30  # number of intervals
    w = 2.0 / M  # width of each interval
    n_points = M
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Average initial values into intervals
        h_init = h_init.reshape(M, -1).mean(axis=1)
    else:
        # Fallback initialization
        h_init = np.ones(M) * 0.5
        
    # Normalize to satisfy integral constraint
    h_init = h_init * (M / 2) / np.sum(h_init)
    
    # Generate shift indices and overlap matrix
    # shifts go from -M+1 to M-1 (inclusive)
    shifts = list(range(-M+1, M))
    A = np.zeros((len(shifts), M))
    
    # Compute overlap lengths for each interval and shift
    for i in range(M):
        for k_idx, k in enumerate(shifts):
            overlap = 0.0
            # Determine actual overlap for interval i with interval i+k
            if 0 <= i+k < M:
                # Compute overlap of interval i with interval i+k
                # Interval i spans [i*w, (i+1)*w), interval i+k spans [(i+k)*w, (i+k+1)*w)
                left = max(i*w, (i+k)*w)
                right = min((i+1)*w, (i+k+1)*w)
                if right > left:
                    overlap = right - left
            A[k_idx, i] = overlap

    # Optimization using CVXPY
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Create CVXPY variables
    h = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == M / 2,  # integral constraint
        h >= 0,
        h <= 1,
        C >= 0
    ]
    
    # Add overlap constraints
    for k_idx, k in enumerate(shifts):
        # This constraint: sum(h[i] * (1 - h[i+k]) * A[k][i]) <= C
        # But we must handle boundary conditions properly
        
        # For shift k, we compute: sum over i of (h[i] * (1 - h[i+k]) * overlap[i])
        # We handle missing terms by setting them to 0
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i in range(M), i+k in range(M)
            # => i in range(M-k) for k >= 0
            expr = 0
            for i in range(M - k):
                expr += h[i] * (1 - h[i + k]) * A[k_idx, i]
            constraints.append(expr <= C)
        else:
            # i-k in range(M), so i in range(-k) to M
            # => i in range(M+k) for k < 0, and i >= -k
            expr = 0
            for i in range(-k, M):
                expr += h[i] * (1 - h[i + k]) * A[k_idx, i]
            constraints.append(expr <= C)

    # Objective function
    objective = cp.Minimize(C)
    
    # Solve using ECOS or MOSEK
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            C_opt = C.value
            if C_opt < best_c5:
                best_c5 = C_opt
                best_h = h_opt.copy()
        else:
            # If optimization failed, fallback to simple gradient descent
            def evaluate_c5(h_array):
                # Reshape h from interval representation to full sequence
                h_full = np.zeros(n_points)
                for i in range(M):
                    h_full[i::M] = h_array[i]
                # Compute overlap using correlation
                overlap = correlate(h_full, 1 - h_full, mode="full")
                return np.max(overlap) * w

            # Simple gradient-based approach with manual optimization
            h_current = h_init.copy()
            step_size = 0.01
            max_iter = 1000
            
            for i in range(max_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Approximate gradient using finite differences
                grad = np.zeros(M)
                eps = 1e-4
                current_c5 = evaluate_c5(h_current)

                for j in range(M):
                    h_plus = h_current.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (M / 2) / np.sum(h_plus)  # renormalize
                    c5_plus = evaluate_c5(h_plus)
                    grad[j] = (c5_plus - current_c5) / eps

                # Gradient update
                h_current -= step_size * grad
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (M / 2) / np.sum(h_current)  # renormalize

                current_c5 = evaluate_c5(h_current)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()

                if best_c5 <= 0.3807:
                    break
                    
    except:
        # Fallback to gradient-based approach if CVXPY fails
        def evaluate_c5(h_array):
            h_full = np.zeros(n_points)
            for i in range(M):
                h_full[i::M] = h_array[i]
            overlap = correlate(h_full, 1 - h_full, mode="full")
            return np.max(overlap) * w

        h_current = h_init.copy()
        step_size = 0.01
        max_iter = 1000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Approximate gradient using finite differences
            grad = np.zeros(M)
            eps = 1e-4
            current_c5 = evaluate_c5(h_current)

            for j in range(M):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (M / 2) / np.sum(h_plus)
                c5_plus = evaluate_c5(h_plus)
                grad[j] = (c5_plus - current_c5) / eps

            # Gradient update
            h_current -= step_size * grad
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (M / 2) / np.sum(h_current)

            current_c5 = evaluate_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()

            if best_c5 <= 0.3807:
                break
    
    # Final validation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
