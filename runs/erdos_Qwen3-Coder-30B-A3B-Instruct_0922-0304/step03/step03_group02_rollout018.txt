import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import*

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 100 points for higher resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with careful handling
    def objective(h):
        return evaluate_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Main optimization loop
    max_iter = 300
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try multiple local optimizations with different starting points
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb current best solution
            perturbed_h = best_h + np.random.normal(0, 0.02, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            
            # Local optimization
            try:
                result = minimize(
                    objective,
                    perturbed_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 30, 'ftol': 1e-9}
                )
                
                if result.success:
                    current_h = result.x
                    current_h = np.clip(current_h, 0, 1)
                    current_h = current_h * (n_points / 2) / np.sum(current_h)
                    current_c5 = evaluate_c5(current_h)
                    
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
                        
                if best_c5 <= 0.3807:
                    break
                    
            except:
                pass
    
    # Final refinement with a more precise method
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = evaluate_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass

    return (best_h.tolist(), float(best_c5), n_points)
