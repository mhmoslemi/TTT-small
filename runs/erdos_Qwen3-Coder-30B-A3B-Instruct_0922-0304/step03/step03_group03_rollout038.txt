import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: create simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use gradient-based optimization
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Local gradient descent step
        grad = np.zeros_like(h)
        eps = 1e-4
        h_base = h.copy()
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update step
        step_size = 0.005
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final refinement with a more robust optimization
    # Create optimization variables
    h_vars = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        t >= 0
    ]
    
    # Loop through all lags to bind the overlaps
    for k in range(-n_points + 1, n_points):
        # Compute sum of overlap terms for shift k
        overlap_sum = 0
        if k >= 0:
            for i in range(n_points - k):
                overlap_sum += h_vars[i] * (1 - h_vars[i + k])
        else:
            k = -k
            for i in range(n_points - k):
                overlap_sum += h_vars[i + k] * (1 - h_vars[i])
        
        constraints.append(overlap_sum <= t)
        
    # Objective: minimize maximum overlap
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h = np.clip(h_vars.value, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_c5(h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
