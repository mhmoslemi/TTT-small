import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop with PSO + Local refinement
    n_particles = 20
    max_iter = 300
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    initial_learning_rate = 0.005
    local_search_steps = 5
    
    # Initialize particle swarm
    if 'initial_h_values' in globals():
        h_swarm = np.tile(h, (n_particles, 1))
        h_swarm += np.random.normal(0, 0.03, (n_particles, n_points))
    else:
        h_swarm = np.random.rand(n_particles, n_points)
    
    # Normalize each particle to meet sum constraint
    for i in range(n_particles):
        h_swarm[i] = np.clip(h_swarm[i], 0, 1)
        h_swarm[i] = h_swarm[i] * (n_points / 2) / np.sum(h_swarm[i])
    
    velocities = np.zeros_like(h_swarm)
    pbest_positions = h_swarm.copy()
    pbest_fitness = np.array([evaluate_c5(h, dx) for h in h_swarm])
    gbest_position = h_swarm[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # PSO velocity and position updates
        r1, r2 = np.random.rand(2, n_particles)
        velocities = omega * velocities + c1 * r1 * (pbest_positions - h_swarm) + c2 * r2 * (gbest_position - h_swarm)
        h_swarm = h_swarm + velocities
        
        # Feasibility repair
        h_swarm = np.clip(h_swarm, 0, 1)
        for i in range(n_particles):
            h_swarm[i] = h_swarm[i] * (n_points / 2) / np.sum(h_swarm[i])
        
        # Evaluate fitness
        current_fitness = np.array([evaluate_c5(h, dx) for h in h_swarm])
        
        # Update personal bests
        update_mask = current_fitness < pbest_fitness
        pbest_positions[update_mask] = h_swarm[update_mask]
        pbest_fitness[update_mask] = current_fitness[update_mask]
        
        # Update global best
        if np.min(current_fitness) < gbest_fitness:
            gbest_fitness = np.min(current_fitness)
            gbest_position = h_swarm[np.argmin(current_fitness)]
        
        # Local search on top particles
        top_indices = np.argsort(current_fitness)[:2]
        for idx in top_indices:
            for _ in range(local_search_steps):
                if time.time() - start_time > budget_s - 1:
                    break
                h = h_swarm[idx]
                corr = correlate(h, 1 - h, mode="full")
                max_lag_idx = np.argmax(corr)
                
                # Compute finite difference gradient
                grad = np.zeros(n_points)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate_c5(h_plus, dx)
                    grad[i] = (c5_plus - current_fitness[idx]) / eps
                
                # Update with momentum
                h_swarm[idx] -= initial_learning_rate * grad
                h_swarm[idx] = np.clip(h_swarm[idx], 0, 1)
                h_swarm[idx] = h_swarm[idx] * (n_points / 2) / np.sum(h_swarm[idx])
        
        # Track best solution
        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest_position.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
