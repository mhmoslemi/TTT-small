import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop using quadratic programming
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations to find worst case shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)  # Convert index to shift
        
        # Create quadratic program for this shift k
        h_var = cp.Variable(n_points)
        
        # Objective: minimize h[k] * (1 - h[0]) + h[k+1] * (1 - h[1]) + ...
        # For shift k, i from 0 to n-k-1: h[i] * (1 - h[i+k])
        objective = 0
        for i in range(n_points - abs(k)):
            if k >= 0:
                objective += h_var[i] * (1 - h_var[i + k])
            else:
                objective += h_var[i - k] * (1 - h_var[i])
        
        objective = cp.Minimize(objective * dx)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Formulate and solve QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h = h_new
                else:
                    # fallback if optimization fails
                    h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # fallback if optimization fails
                h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback to simple update
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
