import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: triangular wave
        h = 1 - 2 * np.abs(np.arange(n_points) - (n_points - 1) / 2) / (n_points - 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # First particle from initial solution
    swarm.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Additional random particles with triangular initialization
    for _ in range(n_particles - 1):
        # Start with triangular wave
        h_trial = 1 - 2 * np.abs(np.arange(n_points) - (n_points - 1) / 2) / (n_points - 1)
        # Add some random noise
        h_trial += np.random.normal(0, 0.05, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        swarm.append(h_trial.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w_min = 0.4
    w_max = 0.9
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    best_c5 = pbest_scores[best_idx]
    
    # Main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight linearly
        w = w_max - (w_max - w_min) * iter_count / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            
            v_new = (w * velocities[i] + 
                     c_p * r_p * (pbest[i] - swarm[i]) + 
                     c_g * r_g * (gbest - swarm[i]))
            
            # Clip velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            h_new = swarm[i] + v_new
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to satisfy constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_c5(h_new, dx)
            
            # Update personal best
            if c5_new < pbest_scores[i]:
                pbest[i] = h_new.copy()
                pbest_scores[i] = c5_new
                
                # Update global best
                if c5_new < best_c5:
                    best_c5 = c5_new
                    gbest = h_new.copy()
            
            # Update swarm and velocity
            swarm[i] = h_new
            velocities[i] = v_new
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation  
    final_c5 = evaluate_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
