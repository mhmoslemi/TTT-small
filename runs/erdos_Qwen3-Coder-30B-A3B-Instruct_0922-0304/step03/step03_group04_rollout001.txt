import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach with increasing resolution levels
    resolution_levels = [10, 20, 40, 70, 100, 140, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = 10
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimizer settings
    de_popsize = 20
    slsqp_maxiter = 200
    
    # Progressively increase resolution
    for i, n_points in enumerate(resolution_levels):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        # Initialize with up-sampled previous solution or random
        if i == 0:
            h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points)
        else:
            # Up-sample from previous solution
            old_h = h.copy()
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
            # Add some noise
            h += np.random.normal(0, 0.02, n_points)
            
        # Ensure h sums correctly  
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Differential Evolution for global search
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vec):
            return evaluate_c5(h_vec, dx)
        
        try:
            result = differential_evolution(
                objective,
                bounds,
                constraints=cons,
                popsize=de_popsize,
                maxiter=50,
                seed=seed,
                updating='deferred'
            )
            if result.success:
                h = result.x
        except:
            pass

        # Local optimization refinement
        maxiter = min(200, int(slsqp_maxiter * (n_points / 70.0)))
        try:
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-8}
            )
            if result.success:
                h = result.x
        except:
            pass
            
        # Ensure feasibility and evaluate
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation and return
    if best_h is None:
        best_h = h
        
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
