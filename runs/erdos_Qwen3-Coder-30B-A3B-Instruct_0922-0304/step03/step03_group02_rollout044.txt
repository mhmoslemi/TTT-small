import numpy as np
from scipy.linalg import eigh
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a smooth initial guess
        x = np.linspace(0, 2, n_points)
        h = np.exp(-20 * (x - 0.5)**2) + 0.2 * np.sin(5 * np.pi * x) + 0.3
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # SDP-based refinement loop
    for iteration in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate and solve a relaxed SDP problem
        # We'll iterate with Fourier domain correlations for efficiency
        
        # Compute Fourier coefficients for current h
        H = np.fft.fft(h)
        # Correlation in frequency domain
        cross_corr = np.fft.ifft(H * np.conj(H))
        # Real part of correlation (shifted)
        corr_real = np.real(cross_corr)
        # Adjust for proper correlation
        corr_shifted = np.concatenate([corr_real[n_points//2:], corr_real[:n_points//2]])
        # Evaluate max overlap using the shifted correlation
        # But that approach may not be accurate enough. Let's switch approach.
        
        # Revert to original gradient based method with better controls
        # but Wegstein acceleration for faster convergence
        # Simple gradient descent idea with constraints 
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Custom local search with momentum and adaptive steps
        # Compute gradient estimates using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient ascent in negative direction
        step_size = 0.1 * (0.999 ** iteration)
        h_new = h - step_size * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
