import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n / 2) / np.sum(h_plus)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

def backtracking_line_search(h, grad, dx, alpha=0.01, beta=0.5, max_iter=20):
    c5_current = evaluate_c5(h, dx)
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2) / np.sum(h_new)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < c5_current:
            return alpha, h_new
        alpha *= beta
    return alpha, h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create two pulses as fallback
        h = np.zeros(n_points)
        center1, center2 = 0.5, 1.5
        width = 0.3
        x = np.linspace(0, 2, n_points)
        pulse1 = np.exp(-0.5 * ((x - center1) / width)**2)
        pulse2 = np.exp(-0.5 * ((x - center2) / width)**2)
        h = pulse1 + pulse2
        h = np.clip(h, 0, 1)
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Gradient computation
        grad = compute_gradient(h, dx)
        
        # Line search
        alpha, h_new = backtracking_line_search(h, grad, dx)
        
        # Update
        h = h_new
        
        # Projection
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbation
        if no_improvement_count > max_no_improvement:
            pert_magnitude = 1e-3 * (1.0 / (iter_count**0.5))
            noise = np.random.normal(0, pert_magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
