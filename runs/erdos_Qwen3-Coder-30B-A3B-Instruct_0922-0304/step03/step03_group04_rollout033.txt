import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try optimization with different approaches
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different optimization strategies
        if iteration % 3 == 0:
            # Standard gradient-based optimization
            from scipy.optimize import minimize
            bounds = [(0, 1) for _ in range(n_points)]
            target_sum = n_points / 2
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                overlap = correlate(h, 1 - h, mode="full")
                return np.max(overlap) * dx
            
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Gradient descent fallback
                step = 0.001
                h_temp = h.copy()
                for _ in range(20):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_overlap = np.max(corr) * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-5
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp

        else:
            # Alternating optimization between different variables
            # Simple heuristic: random perturbation with projection
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.02)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        current_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final fine-tune using direct optimization
    if time.time() - start_time < budget_s - 2:
        from scipy.optimize import differential_evolution
        
        def objective(h):
            overlap = correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx
            
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            maxiter=50,
            popsize=15,
            seed=seed
        )
        
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = np.max(correlate(h_final, 1 - h_final, mode="full")) * dx
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final

    return (best_h.tolist(), best_c5, n_points)
