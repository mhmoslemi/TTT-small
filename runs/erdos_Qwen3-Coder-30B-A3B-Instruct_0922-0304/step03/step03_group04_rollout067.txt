import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Helper function to project h to feasible region
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Initialize using frequency domain approach
    if time.time() - start_time > budget_s - 10:

    n_candidates = 50

        if time.time() - start_time > budget_s - 10:
            break
        # Use small random perturbations in frequency domain
        h_candidate = h.copy()
        freq_h = fft(h_candidate)
        # Perturb only dominant frequencies
        for idx in dominant_indices:
            if idx < len(freq_h) // 2:
                freq_h[idx] += np.random.normal(0, 0.01) + 1j * np.random.normal(0, 0.01)
                if idx > 0:
                    freq_h[-idx] = np.conj(freq_h[idx])

        h_new = np.real(ifft(freq_h))

        candidates.append(h_new)

            break
        candidate_c5 = evaluate_c5(candidate, dx)

            best_c5 = candidate_c5
            best_h = candidate.copy()

        if best_c5 <= 0.3807:

                            simplex[-1] = x_c

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = simplex[0].copy()
                
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
