import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    selected = []
    for _ in range(len(population) // 2):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_std=0.05):
    mask = np.random.random(len(h)) < mutation_rate
    h = h.copy()
    h[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = repair_h(best_h, n_points)
        population.append(best_h)
    else:
        best_h = np.ones(n_points)
        best_h = repair_h(best_h, n_points)
        population.append(best_h)
    
    # Add random individuals
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_c5(ind, dx) for ind in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create next generation
        next_population = [best_individual.copy()]
        while len(next_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            next_population.extend([child1, child2])
        
        population = next_population[:population_size]
    
    # Final evaluation
    final_c5 = evaluate_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
