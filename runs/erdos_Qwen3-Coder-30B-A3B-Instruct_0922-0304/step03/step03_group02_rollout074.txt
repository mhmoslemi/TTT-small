import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill rest of population with random individuals
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    def mutate(individual, mutation_rate=0.2, noise_std=0.05):
        mutated = individual.copy()
        mask = np.random.random(n_points) < mutation_rate
        mutated[mask] += np.random.normal(0, noise_std, np.sum(mask))
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2, crossover_rate=0.8):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def tournament_selection(pop, k=3):
        selected = []
        for _ in range(len(pop)):
            tournament = np.random.choice(len(pop), k)
            best = min(tournament, key=lambda i: evaluate_c5(pop[i], dx))
            selected.append(pop[best].copy())
        return selected
    
    best_c5 = float('inf')
    best_h = None
    
    max_generations = 1000
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(ind, dx) for ind in population]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population)
        
        # Create new population through crossover and mutation
        new_population = selected[:population_size//2].copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(population_size, 2, replace=False)
            child1, child2 = crossover(selected[parent1], selected[parent2])
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
    
    # Final evaluation of best individual
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
