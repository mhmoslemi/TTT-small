import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_individual(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def genetic_algorithm(seed, budget_s, n_points=100):
    np.random.seed(seed)
    start_time = time.time()
    
    pop_size = 100
    n_generations = 1000
    crossover_rate = 0.7
    mutation_rate = 0.2
    tournament_size = 3
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill remaining population with random individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair_individual(h, n_points)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    dx = 2.0 / n_points
    
    for generation in range(n_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection with tournament
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i + 1] if i + 1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
            
            child1 = repair_individual(child1, n_points)
            child2 = repair_individual(child2, n_points)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5_bound = genetic_algorithm(seed, budget_s)
    n_points = len(h)
    return (h.tolist(), c5_bound, n_points)
