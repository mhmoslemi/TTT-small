import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is within bounds and sum equals n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def mutate_individual(h, mutation_rate=0.1, mutation_strength=0.1):
    """Apply mutation to an individual"""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h_mut[mask] += np.random.normal(0, mutation_strength, np.sum(mask))
    return repair_h(h_mut, len(h))

def crossover(parent1, parent2, crossover_rate=0.7):
    """Basic crossover between two individuals"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def ga_optimize(h0, dx, n_points, pop_size=50, generations=1000, mutation_rate=0.1, elite_size=10):
    """Genetic algorithm optimization"""
    population = [repair_h(h0.copy(), n_points) for _ in range(pop_size)]
    
    best_solution = h0.copy()
    best_c5 = evaluate_c5(best_solution, dx)
    
    for gen in range(generations):
        # Evaluate fitness (minimize negative overlap)
        fitness_scores = [-evaluate_c5(ind, dx) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        
        # Track best solution
        current_best = population[0]
        current_c5 = -fitness_scores[sorted_indices[0]]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = current_best.copy()
        
        # Elitism
        elites = population[:elite_size]
        
        # Create new population
        new_population = elites.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parents = np.random.choice(population, size=2, replace=False)
            child1, child2 = crossover(parents[0], parents[1])
            child1 = mutate_individual(child1, mutation_rate)
            child2 = mutate_individual(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        
        if gen % 50 == 0 and best_c5 <= 0.3807:
            break
    
    return best_solution, best_c5

    # Ensure correct sum constraint
    h = repair_h(h, n_points)
    
    # GA phase - 80% of time budget
    remaining_time = budget_s * 0.8
    ga_start = time.time()
    
    ga_h, ga_c5 = ga_optimize(h, dx, n_points, pop_size=50, generations=int(remaining_time / 20))
    
    # Local refinement from GA solution - 20% of time
    if time.time() - start_time < budget_s * 0.9:
        # Apply gradient-based local search to the best GA solution
        h_current = ga_h.copy()
        
        # Refinement using gradient descent approach
        step_size = 0.01
        max_iter = 1000
        for _ in range(max_iter):
            if time.time() - start_time >= budget_s - 1:
                break
                
            # Compute gradient
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Approximate gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(h_current)
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update with gradient descent step
            h_current -= step_size * grad
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)

        if refined_c5 < ga_c5:
            ga_c5 = refined_c5
            ga_h = h_current.copy()

    return (ga_h.tolist(), ga_c5, n_points)
