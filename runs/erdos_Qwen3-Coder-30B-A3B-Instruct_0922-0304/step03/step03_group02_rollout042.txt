import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective for scipy.optimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Define constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Main optimization loop with multiple strategies
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different approaches
        if iteration % 5 == 0:
            # Random perturbation with small amplitude
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            h = h_perturbed
        
        # Optimize with SLSQP
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-9}
        )
        
        if result.success:
            h = result.x
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate current solution
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        else:
            # Fallback to previous best if optimization fails
            h = best_h.copy()
    
    # Final evaluation with better optimization
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h, dx)
    
    # Attempt local refinement at the end
    if time.time() - start_time < budget_s - 1:
        # Second phase with more aggressive optimization
        def objective_local(h):
            return evaluate_c5(h, dx)
            
        result_fine = minimize(
            objective_local,
            final_h,
            method='Nelder-Mead',
            options={'maxiter': 100, 'adaptive': True}
        )
        
        if result_fine.success:
            final_h = result_fine.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
