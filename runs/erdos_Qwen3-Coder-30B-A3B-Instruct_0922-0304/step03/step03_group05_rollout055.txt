import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 2000
    n_initials = 5
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def objective(h):
        return compute_c5(h)

    # Create multiple initial solutions
    initial_solutions = [project_h(h + np.random.normal(0, 0.05, n_points)) for _ in range(n_initials)]
    initial_solutions.append(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try each initial solution
        for sol in initial_solutions:
            if time.time() - start_time > budget_s - 1:
                break
            h = sol.copy()
            
            # Local optimization
            result = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h = project_h(result.x)
            else:
                h = project_h(h + np.random.normal(0, 0.01, n_points))
                
            # Evaluate and update best
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            iteration += 1
            
        # Basin hopping with adaptive step size
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the best solution with decreasing step size
        step_sigma = 0.05 * np.exp(-iteration / 1000)
        h_perturbed = project_h(best_h + np.random.normal(0, step_sigma, n_points))
        
        # Local optimization on perturbed
        result = minimize(
            objective, 
            h_perturbed, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            h = project_h(result.x)
        else:
            h = project_h(h_perturbed)
            
        # Update best
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final refinement if time allows
    if time.time() - start_time < budget_s - 2:
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if result.success:
            best_h = project_h(result.x)
            best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n_points)
