import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Resolution levels to try
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Precompute dx for each resolution
    dx_vals = [2.0 / n for n in resolutions]
    
    # Global search parameters
    de_popsize = 20
    slsqp_maxiter = 200
    
    h_prev = None
    
    for i, (n_points, dx) in enumerate(zip(resolutions, dx_vals)):
        if time.time() - start_time > budget_s - 1:
            break

            # Start with uniform distribution or initial_h_values if available
            if 'initial_h_values' in globals():

                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:

        else:
            # Upsample previous solution
            h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(h_prev)), h_prev)
            # Add noise for exploration
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        h_prev = h.copy()
        
        # Define bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2

        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Try Differential Evolution first for global search
        try:
            result_de = differential_evolution(
                objective, 
                bounds, 
                strategy='best1bin', 
                maxiter=100, 
                popsize=de_popsize, 
                disp=False
            )
            if result_de.success:
                h = result_de.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
        
        # Then SLSQP for refinement
        try:
            result_slsqp = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': slsqp_maxiter, 'ftol': 1e-8}
            )
            if result_slsqp.success:
                h = result_slsqp.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
        
        # Final evaluation
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n = n_points
            
            # Early exit if we've achieved the target
            if best_c5 <= 0.3807:
                break
    
    # Cleanup solution if needed
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (best_n / 2) / np.sum(best_h)
        return (best_h.tolist(), best_c5, best_n)
    
    # Fallback to the last try if nothing was found
    if 'h_prev' in locals():
        h = h_prev
        h = np.clip(h, 0, 1)
        h = h * (best_n / 2) / np.sum(h)
        final_c5 = evaluate_c5(h, dx)
        return (h.tolist(), final_c5, best_n)

    final_c5 = evaluate_c5(h, dx)
