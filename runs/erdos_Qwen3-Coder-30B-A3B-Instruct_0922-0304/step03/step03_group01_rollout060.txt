import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7

            h = np.random.random(n_points)
            h[:n_points//2] = np.clip(h[:n_points//2], 0, 1)
            h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)  # Normalize sum
        return h
    
    # Create initial population
    population = [create_individual() for _ in range(pop_size)]
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    
    # Evolution loop
    gen_no_improve = 0
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (negative because we minimize)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Selection (tournament)
        def tournament_selection(k=3):
            selected = []
            for _ in range(pop_size):
                competitors = np.random.choice(pop_size, k)
                best_comp = np.argmax([fitness[i] for i in competitors])
                selected.append(population[competitors[best_comp]].copy())
            return selected
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_count = int(elitism_rate * pop_size)
        elite_indices = np.argsort(fitness)[-elite_count:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        selected = tournament_selection()
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
                
                # Apply mutations
                for child in [child1, child2]:
                    if np.random.rand() < mutation_rate:
                        idx = np.random.randint(n_points)
                        child[idx] = np.clip(child[idx] + np.random.normal(0, 0.05), 0, 1)
                    
                    # Repair
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
            else:
                # Clone individual
                child = selected[np.random.randint(pop_size)].copy()
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(n_points)
                    child[idx] = np.clip(child[idx] + np.random.normal(0, 0.05), 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Truncate to exact size
        population = new_population[:pop_size]

                gen_no_improve += 1
            else:
                gen_no_improve = 0
        previous_best = best_c5
        
        if gen_no_improve >= 50:
            break
    
    # Local refinement with gradient method
    if best_h is not None:
        h_refined = best_h.copy()
        best_c5_refined = best_c5
        
        # Simple gradient descent with projection
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient approximation
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update
            step_size = 0.001
            h_refined -= step_size * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            # Evaluate
            c5_new = evaluate_c5(h_refined, dx)
            if c5_new < best_c5_refined:
                best_c5_refined = c5_new
                best_h = h_refined.copy()
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
