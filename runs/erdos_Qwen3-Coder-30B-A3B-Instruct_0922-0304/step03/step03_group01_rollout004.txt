import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter_pso = 200
    
    # Initialize PSO swarm
    positions = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    gbest_position = None
    gbest_value = float('inf')
    
    # Enable PSO fallback for cases where it might stagnate
    for i in range(swarm_size):
        # Create initial particle position
        pos = h.copy()
        # Add slight noise for diversity
        if i > 0:
            noise = np.random.normal(0, 0.02, n_points)
            pos += noise
            pos = np.clip(pos, 0, 1)
            # Ensure sum constraint
            pos = pos * (n_points / 2) / np.sum(pos)
        positions.append(pos.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(pos.copy())
        pbest_values.append(evaluate_c5(pos, dx))
        if pbest_values[-1] < gbest_value:
            gbest_value = pbest_values[-1]
            gbest_position = pos.copy()
    
    # PSO main loop
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Update each particle
        for i in range(swarm_size):
            pos = positions[i]
            vel = velocities[i]
            pbest_pos = pbest_positions[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (inertia * vel + 
                   cognitive * r1 * (pbest_pos - pos) + 
                   social * r2 * (gbest_position - pos))
            
            # Update position
            pos = pos + vel
            pos = np.clip(pos, 0, 1)
            # Apply sum constraint
            pos = pos * (n_points / 2) / np.sum(pos)
            
            # Update position and velocity
            positions[i] = pos
            velocities[i] = vel
            
            # Evaluate fitness
            fitness = evaluate_c5(pos, dx)
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest_positions[i] = pos.copy()
                
                # Update global best
                if fitness < gbest_value:
                    gbest_value = fitness
                    gbest_position = pos.copy()
        
        # Early termination if target achieved
        if gbest_value <= 0.3807:
            break
    
    # Refinement with SLSQP
    if gbest_position is not None:
        h = gbest_position.copy()
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Local optimization with SLSQP
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 500, 'ftol': 1e-9})
            if result.success:
                h = result.x
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                # Revert to PSO best if SLSQP fails
                best_c5 = gbest_value
                best_h = gbest_position.copy()
        except Exception as e:
            # Fallback to PSO best in case of error
            best_c5 = gbest_value
            best_h = gbest_position.copy()

    # Final check if no good result was found
    if best_c5 == float('inf'):
        best_c5 = evaluate_c5(h, dx)
        best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
