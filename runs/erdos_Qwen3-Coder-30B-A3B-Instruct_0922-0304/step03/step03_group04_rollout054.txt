import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for multi-resolution optimization
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if res_idx == 0:
            if 'initial_h_values' in globals() and len(initial_h_values) == 70:
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
            else:
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
        else:
            # Up-sample the previous solution
            prev_h = best_h
            h = np.zeros(n_points)
            for i in range(n_points):
                idx = min(i * len(prev_h) // n_points, len(prev_h) - 1)
                h[i] = prev_h[idx]
            
            # Add some noise
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Global optimization with Differential Evolution
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Use DE for exploration
        de_result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            seed=seed,
            maxiter=50,
            popsize=20,
            strategy='best1bin',
            disp=False
        )
        
        if de_result.success:
            h = de_result.x
            # Enforce constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
        
        # Local refinement with SLSQP
        for iter_local in range(3):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                # Enforce constraints
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                    if best_c5 <= 0.3807:
                        break
            else:
                # Use gradient descent fallback
                step = 0.005
                iterations = 100
                h_temp = h.copy()
                
                for _ in range(iterations):
                    if time.time() - start_time > budget_s - 1:
                        break
                        
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    
                h = h_temp
        
        # If improvement stall, restart with perturbed version
        if res_idx > 0:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            current_c5 = evaluate_c5(h_perturbed, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Final check
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, 2.0 / n_points)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback
        n_points = 70
        dx = 2.0 / n_points
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = evaluate_c5(h, dx)
        return (h.tolist(), final_c5, n_points)
