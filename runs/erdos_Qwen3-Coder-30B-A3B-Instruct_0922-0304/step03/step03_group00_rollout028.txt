import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / (n_points/4))
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute indices
    indices = np.arange(n_points)
    
    # Main optimization loop
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find the worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Formulate QP for this shift
        # Objective: min (h^T Q h + c^T h) where Q is derived from shift k
        # But we will use cvxpy for more robust solving
        
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # We want to minimize the overlap at shift k = max_shift
        # This corresponds to sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) * dx
        obj_coeffs = np.zeros(n_points)
        k = max_shift
        relevant_indices = indices[:-abs(k)] if k >= 0 else indices[abs(k):]
        if k >= 0:
            obj_coeffs[relevant_indices] = -h[relevant_indices + k]
        else:
            obj_coeffs[relevant_indices] = -h[relevant_indices - k]
        
        objective = cp.Minimize(obj_coeffs.T @ h_var)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Add some noise to prevent cycling
                h = np.clip(h_new + np.random.normal(0, 0.001, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to simple gradient descent for this specific shift
                step = 0.01
                h_temp = h.copy()
                max_idx = np.argmax(corr)
                shift = max_idx - (n_points - 1)
                gradient = np.zeros(n_points)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    gradient[i] = (c5_plus - current_c5) / eps
                
                h_temp -= step * gradient
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
                
        except Exception as e:
            # Fallback on error
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
