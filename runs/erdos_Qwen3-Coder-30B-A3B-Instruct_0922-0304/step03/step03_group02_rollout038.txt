import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_algorithm(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.7
    mutation_rate = 0.2
    tournament_size = 3
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    # Add random initial candidates
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    # Main GA loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(c5)
        
        # Find best in current generation
        min_fitness = min(fitness_scores)
        best_idx = fitness_scores.index(min_fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[best_idx].copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        # Selection (tournament)
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate and i+1 < population_size:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([new_population[i][:crossover_point], 
                                       new_population[i+1][crossover_point:]])
                child2 = np.concatenate([new_population[i+1][:crossover_point], 
                                       new_population[i][crossover_point:]])
                new_population[i] = child1
                new_population[i+1] = child2
            
            # Mutation
            for j in range(population_size):
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    new_population[j] += noise
                    new_population[j] = repair_h(new_population[j], n_points)
        
        population = new_population
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h, c5 = genetic_algorithm(seed, budget_s)
    return (h.tolist(), c5, len(h))
