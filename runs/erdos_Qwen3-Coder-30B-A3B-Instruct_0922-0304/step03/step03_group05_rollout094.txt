import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: step function with random perturbations
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Use a variant approach with specialized optimization
    # Introduce pulse-based representation to escape local minima
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def perturb_h(h, amplitude=0.05):
        """Add small random perturbations to h"""
        h_pert = h + np.random.normal(0, amplitude, len(h))
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        return h_pert

    def optimize_h_via_local_search(h, max_iter=1000):
        """Optimize using alternating projections and local search"""
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break

            # Randomly perturb and keep better solutions
            new_h = perturb_h(h, 0.02)
            new_c5 = compute_c5(new_h)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                h = new_h.copy()
                
            # Occasionally do global gradient based perturbations
            if i % 30 == 0:
                # Slight modification to current best_h
                h = perturb_h(best_h, 0.01)
            else:
                h = perturb_h(h, 0.01)
                
            # Early stopping if target achieved
            if best_c5 <= 0.3807:
                break
                
        return best_h, best_c5
    
    # Initial optimization
    best_h, best_c5 = optimize_h_via_local_search(h)
    
    # Attempt a second stage with refinement centered around best solution
    if time.time() - start_time < budget_s - 2:
        # Fine-tune around the best so far
        fine_h = perturb_h(best_h, 0.005)
        fine_h = np.clip(fine_h, 0, 1)
        fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
        fine_c5 = compute_c5(fine_h)
        
        if fine_c5 < best_c5:
            best_c5 = fine_c5
            best_h = fine_h
    
    return (best_h.tolist(), best_c5, n_points)
