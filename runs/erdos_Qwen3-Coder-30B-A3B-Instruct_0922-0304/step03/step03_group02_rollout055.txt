import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000  # CMA-ES will generate many evals, allow high iter count
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = repair_h(h, n_points)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES setup
    sigma = 0.1
    lambda_ = 100  # Population size
    
    # Initialize CMA-ES with initial vector and sigma
    es = CMAEvolutionStrategy(h, sigma, {'popsize': lambda_, 'seed': seed})
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Ask for new candidate solutions
        solutions = es.ask()
        
        # Evaluate each solution
        evals = []
        for sol in solutions:
            sol = repair_h(sol, n_points)
            c5 = evaluate_c5(sol, dx)
            evals.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = sol.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        # Tell CMA-ES about the evaluated solutions
        es.tell(solutions, evals)
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
