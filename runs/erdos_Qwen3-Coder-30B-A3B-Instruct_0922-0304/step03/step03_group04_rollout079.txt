import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for multi-resolution optimization
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if res_idx == 0:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Up-sample previous solution
            old_h = best_h
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(old_h)), old_h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Constraints and bounds
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Initial phase: Differential Evolution for global search
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=20,
            seed=seed + res_idx,
            tol=1e-6,
            recombination=0.7
        )
        
        if de_result.success:
            h = de_result.x
        else:
            # Fallback to SLSQP if DE fails
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            if result.success:
                h = result.x

        # Refinement phase: Local optimization
        result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 <= 0.3807:
                    break

        # Perturb slightly for variance
        if res_idx > 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h_pert = h + perturb
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            current_c5_pert = evaluate_c5(h_pert, dx)
            if current_c5_pert < best_c5:
                best_c5 = current_c5_pert
                best_h = h_pert.copy()
                
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        final_c5 = float('inf')
        
    return (best_h.tolist(), final_c5, n_points)
