import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: two pulses
        h = np.zeros(n_points)
        half_width = n_points // 4
        h[n_points//4 : 3*n_points//4] = 0.5
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation helper
    def compute_gradient(h, eps=1e-4):
        h = h.copy()
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        if base_c5 <= 0.3807:
            return np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_c5 = evaluate_c5(h_plus, dx)
            grad[i] = (plus_c5 - base_c5) / eps
        return grad
    
    # Line search
    def line_search(h, grad, alpha_max=0.02, backtrack=0.5, max_iter=20):
        alpha = alpha_max
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < evaluate_c5(h, dx):
                return alpha, h_new
            alpha *= backtrack
        return alpha, h
    
    # Optimization loop
    iterations = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        if no_improvement >= max_no_improvement:
            break
            
        # Compute current C5 and gradient
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if current_c5 <= 0.3807:
            break
            
        grad = compute_gradient(h)
        if np.allclose(grad, 0):
            break
            
        alpha, h_new = line_search(h, grad)
        h = h_new
        
        # Perturbation if stuck
        if no_improvement % 50 == 0 and no_improvement > 0:
            pert_magnitude = 1e-3 * (1.0 / (1.0 + iterations * 0.01))
            noise = np.random.normal(0, pert_magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        iterations += 1
    
    # Final evaluation  
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
