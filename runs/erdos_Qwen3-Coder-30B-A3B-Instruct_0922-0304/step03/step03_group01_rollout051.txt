import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)

    else:

        h = np.ones(n_points)
        h[n_points//3:2*n_points//3] = 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Phase 1: PSO-like search with gradient descent steps
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_limit = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        # Initialize particles with some diversity
        particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        swarm.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_scores.append(objective(particle))
    
    # Global best
    gb_idx = np.argmin(pbest_scores)
    gbest = pbest[gb_idx].copy()
    gbest_score = pbest_scores[gb_idx]
    
    # PSO iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (pbest[i] - swarm[i]) + 
                            social * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Keep in bounds and normalize
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate objective
            current_score = objective(swarm[i])
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = current_score
                
                # Update global best
                if current_score < gbest_score:
                    gbest = swarm[i].copy()
                    gbest_score = current_score
        
        # Local refinement with SLSQP
        if iter_num % 5 == 0:
            refinement_h = gbest.copy()
            refinement_result = minimize(objective, refinement_h, 
                                       method='SLSQP', bounds=bounds, 
                                       constraints=cons, 
                                       options={'maxiter': 10, 'ftol': 1e-9})
            if refinement_result.success:
                refined_h = refinement_result.x
                refined_c5 = objective(refined_h)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
            else:
                # If refinement fails, use the global best
                refined_c5 = objective(gbest)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = gbest.copy()

            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
    
    # Final SLSQP refinement
    final_result = minimize(objective, best_h.copy(), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-9})

        final_h = final_result.x
        final_c5 = objective(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    # Final correction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
