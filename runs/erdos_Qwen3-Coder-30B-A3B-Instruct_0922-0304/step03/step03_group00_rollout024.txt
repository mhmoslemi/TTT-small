import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, dx):
    """Compute gradient with respect to shift k"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # h[i] * (1 - h[i + k]) for i = 0 to n - k - 1
        for i in range(n - k):
            grad[i] += 1 - h[i + k]
            if i + k < n:
                grad[i + k] -= h[i]
    else:
        # h[i] * (1 - h[i + k]) for i = -k to n - 1
        for i in range(-k, n):
            grad[i] += 1 - h[i + k]
            if i + k < n and i + k >= 0:
                grad[i + k] -= h[i]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 50000
    
    # Initialize h from initial_h_values if available

        h = project_h(h, n_points)
    else:

        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = (i + 1) / (mid + 1)
            h[n_points - 1 - i] = (i + 1) / (mid + 1)
        h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k_index = np.argmax(overlap)
        max_overlap = overlap[max_k_index] * dx
        
        # Convert full correlation index to actual shift k
        k = max_k_index - (n_points - 1)
        
        # Compute gradient for the maximal shift
        grad = compute_gradient(h, k, dx)
        
        # Adaptive step size
        step_size = 0.01
        best_step_size = step_size
        best_new_h = h.copy()
        best_new_c5 = max_overlap
        
        # Try step sizes to find a better solution
        for step in [step_size, step_size*2, step_size*0.5]:
            h_trial = h - step * grad
            h_trial = project_h(h_trial, n_points)
            new_c5 = evaluate_c5(h_trial, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_new_h = h_trial.copy()
                best_step_size = step
                
        # Update h with best found step
        h = best_new_h.copy()
        
        # Evaluate and update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
