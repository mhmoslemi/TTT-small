import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Fallback: symmetric triangular
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = project_h(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 50000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the maximum overlap shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Compute gradients for the max shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # h[i] appears in correlation with h[i+shift] and h[i-shift]
            if 0 <= i + max_shift < n_points:
                grad[i] += 1 - h[i + max_shift]
            if 0 <= i - max_shift < n_points:
                grad[i] += 1 - h[i - max_shift]
            # Account for h[i] terms in the correlation
            grad[i] -= 2 * h[i]
        
        # Adaptive learning rate
        step_size = 0.01 / (1.0 + 0.0001 * iteration)
        
        # Update h
        h_new = h - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Check if new point is better, if so accept it
        current_c5 = evaluate_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Revert to previous h
            h = h_new.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
