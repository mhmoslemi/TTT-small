import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Use a more direct approach with CVXPY for better optimization
    # Discretize into 30 intervals for better resolution, then sample to n_points
    M = 30
    interval_width = 2.0 / M
    h_vars = cp.Variable(M)
    
    # Constraints
    sum_constraint = cp.sum(h_vars * interval_width) == 1.0
    bound_constraints = [0 <= h_vars, h_vars <= 1]
    
    # Create overlap constraints
    max_overlaps = []
    for k in range(2*M):  # Adjusted range for full correlation
        # Compute the contribution to overlap with shift k
        overlap = 0
        for i in range(M):
            j = i + k
            if j < M:
                overlap += h_vars[i] * (1 - h_vars[j]) * interval_width
        max_overlaps.append(overlap)
    
    # Use a new variable for the maximum overlap
    C = cp.Variable()
    overlap_constraints = [overlap <= C for overlap in max_overlaps]
    
    # Objective
    objective = cp.Minimize(C)
    
    # Formulate and solve problem
    prob = cp.Problem(objective, bound_constraints + [sum_constraint] + overlap_constraints)
    
    # Try to solve with different solvers
    solver_options = {'abstol': 1e-4, 'reltol': 1e-4, 'max_iters': 2000}
    try:
        prob.solve(solver=solver_options, verbose=False)
    except Exception as e:
        # Fall back to a greedy approach
        h = h_init.copy()
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        best_c5 = float('inf')
        best_h = h
        
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            # Simple gradient descent on a small step
            current_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Compute a rough gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = np.max(np.correlate(h_plus, 1-h_plus, mode="full")) * dx
                grad[i] = (c5_plus - current_c5) / eps
                
            # Update step
            step_size = 0.001
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        return (best_h.tolist(), best_c5, n_points)
    
    # Extract solution
    h_solution = h_vars.value
    # Convert to step function
    h = np.zeros(n_points)
    for i in range(M):
        start = int(i * n_points / M)
        end = int((i + 1) * n_points / M)
        h[start:end] = h_solution[i]
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    final_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    
    return (h.tolist(), final_c5, n_points)
