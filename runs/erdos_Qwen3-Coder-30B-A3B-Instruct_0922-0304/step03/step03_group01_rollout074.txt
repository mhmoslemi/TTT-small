import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation with finite differences
    def compute_gradient(h, eps=1e-4):
        h_base = h.copy()
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h_base, dx)
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        return grad
    
    # Line search for optimal step size
    def line_search(h, grad, max_iter=20):
        alpha = 0.01  # Initial step size
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < evaluate_c5(h, dx):
                return alpha, h_new
            alpha *= 0.5
        return alpha, h
    
    # Optimization loop
    last_improvement = 0
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Line search
        alpha, h_new = line_search(h, grad)
        
        # Update h
        h = h_new
        
        # Periodic perturbation
        if iteration - last_improvement > 50:
            perturbation_magnitude = 1e-3 * (1.0 / (1.0 + iteration**0.5))
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
