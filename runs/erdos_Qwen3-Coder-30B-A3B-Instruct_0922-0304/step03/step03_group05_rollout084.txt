import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        # Add small noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Initialize multiple points using random perturbations
    candidates = [h.copy()]
    for _ in range(4):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        candidates.append(perturbed)

    # Optimization loop for multiple initializations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i, candidate in enumerate(candidates):
            # Local optimization with SLSQP
            result = minimize(
                objective, 
                candidate, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                new_candidate = result.x
            else:
                new_candidate = candidate
            
            # Ensure feasibility
            new_candidate = np.clip(new_candidate, 0, 1)
            new_candidate = new_candidate * (n_points / 2) / np.sum(new_candidate)
            
            # Evaluate
            current_c5 = compute_c5(new_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_candidate.copy()
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
            
            # Update candidate for next round
            candidates[i] = new_candidate
            
        # Basin hopping: random perturbations
        if iteration % 50 == 0:
            for i in range(len(candidates)):
                # Decrease step size with iterations
                sigma = 0.1 * np.exp(-iteration / 500)
                perturb = np.random.normal(0, sigma, n_points)
                new_candidate = candidates[i] + perturb
                new_candidate = np.clip(new_candidate, 0, 1)
                new_candidate = new_candidate * (n_points / 2) / np.sum(new_candidate)
                candidates[i] = new_candidate
                
    # Try adaptive refinement to higher resolution if time allows
    if time.time() - start_time < budget_s - 2 and n_points == 70:
        # Interpolate to higher resolution
        n_points = 140
        dx = 2.0 / n_points
        # Resample h to higher resolution using linear interpolation
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, 70), best_h)
        # Re-scale
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        best_h = h_upsampled.copy()
        best_c5 = compute_c5(best_h)
        
        # Fine-tune with SLSQP
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
            
        return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
