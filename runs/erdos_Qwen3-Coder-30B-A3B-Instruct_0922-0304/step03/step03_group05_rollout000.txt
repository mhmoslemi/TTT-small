import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        old_points = len(h)
        x_old = np.linspace(0, 2, old_points)
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_old, h)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Generate multiple initializations
    initial_candidates = []
    
    # Parent initialization
    initial_candidates.append(h.copy())
    
    # Random perturbations
    for _ in range(5):
        h_pert = h + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_candidates.append(h_pert)

    # Evaluate all initial candidates
    for h_init in initial_candidates:
        h = h_init.copy()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Optimization loop
    max_iter = 1000
    no_improvement_count = 0
    last_best = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current best
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # If no improvement for a while, try re-initialization
        if no_improvement_count > 100:
            h = best_h.copy() + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
        else:
            # Use the best current solution as starting point
            h = best_h.copy()
            
        # Optimize using SLSQP
        result = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
        else:
            # If optimization fails, use perturbation
            h += np.random.normal(0, 0.01, n_points)
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        best_h = h.copy()

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
