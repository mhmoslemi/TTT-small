import numpy as np
import time

def evaluate_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - smooth pulse
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 15:center + 15] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative update approach
    max_iter = 1000
    epsilon = 0.1
    damping = 0.99
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Update each element using multiplicative updates
        h_new = np.zeros_like(h)
        
        # Compute contribution for each element over all lags
        for i in range(n_points):
            total_contribution = 0.0
            for k in range(-n_points+1, n_points):
                if 0 <= i + k < n_points:
                    idx = k + n_points - 1
                    # Using exponential weights based on overlap contributions
                    exp_neg = np.exp(-S[idx] / epsilon)
                    exp_pos = np.exp(S[idx] / epsilon)
                    # Proportional to negative contribution but penalized
                    total_contribution += exp_neg / (1 + exp_pos)
            
            # Apply update rule
            if total_contribution > 0:
                h_new[i] = h[i] * total_contribution * damping
            else:
                h_new[i] = h[i]
        
        # Repair feasibility
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
