import numpy as np
from scipy.signal import correlate
import time
import cma

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 ≤ h[i] ≤ 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    """Objective function for CMA-ES: minimize C5"""
    h = repair_h(h, n_points)
