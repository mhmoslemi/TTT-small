import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES parameters
    lambda_ = 100  # Population size
    mu = 50        # Number of parents
    sigma = 0.1    # Initial standard deviation
    cc = 4 / (n_points + 4)  # Time constant for centroid update
    cs = (mu + 2) / (n_points + mu + 5)  # Time constant for sigma update
    ccov = 2 / ((n_points + 2)**2 + mu)  # Time constant for covariance matrix update
    chi_n = np.sqrt(n_points) * (1 - 1/(4*n_points) + 1/(21*n_points**2))  # Expected value of norm of unit vector
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))  # Weights for weighted recombination
    weights = weights / np.sum(weights)  # Normalize weights
    
    # Initialize CMA-ES state variables
    mean = h.copy()
    C = np.eye(n_points)  # Covariance matrix
    p_c = np.zeros(n_points)  # Evolution path for centroid
    p_sigma = np.zeros(n_points)  # Evolution path for sigma
    fitness_history = []
    
    # Optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sample lambda individuals from the distribution
        samples = np.random.multivariate_normal(mean, sigma**2 * C, lambda_)
        # Repair samples: clip to [0,1] and scale to sum to n_points/2
        individuals = []
        for s in samples:
            s = np.clip(s, 0, 1)
            s = s * (n_points / 2) / np.sum(s)
            individuals.append(s)
        
        # Evaluate individuals
        fitnesses = []
        for ind in individuals:
            fitness = evaluate_c5(ind, dx)
            fitnesses.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        best_indices = sorted_indices[:mu]
        best_fitness = fitnesses[sorted_indices[0]]
        
        # Update best solution
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = individuals[sorted_indices[0]].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Update mean
        old_mean = mean.copy()
        mean = np.sum([weights[i] * individuals[idx] for i, idx in enumerate(best_indices)], axis=0)
        
        # Update evolution paths
        p_sigma = (1 - cs) * p_sigma + np.sqrt(cs * (2 - cs) * mu) * (mean - old_mean) / sigma
        p_c = (1 - cc) * p_c + np.sqrt(cc * (2 - cc) * mu) * (mean - old_mean) / sigma
        
        # Update covariance matrix
        z = np.array([individuals[idx] - old_mean for idx in best_indices])
        C = (1 - ccov) * C + ccov * (np.outer(p_c, p_c) + 
                                   (1 - 1/mu) * np.dot(z.T, z) / (mu * weights.dot(weights)))
        
        # Update step size
        sigma = sigma * np.exp(cs / chi_n * (np.linalg.norm(p_sigma) / chi_n - 1))
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
