from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        h_new = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(h)), h)
        h = h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Generate multiple initializations
    initial_candidates = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.interp(np.linspace(0, 2, n_points), 
                             np.linspace(0, 2, len(h_parent)), h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_candidates.append(h_parent.copy())
    
    # Random perturbations
    for _ in range(4):
        h_pert = h.copy()
        h_pert += np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_candidates.append(h_pert.copy())
    
    # Main optimization loop
    max_iter = 1000
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate initial candidates and find best
        current_best_c5 = float('inf')
        current_best_h = None
        
        for h_init in initial_candidates:
            h_trial = h_init.copy()
            # Run optimization step
            try:
                result = minimize(
                    objective, 
                    h_trial, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8}
                )
                if result.success:
                    h_opt = result.x
                else:
                    h_opt = h_trial
            except:
                h_opt = h_trial
            
            # Ensure feasibility
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            current_c5 = compute_c5(h_opt)
            if current_c5 < current_best_c5:
                current_best_c5 = current_c5
                current_best_h = h_opt.copy()
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Stop if no improvement for many iterations
        if no_improve_count > max_no_improve:
            break
            
        # Generate new perturbations for next round
        initial_candidates = []

            h_pert = best_h.copy()

            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            initial_candidates.append(h_pert.copy())
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
