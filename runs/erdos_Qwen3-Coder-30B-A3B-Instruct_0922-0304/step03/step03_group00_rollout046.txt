import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points, target_sum):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points, target_sum)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, n_points, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projective gradient descent with adaptive step size
    max_iter = 10000
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find the maximum
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update with momentum
        prev_grad = momentum * prev_grad + (1 - momentum) * grad
        h = h - step_size * prev_grad
        
        # Project back to feasible region
        h = project_h(h, n_points, target_sum)
        
        # Adaptive step size adjustment
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # Increase step size slightly if improvement seen
            step_size = min(0.1, step_size * 1.05)
        else:
            # Decrease step size if no improvement
            step_size *= 0.95
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
