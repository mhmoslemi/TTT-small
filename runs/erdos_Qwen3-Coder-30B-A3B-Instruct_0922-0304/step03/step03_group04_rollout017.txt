import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fftpack import fft, ifft
from scipy.stats import qmc

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Clamp to [0, 1] and renormalize
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return h

def spectral_initialization(n_points):
    # If initial_h_values is available, use it
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points / 2)
        return h
    
    # Default initialization
    h = np.ones(n_points)
    h[n_points//2:] = 0
    h = project_to_feasible(h, n_points / 2)
    return h

def get_spectrum(h):
    # FFT of h
    H = fft(h)
    return H

def perturb_spectrum(H, mask):
    # Perturb the spectrum with noise in masked regions
    H_perturbed = H.copy()
    H_perturbed[mask] += np.random.normal(0, 0.05, np.sum(mask))
    return H_perturbed

def generate_candidates(n_points, n_candidates, target_sum):
    candidates = []
    
    # Use Latin Hypercube Sampling in spectral domain
    sampler = qmc.LatinHypercube(d=2 * n_points, seed=42)
    samples = sampler.random(n_candidates)
    samples = samples * 2 - 1  # Map to [-1, 1]
    
    # Convert samples to complex spectrum
    for i in range(n_candidates):
        # Create complex spectrum
        real_part = samples[i, :n_points]
        imag_part = samples[i, n_points:] 
        H_complex = real_part + 1j * imag_part
        
        # Inverse FFT to get h
        h = ifft(H_complex).real
        h = project_to_feasible(h, target_sum)
        candidates.append(h)
        
    return candidates

def refine_candidates(candidates, n_points, dx, target_sum):
    best_candidates = []
    for i, h in enumerate(candidates):
        h_projected = project_to_feasible(h, target_sum)
        c5 = evaluate_c5(h_projected, dx)
        best_candidates.append((c5, h_projected))
    best_candidates.sort(key=lambda x: x[0])
    return best_candidates[:max(1, len(best_candidates) // 10)]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    h = spectral_initialization(n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Spectral initialization
    H = get_spectrum(h)
    init_spectrum = np.abs(H)
    
    # 1. Initialize with LHS candidates
    candidates = generate_candidates(n_points, 50, target_sum)
    
    # 2. Mutate candidates and evaluate
    for iteration in range(100):  # 100 mutations of 50 candidates
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate and keep best
        for i, h_candidate in enumerate(candidates):
            c5 = evaluate_c5(h_candidate, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Mutate candidates
        new_candidates = []
        for h_candidate in candidates:
            # Modern spectral mutation
            H = get_spectrum(h_candidate)
            # Determine mask based on initial spectrum
            mask = init_spectrum > np.percentile(init_spectrum, 70)
            H_mutated = perturb_spectrum(H, mask)
            h_mutated = ifft(H_mutated).real
            h_mutated = project_to_feasible(h_mutated, target_sum)
            new_candidates.append(h_mutated)
            
        candidates = new_candidates
    
    # Now, try local refinement on best candidates
    refined_candidates = refine_candidates(candidates, n_points, dx, target_sum)
    
    # Local refinement with simplex
    for (c5, h_refined) in refined_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Nelder-Mead refinement
        def obj(h):
            h = project_to_feasible(h, target_sum)
            return evaluate_c5(h, dx)
            
        # Start with simplex around the candidate
        simplex_start = np.array([h_refined])
        for _ in range(10):
            # Generate random perturbations
            perturb = np.random.normal(0, 0.05, n_points)
            perturb = project_to_feasible(perturb, target_sum)
            new_h = h_refined + perturb
            new_h = project_to_feasible(new_h, target_sum)
            
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < c5:
                c5 = new_c5
                h_refined = new_h
                
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined.copy()
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
