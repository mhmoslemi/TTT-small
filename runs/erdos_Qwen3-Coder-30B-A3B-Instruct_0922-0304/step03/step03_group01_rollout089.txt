import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    """Create initial population"""
    population = []
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h0 = np.array(initial_h_values)
        h0 = repair_h(h0, n_points)
        population.append(h0)
    
    # Fill remaining population with random samples
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    if np.random.rand() < mutation_rate:
        # Gaussian perturbation
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        # Random reset
        if np.random.rand() < 0.1:
            h = np.random.rand(n_points)
    return repair_h(h, n_points)

def crossover(parent1, parent2):
    """Perform crossover between two parents"""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, 
                      generations, initial_h_values, budget_s):
    """Genetic Algorithm for finding optimal h"""
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h_values)
    best_h = None
    best_c5 = float('inf')
    
    dx = 2.0 / n_points
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)  # Negative because we want to minimize
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        current_best = -fitness[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[0].copy()
        
        # Elitism
        elite_size = max(2, pop_size // 5)
        elite = population[:elite_size]
        
        # Generate next generation
        new_population = elite[:]
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5

def tournament_selection(population, fitness, tournament_size=3):
    """Select individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmax([fitness[i] for i in indices])]
    return population[best_idx]

def refine_with_gradient_descent(h, n_points, dx, budget_s):
    """Refine solution using gradient descent"""
    start_time = time.time()
    max_iter = 10000
    step_size = 0.001
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.2:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-6
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - max_overlap) / eps
        
        # Update
        h = h - step_size * grad
        h = repair_h(h, n_points)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Try genetic algorithm first
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 100
    
    best_h, best_c5 = genetic_algorithm(n_points, pop_size, mutation_rate, 
                                       crossover_rate, generations, 
                                       globals().get('initial_h_values'), 
                                       budget_s)
    
    # Refine with gradient descent
    best_h = refine_with_gradient_descent(best_h, n_points, dx, budget_s)
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure final feasibility
    best_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
