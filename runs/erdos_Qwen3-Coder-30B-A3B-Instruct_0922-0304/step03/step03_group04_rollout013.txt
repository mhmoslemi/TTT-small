import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 30  # Number of intervals
    dx = 2.0 / M
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        # Resample to M intervals
        h_avg = np.zeros(M)
        for i in range(M):
            start_idx = int(i * len(h_initial) / M)
            end_idx = int((i + 1) * len(h_initial) / M)
            h_avg[i] = np.mean(h_initial[start_idx:end_idx])
        h = h_avg.copy()
    else:
        # Fallback initialization
        h = np.ones(M)
        h[M//2:] = 0
        h += np.random.normal(0, 0.05, M)
        h = np.clip(h, 0, 1)
        h = h * (M / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Build matrix of possible overlaps
    # For each shift k, compute overlap between intervals
    A = np.zeros((2*M - 1, M))
    shifts = np.arange(-(M-1), M)
    
    for k_idx, k in enumerate(shifts):
        for i in range(M):
            # Compute overlap of interval i with interval i + k
            if 0 <= i + k < M:
                # Determine overlap length
                if k >= 0:
                    overlap_start = max(0, k * dx)
                    overlap_end = min(dx, (k + 1) * dx)
                    overlap_len = overlap_end - overlap_start
                else:
                    overlap_start = max(0, -k * dx)
                    overlap_end = min(dx, (1 - k) * dx)
                    overlap_len = overlap_end - overlap_start
                A[k_idx, i] = overlap_len
            else:
                A[k_idx, i] = 0
    
    # LP solver
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    constraints = [
        cp.sum(h_vars) == M / 2,
        h_vars >= 0,
        h_vars <= 1,
        C >= 0
    ]
    
    # Add overlap constraints
    for k_idx in range(2*M - 1):
        # Compute sum of h[i] * (1 - h[i+k]) * A[k_idx][i]
        # since h is in intervals, we need to consider the shifted intervals
        overlap_expr = 0
        for i in range(M):
            # Determine i + k in interval indices
            shifted_i = i + shifts[k_idx]
            if 0 <= shifted_i < M:
                # Interval i' = shifted_i
                overlap_expr += h_vars[i] * (1 - h_vars[shifted_i]) * A[k_idx, i]
            else:
                overlap_expr += h_vars[i] * 0 * A[k_idx, i]  # Out of bounds contribution is 0
        
        constraints.append(overlap_expr <= C)
    
    # Minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = [cp.ECOS, cp.SCS, cp.GLPK]
    
    # Try each solver if not infeasible
    for solver in solvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Solve with warm start from previous result
            prob.solve(solver=solver, warm_start=True, max_iters=2000, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                current_h = h_vars.value
                if np.any(np.isnan(current_h)):
                    continue
                current_h = np.clip(current_h, 0, 1)
                # Normalize properly
                current_h = current_h * (M / 2) / np.sum(current_h)
                # Evaluate overlaps
                current_c5 = evaluate_c5(current_h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
        except:
            continue
    
    # Final refinement with simple optimization
    h = best_h.copy()
    bounds = [(0, 1) for _ in range(M)]
    target_sum = M / 2
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use L-BFGS for final refinement
    try:
        from scipy.optimize import minimize
        result = minimize(
            objective,
            h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-8}
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (M / 2) / np.sum(h)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    except:
        pass
    
    # Verification
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, M)
