import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    selected = []
    for _ in range(len(population) // 2):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2):
    if np.random.random() < 0.7:
        point = np.random.randint(1, len(parent1)-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    return parent1.copy(), parent2.copy()

def mutate(individual, mutation_rate=0.2, noise_std=0.05):
    if np.random.random() < mutation_rate:
        for i in range(len(individual)):
            if np.random.random() < 0.2:
                individual[i] += np.random.normal(0, noise_std)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population with constraint repair
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = repair_h(h0, n_points)
        population.append(h0)
    else:
        # Fallback initial h
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = repair_h(h0, n_points)
        population.append(h0)

    # Add random initial individuals
    for _ in range(99):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)

    # Genetic Algorithm main loop
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    for generation in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Update the best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = [best_h.copy()]
        while len(new_population) < 100:
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(repair_h(child1, n_points))
            if len(new_population) < 100:
                new_population.append(repair_h(child2, n_points))
        
        population = new_population
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
