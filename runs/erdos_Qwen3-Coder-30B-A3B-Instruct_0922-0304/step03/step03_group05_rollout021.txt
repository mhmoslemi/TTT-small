import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    max_iter = 1000
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # If no improvement for a while, reduce step size
        if no_improvement_count > 50:
            max_move *= 0.95
            
        # Compute all overlaps to find worst case
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)
        
        # Find top contributors to this overlap
        # The overlap at shift k is sum h[i] * (1 - h[i+k])
        # So we want to reduce h[i] * (1 - h[i+k]) for top i
        
        # For shift max_shift, the overlap sum over i is:
        # sum h[i] * (1 - h[i+max_shift]) for i in [0, n_points-1]
        # Actually, there's a better way:
        # overlap[k] = sum h[i] * (1 - h[(i+k)%n_points])
        # k ranges from -n+1 to n-1
        # Let's compute for direct indices for simplicity
        
        # Recomputing the overlap efficiently
        shift_overlap = correlate(h, 1 - h, mode="full") * dx
        max_shift_idx = np.argmax(shift_overlap)
        # Get the actual shift value
        actual_shift = max_shift_idx - (n_points - 1)
        
        # Get the original overlap values - all overlaps
        overlap_full = correlate(h, 1 - h, mode="full") * dx
        max_over = np.max(overlap_full)
        
        if actual_shift < 0:
            # Convert to forward shift
            forward_shift = -actual_shift
            # Compute max contributions (i, i+forward_shift)
            contributions = h[:-forward_shift] * (1 - h[forward_shift:])
            top_indices = np.argsort(contributions)[-int(n_points * 0.1):][::-1]
            # If not enough, get all
            if len(top_indices) < 10:
                top_indices = np.argsort(contributions)[::-1]
        else:
            # Forward shift
            forward_shift = actual_shift
            contributions = h[:-forward_shift] * (1 - h[forward_shift:])
            top_indices = np.argsort(contributions)[-int(n_points * 0.1):][::-1]
            if len(top_indices) < 10:
                top_indices = np.argsort(contributions)[::-1]
        
        # Reduce the most negative contribution
        for idx in top_indices:
            if idx + actual_shift >= n_points:
                continue
                
            # Compute max adjustment
            orig_idx = idx
            shift_idx = (idx + actual_shift) % n_points
            
            if h[orig_idx] <= 0 or h[shift_idx] >= 1:
                continue
                
            # We want to decrease h[orig_idx] and increase h[shift_idx]
            max_adjust = min(h[orig_idx], 1 - h[shift_idx], max_move)
            
            if max_adjust <= 0:
                continue
                
            # Apply the adjustment
            h[orig_idx] -= max_adjust
            h[shift_idx] += max_adjust
            
        # Maintain the integral constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Clip to feasible range
        h = np.clip(h, 0, 1)
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
