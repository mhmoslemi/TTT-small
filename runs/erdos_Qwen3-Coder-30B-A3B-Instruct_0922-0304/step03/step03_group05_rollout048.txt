import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_onto_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative adjustment approach
    max_iter = 1000
    max_move = 0.01
    patience = 0
    min_patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > min_patience:
            max_move *= 0.95
            patience = 0

        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Create a list of highest contributing indices for the max overlap
        indices = []
        for i in range(n_points):
            k = (max_k - i) % n_points
            indices.append((i, k, h[i] * (1 - h[k])))
        
        # Sort by contribution and take top contributors
        indices.sort(key=lambda x: x[2], reverse=True)
        top_contributors = indices[:max(1, n_points // 20)]
        
        # Perform adjustments
        for i, k, _ in top_contributors:
            # Compute adjustment
            adjust_i = min(h[i], max_move)
            adjust_k = min(1 - h[k], max_move)
            epsilon = min(adjust_i, adjust_k)
            
            if epsilon > 1e-10:
                h[i] -= epsilon
                h[k] += epsilon
                
        # Project back onto feasible set
        h = project_onto_feasible(h)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
