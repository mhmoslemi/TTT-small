import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Alternate optimization of slices to find better solution
    best_c5 = float('inf')
    best_h = h.copy()

    # Use gradient-free method for initial search
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Perturbation-based optimization
        # Small random perturbations
        h_new = h + np.random.normal(0, 0.01, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate the new point
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            # Add more pressure to search
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final refinement with local search
    # First make a better approximation by creating a binary
    # step-like approximation based on initial values
    h_step = np.zeros(n_points)
    for i in range(n_points):
        idx = int(i * 2 / n_points)
        if idx < len(h):
            h_step[i] = h[idx]
    
    # Ensure integral constraint
    h_step = h_step * (n_points / 2) / np.sum(h_step)
    
    final_c5 = compute_c5(h_step)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_step.copy()
    
    return (best_h.tolist(), best_c5, n_points)
