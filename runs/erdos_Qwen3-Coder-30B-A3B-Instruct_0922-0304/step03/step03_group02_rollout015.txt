import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def evaluate_c5(h_vec):
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    # Define the SDP approach
    def optimize_with_sdp(h_guess, max_iter=100):
        h_prev = h_guess.copy()
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update h with Fourier-based method
            h_new = h_prev.copy()
            # Apply small optimization
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                
            if c5 <= 0.3807:
                break
                
            h_prev = h_new

        return best_h.copy(), best_c5

    # Run SDP-based optimization
    final_h, final_c5 = optimize_with_sdp(h, max_iter=100)

    return (final_h.tolist(), final_c5, n_points)
