import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the shift with maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)
        
        # Set up the QP for this specific shift k
        # Variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize sum over shift k of h[i] * (1 - h[i+k])
        # Which is h^T * (1 - h_shifted) where h_shifted is h with shift k
        # For shift k, h[i+k] is h[i] shifted right by k
        # Element-wise product h[i] * (1 - h[i+k])
        
        # Construct matrix for quadratic part
        Q = np.zeros((n_points, n_points))
        if max_k >= 0:
            # h[i] * (1 - h[i + max_k]) 
            for i in range(n_points - max_k):
                Q[i, i + max_k] = 1.0
        else:
            # h[i] * (1 - h[i + max_k]) 
            for i in range(-max_k, n_points):
                Q[i, i + max_k] = 1.0
                
        Q = Q + Q.T  # Symmetrize
        c = np.zeros(n_points)  # Linear part is 0 since we have h*i term only
        obj = cp.Minimize(cp.quad_form(h_vars, Q) - 2 * cp.sum(cp.multiply(h_vars, np.ones(n_points))) * dx)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Create and solve the problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    # Renormalize to meet the constraint
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    # Fallback if optimization fails
                    step = 0.001
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    corr = correlate(h, 1-h, mode="full")
                    max_idx = np.argmax(corr)
                    max_k = max_idx - (n_points - 1)

                    for i in range(n_points):
                        h_plus = h.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1-h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - np.max(corr) * dx) / eps
                    
                    h -= step * grad
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback if QP fails for any reason
            step = 0.001
            grad = np.zeros(n_points)
            eps = 1e-4
            corr = correlate(h, 1-h, mode="full")
            max_idx = np.argmax(corr)
            max_k = max_idx - (n_points - 1)

            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1-h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - np.max(corr) * dx) / eps
            
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
