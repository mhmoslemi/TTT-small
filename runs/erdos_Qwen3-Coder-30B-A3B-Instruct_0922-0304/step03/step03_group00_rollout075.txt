import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop using quadratic programming
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the maximum overlap shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        # Get the shift k for the maximum overlap
        k = max_idx - (n_points - 1)
        
        # Create QP variables
        h_var = cp.Variable(n_points)
        
        # Set up the quadratic objective for this specific shift
        # sum(h[i] * (1 - h[i + k])) over valid indices
        objective = 0
        if k >= 0:
            # shift forward
            for i in range(n_points - k):
                objective += h_var[i] * (1 - h_var[i + k])
        else:
            # shift backward
            for i in range(-k, n_points):
                objective += h_var[i] * (1 - h_var[i + k])
        
        # Add a regularization to make the QP solution more stable
        objective += 1e-8 * cp.sum_squares(h_var - h)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve QP
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            h_new = h_var.value
            if h_new is not None:
                h = h_new
        except:
            # Fallback: simple gradient step
            step = 0.005
            gradient = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                gradient[i] = (c5_plus - current_c5) / eps
                
            h -= step * gradient
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Ensure we never go out of bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
