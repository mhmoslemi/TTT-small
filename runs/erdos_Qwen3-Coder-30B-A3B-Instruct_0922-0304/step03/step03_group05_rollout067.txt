import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_c5_fast(h):
        # Roll the overlap to align with the rightmost valid overlap
        # In principle, we pad with zeros in the center
        # For this problem: we minimize max of all correlations
        # So we compute all correlations and take max
        o = correlate(h, 1 - h, mode="full")
        return np.max(o) * dx

    # Simple gradient-based optimization with restarts
    def optimize_step(h, max_iter=20):
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Try small perturbations
            perturb = np.random.normal(0, 0.01, n_points)
            h_try = h + perturb
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try * (n_points / 2) / np.sum(h_try)
            
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                
        return best_h, best_c5

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a simple local search to escape local minima
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Take several optimization steps
        current_h, current_c5 = optimize_step(h, 10)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        h = current_h.copy()
        
    # Final boundary check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
