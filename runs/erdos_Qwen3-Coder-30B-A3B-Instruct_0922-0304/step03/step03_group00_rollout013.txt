import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def evaluate_c5(h_vals):
        overlap = correlate(h_vals, 1 - h_vals, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    # SDP-based optimization approach
    max_iter = 300
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate and solve SDP relaxation
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraint: sum(h) = n_points/2
        constraints = [cp.sum(h_vars) == n_points / 2]
        
        # Constraint: 0 <= h[i] <= 1
        constraints += [h_vars >= 0, h_vars <= 1]
        
        # Build overlap constraints using SDP relaxation
        for k in range(-n_points+1, n_points):
            # Compute indices for h[i] * (1 - h[i+k]) with proper handling of boundary
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in [0, n-k-1]
                indices1 = np.arange(n_points - k)
                indices2 = np.arange(k, n_points)
            else:
                # h[i] * (1 - h[i+k]) for i in [0, n+k-1]
                indices1 = np.arange(n_points + k)
                indices2 = np.arange(-k, n_points)
            
            # Careful handling of correlation indices
            if k >= 0:
                # indices1 from 0 to n-k-1, indices2 from k to n-1
                corr_idx = np.arange(n_points - k)
                x = h_vars[corr_idx]
                y = h_vars[corr_idx + k]
                # h[i] * (1 - h[i+k]) which is x * (1 - y) = x - x*y
                term = x - cp.multiply(x, y)
            else:
                # indices1 from 0 to n+k-1, indices2 from -k to n-1
                corr_idx = np.arange(n_points + k)
                x = h_vars[corr_idx]
                y = h_vars[corr_idx - k]
                # h[i] * (1 - h[i+k]) which is x * (1 - y) = x - x*y
                term = x - cp.multiply(x, y)
                
            # Note: We need to ensure our SDP model properly captures this constraint
            # For each fixed k, we must have:
            # sum_{i, i+k valid} h[i] * (1 - h[i+k]) <= C5 * dx
            # Since this causes double-counting in the full correlation, we only need
            # 1 specific single scalar constraint from full correlation
            # Better to extract max value with a more direct approach
            
        # We'll approach it differently. Let's use direct optimization with trust-region
        def objective_slsqp(h_vec):
            # Compute overlaps and get max
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return -np.max(corr) * dx  # negative because minimize
        
        bounds = [(0., 1.) for _ in range(n_points)]
        # For large n it may benefit to resolve from outside
        try:
            # Use trust-constr if available; otherwise fallback to L-BFGS
            import scipy.optimize as opt
            res = opt.minimize(objective_slsqp, h.copy(), 
                              method='trust-constr',
                              bounds=bounds,
                              constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
                              options={'maxiter': 100, 'gtol': 1e-8, 'xtol': 1e-8})
        except:
            # Fallback to L-BFGS
            res = opt.minimize(objective_slsqp, h.copy(), 
                              method='L-BFGS-B',
                              bounds=bounds,
                              constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
                              options={'maxiter': 100})
        
        if res.success:
            h = res.x
        else:
            # Simple gradient ascent direction
            # Evaluate gradient via finite diferences
            current_c5 = evaluate_c5(h)
            lr = 0.002
            h_grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_temp = h.copy()
                h_temp[i] += eps
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                corr_plus = correlate(h_temp, 1 - h_temp, mode="full")
                c5_plus = np.max(corr_plus) * dx
                h_grad[i] = (c5_plus - current_c5) / eps
            h -= lr * h_grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
