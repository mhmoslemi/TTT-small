import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with convex relaxation approach
    # Use 30 intervals for more resolution than original approach
    M = 30  # number of intervals
    
    # Define interval boundaries
    intervals = np.linspace(0, 2, M + 1)
    widths = np.diff(intervals)
    
    # Create step function from interval heights
    def step_from_intervals(h_intervals):
        h_step = np.zeros(n_points)
        for i in range(M):
            start_idx = int(np.round(intervals[i] / 2 * n_points))
            end_idx = int(np.round(intervals[i+1] / 2 * n_points))
            h_step[start_idx:end_idx] = h_intervals[i]
        return h_step
    
    # Generate overlap coefficients
    # We'll build a set of linear constraints
    def compute_overlap_constraints(M, widths, n_points):
        # For full convolution-style overlaps
        L = 2 * n_points  # length of full convolution result
        # Create an overlap matrix based on intervals
        num_shifts = 2 * M  # maximum possible shifts
        overlap_coeffs = []
        shifts = []
        
        # For i-th interval in h, we consider its contribution to all possible overlaps
        for k in range(1, 2*M):  # shifts to consider
            coeffs = np.zeros(M)
            # Numerically approximate shift effect using our fixed intervals
            # For each interval j, compute overlap contribution to shift k
            for j in range(M):
                # overlap with interval (j+k) for shift k
                start = intervals[j]
                end = intervals[j+1]
                if (start + k/100.0) < 2 and (end + k/100.0) >= 0:
                    # Approximate overlap as the width where both are non-zero
                    overlap_start = max(0, start + k/100.0)
                    overlap_end = min(2, end + k/100.0)
                    if overlap_end > overlap_start:
                        # Contribution to shift k for interval j is
                        # (overlap width) * (1 - h_j) = width * (1-h_j)
                        coeffs[j] = overlap_end - overlap_start
                else:
                    coeffs[j] = 0.0
            
            overlap_coeffs.append(coeffs)
            shifts.append(k)
            
        return np.array(overlap_coeffs), np.array(shifts)
    
    # Try convex optimization
    try:
        # Variables
        h_vars = Variable(M)
        C = Variable()
        
        # Constraints for step function
        prob = Problem(Minimize(C),
                       [sum(h_vars * widths) == 1,  # integral constraint
                        h_vars >= 0, h_vars <= 1,   # bounds
                        ])

        # Try a different approach: numerical evaluation with gradient descent
        best_h_convex = h.copy()
        best_c5_convex = evaluate_c5(best_h_convex, dx)
        
        # Create lots of candidate h functions
        # Here we'll sample several random convex combinations and numerical optimization
        n_samples = 30000
        best_c5 = float('inf')
        best_h = h.copy()
        
        # Sample possibilities for convex relaxation
        for iter in range(n_samples):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Create random interval heights that satisfy the sum constraint
            h_sample = np.random.rand(M)
            h_sample = h_sample * (n_points / 2) / np.sum(h_sample * widths)
            
            # Ensure constraint
            h_sample = np.clip(h_sample, 0, 1)
            h_sample = h_sample * (n_points / 2) / np.sum(h_sample * widths)
            
            h_step = step_from_intervals(h_sample)
            
            # Evaluate C5
            c5 = evaluate_c5(h_step, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_step.copy()
                
            # Early stopping
            if best_c5 <= 0.3807:
                break
                
    except:
        # Fallback to simple gradient descent
        h = h.copy()
        
        # Gradient descend with fixed step size
        step = 0.001
        iterations = 1000
        
        for i in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlaps
            corr = correlate(h, 1 - h, mode="full")
            # Get max overlap and its position
            max_overlap = np.max(corr)
            max_idx = np.argmax(corr) - (2*n_points - 1)  # shift
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus)
                grad[j] = (c5_plus - max_overlap) / eps
                
            # Update h
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
