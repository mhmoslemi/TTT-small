        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

        h = np.array(h)

        overlap = correlate(h, 1 - h, mode="full")

    best_c5 = float('inf')
    best_h = h.copy()

    # Optimization loop with adaptive strategy
    max_iter = 1000
    iter_count = 0

    # to escape local optima
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        current_c5 = compute_c5_with_custom_method(h)
        if current_c5 < best_c5:
            best_c5 = current_c5

        if iter_count % 5 == 0:

            # Use finite differences to estimate gradient
            grad = np.zeros_like(h)
