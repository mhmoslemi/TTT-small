import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start with perturbations
    initial_solutions = [h.copy()]
    for _ in range(4):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_solutions.append(perturbed)
    
    # Local optimization for each starting point
    for i, init_h in enumerate(initial_solutions):
        current_h = init_h.copy()
        for _ in range(50):  # iterations per start
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(
                objective, 
                current_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 5, 'ftol': 1e-8}
            )
            
            if result.success:
                current_h = result.x
            else:
                current_h += np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
            # Track best
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            if best_c5 <= 0.3807:
                break

    # Basin hopping
    current_h = best_h.copy()
    step_size = 0.1
    cooling_rate = 0.99
    max_basin_iters = 1000
    for i in range(max_basin_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb solution
        perturbation = np.random.normal(0, step_size, n_points)
        new_h = current_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Local optimization on perturbed
        result = minimize(
            objective, 
            new_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            new_h = result.x
        else:
            # If optimization fails, just project
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Update current if better
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            current_h = new_h.copy()
        else:
            current_h = new_h.copy()
        
        # Anneal step size
        step_size *= cooling_rate
        
        if best_c5 <= 0.3807:
            break

    # Final refinement
    result = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if result.success:
        best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
