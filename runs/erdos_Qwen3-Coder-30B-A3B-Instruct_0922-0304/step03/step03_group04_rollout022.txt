import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet representation
    wavelet = 'haar'
    max_levels = int(np.log2(n_points)) - 1
    if max_levels < 3:
        max_levels = 3
    
    # Iterative optimization with wavelet domain
    max_iter = 20
    sparsity_factor = 0.1
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Wavelet transform
        coeffs = pywt.wavedec(h, wavelet, level=max_levels)
        c_flat = pywt.vectorize_coeffs(coeffs)
        
        # Soft-thresholding to encourage sparsity
        threshold = sparsity_factor * np.std(c_flat)
        if threshold > 0:
            c_flat = pywt.threshold(c_flat, threshold, mode='soft')
            coeffs = pywt.unvectorize_coeffs(c_flat, coeffs)
            h_new = pywt.waverec(coeffs, wavelet)
            if len(h_new) > n_points:
                h_new = h_new[:n_points]
            elif len(h_new) < n_points:
                h_new = np.pad(h_new, (0, n_points - len(h_new)), mode='constant')
        else:
            h_new = h.copy()
            
        # Projection to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Local refinement using gradient descent
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vec):
            return evaluate_c5(h_vec, dx)
        
        # Optimization using SLSQP
        result = minimize(
            objective,
            h_new.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 10, 'ftol': 1e-6}
        )
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        else:
            h_refined = h_new.copy()
        
        current_c5 = evaluate_c5(h_refined, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
            
            # Check for target
            if best_c5 <= 0.3807:
                break
        else:
            # Slight perturbation if no improvement
            perturbation = np.random.normal(0, 0.001, n_points)
            h_perturbed = h_refined + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_c5(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
        
        # Increase sparsity factor if not improving
        if iteration % 5 == 0 and iteration > 0:
            sparsity_factor *= 1.1
            
        h = h_refined.copy()

    return (best_h.tolist(), best_c5, n_points)
