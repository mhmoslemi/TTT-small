import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    swarm.append(h0.copy())
    velocities.append(np.zeros(n_points))
    
    # Add triangular functions
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points//2) / n_points)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start, w_end = 0.9, 0.4
    c_p, c_g = 1.5, 1.5
    v_max = 0.1
    
    # Initialize personal bests and global best
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_score
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand()
            r_g = np.random.rand()
            
            vel_update = (w * velocities[i] + 
                         c_p * r_p * (pbest[i] - swarm[i]) + 
                         c_g * r_g * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(vel_update, -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Ensure sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            score = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = swarm[i].copy()
                    gbest_score = score
                    
                    if gbest_score < best_c5:
                        best_c5 = gbest_score
                        best_h = gbest.copy()
            
            # Early exit
            if gbest_score <= 0.3807:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
