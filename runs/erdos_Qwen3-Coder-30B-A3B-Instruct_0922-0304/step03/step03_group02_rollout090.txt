import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Alternative: SDP-based approach for better convergence
    def sdp_optimize_h(start_h, n_points, dx, max_iter=100):
        h = start_h.copy()
        # SDP formulation using CVXPY
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(cp.Variable(2*n_points - 1)))
        
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Define constraint for the overlap
        for k in range(1, n_points):
            corr = cp.Variable(n_points)  # Infinite correlation constraint
            # Simplified: use empirical correlation approximation
            corr_sum = cp.sum(cp.multiply(h_var, cp.roll(h_var, k)))
            constraints.append(corr_sum <= 1 - 0.3807)  # target
            
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=CP SensitivityAnalysis=True, verbose=False)
            if prob.status == 'optimal':
                return h_var.value
        except:
            pass
        return h  # fallback
    
    # SGD-based local search
    def local_search(h, dx, steps=50):
        step_size = 0.01
        for _ in range(steps):
            if time.time() - start_time > budget_s - 1:
                break
            # Gradient approximation
            c5_orig = evaluate_c5(h, dx)
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_eps = h.copy()
                h_eps[i] += eps
                h_eps = np.clip(h_eps, 0, 1)
                h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
                c5_eps = evaluate_c5(h_eps, dx)
                grad[i] = (c5_eps - c5_orig) / eps
            
            # Update step with projection
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate
            c5_new = evaluate_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
        return h
    
    # Main optimization loop
    max_iter = 300
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local search
        h = local_search(h, dx, steps=30)
        
        # Try SDP warm-start
        if iteration % 10 == 0 and 'initial_h_values' in globals():
            try:
                new_h = sdp_optimize_h(h, n_points, dx)
                if np.all(new_h >= 0) and np.all(new_h <= 1):
                    h = new_h.copy()
                    h = h * (n_points / 2) / np.sum(h)
            except:
                pass
                
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
