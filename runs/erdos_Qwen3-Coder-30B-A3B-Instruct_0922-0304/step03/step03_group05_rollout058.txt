import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start: try multiple initializations
    initial_solutions = []
    initial_solutions.append(h.copy())
    
    # Generate perturbed initializations
    for _ in range(4):
        h_pert = h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_solutions.append(h_pert.copy())
    
    # Optimize each initialization
    for h_start in initial_solutions:
        h_current = h_start.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            if result.success:
                h_current = result.x
            else:
                h_current += np.random.normal(0, 0.01, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Basin hopping: perturb the best solution and refine
    h_current = best_h.copy()
    iteration = 0
    T = 1000  # cooling time constant
    
    # Use a larger number of steps and gradually reduce perturbation
    while time.time() - start_time < budget_s - 1:
        if iteration % 100 == 0 and iteration > 0:
            if n_points == 70 and (time.time() - start_time) < budget_s - 100:
                # Try to double resolution if time allows
                n_points = 140
                dx = 2.0 / n_points
                h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                      np.linspace(0, 2, len(best_h)), best_h)
                h_current = h_upsampled.copy()
                # Scale for constraint
                h_current = h_current * (n_points / 2) / np.sum(h_current)
                bounds = [(0, 1) for _ in range(n_points)]
        
        # Perturb the solution
        sigma = 0.1 * np.exp(-iteration / T)
        h_pert = h_current + np.random.normal(0, sigma, len(h_current))
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        # Local optimization
        result = minimize(
            objective, 
            h_pert, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
        else:
            h_new = h_pert
            
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        current_c5 = compute_c5(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            if best_c5 <= 0.3807:
                break
                
        h_current = h_new.copy()
        iteration += 1
    
    # Final refinement using higher tolerance
    final_result = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if final_result.success:
        best_h = final_result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
