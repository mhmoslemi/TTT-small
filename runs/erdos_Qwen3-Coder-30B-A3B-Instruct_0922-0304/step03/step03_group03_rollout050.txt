def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Dual-based optimization with local refinement
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)

            best_c5 = current_c5

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use a variant of projected gradient descent with adaptive step size
        step = 0.01
        for _ in range(100):  # Local steps
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * dx
            max_idx = np.argmax(corr)
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h)

            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update with projected gradient
            h_new = h - step * grad
            h_new = np.clip(h_new, 0, 1)

            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
            else:
                step *= 0.9
                
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
