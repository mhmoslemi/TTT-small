import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    resolutions = [10, 20, 40, 70, 100, 140, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Use provided initial_h_values if available
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (70 / 2) / np.sum(initial_h)
        
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if n_points == resolutions[0]:
            if initial_h is not None:
                h = initial_h.copy()
            else:
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Up-sample the previous solution
            old_h = h.copy()
            h = np.zeros(n_points)
            for i in range(n_points):
                idx = int(i * len(old_h) / n_points)
                h[i] = old_h[idx]
            # Add some noise
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Optimization: Differential Evolution for global search
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Run differential evolution
        de_result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            seed=seed,
            maxiter=100,
            popsize=20,
            tol=1e-6
        )
        
        if de_result.success:
            h = de_result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points
                
        # Refinement with SLSQP
        result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points
                
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, best_n_points)
