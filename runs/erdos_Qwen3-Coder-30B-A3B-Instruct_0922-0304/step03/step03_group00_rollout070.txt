import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k_shift, n_points, dx):
    """Compute gradient of overlap integral with respect to h for given shift k"""
    grad = np.zeros(n_points)
    # For shift k, the overlap term is h[i] * (1 - h[i+k])
    # So partial derivative d/dh[i] = (1 - h[i+k]) * dx
    # and d/dh[i+k] = -h[i] * dx
    
    # When k > 0, h[i] appears in h[i] and h[i+k] terms
    # When k < 0, h[i] appears in h[i] and h[i+k] terms
    # But since we're looking at positions where the overlap contributes,
    # we compute the elements where i+k is within bounds
    
    # Apply to indices where both i and i+k are valid
    for i in range(n_points):
        if i + k_shift >= 0 and i + k_shift < n_points:
            # Gradient term from overlap term h[i] * (1 - h[i+k])
            grad[i] += (1 - h[i + k_shift]) * dx
            # Also the gradient from the other term (which is actually the
            # derivative of h[i] * (1 - h[i+k]) w.r.t h[i])
            # and we need to add -h[i] * dx from the derivative of h[i+k] term
            # But that's done in reverse iteration so we don't double count.
            # Actually, for derivative w.r.t h[i]:
            # if the term is h[i] * (1 - h[i+k]), then d/dh[i] = (1 - h[i+k])
            # if the term is h[i+k] * (1 - h[i]), then d/dh[i] = -h[i+k]
    return grad

def adaptive_step_size(prev_c5, current_c5, step_size, improvement_factor=0.95):
    """Adaptively adjust step size based on improvement in C5"""
    if current_c5 < prev_c5:
        # Improved, increase step size
        return min(step_size * 1.05, 0.1)
    else:
        # No improvement, decrease step size
        return max(step_size * improvement_factor, 1e-6)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Fallback initialization: symmetric triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = 2 * i / (mid - 1) if mid > 1 else 0
            else:
                h[i] = 2 * (n_points - i - 1) / (n_points - mid - 1) if n_points - mid > 1 else 0
        h = project_h(h, n_points)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projected gradient descent parameters
    step_size = 0.01
    max_iter = 50000
    min_improvement = 1e-8
    
    prev_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the shift k that causes maximum overlap
        overlap_full = correlate(h, 1 - h, mode="full")
        k_shift = np.argmax(overlap_full) - (n_points - 1)
        max_overlap = overlap_full[np.argmax(overlap_full)] * dx
        
        # Compute gradient
        grad = compute_gradient(h, k_shift, n_points, dx)
        
        # Update h using gradient descent
        h_new = h - step_size * grad
        
        # Project h_new to feasible region
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_c5(h_new, dx)
        
        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update step size
        step_size = adaptive_step_size(prev_c5, current_c5, step_size)
        prev_c5 = current_c5
        
        # Update h
        h = h_new
        
        # Check if improvement is negligible
        if abs(prev_c5 - current_c5) < min_improvement:
            step_size = max(step_size * 0.9, 1e-6)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
