import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)

        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Meta-heuristic: simulated annealing with local search
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    def local_search(h_vec, max_iter=100):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Gradient approximation with small perturbations
            grad = np.zeros_like(h_vec)
            eps = 1e-5
            current_val = objective(h)
            for i in range(len(h_vec)):
                pert = np.zeros_like(h_vec)
                pert[i] = eps
                h_plus = h_vec + pert
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                val_plus = objective(h_plus)
                grad[i] = (val_plus - current_val) / eps
            # Update with gradient and project back
            step_size = 0.01
            h_vec -= step_size * grad
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec

    # Simulated annealing components
    T_initial = 1.0
    T_final = 1e-6
    alpha = 0.95
    n_iter_T = 100

    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    T = T_initial
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Local search at current temperature
        for _ in range(n_iter_T):
            # Generate neighbor
            neighbor_h = current_h.copy()
            idx = np.random.randint(n_points)
            neighbor_h[idx] += np.random.normal(0, 0.02)
            
            # Project neighbor to valid domain
            neighbor_h = np.clip(neighbor_h, 0, 1)
            neighbor_h = neighbor_h * (n_points / 2) / np.sum(neighbor_h)
            
            neighbor_c5 = evaluate_c5(neighbor_h, dx)
            
            # Acceptance probability
            if neighbor_c5 < current_c5:
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
            else:
                delta = neighbor_c5 - current_c5
                prob = np.exp(-delta / T)
                if np.random.rand() < prob:
                    current_h = neighbor_h.copy()
                    current_c5 = neighbor_c5
            
            # Update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:

        # Local refinement
        current_h = local_search(current_h)
        current_c5 = evaluate_c5(current_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Cool down temperature
        T *= alpha
        if T < T_final:
            break
        
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
