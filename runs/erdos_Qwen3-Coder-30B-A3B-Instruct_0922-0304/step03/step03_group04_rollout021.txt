import numpy as np
from scipy.signal import correlate
from pywt import wavedec, waverec
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet optimization parameters
    max_wavelet_iter = 20
    max_spatial_refine = 2
    lambda_thresh = 0.1
    learning_rate = 0.01
    wavelet_level = min(6, int(np.log2(n_points)) - 1)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Wavelet domain optimization
    for wavelet_iter in range(max_wavelet_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Wavelet decomposition
        coeffs = wavedec(h, 'haar', level=wavelet_level)
        # Threshold coefficients
        thresholded_coeffs = [soft_threshold(c, lambda_thresh) for c in coeffs]
        # Reconstruct
        h_wavelet = waverec(thresholded_coeffs, 'haar')[:n_points]
        h_wavelet = np.clip(h_wavelet, 0, 1)
        h_wavelet = h_wavelet * (n_points / 2) / np.sum(h_wavelet)
        
        # Spatial refinement
        for refine_iter in range(max_spatial_refine):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Gradient based refinement
            # Sample a small subset of points for gradient-based optimization
            indices = np.argsort(np.abs(h_wavelet))[::-1][:n_points//5]
            indices = np.sort(indices)
            
            # Update using gradient descent on subset
            h_refined = h_wavelet.copy()
            step_size = 0.005
            
            # Compute current overlap gradient via finite difference
            corr_current = correlate(h_refined, 1 - h_refined, mode="full")
            max_current = np.max(corr_current) * dx
            
            for i in indices[:10]:  # Update 10 points at most
                eps = 1e-4
                h_plus = h_refined.copy()
                h_plus[i] = np.clip(h_plus[i] + eps, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad = (c5_plus - max_current) / eps
                
                h_refined[i] -= step_size * grad
                h_refined[i] = np.clip(h_refined[i], 0, 1)
                
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            h_wavelet = h_refined.copy()
            
        h = h_wavelet.copy()
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Increase threshold if no improvement in recent iterations
        if wavelet_iter % 5 == 0 and wavelet_iter > 0:
            lambda_thresh *= 1.2

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
