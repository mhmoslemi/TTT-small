import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Create multiple initial perturbations
    num_starts = 5
    initial_hs = [h.copy()]
    for _ in range(num_starts - 1):
        h_pert = h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_hs.append(h_pert)

    # Multi-start optimization
    for h_start in initial_hs:
        h_current = h_start.copy()
        for iteration in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                h_current += np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Evaluate
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            # Early exit
            if best_c5 <= 0.3807:
                break
                
    # Basin hopping refinement
    h_current = best_h.copy()
    step_size = 0.1
    cooling_rate = 0.995
    max_basin_iter = 1000
    
    for iteration in range(max_basin_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = h_current + np.random.normal(0, step_size, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        # Local optimization
        result = minimize(
            objective, 
            h_pert, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        
        if result.success:
            h_candidate = result.x
        else:
            h_candidate = h_pert
            
        # Ensure feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()
            
        # Update step size (cooling)
        step_size *= cooling_rate
        
        # Early exit
        if best_c5 <= 0.3807:
            break

    # Final refinement with tighter tolerances
    if time.time() - start_time < budget_s - 1:
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            best_h = result.x
            
        # Final feasibility check
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
