import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is within bounds and sum equals n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover"""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_std=0.05):
    """Gaussian noise mutation"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, noise_std)
    return mutated

def local_optimization(h, n_points, dx, max_iter=50):
    """Refine using scipy's SLSQP"""
    def objective(h_vec):
        return evaluate_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-9}
        )
        if result.success:
            return repair_h(result.x, n_points)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        population.append(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        population.append(h0)
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of entire population
        fitnesses = []
        for h in population:
            fitness = evaluate_c5(h, dx)
            fitnesses.append(fitness)
        
        # Find best in current generation
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        current_best_h = population[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
        
        # Early stop if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Selection and reproduction
        new_population = []
        
        # Keep top 50% (elitism)
        elite_indices = np.argsort(fitnesses)[:population_size // 2]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Local search on top 10%
        num_top = population_size // 10
        top_indices = np.argsort(fitnesses)[:num_top]
        for idx in top_indices:
            refined = local_optimization(population[idx], n_points, dx)
            new_population[idx] = refined
        
        population = new_population[:population_size]
        
        # Periodic debug prints
        if generation % 100 == 0:
            print(f"Gen {generation}: Best C5 = {best_c5:.6f}")
    
    # Final step: refine best with local optimization
    if best_h is not None:
        best_h = local_optimization(best_h, n_points, dx, max_iter=100)
        best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
