import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure sum(h) = n_points / 2 and h_i ∈ [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    h_mut = h.copy()
    mask = np.random.rand(n_points) < mutation_rate
    # Gaussian perturbation
    h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
    # Random reset for some positions
    reset_mask = np.random.rand(n_points) < 0.1
    h_mut[reset_mask] = np.random.rand(np.sum(reset_mask))
    h_mut = repair_h(h_mut, n_points)
    return h_mut

def crossover_individuals(h1, h2, crossover_rate):
    """Perform crossover between two individuals"""
    if np.random.rand() > crossover_rate:
        return h1.copy(), h2.copy()
    
    # Uniform crossover
    mask = np.random.rand(len(h1)) < 0.5
    h1_child = np.where(mask, h1, h2)
    h2_child = np.where(mask, h2, h1)
    
    return h1_child, h2_child

def genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_generations, dx):
    """Genetic Algorithm for minimizing C5"""
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = [np.ones(n_points)]
    
    # Fill population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_solution = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        # Evaluate fitness (negative C5)
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)
            if c5 < best_c5:
                best_c5 = c5
                best_solution = h.copy()
        
        # Check termination
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Crossover and mutation
        new_population = []
        # Elitism
        elite_count = max(1, pop_size // 5)
        elite_indices = np.argsort(fitness)[-elite_count:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover_individuals(parent1, parent2, crossover_rate)
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            new_population.extend([child1, child2])
        
        # Trim population to exact size
        population = new_population[:pop_size]
        
        if generation % 20 == 0:
            print(f"Generation {generation}: Best C5 = {best_c5:.6f}")
    
    return best_solution, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_generations = 200
    
    # Run GA
    ga_solution, ga_c5 = genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_generations, dx)
    
    # Local refinement with gradient descent if time allows
    if time.time() - start_time < budget_s * 0.8:
        best_solution = ga_solution.copy()
        best_c5 = ga_c5
        h = ga_solution.copy()
        
        # Gradient descent
        step = 0.005
        max_iter = 1000
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient numerically
            grad = np.zeros_like(h)
            eps = 1e-4
            base_corr = correlate(h, 1 - h, mode="full")
            base_c5 = np.max(base_corr) * dx
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - base_c5) / eps
            
            # Update step
            h -= step * grad
            h = repair_h(h, n_points)
            
            # Evaluate
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = h.copy()
                
            if best_c5 <= 0.3807:
                break
        ga_solution = best_solution
        ga_c5 = best_c5
    
    return (ga_solution.tolist(), ga_c5, n_points)
