import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    swarm.append(h0)
    velocities.append(np.zeros(n_points))
    
    # Add other particles with triangular shapes
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        mid = n_points // 2
        for j in range(n_points):
            h[j] = 1 - 2 * abs(j - mid) / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Track global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    best_c5 = gbest_score
    best_h = gbest.copy()
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c_p * r1 * (pbest[i] - h) +
                 c_g * r2 * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            
            # Normalize to satisfy constraint
            h = h * (n_points / 2) / np.sum(h)
            
            # Store updated values
            swarm[i] = h
            velocities[i] = v
            
            # Evaluate
            score = evaluate_c5(h, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h.copy()
                pbest_scores[i] = score
                
                # Update global best if needed
                if score < gbest_score:
                    gbest = h.copy()
                    gbest_score = score
                    if gbest_score < best_c5:
                        best_c5 = gbest_score
                        best_h = gbest.copy()
                        
            # Early exit if target hit
            if best_c5 <= 0.3807:
                break
                
    return (best_h.tolist(), best_c5, n_points)
