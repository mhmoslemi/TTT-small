import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution levels to try
    resolutions = [10, 20, 40, 70, 100, 140, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Start with the best known construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
    else:
        initial_h = np.ones(10)
        initial_h[5:] = 0
        initial_h = np.clip(initial_h, 0, 1)
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        h = initial_h.copy()
        
        # Resize h to n_points
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 1, len(h)), np.linspace(0, 1, n_points), h)
        
        # Normalize to target sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Use Differential Evolution for global search
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=20,
            seed=seed,
            disp=False
        )
        
        if de_result.success:
            h = de_result.x.copy()
            # Ensure sum constraint is met
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points
                
            # Refine with SLSQP
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8},
                tol=1e-8
            )
            
            if result.success:
                h = result.x.copy()
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    best_n_points = n_points
            
            # If we're close to target, try more refinement
            if best_c5 <= 0.3807:
                break

        else:
            # Fallback optimization if DE failed
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x.copy()
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    best_n_points = n_points
        
        # Set next initial guess as current best
        if best_h is not None:
            initial_h = best_h.copy()
            initial_h = np.clip(initial_h, 0, 1)
            # Resize if needed for next resolution step
            if len(initial_h) < n_points:
                temp = np.interp(np.linspace(0, 1, len(initial_h)), np.linspace(0, 1, n_points), initial_h)
                initial_h = temp
            else:
                initial_h = initial_h[:n_points]
    
    # Final evaluation
    if best_h is None:
        best_h = np.ones(70)
        best_h[35:] = 0
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (70 / 2) / np.sum(best_h)
        best_n_points = 70
        best_c5 = evaluate_c5(best_h, 2.0 / 70)
    
    return (best_h.tolist(), best_c5, best_n_points)
