import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 5000

    # SDP Approach
    def sdp_solve(h0, n_points, dx):
        h = cp.Variable(n_points)
        C5 = cp.Variable()

        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2, C5 >= 0]
        
        # For each k, add constraint sum h[i] * (1 - h[i+k])
        for k in range(-n_points+1, n_points):
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0,...,n_points-1-k
                term = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                # h[i] * (1 - h[i+k]) for i = -k,...,n_points-1
                term = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
            constraints.append(C5 >= term * dx)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value, prob.value
        except:
            pass
        return h0, float('inf')

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # SDP optimization
        updated_h, c5 = sdp_solve(h, n_points, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = updated_h.copy()
        
        # If current solution satisfies target or close, just continue refining
        if best_c5 <= 0.3807:
            break
            
        h = updated_h
        
        # Local gradient descent to refine
        step = 0.002
        for _ in range(20):
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Periodically run SDP again to restart
        if iteration % 30 == 0:
            h, c5 = sdp_solve(h, n_points, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final evaluation
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
