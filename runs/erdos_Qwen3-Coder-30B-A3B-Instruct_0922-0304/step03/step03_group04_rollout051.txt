import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points_list = [10, 20, 40, 70, 100, 140, 200]
    dx_list = [2.0 / n_points for n_points in n_points_list]
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        n_points = len(h)
        dx = 2.0 / n_points
        h = h * (n_points / 2) / np.sum(h)
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx)
    else:
        # Fallback: random initialization
        n_points = 70
        dx = 2.0 / n_points
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx)
    
    # Iterate through resolutions
    for i, (n_points, dx) in enumerate(zip(n_points_list, dx_list)):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use previous solution as initialization for current resolution (up-sample if needed)
        if i > 0:
            # Up-sample previous solution
            prev_h = best_h.copy()
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(prev_h)), prev_h)
            # Ensure sum constraint
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = best_h.copy()
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
    
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # First, try Differential Evolution for global search
        try:
            de_result = differential_evolution(
                objective,
                bounds,
                maxiter=100,
                popsize=20,
                seed=seed + i,
                disp=False
            )
            if de_result.success:
                h = de_result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass

        # Then refine with SLSQP
        try:
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass

        # Final check on C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if we're close to target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, 2.0 / len(best_h))
    
    return (best_h.tolist(), final_c5, len(best_h))
