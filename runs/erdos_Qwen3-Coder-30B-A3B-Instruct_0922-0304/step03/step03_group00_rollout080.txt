import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Add triangular functions
    for i in range(n_particles - 1):
        h = np.zeros(n_points)
        mid = n_points // 2
        for j in range(n_points):
            h[j] = 1 - 2 * abs(j - mid) / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal best
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Track global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    best_c5 = pbest_scores[best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c_p * r_p * (pbest[i] - swarm[i]) + 
                           c_g * r_g * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Normalize sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate current fitness
            current_c5 = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if current_c5 < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = current_c5
                
                # Update global best
                if current_c5 < best_c5:
                    gbest = swarm[i].copy()
                    best_c5 = current_c5
                    
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Return best solution
    final_c5 = evaluate_c5(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
