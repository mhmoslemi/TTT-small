import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: alternating pattern
        h = np.ones(n_points)
        h[::2] = 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # PSO initialization
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # Initial constraint satisfaction
    for i in range(swarm_size):
        positions[i] = np.clip(positions[i], 0, 1)
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # PSO main loop
    pso_end_time = start_time + budget_s * 0.8
    
    for iteration in range(max_iter_pso):
        if time.time() > pso_end_time:
            break
            
        # Evaluate particles
        for i in range(swarm_size):
            pos = positions[i]
            score = evaluate_c5(pos, dx)
            
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = pos.copy()
                
            if score < global_best_score:
                global_best_score = score
                global_best_position = pos.copy()
                
        # Update positions and velocities
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                           social * r2 * (global_best_position - positions[i]))
            
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Update best if needed
            score = evaluate_c5(positions[i], dx)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = positions[i].copy()
                
            if score < global_best_score:
                global_best_score = score
                global_best_position = positions[i].copy()
                
    # SLSQP refinement
    if global_best_position is not None:
        h = global_best_position.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
            
        # Run SLSQP with multiple restarts
        max_iter_slsqp = 1000
        restarts = 3
        
        for restart in range(restarts):
            if time.time() > start_time + budget_s - 5:
                break
                
            # Local optimization
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': max_iter_slsqp//restarts, 'ftol': 1e-12}, 
                             callback=None)
            
            if result.success:
                h = result.x
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                if best_c5 <= 0.3807:
                    break
            else:
                # If optimization fails, continue with current best
                break
                
            # Perturb for next restart
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation with best if no improvement from SLSQP
    if best_c5 == float('inf'):
        final_c5 = evaluate_c5(best_h, dx)
    else:
        final_c5 = best_c5
    
    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
