import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init)
    
    # Add more diverse initial particles
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        if i % 3 == 0:
            # Triangular shape
            for j in range(n_points):
                h[j] = max(0, 1 - 2 * abs(j - n_points/2) / n_points)
        elif i % 3 == 1:
            # Step function
            h[:n_points//2] = 1
        else:
            # Random
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Initialize personal bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Main PSO loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] 
                           + c_p * r_p * (pbest[i] - swarm[i])
                           + c_g * r_g * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Enforce bounds
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Enforce constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            score = evaluate_c5(swarm[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = swarm[i].copy()
                    gbest_score = score
                    
        # Early termination
        if gbest_score <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
