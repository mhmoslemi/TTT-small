import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Initialize variables for pulse-based optimization
    M = 5  # Number of pulses
    pulse_heights = np.ones(M) * 0.5  # Initial heights
    pulse_positions = np.linspace(0.2, 1.8, M)  # Initial positions
    pulse_widths = np.ones(M) * 0.3  # Initial widths

    def pulse_to_h(pulse_heights, pulse_positions, pulse_widths):
        h = np.zeros(n_points)
        for i in range(M):
            # Determine indices for pulse
            left_idx = int(max(0, (pulse_positions[i] - pulse_widths[i]/2) / dx))
            right_idx = int(min(n_points, (pulse_positions[i] + pulse_widths[i]/2) / dx))
            if right_idx > left_idx:
                h[left_idx:right_idx] = pulse_heights[i]
        return h

    def project_pulse_params(pulse_heights, pulse_positions, pulse_widths):
        # Ensure heights in [0,1]
        pulse_heights = np.clip(pulse_heights, 0, 1)
        # Ensure positions in [0,2]
        pulse_positions = np.clip(pulse_positions, 0, 2)
        # Ensure widths positive
        pulse_widths = np.maximum(pulse_widths, 1e-6)
        # Renormalize
        total_area = np.sum(pulse_heights * pulse_widths)
        if total_area > 0:
            pulse_heights = pulse_heights * (n_points/2) / total_area
        return pulse_heights, pulse_positions, pulse_widths

    def optimize_pulse_heights(pulse_heights, pulse_positions, pulse_widths):
        # Assume fixed: positions, widths
        h_initial = pulse_to_h(pulse_heights, pulse_positions, pulse_widths)
        h_result = h_initial.copy()
        # Simple optimization by gradient descent on sum of overlaps
        # This is simplified - full optimization would be more complex
        step_size = 0.01
        max_iter = 20
        for _ in range(max_iter):
            # Compute overlaps for current h
            overlap = correlate(h_result, 1 - h_result, mode="full")
            max_idx = np.argmax(overlap)
            # Simple gradient ascent on constraint positions
            # Directly adjust heights according to impact
            for i in range(M):
                # Check impact of changing height of pulse i
                # We use a simplified approach by testing neighbor changes
                h_test = h_result.copy()
                h_test[int(max(0, (pulse_positions[i] - pulse_widths[i]/2) / dx)):
                       int(min(n_points, (pulse_positions[i] + pulse_widths[i]/2) / dx))] += step_size
                h_test = np.clip(h_test, 0, 1)
                overlap_test = correlate(h_test, 1 - h_test, mode="full")
                if np.max(overlap_test) * dx < np.max(overlap) * dx:
                    h_result = h_test
                else:
                    h_test[int(max(0, (pulse_positions[i] - pulse_widths[i]/2) / dx)):
                           int(min(n_points, (pulse_positions[i] + pulse_widths[i]/2) / dx))] -= 2*step_size
                    h_test = np.clip(h_test, 0, 1)
                    overlap_test = correlate(h_test, 1 - h_test, mode="full")
                    if np.max(overlap_test) * dx < np.max(overlap) * dx:
                        h_result = h_test
            # Project back to constraint
            pulse_heights = pulse_heights * (n_points/2) / np.sum(h_result)
            h_result = pulse_to_h(pulse_heights, pulse_positions, pulse_widths)
        return h_result
    
    # Entry point
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try pulse-based approach
        # Update pulse parameters
        h_pulse = pulse_to_h(pulse_heights, pulse_positions, pulse_widths)
        if np.sum(h_pulse) > 0:
            h_pulse = h_pulse * (n_points / 2) / np.sum(h_pulse)
            
        current_c5 = compute_c5(h_pulse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_pulse.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # At each iteration, update one component
        # 1. Optimize pulse heights (since positions and widths fixed)
        h_optimized = optimize_pulse_heights(pulse_heights, pulse_positions, pulse_widths)
        h_pulse = h_optimized
        
        # 2. Try to improve by adjusting positions
        # Compute overlaps for current configuration
        overlap = correlate(h_pulse, 1 - h_pulse, mode="full") 
        max_pos = np.argmax(overlap)
        # Move for a small perturbation in position
        epsilon = 0.01
        move_dir = 1 if (max_pos > len(overlap) // 2) else -1
        for i in range(M):
            pulse_positions[i] += move_dir * epsilon
            if pulse_positions[i] < 0:
                pulse_positions[i] = 0
            if pulse_positions[i] > 2:
                pulse_positions[i] = 2

        # 3. Update pulse heights
        pulse_heights = np.ones_like(pulse_heights)
        h_pulse = pulse_to_h(pulse_heights, pulse_positions, pulse_widths)
        h_pulse = h_pulse * (n_points / 2) / np.sum(h_pulse)
        
        h_optimized = optimize_pulse_heights(pulse_heights, pulse_positions, pulse_widths)
        h_pulse = h_optimized

        # Project everything back and normalize
        pulse_heights, pulse_positions, pulse_widths = project_pulse_params(pulse_heights, pulse_positions, pulse_widths)

        # Final update
        h_pulse = pulse_to_h(pulse_heights, pulse_positions, pulse_widths)
        if np.sum(h_pulse) > 0:
            h_pulse = h_pulse * (n_points / 2) / np.sum(h_pulse)
        
        current_c5 = compute_c5(h_pulse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_pulse.copy()
    
    return (best_h.tolist(), best_c5, n_points)
