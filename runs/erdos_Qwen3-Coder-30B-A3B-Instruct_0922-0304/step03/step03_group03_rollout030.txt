def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Hybrid approach: try local search first
        # Local refinement step using projected gradient
        step = 0.01
        for _ in range(50):
            corr = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * dx
            max_idx = np.argmax(corr)
            
            # Approximate gradient via finite difference
            grad = np.zeros_like(h)
            eps = 1e-4
            h_eval = h.copy()
            for i in range(n_points):
                h_plus = h_eval.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full") 
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Periodically check if one should try a different approach
            if (_ % 10 == 0):
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        # Periodically try a more global optimization approach
        if (iteration % 10 == 0) and iteration > 0:
            # Try to use a modified approach: simple projection with smarter steps
            # This avoids the heavy SDP/Worst-case constraint approach
            # Instead concentrate on an adaptive step gradient descent
            
            # Take a large step in the direction of improving overall overlap
            h_mod = h.copy()
            
            # Modify by simple directional momentum and re-init if needed
            corr = correlate(h_mod, 1 - h_mod, mode="full")
            max_overlap = np.max(corr) * dx
            
            # Try to reduce the maximum overlap
            proposal = h_mod.copy()
            for i in range(n_points):
                # Compute change if we relax this variable
                h_temp = h_mod.copy()
                h_temp[i] = max(0, h_mod[i] - 0.01)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                corr_temp = correlate(h_temp, 1 - h_temp, mode="full")
                c5_temp = np.max(corr_temp) * dx
                
                if c5_temp < current_c5:
                    proposal = h_temp.copy()
            
            h = proposal
            
            # Repair only non-negativity and scaling
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
