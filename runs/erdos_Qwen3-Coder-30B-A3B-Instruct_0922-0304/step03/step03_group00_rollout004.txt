import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current max overlap and corresponding shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Identify shift k that maximizes overlap
        # overlap is of length 2*n_points - 1, centered at n_points-1
        k = max_idx - (n_points - 1)
        
        # Formulate QP for this shift k
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
        
        # Add constraint for sum
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Add bounds
        constraints += [h_var >= 0, h_var <= 1]
        
        # For shift k, form the objective involving convolution
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i = 0 to n_points-k-1  
            # This term contributes to the overlap at shift k
            # So we minimize the sum of these products
            expr = 0
            for i in range(n_points - k):
                expr += h_var[i] * (1 - h_var[i + k]) * dx
        else:
            # k is negative - h[i+k] * (1 - h[i]) for i = -k to n_points-1
            expr = 0
            for i in range(-k, n_points):
                expr += h_var[i + k] * (1 - h_var[i]) * dx
                
        objective = cp.Minimize(expr)
        
        # Rebuild the full QP problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Ensure it's within bounds
                h_new = np.clip(h_new, 0, 1)
                # Re-normalize to satisfy integral constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If QP fails, use gradient or fallback approach
                step = 0.001
                iterations = 20
                h_temp = h.copy()                
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    k = max_idx - (n_points - 1)
                    # Compute gradient approx
                    grad = np.zeros_like(h_temp)
                    eps = 1e-5
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback gradient-like approach
            step = 0.001
            iterations = 20
            h_temp = h.copy()                
            for _ in range(iterations):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                k = max_idx - (n_points - 1)
                # Compute gradient approx
                grad = np.zeros_like(h_temp)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
