import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution optimization with increasing n_points
    resolutions = [10, 20, 40, 70, 100, 140, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h from initial_h_values if available
        if 'initial_h_values' in globals() and n_points == 70:
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Start with a simple step function
            h = np.ones(n_points)
            h[n_points//2:] = 0
            if n_points >= 20:
                # Add some noise for diversity
                h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Differential Evolution for global search
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        max_iter_de = min(100, max(20, 500 // n_points))
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Use DE with populations, then SLSQP refinement
        result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            seed=seed,
            maxiter=max_iter_de,
            popsize=20
        )
        
        if result.success:
            h = result.x
        else:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Refine with SLSQP
        max_iter_slsqp = min(200, max(20, 1000 // n_points))
        for _ in range(5):  # Multiple restarts
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter_slsqp, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_n_points = n_points
                
            if best_c5 <= 0.3807:
                break
                
        # If improvement stalls, perturb a bit and retry once
        if best_c5 <= 0.3807:
            break
            
        # Perturb solution slightly to avoid local minima
        if n_points < 200:
            noise_scale = 0.02
            h = h + np.random.normal(0, noise_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    if best_h is None:
        # Use a simple initialization if nothing was found
        n_points = 70
        dx = 2.0 / n_points
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        best_h = h
        best_c5 = evaluate_c5(h, dx)
        best_n_points = n_points
        
    return (best_h.tolist(), best_c5, best_n_points)
