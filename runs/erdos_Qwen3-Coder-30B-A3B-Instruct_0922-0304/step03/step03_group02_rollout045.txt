import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    noise_sigma = 0.05
    local_search_fraction = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Add random initial individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    # Function to repair a solution to satisfy constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Genetic algorithm
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for entire population
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitnesses.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Select top individuals for next generation
        elite_size = population_size // 2
        selected = population[:elite_size]
        
        # Local search on top few individuals
        local_search_count = int(elite_size * local_search_fraction)
        for i in range(local_search_count):
            h = selected[i].copy()
            h = repair(h)
            
            # Simple local optimization: gradient descent
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute gradient approximation
                direction = np.zeros_like(h)
                eps = 1e-6
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair(h_plus)
                    c5_plus = evaluate_c5(h_plus, dx)
                    direction[j] = (c5_plus - fitnesses[i]) / eps
                    
                # Update with gradient ascent
                h = h + 0.1 * direction
                h = repair(h)
                
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < fitnesses[i]:
                    fitnesses[i] = new_c5
                    selected[i] = h.copy()
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        while len(new_population) < population_size:
            # Tournament selection
            def tournament_select():
                indices = np.random.choice(len(selected), tournament_size)
                best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
                return selected[best_idx].copy()
            
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                offspring = [child1, child2]
            else:
                offspring = [parent1, parent2]
            
            # Mutation
            for child in offspring:
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.normal(0, noise_sigma)
                child = repair(child)
                new_population.append(child)
                
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # If no improvement, fallback to initial construction
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            final_c5 = evaluate_c5(h, dx)
            best_h = h
        else:
            # Fallback to random construction
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            final_c5 = evaluate_c5(h, dx)
            best_h = h
    
    return (best_h.tolist(), final_c5, n_points)
