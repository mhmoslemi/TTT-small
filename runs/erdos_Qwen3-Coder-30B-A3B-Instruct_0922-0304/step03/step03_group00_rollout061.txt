import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define the SDP model
    def solve_sdp(h0):
        # Variables
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1,
            C5 >= 0
        ]
        
        # Add overlap constraints
        # Compute full correlation with 1-h 
        for k in range(-n_points+1, n_points):
            # This is a quadratic constraint - we need to approximate or relax
            # For simplicity, we will do this approximately with feasible outer approximation
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0..n-k-1
                # Summing this gives the overlap
                overlap = 0
                for i in range(n_points - k):
                    overlap += h[i] * (1 - h[i + k])
                constraints.append(C5 >= overlap * dx)
            else:
                # h[i+k] * (1 - h[i]) for i = -k..n-1
                overlap = 0
                for i in range(-k, n_points):
                    overlap += h[i + k] * (1 - h[i])
                constraints.append(C5 >= overlap * dx)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                return h.value, prob.value
        except:
            pass
        return None, None

    # Coordinate ascent with gradient-based local refinement
    step_size = 0.01
    max_inner_iter = 100
    for iter in range(1000):
        if time.time() - start_time > budget_s - 1:
            break

        # Solve SDP relaxation
        h_new, c5_sdp = solve_sdp(h)
        if h_new is not None:
            h = h_new
            c5_new = c5_sdp 
        else:
            # Fallback: use numerical optimization
            # Simple projected gradient descent
            for inner_iter in range(max_inner_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                # Evaluate current overlaps
                corr = correlate(h, 1 - h, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx

                # Gradient (finite diff) estimation
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                # Update with step size
                h -= step_size * grad
                # Project back to feasible region
                h = np.clip(h, 0, 1)
                # Project to correct integral
                h = h * (n_points / 2) / np.sum(h)
            
            c5_new = np.max(correlate(h, 1 - h, mode="full")) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy().tolist()
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h, best_c5, n_points)
