import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum
from cvxpy import Semidef, trace, multiply

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # SDP-based approach
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp_sum(h_vars) == n_points / 2,
            t >= 0
        ]
        
        # Build quadratic constraints using SDP relaxation
        # We construct a semidefinite matrix relaxation
        # Z = [ I  H ]
        #     [ H^T  I ]
        # where H[i,j] = h[i]*(1-h[j]) but we don't use H directly
        # Instead, we use the fact that h[i]*(1-h[j]) <= h[i] + 1 - h[j] - h[i]*h[j]
        
        # For overlap constraint, we approximate the maximum
        # By considering the dual approach more simply:
        # Overlap is max_k sum_{i} h[i] * (1-h[i+k]) 
        # Use dot product approximation in a more robust way
        
        # Try a simpler approach: direct convex relaxation for key quantities
        # Approximate the h by shuffling and making constrained approximations
        
        # For each k, build constraint sum h[i] * (1 - h[i+k]) <= t defined over lags
        max_lag = n_points - 1
        overlap_constraints = []
        # Define the overlap sum easily via sliding window 
        for k in range(-max_lag, max_lag + 1):
            # Compute this overlap in a way that works in relaxed framework
            # k=0: 0 overlap
            if k == 0:
                continue
            elif k > 0:
                # This means sum over i from 0 to n-k (index >= i+k)
                # This is the classical case: h[i] * (1 - h[i+k])
                # We define a special constraint that approximates the maximum
                expr = cp_sum(cp.multiply(h_vars[0:(n_points-k)], 
                                        1 - h_vars[k:n_points]))
                constraints.append(expr <= t)
            else:
                # k < 0 becomes k = -l, so h[i] * (1 - h[i-l]) 
                l = -k
                expr = cp_sum(cp.multiply(h_vars[l:n_points], 
                                        1 - h_vars[0:(n_points-l)]))
                constraints.append(expr <= t)

        # Objective: minimize the max overlap bound
        prob = Problem(Minimize(t), constraints)
        
        try:
            # Portability with solvers
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status in [cp.OPTIMAL]:
                h_new = h_vars.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    # Repair the sum
                    h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback for rare solver failures: local optimization
            pass
            
        # Local refinement step as before, but more efficient gradient descent
        step = 0.005
        local_iters = 20
        for _ in range(local_iters):
            if time.time() - start_time > budget_s - 1:
                break
                
            corr = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * dx
            max_idx = np.argmax(corr)
            
            # Compute gradient more efficiently with finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            h_eval = h.copy()
            
            for i in range(n_points):
                h_plus = h_eval.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
