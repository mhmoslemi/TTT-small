import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_individual(h, rate=0.1, mutation_type='gaussian'):
    h_mut = h.copy()
    n_points = len(h)
    mask = np.random.random(n_points) < rate
    if mutation_type == 'gaussian':
        h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
    elif mutation_type == 'reset':
        h_mut[mask] = np.random.random(np.sum(mask))
    return repair_h(h_mut, np.sum(h))

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def ga_optimize(n_points, pop_size=50, generations=100, mutation_rate=0.1, 
                crossover_rate=0.7, elitism_rate=0.2, target_sum=1.0, dx=2.0/n_points, 
                budget_s=600, start_time=None):
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = [np.ones(n_points)]
    
    # Fill population
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = repair_h(h, target_sum)
        population.append(h)
    
    # Ensure each individual sums to target
    for i in range(len(population)):
        population[i] = repair_h(population[i], target_sum)
    
    best_fitness = float('inf')
    best_individual = None
    
    # Evolution loop
    for gen in range(generations):
        if start_time and time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for ind in population:
            fitness = evaluate_c5(ind, dx)
            fitness_scores.append(fitness)
        
        # Track best individual
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_fitness:
            best_fitness = fitness_scores[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Check termination condition
        if best_fitness <= 0.3807:
            break
            
        # Elitism
        elite_count = max(1, int(elitism_rate * pop_size))
        new_population = population[:elite_count]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection (tournament)
            tournament_size = 3
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            if np.random.random() < crossover_rate and len(population) > 1:
                parent2_idx = np.random.choice([i for i in range(pop_size) if i != winner_idx])
                parent2 = population[parent2_idx]
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate_individual(child1, mutation_rate)
                child2 = mutate_individual(child2, mutation_rate)
                new_population.extend([child1, child2])
            else:
                child = mutate_individual(parent1, mutation_rate)
                new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Ensure all individuals sum correctly
        for i in range(len(population)):
            population[i] = repair_h(population[i], target_sum)
    
    if best_individual is None:
        # Fallback to final population
        fitness_scores = [evaluate_c5(ind, dx) for ind in population]
        best_idx = np.argmin(fitness_scores)
        best_individual = population[best_idx]
        best_fitness = fitness_scores[best_idx]
    
    return best_individual, best_fitness

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

        n_points, 
        pop_size=50, 
        generations=100, 
        mutation_rate=0.1, 
        crossover_rate=0.7, 
        elitism_rate=0.2,
        target_sum=target_sum,
        dx=dx,
        budget_s=budget_s,
        start_time=start_time
    )
    
    # Local refinement using gradient descent if needed
    if time.time() - start_time < budget_s - 2:
        refined_h = ga_h.copy()
        step = 0.005
        max_iter = 100
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
            corr = correlate(refined_h, 1 - refined_h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(refined_h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = refined_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * target_sum / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - max_overlap) / eps
            
            refined_h -= step * grad
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * target_sum / np.sum(refined_h)
            
            current_c5 = evaluate_c5(refined_h, dx)
            if current_c5 < ga_c5:
                ga_h = refined_h.copy()
                ga_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_c5(ga_h, dx)
    
    return (ga_h.tolist(), final_c5, n_points)
