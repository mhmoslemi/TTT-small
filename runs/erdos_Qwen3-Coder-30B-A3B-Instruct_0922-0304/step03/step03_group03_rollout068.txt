import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Multiplicative updates with temperature and damping
    epsilon = 0.1
    damping = 0.99
    max_iter = 1000
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        corr = correlate(h, 1 - h, mode="full")
        S = corr * dx  # S_k values
        
        # Compute multiplicative factors for update
        exp_neg_S = np.exp(-S / epsilon)
        exp_pos_S = np.exp(S / epsilon)
        
        # Initialize update factors
        update_factor = np.ones(n_points)
        
        # Build update factor for each point
        for i in range(n_points):
            factor = 1.0
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    # Only process valid lag indices
                    idx = k + n_points  # shift to positive index for full correlation
                    factor *= exp_neg_S[idx] / (1 + exp_pos_S[idx])
            update_factor[i] = factor
            
        # Apply updates
        h = h * update_factor * damping
        
        # Repair feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate and track best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
