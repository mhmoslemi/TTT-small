import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start with different initializations
    initial_candidates = [h.copy()]
    
    # Add random perturbations
    for _ in range(4):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        initial_candidates.append(perturbed)

    # Optimize each initialization
    for i, init_h in enumerate(initial_candidates):
        h_current = init_h.copy()
        for iteration in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Local optimization with SLSQP
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
            else:
                h_current += np.random.normal(0, 0.01, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Evaluate and update best
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break

    # Basin hopping: perturb and optimize the best solution
    h_current = best_h.copy()
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random perturbation with decreasing scale
        scale = max(0.01, 0.1 * np.exp(-iteration / 500))
        perturbation = np.random.normal(0, scale, n_points)
        h_perturbed = h_current + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local optimization
        result = minimize(
            objective, 
            h_perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
        else:
            h_opt = h_perturbed
        
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Evaluate and update best
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        h_current = h_opt

    # Final refinement with tighter tolerance
    result = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if result.success:
        best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
