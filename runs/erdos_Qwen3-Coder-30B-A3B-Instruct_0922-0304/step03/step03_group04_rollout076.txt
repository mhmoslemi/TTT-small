import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to ensure constraints"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 10

    bounds = [(0, 1) for _ in range(n_points)]
    target_sum = n_points / 2
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return evaluate_c5(h, 2.0 / n_points)
    
    # Multi-resolution optimization
    resolutions = []
    current_n = 10
    while current_n <= 200:
        resolutions.append(current_n)
        current_n *= 2
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        h = np.ones(n_points)
        
        # Initialize from previous solution if available, otherwise from initial_h_values or random
        if res_idx == 0:

        else:
            # Upsample from previous solution
            prev_h = best_h if best_h is not None else np.ones(n_points)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(prev_h)), prev_h)
            h = project_h(h, n_points)
            
        # Differential Evolution for global search
        de_result = differential_evolution(

        )

            h = de_result.x
            h = project_h(h, n_points)
            
        # Refinement with SLSQP
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                h = project_h(h, n_points)

            else:
                # Perturb and restart
                h = h + np.random.normal(0, 0.01, n_points)
                h = project_h(h, n_points)
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, 2.0 / len(best_h))

        h_fallback = np.ones(n_points) * 0.5

        final_c5 = evaluate_c5(h_fallback, 2.0 / n_points)
