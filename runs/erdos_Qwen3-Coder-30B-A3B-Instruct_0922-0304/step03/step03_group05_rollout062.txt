import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_autocorr(h):
        # Compute full autocorrelation
        autocorr = correlate(h, h, mode="full")
        # Select lags from -n+1 to n-1, but focus on [0, 2*n-2] for shift k=0..n-1
        # We want to get shifts in [0, n-1] because they represent x+k where x in [0,2]
        # However, modes full gives the full range: -n+1 to n-1  
        # For our overlap integral it's: overlap[k] = \int h(x)(1-h(x+k)) 
        # We reverse the computation to be accurate.
        # Multiple by dx, we want to match original correlation definition with dx scaling.
        return autocorr

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    max_iter = 200  # Changed from 1000 to try more iterations within budget
    patience = 0  # Track for early stopping
    
    # Iteratively adjust h to improve min autocorrelation 
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0  # Reset patience if improvement
        else:
            patience += 1
            # Early exit if no improvement in 20 iterations
            if patience > 20:
                break
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute all autocorrelations and find the k with the smallest
        # The number of unique lags in "full" correlation mode is 2*n-1, and we inspect for
        # overlap values in [0, 2*dx, 3*dx, ..., (n-1)*dx] . 
        # So really we're interested in autocorr where indices correspond to 0 to n-1 shifts in full array
        # Let's denote i: 0 to 2*n-2
        # For shifting k = 0 to n-1 we want the corresponding autocorrelation value at shift k. 
        # In scipy correlate with mode full, the i=0 at 0 shift, i=1 ... n-1 at shift 1, ..., i=n at shift 0 of h folded
        # Shift on [0,...,n-1] maps to positions [n-1, n, ..., 2*n-2] in full array.
        # The max overlap is from i=n-1 to i=2*n-2: those are the k=0 to k=n-1 shifts that make sense for our integral.
        # But remember I'm talking about integrate h(x) * (1 - h(x+k))
        # This kind of overlap integral matches not the direct correlation function but 
        #    Corr(h, 1-h) = integrate h(x) * (1 - h(x+k)) = [integrate h(x) - integrate h(x)h(x+k)]
        # The result gives correlation of h, 1-h
        all_autocorr = correlate(h, 1-h, mode="full")
        # Convert correlation to overlap values at lags k=0,...,n-1
        # The effective shift k corresponds to index i = n-1+k in the full autocorrelation
        # For subarray i = n-1, ..., 2*n-2 
        # Aligned for k=0 to k=n-1 maps to indices n-1 to 2*n-2
        # The overlap integral is computed as:
        # When i ∈ [0, 2*n-2], the lag ∈ [i − n + 1, n−1] = [-n+1, ..., n-1], but it's still conv with h, 1-h
        # But simply we want overlap values from k = 0 to k = n-1. 
        # In scipy correlate(h, 1-h, mode="full"), i = 0..2n-2. 
        # These are lags: from 0 to n-1 in positive direction (0..n-1) when i = n-1 to 2n-2. So this يعني
        # We transpose: for lag k in [0, n-1], i = n-1+k. This maps k=0 to i=n-1, ..., k=n-1 to i=2*n-2  
        # Therefore linspace(index_start, index_end) = [n-1, n, ..., 2*n-2]
        overlap_vals = all_autocorr[n-1:2*n-1]  # These are the actual overlap integrals corresponding to shifts k=0 to n-1  
        overlap_vals = overlap_vals * dx  # Apply dx to get same units as integral
        min_overlap_idx = np.argmin(overlap_vals)
        # Min overlap occurs at lag = min_overlap_idx (0,1,...,n-1)  
        min_overlap_val = overlap_vals[min_overlap_idx]
        
        # Pick top 3 to 10 lags with worst overlaps (can be 10% of lags max)
        num_lags = max(3, min(10, n_points // 10))  # 3 to 10 lags or 10% of n
        top_indicators = np.argsort(overlap_vals)[:num_lags]  # indices with lowest overlaps
        
        # Now, compute total number of indices to consider
        # Compute the reduced problem to raise minimum value at top_indicators
        # Create indices for these bad lag indices in the full autocorr array
        relevant_indices = [n-1 + i for i in top_indicators]  
        # The unknowns for optimization are still h
        # Which can be done using direct optimization

        # Craft a convex relaxation or directional update
        # Alternatively, perturb h to soften a few localized problems
        # If current C5 is high (bad), let's refine only the problematic lags
        # Let's use the following heuristic:
        # Instead of full minimization each step, do projection based on least overlap
        # Try to adjust h to reduce the value of min_overlap_val

        # Instead of full optimization, try small perturbation in direction of improvement for specific lag
        # Try TrustConstr with box constraints / or ADMM-like idea

        # Form a subproblem minimizing max of top overlaps
        def subobjective(h):
            # Return the maximum of overlap values for the selected indices
            w = correlate(h, 1-h, mode="full")
            overlap_vals = w[n-1:2*n-1] * dx
            return np.max([overlap_vals[i] for i in top_indicators])
        
        # Define initial point is current h 
        # The search is still in feasible set: h sum = 35, h_i in [0,1]
        result = minimize(
            subobjective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        
        # If the optimization improves, accept the change
        if result.success:
            h_new = result.x
            # Ensure feasibility and update
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new 
        else:
            # Perturb with small noise if optimization fails
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
