import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try to improve with SDP approach
    try:
        # Construct SDP problem
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraint: sum(h) = n_points/2
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Constraints: h in [0,1]
        constraints.extend([h_var >= 0, h_var <= 1])
        
        # Constraint: C5 >= overlap for all k
        full_corr = np.zeros(2 * n_points - 1)
        for k in range(-n_points + 1, n_points):
            corr = np.zeros(2 * n_points - 1)
            if k >= 0:
                corr[k:k + n_points] = h_var[:n_points - k]
            else:
                corr[-k:] = h_var[-k:]
            full_corr += corr * (1 - h_var)
        
        # Use formulation using auxiliary variables for better matrix structure
        # Let's try a simpler approach: direct constraint using multiple inequalities
        
        # Create all k shifts explicitly and add inequality constraints
        for k in range(-n_points + 1, n_points):
            # This is the overlap for shift k: sum(h[i] * (1 - h[i+k]))
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0..n_points-k-1
                # Only add constraint if valid
                if n_points - k > 0:
                    idx1 = np.arange(n_points - k)
                    idx2 = np.arange(k, n_points)
                    overlap_expr = cp.sum(cp.multiply(h_var[idx1], 1 - h_var[idx2]))
                else:
                    overlap_expr = 0
            else:
                # h[i] * (1 - h[i+k]) for i = -k..n_points-1
                idx1 = np.arange(-k, n_points)
                idx2 = np.arange(n_points + k)
                overlap_expr = cp.sum(cp.multiply(h_var[idx1], 1 - h_var[idx2]))
            
            constraints.append(C5 >= overlap_expr)
            
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Solve SDP relaxation
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            best_h = np.array(h_var.value).flatten()
            best_c5 = prob.value
            # Ensure it is still in valid range
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            return (best_h.tolist(), best_c5, n_points)
    except Exception as e:
        pass
    
    # Fallback to a hybrid optimization strategy: gradient descent with restarts
    def evaluate_c5(h, dx):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def objective(h):
        return evaluate_c5(h, dx)
    
    # Use a SA-like cooling strategy with local search
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    best_c5 = current_c5
    best_h = current_h.copy()
    
    max_iter = 5000
    initial_temp = 0.1
    cooling_rate = 0.999
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Simulated Annealing step
        temp = initial_temp * (cooling_rate ** iteration)
        h_trial = current_h.copy()
        idx = np.random.randint(0, n_points)
        perturbation = np.random.normal(0, temp * 0.1)
        h_trial[idx] += perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        trial_c5 = evaluate_c5(h_trial, dx)
        if trial_c5 < current_c5 or np.random.rand() < np.exp(-(trial_c5 - current_c5) / temp):
            current_h = h_trial
            current_c5 = trial_c5
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
