import numpy as np
from scipy.signal import correlate
import time
import pywt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet transform setup
    wavelet = 'haar'
    max_level = int(np.log2(n_points)) - 1
    if max_level < 4:
        max_level = 4
    
    # Wavelet optimization
    for iter_wavelet in range(15):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute wavelet coefficients
        try:
            coeffs = pywt.wavedec(h, wavelet, level=max_level)
            # Reconstruct interpolating matrix (simplified)
            h_wavelet = coeffs[0][:n_points] if len(coeffs[0]) >= n_points else np.pad(coeffs[0], (0, n_points - len(coeffs[0])))
            for i in range(1, len(coeffs)):
                h_wavelet = np.pad(h_wavelet, (0, len(coeffs[i])), mode='constant')
                h_wavelet[:len(coeffs[i])] = coeffs[i]
                
            # Gradient estimation in wavelet domain
            # For simplicity, use a finite difference approximation
            eps = 1e-4
            grad = np.zeros_like(h_wavelet)
            h_temp = h.copy()
            c5_base = evaluate_c5(h, dx)
            
            # Approximate gradient through finite differences along coefficients
            for i in range(len(h_wavelet)):
                coeffs_plus = coeffs.copy()
                coeffs_plus[0] = h_wavelet.copy()
                coeffs_plus[0][i] += eps
                h_plus = pywt.waverec(coeffs_plus, wavelet)
                h_plus = h_plus[:n_points]
                # Clip to bounds
                h_plus = np.clip(h_plus, 0, 1)
                # Normalize
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5_base) / eps
            
            # Apply soft thresholding
            lambda_thresh = 0.1
            h_wavelet = soft_threshold(h_wavelet - 0.1 * grad, lambda_thresh)
            # Reconstruct h from updated coefficients
            h = pywt.waverec([h_wavelet[:len(coeffs[0])]] + coeffs[1:], wavelet)
            h = h[:n_points]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        except Exception:
            # fallback to simple gradient descent on h
            step = 0.002
            num_iter = 20
            for _ in range(num_iter):
                corr = correlate(h, 1 - h, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                h -= step * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

            break

    # Final refinement using local optimizer
    for _ in range(20):
        if time.time() - start_time > budget_s - 5:
            break
        step_size = 0.001
        num_iter = 50
        
        # Local gradient descent
        for _ in range(num_iter):
            corr = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * dx
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
