import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define the objective function for optimization
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        c5 = evaluate_c5(h_vec, dx)
        return c5
    
    # Define constraint function
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Optimization using scipy minimize
    constraints = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Main optimization loop
    max_iter = 500
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        if iter_count >= max_iter:
            break
            
        # Local optimization
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 10})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            pass
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Local refinement with gradient steps
        step_size = 0.01
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            
            # Compute gradient via finite difference
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - current_c5) / eps
                
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        iter_count += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
