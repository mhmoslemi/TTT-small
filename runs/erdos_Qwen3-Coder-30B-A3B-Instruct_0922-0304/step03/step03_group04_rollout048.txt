import numpy as np
from scipy.signal import correlate
from pywt import wavedec, waverec
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: smooth step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet configurations
    wavelet = 'haar'
    levels = 6
    max_wavelet_iter = 50
    max_spatial_refine = 10

    # Wavelet-based optimization
    for iter_count in range(max_wavelet_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Decompose h in wavelet basis
        try:
            coeffs = wavedec(h, wavelet, level=levels)
        except:
            break
            
        # Soft thresholding in wavelet domain
        threshold = 0.05
        new_coeffs = []
        for c in coeffs:
            # Apply soft thresholding
            new_c = np.sign(c) * np.maximum(np.abs(c) - threshold, 0)
            new_coeffs.append(new_c)
            
        # Reconstruct h
        try:
            h_wave = waverec(new_coeffs, wavelet)
            if len(h_wave) > n_points:
                h_wave = h_wave[:n_points]
            elif len(h_wave) < n_points:
                h_wave = np.pad(h_wave, (0, n_points - len(h_wave)), 'constant')
        except:
            # If reconstruction fails, use original h
            h_wave = h.copy()
            
        # Clip and normalize
        h_wave = np.clip(h_wave, 0, 1)
        h_wave = h_wave * (n_points / 2) / np.sum(h_wave)
        h = h_wave
        
        # Evaluate C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            # Slight perturbation to escape local minima
            noise = np.random.normal(0, 0.001, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Spatial refinement steps
        if iter_count % 5 == 0 and iter_count > 0:
            # Select most significant wavelet coefficients for spatial refinement
            all_coeffs = np.concatenate(coeffs)
            sorted_indices = np.argsort(np.abs(all_coeffs))[::-1]
            top_indices = sorted_indices[:max(10, len(sorted_indices)//5)]
            
            # Refine those positions using gradient descent
            h_temp = h.copy()
            for _ in range(20):
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Simple gradient approximation
                grad = np.zeros(n_points)
                eps = 1e-4
                for i in top_indices[:20]:  # Limit to top elements
                    if i >= n_points:
                        continue
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Update
                step = 0.01
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                if time.time() - start_time > budget_s - 5:
                    break
                    
            h = h_temp
            
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
