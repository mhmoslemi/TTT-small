import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Local refinement using conjugate gradient approach
        # Gradient approximation using finite differences
        eps = 1e-4
        step_size = 0.01
        
        # Compute current gradient
        grad = np.zeros_like(h)
        current_c5 = evaluate_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step with adaptive step size
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Alternate with simple geometric adjustment
        # Linear combination with save previous solution
        if iteration % 5 == 0:
            h = 0.9 * h + 0.1 * best_h
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
