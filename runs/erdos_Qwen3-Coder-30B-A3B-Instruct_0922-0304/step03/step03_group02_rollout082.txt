import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters and setup
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = repair_h(h, n_points)
    else:

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h):
        h_repaired = repair_h(h, n_points)
        c5 = evaluate_c5(h_repaired, dx)
        return c5
    
    # CMA-ES parameters
    popsize = 100
    sigma = 0.1
    mean = h.copy()
    
    # Initialize CMA-ES
    es = CMAEvolutionStrategy(mean, sigma, {'popsize': popsize})
    
    # Optimization loop
    max_iter = budget_s * 10  # Conservative estimate
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate new population
        X = es.ask()
        fitness = []
        
        # Evaluate all individuals
        for ind in X:
            fit = objective(ind)
            fitness.append(fit)
            
        # Tell CMA-ES about fitness
        es.tell(X, fitness)
        
        # Update best solution
        current_best = es.result.xbest
        current_best = repair_h(current_best, n_points)
        current_c5 = evaluate_c5(current_best, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()

    # Final validation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
