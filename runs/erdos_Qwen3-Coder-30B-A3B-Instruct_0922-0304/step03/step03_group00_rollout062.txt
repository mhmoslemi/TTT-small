        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Deep dive into SDP approach
    def compute_c5_from_h(h_values):
        h = np.array(h_values)
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Try different initialization strategies
    init_strategies = []
    
    if 'initial_h_values' in globals():
        init_strategies.append(('parent', np.array(initial_h_values)))
    
    # Add some structure-based strategies
    init_strategies.append(('uniform', np.ones(n_points) * 0.5))
    
    # Triangular
    tri = np.minimum(np.arange(n_points), np.arange(n_points, 0, -1))
    init_strategies.append(('triangular', tri / np.sum(tri) * (n_points / 2)))
    
    # Step function
    step = np.zeros(n_points)
    step[:n_points//2] = 1.0
    init_strategies.append(('step', step))

    base_sol = []
    
    for name, h_init in init_strategies:
        if time.time() - start_time > budget_s - 2:

        h_start = h_init.copy()
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Simple gradient descent using computed gradient of overlaps
        max_iter = 1000
        step_size = 0.005
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break

            corr = correlate(h_start, 1 - h_start, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx

            # Compute gradient using finite differences
            grad = np.zeros_like(h_start)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_start.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps

            # Gradient step
            h_start -= step_size * grad
            h_start = np.clip(h_start, 0, 1)

        c5 = compute_c5_from_h(h_start)
        if c5 < best_c5:

            best_h = h_start.copy()
            
        # Binary search in small range around current best
        if best_c5 <= 0.3807:
            break

    # Final evaluation  
    final_c5 = compute_c5_from_h(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
