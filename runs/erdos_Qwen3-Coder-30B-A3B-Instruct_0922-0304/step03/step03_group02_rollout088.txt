import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def tournament_selection(population, fitness, tournament_size=3):
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_std=0.05):
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, noise_std)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
        population.append(h)
    
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([repair_h(child1, n_points), repair_h(child2, n_points)])
        
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        # Fallback if no solution was found
        best_h = population[0] if population else np.ones(n_points) * 0.5
        final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
