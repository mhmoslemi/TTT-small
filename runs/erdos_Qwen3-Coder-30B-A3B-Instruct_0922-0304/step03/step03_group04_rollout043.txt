import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fft import fft, ifft

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    start_time = time.time()

    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = h.copy()
    best_h = np.clip(best_h, 0, 1)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Generate diverse initial samples using Latin Hypercube in frequency domain
    def generate_latin_hypercube_samples(n_samples):
        samples = []
        for _ in range(n_samples):
            h_sample = np.random.rand(n_points)
            h_sample = h_sample * (n_points / 2) / np.sum(h_sample)
            h_sample = np.clip(h_sample, 0, 1)
            samples.append(h_sample)
        return samples
    
    # Spectrally guided mutation
    def mutate_spectrally(h_seed):
        h_freq = fft(h_seed)
        
        # Randomly mask frequency components
        mask = np.random.rand(n_points) > 0.5
        h_freq_masked = h_freq * mask
        
        # Add noise in masked regions
        noise = np.random.normal(0, 0.05, n_points) + 1j * np.random.normal(0, 0.05, n_points)
        h_freq_masked += noise * (1 - mask)
        
        # Inverse FFT and project
        h_new = ifft(h_freq_masked).real
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Optimization steps
    mutated_samples = generate_latin_hypercube_samples(50)
    
    # Initialize with best from samples
    for h_sample in mutated_samples:
        current_c5 = evaluate_c5(h_sample, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_sample.copy()
    
    # Add a layer of random search in the domain
    max_random_iter = 1000
    random_iter_count = 0
    
    while time.time() - start_time < budget_s - 10 and random_iter_count < max_random_iter:
        h_candidate = mutate_spectrally(best_h)
        
        # Evaluate and accept if better
        current_c5 = evaluate_c5(h_candidate, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()
        
        # Early termination for target
        if best_c5 <= 0.3807:

        return evaluate_c5(h, dx)
    
    # Evaluate final
    final_c5 = evaluate_c5(best_h, dx)
