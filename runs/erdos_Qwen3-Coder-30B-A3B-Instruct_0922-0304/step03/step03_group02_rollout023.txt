import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.2
    tournament_size = 3
    noise_sigma = 0.05
    
    # Initialize population
    population = []
    
    # If initial_h_values available, use it as first individual
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Generate rest of population randomly
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    # Genetic algorithm loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for h in population:
            current_c5 = evaluate_c5(h, dx)
            fitness_scores.append(current_c5)
        
        # Track best individual
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Select parents using tournament selection
        selected_parents = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_parents.append(population[winner_idx].copy())
        
        # Create offspring through crossover and mutation
        new_population = []
        
        # Keep top 50% of population (elitism)
        sorted_indices = np.argsort(fitness_scores)
        for i in range(population_size // 2):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring to fill remaining slots
        while len(new_population) < population_size:
            # Select two parents
            parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
            parent1 = selected_parents[parent1_idx]
            parent2 = selected_parents[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mutation_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                    for idx in mutation_indices:
                        child[idx] += np.random.normal(0, noise_sigma)
                
                # Repair constraints
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
        
        # Refine top individuals using local search (gradient descent)
        top_individuals = sorted_indices[:population_size // 10]
        for idx in top_individuals:
            h = population[idx].copy()
            
            # Define constraint for sum(h) = n_points / 2
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            # Define bounds for each h[i]
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Define objective for scipy.optimize
            def objective(h):
                return evaluate_c5(h, dx)
            
            # Define constraint dictionary
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
            # Local optimization
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                population[idx] = h_opt.copy()
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
