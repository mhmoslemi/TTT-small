import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - balanced alternating pattern
        h = np.ones(n_points)
        h[::2] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points)
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # Constraint handling for PSO
    def clamp_and_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return x
    
    def evaluate_particle(particle):
        particle = clamp_and_normalize(particle)
        return evaluate_c5(particle, dx)
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Evaluate particle
            score = evaluate_particle(positions[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = positions[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = positions[i].copy()
                
            # Update best solution
            if score < best_c5:
                best_c5 = score
                best_h = positions[i].copy()
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (personal_best_positions[i] - positions[i]) + 
                            social * r2 * (global_best_position - positions[i]))
            
            positions[i] += velocities[i]
            
            # Keep positions within bounds and normalize
            positions[i] = clamp_and_normalize(positions[i])
    
    # SLSQP refinement
    if global_best_position is not None:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Use global best as starting point
        h_refine = global_best_position.copy()
        
        max_slsqp_iter = 200
        for iter_slsqp in range(max_slsqp_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(objective, h_refine, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 50, 'ftol': 1e-9})
            
            if result.success:
                h_refine = result.x
                current_c5 = objective(h_refine)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_refine.copy()
            else:
                # Local gradient descent if SLSQP fails
                step = 0.001
                iter_local = 20
                
                h_temp = h_refine.copy()
                for _ in range(iter_local):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = clamp_and_normalize(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = clamp_and_normalize(h_temp)
                
                h_refine = h_temp
                
            h_refine = clamp_and_normalize(h_refine)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
