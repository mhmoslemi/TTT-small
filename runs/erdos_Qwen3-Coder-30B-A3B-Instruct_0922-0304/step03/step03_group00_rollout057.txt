import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    swarm.append(h_init)
    velocities.append(np.zeros(n_points))
    
    # Add more diverse initial particles
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        # Create a triangular shaped function
        mid = n_points // 2
        for j in range(n_points):
            h[j] = max(0, 1 - 2 * abs(j - mid) / n_points)
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests and global best
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_values = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    best_c5 = gbest_value
    best_h = gbest.copy()
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c_p * r1 * (pbest[i] - h) + c_g * r2 * (gbest - h)
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            
            # Ensure the sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            new_c5 = evaluate_c5(h_new, dx)
            
            # Update personal best
            if new_c5 < pbest_values[i]:
                pbest[i] = h_new.copy()
                pbest_values[i] = new_c5
            
            # Update global best
            if new_c5 < gbest_value:
                gbest = h_new.copy()
                gbest_value = new_c5
                
                if gbest_value <= 0.3807:
                    break
            
            # Update swarm
            swarm[i] = h_new
            velocities[i] = v
            
            # Update best solution found so far
            if gbest_value < best_c5:
                best_c5 = gbest_value
                best_h = gbest.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
