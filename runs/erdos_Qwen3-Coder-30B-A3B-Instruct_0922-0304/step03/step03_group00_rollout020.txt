import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points, target_sum)
    else:
        # Create a symmetric triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / (mid - 1)
            h[n_points - 1 - i] = i / (mid - 1)
        h = project_h(h, n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization
    max_iter = 10000
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap to find maximum
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient analytically for the maximum overlap shift
        k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        # Gradient calculation for shift k
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += 1 - h[i + k]
            if 0 <= i - k < n_points:
                grad[i] += h[i - k] - 1
        
        # Update h using gradient descent
        h = h - step_size * grad
        
        # Project back to feasible region
        h = project_h(h, n_points, target_sum)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size adjustment
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.95
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
