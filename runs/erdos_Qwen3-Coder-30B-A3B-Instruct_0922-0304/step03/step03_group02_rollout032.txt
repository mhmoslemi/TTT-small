import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_solution(h, n_points):
    """Repair solution to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = repair_solution(h, n_points)
    else:
        # Fallback initialization - piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = repair_solution(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define objective function for CMA-ES
    def objective(h):
        h_repaired = repair_solution(h, n_points)
        return evaluate_c5(h_repaired, dx)
    
    # Set up CMA-ES
    opts = {
        'bounds': [0, 1],
        'verbose': -1,
        'popsize': 100,
        'seed': seed
    }
    
    # Use initial solution as mean
    mean = h.copy()
    es = CMAEvolutionStrategy(mean, 0.1, opts)
    
    # Optimization loop
    max_iter = max(1000, int(budget_s * 5))
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Ask for new population
        X = es.ask()
        fitness = []
        
        # Evaluate each solution
        for x in X:
            f = objective(x)
            fitness.append(f)
            
        # Tell fitness values
        es.tell(X, fitness)
        
        # Get best solution from current generation
        current_best_x = es.best.x
        current_best_h = repair_solution(current_best_x, n_points)
        current_c5 = evaluate_c5(current_best_h, dx)
        
        # Update global best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best_h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_h = repair_solution(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
