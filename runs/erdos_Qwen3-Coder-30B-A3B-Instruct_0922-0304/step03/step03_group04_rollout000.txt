import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution levels to try
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Initialize from parent if available
    current_h = None
    current_n_points = 0
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        target_sum = n_points / 2
        
        # Initialize from previous solution if available
        if current_h is not None and current_n_points < n_points:
            # Up-sample the previous solution
            h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, current_n_points), 
                                  current_h)
            h = h_upsampled
        elif 'initial_h_values' in globals():
            # Use the provided initial construction
            h = np.array(initial_h_values)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Generate a random initial guess
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        
        h = project_h(h, target_sum)
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Bounds for variables
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Stage 1: Global search using Differential Evolution
        de_result = differential_evolution(
            objective, 
            bounds, 
            constraints=cons,
            popsize=20,
            maxiter=50,
            seed=seed,
            disp=False
        )
        
        if de_result.success:
            h = de_result.x
            h = project_h(h, target_sum)
        else:
            # Fallback: local search
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            if result.success:
                h = result.x
                h = project_h(h, target_sum)
        
        # Stage 2: Refinement to improve local minimum
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
            h = project_h(h, target_sum)
            
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n_points = n_points
            
        if best_c5 <= 0.3807:
            break
            
        # Update current solution for next iteration
        current_h = h.copy()
        current_n_points = n_points
    
    return (best_h.tolist(), best_c5, best_n_points)
