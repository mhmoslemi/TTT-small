import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy sum constraint and bounds"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_h(h, rate=0.1, mutation_type='gaussian'):
    """Apply mutation to h"""
    h_mut = h.copy()
    if mutation_type == 'gaussian':
        mask = np.random.random(len(h)) < rate
        h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
    elif mutation_type == 'reset':
        mask = np.random.random(len(h)) < rate
        h_mut[mask] = np.random.random(np.sum(mask))
    return repair_h(h_mut, len(h))

def crossover_h(h1, h2, rate=0.7):
    """Perform crossover between two solutions"""
    if np.random.random() > rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1)-1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return repair_h(h1_new, len(h1)), repair_h(h2_new, len(h2))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    generations = 1000
    elite_size = pop_size // 5
    mutation_rate = 0.1
    crossover_rate = 0.7
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial population
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = mutate_h(h0, rate=0.2)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Stats for early stopping
    no_improvement_count = 0
    last_best = float('inf')
    
    # Genetic algorithm
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)  # Negative because we want to minimize
        
        # Sort by fitness
        idxs = np.argsort(fitness)[::-1]
        sorted_pop = [population[i] for i in idxs]
        sorted_fit = [fitness[i] for i in idxs]
        
        # Update best
        current_best = -sorted_fit[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = sorted_pop[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 50:
            break
            
        if best_c5 <= 0.3807:
            break
        
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(pop_size):
            candidates = np.random.choice(pop_size, tournament_size)
            winner = candidates[np.argmax([sorted_fit[i] for i in candidates])]
            selected.append(sorted_pop[winner])
        
        # Create new population
        new_pop = sorted_pop[:elite_size]  # Elitism
        
        # Fill rest with crossover and mutation
        while len(new_pop) < pop_size:
            parent1, parent2 = np.random.choice(pop_size, 2, replace=False)
            child1, child2 = crossover_h(selected[parent1], selected[parent2], 
                                       crossover_rate)
            child1 = mutate_h(child1, mutation_rate)
            child2 = mutate_h(child2, mutation_rate)
            
            new_pop.extend([child1, child2])
        
        # Trim to exact population size
        population = new_pop[:pop_size]
    
    # Local refinement with gradient descent
    if time.time() - start_time < budget_s - 1:
        # Use best GA solution as starting point
        h_refine = best_h.copy()
        step = 0.005
        
        # Run for a few iterations
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            
            corr = correlate(h_refine, 1 - h_refine, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Estimate gradient
            grad = np.zeros_like(h_refine)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Gradient descent step
            h_refine -= step * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        # Check if refinement improved
        refined_c5 = evaluate_c5(h_refine, dx)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = h_refine.copy()
    
    return (best_h.tolist(), best_c5, n_points)
