from scipy.signal import correlate
from scipy.optimize import minimize
import time

    start_time = time.time()

    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_autocorr(h):
        return correlate(h, h, mode="full")

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    max_iter = 1000
    patience = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # If no improvement for a while, perturb
        if patience > 20:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            continue

        auto_corr = compute_autocorr(h)
        # Focus on top 10% of minimal correlations

        def subproblem_objective(h_step):
            h_new = h + h_step
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return -compute_c5(h_new)
            
        # Use SLSQP with fewer iterations
        result = minimize(
            subproblem_objective, 
            np.zeros(n_points), 
            method='SLSQP', 
            bounds=[(-0.1, 0.1) for _ in range(n_points)],
            constraints=[], # Not using equality constraint directly in subproblem
            options={'maxiter': 30, 'ftol': 1e-6}
        )

            h_new = h + h_step

            h_new = h_new * (n_points / 2) / np.sum(h_new)

            if new_c5 < current_c5:

            # If optimization fails, use a small perturbation
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
