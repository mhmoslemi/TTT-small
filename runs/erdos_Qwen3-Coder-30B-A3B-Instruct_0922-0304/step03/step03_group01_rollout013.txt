import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy sum constraint"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    mutation_mask = np.random.random(n_points) < mutation_rate
    h_mut = h.copy()
    h_mut[mutation_mask] += np.random.normal(0, 0.05, np.sum(mutation_mask))
    return repair_h(h_mut, n_points)

def crossover_individuals(h1, h2, crossover_rate):
    """Perform crossover between two individuals"""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
        
    crossover_point = np.random.randint(1, len(h1) - 1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return h1_new, h2_new

def ga_optimize(seed, budget_s, n_points):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 1000
    elite_size = pop_size // 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points) * 0.5
        base_h[:n_points//3] = 0.8
        base_h[2*n_points//3:] = 0.2
    
    for _ in range(pop_size):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evolution loop
    best_c5 = float('inf')
    best_individual = None
    
    dx = 2.0 / n_points
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(-c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Track best
        if -fitness_scores[0] < best_c5:
            best_c5 = -fitness_scores[0]
            best_individual = population[0].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness_scores, 3)
            parent2 = tournament_selection(population, fitness_scores, 3)
            
            child1, child2 = crossover_individuals(parent1, parent2, crossover_rate)
            
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            
            offspring.extend([child1, child2])
            
        # Create new population
        new_population = elite + offspring[:pop_size - elite_size]
        population = new_population
        
    return best_individual, best_c5

def tournament_selection(population, fitness_scores, k):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmax([fitness_scores[i] for i in indices])]
    return population[best_idx]

def local_refine(h, dx, n_points):
    """Refine using gradient-based approach"""
    # Simple gradient descent with projections
    max_iter = 1000
    step_size = 0.002
    
    for _ in range(max_iter):
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Estimate gradient via finite difference
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        # Update
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping for small gradient
        if np.linalg.norm(grad) < 1e-6:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # GA optimization
    ga_h, ga_c5 = ga_optimize(seed, budget_s * 0.8, n_points)
    
    # Local refinement
    refined_h = local_refine(ga_h, dx, n_points)
    final_c5 = evaluate_c5(refined_h, dx)
    
    # Return the best found
    best_h = refined_h if final_c5 < ga_c5 else ga_h
    best_c5 = min(final_c5, ga_c5)
    
    return (best_h.tolist(), best_c5, n_points)
