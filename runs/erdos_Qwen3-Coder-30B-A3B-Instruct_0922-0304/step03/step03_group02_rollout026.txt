import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # number of particles
    max_iter = int(budget_s * 20)  # Rough estimate for iterations
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize positions and velocities for PSO
    positions = np.random.uniform(0, 1, (m, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (m, n_points))
    
    # Scale each particle to satisfy the constraint
    for i in range(m):
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # Personal best positions and their objective values
    pbest_positions = positions.copy()
    pbest_values = np.array([evaluate_c5(pos, dx) for pos in positions])
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) + 
                            c2 * r2 * (gbest_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Clip to bounds
            positions[i] = np.clip(positions[i], 0, 1)
            
            # Maintain sum constraint
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate
            current_c5 = evaluate_c5(positions[i], dx)
            
            # Update pbest
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = positions[i].copy()
            
            # Update gbest
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest_position = positions[i].copy()
            
            # Early exit if target achieved
            if gbest_value <= 0.3807:
                break
        
        # Early exit if target achieved
        if gbest_value <= 0.3807:
            break
    
    # Return best solution
    return (gbest_position.tolist(), gbest_value, n_points)
