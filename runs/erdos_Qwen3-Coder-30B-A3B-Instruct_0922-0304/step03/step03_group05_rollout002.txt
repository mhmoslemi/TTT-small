from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        if len(h) != n_points:
            h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Generate multiple initial points
    initial_points = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            h_parent = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_points.append(h_parent.copy())
    
    # Random perturbations
    for _ in range(4):
        h_pert = h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_points.append(h_pert.copy())
    
    # Fallback if no parent
    if len(initial_points) == 0:
        fallback_h = np.ones(n_points)
        fallback_h[n_points//2:] = 0
        fallback_h = np.clip(fallback_h, 0, 1)
        fallback_h = fallback_h * (n_points / 2) / np.sum(fallback_h)
        initial_points.append(fallback_h.copy())

    # Optimization loop
    max_iter = 1000
    iter_count = 0
    improvement_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate all initial points
        for i, h_current in enumerate(initial_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Make a copy and ensure feasibility
            h_local = h_current.copy()
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            
            # Local optimization
            result = minimize(
                objective, 
                h_local, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
            else:
                h_opt = h_local
            
            # Ensure feasibility
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Evaluate and compare
            current_c5 = compute_c5(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                improvement_count = 0
                last_improvement = iter_count
            else:
                improvement_count += 1
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        iter_count += 1
        
        # Break early if no improvement in a while
        if improvement_count > 100:
            break
            
        # Generate new random point for diversity
        h_new = np.random.rand(n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        initial_points.append(h_new)
        if len(initial_points) > 5:
            initial_points.pop(0)
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
