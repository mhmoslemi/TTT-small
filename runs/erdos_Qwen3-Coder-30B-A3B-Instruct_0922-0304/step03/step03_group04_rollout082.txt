```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use convex optimization approach with a piecewise constant representation
    # For better exploration, we'll start with 20 intervals
    M = 20
    # Create piecewise constant intervals over [0,2]
    intervals = np.linspace(0, 2, M + 1)
    widths = np.diff(intervals)
    
    # Use CVXPY to solve the convex relaxation
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints on h: 0 <= h_i <= 1
    constraints = [h_vars >= 0, h_vars <= 1]
    
    # Constraint on integral: sum(h_i * w_i) = 1
    integral_constraint = cp.sum(cp.multiply(h_vars, widths)) == 1
    constraints.append(integral_constraint)
    
    # Compute overlap constraints for each shift
    # For each shift k, we need to check how much overlaps occur
    # This involves computing overlaps between every interval and its shifted version
    max_shift = M  # Maximum shift we'll consider
    
    # Precompute overlaps for better efficiency
    # Generate potential shifts for overlaps
    shift_indices = np.arange(-M, M + 1)
    
    # For each possible shift k, compute the overlap between intervals
    # Create overlap matrix for each shift
    overlaps = []
    
    for k in shift_indices:
        # we compute for each pair of intervals how much they overlap
        # when one shifts by k * widths[0] 
        # since we are dealing with piecewise constant, this amounts to 
        # checking shifted interval indices
        row = []
        for i in range(M):
            if i + k >= 0 and i + k < M:
                # how much does interval i overlap with interval i+k after shift
                # This is proportional to the minimum of their widths,
                # but we need to handle the fact that the shift translates one interval
                # For interval i,  interval i+k after shift: we compute their intersection
                # Handle with a small grid on one interval to get slope of overlaps
                # manually approximating overlaps based on shifted boundaries
                i_start = intervals[i]
                i_end = intervals[i+1] 
                k_start = intervals[i+k]
                k_end = intervals[i+k+1]
                
                # Assume intervals aligned, compute sqrt of dot product effect due to similarity
                # Simplified overlap effect as a 'half' of their minimum width (the effective amount part)
                # actually, for linear relationship: they overlap if interval i and i+k have positive intersection
                # we just approximate the effect with computed overlap
                overlap_length = max(0, min(i_end, k_end) - max(i_start, k_start))
                # normalize by actual width of the first one, for average 'weight'
                row.append(overlap_length / widths[i] if widths[i] > 0 else 0)
            else:
                row.append(0)
        overlaps.append(row)
    
    overlaps = np.array(overlaps)
    
    # Saturation: Check how much they contribute based on elements
    # So that for k-th shift:
    # cost = sum(h[i] * (1 - h[i + k]) * overlap_k) <= C
    # This turns into affine constraints on h
    for k_idx, k in enumerate(shift_indices):
        if k_idx >= len(overlaps):
            continue
        # Get the overlap vector for this shift k_idx
        overlap_vec = overlaps[k_idx]  # This should be for valid i, elements represent overlap with i+k
        # But that's for overlaps when shifting, needs be directional
        # We treat as sum(h[i] * (1-h[i+k]) * overlap_vec[i]) <= C
        # But because shifting, when we say h[i] and h[i+k], we only add if i+k is valid for the index of h_vars
        # We do it differently to be efficient:
        # Let's iteratively check for applicable shifts
        
        # We loop over pair indices and compute their normalized contribution
        expr = 0
        for i in range(M):
            if i + k >= 0 and i + k < M:
                expr += h_vars[i] * (1 - h_vars[i + k]) * overlap_vec[i]
            elif i + k < 0 or i + k >= M:
                # they don't contribute properly
                continue        
        constraints.append(expr <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Since we're just testing the feasibility, we'll solve directly with ECOS 
    # which handles small to medium problems well
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_solution = h_vars.value
            # convert back to full samples
            h_full = np.zeros(n_points)
            # map each full point to an interval
            points = np.linspace(0, 2, n_points + 1)
            for i in range(n_points):
                # Find which interval point lies in
                idx = np.searchsorted(intervals, points[i], side="right") - 1
                if idx < 0:
                    idx = 0
                elif idx >= M:
                    idx = M-1
                h_full[i] = h_solution[idx]
            
            # enforce sum condition
            h_full = h_full * (n_points / 2) / np.sum(h_full)
            h_full = np.clip(h_full, 0, 1)
            
            # Evaluate C5
            current_c5 = evaluate_c5(h_full, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_full.copy()
                
            # Early return if good enough
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
    except Exception as e:
        pass

    # Fallback to local optimization if needed
    # Consider uniform intervals initially
    now = time.time()
    if now - start_time > budget_s - 10:
        return (best_h.tolist(), best_c5, n_points)
    
    # Reinitialize 
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    bounds = [(0, 1) for _ in range(n_points)]
    target_sum = n_points / 2
    
    def constraint_sum(hv):
        return np.sum(hv) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(hv):
        return evaluate_c5(hv, dx)
    
    # Use L-BFGS-B with constraint
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        try:
            result = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
            else:
                break  # Stop optimization on failure
        except:
            break

    return (best_h.tolist(), best_c5, n_points)
```
