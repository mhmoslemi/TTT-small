import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    step_size = 0.01
    stagnation_count = 0
    stagnation_threshold = 20
    decay_steps = 200
    decay_factor = 0.5
    
    # Gradient descent with projection
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_overlap_plus = np.max(corr_plus) * dx
            grad[i] = (max_overlap_plus - max_overlap) / eps
        
        # Descent step
        h_new = h - step_size * grad
        
        # Project onto [0,1]^n_points
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update h for next iteration
        h = h_new
        
        # Adaptive step size decay
        if iteration > decay_steps and iteration % decay_steps == 0:
            step_size *= decay_factor
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Stagnation check
        if stagnation_count > stagnation_threshold:
            # Try a small random perturbation
            h += np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
