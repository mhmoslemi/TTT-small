import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fft import fft, ifft
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def create_initial_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a smooth step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add a smooth transition
        transition_width = n_points // 10
        for i in range(transition_width):
            h[n_points//2 - transition_width + i] = i / transition_width
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    return h

def latin_hypercube_sample(n_points, n_samples):
    samples = []
    for i in range(n_samples):
        sample = np.random.random(n_points)
        # Shuffle within each dimension
        for j in range(n_points):
            sample[j] = (j + sample[j]) / n_points
        samples.append(sample)
    return np.array(samples)

def project_to_feasible(h, target_sum):
    # Project to [0,1] and renormalize
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return h

def spectral_inplace_mutation(h, n_points, target_sum):
    # Sample in frequency domain
    H = fft(h)
    # Add noise to random frequencies
    noise = np.random.normal(0, 0.05, n_points) + 1j * np.random.normal(0, 0.05, n_points)
    # Scale the noise based on magnitude
    mask = np.random.choice(n_points, size=n_points//2, replace=False)
    H[mask] += noise[mask]
    
    # Inverse transform
    h_new = ifft(H).real
    return project_to_feasible(h_new, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    h = create_initial_h(n_points)
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Spectral-based initialization
    if time.time() - start_time > budget_s - 50:
        return (best_h.tolist(), best_c5, n_points)
    
    # Generate candidates using modified sampling
    n_candidates = 50
    candidates = []
    for i in range(n_candidates):
        candidate = spectral_inplace_mutation(h, n_points, target_sum)
        candidates.append(candidate)
        if time.time() - start_time > budget_s - 50:
            break
    
    # Evaluate all candidates
    if len(candidates) > 0:
        candidate_c5s = []
        for c in candidates:
            try:
                c5_val = evaluate_c5(c, dx)
                candidate_c5s.append((c, c5_val))
            except:
                continue
        
        if len(candidate_c5s) > 0:
            # Sort and take top 10%
            candidate_c5s.sort(key=lambda x: x[1])
            top_candidates = [c[0] for c in candidate_c5s[:max(1, len(candidate_c5s)//10)]]
            
            # Refine top candidates using local search
            for candidate_h in top_candidates:
                if time.time() - start_time > budget_s - 30:
                    break
                    
                def objective(h):
                    return evaluate_c5(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                result = minimize(
                    objective,
                    candidate_h.copy(),
                    method='Nelder-Mead',
                    options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-6}
                )
                
                if result.success:
                    h_refined = result.x
                    h_refined = project_to_feasible(h_refined, target_sum)
                    refined_c5 = evaluate_c5(h_refined, dx)
                    if refined_c5 < best_c5:
                        best_c5 = refined_c5
                        best_h = h_refined.copy()
                
                if best_c5 <= 0.3807:
                    break
    
    # Final sweep if needed
    while time.time() - start_time < budget_s - 5:
        # Create generic mutated structure
        h_new = spectral_inplace_mutation(best_h, n_points, target_sum)
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
