import numpy as np
from scipy.signal import correlate
from pywt import wavedec, waverec
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet decomposition
    wavelet = 'haar'
    max_level = min(6, int(np.log2(n_points)))
    coeffs = wavedec(h, wavelet, level=max_level)
    theta = np.concatenate(coeffs)
    
    # Optimization parameters
    learning_rate = 0.01
    threshold = 0.1
    max_iter = 30
    spatial_refinement_iter = 2
    
    def compute_gradient(theta, h, dx):
        # Reconstruct h from theta
        coeffs_reconstructed = [theta[:len(coeffs[0])]]
        cum_len = len(coeffs[0])
        for i in range(1, len(coeffs)):
            coeffs_reconstructed.append(theta[cum_len:cum_len + len(coeffs[i])])
            cum_len += len(coeffs[i])
        h_recon = waverec(coeffs_reconstructed, wavelet)
        if len(h_recon) > n_points:
            h_recon = h_recon[:n_points]
        elif len(h_recon) < n_points:
            h_recon = np.pad(h_recon, (0, n_points - len(h_recon)), 'constant')
        
        # Compute gradient using finite differences
        grad = np.zeros_like(theta)
        eps = 1e-4
        for i in range(len(theta)):
            theta_plus = theta.copy()
            theta_plus[i] += eps
            coeffs_plus = [theta_plus[:len(coeffs[0])]]
            cum_len = len(coeffs[0])
            for j in range(1, len(coeffs)):
                coeffs_plus.append(theta_plus[cum_len:cum_len + len(coeffs[j])])
                cum_len += len(coeffs[j])
            h_plus = waverec(coeffs_plus, wavelet)
            if len(h_plus) > n_points:
                h_plus = h_plus[:n_points]
            elif len(h_plus) < n_points:
                h_plus = np.pad(h_plus, (0, n_points - len(h_plus)), 'constant')
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - evaluate_c5(h, dx)) / eps
            
        return grad
    
    # Main optimization loop
    iteration = 0
    no_improvement_count = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Wavelet domain optimization
        if iteration % 2 == 0:
            grad = compute_gradient(theta, h, dx)
            theta = theta - learning_rate * grad
            
            # Soft thresholding
            theta = soft_threshold(theta, threshold)
            
            # Reconstruct h
            coeffs_reconstructed = [theta[:len(coeffs[0])]]
            cum_len = len(coeffs[0])
            for i in range(1, len(coeffs)):
                coeffs_reconstructed.append(theta[cum_len:cum_len + len(coeffs[i])])
                cum_len += len(coeffs[i])
            h = waverec(coeffs_reconstructed, wavelet)
            if len(h) > n_points:
                h = h[:n_points]
            elif len(h) < n_points:
                h = np.pad(h, (0, n_points - len(h)), 'constant')
            
            # Clip and normalize
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

            coeffs = wavedec(h, wavelet, level=max_level)

                best_h = h.copy()

            else:
                no_improvement_count += 1

            # Select top 20% significant coefficients
            abs_theta = np.abs(theta)
            threshold_idx = np.argpartition(-abs_theta, int(0.2 * len(theta)))[:int(0.2 * len(theta))]
            theta[threshold_idx] = 0
            
            # Reconstruct
            coeffs_reconstructed = [theta[:len(coeffs[0])]]
            cum_len = len(coeffs[0])
            for i in range(1, len(coeffs)):
                coeffs_reconstructed.append(theta[cum_len:cum_len + len(coeffs[i])])
                cum_len += len(coeffs[i])
            h = waverec(coeffs_reconstructed, wavelet)
            if len(h) > n_points:
                h = h[:n_points]
            elif len(h) < n_points:
                h = np.pad(h, (0, n_points - len(h)), 'constant')
            
            # Clip and normalize
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Note : Instead of complex spatial refinement, just continue with wavelet domain transforms
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1

        # Adaptive threshold
        if no_improvement_count > 10:
            threshold *= 1.5
            no_improvement_count = 0
            
        iteration += 1

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
