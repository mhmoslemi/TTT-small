import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_erdos_solution(h):
    # Mimic the evaluation but using the provided function
    dx = 2.0 / len(h)
    return evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation with finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Line search
    def line_search(h, grad, max_iter=20):
        eta = 0.01
        alpha = 0.5
        for i in range(max_iter):
            h_new = h - eta * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < evaluate_c5(h, dx):
                return h_new, c5_new
            else:
                eta *= alpha
        return h, evaluate_c5(h, dx)
    
    # Optimization loop
    last_improvement = 0
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Gradient computation
        grad = compute_gradient(h)
        
        # Line search
        h, c5 = line_search(h, grad)
        
        # Projection to maintain constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturbation to escape local minima
        if last_improvement > 50:
            noise_magnitude = 1e-3 / np.sqrt(iteration + 1)
            noise = np.random.normal(0, noise_magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = 0

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
