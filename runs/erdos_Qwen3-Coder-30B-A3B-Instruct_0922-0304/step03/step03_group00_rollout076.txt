import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def optimize_for_shift(k):
        # Create QP problem for shift k
        h_var = cp.Variable(n_points)
        
        # Objective: minimize sum over valid indices of h[i] * (1 - h[i + k])
        # This handles the cross-correlation term for shift k
        objective = 0
        for i in range(n_points - abs(k)):
            if k >= 0:
                objective += h_var[i] * (1 - h_var[i + k])
            else:
                objective += h_var[i - k] * (1 - h_var[i])
        
        # The multiplier dx is implicit in the QP solution scale
        prob = cp.Problem(cp.Minimize(objective), [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ])
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            return h_var.value
        except:
            return None
    
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find shift with maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        # Convert index to shift
        k = max_idx - (n_points - 1)
        
        # Optimize for this shift
        h_new = optimize_for_shift(k)
        if h_new is not None:
            h = np.clip(h_new, 0, 1)
            # Renormalize to maintain integral constraint
            h = h * (n_points / 2) / np.sum(h)
        else:
            # If QP fails, perform gradient-based update
            eps = 1e-5
            grad = np.zeros_like(h)
            current_val = evaluate_c5(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                val_plus = evaluate_c5(h_plus, dx)
                grad[i] = (val_plus - current_val) / eps
                
            step = 0.01
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
