import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    gbest = None
    gbest_value = float('inf')
    
    for _ in range(swarm_size):
        # Initialize particle with corrected constraints
        particle = h.copy() + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        swarm.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        pbest_values.append(evaluate_c5(particle, dx))
        if pbest_values[-1] < gbest_value:
            gbest_value = pbest_values[-1]
            gbest = particle.copy()
    
    # PSO iterations
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            velocity = (inertia * velocity + 
                       cognitive * r1 * (pbest[i] - particle) + 
                       social * r2 * (gbest - particle))
            
            # Update position
            particle += velocity
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate
            current_value = evaluate_c5(particle, dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particle.copy()
                pbest_values[i] = current_value
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest = particle.copy()
            
            # Update swarm
            swarm[i] = particle.copy()
            velocities[i] = velocity.copy()
    
    # SLSQP Refinement
    if gbest is not None:
        # Optimization constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint: sum(h) = n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Optimize using SLSQP with best PSO solution
        result = minimize(objective, gbest, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            final_h = result.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
        else:
            best_h = gbest.copy()
            best_c5 = evaluate_c5(best_h, dx)
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
