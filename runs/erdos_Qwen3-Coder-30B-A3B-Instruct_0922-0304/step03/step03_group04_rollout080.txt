import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 30  # Number of intervals
    n_points = 2 * M  # Discretization points
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Rescale to satisfy sum constraint
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Compute interval averages
        h = np.zeros(M)
        for i in range(M):
            start_idx = i * 2
            end_idx = (i + 1) * 2
            h[i] = np.mean(h_init[start_idx:end_idx])
    else:
        # Fallback initialization
        h = np.ones(M) * 0.5

    # Define interval widths
    widths = np.ones(M) * (2.0 / M)
    
    # Precompute overlap matrix
    def compute_overlap_matrix(M, widths):
        # For each pair of intervals, compute overlap for various shifts
        # We will compute C5 by constraint matrix method
        O = np.zeros((M, M))
        for i in range(M):
            for j in range(M):
                start_i, end_i = i * widths[0], (i+1) * widths[0]
                start_j, end_j = j * widths[0], (j+1) * widths[0]
                overlap = max(0, min(end_i, end_j) - max(start_i, start_j))
                O[i, j] = overlap
        return O

    best_c5 = float('inf')
    best_h = h.copy()
    
    # LP approach
    while time.time() - start_time < budget_s - 5:
        # Define variables
        h_vars = cp.Variable(M)
        C = cp.Variable()
        
        # Build constraints
        constraints = [
            cp.sum(cp.multiply(h_vars, widths)) == 1,
            h_vars >= 0,
            h_vars <= 1,
            C >= 0
        ]
        
        # Compute all possible overlaps for shift constraints
        # Modeled: h_i * (1 - h_j) * overlap_ij <= C for all relevant i,j
        # For each shift k, we compute for all i: h_i * (1 - h_{i+k}) 
        shifts = []
        shift_constraints = []
        
        # Compute overlap contribution matrices: 
        # Each entry [i][j] corresponds to contribution of h_i*(1-h_j) when shifted k
        # But with alignment: for shift k, h_i contributes to overlap with h_{i+k}
        # i from 0 to M-1, j from 0 to M-1, constraints of the form 
        # sum_{i,j; j = i+k} h_i * (1 - h_j) * overlap_ij <= C
        
        max_shift = M - 1
        # Precompute state for overlap lookup
        overlap_contributions = {}
        for k in range(-max_shift, max_shift + 1):
            overlap_contributions[k] = np.zeros((M, M))

        # For each shift, compute overlap contributions for all pairs (i,j) where j = i + k
        for k in range(-max_shift, max_shift + 1):
            contrib = np.zeros((M, M))
            for i in range(M):
                j = i + k
                if 0 <= j < M:
                    # This computes overlap of interval i with interval i+k
                    # But in our case, we're looking just at the shifted contribution
                    # The total overlap integral involves i from [0, M-1], j = i+k
                    # so look at 1 - h[j], and h[i]
                    contrib[i, j] = 1  # Use 1 for simplicity to indicate boundary
                else:
                    contrib[i, j] = 0
            overlap_contributions[k] = contrib
        
        # Now express each shift constraint as: sum_{i, j = i+k} h_i * (1-h_j) * overlap_ij <= C
        # Overlap_ij = max(0, width - |i_width - j_width|) up to a 1D notion
        # But in the LP approach, treat it as: take matrix of all h_i * (1-h_j) * constrictio_n 
        # over all valid i,j pairs. We simplify and set it up more precisely:
        # We are computing max_{t=0 to M-1} sum_{i=0 to M-1} h_i * (1-h_{i+t}) * overlap_i^t
        # Instead of this, but more practical:
        # The fundamental constraint is for all valid k:
        # sum_{i} h_i * (1-h_{i+k}) * o_ij <= C
        # where o_ij = 1 if they overlap at shift_k location, o_ij = 0 otherwise.
        # But we still need to compute social importances
        
        # BUILDING MONO LINEAR APPROACH:
        # Instead of using all correlation tricks, we just optimize directly via a simpler LP:
        # Maybe go beyond this and just optimize via full convolution similarly to the gradient descent
        # but waste no space or time in necessarily invalid top implementations.
        
        # A simpler and more direct approach in the less-structured settings:
        # Recover approach from original working code, correct small key bug.
        h_optimized = h.copy()
        
        # Convolution basis method - make larger but workable version:
        # Convert to 2 * M discretized points from h_j
        h_disc = np.zeros(n_points)
        for i in range(M):
            start = 2*i
            end = 2*(i+1)
            h_disc[start:end] = h_optimized[i]
            
        # Ensure constraint:
        h_disc = h_disc * (n_points / 2) / np.sum(h_disc)
        
        # Evaluate
        c5_val = np.max(np.correlate(h_disc, 1 - h_disc, mode="full")) * dx
        
        # Optimization via small perturbations on first solution
        stepsize = 0.001
        iter_total = 0
        max_iters = 2000
        
        while iter_total < max_iters and time.time() - start_time < budget_s - 5:
            h_current = h_disc.copy()
            # Correlation
            corr = np.correlate(h_current, 1 - h_current, mode="full")
            max_corr = np.max(corr)
            max_idx = np.argmax(corr)
            
            # Gradient approximation from finite difference
            h_grad = np.zeros_like(h_current)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] = min(1, h_plus[i] + eps)
                h_plus[i] = max(0, h_plus[i])
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                h_grad[i] = (c5_plus - c5_val) / eps
            
            # Update
            h_current -= stepsize * h_grad
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Compute new value
            corr = np.correlate(h_current, 1 - h_current, mode="full")
            c5_new = np.max(corr) * dx
            
            if c5_new < c5_val:
                h_disc = h_current.copy()
                c5_val = c5_new
                
            iter_total += 1
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_disc.copy()
                
            if c5_val <= 0.3807:
                break
                
    # Final check on best solution using standard measure
    best_corr = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(best_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
