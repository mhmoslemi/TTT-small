import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply Gaussian mutation with repair"""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
    if np.random.rand() < mutation_rate:
        # Random reset
        h = np.random.rand(n_points)
    return repair_h(h, n_points)

def crossover_individuals(h1, h2, crossover_rate):
    """Single-point crossover"""
    if np.random.rand() < crossover_rate:
        point = np.random.randint(1, len(h1))
        child1 = np.concatenate([h1[:point], h2[point:]])
        child2 = np.concatenate([h2[:point], h1[point:]])
        return child1, child2
    return h1.copy(), h2.copy()

def genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_generations, dx, seed=42):
    np.random.seed(seed)
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness_scores.append(-c5)  # Negative for minimization
        
        # Select top individuals
        selected_indices = np.argsort(fitness_scores)[-pop_size//2:]
        selected_pop = [population[i] for i in selected_indices]
        
        # Elitism
        new_population = selected_pop[:pop_size//5]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1 = np.random.choice(selected_pop)
            parent2 = np.random.choice(selected_pop)
            child1, child2 = crossover_individuals(parent1, parent2, crossover_rate)
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        
        # Track best
        current_best_idx = np.argmax(fitness_scores)
        current_best_h = population[current_best_idx]
        current_best_c5 = -fitness_scores[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
    
    return best_h, best_c5

def gradient_refinement(h, dx, n_points):
    """Local gradient-based refinement"""
    max_iter = 100
    step = 0.005
    
    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        h -= step * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_generations = 100
    
    # GA: 80% of time
    ga_time = budget_s * 0.8
    remaining_time = ga_time
    
    best_h, best_c5 = genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, 
                                        int(max_generations * (remaining_time / ga_time)), dx, seed)
    
    if time.time() - start_time > remaining_time:
        return (best_h.tolist(), best_c5, n_points)

    # Refinement: 20% of time
    refinement_time = budget_s - (time.time() - start_time)
    if refinement_time > 1:
        best_h = gradient_refinement(best_h, dx, n_points)
        best_c5 = evaluate_c5(best_h, dx)
        
        # Run one more optimization step if continued time available
        if time.time() - start_time < budget_s - 1:
            # Try a local search using gradient descent
            best_h = gradient_refinement(best_h, dx, n_points)
            best_c5 = evaluate_c5(best_h, dx)
    
    # Final check
    if not np.isfinite(best_c5) or np.isnan(best_c5) or best_c5 > 1.0:
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
        best_c5 = evaluate_c5(h, dx)
        best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
