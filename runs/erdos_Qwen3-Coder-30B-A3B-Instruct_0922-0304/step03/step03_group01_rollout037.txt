import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    h_mut = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Either add noise or reset
            if np.random.random() < 0.5:
                h_mut[i] += np.random.normal(0, 0.05)
            else:
                h_mut[i] = np.random.random()
    return repair_h(h_mut, n_points)

def crossover(parent1, parent2):
    """Blend two parents to create offspring"""
    alpha = np.random.random()
    offspring = alpha * parent1 + (1 - alpha) * parent2
    return offspring

def genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_gen, dx, start_time, budget_s):
    """Genetic Algorithm for minimizing C5"""
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population = [repair_h(h, n_points) for h in population]
        while len(population) < pop_size:
            h = np.random.random(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    else:
        population = []
        for _ in range(pop_size):
            h = np.random.random(n_points)
            h = repair_h(h, n_points)
            population.append(h)
    
    best_c5 = float('inf')
    best_h = None

    for generation in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (negative C5)
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitnesses.append(-c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)[::-1]
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if -fitnesses[0] < best_c5:
            best_c5 = -fitnesses[0]
            best_h = population[0].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection
        elite_size = max(1, pop_size // 5)
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_size:
            if np.random.random() < crossover_rate:
                # Crossover
                p1, p2 = np.random.choice(elite, 2, replace=False)
                child = crossover(p1, p2)
                child = repair_h(child, n_points)
            else:
                # Mutation
                parent = np.random.choice(elite)
                child = mutate_individual(parent, mutation_rate, n_points)
                
            offspring.append(child)
            
        # Create new population
        population = elite + offspring
        
        # Mutate a few more individuals
        for i in range(len(population)):
            if np.random.random() < mutation_rate * 0.5:
                population[i] = mutate_individual(population[i], mutation_rate, n_points)
    
    return best_h, best_c5

def local_refinement(h, dx, max_iter=200):
    """Local gradient descent refinement"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for _ in range(max_iter):
        # Compute gradient
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        
        # Approximate gradient via finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-4
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap * dx) / eps
        
        # Gradient step
        step_size = 0.1
        best_h -= step_size * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
        
        # Evaluate improvement
        new_c5 = evaluate_c5(best_h, dx)
        if new_c5 >= best_c5:
            break
        best_c5 = new_c5
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_gen = 100
    
    # Run GA
    ga_h, ga_c5 = genetic_algorithm(n_points, pop_size, mutation_rate, crossover_rate, max_gen, dx, start_time, budget_s)
    
    # Local refinement
    refined_h, refined_c5 = local_refinement(ga_h, dx, max_iter=200)
    
    # Final evaluation
    final_h = refined_h
    final_c5 = evaluate_c5(final_h, dx)
    
    # Return best solution found
    return (final_h.tolist(), final_c5, n_points)
