```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def soft_threshold(x, threshold):
    return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Haar wavelet transform
    wavelet = 'haar'
    max_level = int(np.log2(n_points)) - 1
    if max_level < 3:
        max_level = 3
    wavelet_coeffs = pywt.wavedecn(h, wavelet, level=max_level)
    
    # Flatten coefficients
    coeff_list = pywt.coeffs_to_array(wavelet_coeffs)[0]
    flat_coeffs = coeff_list.flatten()
    original_shape = flat_coeffs.shape
    
    # Wavelet optimization loop
    for iteration in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Reconstruct h from coefficients
        coeffs = pywt.array_to_coeffs(flat_coeffs.reshape(original_shape), wavelet_coeffs, output_format='wavedecn')
        h = pywt.waverecn(coeffs, wavelet).real
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(flat_coeffs)
        for i in range(len(flat_coeffs)):
            flat_coeffs_plus = flat_coeffs.copy()
            flat_coeffs_plus[i] += eps
            coeffs_plus = pywt.array_to_coeffs(flat_coeffs_plus.reshape(original_shape), wavelet_coeffs, output_format='wavedecn')
            h_plus = pywt.waverecn(coeffs_plus, wavelet).real
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update coefficients with soft thresholding
        step_size = 0.1
        threshold = 0.05
        flat_coeffs = soft_threshold(flat_coeffs - step_size * grad, threshold)
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
