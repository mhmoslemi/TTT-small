import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop using QP-based approach
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find the shift that produces maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (len(h) - 1)  # Convert to shift index
        
        # Set up QP for this shift
        h_var = cp.Variable(n_points)
        
        # Objective: minimize the overlap at this shift
        # For shift k, we want to minimize sum_i h[i] * (1 - h[i + k])
        # where the sum runs over valid indices.
        
        # Create quadratic term matrix Q and linear term vector c
        Q = np.zeros((n_points, n_points))
        c = np.zeros(n_points)
        
        # Build quadratic matrix and linear vector for the specific shift
        if max_shift >= 0:
            # Positive shift: h[i] * (1 - h[i + shift])
            for i in range(n_points - max_shift):
                Q[i, i + max_shift] = -1.0
                Q[i + max_shift, i] = -1.0
                c[i] = 1.0  # Linear term
        else:
            # Negative shift: h[i] * (1 - h[i - shift])
            for i in range(-max_shift, n_points):
                Q[i, i + max_shift] = -1.0
                Q[i + max_shift, i] = -1.0
                c[i] = 1.0  # Linear term
        
        # QP problem
        objective = cp.Minimize(cp.quad_form(h_var, Q) + c.T @ h_var)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve QP problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h = h_new.copy()
            else:
                # If QP fails, do a simple gradient descent
                step = 0.005
                iterations = 200
                h_temp = h.copy()
                
                for _ in range(iterations):
                    # Compute overlaps
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    # Compute gradient
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    # Update h
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                h = h_temp
        except:
            # Fallback: gradient descent
            step = 0.005
            iterations = 200
            h_temp = h.copy()
            
            for _ in range(iterations):
                # Compute overlaps
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                # Update h
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            h = h_temp
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
