import numpy as np
from scipy.signal import correlate
import time
from cma import CMAEvolutionStrategy
import math

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def objective_cma(h, n_points, dx):
    """Objective function for CMA-ES optimization"""
    h = repair_h(h, n_points)
    return evaluate_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:

        h0 = np.ones(n_points) * 0.5
    
    # Set up CMA-ES
    es = CMAEvolutionStrategy(h0, 0.1, {'popsize': 100, 'seed': seed})
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Run optimization
    max_iter = int(budget_s * 10)  # Roughly 10 evaluations per second
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get a batch of candidate solutions
        solutions = es.ask()
        fitnesses = []
        
        for sol in solutions:
            # Repair and evaluate
            repaired_sol = repair_h(sol, n_points)
            fitness = evaluate_c5(repaired_sol, dx)
            fitnesses.append(fitness)
            
            # Update best solution
            if fitness < best_c5:
                best_c5 = fitness
                best_h = repaired_sol.copy()
                
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Tell CMA-ES about the fitnesses
        es.tell(solutions, fitnesses)
        
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
