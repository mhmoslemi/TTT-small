import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_overlap_at_shift(h, shift):
        # Compute overlap for a specific shift
        n = len(h)
        if shift >= n:
            shift = shift % n
        result = 0.0
        for i in range(n):
            j = (i + shift) % n
            result += h[i] * (1 - h[j])
        return result * dx

    def identify_top_contributors(h, max_shift, num_top=5):
        # Find top contributors to current maximum overlap
        n = len(h)
        max_overlap = -1
        best_shift = 0
        
        # Check all possible shifts
        for shift in range(n):
            overlap_val = compute_overlap_at_shift(h, shift)
            if overlap_val > max_overlap:
                max_overlap = overlap_val
                best_shift = shift
        
        # Now find indices with highest product h[i] * (1 - h[i+shift])
        contributions = []
        for i in range(n):
            j = (i + best_shift) % n
            contrib = h[i] * (1 - h[j])
            contributions.append((contrib, i, j))
        
        contributions.sort(reverse=True)
        return [idx for _, idx, _ in contributions[:num_top]]

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100

    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved  
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Reduce step size if no improvement
            max_move *= 0.95
            no_improvement_count = 0
            
        # Find top contributors
        top_indices = identify_top_contributors(h, n_points)
        
        # Adjust values
        for i in top_indices:
            # Find the corresponding j for this shift
            n = len(h)
            # We'll adjust h[i] and h[j] (next in the overlap for max shift)
            # It's better to be more precise but for simplicity just adjust
            # we can simulate adjustment by going through the computation
            
            # In each iteration, we'll do a more sophisticated local search
            # Try to decrease h[i] and increase h[j] where j = (i + shift) % n
            shift = 0  # Find the shift that produced max overlap
            max_overlap = -1
            for s in range(n):
                overlap_val = compute_overlap_at_shift(h, s)
                if overlap_val > max_overlap:
                    max_overlap = overlap_val
                    shift = s
                    
            j = (i + shift) % n
            
            epsilon = min(max_move, h[i], 1 - h[j])
            if epsilon > 1e-8:
                h[i] -= epsilon
                h[j] += epsilon
                
                # Ensure sum constraint
                current_sum = np.sum(h)
                diff = current_sum - n_points / 2
                if abs(diff) > 1e-8:
                    # Adjust all other entries to make sum correct
                    # Distribute evenly for simplicity
                    adjustment = diff / n
                    h += adjustment
                    h = np.clip(h, 0, 1)
        
        # Normalize sum before next iteration
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
