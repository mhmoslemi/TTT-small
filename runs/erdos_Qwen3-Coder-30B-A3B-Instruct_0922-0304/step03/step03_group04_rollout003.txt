import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for multi-resolution approach
    resolutions = [10, 20, 40, 70, 100, 150, 200]
    max_iter_per_res = 50
    population_size = 20
    slsqp_maxiter = 200
    stall_tolerance = 5
    perturb_strength = 0.01
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Start with coarse resolution
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize h
        if 'initial_h_values' in globals() and best_h is None:
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        elif best_h is not None:
            # Up-sample previous solution
            old_h = best_h
            old_n = best_n_points
            h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, old_n), old_h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Default initialization
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Constraints for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        target_sum = n_points / 2
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        stall_count = 0
        prev_best = float('inf')
        
        for iteration in range(max_iter_per_res):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Global search with Differential Evolution
            result_de = differential_evolution(
                objective,
                bounds,
                seed=seed + iteration,
                maxiter=10,
                popsize=population_size,
                disp=False
            )
            
            if result_de.success:
                h = result_de.x
                # Ensure constraints are met
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                # Local refinement with SLSQP
                result_slsqp = minimize(
                    objective,
                    h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': slsqp_maxiter, 'ftol': 1e-8}
                )
                
                if result_slsqp.success:
                    h = result_slsqp.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                
                current_c5 = evaluate_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    best_n_points = n_points
                    stall_count = 0
                else:
                    stall_count += 1
                    
                if stall_count > stall_tolerance:
                    # Perturb slightly and restart
                    h += np.random.normal(0, perturb_strength, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    stall_count = 0
                    
                if best_c5 <= 0.3807:
                    break
            else:
                # Fallback to gradient descent if optimization fails
                step = 0.005
                iterations = 50
                h_temp = h.copy()
                
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    
                h = h_temp

    if best_h is None:
        # Return default solution if nothing was found
        n_points = 70
        dx = 2.0 / n_points
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h += np.random.normal(0, 0.05, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, dx)
        best_n_points = n_points
    
    return (best_h.tolist(), best_c5, best_n_points)
