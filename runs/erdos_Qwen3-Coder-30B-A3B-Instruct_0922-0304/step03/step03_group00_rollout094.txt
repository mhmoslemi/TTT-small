import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - triangular-like shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1 - abs(i - mid) / (mid + 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting-plane approach
    # Track the maximum overlap found in previous iterations
    max_overlap_found = 0.0
    
    # Limit total iterations to avoid exceeding time budget
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Find the shift that causes maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        max_overlap_found = max(max_overlap_found, max_overlap)
        
        # Solve a new LP where we force this overlap to be lower
        # This is a linear relaxation of the problem by adding a cutting plane
        
        # Create variables
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints for h_vars
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2,
            C5_var >= 0
        ]

        # Find when overlap is maximized
        shift = max_idx - (n_points - 1)  # shift from full correlation
        
        # Add specific constraint that limits overlap at shift
        # For shift k, we have sum h[i] * (1 - h[i+k]) <= C5*dx 
        # Linearize: h[i] * (1 - h[i+k]) ~= h[i] - h[i]*h[i+k] 
        # Since h[i+k] is constant during update of h[i], we approximate
        
        # This is a linear approximation:
        # sum_{i} h[i] - h[i]*h[i+k] <= C5*dx
        # sum_{i} h[i] - sum_{i} h[i]*h[i+k] <= C5*dx
        # Since sum_{i} h[i] is fixed:
        # sum_{i} h[i]*h[i+k] >= sum_{i} h[i] - C5*dx
        
        # The LP version depends on either direct integer programming or adding cutting planes
        # to reduce subproblem complexity, but let's implement a smarter solution:
        
        # We make small changes to h and use gradient descent-like approach
        # on the maximum overlap term.
        
        # Compute current overlaps for better understanding
        # This will guide our descent
        
        # As a heuristic, we try to reduce largest overlap by directly changing h
        # Search for the affected intervals
        
        # Identify the peak overlap region
        src_indices = np.arange(n_points)
        if shift >= 0:
            dst_indices = src_indices[shift:]
            src_indices = src_indices[:-shift]
        else:
            dst_indices = src_indices[-shift:]
            src_indices = src_indices[:shift]
            
        # Now build a simple gradient descent approach with line search
        # we climb the gradient to reduce maximum overlap
        
        # Compute current gradient implicitly via finite differences on h
        # For rough accuracy, let's just do a simple descent step
        
        # This is a modified version of the previous, using cvxpy for one update step
        try:
            h_old = h.copy()

            # Use finite differences for gradient
            eps = 1e-4
            gradients = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")

                # Approximate gradient

        pass
    else:

        best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
