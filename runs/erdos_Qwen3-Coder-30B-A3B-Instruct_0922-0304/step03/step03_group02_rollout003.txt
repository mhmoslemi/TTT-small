import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # Number of particles
    max_iter = int(budget_s * 20)  # Roughly 20 iterations per second
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(m):
        particle = h0 + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(particle.copy())
        pbest_values.append(evaluate_c5(particle, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive coefficient
    c2 = 1.5  # Social coefficient
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            particle = particles[i]
            velocity = velocities[i]
            pbest = pbest_positions[i]
            
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocity = (w * velocity 
                       + c1 * r1 * (pbest - particle) 
                       + c2 * r2 * (gbest_position - particle))
            
            # Update position
            particle = particle + velocity
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate
            current_c5 = evaluate_c5(particle, dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_positions[i] = particle.copy()
                pbest_values[i] = current_c5
            
            # Update global best
            if current_c5 < gbest_value:
                gbest_position = particle.copy()
                gbest_value = current_c5
                
            # Update particles and velocities
            particles[i] = particle
            velocities[i] = velocity
            
            # Early exit if target achieved
            if gbest_value <= 0.3807:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(gbest_position, dx)
    
    return (gbest_position.tolist(), final_c5, n_points)
