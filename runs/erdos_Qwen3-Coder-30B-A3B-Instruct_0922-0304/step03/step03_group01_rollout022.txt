import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def line_search(h, grad, obj_func, dx, max_iter=20, alpha=0.01, beta=0.5):
    """Backtracking line search to find optimal step size."""
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2) / np.sum(h_new)
        obj_new = obj_func(h_new, dx)
        obj_old = obj_func(h, dx)
        if obj_new <= obj_old:
            return h_new, alpha
        alpha *= beta
    return h, alpha

def compute_gradient(h, dx, epsilon=1e-4):
    """Compute finite difference gradient of the overlap integral."""
    n = len(h)
    grad = np.zeros(n)
    base_obj = evaluate_c5(h, dx)
    
    for i in range(n):
        h_eps = h.copy()
        h_eps[i] += epsilon
        h_eps = np.clip(h_eps, 0, 1)
        h_eps = h_eps * (n / 2) / np.sum(h_eps)
        
        obj_eps = evaluate_c5(h_eps, dx)
        grad[i] = (obj_eps - base_obj) / epsilon
        
    return grad

        half_width = n_points // 4
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        h[mid1 - half_width:mid1 + half_width] = 0.5
        h[mid2 - half_width:mid2 + half_width] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')

    max_iter = 5000

        # Compute overlap and C5
        current_c5 = evaluate_c5(h, dx)
        
        # Early termination if target achieved
        if current_c5 <= 0.3807:
            best_c5 = current_c5
            best_h = h.copy()
            break

        grad = compute_gradient(h, dx)

        h = h_new

        h = np.clip(h, 0, 1)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
