import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.fftpack import fft, ifft

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize with spectral approach for better starting point
    # Compute FFT and identify dominant frequencies
    fft_h = fft(h)
    # Create mask for mid-frequencies that contribute to high overlap
    # We'll use frequency domain to perturb the solution
    freq_mask = np.ones(n_points)
    # Blur the mask (spectrum-guided mutation)
    mid = n_points // 2
    band_width = n_points // 8
    freq_mask[mid-band_width:mid+band_width] = 0.0
    
    # Mutation loop with spectral guidance
    mutation_count = 0
    while time.time() - start_time < budget_s - 30:
        if mutation_count % 10 == 0 and mutation_count > 0:
            # Re-evaluate best solution
            current_c5 = evaluate_c5(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                h = best_h.copy()
            
        # Generate new candidate by perturbing in frequency domain
        fft_h = fft(h)
        # Add noise to mid-frequencies
        noise = np.random.normal(0, 0.01 * np.std(fft_h), n_points)
        # Apply mask
        fft_h_new = fft_h * freq_mask + noise * (1 - freq_mask)
        # Inverse FFT
        h_new = np.real(ifft(fft_h_new))
        # Apply project to [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if best_c5 <= 0.3807:
                break
                
        h = h_new.copy()
        mutation_count += 1
        
        # Occasionally refine with local optimization
        if mutation_count % 50 == 0:
            # Local refinement using Nelder-Mead
            def obj_func(h_vec):
                h_local = h_vec.copy()
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                return evaluate_c5(h_local, dx)
                
            # Create simplex
            simplex = []
            simplex.append(h.copy())
            for i in range(n_points):
                h_pert = h.copy()
                h_pert[i] += np.random.normal(0, 0.05)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                simplex.append(h_pert)

                result = minimize(obj_func, h, method='Nelder-Mead',
                                  options={'maxiter': 50, 'adaptive': True})

                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_c5(h_refined, dx)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined.copy()
                        h = h_refined.copy()
                        if best_c5 <= 0.3807:

    # Final refinement with local search

        h_local = h_vec.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points / 2) / np.sum(h_local)
        return evaluate_c5(h_local, dx)
    
    # Try all the previous best values
    if time.time() - start_time < budget_s - 10:
        num_local_refinements = 10
        for i in range(num_local_refinements):
            h_local = h.copy()
            # Random perturbation
            for j in range(n_points):
                h_local[j] += np.random.normal(0, 0.02)
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            try:
                result = minimize(obj_func, h_local, method='Nelder-Mead',
                                  options={'maxiter': 20, 'adaptive': True})
                if result.success:
                    h_refined = result.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_c5(h_refined, dx)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined.copy()
                        h = h_refined.copy()
                        if best_c5 <= 0.3807:
                            break
            except:
                pass

    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
