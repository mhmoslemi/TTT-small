import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    # Ensure h is within bounds and sum constraint
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_individual(h, n_points, mutation_rate=0.1):
    h_new = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Random reset or Gaussian perturbation
            if np.random.random() < 0.5:
                h_new[i] = np.random.random()
            else:
                h_new[i] += np.random.normal(0, 0.05)
    return repair_h(h_new, n_points)

def crossover(parent1, parent2, n_points, crossover_rate=0.7):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Uniform crossover
    mask = np.random.random(n_points) < 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    
    return repair_h(child1, n_points), repair_h(child2, n_points)

def genetic_algorithm(seed, n_points, budget_s, pop_size=50, generations=1000):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values)
        h = repair_h(h, n_points)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
        h = repair_h(h, n_points)
    
    for _ in range(pop_size):
        population.append(h.copy())
    
    best_fitness = float('inf')
    best_individual = h.copy()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for individual in population:
            fitness = evaluate_c5(individual, 2.0 / n_points)
            fitness_scores.append(fitness)
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_size = 3
            candidates = np.random.choice(pop_size, tournament_size)
            best_candidate = candidates[np.argmin([fitness_scores[i] for i in candidates])]
            selected.append(population[best_candidate])
            
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_count = pop_size // 5
        elite_indices = np.argsort(fitness_scores)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2, n_points)
            child1 = mutate_individual(child1, n_points)
            child2 = mutate_individual(child2, n_points)
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Update best
        current_best_idx = np.argmin(fitness_scores)
        current_best_fitness = fitness_scores[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
            
        if best_fitness <= 0.3807:
            break
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    
    # Genetic Algorithm
    ga_start = time.time()
    best_h, best_c5 = genetic_algorithm(seed, n_points, budget_s)
    
    # Local refinement with gradient descent
    if time.time() - ga_start < budget_s * 0.9:
        # Convert to optimization variables
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Local optimization
        max_iter = 1000
        step_size = 0.001
        
        for iteration in range(max_iter):
            if time.time() - ga_start > budget_s * 0.9:
                break
                
            # Compute gradients
            corr = correlate(current_h, 1 - current_h, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Approximate gradient using finite differences
            grad = np.zeros_like(current_h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = current_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update
            current_h -= step_size * grad
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
            # Evaluate
            current_c5 = evaluate_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                
            # Early stopping
            if abs(current_c5 - best_c5) < 1e-7:
                break
    
    return (best_h.tolist(), best_c5, n_points)
