import numpy as np
from scipy.signal import correlate
from pywt import wavedec, waverec
import time
import pywt

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Wavelet-based optimization
    wavelet = 'haar'
    max_levels = int(np.log2(n_points)) - 1
    if max_levels < 6:
        max_levels = 6
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initial wavelet decomposition
    coeffs = wavedec(h, wavelet, level=max_levels)
    theta = np.concatenate(coeffs)
    
    # Iterative thresholding in wavelet domain
    max_iter = 30
    lambda_threshold = 0.1
    early_stopping = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Reconstruct from wavelet coefficients
        h_recon = waverec([theta], wavelet)
        if len(h_recon) > n_points:
            h_recon = h_recon[:n_points]
        elif len(h_recon) < n_points:
            h_recon = np.pad(h_recon, (0, n_points - len(h_recon)), 'constant')
        
        # Clip and normalize
        h_recon = np.clip(h_recon, 0, 1)
        h_recon = h_recon * (n_points / 2) / np.sum(h_recon)
        
        # Evaluate C5
        c5 = evaluate_c5(h_recon, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_recon.copy()
            last_improvement = iteration
            early_stopping = 0
        else:
            early_stopping += 1 
        
        if early_stopping >= 10:
            lambda_threshold *= 1.5
            early_stopping = 0
            
        if c5 <= 0.3807:
            break
            
        # Compute gradient approximation using finite differences
        grad = np.zeros_like(theta)
        eps = 1e-4
        for i in range(len(theta)):
            theta_plus = theta.copy()
            theta_plus[i] += eps
            h_plus = waverec([theta_plus], wavelet)
            if len(h_plus) > n_points:
                h_plus = h_plus[:n_points]
            elif len(h_plus) < n_points:
                h_plus = np.pad(h_plus, (0, n_points - len(h_plus)), 'constant')
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        # Update wavelet coefficients with gradient step
        step_size = 0.1
        new_theta = theta - step_size * grad
        # Soft thresholding
        new_theta = np.sign(new_theta) * np.maximum(np.abs(new_theta) - lambda_threshold, 0)
        theta = new_theta
        
    # Final spatial refinement
    h_refined = best_h.copy()
    for _ in range(2):  # 2 rounds of spatial refinement
        if time.time() - start_time > budget_s - 5:
            break
            
        # Spatial gradient using finite difference
        grad_spatial = np.zeros_like(h_refined)
        eps = 1e-3
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad_spatial[i] = (c5_plus - best_c5) / eps
        
        # Update with gradient
        lr = 0.01
        h_refined -= lr * grad_spatial
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Recompute C5
        new_c5 = evaluate_c5(h_refined, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()

    return (best_h.tolist(), best_c5, n_points)
