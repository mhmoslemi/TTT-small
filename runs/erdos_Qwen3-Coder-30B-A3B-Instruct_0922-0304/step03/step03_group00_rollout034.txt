import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Add more diverse initializations
    for i in range(1, n_particles):
        # Triangular function initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        for j in range(n_points):
            h[j] = max(0, 1 - 2 * abs(j - mid) / n_points)
        # Random modification
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track personal bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_c5 = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Find global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_c5
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            v = w * v + c_p * r_p * (pbest[i] - h) + c_g * r_g * (gbest - h)
            
            # Apply velocity limits
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            
            # Ensure constraint sum(h) = n_points/2 
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update swarm
            swarm[i] = h_new
            
            # Evaluate new position
            c5_new = evaluate_c5(h_new, dx)
            
            # Update personal best
            if c5_new < pbest_c5[i]:
                pbest[i] = h_new.copy()
                pbest_c5[i] = c5_new
                
                # Update global best if needed
                if c5_new < gbest_c5:
                    gbest = h_new.copy()
                    gbest_c5 = c5_new
                    
                    # Check for target
                    if gbest_c5 <= 0.3807:
                        best_h = gbest.copy()
                        best_c5 = gbest_c5
                        break
        
        # Update best solution so far
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
    
    return (best_h.tolist(), best_c5, n_points)
