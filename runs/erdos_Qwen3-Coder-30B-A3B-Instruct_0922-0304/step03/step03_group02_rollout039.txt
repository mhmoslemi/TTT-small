import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def ga_optimize(seed=42, budget_s=600, n_points=100):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = repair_h(h0, n_points)
        population.append(h0)
    
    # Fill population with random samples
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Parameters
    crossover_rate = 0.7
    mutation_rate = 0.2
    tournament_size = 3
    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = None
    
    max_generations = 1000
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection: tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        new_population = []
        
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
            
            new_population.append(repair_h(child1, n_points))
            new_population.append(repair_h(child2, n_points))
        
        population = new_population
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    h_best, c5_best = ga_optimize(seed, budget_s)
    n_points = len(h_best)
    dx = 2.0 / n_points
    # Final validation
    if np.abs(np.sum(h_best) - n_points / 2) > 1e-6:
        h_best = repair_h(h_best, n_points)
    
    c5_best = evaluate_c5(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
