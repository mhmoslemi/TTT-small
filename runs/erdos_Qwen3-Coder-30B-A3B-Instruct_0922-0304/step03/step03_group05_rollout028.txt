import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        # Interpolate to new resolution
        h_parent = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h = np.clip(h_parent, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start approach
    initial_h_list = [h.copy()]
    
    # Add some random perturbations to escape local minima
    for _ in range(5):
        h_pert = h + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_h_list.append(h_pert)

    # Evaluate all initial candidates
    for h_init in initial_h_list:
        current_h = h_init.copy()
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    # Main optimization loop
    max_iter = 1000
    iter_no_improve = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Optimize each candidate
        for h_template in initial_h_list:
            h = h_template.copy()
            
            # Optimize using SLSQP
            result = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h = result.x
            else:
                # If optimization fails, use a small perturbation
                h += np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                iter_no_improve = 0
            else:
                iter_no_improve += 1
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if iter_no_improve > 100:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
