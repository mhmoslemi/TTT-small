import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    swarm_size = 50
    max_iter_pso = 200
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Initialize PSO swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        else:
            h_init = np.random.rand(n_points) * 0.5
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(h_init.copy())
        pbest_vals.append(evaluate_c5(h_init, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    for iter_pso in range(max_iter_pso):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraint: sum(h) = n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Optimize the best PSO result
        result = minimize(objective, gbest, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            h_refined = result.x
            current_c5 = objective(h_refined)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined.copy()
        else:
            # Fallback to PSO best
            best_c5 = gbest_val
            best_h = gbest.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
