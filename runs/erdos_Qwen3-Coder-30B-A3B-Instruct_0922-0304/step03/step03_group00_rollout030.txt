import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        swarm.append(h0.copy())
        velocities.append(np.zeros(n_points))
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        swarm.append(h0.copy())
        velocities.append(np.zeros(n_points))
    
    # Generate other particles with triangular functions
    for i in range(1, n_particles):
        h = np.zeros(n_points)
        idx = np.arange(n_points)
        h = 1 - 2 * np.abs(idx - (n_points - 1) / 2) / (n_points - 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Keep track of personal bests
    pbest = [swarm[i].copy() for i in range(n_particles)]
    pbest_c5 = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_c5 = gbest_c5
    best_h = gbest.copy()
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Update inertia
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        # Update each particle
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c_p * r1 * (pbest[i] - h) + c_g * r2 * (gbest - h)
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to satisfy constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_c5(h_new, dx)
            
            # Update personal best
            if c5_new < pbest_c5[i]:
                pbest[i] = h_new.copy()
                pbest_c5[i] = c5_new
                
                # Update global best
                if c5_new < gbest_c5:
                    gbest = h_new.copy()
                    gbest_c5 = c5_new
                    
                    if gbest_c5 <= 0.3807:
                        break
            
            # Store updated values
            swarm[i] = h_new
            velocities[i] = v
            
            if gbest_c5 <= 0.3807:
                break
        
        # Early exit
        if gbest_c5 <= 0.3807:
            break
    
    # Return best solution
    return (gbest.tolist(), gbest_c5, n_points)
