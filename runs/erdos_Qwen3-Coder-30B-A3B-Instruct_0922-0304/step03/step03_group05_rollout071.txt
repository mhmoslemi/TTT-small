import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adjustment parameters
    max_move = 0.01
    patience = 0
    max_patience = 100
    
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Track best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Check if target achieved
        if best_c5 <= 0.3807:
            break
            
        if patience > max_patience:
            # Reset with new initialization
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            max_move = 0.01
            
        # Focus on indices contributing most to maximum overlap
        # The maximum overlap is determined by shift k = max_k - (n_points - 1)
        shift = max_k - (n_points - 1)
        
        # Create a contribution matrix to find top contributors
        contributions = np.zeros(n_points)
        for i in range(n_points):
            j = (i + shift) % n_points
            contributions[i] = h[i] * (1 - h[j])
            
        # Get indices with highest contributions
        top_indices = np.argsort(contributions)[-int(n_points * 0.05):]
        
        # Make adjustments to reduce overlap
        for i in top_indices:
            j = (i + shift) % n_points
            # Compute feasible adjustment
            decrease = min(h[i], max_move)
            increase = min(1 - h[j], max_move)
            adjustment = min(decrease, increase)
            
            if adjustment > 1e-8:
                h[i] -= adjustment
                h[j] += adjustment
                
        # Ensure constraint sum(h) = n_points / 2
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Ensure bounds
        h = np.clip(h, 0, 1)
        
        # Reduce max_move if no improvement recent
        if patience > 10:
            max_move *= 0.95
            
    # Final evaluation with best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
