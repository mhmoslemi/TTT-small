import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        h_new = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(h)), h)
        h = h_new
    else:
        # Fallback initialization: triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h += np.random.normal(0, 0.05, n_points)
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Generate multiple initial solutions
    initial_solutions = []
    
    # 1. Parent solution if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h_parent)), h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_solutions.append(h_parent.copy())
    
    # 2. Random perturbations
    for _ in range(4):
        h_pert = h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_solutions.append(h_pert.copy())
    
    # 3. Alternative initialization: piecewise constant
    h_alternative = np.ones(n_points)
    h_alternative[n_points//3:] = 0
    h_alternative = np.clip(h_alternative, 0, 1)
    h_alternative = h_alternative * (n_points / 2) / np.sum(h_alternative)
    initial_solutions.append(h_alternative.copy())

    # Evaluate all initial candidates
    for h_start in initial_solutions:
        h_current = h_start.copy()
        
        # Optimization loop for this starting point
        for iter_count in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate current solution
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
            # Optimize using SLSQP
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
                # Ensure feasibility
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
            else:
                # If optimization fails, use a small perturbation
                h_current += np.random.normal(0, 0.01, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
