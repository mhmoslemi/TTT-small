from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time
import random

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.clip(h, 0, 1)

        mask = np.ones(n_points, dtype=bool)
        mask[dominant_freq_indices] = False
        H_masked = H.copy()

                vertices[-1] = e_point

            # Contraction or shrinkage
            if r_value < values[-1]:

            else:
                c_point = centroid + 0.5 * (vertices[-1] - centroid)
            c_point = project_h(c_point)
            c_value = objective(c_point)
            if c_value < values[-1]:
                vertices[-1] = c_point
            else:
                # Shrink
                for i in range(1, len(vertices)):
                    vertices[i] = vertices[0] + 0.5 * (vertices[i] - vertices[0])
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
