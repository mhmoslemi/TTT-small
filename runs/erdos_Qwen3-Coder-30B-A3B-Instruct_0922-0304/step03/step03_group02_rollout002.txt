import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    m = 50  # number of particles
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    else:
        h_ref = np.ones(n_points)
        h_ref[n_points//2:] = 0
        h_ref += np.random.normal(0, 0.05, n_points)
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(m):
        if i == 0:
            h = h_ref.copy()
        else:
            h = h_ref + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_c5(h, dx))
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    max_iter = int(budget_s * 20)  # ~20 iterations/second
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
                
            # Update global best
            if current_c5 < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_c5
                
                # Early exit if target achieved
                if gbest_value <= 0.3807:
                    break
        
        if gbest_value <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_value, n_points)
