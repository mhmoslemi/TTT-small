import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    start_time = time.time()

    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Trust region optimization
    max_iter = 1000
    trust_radius = 0.1
    trust_radius_min = 1e-6
    trust_radius_max = 1.0
    expansion_factor = 2.0
    contraction_factor = 0.5
    max_no_improve = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        h_plus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
            h_plus[i] = h[i]  # Reset
            
        # Compute projected step using trust region
        step = -grad
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        # Try this step
        h_trial = h + step
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)

        # Accept the step if it's beneficial
        if c5_trial < current_c5:
            h = h_trial
            trust_radius = min(trust_radius * expansion_factor, trust_radius_max)
        else:
            trust_radius = max(trust_radius * contraction_factor, trust_radius_min)
            
        # Periodic perturbation to escape local minima
        if iteration % max_no_improve == max_no_improve - 1:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
