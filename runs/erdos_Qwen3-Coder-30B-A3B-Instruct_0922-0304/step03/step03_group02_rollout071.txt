import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - step function with some noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective for scipy.optimize
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Define constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Main optimization loop with adaptive strategy
    max_iter = 1000
    local_iter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive: perform local search more frequently in early iterations
        local_search = (iteration < 200) or (iteration % 10 == 0)
        
        if local_search:
            # Local optimization with constraints
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 20, 'ftol': 1e-9}
            )
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Random perturbation for exploration
            h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
                
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        # Occasionally restart with a better random start if stuck
        if iteration > 0 and iteration % 50 == 0 and local_iter < 20:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            local_iter += 1
    
    # Final refinement with focused optimization
    if time.time() - start_time < budget_s - 0.1:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = evaluate_c5(final_h, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()

    return (best_h.tolist(), best_c5, n_points)
