import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape centered at 1
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = 1.0 - abs(i - mid) / (n_points / 2)
        # Add some noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints and bounds
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start strategy with 5 random initializations
    initial_solutions = [h.copy()]

    for _ in range(4):
        h_perturbed = h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        initial_solutions.append(h_perturbed)

    # Optimization loop for each initial solution
    for h_init in initial_solutions:
        h_current = h_init.copy()
        
        for _ in range(50):  # SLSQP iterations
            if time.time() - start_time > budget_s - 1:
                break
                
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h_current = result.x
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
            else:
                # Fallback to small perturbation
                h_current += np.random.normal(0, 0.01, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)

            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()

        if best_c5 <= 0.3807:
            break

    # Basin hopping with exponential decay step size
    step_sigma = 0.1
    decay_rate = 0.995
    max_basin_iters = 1000

    h_current = best_h.copy()
    for iteration in range(max_basin_iters):
        if time.time() - start_time > budget_s - 1:
            break

        # Random perturbation
        perturbation = np.random.normal(0, step_sigma, n_points)
        h_perturbed = h_current + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)

        # Local optimization
        result = minimize(
            objective, 
            h_perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8}
        )

        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        else:
            h_candidate = h_perturbed

        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()

        # Update step size
        step_sigma *= decay_rate

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Fine-tune with higher precision
    if time.time() - start_time < budget_s - 1 and n_points == 70:
        # Interpolate to higher resolution
        n_points_high = 140
        dx_high = 2.0 / n_points_high
        h_high = np.interp(np.linspace(0, 2, n_points_high), 
                          np.linspace(0, 2, n_points), best_h)
        h_high = np.clip(h_high, 0, 1)
        # Scale to match integral constraint
        h_high = h_high * (n_points_high / 2) / np.sum(h_high)

        # Refine with SLSQP at high resolution
        def compute_c5_high(h):
            overlap = correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx_high

        # Constraints and bounds
        def constraint_sum_high(h):
            return np.sum(h) - n_points_high / 2

        constraints_high = [{'type': 'eq', 'fun': constraint_sum_high}]
        bounds_high = [(0, 1) for _ in range(n_points_high)]
        
        # Primal approach: use smaller steps
        result = minimize(
            compute_c5_high, 
            h_high.copy(), 
            method='SLSQP', 
            bounds=bounds_high, 
            constraints=constraints_high,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points_high / 2) / np.sum(h_final)
            final_c5 = compute_c5_high(h_final)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final
                n_points = n_points_high

    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
