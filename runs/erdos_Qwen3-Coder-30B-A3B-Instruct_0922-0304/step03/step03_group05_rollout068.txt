        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def apply_constraints(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()

    # Main optimization loop
    for iteration in range(max_iter):

        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Create a more sophisticated optimization step 
        # by using a combination of local search and perturbation

        perturb = np.random.normal(0, 0.02, n_points)
        h_perturbed = h + perturb
        h_perturbed = apply_constraints(h_perturbed)
        
        # Compare with current solution
        c5_perturbed = compute_c5(h_perturbed)
        if c5_perturbed < current_c5:
            h = h_perturbed
            
        # Additional heuristic: try to improve by weight adjustment
        # Simulated annealing approach with multiple restarts
