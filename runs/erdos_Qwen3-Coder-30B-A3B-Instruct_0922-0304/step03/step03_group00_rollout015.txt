```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use cvxpy to solve LPs
    def solve_lp_with_constraints(h_prev, max_k, C5_bound):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize C5
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add constraints for all shifts; for efficiency, we'll iterate
        # through relevant shifts only (center shifts first)
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue  # skip shift 0
            # Add constraint: sum h[i] * (1 - h[i + k]) <= C5_bound * dx
            # This is a bit tricky due to indexing, so compute manually in numpy
            shifted_indices = np.arange(n_points)
            if k > 0:
                indices = shifted_indices[k:].astype(int)
                # Zero-pad the h and 1-h arrays to don't miss any entries  
                arr1 = h_vars[:len(indices)]
                arr2 = h_vars[k:k+len(indices)]
                constraint_expr = cp.sum(cp.multiply(arr1, 1 - arr2)) <= C5_bound * dx
            elif k < 0:
                indices = shifted_indices[:n_points + k]
                # Zero-pad the h and 1-h arrays to don't miss any entries  
                arr1 = h_vars[-k:].astype(int)
                arr2 = h_vars[:len(indices)].astype(int)
                constraint_expr = cp.sum(cp.multiply(arr1, 1 - arr2)) <= C5_bound * dx
            else:
                continue
            constraints.append(constraint_expr)
            
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                new_h = h_vars.value
                return np.array(new_h), prob.value
        except:
            pass
        
        return h_prev, C5_bound  # fallback

    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Find maximum overlap shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(h) - 1)
        
        # Check if constraint for k already exists
        # Instead of adding constraints iteratively, we'll 
        # focus on improving the choice of initial h or relaxing approach
        
        # Here we do a local perturbation instead of LP solve
        # Since LP approach is complex, let's just try to optimize locally
        # Current strategy: gradient descent + projection
        step = 0.005
        for _ in range(50):
            corr = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            max_k = max_idx - (len(h) - 1)
            max_overlap = corr[max_idx] * dx
            
            # Estimate gradient w.r.t. h
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            # Update h
            h -= step * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Re-evaluate
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Make sure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)

    return (best_h.tolist(), final_c5, n_points)
```
