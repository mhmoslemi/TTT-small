import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Fourier-based SDP approach
    # Define the SDP relaxation
    # h is a vector of length n_points
    # We model h as a rank-1 matrix Y = h * h^T with constraints
    
    for iteration in range(100):  # Limited iterations
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update h to be feasible
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Simple heuristic-based update to explore neighborhood
        # Add small random perturbations and project
        perturbation = np.random.normal(0, 0.005, n_points)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept if better
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < current_c5:
            h = h_new

    # Final check and return best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
