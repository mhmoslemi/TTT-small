import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        # Interpolate to new resolution
        h_parent = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h_parent)), 
                            h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
    else:
        # Fallback: triangular with center peak
        x = np.linspace(0, 2, n_points)
        h_parent = np.maximum(0, 1 - np.abs(x - 1))
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)

    def compute_c5(h):
        # compute convolution of h with (1-h) using full mode
        overlap = correlate(h, 1 - h, mode="full")
        # we want maximum of the overlap, scaled by dx
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def project(h):
        # clip to bounds
        h = np.clip(h, 0, 1)
        # scale to ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Generate multiple initializations
    initial_hs = []
    initial_hs.append(h_parent.copy())
    
    # Add some random perturbations
    for _ in range(4):
        h_pert = h_parent + np.random.normal(0, 0.1, n_points)
        initial_hs.append(project(h_pert))
    
    best_c5 = float('inf')
    best_h = h_parent.copy()

    # Try multiple initial points
    for h_init in initial_hs:
        h = h_init.copy()
        local_best_c5 = float('inf')
        local_best_h = h.copy()
        
        # Local optimization
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        for _ in range(100):  # limited iters per initialization
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h)
            if current_c5 < local_best_c5:
                local_best_c5 = current_c5
                local_best_h = h.copy()
            
            # If target achieved, exit early
            if local_best_c5 <= 0.3807:
                break
                
            # SLSQP optimization step
            try:
                result = minimize(
                    objective, 
                    h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints,
                    options={'maxiter': 5, 'ftol': 1e-8}
                )
                if result.success:
                    h = result.x
                else:
                    # Perturb solution manually if optimization fails
                    h += np.random.normal(0, 0.01, n_points)
            except:
                # fallback
                h += np.random.normal(0, 0.01, n_points)
                
            # Ensure feasibility
            h = project(h)
            
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final refinement
    if time.time() - start_time < budget_s - 1:
        h_fine = best_h.copy()
        # Line-search along individual directions
        step_size = 0.01
        for _ in range(50):  # small number of line search iterations
            if time.time() - start_time > budget_s - 1:
                break
            c5_before = compute_c5(h_fine)
            best_direction = None
            best_c5_improvement = 0
            
            # Stochastic direction search
            for _ in range(10):
                direction = np.random.normal(0, 1, n_points)
                direction = project(direction) - project(np.zeros(n_points))
                # Normalize
                norm = np.linalg.norm(direction)
                if norm > 1e-8:
                    direction = direction / norm
                # Try moving along this direction
                h_plus = project(h_fine + step_size * direction)
                h_minus = project(h_fine - step_size * direction)
                
                c5_plus = compute_c5(h_plus)
                c5_minus = compute_c5(h_minus)
                
                if c5_plus < c5_before and c5_plus < c5_before - best_c5_improvement:
                    best_c5_improvement = c5_before - c5_plus
                    best_direction = direction
                elif c5_minus < c5_before and c5_minus < c5_before - best_c5_improvement:
                    best_c5_improvement = c5_before - c5_minus
                    best_direction = -direction
                                    
            if best_direction is not None:
                h_fine = project(h_fine + 0.5 * step_size * best_direction)
            else:
                break
                
        final_c5 = compute_c5(h_fine)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_fine.copy()
    
    return (best_h.tolist(), best_c5, n_points)
