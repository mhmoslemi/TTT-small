import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based adjustment approach
    max_iter = 1000
    max_move = 0.01
    patience = 0
    max_patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Reduce learning rate if stuck
                max_move *= 0.95
                patience = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Identify top contributing positions
        top_contributors = []
        for i in range(n_points):
            # Compute contribution to max_k shift
            k = max_k - (n_points - 1)  # adjust index to actual shift
            j = (i + k) % n_points
            contrib = h[i] * (1 - h[j])
            top_contributors.append((contrib, i, j))
            
        top_contributors.sort(reverse=True)
        top_indices = top_contributors[:max(1, n_points // 20)]
        
        # Adjust h values
        total_adjustment = 0
        for contrib, i, j in top_indices:
            proposed_move = min(max_move, h[i], 1 - h[j])
            if proposed_move > 1e-8:
                h[i] -= proposed_move
                h[j] += proposed_move
                total_adjustment += proposed_move
                
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
