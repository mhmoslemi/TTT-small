import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - smooth step function
        h = np.zeros(n_points)
        center = n_points // 2
        h[center - 10:center + 10] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiplicative update approach
    epsilon = 0.05
    damping = 0.995
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps for all lags
        corr = correlate(h, 1 - h, mode="full")
        S = corr * dx
        
        # Create updates using multiplicative rule
        h_new = np.zeros_like(h)
        
        # For each position, compute weighted update based on all lags
        for i in range(n_points):
            # Cumulative contribution from all lags
            total_pos = 0.0
            total_neg = 0.0
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    if k >= 0:
                        lag_idx = k
                    else:
                        lag_idx = n_points + k
                    # Contribution from this lag
                    contribution = np.exp(-S[lag_idx] / epsilon) / (1 + np.exp(S[lag_idx] / epsilon))
                    total_pos += contribution
                    total_neg += 1 / (1 + np.exp(S[lag_idx] / epsilon))
            
            # Multiplicative update
            if total_neg > 0:
                factor = total_pos / total_neg
            else:
                factor = 1.0
                
            h_new[i] = h[i] * factor * damping
            
        # Apply update and maintain feasibility
        h = np.clip(h_new, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
