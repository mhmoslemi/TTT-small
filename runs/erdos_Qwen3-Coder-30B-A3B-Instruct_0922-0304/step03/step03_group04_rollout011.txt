import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    M = 30  # Number of intervals
    dx = 2.0 / M
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        # Convert to interval averages
        h = np.zeros(M)
        for i in range(M):
            start_idx = int(i * len(h_initial) / M)
            end_idx = int((i + 1) * len(h_initial) / M)
            h[i] = np.mean(h_initial[start_idx:end_idx])
    else:
        h = np.ones(M) * 0.5
    
    # Normalize sum to 1
    h = h * (M / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Build overlap matrix
    A = np.zeros((2*M-1, M))
    for k in range(2*M-1):
        shift = k - (M - 1)
        for i in range(M):
            # Compute overlap of interval i with shifted interval i + shift
            start_i = i * dx
            end_i = (i + 1) * dx
            start_j = (i + shift) * dx
            end_j = (i + shift + 1) * dx
            
            if i + shift < 0 or i + shift >= M:
                # Out of bounds, overlap is 0
                overlap = 0
            else:
                # Compute intersection
                overlap_start = max(start_i, start_j)
                overlap_end = min(end_i, end_j)
                if overlap_end > overlap_start:
                    overlap = overlap_end - overlap_start
                else:
                    overlap = 0
            A[k, i] = overlap
    
    # Solve LP
    # Variables
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_vars) == M / 2,  # Sum constraint
        h_vars >= 0,              # non-negativity
        h_vars <= 1,              # upper bound
        C >= 0                    # C should be nonnegative
    ]
    
    # Add overlap constraints
    for k in range(2*M-1):
        # Convex combination: sum_{i} h_i * (1 - h_{i+k}) * A[k,i]
        # Since h_{i+k} might not exist (if i+k >= M), we'll check in the computation
        # However, for simplicity, make a separate constraint for the overlap matrix terms
        constraint_expr = 0
        for i in range(M):
            # The shifted index
            shifted_idx = i + (k - (M - 1))
            if 0 <= shifted_idx < M:
                constraint_expr += h_vars[i] * (1 - h_vars[shifted_idx]) * A[k, i]
            else:
                constraint_expr += h_vars[i] * A[k, i]  # 0 * anything = 0
        constraints.append(constraint_expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    problem = cp.Problem(objective, constraints)
    
    # Solve
    try:
        problem.solve(solver=cp.ECOS, abstol=1e-5, reltol=1e-5, max_iters=1000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_vars.value
            C_opt = C.value
            h_opt = np.clip(h_opt, 0, 1)
            # Ensure integral constraint:
            h_opt = h_opt * (M / 2) / np.sum(h_opt)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        else:
            # Fallback to gradient descent on the discretized problem
            h_temp = h.copy()
            # Simple gradient descent loop
            for _ in range(500):
                if time.time() - start_time > budget_s - 1:
                    break
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Simple gradient ascent using finite differences
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(M):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                # Update
                h_temp -= 0.01 * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (M / 2) / np.sum(h_temp)
                
                current_c5 = evaluate_c5(h_temp, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_temp.copy()
    except:
        # Fallback to gradient descent only if LP fails
        h_temp = h.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Gradient
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(M):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update
            h_temp -= 0.01 * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (M / 2) / np.sum(h_temp)
            
            current_c5 = evaluate_c5(h_temp, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_temp.copy()
    
    if best_c5 == float('inf'):
        best_c5 = evaluate_c5(best_h, dx)
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, M)
