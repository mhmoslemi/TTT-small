def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    best_c5 = float('inf')

        max_steps = 20

                break
                
            # Evaluate current overlap: compute all shifts once
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Approximate gradient using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-4
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += eps

                current_c5 = new_c5
            else:
                # Reduce learning rate and try again
                learning_rate *= 0.5
                if learning_rate < 1e-8:
