import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use interval-based convex optimization approach
    M = 30  # Number of intervals
    w = 2.0 / M  # Width of each interval
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Average initial values over intervals
        h_init_avg = np.zeros(M)
        for i in range(M):
            start_idx = int(i * len(h_init) / M)
            end_idx = int((i + 1) * len(h_init) / M)
            h_init_avg[i] = np.mean(h_init[start_idx:end_idx])
        h = np.clip(h_init_avg, 0, 1)
        h = h * (M / 2) / np.sum(h)
    else:
        # Fallback initialization with piecewise constant function
        h = np.ones(M)
        h[M//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (M / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Construct overlap matrix (2M-1 shifts, M intervals)
    # A[k][i] is overlap length between interval i and shifted interval (i+k)
    A = np.zeros((2*M-1, M))
    
    # For each shift k
    for k in range(1-2*M+1, 2*M-1):
        # For each interval i
        for i in range(M):
            # Calculate overlap between interval i and interval (i+k)
            # Interval i spans [i*w, (i+1)*w)
            # Interval (i+k) spans [(i+k)*w, (i+k+1)*w)
            if 0 <= i+k < M and 0 <= i < M:
                start_a = i * w
                end_a = (i+1) * w
                start_b = (i+k) * w
                end_b = (i+k+1) * w
                
                overlap_start = max(start_a, start_b)
                overlap_end = min(end_a, end_b)
                
                if overlap_start < overlap_end:
                    A[k+2*M-2][i] = overlap_end - overlap_start
                else:
                    A[k+2*M-2][i] = 0
            else:
                A[k+2*M-2][i] = 0
    
    # Convex optimization approach
    # Variables
    h_vars = cp.Variable(M)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_vars * w) == 1,   # Integral constraint
        h_vars >= 0,
        h_vars <= 1,
        C >= 0
    ]
    
    # Add overlap constraints for each shift
    for k in range(2*M-1):
        # For each k, compute sum of h_i * (1 - h_{i+k}) * A[k][i]
        # We need to implement conditional indexing for overlapping intervals
        row_constraint = 0
        for i in range(M):
            # Compute shift: i + k - (2*M-2) 
            shift_idx = i + k - (2*M - 2)
            if 0 <= shift_idx < M:
                # If h[i+k] corresponds to variable h_vars[shift_idx]
                row_constraint += h_vars[i] * (1 - h_vars[shift_idx]) * A[k][i]
            else:
                # This should not happen in valid region
                pass
        constraints.append(row_constraint <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=500, feastol=1e-6)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_opt = h_vars.value
            c5_opt = C.value
            # Minimize over representation error and integrate
            h_step = np.clip(h_opt, 0, 1)
            h_step = h_step * (M / 2) / np.sum(h_step)
            
            c5_val = evaluate_c5(h_step, w)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_step.copy()
    except Exception as e:
        # Fallback to original algorithm approach if LP fails
        pass
    
    # If regular optimization didn't work, use gradient descent approach
    # Try on the best solution found
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the solution slightly and evaluate
        h_new = h.copy() + np.random.normal(0, 0.01, M)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (M / 2) / np.sum(h_new)
        
        c5_new = evaluate_c5(h_new, w)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Early exit if achieving target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation using original method for quality check
    final_c5 = evaluate_c5(best_h, w)
    
    # If the record isn't beaten, just return the best solution
    return (best_h.tolist(), final_c5, M)
