import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h_vec):
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(overlap) * dx

    # Variable learning rate and momentum for adaptive step
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate and momentum
        if iteration - last_improvement > 50:
            lr *= 0.95
            momentum *= 0.99
            
        # Compute gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
