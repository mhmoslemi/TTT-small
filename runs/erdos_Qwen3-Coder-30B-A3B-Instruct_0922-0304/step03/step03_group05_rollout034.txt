import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start with basin hopping
    n_starts = 5
    initial_h_list = [h.copy()]
    
    # Generate perturbed initializations
    for _ in range(n_starts - 1):
        perturbed_h = h + np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        initial_h_list.append(perturbed_h)

    # Optimization loop
    max_iter = 2000
    iter_count = 0
    local_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select a random initial point
        h = initial_h_list[np.random.randint(len(initial_h_list))]
        
        # Local optimization with SLSQP
        result = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': local_iter, 'ftol': 1e-8}
        )
        
        current_h = result.x if result.success else h
        
        # Ensure feasibility
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Evaluate current solution
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Basin hopping - random perturbation with decaying step size
        step_size = 0.1 * np.exp(-iter_count / 1000)
        perturbed_h = current_h + np.random.normal(0, step_size, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        initial_h_list.append(perturbed_h)
        
        iter_count += 1

    # Final refinement if we have time
    if time.time() - start_time < budget_s - 1:
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
