import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 10
    
    # Resolution schedule
    resolutions = []
    while n_points <= 200:
        resolutions.append(n_points)
        n_points *= 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
    else:
        h = np.ones(10)
        h[5:] = 0
        h += np.random.normal(0, 0.05, 10)
        h = np.clip(h, 0, 1)
        h = h * (10 / 2) / np.sum(h)
    
    target_sum = len(h) / 2
    dx = 2.0 / len(h)
    
    # Constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(len(h))]
    
    for i, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Up-sample if needed
        if n_points > len(h):
            # Linear interpolation
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x_old, h)
            h = h_new
        elif n_points < len(h):
            # Down-sample by averaging
            steps = len(h) // n_points
            h = np.array([np.mean(h[i*steps:(i+1)*steps]) for i in range(n_points)])
            
        dx = 2.0 / n_points
        target_sum = n_points / 2
        
        # Update constraints and bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h, dx)
            
        # Try multiple random restarts to find better minima
        best_iter_c5 = float('inf')
        best_iter_h = h.copy()
        
        # Differential Evolution for global search
        de_bounds = [(0, 1) for _ in range(n_points)]
        de_result = differential_evolution(
            objective,
            de_bounds,
            maxiter=50,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed + i
        )
        
        if de_result.success:
            h = de_result.x
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            
        # Refinement with SLSQP
        result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
        else:
            # If optimization fails, try to modify existing h
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Ensure constraints are met
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_iter_c5:
            best_iter_c5 = current_c5
            best_iter_h = h.copy()
            
        # Additional check with local optimization
        for _ in range(2):
            # Small random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h_perturbed = h + perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            current_c5_pert = evaluate_c5(h_perturbed, dx)
            if current_c5_pert < best_iter_c5:
                best_iter_c5 = current_c5_pert
                best_iter_h = h_perturbed.copy()
                
        if best_iter_c5 < best_c5:
            best_c5 = best_iter_c5
            best_h = best_iter_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        return (best_h.tolist(), final_c5, len(best_h))
    else:
        return ([0.5]*10, 0.5, 10)
