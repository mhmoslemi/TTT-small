import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adjustment parameters
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            max_move *= 0.95
            no_improvement_count = 0
            
        # Compute overlaps for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        # Only consider first 2*n_points-1 elements since we're computing for all k
        overlap = overlap_full[n_points-1:]
        
        # Find the shift with maximum overlap
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Find indices contributing most to max overlap
        # This means indices i where h[i] * (1 - h[i + k_max]) is large
        # But in the full correlation, we need to be careful about wraparound
        # For the correlation at shift k_max, this is h[i] * (1 - h[(i+k_max) % n_points])
        # So we look at elements in h where h[i] * (1 - h[i+k_max]) is max
        
        # Compute the contribution for each index i
        contributions = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            contributions[i] = h[i] * (1 - h[j])
            
        # Select top contributing indices
        top_indices = np.argsort(contributions)[-int(n_points * 0.05):][::-1]
        
        # Perform adjustments
        for i in top_indices:
            # Current contribution at this index
            j = (i + k_max) % n_points
            
            # Compute maximum adjustment allowed
            max_decrease = min(h[i], max_move)
            max_increase = min(1 - h[j], max_move)
            
            # Choose largest possible adjustment
            epsilon = min(max_decrease, max_increase)
            
            if epsilon > 0:
                # Perform the adjustment
                h[i] -= epsilon
                h[j] += epsilon

        # Normalize to maintain constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Ensure bounds
        h = np.clip(h, 0, 1)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
