import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with structured pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add some noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO Parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # PSO swarm initialization
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    global_best = None
    global_best_value = float('inf')
    
    for _ in range(swarm_size):
        # Initialize particle with slightly perturbed version of h
        particle = h + np.random.normal(0, 0.02, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        p_best = particle.copy()
        p_best_val = evaluate_c5(particle, dx)
        personal_best.append(p_best)
        personal_best_values.append(p_best_val)
        if p_best_val < global_best_value:
            global_best = p_best.copy()
            global_best_value = p_best_val
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            particle = swarm[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            vel = (inertia * vel + 
                   cognitive * r1 * (personal_best[i] - particle) + 
                   social * r2 * (global_best - particle))
            
            # Update particle position
            particle += vel
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate
            current_val = evaluate_c5(particle, dx)
            
            # Update personal best
            if current_val < personal_best_values[i]:
                personal_best[i] = particle.copy()
                personal_best_values[i] = current_val
                
                # Update global best
                if current_val < global_best_value:
                    global_best = particle.copy()
                    global_best_value = current_val
            
            # Save particle
            swarm[i] = particle
            velocities[i] = vel
    
    # Refinement with SLSQP
    if global_best_value < best_c5:
        best_c5 = global_best_value
        best_h = global_best.copy()
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # SLSQP refinement with best solution as starting point
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                     options={'maxiter': 500, 'ftol': 1e-9}, callback=None)
    
    if result.success:
        ref_h = result.x
        ref_c5 = objective(ref_h)
        if ref_c5 < best_c5:
            best_c5 = ref_c5
            best_h = ref_h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
