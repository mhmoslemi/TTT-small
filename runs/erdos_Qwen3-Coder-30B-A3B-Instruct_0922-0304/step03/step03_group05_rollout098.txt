import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adjustment parameters
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)  # shift k corresponding to max overlap
        
        # Find indices that contribute most to max overlap
        top_contributors = []
        for i in range(n_points):
            j = (i + max_k) % n_points
            contribution = h[i] * (1 - h[j])
            top_contributors.append((contribution, i, j))
        
        top_contributors.sort(reverse=True)
        top_indices = top_contributors[:max(1, n_points // 20)]  # Top 5% or so
        
        # Adjust top contributors
        old_h = h.copy()
        adjusted = False
        
        for contrib, i, j in top_indices:
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute max adjustment that keeps constraints satisfied
            decrease = min(h[i], max_move)
            increase = min(1 - h[j], max_move)
            epsilon = min(decrease, increase)
            
            if epsilon > 1e-10:
                h[i] -= epsilon
                h[j] += epsilon
                adjusted = True
        
        # Ensure sum constraint and bounds
        if adjusted:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Reduce adjustment if no improvement
        if no_improvement_count > 50:
            max_move *= 0.95
            if max_move < 1e-6:
                max_move = 1e-6
            no_improvement_count = 0
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
