import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Adjustment parameters
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Reduce step size
            max_move *= 0.95
            no_improvement_count = 0
            
        # Compute all overlaps to identify problematic indices
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        # Get the shift k that produces maximum overlap
        k = max_idx - (n_points - 1)
        
        # Identify indices where h[i] * (1 - h[i+k]) contributes most to the max overlap
        # This requires computing all products h[i] * (1 - h[(i+k) % n_points]) 
        # and selecting top contributors to reduce his contribution
        products = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            products[i] = h[i] * (1 - h[j])
            
        # Sort indices by their product contribution
        top_indices = np.argsort(products)[::-1][:max(1, n_points // 20)]
        
        # Adjust h values
        for i in top_indices:
            j = (i + k) % n_points
            # Compute maximum adjustment to reduce h[i] * (1 - h[j])
            # We want to reduce h[i] and increase h[j] if possible
            decrease = min(max_move, h[i])
            increase = min(max_move, 1 - h[j])
            if decrease > 0 and increase > 0:
                adjust = min(decrease, increase)
                h[i] -= adjust
                h[j] += adjust
                
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        # Maintain exact sum
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
