from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def adjust_h(h, k, i, epsilon):
        new_h = h.copy()
        # Adjust values to reduce overlap
        new_h[i] -= epsilon
        new_h[(i + k) % n_points] += epsilon
        return np.clip(new_h, 0, 1)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def project_h(h):
        # Ensure sum is correct and h values are in [0,1]
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_move = 0.01
    no_improvement_count = 0
    max_no_improvement = 100

    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Reduce step size if no improvement for many iterations
            max_move *= 0.95
            no_improvement_count = 0
            
        # Compute overlap matrix
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        overlap = overlap[:n_points]  # Only positive shifts

        # Identify the critical shift and top contributing indices
        k = np.argmax(overlap)
        # Get top contributing indices
        contrib_indices = np.argsort(h * (1 - h))[-int(n_points * 0.05):][::-1]
        
        # For each high contributor, attempt to reduce overlap
        for i in contrib_indices:
            if i >= n_points:
                continue
                
            # How much can we adjust without violating constraints
            epsilon = min(h[i], 1 - h[(i + k) % n_points], max_move)
            
            if epsilon > 1e-8:
                new_h = adjust_h(h, k, i, epsilon)
                new_c5 = compute_c5(new_h)
                
                if new_c5 < current_c5:
                    h = new_h
                    break  # Take first improving step
        else:
            # If no improvement, perturb slightly
            perturbation = np.random.normal(0, 0.001, n_points)
            h += perturbation
            h = project_h(h)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Make sure final solution respects the constraints
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
