import numpy as np
from scipy.signal import correlate
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main cutting-plane loop
    max_iter = 5000
    constraints = []
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)  # Convert index to shift
        max_overlap = overlap[max_idx] * dx
        
        # Check if we've improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Add constraint for this shift k
        # Constraint: sum_{i} h[i] * (1 - h[i+k]) <= C5 * dx
        # Linearize: h[i] * (1 - h[i+k]) <= (1 - h[i+k]) * h[i] = (h[i] - h[i] * h[i+k])
        # Approximation: if h[i+k] is fixed, linearize around current h
        # For simplicity: use a linear approximation as constraint
        a = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                a[i] = 1 - h[j]  # Contribution to constraint from h[i]

        constraints.append((a, max_overlap * dx))
        
        # Solve new LP
        A_ub = np.array([c[0] for c in constraints])
        b_ub = np.array([c[1] for c in constraints])
        c = np.ones(n_points)  # Minimize sum h (or C5)
        
        # Linear programming bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Add sum constraint
        A_eq = np.ones((1, n_points))
        b_eq = np.array([n_points / 2])
        A_ub = np.vstack([A_ub, A_eq])
        b_ub = np.append(b_ub, b_eq)
        
        try:
            # Solve LP
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                           bounds=bounds, method='highs')
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If solution fails, fall back to random restart
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback if optimization fails
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    overlap = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
