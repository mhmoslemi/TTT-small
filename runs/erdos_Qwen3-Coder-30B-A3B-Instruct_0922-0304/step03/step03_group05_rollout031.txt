import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Multi-start with basin hopping
    num_starts = 5
    initial_candidates = [h.copy()]
    
    # Generate perturbed initial candidates
    for _ in range(num_starts - 1):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        initial_candidates.append(project_to_feasible(perturbed))

    # Local optimization on each candidate
    for candidate in initial_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(
            compute_c5, 
            candidate, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            optimized = project_to_feasible(result.x)
        else:
            optimized = project_to_feasible(candidate + np.random.normal(0, 0.01, n_points))
        
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized.copy()

    # Basin hopping with decreasing perturbation
    current_h = best_h.copy()
    sigma = 0.1
    cooling_factor = 0.999
    max_basin_iter = 1000
    
    for iteration in range(max_basin_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        perturbed = current_h + np.random.normal(0, sigma, n_points)
        perturbed = project_to_feasible(perturbed)
        
        # Local optimization
        result = minimize(
            compute_c5, 
            perturbed, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            optimized = project_to_feasible(result.x)
        else:
            optimized = project_to_feasible(perturbed + np.random.normal(0, 0.01, n_points))
            
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized.copy()
            sigma = 0.1  # Reset sigma if improvement found
        else:
            sigma *= cooling_factor  # Gradually decrease perturbation
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Adaptive resolution - double grid if time permits
    if time.time() - start_time < budget_s - 5 and n_points == 70:
        # Interpolate to higher resolution
        h_fine = np.interp(np.linspace(0, 2, 140), np.linspace(0, 2, 70), best_h)
        # Project to feasible region
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (140 / 2) / np.sum(h_fine)
        n_points = 140
        dx = 2.0 / n_points
        
        # Local optimization at higher resolution
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(
                compute_c5, 
                h_fine, 
                method='SLSQP', 
                bounds=[(0, 1) for _ in range(140)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - 140 / 2}],
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            
            if result.success:
                h_fine = project_to_feasible(result.x)
            else:
                break
                
            current_c5 = compute_c5(h_fine)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_fine.copy()

    # Final refinement
    final_result = minimize(
        compute_c5, 
        best_h, 
        method='SLSQP', 
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if final_result.success:
        best_h = project_to_feasible(final_result.x)
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
