import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the worst shift
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Use cvxpy to solve the QP for this shift
        h_var = cp.Variable(n_points)
        
        # Constraint: sum of h equals n_points/2
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Bound constraints
        constraints += [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Objective: minimize overlap at the max_k shift
        # We need to construct the quadratic form of the overlap at shift max_k
        # overlap at shift k is sum_{i} h[i] * (1 - h[i+k]) 
        # if k is positive, i ranges from 0 to n_points-k-1
        # if k is negative, i ranges from -k to n_points-1
        # if k is 0, i ranges from 0 to n_points-1
        objective = 0
        if max_k < n_points:
            # positive shift
            for i in range(n_points - max_k):
                objective += h_var[i] * (1 - h_var[i + max_k])
        else:
            # negative shift, needs to be converted
            shift_k = max_k - 2 * (n_points - 1)
            for i in range(-shift_k, n_points):
                objective += h_var[i] * (1 - h_var[i - shift_k])
        
        objective = cp.Minimize(objective * dx)
        
        # Solve the QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            h_new = h_var.value
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        except:
            # fallback to simple gradient step if QP fails
            step = 0.01
            iterations = 100
            h_new = h.copy()
            
            for _ in range(iterations):
                # Compute overlaps
                corr = correlate(h_new, 1 - h_new, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_new)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_new.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                # Update h
                h_new -= step * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            h = h_new
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
