    start_time = time.time()

    dx = 2.0 / n_points

        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-5:mid+5] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Use a more efficient approach with precomputed correlations
        # Correlation with reversed array (1-h) 
        h_reversed = 1.0 - h
        overlap = correlate(h, h_reversed, mode="full")
        # Extract valid overlaps (in range [0, 2])  
        max_overlap = np.max(overlap)
        return max_overlap * dx

    def project_feasible(h):
        # Ensure h is in [0,1] and integral equals 1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h

        # For now, keep it simple as direct function

    best_c5 = float('inf')
    best_h = h.copy()

    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Local optimization by small random perturbations
        h_trial = h.copy()
        # Add noise uniformly on [-0.05, 0.05]
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_trial += noise
        h_trial = project_feasible(h_trial)
        
        # Replace current solution if it's better
        trial_c5 = compute_c5(h_trial)
        if trial_c5 < current_c5:
            h = h_trial
        else:
            # Random exploration of the space 
            h_pert = h.copy()
            idx = np.random.randint(0, n_points)
            # Change a segment 
            start_idx = max(0, idx - 3)
            end_idx = min(n_points, idx + 3)
            for i in range(start_idx, end_idx):
                h_pert[i] += np.random.normal(0, 0.02)
            h_pert = project_feasible(h_pert)
            pert_c5 = compute_c5(h_pert)
            if pert_c5 < current_c5:
                h = h_pert

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
