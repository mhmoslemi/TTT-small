import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraint(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2"""
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_particles = 15
    max_iter = 3000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = project_to_constraint(h0, n_points)
        particles.append(h0.copy())
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = np.clip(h0, 0, 1)
        h0 = project_to_constraint(h0, n_points)
        particles.append(h0.copy())
    
    # Add other particles with triangular functions
    for i in range(n_particles - 1):
        h = np.zeros(n_points)
        mid = n_points // 2
        for j in range(n_points):
            h[j] = 1.0 - 2.0 * abs(j - mid) / n_points
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, n_points)
        particles.append(h.copy())
    
    # Initialize velocities
    for i in range(n_particles):
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Track best solutions
    pbest = [p.copy() for p in particles]
    pbest_fitness = [evaluate_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            v = (w * v 
                 + c_p * r_p * (pbest[i] - h)
                 + c_g * r_g * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraint(h_new, n_points)
            particles[i] = h_new
            
            # Evaluate fitness
            fitness = evaluate_c5(h_new, dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = h_new.copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = h_new.copy()
                    gbest_fitness = fitness
        
        # Early termination
        if gbest_fitness <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_fitness, n_points)
