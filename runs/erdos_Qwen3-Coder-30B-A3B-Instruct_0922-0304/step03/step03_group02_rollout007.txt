from scipy.signal import correlate
import time
from scipy.optimize import minimize
import math

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use a combination of methods for robust optimization
    max_iter = 1000
    iteration = 0

        h_noisy = h + np.random.normal(0, 0.01, n_points)
        h_noisy = np.clip(h_noisy, 0, 1)

        result = minimize(

            method='SLSQP',

            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-9}
        )
        
        if result.success:
