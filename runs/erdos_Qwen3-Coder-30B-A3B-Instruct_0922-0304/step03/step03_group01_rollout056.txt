import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: create a balanced two-pulse construction
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with gradient descent and projection
    step_size = 0.01
    patience = 0
    patience_limit = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Gradient descent step
        h_new = h - step_size * grad
        
        # Project onto [0,1]^n_points
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept new point if it improves objective
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
            if best_c5 <= 0.3807:
                break
        else:
            patience += 1
            
        h = h_new
        
        # Adaptive step size
        if iteration % 200 == 0 and iteration > 0:
            step_size *= 0.5
            
        if patience >= patience_limit:
            step_size *= 0.5
            patience = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
