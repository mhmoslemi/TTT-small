import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import pywt
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 64
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Wavelet-based refinement
    wavelet = 'haar'
    max_levels = int(np.log2(n_points)) - 1
    if max_levels < 1:
        max_levels = 1
    
    # Precompute wavelet transform matrices
    wavelet_coeffs = pywt.wavedec(h, wavelet, level=max_levels)
    coeffs_sizes = [len(c) for c in wavelet_coeffs]
    
    # Optimization parameters
    max_wavelet_iter = 30
    max_spatial_refine = 2
    
    # Iterative optimization
    for wavelet_iter in range(max_wavelet_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Wavelet domain optimization
        try:
            # Transform to wavelet domain
            coeffs = pywt.wavedec(h, wavelet, level=max_levels)
            
            # Apply soft thresholding
            threshold = 0.1 * (1.0 + 0.01 * wavelet_iter)
            new_coeffs = []
            for c in coeffs:
                new_coeffs.append(pywt.threshold(c, threshold, mode='soft'))
            
            # Reconstruct
            h_wavelet = pywt.waverec(new_coeffs, wavelet)
            if len(h_wavelet) > n_points:
                h_wavelet = h_wavelet[:n_points]
            elif len(h_wavelet) < n_points:
                pad = n_points - len(h_wavelet)
                h_wavelet = np.pad(h_wavelet, (0, pad), mode='constant')
            
            # Project back to feasible domain
            h = np.clip(h_wavelet, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        except Exception:
            pass
            
        # Spatial refinement with gradient ascent
        for refine_iter in range(max_spatial_refine):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Get top wavelet features
            coeffs = pywt.wavedec(h, wavelet, level=max_levels)
            magnitudes = [np.abs(c).mean() for c in coeffs]
            max_magnitude_idx = np.argmax(magnitudes)
            
            # Refine only top features
            h_refined = h.copy()
            lr = 0.01 + 0.001 * refine_iter
            
            # Simple gradient approximation
            eps = 1e-4
            base_c5 = evaluate_c5(h_refined, dx)
            grad = np.zeros_like(h_refined)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            
            h_refined -= lr * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            h = h_refined
            
        # Evaluate and update best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
