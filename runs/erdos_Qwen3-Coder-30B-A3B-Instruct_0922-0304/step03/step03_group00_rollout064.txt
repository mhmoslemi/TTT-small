import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    total_sum = n_points / 2.0

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * total_sum / np.sum(h_initial)
    else:
        # Fallback: simple construction
        h_initial = np.ones(n_points)
        h_initial[n_points//2:] = 0
        h_initial = h_initial * total_sum / np.sum(h_initial)

    best_h = h_initial.copy()
    best_c5 = float('inf')

    # Set up SDP relaxation
    h = Variable(n_points)
    C5 = Variable()

    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == total_sum, C5 >= 0]

    # Loop over different values of k to add constraints
    for k in range(-n_points + 1, n_points):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute overlap efficiently using correlation
        # Use sliding window formula for efficiency
        if k >= 0:
            corr_term = 0
            for i in range(n_points - k):
                corr_term += h[i] * (1 - h[i + k])
        else:
            corr_term = 0
            for i in range(-k, n_points):
                corr_term += h[i] * (1 - h[i + k])
        constraints.append(C5 >= corr_term)

    # Objective
    objective = Minimize(C5)

    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=1000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_sol = h.value
            c5_estimate = prob.value
            if c5_estimate < best_c5:
                best_c5 = c5_estimate
                best_h = np.maximum(0, np.minimum(1, h_sol))
        else:
            # Fallback to current construction
            best_h = h_initial.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        best_h = h_initial.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)

    # Refine solution with local optimization if time allows
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - total_sum}
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 50, 'ftol': 1e-9})
        if result.success:
            fine_h = result.x
            fine_c5 = evaluate_erdos_solution(fine_h, dx)
            if fine_c5 < best_c5:
                best_c5 = fine_c5
                best_h = fine_h

    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
