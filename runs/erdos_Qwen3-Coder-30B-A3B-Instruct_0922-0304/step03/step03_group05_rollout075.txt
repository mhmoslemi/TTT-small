import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    # Generate multiple initial points
    initial_points = []
    
    # Parent initialization
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_parent)), h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_points.append(h_parent)
    
    # Random perturbations
    for _ in range(4):
        h_rand = h.copy()
        noise = np.random.normal(0, 0.1 * np.mean(h), n_points)
        h_rand += noise
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_points.append(h_rand)
    
    # Additional fallback if needed
    if len(initial_points) < 5:
        h_fallback = np.ones(n_points)
        h_fallback[n_points//2:] = 0
        h_fallback += np.random.normal(0, 0.1, n_points)
        h_fallback = np.clip(h_fallback, 0, 1)
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        initial_points.append(h_fallback)

    # Main optimization loop
    max_iter = 100
    iter_count = 0
    no_improve_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate all initial points and pick the best
        current_best_c5 = float('inf')
        current_best_h = h.copy()
        for i, h_init in enumerate(initial_points):
            h_current = h_init.copy()
            # Local optimization
            result = minimize(
                objective, 
                h_current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                h_opt = result.x
            else:
                h_opt = h_current
            
            # Ensure feasibility
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Evaluate and update best
            c5_val = compute_c5(h_opt)
            if c5_val < current_best_c5:
                current_best_c5 = c5_val
                current_best_h = h_opt.copy()
                
        # Update global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # For next iteration, generate new random starts around the current best
        initial_points = []
        for _ in range(5):
            h_new = best_h.copy()
            noise = np.random.normal(0, 0.05 * np.mean(best_h), n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            initial_points.append(h_new)
            
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
