import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the worst shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k = max_idx - (len(h) - 1)
        
        # Prepare QP formulation
        # The objective is to minimize sum_{i} h[i] * (1 - h[i + k])
        # where k is the shift that currently gives maximum overlap
        
        # Construct interaction matrix for the specific shift k
        # For shift k: we minimize sum_{i} h[i] * (1 - h[i+k]) * dx
        # Which is sum_{i} h[i] - h[i] * h[i+k] * dx
        # Note: In our case we can sacrifice some precision for simplicity
        
        # Create variable and constraints for CVXPY
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize sum of h[i] * (1 - h[i+k]) over all valid i
        # Create objective function
        objective_expr = 0
        for i in range(n_points):
            if i + k < n_points and i + k >= 0:
                objective_expr += h_vars[i] * (1 - h_vars[i + k]) * dx
                
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Formulate the QP
        objective = cp.Minimize(objective_expr)
        prob = cp.Problem(objective, constraints)
        
        # Solve it
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_vars.value
                h = h_new.copy()
            else:
                # If QP solver fails, use gradient descent
                step = 0.005
                iterations = 100
                h_temp = h.copy()
                
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    # Compute gradient using finite differences
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                        
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    
                h = h_temp
        except:
            # Fallback
            step = 0.005
            iterations = 100
            h_temp = h.copy()
            
            for _ in range(iterations):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                    
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
            h = h_temp
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
