import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - balanced pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Trust region optimization with projected gradient descent
    max_iter = 5000
    tr_radius = 0.1
    tr_radius_min = 1e-6
    expansion_factor = 2.0
    contraction_factor = 0.5
    early_stop_patience = 50
    patience_counter = 0
    
    # Gradient estimation and update helper
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        h_base = h.copy()
        base_c5 = evaluate_c5(h_base, dx)
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Trust region step
    def tr_step(h_prev, grad, tr_radius):
        # Simple gradient descent step with trust region
        step = -grad * tr_radius / np.linalg.norm(grad)
        h_new = h_prev + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Compute gradient
        grad = compute_gradient(h)
        
        # Compute step
        h_new = tr_step(h, grad, tr_radius)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject step based on trust region
        if new_c5 < current_c5:
            h = h_new
            tr_radius = min(tr_radius * expansion_factor, 0.5)
            patience_counter = 0
        else:
            tr_radius = max(tr_radius * contraction_factor, tr_radius_min)
            patience_counter += 1
            if patience_counter > early_stop_patience:
                # Perturb h slightly if no progress
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                patience_counter = 0

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
