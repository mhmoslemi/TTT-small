import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def repair_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Mutation: Gaussian perturbation + random reset"""
    h_new = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # 50% chance of Gaussian perturbation
            if np.random.random() < 0.5:
                h_new[i] += np.random.normal(0, 0.02)
            else:
                # 50% chance of random reset
                h_new[i] = np.random.random()
    return repair_h(h_new, n_points)

def crossover(parent1, parent2, crossover_rate, n_points):
    """Order-one crossover"""
    if np.random.random() >= crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point1 = np.random.randint(1, n_points - 1)
    point2 = np.random.randint(point1, n_points)
    
    child1 = parent1.copy()
    child2 = parent2.copy()
    
    child1[point1:point2] = parent2[point1:point2]
    child2[point1:point2] = parent1[point1:point2]
    
    child1 = repair_h(child1, n_points)
    child2 = repair_h(child2, n_points)
    
    return child1, child2

    else:
        population = [np.ones(n_points) * 0.5]
    
    # Fill up with random individuals
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_solution = None
    best_c5 = float('inf')
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = [evaluate_c5(individual, dx) for individual in population]
        
        # Find best solution in this generation
        gen_best_idx = np.argmin(fitness_scores)
        gen_best_c5 = fitness_scores[gen_best_idx]
        if gen_best_c5 < best_c5:
            best_c5 = gen_best_c5
            best_solution = population[gen_best_idx].copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        
        # Elitism: keep top 20%
        elite_size = max(1, int(0.2 * pop_size))
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate, n_points)
            child1 = mutate_individual(child1, mutation_rate, n_points)
            child2 = mutate_individual(child2, mutation_rate, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return best_solution, best_c5
