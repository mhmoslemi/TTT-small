import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Early stopping
    patience = 0
    max_patience = 10
    
    # Iterative refinement with quadratic programming
    max_iterations = 60
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break
        
        # Compute gradient approximation using finite differences
        epsilon = 1e-4
        grad = np.zeros(n_points)
        
        # Evaluate gradient: df/dh_i ≈ [f(h+εe_i) - f(h-εe_i)] / (2ε)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * epsilon)
        
        # Formulate and solve quadratic program
        # Minimize: 0.5 * d^T * H * d + grad^T * d
        # Subject to: 0 <= d_i <= 1, sum(d) = 0 (since we optimize relative to h)
        # This is a constrained QP where we want to minimize a quadratic approximation
        # of the objective
        
        # For simplicity, we use a projected gradient descent approach
        # We'll do a line search with multiple steps to find good descent direction
        
        # Start with gradient descent step
        step_size = 0.01
        new_h = h - step_size * grad
        
        # Project back to feasible set
        new_h = project_to_feasible(new_h, target_sum)
        
        # Try a few line search steps
        best_new_h = new_h.copy()
        best_new_c5 = compute_c5(new_h, dx)
        
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            test_h = h - alpha * grad
            test_h = project_to_feasible(test_h, target_sum)
            test_c5 = compute_c5(test_h, dx)
            if test_c5 < best_new_c5:
                best_new_c5 = test_c5
                best_new_h = test_h.copy()
        
        h = best_new_h.copy()
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
