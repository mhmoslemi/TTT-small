import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Adaptive coordinate-wise updates
    max_iters = 100000
    iter_count = 0
    no_improvement_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        if no_improvement_count > 1000:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Find gradients (approximate)
        # For each point i, compute gradient direction
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Compute gradient component for h[i]
            # Delta[0] = h[i] - (1-h[i]) = 2*h[i] - 1
            # But we want the effect on max overlap
            gradients[i] = overlap[i] * (1 - 2 * h[i])
            
        # Select top coordinates by absolute gradient
        top_indices = np.argsort(np.abs(gradients))[::-1][:max(1, n_points // 10)]
        
        # Update selected coordinates
        updated = False
        for i in top_indices:
            if time.time() - start_time > budget_s - 2:
                break
                
            orig_h = h[i]
            
            # Compute step size to decrease gradient
            # This is simplified: stay within bounds
            # Negative direction is better if gradient > 0
            alpha = min(1.0, max(0.0, -gradients[i] / (1.0 + np.abs(gradients[i]))))
            
            # Update with clamping
            h[i] = np.clip(orig_h + alpha * (1 - 2 * h[i]), 0, 1)
            
            # Renormalize to maintain sum constraint
            h = normalize_h(h, target_sum)
            
            # Check result
            new_c5 = compute_c5(h, dx)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                updated = True
                last_improvement = iter_count
                
                if best_c5 <= 0.3807:
                    break
            else:
                # Revert
                h[i] = orig_h
                
        if updated:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        iter_count += 1
        
        # Periodic full objective check 
        if iter_count % 1000 == 0:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
    return (best_h.tolist(), best_c5, n_points)
