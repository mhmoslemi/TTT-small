import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a smooth step function
        h = np.zeros(n_points)
        # Create a smooth step from 0 to 1
        x = np.linspace(0, 2, n_points)
        # Centered ramp function 
        h = np.maximum(0, np.minimum(1, (x - 0.5) * 2))
        # Add slight randomness
        h += np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, target_sum)
    
    # Remez-like iterative refinement
    max_iterations = 30
    alpha = 0.1
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap function
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        
        # Create a phase inversion pattern based on dominant frequency
        # Estimate dominant frequency from the peak in overlap
        # Use the center part from both ends to get an estimate of the oscillator
        lag = max_lag_idx - (n_points - 1)  # convert to lag index
        if lag == 0:
            # Avoid division by zero
            freq = 0.1
        else:
            # Estimate dominant frequency from overlap peak
            freq = 1.0 / (abs(lag) + 0.1)
            
        # Generate sinusoidal pattern around the fundamental
        x = np.linspace(0, 2, n_points)
        pattern = np.sin(2 * np.pi * freq * x + np.random.rand() * 2 * np.pi)
        # Apply pattern as gradient update
        h_new = h + alpha * pattern
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, target_sum)
        
        # Accept or reject update
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            if best_c5 <= 0.3807:
                break
            
        # Reduce step size
        alpha *= 0.95
    
    # Final refinement with local optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # SLSQP can get trapped in local minima, try multiple restarts
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        if result.success:
            h_test = result.x
            h_test = np.clip(h_test, 0, 1)
            h_test = project_to_feasible(h_test, target_sum)
            test_c5 = compute_c5(h_test, dx)
            if test_c5 < best_c5:
                best_h = h_test.copy()
                best_c5 = test_c5
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
