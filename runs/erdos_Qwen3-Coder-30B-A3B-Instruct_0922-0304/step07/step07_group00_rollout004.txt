import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute exact C5 value using correlate."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute approximate gradient using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-8
    
    # Compute forward difference approximation
    h_plus = h.copy()
    h_minus = h.copy()
    
    for i in range(n):
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = np.clip(h_plus, 0, 1)
        h_minus = np.clip(h_minus, 0, 1)
        h_plus = normalize_h(h_plus, len(h) / 2)
        h_minus = normalize_h(h_minus, len(h) / 2)
        
        c5_plus = compute_c5_exact(h_plus, dx)
        c5_minus = compute_c5_exact(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        h_plus[i] -= eps
        h_minus[i] += eps
        
    return grad

def adam_update(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer update."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return h, m, v

    max_iter = 10000

    # Adam parameters
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 1
    
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (use finite differences for robustness)
        grad = compute_gradient(h, dx)
        
        # Update with Adam
        h, m, v = adam_update(h, grad, m, v, t, lr)
        t += 1
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

        current_c5 = compute_c5_exact(h, dx)
        if current_c5 < best_c5:
