import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
    
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    max_iter = 3000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement = 0
    
    # Initialize momentum
    velocity = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Compute gradient (this is the shift of (1-h) that achieves max overlap)
        # This is the derivative of overlap w.r.t. h
        grad = np.zeros_like(h)
        if max_k < len(h):
            # Shift (1-h) to the right by max_k positions
            grad[max_k:] = 1 - h[:-max_k] if max_k > 0 else 1 - h
        else:
            # Need to handle the wrap-around case explicitly
            # In the full convolution, we have overlap values indexed from 0 to 2*n-2
            # Translate max_k (from 0 to 2*n-2) into proper offset
            offset = max_k - (n_points - 1)  # offset in the range -n+1 to n-1
            if offset > 0:
                grad[:n_points - offset] += (1 - h[offset:])
            elif offset < 0:
                grad[-offset:] += (1 - h[:n_points + offset])
            else:
                grad[:] += (1 - h[:])
        
        # Project onto feasible region
        velocity = momentum * velocity - learning_rate * grad
        h_new = h + velocity
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum) > 1e-10:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate C5
        overlap_new = correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(overlap_new) * dx
        
        # Accept new solution if it is better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        h = h_new
        
        # Adapt learning rate if no improvement
        if no_improvement > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-5:
                learning_rate = 1e-5
            no_improvement = 0
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
