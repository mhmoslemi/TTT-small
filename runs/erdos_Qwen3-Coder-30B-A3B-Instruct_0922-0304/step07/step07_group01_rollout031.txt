import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Initialize with a step-like function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Stage 1: Convex relaxation optimization
    # We will use CVXPY to solve a convex relaxation
    # But to reduce time, we'll do a single optimization pass
    # Define the problem variables
    h_cvx = cp.Variable(n_points)
    
    # Objective: minimize the maximum overlap
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == target_sum,
        C >= 0
    ]
    
    # For a simplified convex approach, we use a modification
    # Direct optimization using gradient descent after relaxation
    
    # Use projected gradient descent for refinement
    lr = 0.01
    max_iter = 1000
    decay_rate = 0.999
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute gradient approximations using finite differences
        h_current = best_h.copy()
        c5_current = compute_c5(h_current, dx)
        
        # Use simpler approach: make small perturbations and evaluate
        # For efficient gradient estimation (1D case), step through points

        # Compute overlap for each shift
        overlaps = np.zeros((n_points + 1,))  # full convolution

            # Compute negative overlap for current point
            if shift < len(h_current):
                # Slide h[shift:] on h[:n_points-shift] 
                overlap = np.sum(h_current[shift:] * (1 - h_current[:n_points-shift]))
            else:
                overlap = 0
            overlaps[shift] = overlap
        
        best_overlap = np.max(overlaps)
        best_idx = np.argmax(overlaps)
        
        # Perturb h in direction opposite to gradient
        h_new = h_current.copy()
        if best_idx <= n_points//2:
            # Gradient is positive, lower the values
            h_new += lr * (0.5 - h_new)
        else:
            # Gradient is negative, raise the values
            h_new += lr * (0.5 - h_new)
            
        h_new = project_to_constraints(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new

            lr = max(lr * decay_rate, 1e-6)

        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
