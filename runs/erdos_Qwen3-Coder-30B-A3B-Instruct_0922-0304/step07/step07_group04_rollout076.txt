import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function: piecewise constant
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with adaptive regularization
    max_iter = 10000
    eta = 1e-3  # learning rate
    beta = 0.9  # momentum
    lambda_reg = 1e-4  # regularization parameter
    gamma = 0.5  # backtracking factor
    
    # Precompute shift indices to speed up gradient computation
    all_shifts = []
    for k in range(-n_points + 1, n_points):
        shift_indices = np.arange(n_points)
        if k >= 0:
            shift_indices = (shift_indices - k) % n_points
        else:
            shift_indices = (shift_indices - k) % n_points
        all_shifts.append(shift_indices)
    
    # Momentum term
    v = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient using finite differences
        h_eval = h.copy()
        current_c5 = evaluate_overlap(h_eval, dx)
        
        # Estimate gradient numerically
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            sum_plus = np.sum(h_plus)
            if sum_plus > 0:
                h_plus = h_plus * (target_sum / sum_plus)
            c5_plus = evaluate_overlap(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Add regularization for constraint
        penalty_grad = np.ones(n_points) * (np.sum(h) - target_sum) * lambda_reg
        grad += penalty_grad
        
        # Update with momentum
        v = beta * v - eta * grad
        h_new = h + v
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        current_sum_new = np.sum(h_new)
        if current_sum_new > 0:
            h_new = h_new * (target_sum / current_sum_new)
        
        # Backtracking line search
        alpha = 1.0
        while True:
            h_trial = h + alpha * v  # do not apply momentum here
            h_trial = np.clip(h_trial, 0, 1)
            sum_trial = np.sum(h_trial)
            if sum_trial > 0:
                h_trial = h_trial * (target_sum / sum_trial)
            trial_c5 = evaluate_overlap(h_trial, dx)
            
            if trial_c5 <= current_c5 + 1e-8 or alpha < 1e-6:
                h = h_trial
                break
            alpha *= gamma
        
        # Update regularization
        lambda_reg *= 1.1
        
        # Early stopping based on improvement
        new_c5 = evaluate_overlap(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
