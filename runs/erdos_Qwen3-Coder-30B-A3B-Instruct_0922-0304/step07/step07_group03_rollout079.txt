import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # ES parameters
    population_size = 20
    elite_fraction = 0.5
    max_generations = 5000
    initial_lr = 0.1
    final_lr = 0.01
    
    # Initialize population
    population = []
    fitness_scores = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        
        individual = project_to_constraints(individual, target_sum)
        population.append(individual)
        fitness_scores.append(compute_c5(individual, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness_scores)
    best_h = population[best_idx].copy()
    best_c5 = fitness_scores[best_idx]
    
    # Evolutionary process
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        elite_indices = sorted_indices[:int(population_size * elite_fraction)]
        
        # Update learning rate
        lr = initial_lr - (initial_lr - final_lr) * generation / max_generations
        
        # Generate new population
        new_population = [population[i] for i in elite_indices]
        new_fitness = [fitness_scores[i] for i in elite_indices]
        
        # Mutate elite individuals
        for i in range(int(population_size * elite_fraction), population_size):
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Gaussian mutation
            noise = np.random.normal(0, lr, n_points)
            mutated = parent + noise
            
            # Project to constraints
            mutated = project_to_constraints(mutated, target_sum)
            
            new_population.append(mutated)
            new_fitness.append(compute_c5(mutated, dx))
        
        # Update population
        population = new_population
        fitness_scores = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness_scores[current_best_idx]
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
