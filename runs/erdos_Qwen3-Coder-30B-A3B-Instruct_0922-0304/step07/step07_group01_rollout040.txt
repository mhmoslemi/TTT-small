import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint using projection
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        # Distribute the difference
        if diff > 0:
            # Need to increase values
            h += diff / len(h)
        else:
            # Need to decrease values
            h += diff / len(h)
        # Re-project to ensure bounds
        h = np.clip(h, 0, 1)
        # Apply small adjustment to maintain sum constraint exactly
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h
