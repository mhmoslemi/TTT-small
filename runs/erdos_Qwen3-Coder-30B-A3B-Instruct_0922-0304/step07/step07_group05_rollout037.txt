import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Start with 40% of particles as variations of initial
        particles.append(h.copy())
        for _ in range(int(0.4 * population_size) - 1):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
    
    # Fill the rest of population with random feasible solutions
    while len(particles) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h)
    
    # PSO parameters
    w = 0.8
    c1 = c2 = 1.5
    inertia_decay = 0.98
    
    # Initialize personal bests and global best
    personal_bests = [p.copy() for p in particles]
    personal_best_fitnesses = [compute_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(personal_best_fitnesses)
    gbest = personal_bests[gbest_idx].copy()
    gbest_fitness = personal_best_fitnesses[gbest_idx]
    
    # Initialize velocities
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(population_size)]
    
    max_iterations = 60
    stagnation_count = 0
    stagnation_threshold = 20
    
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_bests[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fitness
                personal_bests[i] = particles[i].copy()
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest = particles[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            
            # Early termination if target achieved
            if gbest_fitness <= 0.3807:
                break
                
        # Update inertia weight
        w *= inertia_decay
        
        # Stagnation detection
        if stagnation_count > stagnation_threshold:
            # Increase exploration
            for i in range(population_size):
                velocities[i] += np.random.normal(0, 0.05, n_points)
            stagnation_count = 0
    
    # Final refinement with gradient descent on best
    best_h = gbest.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(20):  # Run a few rounds of local refinement
        result = minimize(objective, best_h.copy(), method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
