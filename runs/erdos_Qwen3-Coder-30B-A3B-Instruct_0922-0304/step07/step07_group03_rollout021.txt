import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        # Scale down
        h = h * (target_sum / current_sum)
    else:
        # Distribute deficit
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Start with a sin-based construction
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Alternating minimization parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        c = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(c)
        
        # Form gradient for update
        # Take the derivative of the overlap with respect to h
        # For a fixed k, max overlap is related to h[i] for i k positions away
        # Let's use a simplified gradient
        grad_h = np.zeros_like(h)
        # This step works in a simplified way for update
        for i in range(len(h)):
            idx = (i + max_k - len(h) + 1) % len(h)  # assuming the overlap at max_k
            if 0 <= idx < len(h):
                grad_h[i] -= (1 - h[idx])  # simplified
        # Real update
        h_candidate = h - alpha * grad_h
        
        # Add random perturbation 
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to feasible set
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h = h_candidate.copy()
            
        # Decrease learning rates
        alpha = alpha / (1 + iteration / tau)
        epsilon = epsilon / (1 + iteration / tau)
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
