import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Distribute the difference evenly
    h = np.clip(h + diff / len(h), 0, 1)
    # Final adjustment for precision
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_sum_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.full(n_points, 0.5)
        
    # PGD parameters
    alpha = 0.1
    beta = 0.9
    momentum = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient: (1-h) shifted by k_max
        # For full correlation, the gradient shape is same as h
        grad = np.zeros_like(h)
        if k_max < len(h):
            # Shift by k_max from the right
            grad[k_max:] = 1 - h[:-k_max]
        if k_max >= len(h):
            # Shift by (len(h) - k_max) from the left
            offset = k_max - len(h) + 1
            grad[:len(h)-offset] = 1 - h[offset:]
        else:
            # Standard case
            grad[k_max:] = 1 - h[:-k_max]
            
        # Apply Nesterov momentum
        momentum = beta * momentum - alpha * grad
        h_new = h + momentum
        
        # Clipping
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_sum_constraint(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        
        # Update
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new
        iteration += 1
        
        # Adaptive learning rate
        if no_improvement_count > 100:
            alpha *= 0.95
            alpha = max(alpha, 1e-5)
            no_improvement_count = 0
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Periodic check
        if iteration % 10 == 0:
            if time.time() - start_time > budget_s - 2:
                break
    
    return (best_h.tolist(), best_c5, n_points)
