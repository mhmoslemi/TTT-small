import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, dx, maxiter=1000):
    """Perform local optimization using L-BFGS-B."""
    def objective(h_flat):
        h_corrected = h_flat.copy()
        h_corrected = project_to_constraints(h_corrected, n_points / 2)
        return compute_c5(h_corrected, dx)
    
    def gradient(h_flat):
        # Simple finite difference gradient approximation
        eps = 1e-6
        grad = np.zeros_like(h_flat)
        for i in range(len(h_flat)):
            h_plus = h_flat.copy()
            h_plus[i] += eps
            h_minus = h_flat.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    bounds = [(0, 1) for _ in range(len(h))]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': maxiter, 'gtol': 1e-6}, 
                      jac=gradient)
    
    if result.success:
        h_opt = project_to_constraints(result.x, n_points / 2)
        return h_opt
    
    return h

    pop_size = 20
    sigma_initial = 0.1

    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            p = np.array(initial_h_values)

    best_idx = np.argmin(fitness)
