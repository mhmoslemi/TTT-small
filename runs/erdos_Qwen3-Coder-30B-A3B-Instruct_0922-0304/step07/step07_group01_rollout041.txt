import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, c5_target):
    """Compute gradient of the overlap integral approximation."""
    # Compute the gradient using finite differences
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_to_constraints(h_plus, len(h)/2)
        h_minus = project_to_constraints(h_minus, len(h)/2)
        grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Initialize with a step-like function and add some randomness
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Initial evaluation
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute gradient
        grad = compute_gradient(h, dx, best_c5)
        
        # Update velocity with momentum
        velocity = momentum * velocity - learning_rate * grad
        
        # Update parameters
        h = h + velocity
        h = project_to_constraints(h, target_sum)
        
        # Evaluate new solution
        c5 = compute_c5(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
