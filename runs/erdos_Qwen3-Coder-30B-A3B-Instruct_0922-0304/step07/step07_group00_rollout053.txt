import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    """Compute gradient of the objective function."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all overlaps efficiently using correlation
    # overlap[k] = sum h[i] * (1 - h[i+k]) for valid i
    # This is equivalent to correlate(h, 1-h, mode='full')
    overlap = correlate(h, 1 - h, mode="full")
    
    # The maximum overlap k index
    max_k_idx = np.argmax(overlap)
    
    # We want to differentiate the maximum overlap
    # This approximates the gradient
    for i in range(n):
        # Compute partial derivative of overlap with respect to h[i]
        for k in range(-n+1, n):  # indices for the full convolution
            if 0 <= i + k < n:
                # d/dh[i] (h[i] * (1 - h[i+k])) = 1 - 2*h[i+k]
                # But we must consider this for all overlaps
                # For simplicity, we use a proxy gradient based on the maximum
                if k == max_k_idx - (n-1):
                    grad[i] += 1 - 2 * h[i + k] if 0 <= i + k < n else 0

    # Normalize the gradient
    if np.linalg.norm(grad) > 0:
        grad = grad / np.linalg.norm(grad)
    
    return grad
