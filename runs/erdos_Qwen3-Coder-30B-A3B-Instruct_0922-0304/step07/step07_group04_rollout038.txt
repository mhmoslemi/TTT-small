import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-8
    learning_rate = 0.01
    max_iter = int(budget_s * 500)
    
    # Perturbation and gradient computations
    def compute_gradient(h, dx):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        base_c5 = evaluate_overlap(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
                
            c5_plus = evaluate_overlap(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        return grad
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h
        h_new = h.copy()
        # Add gradient-guided noise
        grad = compute_gradient(h, dx)
        noise = np.random.normal(0, learning_rate * T, n_points)
        h_new += noise - 0.1 * grad * learning_rate
        
        # Keep within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new overlap
        c5_new = evaluate_overlap(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_E = c5_new - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        
        # Cooling
        T = max(min_T, T * cooling_rate)
        
        # Adjust learning rate
        learning_rate = max(1e-5, learning_rate * 0.9999)
    
    # Final local gradient-based refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
