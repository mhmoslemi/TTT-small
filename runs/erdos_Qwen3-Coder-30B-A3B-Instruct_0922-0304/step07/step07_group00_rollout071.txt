import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    
    # Frank-Wolfe algorithm
    max_iter = 5000
    iter_count = 0
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx

        # Track best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = iter_count
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Linear approximation gradient
        # For convolution h * (1-h), the gradient for shift k_max is (1-2*h) shifted by k_max
        grad = np.zeros(n_points)
        if k_max < n_points:
            grad[k_max:] = 1 - 2 * h[:n_points - k_max]
        else:
            grad[:n_points - k_max] = 1 - 2 * h[k_max - n_points:]
        
        # Solve linear program: min s^T grad s.t. sum(s) = 0 and 0 <= s <= 1
        # Use simplex method (linprog) to solve
        # In practice, since constraints are simple, we can use coordinate descent
        # but here we use a simple approach:
        # find the most negative component to increase and a most positive to decrease
        # Assume grad is linearized at current point
        
        # Find search direction: descending direction with sum constraint
        s = -grad
        # Ensure sum(s) = 0
        s = s - np.mean(s)
        # Ensure bounds [0,1]
        s = np.clip(s, 0, 1)
        # Renormalize to satisfy integral constraint
        s = normalize_h(s, target_sum)
        
        # Line search
        alpha = 0.5
        max_alpha = 1.0
        min_alpha = 1e-6
        best_alpha = alpha
        
        temp_h = h.copy()
        temp_h = temp_h + alpha * (s - h)
        temp_h = np.clip(temp_h, 0, 1)
        temp_h = normalize_h(temp_h, target_sum)
        temp_c5 = compute_c5(temp_h, dx)
        
        while temp_c5 >= max_overlap and alpha > min_alpha:
            alpha /= 2
            temp_h = h + alpha * (s - h)
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = normalize_h(temp_h, target_sum)
            temp_c5 = compute_c5(temp_h, dx)
            
        if alpha <= min_alpha:
            break
            
        # Update h
        h = h + alpha * (s - h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iter_count += 1
        
        # Termination condition
        if iter_count - last_improvement > 500:
            break
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
