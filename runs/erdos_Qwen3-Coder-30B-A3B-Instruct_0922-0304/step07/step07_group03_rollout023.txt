import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit proportionally
    deficit = target_sum - current_sum
    if deficit > 0:
        # Need to increase values
        h += deficit * h / current_sum
    else:
        # Need to decrease values
        h *= target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):

        h = project_to_constraints(h, target_sum)
    
    # Learning rate and perturbation parameters
    alpha = 0.2
    epsilon = 0.1
    decay_tau = 1000
    
    best_h = h.copy()

        # Compute overlap

        max_overlap = np.max(overlap)
        
        # Compute gradient for update direction
        # We want to reduce max overlap, so we consider the "problematic" correlation values

        # Simplified strategy: gradient is simply 1-h (approximates local negative impact of increasing h)
        grad = -(1 - h)
        
        # Update with gradient descent
        h_candidate = h - alpha * grad

        h_candidate = project_to_constraints(h_candidate, target_sum)

    final_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
