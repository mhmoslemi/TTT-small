import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced local gradient descent approach
    def gradient_step(h_vec, step_size=1e-3, momentum=0.9, reg_lambda=1e-4):
        """Perform gradient step with momentum."""
        # Compute current overlap
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        max_k = np.argmax(overlap)
        max_val = overlap[max_k]
        
        # For gradient approximation, we'll compute partial derivatives
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            
            # Ensure in bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Normalize
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            # Evaluate overlaps
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (overlap_plus[max_k] - overlap_minus[max_k]) / (2 * eps)
        
        # Apply momentum
        if not hasattr(gradient_step, 'velocity'):
            gradient_step.velocity = np.zeros_like(h_vec)
        gradient_step.velocity = momentum * gradient_step.velocity - step_size * grad
        
        new_h = h_vec + gradient_step.velocity
        
        # Apply regularization to maintain constraints
        new_h = new_h - reg_lambda * (np.sum(new_h) - target_sum)
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize
        h_sum = np.sum(new_h)
        if h_sum > 0:
            new_h = new_h * (target_sum / h_sum)
            
        return new_h
    
    # Optimization loop
    iter_count = 0
    max_iter = 3000
    step_size = 1e-3
    momentum = 0.9
    reg_lambda = 1e-4
    
    while time.time() - start_time < budget_s - 2 and iter_count < max_iter:

        old_h = best_h.copy()
        old_c5 = best_c5
        
        # Perform step
        new_h = gradient_step(best_h, step_size, momentum, reg_lambda)
        new_c5 = evaluate_overlap(new_h, dx)

        if new_c5 < best_c5:

        # Update reg_lambda if we're getting constraint violations
        if np.abs(np.sum(best_h) - target_sum) > 1e-3:
            reg_lambda *= 2.0

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
