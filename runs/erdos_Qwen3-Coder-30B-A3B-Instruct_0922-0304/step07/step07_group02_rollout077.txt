import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy sum constraint
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization using CVXPY
    # Variables
    h_var = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum
    ]
    
    # Objective: minimize max_k (h * shift_k(1-h))

        # Try a simple local search for improvement

        improved = False
