import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Local refinement using COBYLA method
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try multiple starting points to avoid local optima
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Random perturbation
        h_init = h + np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init_sum = np.sum(h_init)
        if h_init_sum > 0:
            h_init = h_init * (target_sum / h_init_sum)
            
        try:
            result = minimize(objective, h_init, method='COBYLA', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 500, 'ftol': 1e-8})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
        except:
            pass

    # Try one more optimization with SLSQP if time allows
    if time.time() - start_time < budget_s - 5:
        try:
            result = minimize(objective, best_h, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-8})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
        except:
            pass

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
