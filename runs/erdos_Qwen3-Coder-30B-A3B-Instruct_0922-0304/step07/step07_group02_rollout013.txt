import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.8
    crossover_probability = 0.7
    max_iterations = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(population_size):
        if 'initial_h_values' in globals() and i == 0:
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
        population.append(particle)
        fitness.append(compute_c5(particle, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_solution = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        for i in range(population_size):
            # Select three distinct random individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = project_to_constraint(mutant, target_sum)
            
            # Crossover
            trial = np.copy(population[i])
            jrand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_probability or j == jrand:
                    trial[j] = mutant[j]
            
            # Ensure trial satisfies constraints
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate trial
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                population[i] = trial.copy()
                fitness[i] = trial_c5
                
                # Update global best
                if trial_c5 < best_value:
                    best_value = trial_c5
                    best_solution = trial.copy()
        
        iteration += 1
        
        # Periodic check for early termination
        if best_value <= 0.3807:
            break
    
    return (best_solution.tolist(), best_value, n_points)
