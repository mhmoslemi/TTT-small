import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # DE parameters
    pop_size = 50
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    if 'initial_h_values' in globals():
        # Seed with initial solution
        population.append(h.copy())
        fitness.append(compute_c5(h, dx))
        # Add random variations
        for _ in range(pop_size - 1):

            h_rand += np.random.normal(0, 0.01, n_points)

            h_rand = normalize_h(h_rand, target_sum)
            population.append(h_rand.copy())
            fitness.append(compute_c5(h_rand, dx))
    else:
        # Random population
        for _ in range(pop_size):
            h_rand = np.random.rand(n_points)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = normalize_h(h_rand, target_sum)
            population.append(h_rand.copy())
            fitness.append(compute_c5(h_rand, dx))
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]

        new_population = []

            candidates = list(range(pop_size))

            mutant = population[a] + mutation_factor * (population[b] - population[c])
