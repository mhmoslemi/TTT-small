import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population using provided initial_h_values if available
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add some variations to the initial solution
        for _ in range(population_size // 5):
            variation = h + np.random.normal(0, 0.03, n_points)
            variation = project_to_feasible(variation, target_sum)
            population.append(variation)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add some variations to the initial solution
        for _ in range(population_size // 5):
            variation = h + np.random.normal(0, 0.03, n_points)
            variation = project_to_feasible(variation, target_sum)
            population.append(variation)
    
    # Fill remaining population with random individuals
    while len(population) < population_size:
        h_random = np.random.rand(n_points)
        h_random = project_to_feasible(h_random, target_sum)
        population.append(h_random)
    
    # PSO parameters
    inertia_w = 0.8
    cognitive_coef = 1.5
    social_coef = 1.5
    max_iter = 60
    
    # Initialize PSO
    particles = np.array(population)
    velocities = np.random.uniform(-0.1, 0.1, (population_size, n_points))
    best_positions = particles.copy()
    best_values = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(best_values)
    global_best = particles[global_best_idx].copy()
    global_best_value = best_values[global_best_idx]
    
    # PSO iterations
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                           cognitive_coef * r1 * (best_positions[i] - particles[i]) +
                           social_coef * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate new position
            new_value = compute_c5(particles[i], dx)
            if new_value < best_values[i]:
                best_values[i] = new_value
                best_positions[i] = particles[i].copy()
                if new_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = new_value
                    stagnation_count = 0
        
        # Update inertia weight
        inertia_w *= 0.98
        
        # Check for stagnation
        stagnation_count += 1
        if stagnation_count >= max_stagnation:
            break
    
    # Final refinement: local search with gradient descent
    best_h = global_best.copy()
    best_c5 = global_best_value
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            new_c5 = compute_c5(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                break
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
