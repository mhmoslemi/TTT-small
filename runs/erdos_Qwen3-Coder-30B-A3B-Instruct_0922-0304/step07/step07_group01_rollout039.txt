import numpy as np
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
    # Using correlation to compute the overlap integral
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint and bounds"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to exact sum
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Simulated Annealing parameters
    T0 = 0.1 * compute_c5(h, dx)
    T = T0
    alpha = 0.99
    max_iter = 10000
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Generate candidate solution by perturbing current solution
        # Add uniform noise
        noise = np.random.uniform(-T, T, n_points)
        h_candidate = h + noise
        # Project to constraints
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Compute candidate C5
        candidate_c5 = compute_c5(h_candidate, dx)
        
        # Accept or reject the candidate
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_candidate
            current_c5 = candidate_c5
            
            # Update the best solution
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Cool down the temperature
        T *= alpha
        iter_count += 1
        
        # Check if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Return the best solution found
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
