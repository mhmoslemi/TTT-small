import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def gradient_descent_optimization(h, dx, max_iter=10000, lr=0.01, tol=1e-8, patience=1000):
    """Optimize h using gradient descent with adaptive learning rate."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    patience_counter = 0
    current_lr = lr
    
    for i in range(max_iter):
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        
        # Gradient approximation for each point
        grad = np.zeros_like(h)
        n = len(h)
        
        # Compute finite difference gradient
        for j in range(n):
            # Forward difference approximation
            h_plus = h.copy()
            h_plus[j] += 1e-6
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, n/2)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[j] = (overlap_plus[max_idx] - max_overlap) / 1e-6
        
        # Update step with learning rate
        h = h - current_lr * grad
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n/2)
        
        # Check for improvement
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adaptive learning rate decay
        if patience_counter > 100:
            current_lr *= 0.999
            patience_counter = 0
            
        if patience_counter > patience:
            break
            
        if best_c5 <= 0.3807:
            break
            
    return best_h, best_c5
