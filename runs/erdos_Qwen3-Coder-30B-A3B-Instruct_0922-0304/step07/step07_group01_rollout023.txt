import numpy as np
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Alternative initialization
        h = np.ones(n_points) * 0.5
        h = project_to_constraints(h, target_sum)
    
    # Start with initial solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try different approaches to improve solution
    num_iterations = 500
    for it in range(num_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random perturbation
        perturbation = np.random.normal(0, 0.02, n_points)
        h_trial = h + perturbation
        h_trial = project_to_constraints(h_trial, target_sum)
        
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
            
    # Try convex optimization in Fourier domain (even simpler approach)
    if time.time() - start_time < budget_s - 5:
        # Use a simplified convex approach to further refine
        h_refined = best_h.copy()
        for refine_iter in range(100):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Simple gradient descent moves
            step_size = 0.001
            # Small random steps in constrained space
            h_new = h_refined + np.random.normal(0, 0.005, n_points)
            h_new = project_to_constraints(h_new, target_sum)
            
            c5_new = compute_c5(h_new, dx)
            if c5_new < compute_c5(h_refined, dx):
                h_refined = h_new
                
        best_c5 = compute_c5(h_refined, dx)
        best_h = h_refined
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
