import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute exact C5 value using correlation."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the penalty function (approximation)."""
    # Compute correlations manually for gradient
    grad = np.zeros_like(h)
    # For simplicity, we compute exact correlation with 1-h,
    # then approximate gradient using finite differences
    # This simulates the derivative contribution
    h_extended = np.concatenate([h, h[:n_points-1]])  # Circular padding
    one_minus_h_extended = np.concatenate([1-h, (1-h)[:n_points-1]])
    
    # Compute overlap for all lags
    overlap = np.zeros(2 * n_points - 1)
    for i in range(2 * n_points - 1):
        overlap[i] = np.sum(h_extended[i:i+n_points] * one_minus_h_extended[:n_points])
    
    # Find maximum overlap and its position
    max_pos = np.argmax(overlap)
    max_overlap = overlap[max_pos]
    
    # Construct gradient contribution at max overlap
    # Since position depends on conv:
    # For max_pos in [0, 2*n_points - 2], 
    # max_overlap arises from sum_{j=0}^{n-1} h[j] * (1 - h[j + max_pos - n + 1])
    # The gradient of this term wrt h[i] is:
    # if i is in the active set: - for h[i + max_pos - n + 1] term (times dx ignored)
    # if i + max_pos - n + 1 = j then += 1 if j in [0, n-1]
    # Let's compute explicitly the gradient for i = k, if that's relevant
    # This is approximate, but better than no gradient at all
    
    # Approximation: compute gradient using finite difference
    eps = 1e-6
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = normalize_h(h_plus, n_points / 2)
        c5_plus = compute_c5_exact(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = normalize_h(h_minus, n_points / 2)
        c5_minus = compute_c5_exact(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from global initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    # Adam-like optimization with fixed learning rate
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Iterate until time exhausted or target reached
    iter_count = 0
    no_improve_count = 0
    while time.time() - start_time < budget_s - 1 and best_c5 > 0.3807:
        # Compute gradient (using finite differences)
        grad = compute_gradient(h, dx, n_points)
        
        # Update Adam momentum
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        
        # Bias correction
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        
        # Update parameters
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Compute new C5
        c5 = compute_c5_exact(h, dx)
        
        # Update best solution if improving
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adjust learning rate if not improving
        if no_improve_count % 200 == 0:
            learning_rate *= 0.95
            
        iter_count += 1
        
        # Early stop if we've gone too many steps without improvement
        if no_improve_count > 1000:
            break
    
    if best_c5 <= 0.3807:
        # Ensure final solution is properly normalized and clipped
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
        best_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
