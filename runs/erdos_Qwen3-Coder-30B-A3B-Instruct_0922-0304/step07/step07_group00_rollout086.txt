import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a simple initial solution
        h = np.full(n_points, 0.5)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Create basis functions
    def create_basis_functions(n_basis):
        basis = []
        for i in range(n_basis):
            b = np.zeros(n_points)
            # Create a simple pulse with varying width and position
            width = max(2, n_points // (4 + i % 4))
            start_pos = (i * n_points // (n_basis * 2)) % (n_points - width)
            b[start_pos:start_pos + width] = 1.0
            # Normalize the basis function to match integral constraint
            if np.sum(b) > 0:
                b = b * (target_sum / np.sum(b))
            basis.append(b)
        return basis
    
    # Precompute basis overlaps
    def compute_basis_overlaps(basis_funcs):
        n_basis = len(basis_funcs)
        A = np.zeros((n_basis, n_basis, 2*n_points - 1))
        for i, bi in enumerate(basis_funcs):
            for j, bj in enumerate(basis_funcs):
                A[i,j,:] = correlate(bi, bj, mode="full")
        return A
    
    # Start with 16 basis functions
    n_basis = 16
    basis_funcs = create_basis_functions(n_basis)
    A = compute_basis_overlaps(basis_funcs)
    
    # Resolve the basis components to approximate the initial h
    def solve_basis_decomp(h, basis_funcs, A):
        n_basis = len(basis_funcs)
        # Initialize lambda
        lamb = Variable(n_basis)
        # Objective: minimize squared difference
        obj = Minimize(sum_squares(basis_funcs @ lamb - h))
        # Constraints
        constraints = [sum(lamb) == 1, lamb >= 0]
        
        # Solve the problem
        prob = Problem(obj, constraints)
        prob.solve(solver=ECOS, verbose=False)
        
        if prob.status == 'optimal':
            return lamb.value
        else:
            return np.ones(n_basis) / n_basis  # fallback
    
    # Initialize lambda
    lamb = solve_basis_decomp(h, basis_funcs, A)
    
    # Define the dual variables and outer loop
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    iters_no_improvement = 0
    max_no_improvement = 100
    
    # Main loop
    while time.time() - start_time < budget_s - 2:
        # Update h from the basis functions
        h = basis_funcs @ lamb
        
        # Compute the current overlap
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iters_no_improvement = 0
        else:
            iters_no_improvement += 1
            
        # Stop if we've reached or exceeded our goal
        if best_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if iters_no_improvement > max_no_improvement:
            break
            
        # Form the dual problem objective based on current h
        # Need to compute the overlap matrix
        B = np.zeros((2*n_points - 1, n_basis, n_basis))
        for k in range(2*n_points - 1):  # number of shifts
            # We approximate h(x) as sum_i lambda_i * basis_i(x)
            for i in range(n_basis):
                for j in range(n_basis):
                    # b_i and b_j can be displaced by k
                    # overlap kernel effect: int b_i(x) * b_j(x+k) dx
                    # This requires indexing to compute at shift k
                    # We'll approximate with correct indices
                    B[k, i, j] = A[i, j, k]
        
        # Compute true overlap for this lambda solution
        overlap_totals = np.zeros(2*n_points - 1)
        for k in range(2*n_points - 1):
            # Lambda is updated to reduce this value
            temp = np.zeros((n_basis, n_basis))
            # Compute for k
            for i in range(n_basis):
                for j in range(n_basis):
                    temp[i, j] = B[k, i, j] * (1 - A[j, j, k])  # this is a best guess
            overlap_totals[k] = np.sum(lamb[:, None] * temp * lamb[None, :])
        
        # Estimate the max overlap as the worst case (simplified)
        maxval = np.max(overlap_totals)
        
        # Update Lambda using a direct convex approach
        # We solve: minimize sum_k mu_k * (sum_ij lamb_i lamb_j B[k, i, j]) over lamb
        # subject to sum(lamb) = 1, lamb >= 0
        # For numerical stablity and completely simplify, use just one update
        # 

        # Compute the gradient of the objective w.r.t. Lambda
        # Guide: if I fix the first few (or middle) of lamb, I can compute what other lamb contribute
        # This approach applies the log-sum-exp trick
        # But we can just use cvxpy for simplicity
        
        # Create new dual weight vector
        weights = np.ones(2*n_points - 1)
        
        # Minimize sum_k weights[k] * \sum_{i,j} lamb_i lamb_j * A[i,j][k] ?
        # This is actually the max of overlap integral for each k.
        # For simplicity, let's try linearization using an initial h,
        # Then solve the quadratic optimization:
        # minimize \sum_k weights_k * sum_ij lamb_i lamb_j * (A[i,j][k] - A[i,j][k]*A[j,j][k])
        weights = weights / np.sum(weights)
        
        # Duality - use a relaxation approach
        # Dual updates: this is not direct but constaints too.
        # For a constrained optimization, if we keep a Lagrange dual vector and update:
        #  We need to compute the actual constraint violation
        # but here yields strong duality this might be ok
        
        # Do this simple approach for now:
        # Constraint for one k (overlaps)
        # max_k int h(x)(1-h(x+k)) dx = ?
        
        # Solve for new lambda updating from the metric between current overlap and what's achievable
        # Compute gradient dy/d lamb for max_k over all k positions of the entries:
        # We do simple descent
        # In principle, gradient = 2 * B * lambda (or close to)  
        
        # Somewhat naively but specific work:
        # Compute indicator over overlap
        B_sum = np.zeros((n_basis, n_basis))
        # B_sum = sum_k w_k * B_k
        for k in range(2*n_points - 1):
            B_sum += weights[k] * B[k, :, :]
        
        # Approximate the gradient of the overlap term for each k separately
        # Note there isn't a clean closed form for the gradient of the max,
        # But under suitable conditions, gradient uplifts direction of that entry
        # Let's pick the k that currently maximizes overlap, and adjust lamb accordingly
        
        max_k_idx = np.argmax(overlap_totals)
        # Gradient of total overlap is B[max_k_idx, :] * lamb with proper indexing post-processing
        grad = 2 * B[max_k_idx, :, :] @ lamb
        # Adjust lamb to reduce this gradient (simple descent)
        step = 0.05
        
        lamb_new = lamb - step * grad
        # Enforce simplex constraint (maintain sum to 1)
        lamb_new = np.maximum(lamb_new, 0.0)
        lamb_new = lamb_new / np.sum(lamb_new)
        lamb = lamb_new
        
        # Possibly add new basis functions if old basis set is inadequate
        # Based on large constraint violations (this generally overcomplicates)
        # Temporarily skip augmentation
        
    # Refine best h
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
