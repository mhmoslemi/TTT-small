import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create diverse starting population
        particles.append(h.copy())
        for _ in range(population_size // 3):
            perturbation = np.random.normal(0, 0.02, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
            particles.append(h_perturbed.copy())
        # Fill the rest
        for _ in range(population_size - len(particles)):
            h_random = np.random.rand(n_points)
            h_random = np.clip(h_random, 0, 1)
            h_random = normalize_h(h_random, target_sum)
            particles.append(h_random.copy())
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create diverse starting population
        particles.append(h.copy())
        for _ in range(population_size // 3):
            perturbation = np.random.normal(0, 0.02, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
            particles.append(h_perturbed.copy())
        # Fill the rest with random feasible solutions
        for _ in range(population_size - len(particles)):
            h_random = np.random.rand(n_points)
            h_random = np.clip(h_random, 0, 1)
            h_random = normalize_h(h_random, target_sum)
            particles.append(h_random.copy())

    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # Initialize personal bests
    pbest = [particle.copy() for particle in particles]
    pbest_fitness = [compute_c5(p, dx) for p in pbest]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    max_iterations = 60
    stagnation_count = 0
    stagnation_threshold = 10
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    stagnation_count = 0
        
        stagnation_count += 1
        if stagnation_count > stagnation_threshold:
            # Reset some particles that haven't improved
            for i in range(population_size):
                if np.random.rand() < 0.2:
                    h_random = np.random.rand(n_points)
                    h_random = np.clip(h_random, 0, 1)
                    h_random = normalize_h(h_random, target_sum)
                    particles[i] = h_random.copy()
                    velocities[i] = np.zeros(n_points)
            stagnation_count = 0
            
        if gbest_fitness <= 0.3807:
            break

    # Final refinement with gradient descent for the best individual
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Run local refinement several times with different restarts
    best_h = gbest.copy()
    best_c5 = gbest_fitness
    
    for _ in range(10):  # Run multiple rounds
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local search from current best
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5

    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
