import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Make sure we don't go over budget before the first check
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 1.0:
        return (best_h.tolist(), best_c5, n_points)
    
    # Local quadratic optimization
    max_iterations = 60
    patience = 0
    patience_limit = 10
    
    for iteration in range(max_iterations):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1.0:
            break
            
        # Compute gradient approximation
        current_c5 = compute_c5(h, dx)
        epsilon = 1e-4
        
        # Compute gradient with finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Formulate quadratic program
        # minimize 0.5 * d^T * H * d + grad^T * d
        # subject to 0 <= d <= 1, sum(d) = 0
        
        # We solve for the descent direction. The Hessian approximation is assumed to be identity.
        # This is a simplified approach making it computationally efficient.
        P = np.eye(n_points)
        q = grad
        
        # CVXPY quadratic program
        d = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            d >= 0,
            d <= 1,
            cp.sum(d) == 0
        ]
        
        # Objective - minimize quadratic approximation
        objective = cp.Minimize(0.5 * cp.quad_form(d, P) + q.T @ d)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            descent_dir = d.value
        except:
            # If QP fails, use gradient descent
            descent_dir = -grad
            # Project to feasible set
            descent_dir = descent_dir - np.mean(descent_dir)  # Zero mean
        
        # Line search
        alpha = 1.0
        max_alpha = 1.0
        min_alpha = 1e-6
        backtracking_factor = 0.5
        max_line_search = 20
        
        for _ in range(max_line_search):
            h_new = h + alpha * descent_dir
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < current_c5:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                break
            else:
                alpha *= backtracking_factor
                if alpha < min_alpha:
                    break
        
        if patience >= patience_limit:
            break
            
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
