```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    h = h * (target_sum / np.sum(h))  # Normalize to exact sum
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Early termination if already good
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Iterative improvement using quadratic approximation
    patience = 0
    max_patience = 10
    max_iterations = 60
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        base_c5 = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] = max(0.0, h_minus[i] - eps)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Quadratic approximation: f(x + d) ≈ f(x) + grad^T * d + 0.5 * d^T * H * d
        # We don't compute the full Hessian, just use a diagonal approximation
        
        # Create and solve quadratic program for descent direction
        # min (1/2) * d^T * H * d + grad^T * d
        # s.t. 0 <= h + d <= 1, sum(d) = 0, sum(h + d) = target_sum
        
        # Let d = new_h - h, so sum(d) = 0, sum(h + d) = sum(h) = target_sum
        # So we're solving: min (1/2) * d^T * H * d + grad^T * d 
        # s.t 0 <= h + d <= 1 
        
        # Define variables
        d = cp.Variable(n_points)
        
        # Approximate Hessian as diagonal matrix with small values to encourage 
        # small steps and reduce oscillation
        H_diag = np.ones(n_points) * 1e-6  # Very small diagonal
        objective = cp.Minimize(0.5 * cp.quad_form(d, np.diag(H_diag)) + grad.T @ d)
        
        # Constraints
        constraints = [
            h + d >= 0,
            h + d <= 1
        ]
        
        # Create and solve QP
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                step_dir = d.value
            else:
                # Fallback to gradient descent with small step
                step_dir = -grad * 0.01
        except:
            # Fallback to gradient descent
            step_dir = -grad * 0.01
            
        # Perform line search along the direction
        alpha = 1.0
        max_alpha = 1.0
        step_size = 0.1
        
        for _ in range(20):  # Limit line search iterations
            new_h = h + alpha * step_dir
            
            # Clip to [0,1] range
            new_h = np.clip(new_h, 0, 1)
            
            # Project to sum = target_sum
            new_h = new_h * (target_sum / np.sum(new_h))
            
            new_c5 = compute_c5(new_h, dx)
            
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                patience = 0
                h = new_h
                break
            else:
                alpha *= step_size
                
        if alpha < 1e-8:
            patience += 1
            if patience >= max_patience:
                break
        else:
            patience = 0
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final refinement using scipy optimize
    def objective(h_vec):
        h = h_vec.copy()
        h = h * (target_sum / np.sum(h))  # Maintain sum constraint 
        return compute_c5(h, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Use the current best as starting point
    h_refined = best_h.copy()
    h_refined = h_refined * (target_sum / np.sum(h_refined))
    
    result = minimize(objective, h_refined, method='SLSQP', 
                      bounds=bounds, constraints=constraint, 
                      options={'maxiter': 50, 'ftol': 1e-8})
    
    if result.success:
        final_h = result.x.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (target_sum / np.sum(final_h))
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
```
