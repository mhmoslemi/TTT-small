import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, max_iter=1000):
    """Apply local optimization to improve a candidate."""
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_func(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter}
    )
    
    if result.success:
        return project_to_constraints(result.x, target_sum)
    return project_to_constraints(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial population
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # ES parameters
    population_size = 20
    sigma_initial = 0.1
    sigma_final = 0.01
    max_generations = 2000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbed = h + np.random.normal(0, sigma_initial, n_points)
        population.append(project_to_constraints(perturbed, target_sum))
    
    # Evaluate initial population
    fitnesses = [compute_c5(ind, dx) for ind in population]
    
    best_individual = population[np.argmin(fitnesses)].copy()
    best_fitness = min(fitnesses)
    
    # ES loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update sigma
        sigma = sigma_initial - (sigma_initial - sigma_final) * generation / max_generations
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_size = population_size // 2
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate new population with mutation
        new_population = elite.copy()
        
        for _ in range(population_size - elite_size):
            parent = elite[np.random.randint(0, len(elite))]
            mutated = parent + np.random.normal(0, sigma, n_points)
            new_individual = project_to_constraints(mutated, target_sum)
            new_population.append(new_individual)
            
        # Local optimization for best individuals
        for i, individual in enumerate(new_population):
            if compute_c5(individual, dx) < 0.381:
                new_population[i] = local_optimize(individual, target_sum, dx)
        
        # Evaluate new population
        fitnesses = [compute_c5(ind, dx) for ind in new_population]
        
        # Update best individual
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_individual = new_population[current_best_idx].copy()
            best_fitness = fitnesses[current_best_idx]
            
        # Early termination
        if best_fitness <= 0.3807:
            break
            
        # Replace population
        population = new_population.copy()
    
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
