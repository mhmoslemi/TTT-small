import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, dx, target_sum, maxiter=1000):
    """Use L-BFGS-B for local optimization with box constraints."""
    def objective(x):
        return compute_c5(x, dx)
    
    def jac(x):
        # Numerical gradient (simple finite difference)
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = project_to_constraints(x_plus, target_sum)
            x_minus = x.copy()
            x_minus[i] -= eps
            x_minus = project_to_constraints(x_minus, target_sum)
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad
    
    bounds = [(0, 1) for _ in range(len(h))]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-12})
    return result.x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # ES parameters
    pop_size = 20
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_generations = 2000
    
    # Initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_mut = h + np.random.normal(0, mutation_rate_start, n_points)
        h_mut = project_to_constraints(h_mut, target_sum)
        population.append(h_mut)
    
    # Evaluate initial population
    fitness = [compute_c5(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Evolutionary strategy loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection: Keep top 50%
        selected_indices = np.argsort(fitness)[:pop_size//2]
        selected_population = [population[i] for i in selected_indices]
        selected_fitness = [fitness[i] for i in selected_indices]
        
        # Update mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * generation / max_generations
        
        # Create new population
        new_population = selected_population.copy()
        new_fitness = selected_fitness.copy()
        
        # Generate offspring
        for _ in range(pop_size - len(selected_population)):
            parent = selected_population[np.random.randint(0, len(selected_population))]
            offspring = parent + np.random.normal(0, mutation_rate, n_points)
            offspring = project_to_constraints(offspring, target_sum)
            new_population.append(offspring)
        
        # Evaluate new population
        new_fitness = [compute_c5(ind, dx) for ind in new_population]
        
        # Update best if needed
        min_idx = np.argmin(new_fitness)
        if new_fitness[min_idx] < best_fitness:
            best_individual = new_population[min_idx].copy()
            best_fitness = new_fitness[min_idx]
        
        # Local optimization for good candidates
        for i, ind in enumerate(new_population):
            if new_fitness[i] < 0.381:  # Only optimize promising candidates
                if time.time() - start_time > budget_s - 1:
                    break
                local_result = local_optimize(ind, dx, target_sum)
                local_c5 = compute_c5(local_result, dx)
                if local_c5 < new_fitness[i]:
                    new_population[i] = local_result
                    new_fitness[i] = local_c5
                    if local_c5 < best_fitness:
                        best_individual = local_result.copy()
                        best_fitness = local_c5
        
        # Update population for next generation
        population = new_population
        fitness = new_fitness
        
        # Early stopping
        if best_fitness <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
