import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm
    swarm_size = 100
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # 40% of particles are variations of the initial solution
        for _ in range(int(swarm_size * 0.4)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation.copy())
        # Fill with random particles
        for _ in range(swarm_size - len(particles)):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand.copy())
    else:
        # All random particles
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(swarm_size):
        velocities.append(np.random.normal(0, 0.01, n_points))
    
    # Initialize personal bests
    personal_bests = [particle.copy() for particle in particles]
    personal_best_scores = [compute_c5(particle, dx) for particle in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO parameters
    w = 0.8  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    max_iter = 60
    iter_count = 0
    
    # Main PSO loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_bests[i] - particles[i])
            social = c2 * r2 * (global_best - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply boundary constraints
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Ensure sum constraint
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            current_score = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_bests[i] = particles[i].copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = particles[i].copy()
            
            # Early stopping
            if global_best_score <= 0.3807:
                break
        
        # Update parameters
        w = 0.8 * (0.98 ** iter_count)
        iter_count += 1
        
        if global_best_score <= 0.3807:
            break
    
    # Local refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    refined_h = global_best.copy()
    best_refined_c5 = global_best_score
    
    # Run several rounds of local optimization 
    for attempt in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, refined_h, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            new_score = compute_c5(refined_h, dx)
            if new_score < best_refined_c5:
                best_refined_c5 = new_score
                global_best = refined_h.copy()
        else:
            break
    
    final_h = global_best.copy()
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
