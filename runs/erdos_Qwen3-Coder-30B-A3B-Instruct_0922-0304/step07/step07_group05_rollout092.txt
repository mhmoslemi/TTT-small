import numpy as np
from scipy.fft import fft, ifft
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    # Using correlate for the cross-correlation with 1-h
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create smooth step function as initial guess
        h = np.zeros(n_points)
        # Centered triangular pulse
        mid = n_points // 2
        width = n_points // 4
        for i in range(n_points):
            if i < mid - width or i > mid + width:
                h[i] = 0
            elif i <= mid:
                h[i] = (i - (mid - width)) / width
            else:
                h[i] = (1 - (i - mid) / width)
        # Normalize to required sum
        h = h * (target_sum / np.sum(h))
        
    # Iterative refinement
    max_iterations = 50
    alpha = 0.1
    decay = 0.95
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        max_lag = np.argmax(overlap)
        max_overlap = np.max(overlap)
        
        # Weight function based on overlap
        # Weights are inversely proportional to current overlap values
        weight = np.zeros(2 * n_points - 1)
        weight[max_lag] = 1.0
        # Smooth weighting - less aggressive
        for i in range(1, 10):
            if max_lag + i < len(weight):
                weight[max_lag + i] = 1.0 / (1.0 + i / 5.0)
            if max_lag - i >= 0:
                weight[max_lag - i] = 1.0 / (1.0 + i / 5.0)
                
        # Apply the weighting to compute weighted update
        # Theoretical update: adjust current h to reduce overlap at max_lag
        # We do a simplified frequency-domain-inspired update:
        # High-frequency component values are adjusted more
        # Simple phase shift can be added to meet the goal
        freq_h = fft(h)
        # Create a high-pass component (remove DC)
        freq_h[0] = 0  # Zero DC component
        # Inverse FFT and add to original to create modified version
        modulation = np.real(ifft(freq_h * 1j)).real  # Frequency modulation
        modulation = modulation / (np.max(np.abs(modulation)) + 1e-10)
        
        h_new = h - alpha * modulation
        
        # Ensure h remains in feasible range
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to required sum
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Update h
        h = h_new.copy()
        
        # Update best solution if better found
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
        alpha *= decay  # Decrease step size
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
