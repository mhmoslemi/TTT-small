import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing approach
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    min_T = 1e-8
    max_iter = 100000
    lr = 0.01
    penalty_weight1 = 100.0
    penalty_weight2 = 100.0
    
    def compute_energy(h_vec, target_sum, dx):
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        c5 = np.max(overlap) * dx
        
        # Constraint penalties
        clipped = np.clip(h_vec, 0, 1)
        bound_penalty = np.sum((h_vec - clipped)**2)
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        
        total_energy = c5 + penalty_weight1 * bound_penalty + penalty_weight2 * sum_penalty
        return total_energy, c5
    
    def perturb(h_vec, T, lr):
        # Perturb with Gaussian noise scaled by temperature
        noise = np.random.normal(0, lr * T, n_points)
        new_h = h_vec + noise
        return np.clip(new_h, 0, 1)
    
    current_h = h.copy()
    current_energy, current_c5 = compute_energy(current_h, target_sum, dx)
    
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
        
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        # Generate new candidate
        new_h = perturb(current_h, T, lr)
        
        # Normalize sum
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        
        # Compute energy of new point
        new_energy, new_c5 = compute_energy(new_h, target_sum, dx)
        
        # Accept or reject
        if new_energy < current_energy or np.random.rand() < np.exp(-(new_energy - current_energy) / T):
            current_h = new_h
            current_energy = new_energy
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Update temperature
        T = max(min_T, T * cooling_rate)
        
        iter_count += 1
    
    # Final gradient-based refinement
    def objective(h_vec):
        return evaluate_overlap(h_vec, dx)
    
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Use the best result so far as starting point
    refined_h = best_h.copy()
    try:
        result = minimize(objective, refined_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
