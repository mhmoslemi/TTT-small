import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        scale = target_sum / current_sum
        h = h * scale
    else:
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Parameters
    max_iter = 3000
    alpha = 0.2
    alpha_decay = 1000
    epsilon = 0.1
    epsilon_decay = 1000
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main loop
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation to get overlap
        c = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(c)
        
        # Gradient step: try to reduce the maximum overlap
        # Gradient with respect to h: -delta(k) where k is the shift of max overlap
        # We approximate by adding small negative gradient at relevant positions
        gradient = np.zeros(n_points)
        # Determine range of indices that affect the max overlap
        # We could be smarter here, but a simple approximation works well:
        # Add gradients for indices in the top 5% of overlap region
        threshold = np.percentile(c, 95)
        relevant_indices = np.where(c >= threshold)[0]
        if len(relevant_indices) > 0:
            # Approximate gradient as negative one-hot at max-overlap position
            # This is a crude but effective approach
            k = max_idx - (n_points - 1)  # The actual shift
            # Add gradient in reverse direction
            for idx in relevant_indices:
                rel_idx = idx - (n_points - 1)  # Convert to shift
                if abs(rel_idx - k) < 10:  # Only focus on nearby indices
                    # Derivative term should be -sum of (1-h_{i+k}) over all i 
                    # But we compute it for a single sample
                    gradient[idx - (n_points - 1) + (n_points - 1)] -= 1.0
                    
        # Also consider smooth gradient
        gradient_h = - (1 - h)  # This is not quite correct, but a good start

        # Simple gradient descent step
        h_candidate = h - alpha * gradient_h
        
        # Add perturbation to escape local minima
        perturb = epsilon * (2 * np.random.rand(n_points) - 1)
        h_candidate += perturb
        
        # Apply constraints
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        new_c5 = compute_c5(h_candidate, dx)
        
        # Update if better
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            alpha = max(0.001, alpha * 0.99)
            epsilon = max(0.01, epsilon * 0.99)
        else:
            # Revert to previous h, reduce learning rate
            alpha *= 0.995
            epsilon *= 0.995
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate decay
        alpha *= (1.0 - 1.0 / (1.0 + t / alpha_decay))
        epsilon *= (1.0 - 1.0 / (1.0 + t / epsilon_decay))
        
    return (best_h.tolist(), best_c5, n_points)
