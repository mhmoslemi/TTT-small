import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient-based optimization with constraint handling
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    def gradient(h_vector):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vector)
        for i in range(len(h_vector)):
            h_plus = h_vector.copy()
            h_plus[i] += eps
            h_minus = h_vector.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Use projected gradient descent
    max_iter = 2000
    lr = 0.01
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = gradient(best_h)
        
        # Update step (projected gradient descent)
        new_h = best_h - lr * grad
        
        # Project to bounds [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Project to sum constraint
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        
        # Check improvement
        new_c5 = evaluate_overlap(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            lr = max(1e-6, lr * 0.999)  # Adaptive learning rate
        else:
            lr *= 0.9  # Reduce learning rate if no improvement
            
        # Early stopping based on improvement threshold
        if abs(new_c5 - best_c5) < 1e-8:
            break
    
    # Fine-tune with scipy optimization
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def obj(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(obj, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-10})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
