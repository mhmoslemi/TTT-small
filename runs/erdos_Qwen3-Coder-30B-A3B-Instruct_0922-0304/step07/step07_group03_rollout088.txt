import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Optimization parameters
    max_iter = 20000
    alpha = 0.1
    decay = 1000.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        # Gradient based on maximum value
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Approximate gradient using finite differences
            if i < len(h) - 1:
                grad[i] = corr[max_idx] - corr[max_idx + 1]
            else:
                grad[i] = 0
        
        # Update with gradient descent
        h = h - alpha * grad
        h = project_to_constraints(h, target_sum)
        
        # Update learning rate
        alpha = 0.1 / (1 + iter_num / decay)
        
        # Keep best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final projection
    final_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
