import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based optimization
    def compute_gradient(h, dx):
        # Compute gradient using finite differences
        # Approximate ∂C5/∂h_i = ∂/∂h_i max_k ∫ h(x)(1-h(x+k)) dx
        # For each point i, we compute contributions from all shifts where i occurs
        
        n = len(h)
        grad = np.zeros(n)
        
        # For each point i
        for i in range(n):
            # For all possible shifts k
            for k in range(-n+1, n):
                # j = (i - k) % n  -> h[i] influence on (1-h[j])
                j = (i - k) % n
                if 0 <= j < n:
                    # contribution: 1 - h[j] (if we shift by k)
                    grad[i] += (1 - h[j]) * dx
        
        return grad
    
    # Optimized gradient descent with adaptive learning rate
    def optimize_h(h, max_iter=1000):
        h = h.copy()
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        
        # Initialize optimizers
        momentum = 0.9
        vel = np.zeros_like(h)
        adaptive_lr = 1e-3
        reg_lambda = 1e-4
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute C5 for current h
            c5_current = evaluate_overlap(h, dx)
            
            if c5_current < best_c5:
                best_h[:] = h[:]
                best_c5 = c5_current
            
            # Compute gradient
            grad = compute_gradient(h, dx)
            
            # Adjust for constraints with regularization
            penalty = np.sum(h) - target_sum
            reg_grad = np.ones_like(h) * penalty * reg_lambda
            
            # Combine gradients
            total_grad = grad + reg_grad
            
            # Update with momentum
            vel = momentum * vel - adaptive_lr * total_grad
            h += vel
            
            # Project onto feasible set [0,1] and enforce sum constraint
            h = np.clip(h, 0, 1)
            
            # Adjust to maintain sum constraint with a small tolerance
            current_sum = np.sum(h)
            if abs(current_sum - target_sum) > 1e-6:
                h = h * (target_sum / current_sum)
            
            # Adapt learning rate based on improvement in last few steps
            if iteration % 10 == 0:
                if iteration > 0 and c5_current > evaluate_overlap(h, dx) - 1e-10:
                    adaptive_lr *= 0.8
                else:
                    adaptive_lr = min(adaptive_lr * 1.05, 1e-2)
                    
            # Increase regularization if needed
            if abs(np.sum(h) - target_sum) > 1e-3:
                reg_lambda = min(reg_lambda * 1.1, 1.0)
        
        return h
    
    # Apply optimization with more aggressive search
    ref_h = optimize_h(best_h, 1000)
    
    # Evaluate final result
    final_c5 = evaluate_overlap(ref_h, dx)
    if final_c5 < best_c5:
        best_h = ref_h
        best_c5 = final_c5
    
    # Return result in required format
    return (best_h.tolist(), best_c5, n_points)
