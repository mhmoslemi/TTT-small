import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    max_iter = 20000
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    lr_schedule = lambda t: 0.1 / (1 + t/1000)
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values - sawtooth-like pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.sin(np.pi * i / n_points))
        h = project_to_constraints(h, target_sum)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation optimization
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        c = correlate(h, 1 - h, mode="full")
        
        # Compute surrogate gradient
        # Simplified gradient for convex relaxation:
        # ∇f = h ⋆ (1-h) where f = ½ \|h ⋆ (1-h)\|²  
        # For numerical stability and efficiency, we use 
        # a scaled version of this gradient.
        grad = c * dx  # Approximate gradient
        
        # Update via projected gradient descent
        lr = lr_schedule(iter_num)
        h_new = h - lr * grad
        
        # Project to feasible region
        h = project_to_constraints(h_new, target_sum)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final refinement - fine-grained search around best
    if best_c5 > 0.3807:
        # Try fine-tuning with a local optimizer
        h_refined = best_h.copy()
        n_refine = n_points * 2  # Increase resolution
        x_refined = np.linspace(0, 2, n_refine)
        h_refined_interp = np.interp(x_refined, np.linspace(0, 2, n_points), h_refined)
        h_refined_interp = project_to_constraints(h_refined_interp, n_refine/2)
        
        for i in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            c_refined = correlate(h_refined_interp, 1 - h_refined_interp, mode="full")
            grad_refined = c_refined * (2.0/n_refine)
            lr = lr_schedule(i)
            h_refined_interp = h_refined_interp - lr * grad_refined
            h_refined_interp = project_to_constraints(h_refined_interp, n_refine/2)
            
            current_c5_refined = compute_c5(h_refined_interp, 2.0/n_refine)
            if current_c5_refined < best_c5:
                best_c5 = current_c5_refined
                # Downsample for final result
                best_h = np.interp(np.linspace(0, 2, n_points), x_refined, h_refined_interp)
                best_h = project_to_constraints(best_h, target_sum)
    
    return (best_h.tolist(), best_c5, n_points)
