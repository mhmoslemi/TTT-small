import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Proportional adjustment
    if current_sum > target_sum:
        # Scale down
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Distribute deficit proportionally
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default settings
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 3000
    
    # Initialize from initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Something structured to start
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Parameters for gradient descent style update
    alpha = 0.2
    epsilon = 0.1
    decay_factor = 1000.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Alternating minimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        c = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(c)
        
        # Smoothing-based descent
        # Approximate gradient using local averaging of 1-h
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # finite difference estimate of 1-h's derivative
            left_val = 1 - h[(i-1) % len(h)]
            right_val = 1 - h[(i+1) % len(h)]
            # gradient from central difference
            grad[i] = (right_val - left_val) / 2.0
            
        # Update h
        h_candidate = h - alpha * grad
        
        # Add random noise for escape
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project onto feasible set
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept improvement with decrease in C5
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Revert to previous if no improvement
            pass
        
        # Adapt learning rate and noise
        alpha = alpha / (1.0 + iter_num / decay_factor)
        epsilon = epsilon / (1.0 + iter_num / decay_factor)
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Return final best
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
