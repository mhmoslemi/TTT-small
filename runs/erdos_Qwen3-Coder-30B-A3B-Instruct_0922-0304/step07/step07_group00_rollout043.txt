import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to satisfy the constraint
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-12))
    
    # Optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Forward pass: compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Compute gradients
        # This is a simplification that approximates gradients for the maximum overlap
        # For exact gradient computation, we would need to compute through the convolution
        # For now, this is a heuristic approximation
        
        # Get indices where h is not near boundaries
        valid_indices = (h > 1e-6) & (h < 1 - 1e-6)
        
        # Approximate gradient using finite differences
        # We compute directional derivative via small perturbations
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            if not valid_indices[i]:
                continue
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 1e-8, 1 - 1e-8)
            h_plus = h_plus / np.sum(h_plus) * target_sum
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 1e-8, 1 - 1e-8)
            h_minus = h_minus / np.sum(h_minus) * target_sum
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update parameters using Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_count+1))
        v_hat = v / (1 - beta2**(iter_count+1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h space
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-12))
        
        iter_count += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
