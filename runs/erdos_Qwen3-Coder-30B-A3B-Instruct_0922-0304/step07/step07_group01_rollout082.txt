import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with step-like function
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint
    h = normalize_h(h, target_sum)
    
    # Stage 1: Convex relaxation
    # Create the problem instance with fewer samples for efficiency
    n_reduction = max(1, n_points // 10)  # Reduce resolution for relaxation
    reduced_n = max(10, n_points // n_reduction)
    
    # Interpolate to lower resolution for faster convex solve
    if reduced_n < n_points:
        x_full = np.linspace(0, 2, n_points)
        x_reduced = np.linspace(0, 2, reduced_n)
        h_reduced = np.interp(x_reduced, x_full, h)
    else:
        h_reduced = h.copy()
    
    # Convex relaxation with CVXPY
    h_cvx = cp.Variable(reduced_n)
    C = cp.Variable()
    
    # Dependency on shifted h
    constr = []
    for k in range(2 * reduced_n - 1):
        # For each lag, sum up all terms h[i] * (1 - h_shifted[i])
        shift = k - (reduced_n - 1)  # Calculate shift
        if shift >= 0:
            shifted = np.concatenate([np.zeros(shift), h_cvx[:reduced_n-shift]])
        else:
            shifted = np.concatenate([h_cvx[-shift:], np.zeros(-shift)])
        # Use element-wise multiplication for correlation-like computation
        # Since this is a bit tricky to model fully in cvxpy, we compute correlation
        # as equivalent constraints by acknowledging that for large n,
        # this is computationally anyway expensive
        # We directly model without excessive convolution
    
    # Simpler approach: We model as minimizing C subject to overlap constraints
    # Create a simpler linearized max-overlap constraint
    
    # Model constraints directly with fewer samples
    h_reduced = cp.Variable(reduced_n, nonneg=True)
    C = cp.Variable()
    
    # Constraint: sum h = n_points / 2 expressed in reduced space
    constr.append(cp.sum(h_reduced) == target_sum)
    
    # Compute overlaps for a sample of key shifts
    effective_shifts = min(100, 2 * reduced_n - 1)
    i_sampled = np.linspace(0, 2 * reduced_n - 2, min(100, effective_shifts), dtype=int)
    
    for i in i_sampled:
        shift = i - (reduced_n - 1)
        if shift >= 0:
            shifted = h_reduced[shift:] + np.zeros(reduced_n)
            shifted[:reduced_n-shift] = h_reduced[:reduced_n-shift]
        else:
            shifted = h_reduced[:shift] + np.zeros(reduced_n)
            shifted[-shift:] = h_reduced[-shift:]
        
        # Linear approximation to maximize: h[i] * (1 - shifted[i])
        overlap = cp.sum(cp.multiply(h_reduced, 1 - shifted))
        constr.append(overlap <= C)
    
    # Solve convex optimization
    prob = cp.Problem(cp.Minimize(C), constr)
    try:
        prob.solve(solver=cp.ECOS, max_iters=500)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_reduced.value
            if h_sol is not None:
                # Interpolate back to full resolution
                h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, reduced_n), h_sol)
                h = np.clip(h_new, 0, 1)
                h = normalize_h(h, target_sum)
    except Exception:
        pass  # Fall back on original h if convex solve fails
    
    # Stage 2: Gradient Descent with projection
    # Define learning rate with decay
    lr = 0.01
    decay_rate = 0.95
    max_iter = 2000
    
    i_iter = 0
    
    while time.time() - start_time < budget_s - 1 and i_iter < max_iter:
        # Compute current value of C
        current_c5 = compute_c5(h, dx)
        
        # Compute gradients for a few selected shifts
        # For gradient direction, evaluate with a small perturbation
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            c_plus = compute_c5(h_plus, dx)
            grad[i] = (c_plus - current_c5) / eps
        
        # Update h with gradient descent and projection
        h = h - lr * grad
        h = project_to_constraints(h, target_sum)
        
        # Adaptive learning rate decay
        if i_iter > 0 and i_iter % 20 == 0:
            lr *= decay_rate
        
        # Early stop if improvement is below threshold
        if current_c5 <= 0.3807:
            break
            
        i_iter += 1
    
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
