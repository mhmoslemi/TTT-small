import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to correct sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Forward pass: compute current h and C5
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Recompute w to maintain feasibility
        w = np.log(h / (1 - h + 1e-8))
        
        # Compute gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Gradient computation for the maximum overlap
        g = np.zeros(n_points)
        for i in range(n_points):
            # Compute contribution to the gradient for max overlap
            if i + max_k < n_points:
                # Derivative of h[i] is h[i] * (1 - h[i]) * (1 - 2 * h[i])
                g[i] += (1 - 2 * h[i]) * (1 - h[i + max_k])
        
        # Scale by dx for the final gradient
        g *= dx
        
        # Update Adam parameters
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        # Update parameters
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Thresholding step
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-8))
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution if found
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final refinement with exact computation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
