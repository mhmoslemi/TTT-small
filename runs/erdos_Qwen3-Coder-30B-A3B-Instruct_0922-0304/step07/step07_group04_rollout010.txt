import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a base function with some structure
        h = np.zeros(n_points)
        # Circular arrangement with some variation
        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-6
    max_iter = int(1e5)
    early_stop_threshold = 1e-8
    early_stop_count = 0
    
    # Annealing loop
    iter_count = 0
    while T > min_T and iter_count < max_iter and (time.time() - start_time < budget_s - 5):
        # Perturb h
        h_new = h.copy()
        
        # Add random noise scaled by temperature
        noise = np.random.normal(0, 0.01 * T, n_points)
        h_new += noise
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        new_sum = np.sum(h_new)
        if new_sum > 0:
            h_new = h_new * (target_sum / new_sum)
        
        # Evaluate new solution
        c5_new = evaluate_overlap(h_new, dx)
        
        # Metropolis acceptance criterion
        delta_c5 = c5_new - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                early_stop_count = 0
            else:
                early_stop_count += 1
        else:
            # Keep old solution
            pass
        
        # Cooling schedule
        T *= cooling_rate
        
        iter_count += 1
        
        # Early stopping condition
        if early_stop_count > 2000:
            break
    
    # Final local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Use best solution as starting point
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
