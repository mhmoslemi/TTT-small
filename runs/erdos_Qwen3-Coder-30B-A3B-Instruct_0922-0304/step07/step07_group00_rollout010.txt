import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cvxsum, norm

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_basis_functions(n_points, n_bases=16):
    """Create a set of basis functions with varying pulse shapes."""
    basis = []
    dx = 2.0 / n_points
    
    # Create step functions with different pulse widths and positions
    pulse_widths = np.linspace(0.1, 0.5, 8)
    positions = np.linspace(0.1, 1.9, 8)
    
    for i, width in enumerate(pulse_widths):
        for j, pos in enumerate(positions):
            phi = np.zeros(n_points)
            start_idx = int(pos / dx)
            end_idx = min(start_idx + int(width / dx), n_points)
            if end_idx > start_idx:
                phi[start_idx:end_idx] = 1.0
            basis.append(phi)
            if len(basis) >= n_bases:
                break
        if len(basis) >= n_bases:
            break
    
    # Normalize each basis function to have integral 1/n_points
    for i, phi in enumerate(basis):
        integral = np.sum(phi) * dx
        if integral > 0:
            basis[i] = phi / integral
    
    # Ensure we have exactly n_bases
    while len(basis) < n_bases:
        basis.append(np.zeros(n_points))
        
    return basis

def compute_overlap_integrals(basis, n_points, dx):
    """Precompute the overlap integrals A_ij^k."""
    n_bases = len(basis)
    A = np.zeros((n_bases, n_bases, 2 * n_points - 1))
    
    for i in range(n_bases):
        for j in range(n_bases):
            for k in range(2 * n_points - 1):
                # Align basis functions for convolution
                if k < n_points:
                    offset = k
                    slice1 = basis[i][:n_points-k]
                    slice2 = basis[j][k:]
                else:
                    offset = k - n_points + 1
                    slice1 = basis[i][k-n_points+1:]
                    slice2 = basis[j][:n_points-k+n_points-1]
                
                if len(slice1) > 0 and len(slice2) > 0:
                    A[i, j, k] = np.sum(slice1 * slice2) * dx
    
    return A

    n_points = 70

    A = compute_overlap_integrals(basis, n_points, dx)
    
    # Define optimization variables
    n_bases = len(basis)
    lambda_var = Variable(n_bases, nonneg=True)
    
    # Initial setup for lambda
    if initial_h is not None:
        # Project initial_h onto basis space
        initial_h_norm = normalize_h(np.array(initial_h), target_sum)
        basis_matrix = np.array(basis).T
        lambda_value = np.linalg.lstsq(basis_matrix, initial_h_norm, rcond=None)[0]
        lambda_value = np.maximum(lambda_value, 0)
        lambda_value = lambda_value / np.sum(lambda_value)
        lambda_var.value = lambda_value
    else:
        lambda_var.value = np.ones(n_bases) / n_bases
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    iteration = 0
    
    # Dual variable vector
    mu = np.zeros(2 * n_points - 1)
    
    # Iteration loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        if lambda_var.value is not None:
            h = basis_matrix @ lambda_var.value
            h = normalize_h(h, target_sum)
            c5_val = compute_c5(h, dx)
            
            # Track best solution
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Update dual variables based on gradient/gradient ascent
            # Compute current overlap for gradients
            # For dual update:
            overlap = correlate(h, 1 - h, mode="full")
            # Use gradient estimation to adjust mu
            # This is a simplified greedy approach
            max_idx = np.argmax(overlap)
            if max_idx < len(mu):

        # This means solving: minimize sum_k mu_k * sum_ij lambda_i lambda_j (A_ij^k)

        # Create quadratic cost optimization problem with existing mu
        cost = 0
        for k in range(2 * n_points - 1):
            cost += mu[k] * cvxsum(
                [lambda_var[i] * lambda_var[j] * A[i, j, k] for i in range(n_bases) for j in range(n_bases)]
            )

        prob = Problem(Minimize(cost + regularization), [cvxsum(lambda_var) == 1])
        
        try:

    # Ensure last computed solution is used
    if lambda_var.value is not None:
        final_h = basis_matrix @ lambda_var.value
        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
        best_c5 = min(best_c5, final_c5)
        best_h = final_h
    
    return (best_h.tolist(), best_c5, n_points)
