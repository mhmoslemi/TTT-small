import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
    
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Track iterations without improvement
    no_improvement_count = 0
    max_no_improvement = 100
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < 3000:
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Compute gradient 
        # The gradient with respect to h is almost 1-h shifted by optimal k
        # but we need to compute it right
        grad = np.zeros(n_points)
        if max_k < n_points:
            # Padding on the left
            grad[max_k:] = 1 - h[:n_points - max_k]
        else:
            # Padding on the right
            k_offset = max_k - n_points + 1
            grad[:n_points - k_offset] = 1 - h[k_offset:]
        
        # Apply momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Project to sum constraint
        h = project_to_constraint(h, target_sum)
        
        # Compute new C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adaptive learning rate
        if no_improvement_count > max_no_improvement:
            learning_rate *= 0.95
            no_improvement_count = 0
            if learning_rate < 1e-5:
                break
        
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
