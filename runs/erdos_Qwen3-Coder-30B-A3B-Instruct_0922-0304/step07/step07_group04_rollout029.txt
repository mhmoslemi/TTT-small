import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based optimization
    def compute_gradient(h, dx):
        """Compute gradient of overlap w.r.t. h."""
        n = len(h)
        grad = np.zeros(n)
        
        # For each point, compute how it contributes to the maximum overlap
        # This is a simplified approximation - in practice, one would precompute all
        # shift correlations and their gradients
        for i in range(n):
            # Compute derivative of overlap at i with respect to h[i]
            # This is a simplistic view, could be made more precise
            # For now, just approximate using symmetric finite differences
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += 1e-5
            h_minus[i] -= 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            overlap_plus = evaluate_overlap(h_plus, dx)
            overlap_minus = evaluate_overlap(h_minus, dx)
            grad[i] = (overlap_plus - overlap_minus) / (2e-5)
        
        return grad
    
    # Apply adaptive gradient descent
    max_iter = 1000
    learning_rate = 1e-3
    momentum = 0.9
    v = np.zeros_like(h)
    lambda_reg = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Compute regularized gradient
        reg_grad = grad + lambda_reg * (np.sum(h) - target_sum)
        
        # Update with momentum
        v = momentum * v - learning_rate * reg_grad
        h = h + v
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        # Check for improvement
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adjust regularization
        if current_sum < target_sum * 0.98 or current_sum > target_sum * 1.02:
            lambda_reg *= 1.1
        else:
            lambda_reg *= 0.99
            
        # Reduce learning rate occasionally
        if iteration % 100 == 0 and learning_rate > 1e-6:
            learning_rate *= 0.9
        
        # Early stopping criteria
        if np.linalg.norm(grad) < 1e-6:
            break
            
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
