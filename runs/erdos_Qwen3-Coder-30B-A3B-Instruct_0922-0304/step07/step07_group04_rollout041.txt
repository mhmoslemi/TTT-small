import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use gradient-based optimization
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Use L-BFGS-B which handles bounds well
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Try different random restarts
    best_solution = best_h.copy()
    best_value = best_c5
    
    # Multiple restarts to avoid local minima
    n_restarts = 5
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Start with some random perturbation
        h_start = np.random.uniform(0.1, 0.9, n_points)
        h_start = h_start * (target_sum / np.sum(h_start))
        
        # Optimize with L-BFGS-B
        try:
            result = minimize(objective, h_start, method='L-BFGS-B', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_value:
                    best_value = final_c5
                    best_solution = refined_h.copy()
        except:
            continue
    
    # Final optimization with SLSQP
    try:
        result = minimize(objective, best_solution, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_value:
                best_value = final_c5
                best_solution = refined_h.copy()
    except:
        pass
        
    # Ensure final solution is valid
    final_h = np.clip(best_solution, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
