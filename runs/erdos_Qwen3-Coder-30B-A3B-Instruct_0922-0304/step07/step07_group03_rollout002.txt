import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_iter = 5000
    
    # Initialize population
    population = []
    for i in range(population_size):
        if 'initial_h_values' in globals():
            h_candidate = np.array(initial_h_values)
        else:
            h_candidate = np.random.uniform(0, 1, n_points)
        h_candidate = project_to_constraints(h_candidate, target_sum)
        population.append(h_candidate)
    
    best_h = None
    best_c5 = np.inf
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate mutation rate with decay
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness_scores.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population through mutation and selection
        new_population = sorted_population[:elite_size]
        
        # Fill rest with mutated elites
        for i in range(elite_size, population_size):
            parent_idx = np.random.choice(elite_size)
            parent = sorted_population[parent_idx]
            
            # Apply Gaussian mutation
            noise = np.random.normal(0, mutation_rate, n_points)
            mutant = parent + noise
            mutant = project_to_constraints(mutant, target_sum)
            new_population.append(mutant)
            
        population = new_population
    
    # Final check
    if best_h is None:
        # Fallback to last population member if something went wrong
        final_h = population[-1]
        final_c5 = compute_c5(final_h, dx)
    else:
        final_c5 = best_c5
        final_h = best_h
    
    return (final_h.tolist(), final_c5, n_points)
