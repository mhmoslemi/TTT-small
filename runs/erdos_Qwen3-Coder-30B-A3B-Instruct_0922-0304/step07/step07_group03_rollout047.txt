import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit/profit
    deficit = target_sum - current_sum
    if deficit > 0:
        # Need to increase values
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
    else:
        # Need to decrease values
        scale = target_sum / current_sum
        h *= scale
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create initial structured pattern
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_feasible(h, target_sum)

    # AM parameters
    max_iter = 3000
    alpha_init = 0.2
    epsilon_init = 0.1
    tau = 1000
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlation
        c = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(c)
        max_val = c[max_idx]
        
        # Compute gradient approx
        gradient = np.zeros(n_points)
        # Simple finite difference approach for gradient
        for i in range(n_points):
            # Approximate derivative with respect to h[i]
            h_plus = h.copy()
            h_plus[i] += 1e-5
            c_plus = correlate(h_plus, 1 - h_plus, mode="full")
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            c_minus = correlate(h_minus, 1 - h_minus, mode="full")
            gradient[i] = (c_plus[max_idx] - c_minus[max_idx]) * 1e5
            
        # Update h based on gradient
        alpha = alpha_init / (1 + t / tau)
        h_candidate = h - alpha * gradient
        
        # Add random perturbation to escape local minima
        epsilon = epsilon_init / (1 + t / tau)
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, target_sum)
        
        # Compute new overlap value
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = new_c5
        
        # Continue with current h
        h = h_candidate.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
