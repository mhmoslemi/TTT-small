import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Initialize relaxed version
    h_relaxed = h + np.random.normal(0, 0.02, n_points)
    h_relaxed = np.clip(h_relaxed, 0, 1)
    h_relaxed = normalize_h(h_relaxed, target_sum)
    
    # Hyperparameters
    alpha = 0.1
    beta = 0.9
    lambda_reg = 1e-4
    
    # Precompute indices for shifts
    shift_indices = {}
    for k in range(-n_points + 1, n_points):
        shift_indices[k] = np.arange(n_points) - k
        shift_indices[k] = np.where(shift_indices[k] < 0, 
                                 shift_indices[k] + n_points, 
                                 shift_indices[k] % n_points)
    
    # Adam-like optimizer state
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    iteration = 0
    improvement_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Calculate gradient estimate via finite differences
        # We estimate ∂C5/∂h[i]
        grad = np.zeros(n_points)
        epsilon = 1e-5
        base_c5 = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_constraint(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / epsilon
        
        # Adam-like update
        m = beta * m + (1 - beta) * grad
        v = beta * v + (1 - beta) * grad ** 2
        m_hat = m / (1 - beta ** iteration)
        v_hat = v / (1 - beta ** iteration)
        
        update = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Update relaxed values
        h_relaxed = h_relaxed - update
        
        # Add regularization
        h_relaxed = h_relaxed - lambda_reg * (h_relaxed - h)
        
        # Project to feasible region
        h = np.clip(h_relaxed, 0, 1)
        h = project_to_constraint(h, target_sum)
        
        # Recompute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvement_count += 1
            last_improvement = iteration
            
        if iteration - last_improvement > 100:
            alpha *= 0.9
            if alpha < 1e-6:
                alpha = 1e-6
            last_improvement = iteration
            
        if current_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
