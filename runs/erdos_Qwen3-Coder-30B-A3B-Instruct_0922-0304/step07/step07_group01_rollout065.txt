import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def gradient_step(h, dx, target_sum, epsilon=1e-6):
    # Compute gradient using finite differences
    grad = np.zeros_like(h)
    h_plus = h.copy()
    h_minus = h.copy()
    
    for i in range(len(h)):
        h_plus[i] += epsilon
        h_minus[i] -= epsilon
        h_plus = project_to_constraints(h_plus, target_sum)
        h_minus = project_to_constraints(h_minus, target_sum)
        
        c5_plus = compute_c5(h_plus, dx)
        c5_minus = compute_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        h_plus[i] -= epsilon
        h_minus[i] += epsilon
    
    return grad
