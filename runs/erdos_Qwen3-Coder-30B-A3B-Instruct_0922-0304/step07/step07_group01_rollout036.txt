import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Simulated Annealing parameters
    T0 = 0.1 * compute_c5(h, dx)
    alpha = 0.99
    T = T0
    max_iter = 10000
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h, dx)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Generate candidate solution by perturbing current solution
        candidate_h = current_h + np.random.uniform(-T, T, n_points)
        candidate_h = project_to_constraints(candidate_h, target_sum)
        candidate_c5 = compute_c5(candidate_h, dx)
        
        # Accept or reject the candidate
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.random.random() < np.exp(-delta / T):
                current_h = candidate_h
                current_c5 = candidate_c5
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        # Cool down temperature
        T *= alpha
        
        # Early stopping for target
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    # Final evaluation and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
