import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 60
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
        # Add some noise to create variation
        for _ in range(population_size//5):
            h = h_init + np.random.normal(0, 0.02, n_points)
            h = project_to_feasible(h, target_sum)
            population.append(h)
        
        # Fill with random individuals
        for _ in range(population_size - population_size//5):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, target_sum)
            population.append(h)
    else:
        # Create a random population with good distribution
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, target_sum)
            population.append(h)

    # Initialize PSO parameters
    inertia_w = 0.8
    c1 = 1.5
    c2 = 1.5
    particles = population.copy()
    
    # Initialize velocities (same size as particles)
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    
    # Initialize personal bests and global best
    personal_bests = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    global_best = personal_bests[np.argmin(personal_best_fitness)].copy()
    global_best_fitness = min(personal_best_fitness)
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                inertia_w * velocities[i] + 
                c1 * r1 * (personal_bests[i] - particles[i]) + 
                c2 * r2 * (global_best - particles[i])
            )
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints and projection
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            current_fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = current_fitness
                personal_bests[i] = particles[i].copy()
                
                # Update global best
                if current_fitness < global_best_fitness:
                    global_best_fitness = current_fitness
                    global_best = particles[i].copy()
        
        # Decrease inertia weight
        inertia_w *= 0.98
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Refinement using local optimization
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            new_c5 = compute_c5(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                break
        else:
            break
            
    return (best_h.tolist(), best_c5, n_points)
