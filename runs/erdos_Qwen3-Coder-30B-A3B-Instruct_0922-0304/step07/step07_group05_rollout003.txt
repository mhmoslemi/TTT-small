import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # PSO parameters
    n_particles = 100
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 60
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Create diverse initial particles
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Fill with variations of the best solution
    for _ in range(n_particles // 3):
        pert = h + np.random.normal(0, 0.05, n_points)
        particles.append(project_to_feasible(pert, target_sum))
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        
    # Fill with random particles
    for _ in range(n_particles - n_particles // 3 - 1):
        rand_h = np.random.rand(n_points)
        rand_h = project_to_feasible(rand_h, target_sum)
        particles.append(rand_h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal best
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(personal_best_fitness)
    gbest = personal_best[gbest_idx].copy()
    gbest_fitness = personal_best_fitness[gbest_idx]
    
    # History for early termination
    history = []
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    history = []
                else:
                    history.append(time.time())
    
    # Core optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Local refinement
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, gbest.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            gbest = result.x
            gbest = np.clip(gbest, 0, 1)
            gbest = normalize_h(gbest, target_sum)
        else:
            break
    
    # Final check
    final_c5 = compute_c5(gbest, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = gbest.copy()
    
    return (best_h.tolist(), best_c5, n_points)
