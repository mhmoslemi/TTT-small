import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Smooth step function as starting point
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        # Add some smoothness
        h = np.convolve(h, np.ones(5)/5, mode='same')
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Remez-like optimization with frequency domain adjustments
    max_iter = 50
    alpha = 0.1
    decay = 0.95
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        
        # Use frequency domain approach for updates
        # Take FFT of h and 1-h
        H = np.fft.fft(h)
        G = np.fft.fft(1 - h)
        corr_fft = H * np.conj(G)
        
        # Inverse FFT to get the correlation in time domain
        corr_time = np.fft.ifft(corr_fft).real
        
        # Find dominant frequency component
        freq_domain = np.fft.fft(corr_time)
        k = np.argmax(np.abs(freq_domain[1:n_points//2])) + 1
        
        # Phase inversion with dominant frequency
        phase_modulation = alpha * np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
        h_new = h + phase_modulation
        
        # Projection to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new candidate
        c5_new = compute_c5(h_new, dx)
        
        # Accept if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        if best_c5 <= 0.3807:
            break
            
        h = h_new.copy()
        alpha *= decay
    
    # Final local optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(5):
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target_sum / np.sum(best_h))
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
