import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient-based optimization with adaptive regularization
    max_iter = 1000
    learning_rate = 1e-3
    momentum = 0.9
    reg_lambda = 1e-4
    tolerance = 1e-8
    
    # Precompute shift indices for overlap evaluation
    shift_indices = []
    for k in range(-n_points + 1, n_points):
        indices = (np.arange(n_points) - k) % n_points
        shift_indices.append(indices)
    
    # Initialize velocity for momentum
    v = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient approximation
        # df/dh_i = sum over k of (1 - h[(i-k)%n]) - sum over k of h[(i+k)%n] 
        # But we only want gradient of the maximum overlap term
        # So we compute gradients for the maximum overlap and update accordingly
        
        # Evaluate current overlaps for each shift
        overlaps = []
        for k, indices in enumerate(shift_indices):
            shift_h = h[indices]
            overlap = np.sum(h * (1 - shift_h))
            overlaps.append(overlap)
        
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Approximate gradient w.r.t. h (for the max overlap term)
        # d/dh_i [h_i * (1 - h_{i-k})] = 1 - h_{i-k} if k >= 0
        # d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} if k <= 0
        grad = np.zeros_like(h)
        
        if max_k < n_points:  # k <= 0 (right shift)
            k = max_k - n_points + 1  # Reverse mapping
            k = -k  # Convert back
            indices = (np.arange(n_points) + k) % n_points
            grad = 1 - h[indices]
        else:  # k > 0 (left shift) 
            k = max_k - n_points + 1  # Compute actual shift
            indices = (np.arange(n_points) - k) % n_points
            grad = 1 - h[indices]
        
        # Add regularization term
        reg_term = reg_lambda * (h - 0.5)  # Trust region type penalty
        grad += reg_term
        
        # Update with momentum
        v = momentum * v - learning_rate * grad
        h_new = h + v
        
        # Project h_new onto feasible region [0,1] and keep sum constraint
        h_new = np.clip(h_new, 0, 1)
        current_sum_new = np.sum(h_new)
        if current_sum_new > 0:
            h_new = h_new * (target_sum / current_sum_new)
        
        # Backtracking line search with adaptivity
        c5_new = evaluate_overlap(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Adaptive lambda based on constraint satisfaction
        if np.abs(np.sum(h_new) - target_sum) > 1e-2:
            reg_lambda = min(reg_lambda * 1.5, 1e-1)
        else:
            reg_lambda = max(reg_lambda * 0.9, 1e-6)
        
        h = h_new.copy()
        
        # Early stopping based on improvement
        if i > 100 and abs(best_c5 - evaluate_overlap(h, dx)) < tolerance * 10:
            break

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
