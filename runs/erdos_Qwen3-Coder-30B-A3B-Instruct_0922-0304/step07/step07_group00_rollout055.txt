import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h

    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Frank-Wolfe style coordinate-wise optimization
    max_iter = 100000
    iter_count = 0
    
    # Track recent improvements to detect stagnation
    recent_improvements = []
    stagnation_threshold = 1000
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute current overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Compute gradient components for each coordinate
        # This is approximated as: \frac{\partial}{\partial h_i} max_k overlap_k = 2 * h_i - 1
        # but we need more precise: for each i, accumulate contributions to all overlap_k where k matches pattern
        # Coordinates contributing to max overlap get priority
        
        # Also track the indexes with most impact
        # for each h[i], compute the potential change with repect to overlapping terms
        
        # Simple gradient estimation approach: for each i, compute how changing h[i] changes overlap terms
        # approximation is simplified for speed: entire gradient computed by numerical difference (not worth it)
        # Instead, we directly compute overlap across all positions to estimate sensitivity
        
        influence = np.zeros(n_points)
        h_plus = h.copy()
        h_minus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = min(1., h[i] + 0.01)
            h_minus[i] = max(0., h[i] - 0.01)
            
            # Compute overlaps with slight perturbations around h[i]
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            # Gradient estimate at i
            grad_estimator = (np.max(overlap_plus) - np.max(overlap_minus)) / 0.02
            
            influence[i] = abs(grad_estimator)
            
            # Reset
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Select the top 10% most influential coordinates
        top_k = max(1, n_points // 10)
        top_indices = np.argsort(influence)[-top_k:][::-1]
        
        # Update each top coordinate with best step
        for i in top_indices:
            # Estimate gradient sign for magnitude of influence is strong
            diff_grad = (overlap[i + n_points - 1] - overlap[i])  # This is simplified
            
            if abs(diff_grad) < 1e-10:
                continue
                
            # Compute step size - see Frank-Wolfe style update
            # We compute target change, then determine optimal step
            # But use direct modification with trust region
            
            # Current value
            h_old = h[i] 
            
            # Simple line search attempt - choose best valid h[i] to reduce max overlap
            step_sizes = np.linspace(-0.05, 0.05, 20) if i == top_indices[0] else np.linspace(-0.02, 0.02, 10)
            best_step = 0.
            best_c5_trial = current_max
            
            for alpha in step_sizes:
                h_test = h.copy()

                h_test = np.clip(h_test, 0, 1)
                h_test = normalize_h(h_test, target_sum)
                
                c5_test = compute_c5(h_test, dx)
                
                if c5_test < best_c5_trial:
                    best_c5_trial = c5_test
                    best_step = alpha
                    
            # Apply the best step if improving
            if best_step != 0:
                h[i] += best_step
                h = np.clip(h, 0, 1)  # Maintain feasibility
                h = normalize_h(h, target_sum)  # Maintain mass constraint
                
                # Check if this reduced our bound
                trial_c5 = compute_c5(h, dx)
                if trial_c5 < best_c5:

                    best_h = h.copy()
                
                # Early stopping
                if best_c5 <= 0.3807:
                    break
        
        iter_count += 1
        
        # Detect stagnation
        recent_improvements.append(best_c5)
        if len(recent_improvements) > 100:
            recent_improvements.pop(0)
            if len(set(round(v, 6) for v in recent_improvements)) == 1:
                # No variation for last 100 steps
                break
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
