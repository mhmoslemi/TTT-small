import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to maintain sum constraint exactly
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Initialize step-like function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Convex relaxation approach
    h_cvx = cp.Variable(n_points)
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == target_sum]
    
    # For now, we'll try a simpler approach with local search
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based refinement
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Try perturbation-based local search
    max_iter = 1000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try moving boundaries of step functions
        h_new = h.copy()
        # Make small adjustments to positions of step changes
        step_positions = np.where(np.diff(np.round(h_new)) != 0)[0]
        if len(step_positions) > 1:
            # Move one step boundary
            pos = step_positions[np.random.randint(len(step_positions))]
            if pos > 0 and pos < n_points - 1:
                delta = np.random.uniform(-0.05, 0.05)
                h_new[pos] += delta
                h_new[pos+1] -= delta
                h_new = project_to_constraints(h_new, target_sum)
                
                new_c5 = compute_c5(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    h = h_new.copy()
        else:
            # All uniform, introduce step
            if np.random.rand() > 0.5:
                mid = n_points // 2
                h_new[mid] = min(1.0, h_new[mid] + 0.1)
                h_new[mid+1] = max(0.0, h_new[mid+1] - 0.1)
                h_new = project_to_constraints(h_new, target_sum)
                new_c5 = compute_c5(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    h = h_new.copy()
        
        # Occasionally add noise to escape local optima
        if np.random.rand() < 0.1:
            h += np.random.normal(0, 0.01, n_points)
            h = project_to_constraints(h, target_sum)
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            
        # Early stopping for target
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
