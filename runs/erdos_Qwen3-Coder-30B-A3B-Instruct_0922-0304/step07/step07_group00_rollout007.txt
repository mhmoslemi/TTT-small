import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based optimization with coordinate-wise updates
    iteration = 0
    last_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < max_no_improvement:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Rank coordinates by gradient impact
        # Gradient: Δ[i] = overlap[i] * (1 - 2*h[i])
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Calculate gradient contribution for i
            grad = 0
            # For each possible k, we need to consider the effect on overlap
            # overlap[i] is the value of the convolution at point i
            # gradient direction is to minimize this by adjusting h[i]
            # Let's compute (1-2*h[i]) for simplicity - this is a poor approximation
            # But it gives us a direction to move in
            gradients[i] = overlap[i] * (1 - 2 * h[i])
        
        # Select top 10% coordinates by |gradient|
        top_indices = np.argsort(np.abs(gradients))[-int(0.1 * n_points):][::-1]
        
        # Update selected coordinates
        for i in top_indices:
            old_val = h[i]
            # heuristic: move in direction of gradient to reduce maximum overlap
            # but keep h[i] in [0, 1]
            step = 0.01 * gradients[i]
            
            # Update h[i] 
            new_h = h.copy()
            new_h[i] = np.clip(h[i] + step, 0, 1)
            
            # Renormalize to keep sum = target_sum
            new_h = normalize_h(new_h, target_sum)
            
            # Compute new C5
            new_c5 = compute_c5(new_h, dx)
            
            # Accept the move if it improves C5
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                last_improvement = iteration
            
        iteration += 1
        
        # Regularly check the actual C5 value
        if iteration % 50 == 0:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if best_c5 <= 0.3807:
                break
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
