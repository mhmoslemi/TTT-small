import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm
    population_size = 100
    max_generations = 500
    tournament_size = population_size // 10
    mutation_rate = 0.1
    max_no_improvement = 50
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = np.random.rand(n_points)
        individual = np.clip(individual, 0, 1)
        individual_sum = np.sum(individual)
        if individual_sum > 0:
            individual = individual * (target_sum / individual_sum)
        population.append(individual)
    
    # Genetic algorithm loop
    no_improvement_count = 0
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            fit = evaluate_overlap(ind, dx)
            fitness.append(fit)
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection - tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < population_size:
            # Selection for crossover
            parents = np.random.choice(population_size, 2, replace=False)
            p1, p2 = selected[parents[0]], selected[parents[1]]
            
            # Blend crossover (0.2 to 0.8 blend)
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.binomial(1, 0.3, n_points)
                noise = np.random.normal(0, 0.05, n_points)
                child += noise * mask
            
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        population = new_population
        mutation_rate = max(0.01, mutation_rate * 0.995)  # Adaptive decay
    
    # Local refinement with SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 100, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
