import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_bounds(h):
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Optimization with CVXPY
    h_var = Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1]
    
    # Objective: minimize max correlation
    max_corr = Variable()
    # We'll use an overestimate approach; the actual computation
    # for large n is both faster and accurate enough.
    
    # Construct a simplified low-rank approximation to limit optimization over the full space
    # Instead, we tweak the initial solution with small perturbations
    
    # Simple gradient-based local optimization
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Adaptive step size using gradient estimates
    step_size = 0.001
    iterations = 0
    max_iter = 1000
    
    # Get the full correlation at initialization
    prev_c5 = best_c5 + 1  # Ensure we iterate at least once
    
    while time.time() - start_time < budget_s - 2 and iterations < max_iter:
        old_h = best_h.copy()
        
        # Compute gradients through finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = old_h.copy()
            h_plus[i] += eps
            h_plus = project_to_bounds(h_plus)
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # Maintain sum
            
            h_minus = old_h.copy()
            h_minus[i] -= eps
            h_minus = project_to_bounds(h_minus)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient descent step
        new_h = old_h - step_size * grad
        new_h = project_to_bounds(new_h)
        new_h = new_h * (target_sum / np.sum(new_h))  # Maintain mass constraint
        
        new_c5 = compute_c5(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            
            # Adapt step size based on progress
            if prev_c5 - new_c5 < 1e-6:
                step_size *= 0.95
        else:
            step_size *= 0.99
            
        prev_c5 = new_c5
        iterations += 1
        
        if best_c5 < 0.3807:
            break
    
    # Final verification
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
