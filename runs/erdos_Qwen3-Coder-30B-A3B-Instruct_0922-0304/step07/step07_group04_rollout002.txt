import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with adaptive regularization
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (target_sum / current_sum)
        return h_vec
    
    # Precompute shift indices
    full_length = 2 * n_points - 1
    lags = np.arange(-n_points + 1, n_points)
    shift_indices = []
    for k in lags:
        indices = np.arange(n_points)
        shifted_indices = (indices - k) % n_points
        shift_indices.append(shifted_indices)
    
    # Gradient computation and update function
    def compute_gradient(h_vec, λ=1e-4):
        # Compute all overlaps for gradient calculation
        # For each lag k, compute derivative w.r.t. h[i]
        # ∂/∂h[i] = (1 - h[(i-k)%n]) for all valid k where i is in support of h
        grad = np.zeros_like(h_vec)
        # For overlap calculation, we compute sum(h[i] * (1 - h[(i-k)%n])) 
        # So ∂/∂h[i] = 1 - h[(i-k)%n] for valid pairs
        for k, indices in zip(lags, shift_indices):
            # Only compute for indices where overlap contributes significantly
            if k >= 0:
                grad[:n_points-k] += 1 - h_vec[indices[:n_points-k]]
            else:
                grad[-k:] += 1 - h_vec[indices[-k:]]
        
        # Add penalty term for constraint violation
        sum_violation = np.sum(h_vec) - target_sum
        grad += λ * sum_violation
        
        return grad
    
    # Adaptive line search with backtracking
    def line_search(h_vec, grad, alpha=1e-3, γ=0.5, max_iter=10):
        for _ in range(max_iter):
            h_new = h_vec - alpha * grad
            h_new = project(h_new)
            c5_new = evaluate_overlap(h_new, dx)
            if c5_new < best_c5 * 0.9999:  # Accept improvement
                return h_new, c5_new, alpha
            alpha *= γ
        return h_vec, evaluate_overlap(h_vec, dx), alpha
    
    # Main optimization loop
    h_current = h.copy()
    c5_current = evaluate_overlap(h_current, dx)
    
    # Nesterov momentum
    v = np.zeros_like(h_current)
    momentum = 0.9
    α = 1e-3
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        # Compute gradient
        grad = compute_gradient(h_current)
        
        # Nesterov accelerated update
        v = momentum * v - α * grad
        h_temp = h_current + momentum * v - α * grad
        
        # Project
        h_temp = project(h_temp)
        
        # Line search
        h_current, c5_current, α = line_search(h_temp, grad, α)
        
        # Update if better solution
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        iter_count += 1
    
    # Final validation
    final_h = project(best_h)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
