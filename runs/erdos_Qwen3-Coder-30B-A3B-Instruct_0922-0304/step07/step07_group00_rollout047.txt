import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute the gradient of overlap at shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    # The gradient of overlap_k at each position i is (1 - 2*h[i+k_max])
    for i in range(n - k_max):
        grad[i] += (1 - 2 * h[i + k_max])
    return grad

def solve_lp_direction(grad, n):
    """Solve the linear program to find the direction."""
    # Minimize grad^T * s subject to:
    # s[i] in [0, 1]
    # sum(s) = 0
    # We treat this as maximizing grad^T * (-s)
    # which means we select the 0 or 1 values that maximize the dot product
    # with negative gradient
    s = np.zeros(n)
    indices = np.argsort(-grad)  # descending order
    zero_indices = []
    one_indices = []
    sum_s = 0
    for i in indices:
        if sum_s == 0:
            # Choose 1 if it's beneficial
            if grad[i] < 0:
                s[i] = 1
                sum_s += 1
                one_indices.append(i)
            else:
                s[i] = 0
                zero_indices.append(i)
        elif sum_s > 0:
            # If we already have sum_s positive, we want to minimize further
            if grad[i] > 0 and sum_s > 0:
                s[i] = 0
                zero_indices.append(i)
                sum_s -= 1
            elif grad[i] < 0:
                # More complex: try to swap one of the ones to zero
                # This is just a heuristic for satisfying the constraint
                if len(one_indices) > 0:
                    s[one_indices[-1]] = 0  # Remove one from 1s
                    one_indices.pop()
                    sum_s -= 1
                    s[i] = 1  # Set this to 1
                    sum_s += 1
                    one_indices.append(i)
            else:
                s[i] = 0
        else:
            # Just set to 0
            s[i] = 0
    
    # Ensure sum = 0
    if sum_s != 0:
        # Modify s to keep sum = 0
        # Pick indices that are at 1
        if len(one_indices) >= 1:
            s[one_indices[0]] = 0
            sum_s -= 1
        if len(zero_indices) >= 1:
            s[zero_indices[0]] = 1
            sum_s += 1
    
    # If still not 0, make some arbitrary adjustment  
    while sum_s != 0:
        # Switch an element from 1 to 0
        if len(one_indices) > 0:
            idx = one_indices.pop()
            s[idx] = 0
            sum_s -= 1
        else:
            break
    return s

def backtracking_line_search(h, direction, grad, k_max, dx, max_alpha=1.0, c1=1e-4):
    """Perform backtracking line search to find a good step size."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    while alpha > 1e-6:
        new_h = h + alpha * (direction - h)
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, len(h) / 2)
        
        new_c5 = compute_c5(new_h, dx)
        
        # Armijo condition: f(x + alpha*d) <= f(x) + c1 * alpha * grad^T * d
        if new_c5 <= current_c5 + c1 * alpha * np.dot(grad, direction - h):
            return alpha, new_h
        alpha *= 0.5
    return alpha, h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Frank-Wolfe algorithm
    max_iter = 5000
    iteration = 0
    c5_history = []
    stagnation_count = 0
    max_stagnation = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient
        grad = compute_gradients(h, k_max)
        
        # Solve for direction
        direction = solve_lp_direction(grad, n_points)
        
        # Perform line search
        alpha, new_h = backtracking_line_search(h, direction, grad, k_max, dx)
        
        # Update h
        h = new_h
        
        # Compute C5 for current
        current_c5 = compute_c5(h, dx)
        c5_history.append(current_c5)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        if stagnation_count > max_stagnation:
            break
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
