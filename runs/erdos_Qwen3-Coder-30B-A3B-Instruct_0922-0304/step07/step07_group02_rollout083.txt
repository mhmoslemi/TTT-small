import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
        h = project_to_constraint(h, target_sum)
    
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    
    max_iter = 3000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        # The maximum overlap occurs at shift k_max
        # The gradient is (1 - h) shifted by k_max
        # We work with the shifted version for gradient update
        
        # Compute gradient: (1 - h) shifted to maximize overlap
        gradient = np.zeros(n_points)
        if k_max < len(h):
            gradient[k_max:] = 1 - h[:-k_max] if k_max > 0 else 1 - h
        else:
            # For the case when k_max >= len(h), we consider shifted overlap
            # This handles the case where the shift is in the 'full' mode
            # We just make the calculation accurate using correlate's semantics
            # Actually, note that for full mode, we want to get the index
            # of maximum overlap in the correct interval and then compute gradient
            # The overlap gives us indices that tell us where the peak occurs
            # Let's be more direct and compute it properly:
            # overlap_index = np.argmax(overlap) 
            # overlap_len = len(overlap)
            # The actual overlap in mode = full is of length 2*n - 1.
            # The true shift k in [0, 2) range starts at overlap[0] for k=0
            # and is shifted by overlap_index - (n-1) positions in the overlap array
            # To get the shift amount in real domain k = (overlap_index - (n-1)) * dx
            # But since index 0 corresponds to the max being in "first", 
            # we can just compute the gradient directly from computation:
            # the max is at index overlap_index, and that corresponds
            # to shifting(1-h) by that amount in overlap, so the gradient is:
            # shift_by = overlap_index - (n-1)  # corresponds to k 
            shift_by = k_max - (n_points - 1)
            shifted_ones_minus_h = np.roll(1 - h, shift_by)
            gradient = shifted_ones_minus_h
        # Apply the gradient descent update  
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        
        # Project to constraint
        h = project_to_constraint(h, target_sum)
        
        # Compute current value
        current_c5 = compute_c5(h, dx)
        
        # Check if this is an improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Learning rate adaptation
        if no_improvement_count > 100:
            learning_rate *= 0.95
            no_improvement_count = 0
            if learning_rate < 1e-5:
                learning_rate = 1e-5
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
