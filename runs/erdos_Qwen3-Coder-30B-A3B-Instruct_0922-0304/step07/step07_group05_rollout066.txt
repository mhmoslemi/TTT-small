import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize to meet sum constraint
        h = h * (target_sum / np.sum(h))
    else:
        # Create smooth initial guess: triangular pulse in the center
        h = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 4
        for i in range(width):
            h[mid - width//2 + i] = i / (width//2)
            h[mid + width//2 - 1 - i] = i / (width//2)
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (target_sum / np.sum(h))
        
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Phase inversion method parameters
    max_iterations = 20
    alpha = 0.1
    decay = 0.95
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(overlap)
        
        # Use the dominant frequency approach from frequency domain
        # Perform FFT to understand dominant components
        h_fft = np.fft.fft(h)
        # Get the dominant frequency (index of peak in magnitude)
        freqs = np.fft.fftfreq(len(h), dx)
        # Since we're dealing with real signals, take the first half only
        # Find dominant positive frequency by looking at magnitude
        mag = np.abs(h_fft[:len(h)//2])
        if len(mag) > 1:
            dominant_freq_idx = np.argmax(mag[1:]) + 1  # skip DC component
        else:
            dominant_freq_idx = 1
            
        # Generate phase inversion signal with dominant frequency
        phase_inversion = alpha * np.sin(2 * np.pi * freqs[dominant_freq_idx] * np.arange(len(h)) * dx)
        
        # Update h with phase inversion
        h_new = h + phase_inversion
        
        # Project onto feasible space
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Check improvement
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        alpha *= decay
        
        # Early stopping if improvement is minimal
        if iteration > 2 and abs(new_c5 - best_c5) < tolerance:
            break
            
        # Ensure feasible solution before next iteration
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    return (best_h.tolist(), best_c5, n_points)
