import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to correct sum
    h = h / np.sum(h) * target_sum
    
    # Transform to log-odds space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Initialize momentum and variance terms
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Transform back to probability space
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Update log-odds for next iteration
        w = np.log(h / (1 - h + 1e-8))
        
        # Evaluate current function value and gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Gradient computation
        g = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from overlap term
            # Only non-zero where the overlap is maximal
            if i + max_idx < n_points:
                g[i] += (1 - 2 * h[i]) * (1 - h[i + max_idx])
            if i + max_idx >= n_points:
                # This case handles circular convolution to avoid boundary issues
                # For overlap[0], we consider element wise contribution
                pass
    
        # Update momentum and variance
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iter_count+1))
        v_hat = v / (1 - beta2**(iter_count+1))
        
        # Update weights
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Recompute function value
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            if best_c5 <= 0.3807:
                break
                
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
