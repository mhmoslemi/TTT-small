import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Adaptive Gaussian Mixture Algorithm (AGMA)
    # Parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize mixture components
    components = []
    if 'initial_h_values' in globals():
        # Initialize first component around the provided solution
        mean = np.array(initial_h_values)
        mean = np.clip(mean, 0, 1)
        mean = normalize_h(mean, target_sum)
        components.append({
            'mean': mean.copy(),
            'cov': np.eye(n_points) * initial_sigma,
            'weight': 1.0/n_components
        })
        # Add random components
        for _ in range(n_components - 1):
            component = {
                'mean': np.random.rand(n_points),
                'cov': np.eye(n_points) * initial_sigma,
                'weight': 1.0/n_components
            }
            component['mean'] = normalize_h(component['mean'], target_sum)
            components.append(component)
    else:
        # All random components
        for _ in range(n_components):
            component = {
                'mean': np.random.rand(n_points),
                'cov': np.eye(n_points) * initial_sigma,
                'weight': 1.0/n_components
            }
            component['mean'] = normalize_h(component['mean'], target_sum)
            components.append(component)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Main AGMA loop
    max_iterations = 6000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Sample from the mixture
        candidates = []
        for _ in range(population_size):
            # Sample component
            weights = [c['weight'] for c in components]
            selected_idx = np.random.choice(len(components), p=weights)
            selected = components[selected_idx]
            
            # Sample from selected component
            try:
                sample = np.random.multivariate_normal(selected['mean'], selected['cov'])
            except np.linalg.LinAlgError:
                # Fallback to simple Gaussian sampling if matrix is singular
                sample = selected['mean'] + np.random.randn(n_points) * np.sqrt(np.diag(selected['cov']))
            
            # Apply constraints
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            
            candidates.append(sample)
        
        # Evaluate candidates
        candidate_scores = []
        for candidate in candidates:
            score = compute_c5(candidate, dx)
            candidate_scores.append((score, candidate))
        
        # Sort by score and keep top
        candidate_scores.sort(key=lambda x: x[0])
        top_candidates = [c[1] for c in candidate_scores[:keep_top]]
        top_scores = [c[0] for c in candidate_scores[:keep_top]]
        
        # Update best
        if top_scores[0] < best_c5:
            best_c5 = top_scores[0]
            best_h = top_candidates[0].copy()
        
        # Update mixture components
        for i, component in enumerate(components):
            if len(top_candidates) == 0:
                break
                
            # Compute new mean as average of top candidates
            new_mean = np.mean(top_candidates, axis=0)
            
            # Compute updated covariance
            if len(top_candidates) > 1:
                # Center the top candidates
                centered = np.array(top_candidates) - new_mean
                new_cov = (1.0 / (len(top_candidates) - 1)) * np.dot(centered.T, centered)
            else:
                new_cov = component['cov']  # No change if only one candidate
            
            # Regularize covariance matrix to keep it positive definite
            np.fill_diagonal(new_cov, np.diag(new_cov) + 1e-8)
            
            # Update with weighted average of old and new
            # Empirical adjustment factor
            alpha = 0.7
            new_mean = alpha * new_mean + (1 - alpha) * component['mean']
            new_cov = alpha * new_cov + (1 - alpha) * component['cov']
            
            # Decay covariances
            new_cov = new_cov * covariance_decay
            
            # Update component
            components[i] = {
                'mean': new_mean,
                'cov': new_cov,
                'weight': 1.0/n_components  # Equal weights (could make adaptive)
            }
        
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
