import numpy as np
from scipy.signal import correlate
import time
from scipy.stats import multivariate_normal
from sklearn.decomposition import PCA

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: scale to target sum
    h = h * (target_sum / current_sum)
    # Clamp to bounds
    h = np.clip(h, 0, 1)
    # Rescale to ensure sum constraint
    h = h * (target_sum / np.sum(h))
    return h

def apply_gaussian_mutation(x, cov_matrix, mutation_scale=0.1):
    """Apply Gaussian mutation to a vector."""
    try:
        mutated = np.random.multivariate_normal(x, cov_matrix * mutation_scale)
        return mutated
    except:
        #fallback in case of singular matrix
        return x + np.random.normal(0, mutation_scale, len(x))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add noise
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_mutation_scale = 0.1
    
    # Initialize Gaussian mixture
    means = [h.copy()]
    covariances = [np.eye(n_points) * initial_mutation_scale]
    weights = [1.0]
    
    # Sample based on mixture model
    def sample_from_gmm():
        #Sample component index based on weights
        comp_idx = np.random.choice(len(weights), p=weights)
        # Sample from chosen component
        sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
        # Clip to bounds
        sample = np.clip(sample, 0, 1)
        # Project to sum constraint
        sample = project_to_constraint(sample, target_sum)
        return sample
    
    # Initialize with a population
    population = [sample_from_gmm() for _ in range(population_size)]
    populations = [population]
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    iteration = 0
    max_iterations = 6000  # effectively limits time
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Evaluate all individuals
        fitnesses = [compute_c5(individual, dx) for individual in population]
        
        # Find best individuals to update model
        sorted_indices = np.argsort(fitnesses)
        top_individuals = [population[i] for i in sorted_indices[:keep_top]]
        top_fitnesses = [fitnesses[i] for i in sorted_indices[:keep_top]]
        
        # Update best solution so far
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Update the GMM
        # Recompute weights
        weights = np.array(top_fitnesses)
        weights = np.exp(-5 * (weights - np.min(weights)) / (np.max(weights) - np.min(weights) + 1e-8))
        weights = weights / np.sum(weights)
        
        # Update means and covariances
        # Use top individuals for updating
        if len(top_individuals) > 1:
            means = []
            covariances = []
            # One mean per top individual as starting point for update  
            for individual in top_individuals:
                means.append(individual.copy())
                # Use PCA to estimate initial covariance
                try:
                    # Get the PCA of the top individuals
                    X = np.array(top_individuals)
                    # For small number of components, to be careful
                    # Simple estimation
                    if len(top_individuals) <= 5:
                        # Fallback to full covariance since we have few samples
                        cov = np.cov(X.T)
                    else:
                        # Use PCA to get full rank
                        pca = PCA(n_components=min(10, len(top_individuals)-1))
                        pca.fit(X)
                        cov = pca.components_.T @ np.diag(pca.explained_variance_) @ pca.components_
                    covariances.append(cov + np.eye(n_points) * 1e-6)  # add small noise to be PD
                except:
                    covariances.append(np.eye(n_points) * initial_mutation_scale)
            
            # Normalize weights again after updating
            weights = np.ones(len(means))
            weights = weights / np.sum(weights)
        
        # Apply decay to covariances
        if len(covariances) > 0:
            covariances = [cov * covariance_decay for cov in covariances]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            if iteration < 50:
                # Early phase: random sampling
                sample = np.clip(np.random.rand(n_points), 0, 1)
                sample = project_to_constraint(sample, target_sum)
            else:
                # Sample from gmm
                sample = sample_from_gmm()
            new_population.append(sample)
        
        population = new_population
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Return final best solution
    return (best_h.tolist(), best_c5, n_points)
