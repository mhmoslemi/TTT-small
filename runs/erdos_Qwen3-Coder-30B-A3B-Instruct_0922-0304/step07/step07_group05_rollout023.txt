```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to meet the sum constraint and bound constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Start with triangular pulse
        h = np.zeros(n_points)
        midpoint = n_points // 2
        for i in range(midpoint):
            h[i] = i / midpoint
            h[n_points - 1 - i] = i / midpoint
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iterative refinement
    max_iterations = 60
    patience = 10
    no_improvement = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience or best_c5 <= 0.3807:
            break
            
        # Quadratic approximation using finite differences
        eps = 1e-4
        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            gradient[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
            
        # Formulate and solve QP
        # Minimize: 0.5 * d^T * H * d + gradient^T * d
        # Subject to: sum(d) = 0, 0 <= h + d <= 1
        
        # Simple QP with box constraints
        d = cp.Variable(n_points)
        
        # Objective: quadratic approximation
        objective = cp.Minimize(0.5 * cp.quad_form(d, np.eye(n_points)) + gradient.T @ d)
        
        # Constraints
        constraints = [
            cp.sum(d) == 0,
            h + d >= 0,
            h + d <= 1
        ]
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                step = d.value
            else:
                # Fallback to simple gradient descent
                step = -0.01 * gradient
        except:
            # Fallback to simple gradient descent
            step = -0.01 * gradient
            
        # Line search
        step_size = 1.0
        max_linesearch = 20
        for _ in range(max_linesearch):
            h_new = h + step_size * step
            h_new = project_to_feasible(h_new, target_sum)
            if compute_c5(h_new, dx) <= current_c5:
                h = h_new
                break
            step_size *= 0.5
            
        # Occasionally close to full iteration, merge with known good
        if iteration % 5 == 0 and iteration > 0:
            h = (h + best_h) / 2
            h = project_to_feasible(h, target_sum)
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
