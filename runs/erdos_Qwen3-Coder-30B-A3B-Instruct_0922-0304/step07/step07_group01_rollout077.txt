import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def step_function_from_continuous(h_cont, n_points, target_sum):
    # Create step function from continuous solution
    h_step = np.zeros(n_points)
    # Get cumulative sum to determine where to place unit steps
    cumsum = np.cumsum(h_cont)
    # Normalize to target sum
    h_step = project_to_sum_constraint(h_step, target_sum)
    
    # Simple binary thresholding approach
    h_step = np.clip(h_cont, 0, 1)
    h_step = project_to_sum_constraint(h_step, target_sum)
    return h_step

def adaptive_local_refinement(h, target_sum, dx, max_iter=50):
    """Refine step function by adjusting positions to minimize C5"""
    h = np.array(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        # Try small perturbations at different positions
        h_trial = h.copy()
        idx = np.random.randint(0, len(h))
        
        # Small random step
        delta = np.random.normal(0, 0.02)
        h_trial[idx] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_to_sum_constraint(h_trial, target_sum)
        
        trial_c5 = compute_c5(h_trial, dx)
        if trial_c5 < best_c5:
            best_h = h_trial
            best_c5 = trial_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_sum_constraint(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convex relaxation using CVXPY
    h_cvx = cp.Variable(n_points, name="h")
    C = cp.Variable(name="C")
    constraints = [
        cp.sum(h_cvx) == target_sum,
        h_cvx >= 0,
        h_cvx <= 1
    ]
    
    # Add constraint for C
    # Approximate with a few key convolution constraints
    # In practice, we'll use a simplified approximation of constraint
    # We'll compute directly convolution for all valid shifts compute C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve convex relaxation 
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        h_cont = h_cvx.value
        if h_cont is not None:
            h_init = h_cont.copy()
        else:
            h_init = h.copy()
    except:
        h_init = h.copy()
    
    # Convert to step function
    h = step_function_from_continuous(h_init, n_points, target_sum)
    
    # Local refinement
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Some refinement iterations
    refine_iters = 100
    for _ in range(refine_iters):
        if time.time() - start_time > budget_s - 1:
            break
        h_trial, c5_trial = adaptive_local_refinement(h, target_sum, dx)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
        h = h_trial
    
    return (best_h.tolist(), best_c5, n_points)
