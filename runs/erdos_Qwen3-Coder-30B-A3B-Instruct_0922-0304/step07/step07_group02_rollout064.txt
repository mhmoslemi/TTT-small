import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Differential Evolution Parameters
    population_size = 50
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, target_sum)
        else:
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, target_sum)
        population.append(individual)
        fitness.append(compute_c5(individual, dx))
    
    # Find initial best
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Evolve population
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct individuals for mutation
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.copy(population[i])
            crossover_mask = np.random.rand(n_points) < crossover_prob
            trial = np.where(crossover_mask, mutant, trial)
            
            # Ensure constraints
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate trial
            trial_fitness = compute_c5(trial, dx)
            
            # Select better individual
            if trial_fitness < fitness[i]:

                new_fitness.append(trial_fitness)
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_individual = trial.copy()
