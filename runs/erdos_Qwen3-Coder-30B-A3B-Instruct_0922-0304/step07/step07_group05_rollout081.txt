import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with the initial solution if available
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Create initial population with some diversity
    for _ in range(population_size):
        if np.random.rand() < 0.4:
            # Start from the initial solution with some variations
            individual = h + np.random.normal(0, 0.01, n_points)
        else:
            # Random feasible solution
            individual = np.random.rand(n_points)
            individual = normalize_h(individual, target_sum)
        individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    max_iter = 60
    
    # Initialize particles
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(population_size)]
    personal_best = [pop.copy() for pop in population]
    personal_best_fitness = [compute_c5(p, dx) for p in population]
    
    # Find global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - population[i]) +
                            c2 * r2 * (global_best - population[i]))
            
            # Update position
            population[i] = population[i] + velocities[i]
            
            # Apply constraints
            population[i] = np.clip(population[i], 0, 1)
            population[i] = normalize_h(population[i], target_sum)
            
            # Evaluate new position
            fitness = compute_c5(population[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = population[i].copy()
                personal_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = population[i].copy()
                global_best_fitness = fitness
                
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Final refinement with gradient descent for the best individual
    best_h = global_best.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(20):  # Run gradient descent for refinement
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
