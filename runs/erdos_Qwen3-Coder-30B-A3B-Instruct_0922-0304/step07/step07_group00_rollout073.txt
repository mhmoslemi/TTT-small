import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute gradient of overlap at shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    # The gradient of overlap_k with respect to h_i is (1 - 2*h[i+k_max]) 
    # but we only consider valid indices
    for i in range(n - k_max):
        grad[i] += (1 - 2 * h[i + k_max])
    return grad

def solve_lp_gradient(grad, n_points):
    """Solve the linear program to find descent direction."""
    # We want to minimize grad^T s subject to sum(s) = 0, 0 <= s[i] <= 1
    # Simple approach: set s[i] = 1 where grad[i] is most negative, 0 otherwise
    # But we need to be careful about sum(s) = 0 constraint
    # Better: set s to be the direction that reduces gradient most
    s = np.zeros(n_points)
    # Set s to be -grad with some constraints
    # but we have sum(s) = 0 so we will adjust
    s = -grad
    # Now adjust to make sum zero while keeping in [0,1]
    s_mean = np.mean(s)
    s = s - s_mean
    s = np.clip(s, 0, 1)
    return s

def backtracking_line_search(h, grad, s, dx, max_alpha=1.0):
    """Perform backtracking line search for sufficient decrease."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    # Reduce alpha until we get a sufficient decrease
    while alpha > 1e-10:
        new_h = h + alpha * (s - h)
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, len(h) / 2)
        new_c5 = compute_c5(new_h, dx)
        if new_c5 < current_c5:
            return alpha
        alpha *= 0.5
    return alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 500
    k_max = 0  # Start with some shift
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute overlaps
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        
        # Compute gradient
        grad = compute_gradients(h, k_max)
        
        # Solve linear program for descent direction
        s = solve_lp_gradient(grad, n_points)
        
        # Ensure s is feasible and sum to zero (approximate)
        s_sum = np.sum(s)
        if abs(s_sum) > 1e-10:
            # Adjust s to be zero mean
            s = s - np.mean(s)
            s = np.clip(s, 0, 1)
        
        # Backtracking line search
        alpha = backtracking_line_search(h, grad, s, dx)
        
        # Update h
        h = h + alpha * (s - h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Check for improvement
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        if no_improve_count > max_no_improve:
            # Try restart with some randomization
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            no_improve_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
