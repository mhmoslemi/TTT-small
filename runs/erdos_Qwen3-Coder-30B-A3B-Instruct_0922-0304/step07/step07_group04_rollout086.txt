import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing with gradient guidance
    T = 1.0
    cooling_rate = 0.999
    lr = 0.01
    lambda1 = 100.0
    lambda2 = 100.0
    
    def compute_energy(h_vec, c5_val):
        # Constraint penalties
        bound_penalty = np.sum(np.clip(h_vec, 0, 1) - h_vec)**2
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return c5_val + lambda1 * bound_penalty + lambda2 * sum_penalty
    
    # Annealing loop
    max_iterations = int(1e5)
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h
        h_new = h.copy()
        # Gradient-guided perturbation
        grad = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            # Compute gradient estimate
            indices = np.arange(n_points)
            shifted_indices = (indices - k) % n_points
            for i in range(n_points):
                grad[i] += (1 - h[shifted_indices[i]]) * (h[i] > 0) - h[i] * (h[shifted_indices[i]] < 1)
        
        noise = np.random.normal(0, lr * 0.1, n_points)
        h_new += lr * grad + noise
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new solution
        c5_new = evaluate_overlap(h_new, dx)
        E_new = compute_energy(h_new, c5_new)
        E_old = compute_energy(h, best_c5)
        
        # Accept with Metropolis criterion
        if c5_new < best_c5:
            h = h_new
            best_h = h_new
            best_c5 = c5_new
        else:
            delta_E = E_new - E_old
            if delta_E < 0 or np.exp(-delta_E / T) > np.random.random():
                h = h_new
                
        # Update temperature and learning rate
        T *= cooling_rate
        lr *= 0.99999
        
    # Final refinement using gradient descent
    if time.time() - start_time < budget_s - 5:
        def obj_func(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        h_refined = best_h.copy()
        lr_refine = 1e-4
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            c5_current = obj_func(h_refined)
            # Simple gradient estimation
            grad_est = np.zeros(n_points)
            for k in range(-n_points + 1, n_points):
                indices = np.arange(n_points)
                shifted_indices = (indices - k) % n_points
                for i in range(n_points):
                    grad_est[i] += (1 - h_refined[shifted_indices[i]]) - h_refined[i]

            h_refined -= lr_refine * grad_est
            h_refined = np.clip(h_refined, 0, 1)
            h_refined_sum = np.sum(h_refined)
            if h_refined_sum > 0:
                h_refined = h_refined * (target_sum / h_refined_sum)
            
            c5_new = obj_func(h_refined)
            if c5_new < best_c5:
                best_h = h_refined
                best_c5 = c5_new
    
    # Final normalization
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
