import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the known construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to correct sum
    h = h / np.sum(h) * target_sum
    
    # Transform to log-odds space for easier optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Optimization parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    max_iter = 10000
    
    # Adam optimizer parameters
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current objective
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient (approximate)
        # This is a simplified derivative estimate using finite differences
        # We're interested in optimizing max over all lags
        g = np.zeros(n_points)
        
        # Compute overlap with all shifts and get the max one
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_lag = max_idx - (len(h) - 1)  # Convert to actual lag
        
        # Gradient approximation based on the max overlap term only
        # This is a simple approach; more sophisticated methods could 
        # consider all terms but that's complex for continuous optimization
        
        # Compute a rough gradient approximation for the max lag term
        # using a finite difference style around elements contributing to max
        # For simplicity, a naive gradient using finite differences via soft threshold
        for k in range(len(h)):
            if max_lag + k < len(h):
                # Gradient direction
                g[k] += h[k] * (1 - 2 * h[k]) * (1 - h[max_lag + k])
        
        # Apply Adam optimization step
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to probability space
        h = sigmoid(w)
        
        # Renormalize to target sum
        h = h / np.sum(h) * target_sum
        
        # Clip to prevent numerical issues
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Update w for next iteration
        w = np.log(h / (1 - h + 1e-8))
        
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            lr *= 0.999
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
