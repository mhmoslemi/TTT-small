import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
    # Differential Evolution parameters
    population_size = 60
    mutation_factor = 0.8
    crossover_probability = 0.7
    max_generations = 3000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual = np.clip(individual, 0, 1)
            individual = individual * (target_sum / np.sum(individual))
        else:
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = individual * (target_sum / np.sum(individual))
        population.append(individual)
    
    # Evaluate initial population
    fitness = []
    for individual in population:
        fitness.append(compute_c5(individual, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Create new population
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct individuals for mutation
            indices = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = x1 + mutation_factor * (x2 - x3)
            
            # Crossover
            trial = np.copy(population[i])
            if np.random.rand() < crossover_probability:
                j = np.random.randint(n_points)
                trial[j] = mutant[j]
            
            # Ensure bounds and constraint
            trial = np.clip(trial, 0, 1)
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate trial
            trial_fitness = compute_c5(trial, dx)
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
            # Update global best
            if trial_fitness < best_fitness:
                best_fitness = trial_fitness
                best_individual = trial.copy()
                
        population = new_population
        fitness = new_fitness
        
        # Periodic checks
        if generation % 100 == 0:
            if best_fitness <= 0.3807:
                break
        
        generation += 1
    
    return (best_individual.tolist(), best_fitness, n_points)
