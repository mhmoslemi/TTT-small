import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Computes the C5 value for a given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Projects h to satisfy bounds and sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def initialize_h(n_points):
    """Initialize h with the provided initial_h_values if available."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    h = initialize_h(n_points)
    h = project_to_constraints(h, target_sum)
    
    # Convex optimization in frequency domain
    # Relax the problem with a smooth approximation
    
    # Try to maximize -C5 to minimize it
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Convex optimization using CVXPY
    # We model the constraints in the frequency domain
    
    # Create a large number of frequency points for accuracy
    n_freq = 2 * n_points  # Oversampling for better resolution
    freq = np.linspace(0, 2*np.pi, n_freq, endpoint=False)
    
    # Formulate the optimization problem in the optimization space
    # More practical approach: optimize directly
    
    # Use gradient based optimization
    max_iters = 5000
    learning_rate = 0.01
    tolerance = 1e-6
    patience = 100
    
    # Initialize with current best
    h_current = h.copy()
    current_c5 = best_c5
    
    no_improvement_count = 0
    
    for i in range(max_iters):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Perturbation-based update
        # Compute gradient of the overlap term w.r.t. h
        # We approximate it via finite differences
        eps = 1e-5
        grad = np.zeros_like(h_current)
        
        # Central difference approximation
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
                
            # Project to ensure constraints
            h_plus = project_to_constraints(h_plus, target_sum)
            h_minus = project_to_constraints(h_minus, target_sum)
                
            c5_plus = evaluate_c5(h_plus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
                
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient step
        h_new = h_current - learning_rate * grad
        
        # Project
        h_new = project_to_constraints(h_new, target_sum)
        
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            h_current = h_new.copy()
            
            # Update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if no_improvement_count > 50:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                break
        
        if no_improvement_count > patience:
            break
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
