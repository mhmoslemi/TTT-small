def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    """Create a set of basis functions for h(x)."""
    basis = []

        widths = np.random.rand(n_pulses) * 0.3 + 0.1
        # Create the pulses
        for j, pos in enumerate(positions):
            width = int(widths[j] * n_points)
            start = max(0, pos - width // 2)
            end = min(n_points, pos + width // 2)
            h[start:end] = 0.5 + 0.5 * np.random.rand()  # Random amplitude
        basis.append(h)
    # Normalize all basis functions to have same integral
    for i in range(len(basis)):
        basis[i] = basis[i] * (n_points / 2) / np.sum(basis[i])
    return basis

def project_to_feasible(h, target_sum):
    """Project h to feasible region."""
    # Ensure it's in [0, 1]
    h = np.clip(h, 0, 1)
    # Ensure correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h
