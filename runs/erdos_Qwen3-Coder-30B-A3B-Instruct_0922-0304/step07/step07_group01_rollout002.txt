import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def initial_step_function(n_points, target_sum):
    # Start with a piece-wise constant construction
    h = np.zeros(n_points)
    # Approximate a step function based on the convolution properties
    # Use a known good pattern and perturb it slightly to avoid local minima
    mid = n_points // 2
    h[:mid] = 1.0
    h = project_to_constraints(h, target_sum)
    return h

def convex_relaxation(n_points, target_sum, dx):
    # Convex relaxation to get a continuous initial solution
    h = cp.Variable(n_points)
    # Objective: maximize C subject to the constraints
    C = cp.Variable()
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        # Constraint for convolution constraints in terms of C
    ]
    
    # This follows a more sophisticated approach using CVXPY with explicit constraints
    # For simplicity and efficiency, we'll solve a relaxed version
    # Solve relaxed continuous optimization
    h_relaxed = cp.Variable(n_points)
    constraints_relaxed = [
        h_relaxed >= 0,
        h_relaxed <= 1,
        cp.sum(h_relaxed) == target_sum
    ]
    
    # Objective: use sum of h to avoid constraints complexity
    objective_relaxed = cp.Minimize(cp.sum(h_relaxed))
    
    prob = cp.Problem(objective_relaxed, constraints_relaxed)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if h_relaxed.value is not None:
            return h_relaxed.value
    except:
        pass
    # Fallback to a simple initialization
    return np.ones(n_points) * (target_sum / n_points)

def perturb_step_function(h, target_sum):
    # Convert to step function with a more sophisticated pattern
    h_c = h.copy()
    h_c = project_to_constraints(h_c, target_sum)
    
    # Apply some perturbation to the step positions
    h_step = h_c.copy()
    # Try to improve the current solution
    for i in range(len(h_step)):
        if i < len(h_step) - 1:
            h_step[i] = h_c[i]
    
    # Now-adjust sum while maintaining bounds
    h_step = project_to_constraints(h_step, target_sum)
    return h_step

def adjust_to_step_function(h, target_sum, dx):
    # Convert continuous to step function, respecting sum constraint
    h_step = np.zeros_like(h)
    # Place a step at location that balances the integral
    mid = len(h) // 2
    h_step[:mid] = 1
    h_step = project_to_constraints(h_step, target_sum)
    return h_step

def smooth_step_function(h, target_sum, dx):
    # Smooth assuming a continuous variation and adjust for sum
    # Thresholding approach for step function
    # With threshold 't', set h_step[i] = 1 for those h[i] >= t
    # Find appropriate t to match sum constraint
    
    # Sort and find threshold
    h_sorted = np.sort(h)[::-1]
    # Find threshold 't' to get the same sum
    cumulative_sum = np.cumsum(h_sorted)
    threshold_index = np.searchsorted(cumulative_sum, target_sum, side='right')
    if threshold_index < len(h_sorted):
        t = h_sorted[threshold_index]
    else:
        t = 0.0
        threshold_index = len(h_sorted)
    
    # Apply threshold
    h_step = np.zeros_like(h)
    h_step[h >= t] = 1.0
    
    # Adjust for sums with constraints
    h_step = project_to_constraints(h_step, target_sum)
    
    return h_step

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = project_to_constraints(h, target_sum)

    try:

        if relaxed_sol is not None:
            h = relaxed_sol.copy()
    except:
        pass
    h = project_to_constraints(h, target_sum)

    # Convert to a well-defined step function
    h = smooth_step_function(h, target_sum, dx)
    h = project_to_constraints(h, target_sum)
    
    # Refinement using gradient-based method

        return compute_c5(h, dx)
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Use a few iterations of local search
    max_iter = 50
    step_size = 0.05
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local perturbation - change one or two elements
        h_new = h.copy()
        # Perturb a few indices
        indices_to_change = np.random.choice(n_points, size=min(5, n_points//5), replace=False)
        for i in indices_to_change:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, step_size), 0, 1)
        
        # Re-project to satisfy constraints
        h_new = project_to_constraints(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)
        
        # Accept better or random solutions
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            # Early termination if target achieved
            if best_c5 <= 0.3807:
                break
    
    # Final refinement through enumeration if time still allows
    if budget_s - (time.time() - start_time) > 5:
        h_final = best_h
        best_c5_final = best_c5
        
        # Try several more local modifications
        num_iter = min(100, int((budget_s - (time.time() - start_time)) / 2))
        for _ in range(num_iter):
            h_try = h_final.copy()
            if np.random.random() < 0.5:
                # Modify one step position
                idx = np.random.randint(0, n_points-1)
                h_try[idx] = np.clip(h_try[idx] + np.random.normal(0, 0.01), 0, 1)
            else:
                # Add some noise to a section
                start_idx = np.random.randint(0, n_points - 5)
                for j in range(start_idx, min(start_idx + 5, n_points)):
                    h_try[j] = np.clip(h_try[j] + np.random.normal(0, 0.01), 0, 1)
                    
            h_try = project_to_constraints(h_try, target_sum)
            candidate_c5 = compute_c5(h_try, dx)
            if candidate_c5 < best_c5_final:
                best_c5_final = candidate_c5
                h_final = h_try
                
        best_h = h_final
        best_c5 = best_c5_final
        
    return (best_h.tolist(), best_c5, n_points)
