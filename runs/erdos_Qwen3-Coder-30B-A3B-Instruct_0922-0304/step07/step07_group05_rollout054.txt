import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with better diversity
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = normalize_h(h_base, target_sum)
    else:
        # Create base function
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1
        h_base += np.random.normal(0, 0.02, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = normalize_h(h_base, target_sum)
    
    # Create diverse initial population
    for i in range(population_size):
        if i < population_size // 5:  # 20% from base solution
            h = h_base + np.random.normal(0, 0.03, n_points)
        else:  # 80% random
            h = np.random.uniform(0, 1, n_points)
            h = normalize_h(h, target_sum)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    best_h = h_base.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 60
    w_start = 0.8
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize PSO particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        h = population[np.random.randint(0, len(population))].copy()
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(h.copy())
        pbest_vals.append(compute_c5(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest_h = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start + (w_end - w_start) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (pbest[i] - particles[i])
                           + c2 * r2 * (gbest_h - particles[i]))
            
            # Apply velocity and limit to max speed
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = fitness
            
            # Update global best
            if fitness < gbest_val:
                gbest_h = particles[i].copy()
                gbest_val = fitness
        
        # Early termination if we reach our target
        if gbest_val <= 0.3807:
            break
    
    # Refinement phase with local search
    refined_h = gbest_h.copy()
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, refined_h.copy(), method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(refined_h, dx)
    return (refined_h.tolist(), final_c5, n_points)
