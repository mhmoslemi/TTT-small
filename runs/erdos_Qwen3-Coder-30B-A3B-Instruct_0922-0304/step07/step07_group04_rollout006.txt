import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Initialize population
    population_size = 100
    population = []
    
    # Add initial solution
    population.append(h.copy())
    
    # Add random solutions
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = np.clip(h_random, 0, 1)
        current_sum = np.sum(h_random)
        if current_sum > 0:
            h_random = h_random * (target_sum / current_sum)
        population.append(h_random)
    
    # Genetic algorithm parameters
    max_generations = 1000
    tournament_size = max(2, population_size // 10)
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    
    generation = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while generation < max_generations and time.time() - start_time < budget_s - 5:
        # Evaluate fitness
        fitness = []
        for individual in population:
            c5_val = evaluate_overlap(individual, dx)
            fitness.append(c5_val)
        
        # Find best individual
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement > max_no_improvement:
            break
            
        # Selection (tournament)
        new_population = []
        
        # Elitism - Keep best individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parents_indices = np.argsort(tournament_fitness)[:2]
            parent1 = population[tournament_indices[parents_indices[0]]]
            parent2 = population[tournament_indices[parents_indices[1]]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Blending crossover
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Copy one parent
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                mask = np.random.rand(n_points) < 0.05  # Some elements mutated
                mutation_strength = 0.1 * (1 - generation / max_generations)  # Adaptive
                noise = np.random.normal(0, mutation_strength, n_points)
                child += noise * mask
                
            # Clip to [0,1]
            child = np.clip(child, 0, 1)
            
            # Renormalize
            current_sum = np.sum(child)
            if current_sum > 0:
                child = child * (target_sum / current_sum)
                
            new_population.append(child)
            
        population = new_population
        generation += 1
        
        # Adapt mutation rate
        if generation % 100 == 0 and mutation_rate > 0.01:
            mutation_rate *= 0.95
    
    # Local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    # Use the best solution found to refine locally
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
