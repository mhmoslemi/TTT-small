import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def project_to_bounds(h):
    return np.clip(h, 0, 1)

def initialize_from_parent(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        return h
    else:
        # Initialize with a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        return h

def convex_relaxation(n_points, target_sum):
    # Convex optimization using CVXPY
    h = cp.Variable(n_points)
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Objective: minimize max convolution value
    # We'll solve the problem differently:
    # minimize t such that all convolutions are <= t
    # But since this is non-convex, we solve with a rough approximation by using 
    # a set of sample points and a smoothing approach.
    
    # For now, use a more direct approach: target similar structure
    h0 = initialize_from_parent(n_points)
    
    # Apply simple continuous relaxation and smooth
    h0 = project_to_bounds(h0)
    h0 = project_to_sum_constraint(h0, target_sum)
    
    return h0

def step_function_from_continuous(h_continuous, n_points, target_sum):
    # Convert continuous solution to binary-like step function
    h = h_continuous.copy()
    h = project_to_bounds(h)
    h = project_to_sum_constraint(h, target_sum)
    return h

def local_refinement(h, n_points, target_sum, dx, max_iter=20):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Small perturbations to positions and values
    step_size = 0.01
    for _ in range(max_iter):
        h_new = best_h.copy()
        # Randomly perturb a few points
        indices = np.random.choice(n_points, size=min(5, n_points//4), replace=False)
        for i in indices:
            pert = np.random.uniform(-step_size, step_size)
            h_new[i] += pert
        h_new = project_to_bounds(h_new)
        h_new = project_to_sum_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial solution from parent or default
    h = initialize_from_parent(n_points)
    h = project_to_bounds(h)
    h = project_to_sum_constraint(h, target_sum)
    
    # Try convex relaxation
    try:
        h_convex = convex_relaxation(n_points, target_sum)
        h = h_convex
        h = project_to_sum_constraint(h, target_sum)
    except Exception as e:
        pass  # Continue with current solution if convex fails
    
    # Local refinement
    best_h, best_c5 = local_refinement(h, n_points, target_sum, dx, max_iter=50)
    
    # Keep optimizing with particle swarm or similar if time permits
    # Limit steps to prevent time overrun
    max_iter_total = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 5 and iter_count < max_iter_total:
        # Generate new candidates with small random changes
        h_candidate = best_h.copy()
        indices = np.random.choice(n_points, size=min(3, n_points//5), replace=False)
        for i in indices:
            pert = np.random.normal(0, 0.02)
            h_candidate[i] += pert
        
        h_candidate = project_to_bounds(h_candidate)
        h_candidate = project_to_sum_constraint(h_candidate, target_sum)
        
        c5_candidate = compute_c5(h_candidate, dx)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
        
        iter_count += 1
    
    # Final validation and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
