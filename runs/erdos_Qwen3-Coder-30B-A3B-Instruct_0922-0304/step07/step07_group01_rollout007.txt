import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def convex_relaxation_initialization(h, target_sum, dx):
    n = len(h)
    # Formulate convex relaxation
    h_cont = cp.Variable(n)
    
    # Objective: minimize max overlap
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_cont >= 0,
        h_cont <= 1,
        cp.sum(h_cont) == target_sum
    ]
    
    # Add overlap constraints
    for k in range(2 * n - 1):
        # This is an approximation of the full overlap constraint
        # We'll use a few sample lags to make this tractable
        if k < n:
            shifted = cp.concat([cp.zeros(k), h_cont[:n-k]], axis=0)
        else:
            shifted = cp.concat([h_cont[k-n+1:], cp.zeros(n-(k-n+1))], axis=0)
        overlap_k = cp.sum(cp.multiply(h_cont, 1 - shifted))
        constraints.append(overlap_k <= C)
    
    # Solve the convex optimization problem
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status == cp.OPTIMAL:
        init_h = h_cont.value
    else:
        # Fallback to the original h
        init_h = h.copy()
    
    # Make it a step function
    threshold = 0.5
    step_h = (init_h > threshold).astype(float)
    step_h = project_to_constraints(step_h, target_sum)
    
    return step_h

def gradient_descent_refinement(h, target_sum, dx, max_iter=1000, learning_rate=0.01):
    n = len(h)
    h_current = h.copy()
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    for iter in range(max_iter):
        # Compute gradients using finite differences
        c5_current = compute_c5(h_current, dx)
        
        gradient = np.zeros(n)
        eps = 1e-5
        
        for i in range(n):
            h_plus = h_current.copy()
            h_plus[i] += eps
            
            h_plus = project_to_constraints(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            gradient[i] = (c5_plus - c5_current) / eps
        
        # Update step with projected gradient descent
        h_new = h_current - learning_rate * gradient
        h_new = project_to_constraints(h_new, target_sum)
        h_current = h_new
        
        c5_new = compute_c5(h_current, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_current.copy()
            
    return best_h, best_c5

    dx = 2.0 / len(h)
    target_sum = len(h) / 2
    
    h = project_to_constraints(h, target_sum)
    
    # Stage 1: Convex relaxation
    if time.time() - start_time < budget_s / 3:
        h = convex_relaxation_initialization(h, target_sum, dx)
        h = project_to_constraints(h, target_sum)
        
    # Stage 2: Gradient descent refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 1:
        h, final_c5 = gradient_descent_refinement(

        )
