import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):

    n = len(h)
    grad = np.zeros(n)

    for i in range(n):
        for k in range(n):
            j = i + k

                # but for gradient, we compute one specific overlap value
                overlap_k = h[i] * (1 - h[j])
                grad[i] += (1 - 2 * h[j]) * dx
                
    return grad
