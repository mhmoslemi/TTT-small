import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, dx, max_iter=1000):
    """Local optimization using L-BFGS-B."""
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h, dx)
    
    def gradient(h_flat):
        # Simple finite difference gradient approximation
        eps = 1e-6
        grad = np.zeros_like(h_flat)
        for i in range(len(h_flat)):
            h_plus = h_flat.copy()
            h_plus[i] += eps
            h_minus = h_flat.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Box constraints
    bounds = [(0, 1) for _ in range(len(h))]
    
    # Initial guess
    x0 = h.flatten()
    
    # Optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': max_iter, 'disp': False},
                      callback=None)
    
    if result.success:
        h_opt = result.x.reshape(h.shape)
        h_opt = project_to_constraints(h_opt, len(h) / 2)
        return h_opt
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # Evolution Strategies parameters
    population_size = 20
    mutation_strength_start = 0.1
    mutation_strength_end = 0.01
    max_generations = 2000
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Create initial population
    for i in range(population_size):
        if 'initial_h_values' in globals():
            h_new = np.array(initial_h_values)
            h_new += np.random.normal(0, mutation_strength_start, n_points)
        else:
            h_new = np.random.uniform(0, 1, n_points)
        h_new = project_to_constraints(h_new, target_sum)
        population.append(h_new)
        population_fitness.append(compute_c5(h_new, dx))
    
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_fitness = population_fitness[best_idx]
    
    # ES main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        for i in range(population_size):
            fitness = compute_c5(population[i], dx)
            population_fitness[i] = fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best
        if population_fitness[0] < best_fitness:
            best_h = population[0].copy()
            best_fitness = population_fitness[0]
        
        # Early termination
        if best_fitness <= 0.3807:
            break
        
        # Selection (elitism)
        elite_size = population_size // 2
        elite = population[:elite_size]
        elite_fitness = population_fitness[:elite_size]
        
        # Mutation with adaptive sigma
        sigma = mutation_strength_start - (mutation_strength_start - mutation_strength_end) * generation / max_generations
        
        # Generate new population
        new_population = elite.copy()
        new_fitness = elite_fitness.copy()
        
        for i in range(elite_size, population_size):
            parent = elite[np.random.randint(0, elite_size)]
            child = parent + np.random.normal(0, sigma, n_points)
            child = project_to_constraints(child, target_sum)
            new_population.append(child)
            new_fitness.append(compute_c5(child, dx))
        
        population = new_population
        population_fitness = new_fitness
        
        # Trigger local optimization on best candidates
        for i in range(min(3, len(population))):
            if population_fitness[i] < 0.381:
                candidate = population[i]
                optimized = local_optimize(candidate, dx, max_iter=1000)
                new_fitness = compute_c5(optimized, dx)
                if new_fitness < population_fitness[i]:
                    population[i] = optimized
                    population_fitness[i] = new_fitness
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
