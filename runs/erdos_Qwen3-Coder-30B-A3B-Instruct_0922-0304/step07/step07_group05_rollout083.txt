import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with initial_h_values if available
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Create initial population
    # Use 40% of population from variations of initial solution
    for _ in range(population_size // 2):
        individual = h + np.random.normal(0, 0.01, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    # Fill remaining with random feasible solutions
    for _ in range(population_size // 2):
        individual = np.random.rand(n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # PSO parameters
    inertia_weight = 0.8
    cognitive_weight = 1.5
    social_weight = 1.5
    max_iter = 60
    
    # Initialize particles and velocities
    positions = np.array(population)
    velocities = np.random.uniform(-0.1, 0.1, (population_size, n_points))
    personal_best_positions = positions.copy()
    personal_best_fitness = [compute_c5(p, dx) for p in positions]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (personal_best_positions[i] - positions[i]) +
                           social_weight * r2 * (global_best_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = normalize_h(positions[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(positions[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best_positions[i] = positions[i].copy()
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best_position = positions[i].copy()
        
        # Adjust inertia weight
        inertia_weight *= 0.98
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Local optimization with gradient descent on best solution
    best_h = global_best_position.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Run gradient descent for local refinement
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50, 'ftol': 1e-9})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
            
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
