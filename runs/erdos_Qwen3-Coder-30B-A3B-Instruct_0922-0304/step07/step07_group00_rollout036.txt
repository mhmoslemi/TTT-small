def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")

def compute_overlap_matrix(basis, dx):

    A = np.zeros((n_basis, n_basis, 2*n_points-1))

            # Convolve basis[i] with (1 - basis[j])
            conv = correlate(basis[i], 1 - basis[j], mode="full")
            A[i,j,:] = conv * dx
    return A
