import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the objective function."""
    # Compute all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    
    # Compute gradient components for each position
    grad = np.zeros(n_points)
    for i in range(n_points):
        for k in range(-n_points + 1, n_points):
            # Get corresponding overlap value
            overlap_idx = k + n_points - 1
            if overlap_idx >= 0 and overlap_idx < len(overlap):
                # Adjusted by dx for the integral approximation
                grad[i] += 2 * overlap[overlap_idx] * dx * (1 - 2 * h[(i + k) % n_points])
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the existing construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimizer state
    lr = 0.01
    decay_rate = 0.999
    patience = 0
    patience_limit = 1000
    last_improvement = 0
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Update
        h = h - lr * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Evaluate
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count
        
        # Update learning rate
        if iter_count % 1000 == 0:
            lr *= decay_rate
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if iter_count - last_improvement > patience_limit:
            break
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
