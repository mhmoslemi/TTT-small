import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Determine initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 100
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    max_iter = 60
    inertia_weight = w_start
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    
    # Generate initial particles
    for i in range(n_particles):
        if i < n_particles // 2 and 'initial_h_values' in globals():
            # Use the initial solution as a base for some particles
            particle = h + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.rand(n_points)
            particle = normalize_h(particle, target_sum)
        particle = np.clip(particle, 0, 1)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        fitness = compute_c5(particle, dx)
        personal_best.append(particle.copy())
        personal_best_fitness.append(fitness)
    
    # Global best initialization
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = inertia_weight
        inertia_weight *= w_decay
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best[i] - particles[i]) + 
                             c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Compute fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
        
        # Early termination if target is reached
        if global_best_fitness <= 0.3807:
            break
    
    # Refine with gradient descent
    best_h = global_best.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Run a few rounds of gradient-based refinement
    best_refined_h = best_h.copy()
    best_refined_c5 = global_best_fitness
    
    for _ in range(20):  # Perform several rounds of local optimization
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_refined_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_refined_h = result.x
            best_refined_h = np.clip(best_refined_h, 0, 1)
            best_refined_h = normalize_h(best_refined_h, target_sum)
            new_c5 = compute_c5(best_refined_h, dx)

                best_refined_c5 = new_c5
        else:
            break
    
    # Final check and return
    final_c5 = compute_c5(best_refined_h, dx)
    return (best_refined_h.tolist(), final_c5, n_points)
