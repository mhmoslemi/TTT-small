import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a smoother initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
        
    # Differential Evolution parameters
    population_size = 60
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iterations = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, target_sum)
        else:
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, target_sum)
        population.append(individual.copy())
        fitness.append(compute_c5(individual, dx))
    
    # Track the best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        if iteration % 100 == 0:
            # Check if we're good enough to stop early
            if best_c5 <= 0.3807:
                break
        
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Mutation: select three different individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Create mutant vector
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.copy(population[i])
            crossover_points = np.random.rand(n_points) < crossover_prob
            trial[crossover_points] = mutant[crossover_points]
            
            # Project to constraint
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
            # Update global best
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5
        
        population = new_population
        fitness = new_fitness
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
