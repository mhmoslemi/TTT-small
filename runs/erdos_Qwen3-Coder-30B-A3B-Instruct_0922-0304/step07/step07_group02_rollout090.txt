import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h_relaxed, target_sum):
    """Project h_relaxed to feasible [0,1] region and enforce sum constraint."""
    h = np.clip(h_relaxed, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to meet sum constraint
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load initial if available
    if 'initial_h_values' in globals():
        h_relaxed = np.array(initial_h_values)
    else:
        # Initialize with alternating pattern
        h_relaxed = np.zeros(n_points)
        h_relaxed[::2] = 1.0
        h_relaxed[1::2] = 0.0
        # Add noise
        h_relaxed += np.random.normal(0, 0.1, n_points)
    
    # Enforce sum constraint on initialization
    h_relaxed = project_to_feasible(h_relaxed, target_sum)
    
    # Parameters
    alpha = 0.1  # learning rate
    beta = 0.9   # momentum
    reg_lambda = 1e-4  # regularization
    
    # Momentum and velocity for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h_relaxed.copy()
    best_c5 = evaluate_c5(best_h, dx)
    patience = 0
    max_patience = 200
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 3000:
        # Compute current C5
        current_c5 = evaluate_c5(h_relaxed, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_relaxed.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break

        # Gradient estimation using finite differences
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h_relaxed.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update using Adam
        m = beta * m + (1 - beta) * grad
        v = beta * v + (1 - beta) * grad**2
        m_hat = m / (1 - beta**(iteration+1))
        v_hat = v / (1 - beta**(iteration+1))
        
        # Apply gradients, regularize
        delta = alpha * m_hat / (np.sqrt(v_hat) + 1e-8) - reg_lambda * (h_relaxed - best_h)
        h_relaxed -= delta
        
        # Apply constraints
        h_relaxed = project_to_feasible(h_relaxed, target_sum)
        
        # Adaptive regularization and learning rate
        if patience > 100:
            alpha *= 0.9
            patience = 0
        
        if iteration % 20 == 0:
            # Increase regularization if necessary
            reg_lambda *= 1.05
            
        iteration += 1
    
    # Final check
    final_h = project_to_feasible(best_h, target_sum)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
