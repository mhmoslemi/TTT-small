import numpy as np
from scipy.signal import correlate
import time
from scipy.spatial.distance import cdist

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def gaussian_mixture_sample(means, covariances, weights, n_samples, n_points):
    """Sample from a mixture of Gaussians."""
    # Sample component indices
    component_indices = np.random.choice(len(weights), size=n_samples, p=weights)
    samples = np.zeros((n_samples, n_points))
    
    for i in range(n_samples):
        idx = component_indices[i]
        samples[i] = np.random.multivariate_normal(means[idx], covariances[idx])
    
    return samples

def update_gmm(means, covariances, weights, selected_samples, n_points):
    """Update GMM parameters based on selected samples."""
    n_components = len(means)
    n_selected = len(selected_samples)
    
    # Update weights (normalized)
    weights = np.array([np.sum(selected_samples == means[i]) for i in range(n_components)])
    weights = weights / np.sum(weights)
    
    # Update means
    updated_means = []
    for i in range(n_components):
        if n_selected > 0:
            mask = np.random.choice(n_selected, size=max(1, n_selected // n_components), replace=True)
            selected_for_component = selected_samples[mask]
            updated_means.append(np.mean(selected_for_component, axis=0))
        else:
            updated_means.append(means[i])
    
    # Update covariances (simplified version)
    updated_covariances = []
    for i in range(n_components):
        if n_selected > 1:
            mask = np.random.choice(n_selected, size=max(1, n_selected // n_components), replace=True)
            selected_for_component = selected_samples[mask]
            center = updated_means[i]
            centered_samples = selected_for_component - center
            # Estimate covariance with regularization
            if len(selected_for_component) == 1:
                cov = np.eye(n_points) * 1e-8
            else:
                cov = np.cov(centered_samples.T) + np.eye(n_points) * 1e-10
            updated_covariances.append(np.clip(cov, 1e-8, None))
        else:
            updated_covariances.append(covariances[i])
    
    return updated_means, updated_covariances, weights

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial construction
        h = np.ones(n_points) * 0.5
        h[:n_points//4] = 1.0
        h[n_points//4:n_points//2] = 0.0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    
    # Initialize GMM components
    means = []
    covariances = []
    weights = np.ones(n_components) / n_components
    
    # Start with a point
    means.append(h.copy())
    covariances.append(np.eye(n_points) * 0.01)
    
    # Add random components
    for _ in range(n_components - 1):
        random_h = np.random.rand(n_points)
        random_h = np.clip(random_h, 0, 1)
        random_h = normalize_h(random_h, target_sum)
        means.append(random_h.copy())
        covariances.append(np.eye(n_points) * 0.01)
    
    # Main optimization loop
    best_solution = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iter = 6000  # Still bounded by time
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample population
        samples = gaussian_mixture_sample(means, covariances, weights, population_size, n_points)
        
        # Evaluate and constrain samples
        fitness_values = []
        constrained_samples = []
        
        for sample in samples:
            # Clip to [0,1]
            sample = np.clip(sample, 0, 1)
            # Project to constraint
            sample = project_to_constraint(sample, target_sum)
            # Compute C5
            fitness = compute_c5(sample, dx)
            fitness_values.append(fitness)
            constrained_samples.append(sample)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_values)
        top_samples = [constrained_samples[i] for i in sorted_indices[:keep_top]]
        top_fitness = [fitness_values[i] for i in sorted_indices[:keep_top]]
        
        # Update best solution
        if top_fitness[0] < best_c5:
            best_c5 = top_fitness[0]
            best_solution = top_samples[0].copy()
            
        # Update GMM
        if len(top_samples) > 0:
            means, covariances, weights = update_gmm(means, covariances, weights, 
                                                   np.array(top_samples), n_points)
            
            # Decay covariances
            for i in range(len(covariances)):
                covariances[i] = covariances[i] * covariance_decay
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_solution.tolist(), best_c5, n_points)
