import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a reasonable starting point
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Annealing parameters
    T = 1.0
    T_min = 1e-8
    alpha = 0.999
    max_iter = int(1e5)
    
    # Penalty weights
    lambda1 = 100.0
    lambda2 = 100.0
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    def compute_energy(h, dx):
        c5 = evaluate_overlap(h, dx)
        # Constraint penalties
        penalty_bound = np.sum((h - np.clip(h, 0, 1)) ** 2)
        penalty_sum = (np.sum(h) - target_sum) ** 2
        return c5 + lambda1 * penalty_bound + lambda2 * penalty_sum
    
    # Simulated Annealing
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h
        h_new = h.copy()
        # Use a small learning rate that decreases with T
        lr = 0.01 * T
        perturbation = np.random.normal(0, lr, n_points)
        h_new += perturbation
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Compute energies
        energy_old = compute_energy(h, dx)
        energy_new = compute_energy(h_new, dx)
        
        # Accept or reject
        delta_e = energy_new - energy_old
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h = h_new.copy()
            if energy_new < best_c5:
                best_h = h_new.copy()
                best_c5 = energy_new
        
        # Cool down
        T = max(T_min, T * alpha)
    
    # Final refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
