import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-6
    max_iter = int(1e5)
    lr = 0.01
    penalty_weight = 100.0
    
    def compute_energy(h_vec):
        c5 = evaluate_overlap(h_vec, dx)
        # Constraint penalties
        bound_penalty = np.sum(np.maximum(0, h_vec - 1)**2 + np.maximum(0, -h_vec)**2)
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        energy = c5 + penalty_weight * (bound_penalty + sum_penalty)
        return energy, c5
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Make a perturbation
        h_new = h.copy()
        perturb = np.random.normal(0, lr * T, n_points)
        h_new += perturb
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Compute energies
        _, c5_old = compute_energy(h)
        energy_new, c5_new = compute_energy(h_new)
        
        # Accept or reject
        delta_E = energy_new - c5_old
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Update temperature
        T = max(min_T, T * cooling_rate)
        
        # Occasionally adjust learning rate
        if i % 1000 == 0:
            lr = max(1e-6, lr * 0.995)
    
    # Gradient refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        def gradient(h_vec):
            # Approximate gradient using finite differences
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (target_sum / h_plus_sum)
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus_sum = np.sum(h_minus)
                if h_minus_sum > 0:
                    h_minus = h_minus * (target_sum / h_minus_sum)
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
            return grad
        
        # Gradient descent
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            grad = gradient(h)
            h = h - 0.005 * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (target_sum / h_sum)
            c5_new = evaluate_overlap(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
