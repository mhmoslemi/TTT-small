import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iterative refinement using quadratic approximation
    max_iterations = 60
    patience = 10
    no_improvement_count = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience or best_c5 <= 0.3807:
            break
            
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            
            grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
        
        # Quadratic approximation: we approximate the objective around current point
        # For simplicity, we construct a QP with a diagonal Hessian
        # Hessian approximation: diagonal Hessian elements from finite differences
        # The actual objective might be complex, so we simplify using a common strategy:
        # Approximate the Hessian as a diagonal matrix with small positive values
        
        # For faster optimization, we can also do coordinate descent or other simple methods
        
        # Instead of computing full Hessian, let's approximate Hessian with simple diagonal
        # and solve a simpler QP:
        # minimize  0.5 * d^T * H * d + grad^T * d
        # subject to: 0 <= h + d <= 1, sum(d) = 0
        
        # Using convex optimization for faster local refinement
        d = cp.Variable(n_points)
        # Directly attempt to minimize gradient descent along estimated direction
        
        # Simple line search
        step_size = 0.1
        new_h = h - step_size * grad
        new_h = project_to_feasible(new_h, target_sum)
        
        # Try a line search
        step_sizes = [0.01, 0.05, 0.1, 0.2, 0.5]
        best_new_h = h.copy()
        best_new_c5 = current_c5
        
        for s in step_sizes:
            candidate_h = h - s * grad
            candidate_h = project_to_feasible(candidate_h, target_sum)
            candidate_c5 = compute_c5(candidate_h, dx)
            if candidate_c5 < best_new_c5:
                best_new_c5 = candidate_c5
                best_new_h = candidate_h.copy()
        
        h = best_new_h.copy()
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
