import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    target_sum = n_points / 2

        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    if 'initial_h_values' in globals():
        means.append(h.copy())
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
    else:
        means.append(h.copy())
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
    
    # Add random components
    for _ in range(n_components - 1):
        comp = np.random.rand(n_points) * 0.5
        means.append(comp)
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iterations = 6000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Sample candidates from GMM
        candidates = []
        candidate_scores = []
        
        for _ in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            mean = means[comp_idx]
            cov = covariances[comp_idx]
            
            # Sample from Gaussian
            sample = np.random.multivariate_normal(mean, cov)
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            
            c5 = compute_c5(sample, dx)
            candidates.append(sample.copy())
            candidate_scores.append(c5)
        
        # Sort by score
        sorted_indices = np.argsort(candidate_scores)
        top_candidates = [candidates[i] for i in sorted_indices[:keep_top]]
        top_scores = [candidate_scores[i] for i in sorted_indices[:keep_top]]
        
        # Update GMM components
        # Update means
        for i in range(n_components):
