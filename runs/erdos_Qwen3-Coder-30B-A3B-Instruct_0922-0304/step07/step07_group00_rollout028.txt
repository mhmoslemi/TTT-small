import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of C5 approximation with respect to h."""
    # This is an approximation - we will use finite differences
    # for a more accurate gradient approximation
    grad = np.zeros_like(h)
    # We'll compute using central differences for better accuracy
    eps = 1e-6
    base_c5 = compute_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = np.clip(h_plus, 0, 1)
        h_minus = np.clip(h_minus, 0, 1)
        h_plus = normalize_h(h_plus, n_points / 2)
        h_minus = normalize_h(h_minus, n_points / 2)
        grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
    
    return grad

def optimize_h_adam(h, dx, n_points, max_iter=10000, lr=0.01, decay_rate=0.999, tol=1e-8):
    """Optimize h using Adam-like gradient descent."""
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    
    for t in range(max_iter):
        if no_improvement_count > 1000:
            break
        grad = compute_gradient(h, dx, n_points)
        
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        h_new = h - step
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2)
        
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if best_c5 <= 0.3807:
            break
            
        h = h_new
        
        lr *= decay_rate
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, n_points / 2)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    while time.time() - start_time < budget_s - 1:
        # Optimize with Adam
        h_opt, c5_opt = optimize_h_adam(h, dx, n_points, max_iter=500, lr=0.01)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
        if best_c5 <= 0.3807:
            break
            
        # Perturb with small noise to escape local minima
        h = best_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    
    # Final verification
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
