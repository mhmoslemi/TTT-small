import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_episodes = 2000
    epsilon_start = 0.1
    epsilon_end = 0.01
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function with smooth transitions
        h = np.zeros(n_points)
        # Place a peak in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # RL-based refinement
    np.random.seed(seed)
    actions = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    # Simple policy network weights
    W1 = np.random.normal(0, 0.1, (2, 32))
    b1 = np.random.normal(0, 0.1, 32)
    W2 = np.random.normal(0, 0.1, (32, len(actions)))
    b2 = np.random.normal(0, 0.1, len(actions))
    
    # Optimization parameters
    learning_rate = 0.001
    episodes_without_improvement = 0
    min_improvement = 1e-6
    
    for episode in range(max_episodes):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate a new solution using policy
        h_new = np.zeros(n_points)
        cumulative_sum = 0.0
        
        # Sample actions with epsilon-greedy policy
        for i in range(n_points):
            # State: (position, cumulative sum)
            state = np.array([i, cumulative_sum])
            state_normalized = np.array([state[0] / n_points, state[1] / target_sum])
            
            # Simple policy network forward pass
            hidden = np.maximum(0, np.dot(W1, state_normalized) + b1)
            logits = np.dot(W2, hidden) + b2
            probs = np.exp(logits - np.max(logits))  # For numerical stability
            probs = probs / np.sum(probs)
            
            # Epsilon-greedy action selection
            epsilon = epsilon_start + (epsilon_end - epsilon_start) * (episode / max_episodes)
            if np.random.rand() < epsilon:
                action_idx = np.random.choice(len(actions), p=probs)
            else:
                action_idx = np.argmax(probs)
                
            action = actions[action_idx]
            h_new[i] = action
            cumulative_sum += action
            
        # Normalize to target sum
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new solution
        c5_new = evaluate_overlap(h_new, dx)
        
        # Update policy if improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            episodes_without_improvement = 0
        else:
            episodes_without_improvement += 1
            
        # Policy update with REINFORCE
        if c5_new < best_c5 - min_improvement:
            # Simple policy gradient update
            g = (best_c5 - c5_new) * 0.001  # Simple reward signal
            
            # Unroll the gradient computation for brevity
            # This simplistic update does not compute exact gradients but directly
            # adjusts weights based on performance signal
            W1 += learning_rate * g * np.ones_like(W1)
            W2 += learning_rate * g * np.ones_like(W2)
            b1 += learning_rate * g * np.ones_like(b1)
            b2 += learning_rate * g * np.ones_like(b2)
            
        # Termination check
        if episodes_without_improvement > 50:
            break
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
