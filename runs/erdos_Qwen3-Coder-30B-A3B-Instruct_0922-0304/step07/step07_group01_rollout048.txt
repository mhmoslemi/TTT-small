import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points) * 0.5

    # Fourier domain approach
    # Use CVXPY for convex optimization in frequency domain
    H = cp.Variable(n_points, complex=True)
    h_var = cp.Variable(n_points)

    # Use the constraints from the problem to build the optimization
    # Minimize max overlap in the frequency domain
    # H is the FFT of h
    # We want to minimize the maximum value of the convolution h * (1-h)

    # Objective (approximated)
    max_overlap = cp.Variable()

    # Constraints for h in [0,1] and sum = target_sum
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
    ]

    # Fourier transform relationships
    # For real signal h, H[0] is real and H[-i] = conj(H[i])
    # Add constraints to keep h close to original (or improve solution) during optimization

    # This is a relax problem in Fourier space; we make a reasonable guess
    # and do projected gradient descent in time domain with re-projection

    # Improved strategy: start with the initial hand-crafted solution
    # then iteratively optimize with projected gradient descent
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Gradient descent in the space of valid h's with projection
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-6

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break

        # Compute gradient estimates (finite differences)
        # Perturb each point slightly to estimate gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        base_c5 = compute_c5(best_h, dx)
        
        # We only compute gradients to reduce computation
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_to_constraints(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        # Update using projected gradient
        new_h = best_h - step_size * grad
        new_h = project_to_constraints(new_h, target_sum)
        new_c5 = compute_c5(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            step_size = max(1e-6, step_size * 1.05)  # Adaptive step size
        else:
            step_size *= 0.9  # Reduce learning rate if no improvement

        # Early termination
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
