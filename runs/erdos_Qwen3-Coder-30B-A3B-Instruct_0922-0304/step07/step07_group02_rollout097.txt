import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function with some randomness
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # GMM initialization
    means = []
    covariances = []
    weights = []
    
    # Initialize with the best solution we have
    if 'initial_h_values' in globals():
        means.append(np.array(initial_h_values).copy())
    else:
        means.append(h.copy())
    covariances.append(np.eye(n_points) * initial_sigma)
    weights.append(1.0)
    
    # Add some random components for diversity
    for _ in range(n_components - 1):
        init_h = np.random.rand(n_points)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, target_sum)
        means.append(init_h.copy())
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iterations = 6000  # To fit within budget
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Sample from the GMM
        candidates = []
        for _ in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            # Sample from component
            sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            candidates.append(sample)
        
        # Evaluate candidates
        evaluated = []
        for c in candidates:
            # Clip to bounds
            c = np.clip(c, 0, 1)
            # Normalize to correct sum
            c = project_to_constraint(c, target_sum)
            # Evaluate
            c5 = compute_c5(c, dx)
            evaluated.append((c, c5))
        
        # Sort by C5 value
        evaluated.sort(key=lambda x: x[1])
        top_candidates = evaluated[:keep_top]
        
        # Update GMM
        if len(top_candidates) > 0:
            # New means - average of top solutions
            new_means = []
            new_covariances = []
            new_weights = []
            
            # Get the selected candidates
            selected_h = [c[0] for c in top_candidates]
            selected_c5 = [c[1] for c in top_candidates]
            
            # Update means (weighted by inverse of C5)
            total_weight = sum(1 / (c5 + 1e-12) for c5 in selected_c5)
            for i, (h_cand, c5) in enumerate(zip(selected_h, selected_c5)):
                weight = 1 / (c5 + 1e-12) / total_weight
                if i == 0:
                    new_means.append(h_cand.copy())
                else:
                    # Add weighted average
                    new_means[0] = (new_means[0] * (1 - weight) + h_cand * weight)
            
            # Update covariances
            if len(selected_h) >= 2:
                # Use all selected points for covariance update
                stacked = np.vstack(selected_h)
                mean_point = np.mean(stacked, axis=0)
                diff = stacked - mean_point
                covariance = np.cov(diff.T)
                # Add regularization for numerical stability
                regularization = 1e-8 * np.eye(n_points)
                covariance += regularization
                new_covariances.append(covariance)
            else:
                new_covariances.append(np.eye(n_points) * initial_sigma)
            
            # Keep old means/covariances if needed
            if len(new_means) < n_components:
                for i in range(len(new_means), n_components):
                    new_means.append(means[0].copy())
                    new_covariances.append(covariances[0].copy())
            
            # Update weights (equally weighted for simplicity)
            new_weights = np.ones(n_components) / n_components
            
            # Update parameters
            means = new_means
            covariances = new_covariances
            weights = new_weights
            
            # Decay covariances
            for i in range(len(covariances)):
                covariances[i] *= covariance_decay
            
            # Update best
            current_best_c5 = min(selected_c5)
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = selected_h[0].copy()
        
        iteration += 1
        
        # Check if we've found a sufficiently good solution
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
