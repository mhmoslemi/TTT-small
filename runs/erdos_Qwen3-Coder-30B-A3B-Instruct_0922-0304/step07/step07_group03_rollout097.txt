import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit proportionally to remaining space
    deficit = target_sum - current_sum
    if deficit > 0:
        # Need to increase values
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
    else:
        # Need to decrease values
        scale = target_sum / current_sum
        h *= scale
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 3000

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
        
    # Parameters
    alpha = 0.2
    epsilon = 0.1
    tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        c = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(c)
        
        # Gradient step: gradient of overlap w.r.t. h
        # We approximate the gradient as: - (1-h) shifted appropriately
        shifted = np.roll(1 - h, max_k - (len(h) - 1))
        grad = -shifted
        
        # Update h
        h_candidate = h - alpha * grad
        
        # Random perturbation
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to feasible region
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = candidate_c5
            alpha = 0.2
            epsilon = 0.1
        else:
            alpha *= 0.999
            epsilon *= 0.999
            
        # Adaptive learning rates
        if i > 0:
            # Decay of parameters
            alpha = max(0.001, alpha * (1 + i / tau))
            epsilon = max(0.001, epsilon * (1 + i / tau))
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
