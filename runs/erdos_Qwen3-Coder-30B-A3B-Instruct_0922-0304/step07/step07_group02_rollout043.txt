import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Adjust values
    h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Relaxation for exploration
    h_relaxed = h.copy()
    
    # Parameters
    learning_rate = 0.1
    momentum = 0.9
    lambda_reg = 1e-4
    max_iter = 3000
    patience = 100
    
    # Initialize momentum terms
    velocity = np.zeros(n_points)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        
        # Compute gradient approximation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        # Compute gradient component-wise
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus_proj = np.clip(h_plus, 0, 1)
            h_plus_proj = project_to_constraint(h_plus_proj, target_sum)
            c5_plus = compute_c5(h_plus_proj, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update velocity with momentum
        velocity = momentum * velocity + learning_rate * grad
        
        # Update relaxed values
        h_relaxed -= velocity
        
        # Project to [0, 1]
        h = np.clip(h_relaxed, 0, 1)
        
        # Enforce sum constraint
        h = project_to_constraint(h, target_sum)
        
        # Update regularization parameter
        if current_c5 < best_c5:
            last_improvement = iteration
            best_c5 = current_c5
            best_h = h.copy()

            if iteration - last_improvement > patience:
                learning_rate *= 0.9
                last_improvement = iteration
                if lambda_reg < 1e-2:
                    lambda_reg *= 1.2
