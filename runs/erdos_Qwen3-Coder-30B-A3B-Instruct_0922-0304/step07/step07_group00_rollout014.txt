import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_basis_functions(n_points, n_basis=16):
    """Generate basis functions with varying pulse structures."""
    basis = []
    for i in range(n_basis):
        b = np.zeros(n_points)
        # Distribute pulses from left to right
        n_pulses = 2 + (i % 4)  # 2,3,4,5 pulses
        positions = np.linspace(0, n_points-1, n_pulses+2)[1:-1].astype(int)
        for pos in positions:
            b[max(0, pos-1):min(n_points, pos+2)] = 1.0
        basis.append(b)
    return np.array(basis)

def compute_overlap_integrals(basis, dx):
    """Precompute A_{ij}^k for all basis functions."""
    n_basis = basis.shape[0]
    n_points = basis.shape[1]
    # Precompute integrals for each shift k
    overlaps = {}
    for k in range(2 * n_points):
        overlaps[k] = np.zeros((n_basis, n_basis))
        for i in range(n_basis):
            for j in range(n_basis):
                shift = basis[i] * np.roll(basis[j], k)
                overlaps[k][i, j] = np.sum(shift) * dx
    return overlaps

def solve_convex_relaxation(basis, overlaps, dx, n_points):
    """Solve the convex relaxation of the optimization problem."""
    n_basis = basis.shape[0]
    # Define variables
    lambdas = cp.Variable(n_basis)
    # Convex relaxation to minimize maximum overlap
    obj = cp.Minimize(cp.sum_squares(lambdas))  # Regularization to avoid degeneracy
    constraints = [
        cp.sum(lambdas) == 1,
        lambdas >= 0,
    ]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=CPLEX, verbose=False)
    if prob.status != OPTIMAL:
        return None
    return lambdas.value

def generate_h_from_lambdas(lambdas, basis, dx):
    """Generate h from convex combination of basis functions."""
    h = np.dot(lambdas, basis)
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Normalize integral to 1
    h = h * (1.0 / np.sum(h) * 2.0)  # Integrates to 1 over [0,2]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 16
    
    # Generate basis functions
    basis = compute_basis_functions(n_points, n_basis)
    
    # Precompute overlap integrals
    overlaps = compute_overlap_integrals(basis, dx)
    
    # Initialize with baseline if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        # Project initial h onto basis
        lambdas = solve_convex_relaxation(basis, overlaps, dx, n_points)
        if lambdas is not None:
            h_projected = generate_h_from_lambdas(lambdas, basis, dx)
            # Try to minimize with improved method
            best_h = h_projected
            best_c5 = evaluate_c5(best_h, dx)
        else:
            best_h = np.random.rand(n_points)
            best_h = np.clip(best_h, 0, 1)
            best_c5 = evaluate_c5(best_h, dx)
    else:
        best_h = np.random.rand(n_points)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = evaluate_c5(best_h, dx)
    
    iter_count = 0
    max_iter = 300
    patience = 0
    best_improvement = 0.0

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Solve convex relaxation
        lambdas = solve_convex_relaxation(basis, overlaps, dx, n_points)
        if lambdas is None:
            # Fallback
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        else:
            h = generate_h_from_lambdas(lambdas, basis, dx)
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            best_improvement = max(best_improvement, 0.380972 - c5)
        else:
            patience += 1
            if patience > 50 and best_improvement < 1e-5:
                break
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
