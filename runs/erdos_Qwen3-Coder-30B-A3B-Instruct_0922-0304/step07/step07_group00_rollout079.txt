import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_c5_exact(h, dx):
    """Compute exact C5 value."""
    return compute_c5(h, dx)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def gradient_step(h, dx, lr=0.01):
    """Perform one gradient descent step."""
    n = len(h)
    # Compute overlap values
    h_flip = 1 - h
    overlap = correlate(h, h_flip, mode="full")
    
    # Compute gradient for each position
    grad = np.zeros(n)
    for i in range(n):
        for k in range(n):
            if 0 <= i + k < n:
                grad[i] += 2 * overlap[n-1+k] * (-2 * h[i+k])
    
    # Apply gradient and update
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, n/2)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5_exact(h, dx)
    
    lr = 0.01
    patience = 0
    max_patience = 1000
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient step with adaptive learning rate
        h_new = gradient_step(h, dx, lr)
        c5_new = compute_c5_exact(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
            # Reduce learning rate if improvement is sustained
            if iteration % 1000 == 0 and lr > 1e-6:
                lr *= 0.999
        else:
            patience += 1
        
        h = h_new.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Break if no improvement for too long
        if patience > max_patience:
            break
    
    final_c5 = compute_c5_exact(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
