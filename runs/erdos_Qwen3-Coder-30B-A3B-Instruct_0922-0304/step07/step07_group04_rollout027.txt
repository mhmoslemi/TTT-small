import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_gradient(h, dx):
    """Compute gradient of overlap w.r.t. h."""
    n = len(h)
    grad = np.zeros_like(h)
    
    # For each position i, compute how much changing h[i] affects max overlap
    # We focus on the gradient for the current maximum overlap, which is a complex operation
    # Here we use finite differences as a proxy for gradient
    eps = 1e-6
    base_overlap = evaluate_overlap(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus_sum = np.sum(h_plus)
        if h_plus_sum > 0:
            h_plus = h_plus * (np.sum(h) / h_plus_sum)
        overlap_plus = evaluate_overlap(h_plus, dx)
        grad[i] = (overlap_plus - base_overlap) / eps
    
    return grad

def project_to_feasible(h, target_sum):
    """Project h onto the feasible set: h_i ∈ [0,1] and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

        h = np.clip(h, 0, 1)
        
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based descent with adaptive regularization
    max_iter = 2000
    eta = 1e-3  # Initial learning rate
    beta = 0.9  # Momentum
    lambda_reg = 1e-4  # Regularization strength

    # Nesterov momentum
    v = np.zeros_like(h)

        if time.time() - start_time > budget_s - 1:

        # Compute current gradient
        grad = compute_gradient(h, dx)
        
        # Apply momentum
        v = beta * v - eta * grad
        h_nesterov = h + v
        
        # Project to feasible set
        h_projected = project_to_feasible(h_nesterov, target_sum)
        
        # Adaptive regularization
        h_regularized = h_projected + lambda_reg * (target_sum - np.sum(h_projected)) / n_points
        
        # Ensure bounds
        h_regularized = np.clip(h_regularized, 0, 1)
        
        # Renormalize
        h_regularized_sum = np.sum(h_regularized)

        # Evaluate overlap
