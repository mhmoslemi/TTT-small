import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a structured initial function
        h = np.zeros(n_points)
        # Create a smooth step function with a ramp
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h[i] = x * 2
            elif x < 1.5:
                h[i] = 1
            else:
                h[i] = 2 - x
        
        # Scale to meet constraint
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Iterative refinement using gradient-like updates
    iterations = 0
    max_iterations = 100
    learning_rate = 0.1
    
    # Create frequency domain representation
    h_padded = np.pad(h, (0, n_points), 'constant')
    
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 1:
        if iterations % 10 == 0:
            # Recompute the full overlap for better accuracy
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            max_lag = max_idx - (len(h) - 1)
            # Some heuristic to identify dominant frequency
            max_val = overlap[max_idx]
        else:
            overlap = correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(overlap)
            max_lag = max_idx - (len(h) - 1)
        
        # Try to improve by modifying h at the position that creates most overlap
        perturbation = np.zeros(len(h))
        # Perturb right at the peak overlap lag
        if max_lag > 0 and max_lag < len(h):
            if max_lag < len(h) // 2:
                # modify leading part
                start_idx = 0
                for i in range(min(max_lag + 1, len(h))):
                    if h[i] > 0.01:
                        perturbation[i] += -learning_rate * 0.2  # reduce
                    else:
                        perturbation[i] += learning_rate * 0.5
            else:
                # modify trailing part
                start_idx = max_lag
                for i in range(start_idx, min(len(h), start_idx + 10)):
                    if h[i] < 0.99:
                        perturbation[i] += learning_rate * 0.2  # increase
                    else:
                        perturbation[i] -= learning_rate * 0.5

        # Update h
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        h = h_new
        learning_rate *= 0.99
        iterations += 1
    
    # Final local refinement
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    result = minimize(objective, best_h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 30})
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        final_c5 = evaluate_c5(best_h, dx)
    else:
        final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
