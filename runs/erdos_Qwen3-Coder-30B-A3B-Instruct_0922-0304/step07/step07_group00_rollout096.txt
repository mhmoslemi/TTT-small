import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute the gradient of overlap_k_max with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    # For the overlap at shift k_max, the gradient is (1-2*h) at the shifted positions
    for i in range(n_points - k_max):
        grad[i] += (1 - 2 * h[i + k_max])
    return grad

def solve_lp_direction(grad, n_points):
    """Solve the linear program to find the direction s."""
    # We want to minimize grad^T * s subject to sum(s) = 0, 0 <= s <= 1
    # This is equivalent to finding s that minimizes the linear function
    # This can be solved greedily:
    # Sort gradient values with indices
    idxs = np.argsort(grad)
    # Current solution
    s = np.zeros(n_points)
    # Try to make s as negative as possible among indices with negative grad
    # and positive as much as possible where grad is positive
    # But also ensure sum(s) = 0
    
    # Let's solve the constrained minimization explicitly:
    # We can use the fact that for sum(s) = 0, the optimal is to make s = 0 or 1
    # This usually means making negative grad indices 0 and positive grad indices 1
    # But this needs to satisfy sum(s) = 0.
    # Instead, we use the standard Frank-Wolfe idea:
    # Take the direction that improves the gradient the most in negative direction
    # But here we need sum(s) = 0, so we compute the individual updates
    s = np.zeros(n_points)
    idxs_neg = np.where(grad < 0)[0]
    idxs_pos = np.where(grad > 0)[0]
    
    # Equalize contributions to get sum = 0
    sum_neg = np.sum(grad[idxs_neg]) if len(idxs_neg) > 0 else 0
    sum_pos = np.sum(grad[idxs_pos]) if len(idxs_pos) > 0 else 0
    
    if sum_neg < 0 and sum_pos > 0:
        # We want sum(s) = 0, so we need to balance positive and negative sides
        # One standard way is to make s[i] = 1 for i in idxs_pos until
        # we get sum = 0 or solution is saturated. Then make s[i] = 0 where expected.
        # For greedy minimization subject to constraints, look at the extreme cases.
        # Since we are constrained to 0 <= s[i] <= 1 and sum(s) = 0,
        # the direction that minimizes single components should adjust towards extremes
        # Let's compute the specific direction:
        # We really want a feasible s with sum = 0 such that grad^T s is minimized
        # We can do this with simpler approach: make s as close to zero/one as possible 
        # while sum(s) = 0.
        # This is like assigning 1s and 0s such that sum is 0.
        # Assign s as follows:
        # Or we determine the maximum positive contribution and make it 1,
        # minimum negative contribution make it 0, but we must then adjust to get sum = 0
        
        # A cleaner approach: Since constraints are sum(s) = 0, we can do:
        # s = we choose s in [0,1]^n, sum(s) = 0.
        # min grad^T s such that 0 <= s[i] <= 1 for all i, sum(s) = 0
        # This is essentially minimizing a linear function (grad^T s) subject to a box and sum = 0
        # The solution will basically try to turn around each entry's direction.
        # What's the minimum possible value of this inner product?
        # - If sum of all grad >= 0, turn s=0 (if that's allowed). No, we can't just zero everything
        # - We want sum(s) = 0 but s in [0,1]. 
        # - Best s in terms of minimizing inner product is if we make s as negative as
        # we can while still satisfying sum(s) = 0, 0 <= s <= 1.
        # But in our case we don't have s directly comparable to the constraints 
        # We want minimize for a fixed grad, and need s such that 
        # sum(s) = 0 and 0 <= s[i] <= 1 for all i
        # A much robust approach:
    
    s = np.zeros(n_points)
    sum_positive = 0
    sum_negative = 0
    # Spread positive weights as much as we can to reduce the inner product
    # If we make one s idx = 0 and one = 1, and their contributions are + and -,
    #贪心法我们尽可能使用正梯度系数设备以0, 负梯度用1来平衡 sum = 0
    for i in range(n_points):
        if grad[i] < 0:
            s[i] = 1
            sum_negative -= grad[i]
        else:
            s[i] = 0
            sum_positive += grad[i]
      
    # Make sure sum(s) = 0. We need more constraints or just ensure this.
    # This problem is the key part. If sum is known to be zero, we cannot do much.
    # For the Frank-Wolfe sense: we imagine shifting along a direction s that is feasible.
    # In standard convex optimization context with constraints sum(h) = 1, sum(s) = 0:
    # minimize grad^T s subject to sum(s) = 0 and 0 <= s[i] <= 1
    
    # Better approach: What we really want is an s that satisfies:
    #  sum(s) = 0  
    # 0 <= s[i] <= 1
    # We want to minimize grad^T s.  
    # One exact method: due to no constraint other than bounds,
    # if sum(s) = 0, s_i ∈ [0,1], the best s that minimizes gradients 
    # is to make most of the s = 0 (for positive grads) and most s = 1 (for negative gradients). 
    # But we must maintain sum=0. 
    # If we have sum of pos_grad = P, neg_grad = N, we actually want:
    # Set s=1 for those in negative gradient side (to make them negative), s=0 for positive. 
    # This gives us just - sum_neg terms of the inner product.
    # Then we might need to modify.
    
    return s

def line_search(h, grad, s, dx, max_steps=10):
    """Perform backtracking line search to find optimal alpha."""
    # Initial objective
    old_c5 = compute_c5(h, dx)
    # delta = s - h
    alpha = 0.5
    for _ in range(max_steps):
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, 0.5 * len(h_new))  # We will normalize for integrals in other places
        new_c5 = compute_c5(h_new, dx)
        # Simple decrease criterion
        if new_c5 < old_c5:
            return alpha, h_new
        alpha /= 2
    return alpha, h

    max_iter = 5000
    improvement_iter = 0
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps and find max shift
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        c5 = overlap[k_max] * dx
        
        # If we are below threshold, we are done
        if c5 <= 0.3807:
            break
            
        if c5 < last_c5:
            improvement_iter = 0
            last_c5 = c5
        else:
            improvement_iter += 1
            if improvement_iter > 500:
                break
                
        # Compute gradient for the max overlap
        grad = compute_gradients(h, k_max)
        
        # Get direction s
        # Easier option: 
        # Since a lot of constraints are bounds [0,1] and sum=0,
        # Frank-Wolfe says: find s in the direction that minimizes linearized objective.
        # This means minimize grad^T s under the constraints sum(s) = 0 and 0<=s<=1.
        # This is a combinatorial problem and can be solved by construction:
        # We want to minimize sum over i of grad_i * s_i, sum(s) = 0, 0 <= s_i <= 1.
        # By turning s_i to extremes with decreasing influence to minimizing the sum.
        
        # Reduce the computation elsewhere to properly compute gradient for Frank-Wolfe
        s = np.zeros(n_points)
        # Slow but correct: the optimal s would be such that:
        # if grad[i] < 0, set s[i] to be as much as possible (1), 
        # just enough to make sum equal to 0.
        # Not straightforward but for simplicity:
        # Directly take a direction idea.
        # Let's try to make small adjustments based on ordering gradients:
        idxs_sorted = np.argsort(grad)
        pos_sum = 0.0
        neg_sum = 0.0
        for i in idxs_sorted:
            if grad[i] < 0:
                s[i] = 1
                neg_sum += grad[i]
            else:
                s[i] = 0
                pos_sum += grad[i]
        
        # Now normalize sum(s) to equal 0
        # Note we're not enforcing sum(s)=0 directly
        # The Frank-Wolfe method actually synthesizes better by just updating h  
        # To get s such that sum(s) = 0, make some 1 and some 0 such that the weights are matched.
        # Since we are trying to make inner product minimized under linear constraints:
        # We can take s such that it makes grad^T s minimized. These conditions mean
        # s[1] has more influence if it is more negative gradient --> set 1 if it is negative, 0 if positive?
        
        # Simplify and allow re-loop: compute a feasible s that achieves sum(s)=0, etc.
        # Simply define s by ignoring other constraints and make sum(s)=0 manually
        if np.sum(s) == 0:
            s = np.ones(n_points)
            s = s / np.sum(s)
            # Make it zero at the start for a more stable change
            # Alternative: update s uniformly to make sum zero
        # Roughly speaking:
        # Let want to sum(s) = 0.
        # Assign s=np.ones(n) for now as candidate and do balancing hand-crafted.
        # A better but slightly slower approach:
        if np.sum(s) != 0:
            # Balance boundaries
            s_diff = np.sum(s)  # This is positive if we used all 1 for negative gradient
            # Adjust to yield sum(s) = 0
            s = s - s_diff / n_points  # This changes s, sum should not be exactly zero
        
        # Alternative: just force s by making a quantity 1 and the rest 0 after adjusting
        # Given Valid s such that sum =0, 0 <= s_i <= 1:
        # most trick: Fast exact approach to enforce sum(s) = 0:
        # We choose s such that its components minimize the dot product with grad subject to constraints.
        # Simple way: choose s = (s_1, ..., s_n) where all s_i = 0 or 1, sum(s) = 0
        # that is, set a few to 1 and others to 0 to make sum = 0.
        weights = np.copy(grad)
        
        # Ensure weights are ordered in increasing order for best assignment
        idxs_weights = np.argsort(weights)
        s = np.zeros(n_points)
        
        # Calculate sum of values that decrease and sum of increasing
        # Try to assign 1s to the positions with most negative gradient to reduce total dot product.
        weights_sum = 0.0
        for idx in idxs_weights:
            if weights[idx] < 0:
                s[idx] = 1
            if weights[idx] > 0:
                s[idx] = 0
            weights_sum += weights[idx] * s[idx]
        
        # Add if sum(s) != 0, adjust
        if np.sum(s) != 0:
            # A compromise ensuring sum = 0:
            # We adjust by making some s[i] = 0 to a higher one
            sum_s = np.sum(s)
            if sum_s > 0:
                # Reduce sum_s elements to 0 to make sum = 0.  
                # Pick largest s indices.
                zero_idxs = np.argpartition(s, -int(sum_s))[-int(sum_s):]
                for i in zero_idxs:
                    s[i] = 0
                
        # Line search
        alpha, h_new = line_search(h, grad, s, dx)
        
        if alpha > 1e-6:
            h = h_new
            # Normalize at the end of iteration
        else:
            # No improvement happened, maybe we stop or issue warning
            pass
            
    # Make sure final state is feasible
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    final_c5 = compute_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
