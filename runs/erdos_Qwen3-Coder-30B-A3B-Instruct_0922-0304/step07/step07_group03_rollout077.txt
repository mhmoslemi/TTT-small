import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    
    # ES parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_iter = 5000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            candidate = np.array(initial_h_values)
            noise_scale = 0.05
            noise = np.random.normal(0, noise_scale, n_points)
            candidate += noise
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (target_sum / np.sum(candidate))
        else:
            candidate = np.random.uniform(0, 1, n_points)
            candidate = candidate * (target_sum / np.sum(candidate))
        population.append(candidate)
    
    # Evaluate initial population
    fitness_values = [compute_c5(individual, dx) for individual in population]
    
    # Best solution so far
    best_idx = np.argmin(fitness_values)
    best_individual = population[best_idx].copy()
    best_fitness = fitness_values[best_idx]
    
    # Main ES loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate current mutation rate
        t = iter_num / max_iter
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * t
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_values)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_values[i] for i in sorted_indices]
        
        # Keep elite
        next_generation = sorted_population[:elite_size]
        
        # Generate new individuals through mutation
        for _ in range(population_size - elite_size):
            # Select parent from elite
            parent_idx = np.random.randint(0, elite_size)
            parent = sorted_population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            
            next_generation.append(child)
        
        # Evaluate new population
        fitness_values = [compute_c5(individual, dx) for individual in next_generation]
        
        # Update best
        current_best_idx = np.argmin(fitness_values)
        if fitness_values[current_best_idx] < best_fitness:
            best_individual = next_generation[current_best_idx].copy()
            best_fitness = fitness_values[current_best_idx]
        
        # Early termination
        if best_fitness <= 0.3807:
            break
        
        population = next_generation
    
    # Final check
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
