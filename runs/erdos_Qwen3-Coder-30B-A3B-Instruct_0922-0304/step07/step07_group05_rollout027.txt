import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with both initial construction & randomness
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, target_sum)
    
    # Create diverse initial population
    for _ in range(population_size):
        if np.random.rand() < 0.2:  # 20% from initial
            individual = h + np.random.normal(0, 0.02, n_points)
        else:
            individual = np.random.rand(n_points)
            individual = individual * 0.5  # Center in [0,0.5]
        individual = project_to_feasible(individual, target_sum)
        population.append(individual)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # PSO parameters
    inertia_w = 0.8
    cognitive_c = 1.5
    social_c = 1.5
    max_iterations = 60
    
    # Initialize particles
    particles = population.copy()
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    personal_best = particles.copy()
    personal_best_scores = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia_w * velocities[i] + 
                           cognitive_c * r1 * (personal_best[i] - particles[i]) + 
                           social_c * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate
            current_score = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = current_score
                
                # Update global best
                if current_score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = current_score
                    stagnation_count = 0
                    
        # Update inertia weight
        inertia_w *= 0.98
        
        # Early termination check
        stagnation_count += 1
        if stagnation_count >= 20:
            break
        
        # Check if we've achieved target
        if global_best_score <= 0.3807:
            break
    
    # Final refinement with local optimization
    best_h = global_best.copy()
    best_c5 = global_best_score
    
    # Simple gradient ascent direction
    def gradient_step(h, step_size=0.01):
        # Compute gradient estimate in a simple way: finite difference
        base_c5 = compute_c5(h, dx)
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        # Step along negative gradient (to minimize)
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, target_sum)
        return h_new
    
    # Perform refinement
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = gradient_step(best_h)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        else:
            # Random perturbation if no improvement
            pert = np.random.normal(0, 0.001, n_points)
            h_new = best_h + pert
            h_new = project_to_feasible(h_new, target_sum)
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
    
    # Final validation
    best_h = project_to_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
