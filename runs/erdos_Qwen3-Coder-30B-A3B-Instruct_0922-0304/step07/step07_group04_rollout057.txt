import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 1000
    tournament_size = population_size // 10
    mutation_rate = 0.1
    max_no_improvement = 50
    
    # Initialize population
    population = []
    population.append(h.copy())
    
    for _ in range(population_size - 1):
        new_h = np.random.rand(n_points)
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        population.append(new_h)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fitness.append(evaluate_overlap(ind, dx))
    
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    no_improvement = 0
    generation = 0
    
    while generation < generations and time.time() - start_time < budget_s - 5:
        # Tournament selection
        new_population = []
        
        # Elitism: keep best solution
        new_population.append(best_h.copy())
        
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone and potentially mutate
            child = population[winner_idx].copy()
            
            # Crossover with another random individual
            other_idx = np.random.randint(0, population_size)
            if other_idx != winner_idx:
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * child + (1 - alpha) * population[other_idx]
                child = np.clip(child, 0, 1)
                
            # Mutation
            if np.random.random() < mutation_rate:
                # Gaussian mutation with adaptive standard deviation
                sigma = 0.1 * (1 - generation / generations)
                mask = np.random.random(n_points) < 0.3
                child += np.random.normal(0, sigma, n_points) * mask
                child = np.clip(child, 0, 1)
            
            # Normalize to sum constraint
            current_sum = np.sum(child)
            if current_sum > 0:
                child = child * (target_sum / current_sum)
            
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_overlap(ind, dx) for ind in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > max_no_improvement:
            mutation_rate = min(0.5, mutation_rate * 1.1)
            no_improvement = 0
            
        generation += 1
    
    # Local refinement of best candidate using SLSQP
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
