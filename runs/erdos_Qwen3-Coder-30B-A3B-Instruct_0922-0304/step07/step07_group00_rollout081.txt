import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Frank-Wolfe style coordinate-wise optimization
    max_iter = 100000
    iter_count = 0
    last_improvement = 0
    n_updates = max(1, n_points // 10)  # Update top 10% coordinates
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute overlap and identify max
        overlap = correlate(h, 1 - h, mode="full")
        current_c5 = np.max(overlap) * dx
        
        # Compute gradient-like components for each point
        gradients = np.zeros(n_points)
        # Direct derivative computation for overlap
        # gradient[i] = d/dh_i (sum_j h[j](1-h[j+i]))
        # This approximates Δ[i] = overlap[i] * (1 - 2 * h[i])
        for i in range(n_points):
            if i < len(overlap):
                gradients[i] = overlap[i] * (1 - 2 * h[i])
        
        # Select top updates
        update_indices = np.argsort(np.abs(gradients))[-n_updates:][::-1]
        
        # Track best candidate
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        # Try updates to selected coordinates
        for i in update_indices:
            h_temp = h.copy()
            # Compute step size based on approximate gradient
            if gradients[i] != 0:
                # Simple line search along gradient direction
                step_size = max(0, min(1, -gradients[i] * 0.1))
                h_temp[i] += step_size * (1 - 2 * h[i])
                h_temp = np.clip(h_temp, 0, 1)
                # Renormalize to maintain constraint
                h_temp = normalize_h(h_temp, target_sum)
                
                candidate_c5 = compute_c5(h_temp, dx)
                if candidate_c5 < best_candidate_c5:
                    best_candidate = h_temp.copy()
                    best_candidate_c5 = candidate_c5
        
        # Update if we found a better solution
        if best_candidate_c5 < best_c5:
            h = best_candidate
            best_c5 = best_candidate_c5
            last_improvement = iter_count
        else:
            # Sometimes we randomly perturb to escape local minima
            if np.random.rand() < 0.05:
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        
        iter_count += 1
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if iter_count - last_improvement > 1000:
            # If no improvement in 1000 iterations, try more exploration
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
    # Final computation
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
