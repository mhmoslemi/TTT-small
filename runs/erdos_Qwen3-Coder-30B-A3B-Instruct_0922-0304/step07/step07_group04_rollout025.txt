import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based optimization
    def compute_gradient(h, dx):
        # Compute gradient of overlap w.r.t. h
        # This is a simplified approximation
        n = len(h)
        grad = np.zeros(n)
        
        # For each position, compute how it affects the maximum overlap
        # This is a rough approximation
        for i in range(n):
            grad[i] = 0.0
            # Consider shifting by various lags k
            for k in range(-n+1, n):
                # Evaluate derivative of overlap term
                shift_k = (i - k) % n
                if 0 <= shift_k < n:
                    grad[i] += (1 - h[shift_k])
        return grad * dx
    
    def project_h(h, target_sum):
        # Project h onto simplex with sum = target_sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        return h
    
    # Adaptive gradient descent
    max_iter = 10000
    learning_rate = 1e-3
    momentum = 0.9
    reg_lambda = 1e-4
    v = np.zeros_like(h)
    prev_c5 = best_c5 + 1
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Add regularization term
        regularization = reg_lambda * (np.sum(h) - target_sum)
        grad += regularization
        
        # Update with momentum
        v = momentum * v - learning_rate * grad
        h = h + v
        
        # Project back to feasible set
        h = project_h(h, target_sum)
        
        # Evaluate current solution
        current_c5 = evaluate_overlap(h, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if current_c5 > prev_c5 + 1e-8:
            learning_rate *= 0.9
        else:
            learning_rate = min(1e-1, learning_rate * 1.01)
            
        # Adaptive regularization
        if np.sum(h) < target_sum - 1e-6 or np.sum(h) > target_sum + 1e-6:
            reg_lambda *= 1.1
        else:
            reg_lambda = max(1e-6, reg_lambda * 0.999)
        
        prev_c5 = current_c5
        
        # Stagnation check
        if stagnation_count > max_stagnation:
            if learning_rate > 1e-6:
                learning_rate *= 0.5
                stagnation_count = 0
            else:
                break
                
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
