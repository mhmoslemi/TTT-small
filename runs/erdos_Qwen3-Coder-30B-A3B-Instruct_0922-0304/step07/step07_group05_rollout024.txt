import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to the feasible region: 0 <= h <= 1 and sum(h) = target_sum."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a more structured initial solution
        h = np.zeros(n_points)
        # Place a triangular pulse
        mid = n_points // 2
        width = n_points // 4
        for i in range(max(0, mid - width//2), min(n_points, mid + width//2)):
            h[i] = 1 - abs(i - mid) / (width//2)
        h = project_to_feasible(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    patience = 0
    max_patience = 10
    
    # Iterative local optimization with quadratic approximations
    for iteration in range(60):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the current overlap at maximum
        overlap = correlate(best_h, 1 - best_h, mode="full")
        max_lag_idx = np.argmax(overlap)
        max_val = overlap[max_lag_idx]
        
        # Linear approximation of gradient at current point
        # Since we are dealing with the maximum correlation, 
        # this is a bit tricky for gradients. We will use a first-order
        # finite difference to approximate sensitivity.
        eps = 1e-4
        grad_approx = np.zeros_like(best_h)
        for i in range(n_points):
            # Perturb h[i]
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[i] = min(1.0, best_h[i] + eps)
            h_minus[i] = max(0.0, best_h[i] - eps)
            
            # Project to feasible region
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            
            # Compute overlap differences
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad_approx[i] = (overlap_plus[max_lag_idx] - overlap_minus[max_lag_idx]) / (2 * eps)
            
        # Set up QP: minimize 0.5 * d^T * H * d + grad^T * d
        # subject to: sum(d) = 0, 0 <= h+d <= 1
        # For simplicity, we'll make a simpler approach, use CVXPY for optimization
        
        # Formulate QP problem (simplified)
        # Minimize: trace(1/2 * d^T * H * d + grad^T * d) 
        # where H is the Hessian (not directly computed, use identity approximation)
        # subject to sum(d) = 0, 0 <= h + d <= 1
        
        d = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(d) + grad_approx.T @ d)
        constraints = [
            cp.sum(d) == 0,
            h + d >= 0,
            h + d <= 1
        ]
        
        # Try to use a more stable solver
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=SCS, verbose=False)
            if prob.status == 'optimal':
                step = d.value
            else:
                step = -grad_approx  # fallback
        except:
            step = -grad_approx  # fallback
            
        # Line search on step size
        alpha = 1.0
        min_alpha = 1e-6
        backtracking_factor = 0.5
        max_backtrack = 20
        
        for _ in range(max_backtrack):
            h_candidate = best_h + alpha * step
            h_candidate = project_to_feasible(h_candidate, target_sum)
            c5_candidate = compute_c5(h_candidate, dx)
            
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                patience = 0
                break
            else:
                alpha *= backtracking_factor
                if alpha < min_alpha:
                    break
                    
        # Check for stagnation
        patience += 1
        if patience >= max_patience:
            break
            
        # Early termination if we've found a very good solution
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
