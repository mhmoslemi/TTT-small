import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Distribute the difference evenly
    h = h + diff / len(h)
    # Clip to [0, 1] and re-scale
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
        
    # PGD parameters
    alpha = 0.1
    beta = 0.9
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stagnation_count = 0
    max_stagnation = 100
    
    iterations = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute all overlap values for different shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        # max_idx corresponds to shift k
        k = max_idx - (len(h) - 1)
        
        # Create gradient: (1-h) shifted by k
        grad = np.zeros_like(h)
        if k >= 0:
            grad[k:] = 1 - h[:-k]
        else:
            grad[:k] = 1 - h[-k:]
        
        # Update with momentum
        momentum = beta * momentum + grad
        h = h - alpha * momentum
        
        # Maintain constraints
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                alpha *= 0.95
                stagnation_count = 0
                if alpha < 1e-5:
                    break
        
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
