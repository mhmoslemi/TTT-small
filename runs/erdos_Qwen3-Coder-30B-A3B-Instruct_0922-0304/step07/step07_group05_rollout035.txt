import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles for PSO
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add some diversity to the initial solution
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation.copy())
        # Fill with random feasible particles
        for _ in range(population_size - len(particles)):
            h_random = np.random.rand(n_points)
            h_random = normalize_h(h_random, target_sum)
            particles.append(h_random.copy())
    else:
        # Create random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            particles.append(h.copy())
    
    # PSO parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iterations = 60
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Initialize global best
    best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[best_idx].copy()
    global_best_fitness = personal_best_fitness[best_idx]
    
    # Velocity for each particle
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # PSO main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (inertia_weight * velocities[i] + 
                            cognitive_coeff * r1 * (personal_best[i] - particles[i]) +
                            social_coeff * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary conditions
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Re-normalize to satisfy constraint
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
        
        # Adaptive inertia weight
        if iteration % 10 == 0:
            inertia_weight *= 0.98
            
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Local refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    best_h = global_best.copy()
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
