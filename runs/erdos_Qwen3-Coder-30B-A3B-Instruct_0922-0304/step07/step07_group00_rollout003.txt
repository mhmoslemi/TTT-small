import numpy as np
from scipy.signal import correlate
import time

def compute_c5_exact(h, dx):
    """Compute the exact C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, target_sum):
    """Compute gradient of the penalty approximation."""
    n = len(h)
    grad = np.zeros(n)
    
    # Simplified gradient approximation based on max overlap
    # For each position, we estimate contribution to maximum overlap
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    
    # Approximate gradient based on a single strong overlap term
    # This is an estimator, not exact derivative
    for i in range(n):
        # Weighted sum of influence from all positions
        g = 0.0
        for k in range(n):
            if 0 <= i+k < n:
                g += 2 * (overlap[max_idx] / dx) * (1 - 2 * h[i+k])
            if 0 <= i-k < n:
                g += 2 * (overlap[max_idx] / dx) * (1 - 2 * h[i-k])
        grad[i] = g
    
    # Normalize gradient
    if np.linalg.norm(grad) > 0:
        grad = grad / np.linalg.norm(grad)
    
    # Apply constraint gradient (sum = target_sum)
    grad_sum = np.ones(n) * (1.0 / target_sum)
    grad = grad - np.dot(grad, grad_sum) * grad_sum
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller dimensionality to focus optimization
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Adam-like optimizer parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and last_improvement < 1000:
        # Compute current C5
        current_c5 = compute_c5_exact(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient 
        grad = compute_gradient(h, dx, target_sum)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        # Update parameters
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        if iteration % 1000 == 0:
            learning_rate *= 0.999
    
    # Final computation of best C5
    final_c5 = compute_c5_exact(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
