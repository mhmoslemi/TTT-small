import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust values to meet the sum exactly
        diff = target_sum - current_sum
        h += diff / len(h)
        # Clamp and re-normalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h *= (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Initialize with a smooth step-like function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        h = project_to_constraints(h, target_sum)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.01
    max_step = 0.05
    tolerance = 1e-6
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            
            # Project perturbed values
            h_plus = project_to_constraints(h_plus, target_sum)
            h_minus = project_to_constraints(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        step = learning_rate * grad
        step = np.clip(step, -max_step, max_step)
        
        # Move
        h_new = h - step
        
        # Project to constraints
        h_new = project_to_constraints(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
