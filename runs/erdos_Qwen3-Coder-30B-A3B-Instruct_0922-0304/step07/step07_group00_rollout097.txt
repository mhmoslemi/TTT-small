def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    # actual gradient of h[i](1-h[i+k_max]) = 1-2*h[i] * delta(i+k_max)  
    # Actually, for overlap at k_max: 
    #        overlap_k_max = sum_i h[i] * (1 - h[i+k_max])
    # Gradient w.r.t h[i] = 1 - h[i+k_max] - h[i+k_max]  = 1 - 2*h[i+k_max] if i+k_max < n
    #                    = 0 if i+k_max >= n (i.e., k_max is large anyway) 
    # this is not correct - the gradient is partial derivative wrt h[i] 
    # Let's do it properly:
    # overlap_k = sum_{i} h[i] * (1 - h[i+k]) 
    # grad_{i'} = \frac{\partial overlap_k}{\partial h_{i'}} = 1 - h[i'+k] if i'+k < n else 0
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1.0 - h[i + k_max]
    return grad

def line_search(h, s, grad, dC5, dx, max_steps=10):
    """Perform backtracking line search for optimal step."""
    # compute <grad, s> = sum_i g_i * s_i
    # min should be where we cancel some effect best:
    # line search: try alpha in [0,1] such that
    # pick maximum step following projected descent objective.  
    # Better for phased-step by relaxation logic for real adjuster.
    # Note that return also s can be global 
    # Let's go for: min  f(h+alpha d) s.t. d is descent direction.
    
    # Step size goal allows to somewhat shift weight:
    # from h to h + step * (s-h) where step is retreat/advance.
    
    # Define approximate contract rate to goal.
    c1 = 0.1  # Armijo condition mix factor
    alpha = 1.0  # Initial step size
    
    h_old = h.copy()
    for _ in range(max_steps):
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, np.sum(h))
        c5_new = compute_c5(h_new, dx)
        # check sufficient decrease
        # Theoretical sufficient decrease condition:
        # F(x +alpha d) <= F(x) + c1 * alpha * <grad, d>
        # Crate an easier envelope based on actual directive values:
        # Check whether gain in reduction despite slight tolerance. 
        # c5_left = F(h)  
        # actual decent: 
        if c5_new < dC5:
            return alpha, h_new
        else:
            alpha *= 0.5  # Multiplier gradient descent algorithm
            
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) = 1
    
    # initialize

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # our framework
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    iteration = 0
    
    # Frank-Wolfe iteration
    last_improvement = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        c5_current = np.max(overlap_full) * dx
        
        # If already hit target, break
        if c5_current <= 0.3807:
            break
            
        # Linear approximation step
        grad = compute_gradients(h, k_max)
        
        # Solve linear sub-problem to get s
        s = solve_linear_program(grad, n_points)
        
        # compute derivative  
        # <s - h, dC5> = <s - h, grad> where dC5 approximated in this neighborhood. 
        # And derivative in change direction is approximately opposite. 
        # Let's treat as error-relative projection approach.
        dC5 = np.sum(grad * (s - h)) 
        # Line search
        alpha, h_new = line_search(h, s, grad, c5_current, dx, max_steps=10)
        
        h = h_new
        
        # Save best
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
            
        if iteration % 100 == 0:
            # periodic check
            if c5_now <= 0.3807:
                break
        
        iteration += 1
        # Early stop on no improvement
        if (iteration - last_improvement) > 500:
            if best_c5 <= 0.3807:
                break
                
    return (best_h.tolist(), best_c5, n_points)
