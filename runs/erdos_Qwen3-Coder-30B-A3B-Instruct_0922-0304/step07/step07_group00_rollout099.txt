import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def get_critical_shift(h, dx):
    """Find the shift that contributes most to the maximum overlap."""
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    k_max = max_idx - len(h) + 1
    return k_max

def line_search(h, s, k_max, dx, alpha_init=0.5, max_iter=10):
    """Perform backtracking line search to find optimal step size."""
    alpha = alpha_init
    current_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < current_c5:
            return alpha, h_new
        alpha *= 0.5
    
    return alpha, h

def frank_wolfe_step(h, k_max, dx):
    """Perform one Frank-Wolfe step for the given shift k_max."""
    n = len(h)
    
    # Create the gradient direction for k_max shift
    # The gradient of overlap_k_max at h is (1 - 2*h) shifted by k_max
    gradient = np.zeros(n)
    if k_max >= 0:
        gradient[k_max:] = 1 - 2 * h[:-k_max] if k_max < n else 1 - 2 * h
    else:
        gradient[:k_max] = 1 - 2 * h[-k_max:] if -k_max < n else 1 - 2 * h
    
    # Solve the linear program: min s^T * gradient, s >= 0, sum(s) = 0
    # This is equivalent to finding s such that s = argmin s^T * gradient
    # with sum(s) = 0, s >= 0, sum(h) = n/2
    # We can choose s to be at the minimum coordinate of gradient (or on the boundary)
    # We choose s to be the vector that minimizes direction change
    
    # The constrained solution in a simplex: revert to h, but with gradient direction
    # For this problem, we want to move in the direction that minimizes gradient term
    # So we take s as the element that gives most negative gradient term in the area of interest
    # Simplest: choose s = 0 except at one point where we reduce h in favor of increasing s
    # to still maintain integral constraint:
    
    # To keep the sum constraint, we can bound k based on how many points are new
    # Simple strategy: choose s such that the extremal point has negative dot product
    # We'll move not away from current h to e_i, but a proper simplex projection.
    
    # heuristically provide a simplex step:
    # If gradient is negative at some point, we prefer to increase the 
    # value at that location, but we need to respect constraints.
    
    # better approach:
    # Let's take s to be a vector with value 1 at the index that minimizes the gradient
    # Plus we still make sure that we also normalize properly
    
    # Let’s try to optimize towards the gradient direction directly, with constraints:
    # Make a minimal simple move: s = argmin_s ((1-2*h)[i+k_max] * s[i])
    # which is about minimizing under s conditions.
    
    # Rough heuristic: compute the direction of steepest decrease in the gradient,
    # while respecting the constraints.
    
    # Simple version:
    # Shift gradient to center - note that it's often a simple weighted average
    s = np.zeros(n)
    
    # Do the standard FW proper component-wise update
    # compute min gradient point
    idx_min = np.argmin(gradient)
    s[idx_min] = 1.0
    
    # now, we adjust so that sum s = 0 but keeping sum h = n/2
    # but the constraint for primal variables is sums: sum(h) = n/2
    # in the line search, we check that the new point satisfies these somehow
    
    return s

def solve_erdos_frank_wolfe(n_points, target_sum, dx, max_iter=5000, quiet=True):
    """Solve using Frank-Wolfe algorithm to minimize C5."""
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    k_max = get_critical_shift(h, dx) # initial shift

    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter:
        iteration += 1 

        # Determine current critical shift
        if iteration % 100 == 0:
            k_max = get_critical_shift(h, dx)

        # Compute the Frank-Wolfe step
        s = frank_wolfe_step(h, k_max, dx)

        # Perform line search
        alpha, h_new = line_search(h, s, k_max, dx)

        # Update h
        h = h_new

        # Update best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration

        if not quiet and iteration % 500 == 0:
            print(f"Iteration {iteration}, C5: {best_c5:.6f}")
            
        if best_c5 <= 0.3807:
            break
        
        if iteration - last_improvement > 500:
            # No progress for many iterations, adjust the critical shift
            k_max = get_critical_shift(h, dx)
            if iteration - last_improvement > 1000:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_h, best_c5 = solve_erdos_frank_wolfe(n_points, target_sum, dx, 
                                            max_iter=int(3000), quiet=True)
    
    # Validate and ensure final feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
