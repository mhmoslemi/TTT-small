import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Simple initialization
        h = np.ones(n_points) * 0.5
    
    # Fourier domain optimization
    def solve_fourier_opt(h, n_points, target_sum, dx):
        # Use CVXPY for convex optimization in frequency domain
        H = cp.Variable(n_points, complex=True)
        h_real = cp.Variable(n_points)
        
        # Define constraints in terms of the real part
        constraints = [
            cp.real(H[0]) == target_sum,
            h_real >= 0,
            h_real <= 1,
            cp.sum(h_real) == target_sum
        ]
        
        # Fourier relationship
        for i in range(n_points):
            if i < n_points // 2 + 1:
                constraints.append(H[i] == cp.sum(h_real * cp.exp(-2j * np.pi * np.arange(n_points) * i / n_points)))
            else:
                constraints.append(H[i] == cp.conj(H[n_points - i]))
        
        # Objective - minimize max overlap
        # We'll do this iteration by iteration, not the explicit Fourier domain approach
        # Let's use a simpler idea:
        # Use gradient-based method but in frequency domain space
        
        return h_real, constraints
    
    # Instead use a simpler relaxation and gradient descent approach
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent in function space with projected updates
    def project_h(h, target_sum):
        # Ensure h is in [0,1] and sum equals target_sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        return h
    
    # Simple local search with gradient descent
    max_iter = 1000
    lr = 0.001
    eps = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
            
        # Compute overlap and its local gradient with respect to h
        # This is a complex operation. For simplicity, we will sample gradient
        # directions and perform local search with a projection.
        
        # Alternate approach: Hartley transform approach or proximal techniques,
        # but we'll continue with a direct approach:
        
        # Perturb h slightly and compute new C5
        candidate_h = best_h.copy()
        # Add small random perturbations
        noise = np.random.normal(0, 0.001, n_points)
        perturbed_h = candidate_h + noise
        # Project
        perturbed_h = project_h(perturbed_h, target_sum)
        
        # Check how it does
        new_c5 = compute_c5(perturbed_h, dx)
        if new_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = new_c5
            
        # Occasionally resort to perturbation around good point
        if iteration % 20 == 0 and np.random.random() < 0.2:
            # Refine around best_h with small local steps
            ref_h = best_h + np.random.normal(0, 0.0005, n_points)
            ref_h = project_h(ref_h, target_sum)
            ref_c5 = compute_c5(ref_h, dx)
            if ref_c5 < best_c5:
                best_h = ref_h
                best_c5 = ref_c5
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_c5), int(n_points))
