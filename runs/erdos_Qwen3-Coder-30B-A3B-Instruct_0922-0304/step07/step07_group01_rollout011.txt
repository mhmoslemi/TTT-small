import numpy as np
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral using correlation."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint exactly."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to match target sum
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

    n_points = 70

    target_sum = n_points / 2

        h[:n_points//2] = 1
        # Add small noise for variety
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Simulated Annealing parameters
    T0 = 0.1 * compute_c5(h, dx)  # Initial temperature
    alpha = 0.99  # Cooling rate
    T = T0
    max_iter = 10000
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Generate new candidate solution by perturbing h
        h_new = h + np.random.uniform(-T, T, n_points)
        h_new = project_to_constraints(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
