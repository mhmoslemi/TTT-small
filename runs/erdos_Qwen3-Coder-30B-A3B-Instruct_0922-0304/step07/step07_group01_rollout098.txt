import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = project_to_constraints(h, target_sum)
    
    # Convex optimization in frequency domain
    # We'll use a simplified approach: minimize the maximum of the overlap
    # by solving a convex program over the Fourier coefficients
    
    # Transform to frequency domain
    H = np.fft.fft(h)
    n_freqs = len(H)
    
    # Variables for optimization
    H_real = Variable(n_freqs)
    H_imag = Variable(n_freqs)
    
    # Constraints
    constraints = []
    
    # Real/imaginary constraints
    for i in range(n_freqs):
        constraints.append(H_real[i] == H[i].real)
        constraints.append(H_imag[i] == H[i].imag)
    
    # Symmetry constraints for real signal
    # For real signal h, H[-k] = H[k]*, and H[0] is real
    for i in range(1, n_freqs//2):
        constraints.append(H_real[i] == H_real[n_freqs-i])
        constraints.append(H_imag[i] == -H_imag[n_freqs-i])
    
    if n_freqs % 2 == 0:
        constraints.append(H_imag[n_freqs//2] == 0)  # Nyquist freq
    
    # Create optimization problem
    # Minimize Z (upper bound on overlap)
    Z = Variable()
    
    # The overlap is the correlation h * (1-h)
    # In freq domain, this is: H * conj(1-H)
    # So overlap frequency domain is: H * (conj(H) - 1)
    # But it's easier to compute directly in time domain
    # Let's optimize to minimize C5 directly by sampling
    
    # Custom approach: optimize with a simple gradient-descent-like method but 
    # ensuring projected feasible solutions
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based local refinement
    max_iter = 1000
    learning_rate = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient direction approximately via finite differences
        current_c5 = compute_c5(best_h, dx)
        
        # Perturb each coordinate individually to estimate gradient direction
        grad_estimate = np.zeros_like(best_h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_to_constraints(h_plus, target_sum)
            h_minus = project_to_constraints(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            grad_estimate[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Move in negative gradient direction
        new_h = best_h - learning_rate * grad_estimate
        
        # Project back to constraints
        new_h = project_to_constraints(new_h, target_sum)
        new_c5 = compute_c5(new_h, dx)
        
        # Accept if improvement
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            # Reduce learning rate for next iteration
            learning_rate *= 0.995
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
