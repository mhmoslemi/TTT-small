import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    h = np.clip(h + diff / len(h), 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize relaxed version
    h_relaxed = h.copy() + np.random.normal(0, 0.05, n_points)
    
    # Parameters
    alpha = 0.1
    beta = 0.9
    lambda_reg = 1e-4
    max_iter = 3000
    
    # Momentum and velocity for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current overlap for each shift
        current_c5 = compute_c5(h, dx)
        
        # Compute gradient w.r.t. relaxed version
        # This should consider the full correlation calculation (simplified)
        # For numerical gradient estimation:
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus_clipped = np.clip(h_plus, 0, 1)
            h_plus_clipped = normalize_h(h_plus_clipped, target_sum)
            c5_plus = compute_c5(h_plus_clipped, dx)
            
            h_minus = h_relaxed.copy()
            h_minus[i] -= eps
            h_minus_clipped = np.clip(h_minus, 0, 1)
            h_minus_clipped = normalize_h(h_minus_clipped, target_sum)
            c5_minus = compute_c5(h_minus_clipped, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with Adam
        m = beta * m + (1 - beta) * grad
        v = beta * v + (1 - beta) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta**(iter_count + 1))
        v_hat = v / (1 - beta**(iter_count + 1))
        
        # Update h_relaxed
        h_relaxed = h_relaxed - alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Regularization term
        h_relaxed = h_relaxed - lambda_reg * (h_relaxed - h)
        
        # Project to feasible region and enforce sum
        h = np.clip(h_relaxed, 0, 1)
        h = project_to_constraint(h, target_sum)
        
        # Update best if improved
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive learning rate and regularization
        if iter_count % 100 == 0 and iter_count > 0:
            alpha *= 0.9
            lambda_reg *= 1.2
        elif iter_count % 1000 == 0:
            lambda_reg *= 0.8
        
        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
