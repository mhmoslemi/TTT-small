import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    pop_size = 100
    max_generations = 1000
    tournament_size = max(2, pop_size // 10)
    mutation_rate = 0.1
    blending_min, blending_max = 0.2, 0.8
    
    # Create initial population
    population = [h.copy()]
    for i in range(pop_size - 1):
        new_h = np.random.uniform(0, 1, n_points)
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        population.append(new_h)
    
    # Evaluate initial population
    fitness = [evaluate_overlap(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # GA loop
    no_improvement_count = 0
    previous_best = best_c5
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            best_in_tournament = min(tournament, key=lambda i: fitness[i])
            selected.append(population[best_in_tournament])
        
        # Crossover and mutation
        new_population = [population[best_idx].copy()]
        for i in range(0, pop_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i + 1] if i + 1 < len(selected) else selected[0]
            
            # Blending crossover
            alpha = np.random.uniform(blending_min, blending_max)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Apply mutation to both children
            mutation_strength = mutation_rate * (0.95 ** generation)
            
            # Mutation
            mask = np.random.random(n_points) < mutation_rate
            if np.any(mask):
                child1 += np.random.normal(0, mutation_strength, n_points) * mask
                child1 = np.clip(child1, 0, 1)
                current_sum = np.sum(child1)
                if current_sum > 0:
                    child1 = child1 * (target_sum / current_sum)
            
            mask = np.random.random(n_points) < mutation_rate
            if np.any(mask):
                child2 += np.random.normal(0, mutation_strength, n_points) * mask
                child2 = np.clip(child2, 0, 1)
                current_sum = np.sum(child2)
                if current_sum > 0:
                    child2 = child2 * (target_sum / current_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:pop_size]
        
        # Evaluate fitness
        fitness = [evaluate_overlap(ind, dx) for ind in new_population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = new_population[best_idx].copy()
            best_c5 = fitness[best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check for stagnation
        if no_improvement_count > 100:
            # Reset mutation rate
            mutation_rate = 0.2
            no_improvement_count = 0
            
        population = new_population
    
    # Local refinement on best solution
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
