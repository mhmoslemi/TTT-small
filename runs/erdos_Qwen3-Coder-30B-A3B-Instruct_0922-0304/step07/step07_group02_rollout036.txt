import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Adjust values to meet the sum constraint
    h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def differential_evolution_step(population, fitness, target_sum, dx, 
                               mutation_factor=0.8, crossover_prob=0.7):
    """Perform one step of differential evolution."""
    n_points = len(population[0])
    new_population = []
    
    for i in range(len(population)):
        # Select three different individuals
        indices = np.random.choice(len(population), 3, replace=False)
        a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
        
        # Mutation
        mutant = a + mutation_factor * (b - c)
        mutant = np.clip(mutant, 0, 1)
        
        # Crossover
        trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
        
        # Project to constraint
        trial = project_to_constraint(trial, target_sum)
        
        # Evaluate
        trial_fitness = compute_c5(trial, dx)
        
        # Selection
        if trial_fitness < fitness[i]:
            new_population.append(trial)
        else:
            new_population.append(population[i])
    
    return new_population

        for _ in range(50):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
            fitness.append(compute_c5(h, dx))
    
    # Add more diverse initial solutions
    for _ in range(49):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
        fitness.append(compute_c5(h, dx))
    
    # Differential evolution parameters
    max_iter = 3000

    crossover_prob = 0.7
    
    # Track best solution
