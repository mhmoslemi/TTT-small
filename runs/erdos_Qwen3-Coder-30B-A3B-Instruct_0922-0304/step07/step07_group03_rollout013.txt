import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = project_to_constraints(h, target_sum)
    
    # LR schedule
    def learning_rate(t, t_max=2000):
        return 0.1 / (1 + t / (t_max/10))
    
    # Gradient descent loop
    max_iter = 2000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        
        # Compute surrogate gradient (approximation)
        # Use the index of maximum overlap to guide gradient
        max_idx = np.argmax(overlap)
        # Construct a small gradient based on correlation at max point
        grad = np.zeros_like(h)
        
        # Compute correlation values efficiently
        for i in range(len(h)):
            # Compute contribution from index i
            contrib = 0
            for k in range(len(h)):
                j = k - i
                if 0 <= j < len(h):
                    contrib += (1 - h[j]) * h[k]
            grad[i] = contrib * 2 * dx  # Approximate gradient
        
        # Apply gradient
        lr = learning_rate(iter_num)
        h = h - lr * grad
        h = project_to_constraints(h, target_sum)
        
        # Early termination
        if compute_c5(h, dx) <= 0.3807:
            break
    
    # Final check and return
    final_c5 = compute_c5(h, dx)
