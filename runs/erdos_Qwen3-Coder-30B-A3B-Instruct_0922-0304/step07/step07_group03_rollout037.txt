import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values using a sawtooth pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Convex optimization with projected gradient descent
    max_iter = 20000
    learning_rate = 0.1
    decay_factor = 1000.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlation
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx]
        
        # Compute gradient of surrogate objective
        # We minimize a surrogate that approximates the max overlap squared
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient at position i
            for k in range(n_points):
                if 0 <= i - k < n_points:
                    grad[i] += 2 * (max_overlap) * dx * (1 - h[i - k])
        
        # Gradient descent step
        h = h - learning_rate * grad / (1 + iter_num / decay_factor)
        
        # Project to feasible region
        h = project_to_constraints(h, target_sum)
        
        # Compute new C5 value
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Multi-resolution refinement
    for resolution in [140, 280]:
        if time.time() - start_time > budget_s - 1:
            break
            
        if resolution > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, resolution), 
                              np.linspace(0, 2, n_points), best_h)
            dx_new = 2.0 / resolution
            target_sum_new = resolution / 2
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, target_sum_new)
            
            # Refine with more iterations at higher resolution
            for _ in range(500):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                overlap = correlate(h_new, 1 - h_new, mode="full")
                max_overlap_idx = np.argmax(overlap)
                max_overlap = overlap[max_overlap_idx]
                
                grad = np.zeros_like(h_new)
                for i in range(resolution):
                    for k in range(resolution):
                        if 0 <= i - k < resolution:
                            grad[i] += 2 * (max_overlap) * dx_new * (1 - h_new[i - k])
                
                h_new = h_new - 0.05 * grad / (1 + _ / 100)
                h_new = project_to_constraints(h_new, target_sum_new)
                
                current_c5 = compute_c5(h_new, dx_new)
                if current_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = current_c5
                    if best_c5 <= 0.3807:
                        break
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
