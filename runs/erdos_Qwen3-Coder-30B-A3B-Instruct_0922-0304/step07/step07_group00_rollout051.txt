import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the objective function."""
    # We approximate the max with a differentiable sum of squares
    # Compute all convolution values
    conv = correlate(h, 1 - h, mode="full")
    # Compute gradient for each position
    grad = np.zeros(n_points)
    # The gradient at position i
    # involves terms from all regions where i contributes (i and i+k are valid)
    for i in range(n_points):
        for k in range(-n_points + 1, n_points):
            if 0 <= i + k < n_points:
                grad[i] += 2 * conv[n_points - 1 + k] * (1 - 2 * h[i + k]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():

    else:

    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Ensure feasible initial solution
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Optimizer parameters
    learning_rate = 0.01
    decay_rate = 0.999
    tolerance = 1e-8
    max_iter = 10000
    last_improvement = 0
    
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # For each iteration
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute gradient using finite differences for robustness
        eps = 1e-6
        grad_approx = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            h_minus = normalize_h(h_minus, target_sum)
            f_plus = compute_c5(h_plus, dx)
            f_minus = compute_c5(h_minus, dx)
            grad_approx[j] = (f_plus - f_minus) / (2 * eps)
        
        # Update step
        h -= learning_rate * grad_approx
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Check for improvement
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Update learning rate
        if i % 1000 == 0:
            learning_rate *= decay_rate
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if last_improvement > 1000:
            break
    
    # Return the best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
