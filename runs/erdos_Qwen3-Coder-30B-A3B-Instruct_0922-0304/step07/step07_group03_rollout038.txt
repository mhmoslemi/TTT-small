import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust to meet target sum
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points):
            h[i] = 0.5 * (1 + np.cos(np.pi * i / n_points))
        h = project_to_constraints(h, target_sum)
    
    # Convex relaxation settings
    max_iter = 20000
    alpha0 = 0.1
    tau = 1000
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute gradient of surrogate objective
        c = correlate(h, 1 - h, mode="full")
        # This is an approximation to gradient computation
        # We'll use a simple update rule based on local correlations
        grad = np.zeros_like(h)
        # Compute approximation of gradient from neighbors
        for i in range(len(h)):
            # Approximate gradient using differences with neighbors  
            left = h[max(0, i-1)]
            right = h[min(len(h)-1, i+1)]
            grad[i] = (right - left) * (2/n_points)
        
        # Update step with adaptive learning rate
        alpha = alpha0 / (1 + iter_num / tau)
        h = h - alpha * grad
        
        # Project back to constraints
        h = project_to_constraints(h, target_sum)
        
        # Compute current value
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Multi-resolution refinement
    for n_points_ref in [140, 280, 560]:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Interpolate to higher resolution
        h_high = np.interp(np.linspace(0, 2, n_points_ref), 
                          np.linspace(0, 2, len(best_h)), best_h)
        
        # Project to constraints at higher resolution
        h_high = project_to_constraints(h_high, n_points_ref / 2)
