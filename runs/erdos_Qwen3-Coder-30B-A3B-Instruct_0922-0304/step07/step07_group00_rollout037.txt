import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_basis_functions(n_points, n_basis=16):
    """Create basis functions with varying pulse widths and positions."""
    basis = []
    for i in range(n_basis):
        # Create a step function basis
        start = i * 2.0 / n_basis
        end = (i + 1) * 2.0 / n_basis
        h = np.zeros(n_points)
        for j in range(n_points):
            x = j * 2.0 / n_points
            if start <= x < end:
                h[j] = 1.0
        # Normalize to integrate to 1/(n_points/2) = 2/n_points
        h = h * (2.0 / n_points) / np.sum(h) if np.sum(h) > 0 else h
        basis.append(h)
    return basis

def compute_overlap_matrix(basis, dx):
    """Compute the overlap matrix A_ij^k."""
    n_basis = len(basis)
    n_points = len(basis[0])
    A = np.zeros((n_basis, n_basis, 2 * n_points - 1))
    for i in range(n_basis):
        for j in range(n_basis):
            # Compute overlap for all shifts
            a = correlate(basis[i], basis[j], mode="full")
            A[i, j] = a
    return A

def reconstruct_h(lambdas, basis):
    """Reconstruct h from basis coefficients."""
    h = np.zeros(len(basis[0]))
    for i, lam in enumerate(lambdas):
        h += lam * basis[i]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create basis functions
    basis = create_basis_functions(n_points, 16)
    A = compute_overlap_matrix(basis, dx)
    
    # Load initial h if available
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    
    # If initial_h, decomp with basis functions
    if initial_h is not None:
        # Solve least squares to initialize lambdas
        B = np.array(basis).T
        lambdas = np.linalg.lstsq(B, initial_h, rcond=None)[0]
        lambdas = np.maximum(lambdas, 0)
        lambdas = lambdas / np.sum(lambdas)
    else:
        # Initialize with uniform distribution
        lambdas = np.ones(len(basis)) / len(basis)
    
    best_h = None
    best_c5 = float('inf')
    
    # Main optimization loop
    max_iterations = 2000
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Reconstruct h
        h = reconstruct_h(lambdas, basis)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Compute C5
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Check for early stopping
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient for updating lambdas using approximation of C5
        n_basis = len(basis)
        K = 2 * n_points - 1
        
        # Compute the gradient of C5 w.r.t. lambdas (approximated)
        dCdlambdas = np.zeros(n_basis)
        for i in range(n_basis):
            for j in range(n_basis):
                # Compute gradient contributions for overlap terms
                # This handles part of the second term in ∫ h(x)(1-h(x+k))dx
                grad_sum = 0
                for k_idx in range(K):
                    # A_{ij}^k terms
                    A_ij_k = A[i, j, k_idx]
                    g_term = A_ij_k - A_ij_k * A[j, j, k_idx]
                    grad_sum += g_term
                
                if i == j:
                    grad_sum = 0  # Simplification
                else:
                    dCdlambdas[i] += -2 * lambdas[j] * grad_sum
                
        # Normalize gradient to prevent explosion
        grad_norm = np.linalg.norm(dCdlambdas)
        if grad_norm > 0:
            dCdlambdas = dCdlambdas / grad_norm
        
        # Update lambdas using gradient descent
        step_size = 0.01 * np.exp(-iteration / 100)
        lambdas = lambdas - step_size * dCdlambdas
        
        # Project onto simplex
        lambdas = np.maximum(lambdas, 0)
        lambdas = lambdas / np.sum(lambdas)
        
    # Ensure final solution is valid
    if best_h is None:
        h = reconstruct_h(lambdas, basis)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        best_h = h
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
