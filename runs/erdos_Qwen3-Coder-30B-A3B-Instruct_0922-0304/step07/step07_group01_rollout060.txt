import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint using projection
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to match target sum
        h = h * (target_sum / current_sum)
        # Clamp again to stay within bounds
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Alternative initialization: create smoother step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        h = project_to_constraints(h, target_sum)
    
    # Simulated Annealing parameters
    T0 = 0.1 * compute_c5(h, dx)
    T = T0
    alpha = 0.99
    max_iter = 10000
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    for i in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create candidate by perturbing current solution
        noise = np.random.uniform(-T, T, n_points)
        candidate_h = h + noise
        candidate_h = project_to_constraints(candidate_h, target_sum)
        candidate_c5 = compute_c5(candidate_h, dx)
        
        # Acceptance probability
        if candidate_c5 < current_c5:
            # Always accept if better
            h = candidate_h
            current_c5 = candidate_c5
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            prob = np.exp(-delta / T)
            if np.random.random() < prob:
                h = candidate_h
                current_c5 = candidate_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Cool temperature
        T *= alpha
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
