import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Frank-Wolfe algorithm
    max_iter = 5000
    line_search_steps = 10
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Linear approximation of gradient
        # The gradient with respect to h for the k-th overlap is (1 - 2*h) shifted by k
        # We shift the gradient back so that it's in the same space as h
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + max_k < len(h):
                grad[i] = 1 - 2 * h[i + max_k]
            else:
                # When we go beyond, add to the beginning (wraparound effect 
                # in the convolution, but let's use direct indexing)
                continue
                
        # Solve linear program: min s^T * grad s.t. sum(s) = 0, 0 <= s <= 1
        # For this, we make a step in the direction that decreases the gradient most
        # We want to go in the direction that locally minimizes the linearized objective
        
        # Gradient is linearized, so we want step direction s that minimizes s^T * grad
        # Subject to sum(s) = 0 and 0 <= s <= 1
        
        # Simple approach: identify the most negative components and set them
        # to zero, then do feasible scaling.
        s = np.zeros_like(h)
        
        # Simple heuristic to select s
        neg_indices = np.where(grad < 0)[0]
        if len(neg_indices) > 0:
            # Choose the most negative
            most_neg = np.argmin(grad)
            s[most_neg] = 1.0  # Step towards the negative gradient
            
            # Ensure sum(s) = 0
            # Here we ensure the linear constraint
            # But to mitigate bias, we actually compute a better step direction
            # Let's take a weighted move towards minimizing the gradient
            s += -grad  # Move anti-gradient
            s = np.clip(s, 0, 1)  # Bound
        
        # Now enforce that sum(s) = 0 (we are looking for a feasible step direction)
        # For linear programming, we could solve this via projection.
        # Since it's simple, we'll do:
        # s = s - mean(s) to make sum(s) = 0, then renormalize to fixed size.
        
        s = s - np.mean(s)  # This ensures sum(s) = 0 now
        s = np.clip(s, -1, 1)
        # Normalize at least to make direction epsilon-size
        norm_s = np.linalg.norm(s)
        if norm_s > 0:
            s = s / norm_s * 0.01  # Make it small step
        else:
            s = np.zeros(n_points)
            
        # Line search to find best alpha
        best_alpha = 0.0
        best_new_h = h.copy()
        best_new_c5 = max_overlap
        
        # Try decreasing alphas
        for alpha in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            new_h = h + alpha * s
            new_h = normalize_h(new_h, target_sum)
            new_h = np.clip(new_h, 0, 1)
            new_c5 = compute_c5(new_h, dx)
            
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
                best_new_h = new_h.copy()
                
            if new_c5 < 0.3807:
                break
                
        if best_alpha > 0:
            h = best_new_h
            if abs(best_new_c5 - last_c5) < 1e-8:
                patience_counter += 1
            else:
                patience_counter = 0
            last_c5 = best_new_c5
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            # May not be converging, switch to something else:
            break
            
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
