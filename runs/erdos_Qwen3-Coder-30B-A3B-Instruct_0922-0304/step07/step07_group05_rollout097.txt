import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iterative refinement using quadratic approximation and QP
    max_iterations = 60
    patience = 10
    patience_counter = 0
    prev_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation and Hessian (simplified)
        # For quadratic model, we'll directly compute a descent direction
        current_c5 = compute_c5(h, dx)
        gradient = np.zeros(n_points)
        
        # Simple finite difference method to estimate gradient
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            gradient[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Form and solve QP: min 0.5 * d^T * H * d + g^T * d
        # subject to A * d = 0, 0 <= d <= 1
        # In our case, we simplify and use gradient descent-like approach
        # But still use quadratic programming to refine
        d = -gradient  # descent direction
        
        # Make sure we enforce that sum stays the same 
        # project to zero-sum space
        d_sum = np.sum(d)
        d -= d_sum / n_points  # ensure sum(d) = 0
        
        # Project d to feasible region
        d = np.clip(d, -h, 1 - h)
        
        # Line search
        alpha_max = 1.0
        for i in range(20):  # max steps
            test_h = h + alpha_max * d
            test_h = project_to_feasible(test_h, target_sum)
            test_c5 = compute_c5(test_h, dx)
            
            if test_c5 < current_c5:
                alpha = alpha_max
                break
            alpha_max /= 2
        else:
            alpha = 0.0
            
        if alpha > 0:
            h = h + alpha * d
            h = project_to_feasible(h, target_sum)
            current_c5 = compute_c5(h, dx)
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                patience_counter = 0
            else:
                patience_counter += 1
                
            if patience_counter >= patience:
                break
        else:
            patience_counter += 1
            if patience_counter >= patience:
                break
    
    # Final computation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
