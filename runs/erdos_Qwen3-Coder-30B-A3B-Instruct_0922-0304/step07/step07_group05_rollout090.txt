import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create smooth step function
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Remez-like optimization
    max_iterations = 20
    alpha = 0.1
    decay = 0.95
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        
        # Synthesize a phase-inverted update
        # Use a sinusoidal with frequency related to the maximum overlap lag
        k = max_lag / n_points
        update = alpha * np.sin(2 * np.pi * k * np.arange(n_points))
        
        # Apply update
        h_new = h + update
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Accept new solution if improvement
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        alpha *= decay
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
