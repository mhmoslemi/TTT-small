import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    tournament_size = max(5, population_size // 20)  # Dynamic tournament size
    max_generations = 1000
    mutation_rate = 0.1
    elite_size = 5
    
    # Create initial population
    population = []
    population_fitness = []
    
    # Add initial solution
    population.append(h.copy())
    population_fitness.append(best_c5)
    
    # Add random solutions
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand_sum = np.sum(h_rand)
        if h_rand_sum > 0:
            h_rand = h_rand * (target_sum / h_rand_sum)
        population.append(h_rand)
        population_fitness.append(evaluate_overlap(h_rand, dx))
    
    # GA main loop
    last_improvement = 0
    generation = 0
    
    while (time.time() - start_time < budget_s - 5) and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best solution
        if population_fitness[0] < best_c5:
            best_c5 = population_fitness[0]
            best_h = population[0].copy()
            last_improvement = generation
            
        # Early stopping if no improvement in last 200 generations
        if generation - last_improvement > 200:
            break
        
        # Create new population
        new_population = [population[i].copy() for i in range(elite_size)]
        new_population_fitness = [population_fitness[i] for i in range(elite_size)]
        
        # Elitism and selection
        while len(new_population) < population_size:
            # Tournament selection
            def tournament_select():
                tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
                tournament_fitness = [population_fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                return population[winner_idx]
            
            # Selection and crossover
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Blend crossover
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                mask = np.random.binomial(1, 0.5, n_points)
                noise = np.random.normal(0, 0.05, n_points) * mask
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize sum
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            new_population.append(child)
            new_population_fitness.append(evaluate_overlap(child, dx))
        
        population = new_population
        population_fitness = new_population_fitness
        
        # Adaptive mutation rate
        if generation % 10 == 0:
            mutation_rate = max(0.01, mutation_rate * 0.98)
        
        generation += 1
    
    # Final local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation  
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
