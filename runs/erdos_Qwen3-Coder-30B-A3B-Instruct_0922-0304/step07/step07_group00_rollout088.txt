import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # If initial_h_values is available, use it as a starting point
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    if initial_h is not None:
        h = np.array(initial_h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simulated annealing style optimization
    temperature = 1.0
    decay_rate = 0.995
    min_temperature = 1e-6
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Create a new candidate by perturbing current solution
        h_new = h.copy()
        
        # Perturb randomly
        perturbation_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in perturbation_indices:
            delta = np.random.normal(0, temperature * 0.1)
            h_new[i] += delta
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject the move
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
                h = h_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Decay temperature
        temperature *= decay_rate
        iteration += 1
        
        # Early termination check
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
