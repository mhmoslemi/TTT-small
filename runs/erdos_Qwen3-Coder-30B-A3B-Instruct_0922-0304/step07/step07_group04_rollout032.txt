import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Custom genetic algorithm for optimization
    population_size = 100
    tournament_size = max(5, population_size // 10)
    max_generations = 1000
    mutation_rate = 0.1
    max_no_improvement = 50
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        candidate = np.random.rand(n_points)
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if candidate_sum > 0:
            candidate = candidate * (target_sum / candidate_sum)
        population.append(candidate)
    
    # Evaluate initial fitness
    fitness = [evaluate_overlap(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # GA loop
    no_improvement_count = 0
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            best_in_tournament = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[best_in_tournament].copy())
        
        # Crossover and mutation
        new_population = [population[best_idx].copy()]  # Elitism
        
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(population_size, 2, replace=False)
            p1, p2 = selected[parent1], selected[parent2]
            
            # Blend crossover
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0, 1)
            
            # Normalize sum
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.3
                noise = np.random.normal(0, 0.05, n_points)
                child += noise * mask
                child = np.clip(child, 0, 1)
                
                # Re-normalize
                child_sum = np.sum(child)
                if child_sum > 0:
                    child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_overlap(ind, dx) for ind in population]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive mutation rate
        if no_improvement_count > 10:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        if no_improvement_count > max_no_improvement:
            break
    
    # Local refinement using gradient descent
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
