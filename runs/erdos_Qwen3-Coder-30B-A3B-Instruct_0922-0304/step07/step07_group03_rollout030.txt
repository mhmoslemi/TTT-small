import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, dx, target_sum, maxiter=1000):
    """Use L-BFGS-B to locally optimize h."""
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    # Convert to list for scipy
    x0 = h.copy()
    
    # Optimization bounds and constraints
    bounds = [(0, 1) for _ in range(len(x0))]
    constraints = {'type': 'eq', 'fun': constraint}
    
    # Call optimizer
    result = minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    # Revert back to h values
    if result.success:
        h_opt = result.x.copy()
    else:
        h_opt = x0.copy()
    
    # Ensure constraints again
    h_opt = project_to_constraints(h_opt, target_sum)
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # Evolution Strategies parameters
    population_size = 20
    generations = 2000
    sigma_init = 0.1
    sigma_final = 0.01
    elite_ratio = 0.5
    
    # Initialize population
    population = []
    for i in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual += np.random.normal(0, sigma_init, n_points)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = normalize_h(individual, target_sum)
        population.append(individual.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    # ES loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate fitness
        fitness = []
        for individual in population:
            fit = compute_c5(individual, dx)
            fitness.append(fit)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # If we've found a good solution, try local optimization
        if fitness[0] < 0.381 and time.time() - start_time < budget_s - 2:
            local_h = local_optimize(population[0], dx, target_sum)
            local_c5 = compute_c5(local_h, dx)
            if local_c5 < best_c5:
                best_c5 = local_c5
                best_h = local_h.copy()
            # Add refined solution to population if it's better than worst
            if local_c5 < fitness[-1]:
                population[-1] = local_h.copy()
        
        # Selection
        elite_size = int(population_size * elite_ratio)
        elite = population[:elite_size]
        
        # Mutation and create offspring
        new_population = elite.copy()
        sigma = sigma_init - (sigma_init - sigma_final) * gen / generations
        
        # Generate new individuals
        while len(new_population) < population_size:
            parent = elite[np.random.randint(0, elite_size)]
            child = parent + np.random.normal(0, sigma, n_points)
            child = project_to_constraints(child, target_sum)
            new_population.append(child)
            
        population = new_population
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
