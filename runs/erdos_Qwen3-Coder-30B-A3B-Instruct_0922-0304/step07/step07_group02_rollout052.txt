def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    # Gradient-based optimization with bounds
    # This is a heuristic because the problem is non-convex
    step_size = 0.01
