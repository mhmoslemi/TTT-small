import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Create initial population
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.01, n_points)
        individual = project_feasible(individual, target_sum)
        population.append(individual)
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = [p.copy() for p in population]
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(population_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO loop
    max_iterations = 60
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        for i in range(population_size):
            r1, r2 = np.random.rand(2)
            # Update velocity
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            
            if stagnation_count >= max_stagnation:
                if time.time() - start_time > budget_s - 1:
                    break
                # Add random perturbation to particles
                for j in range(population_size):
                    if np.random.rand() < 0.1:
                        particles[j] += np.random.normal(0, 0.03, n_points)
                        particles[j] = project_feasible(particles[j], target_sum)
            
        if global_best_fitness <= 0.3807:
            break
            
        if stagnation_count >= max_stagnation:
            # Another randomization step
            for j in range(population_size):
                if np.random.rand() < 0.2:
                    noise = np.random.normal(0, 0.02, n_points)
                    particles[j] = project_feasible(particles[j] + noise, target_sum)

    # Final refinement with local optimization
    refinement_steps = 20
    for _ in range(refinement_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_global = global_best.copy()
        # Simple gradient-like adjustment
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = global_best.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = global_best.copy()
            h_minus[i] -= eps
            h_minus = project_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update using gradient
        step_size = 0.001
        new_global -= step_size * grad
        new_global = project_feasible(new_global, target_sum)
        
        if compute_c5(new_global, dx) < compute_c5(global_best, dx):
            global_best = new_global.copy()
            
    final_c5 = compute_c5(global_best, dx)
    return (global_best.tolist(), final_c5, n_points)
