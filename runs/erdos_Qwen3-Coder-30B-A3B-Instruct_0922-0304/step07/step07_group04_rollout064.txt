import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-6
    alpha = 0.999
    max_iter = int(budget_s * 1000)
    learning_rate = 0.01
    penalty_weight1 = 100.0
    penalty_weight2 = 100.0
    
    def compute_energy(h_vec, c5_val):
        # Constraint penalties
        bound_penalty = np.sum(np.clip(h_vec, 0, 1) - h_vec)**2
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return c5_val + penalty_weight1 * bound_penalty + penalty_weight2 * sum_penalty
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Make a perturbation
        h_new = h.copy()
        perturbation = np.random.normal(0, learning_rate, n_points)
        h_new += perturbation
        
        # Apply penalties via clipping
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new configuration
        c5_new = evaluate_overlap(h_new, dx)
        energy_new = compute_energy(h_new, c5_new)
        energy_old = compute_energy(h, best_c5)
        
        # Metropolis acceptance criterion
        delta_energy = energy_new - energy_old
        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
        # Anneal temperature
        T = max(T * alpha, T_min)
        
        # Adjust learning rate if temperature is low
        if T < 0.01:
            learning_rate = 0.001

    # Gradient-based refinement
    def gradient_descent_refinement(h_vector, max_iter=200):
        h_refined = h_vector.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Simple gradient descent step
            c5_val = evaluate_overlap(h_refined, dx)
            # Approximate gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h_refined)
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (target_sum / h_plus_sum)
                c5_plus = evaluate_overlap(h_plus, dx)
                grad[i] = (c5_plus - c5_val) / eps
            
            # Update with gradient
            h_refined -= 0.001 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined_sum = np.sum(h_refined)
            if h_refined_sum > 0:
                h_refined = h_refined * (target_sum / h_refined_sum)
                
            # Evaluate and update best if improved
            new_c5 = evaluate_overlap(h_refined, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_refined.copy()
                
        return h_refined, best_c5
    
    # Perform refinement if time allows
    if time.time() - start_time < budget_s - 5:
        try:
            h_refined, final_c5 = gradient_descent_refinement(best_h)
            if final_c5 < best_c5:
                best_h = h_refined
                best_c5 = final_c5
        except:
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
