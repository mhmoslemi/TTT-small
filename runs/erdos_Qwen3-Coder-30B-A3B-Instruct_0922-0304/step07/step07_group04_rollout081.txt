import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 1000
    tournament_size = population_size // 10
    mutation_rate = 0.1
    max_no_improvement = 100
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        new_h = np.random.rand(n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * (target_sum / new_h_sum)
        population.append(new_h)
    
    # Store fitness of population
    fitness = [evaluate_overlap(ind, dx) for ind in population]
    
    best_global = best_c5
    no_improvement_count = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            # Blend crossover
            alpha = np.random.uniform(0.2, 0.8)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                mask = np.random.binomial(1, 0.1, n_points)
                noise = np.random.normal(0, 0.05, n_points)
                child1 += noise * mask
                child1 = np.clip(child1, 0, 1)
                child1_sum = np.sum(child1)
                if child1_sum > 0:
                    child1 = child1 * (target_sum / child1_sum)
                    
            if np.random.random() < mutation_rate:
                mask = np.random.binomial(1, 0.1, n_points)
                noise = np.random.normal(0, 0.05, n_points)
                child2 += noise * mask
                child2 = np.clip(child2, 0, 1)
                child2_sum = np.sum(child2)
                if child2_sum > 0:
                    child2 = child2 * (target_sum / child2_sum)
            
            new_population.extend([child1, child2])
        
        # Keep only valid individuals
        valid_population = []
        for ind in new_population:
            if np.all(0 <= ind) and np.all(ind <= 1):
                # Remap to proper sum
                ind_sum = np.sum(ind)
                if ind_sum > 0:
                    ind = ind * (target_sum / ind_sum)
                valid_population.append(ind)
        
        population = valid_population[:population_size]
        
        # Evaluate fitness
        fitness = [evaluate_overlap(ind, dx) for ind in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Adaptive mutation rate
        if gen % 50 == 0 and gen > 0:
            mutation_rate *= 0.95
    
    # Final local refinement with gradient-based optimization for best solution
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
