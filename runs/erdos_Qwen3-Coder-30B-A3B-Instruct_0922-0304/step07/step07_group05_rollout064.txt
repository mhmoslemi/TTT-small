import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    velocities = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Generate population with some diversity around initial
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
        for _ in range(population_size // 5):
            perturbed = h + np.random.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = normalize_h(perturbed, target_sum)
            particles.append(perturbed)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        # Fill the rest with random feasible solutions
        for _ in range(population_size - population_size // 5):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand.copy())
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    else:
        # Create a base function with more structure
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        # Add some randomness for diversity
        h_init += np.random.normal(0, 0.02, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
        particles.append(h_init.copy())
        velocities.append(np.zeros_like(h_init))
        # Rest of particles
        for _ in range(population_size - 1):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand.copy())
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))

    # Particle Swarm Optimization parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iter = 60
    iter_no_improve = 0
    max_no_improve = 20

    # Track best particle
    best_particle = particles[0].copy()
    best_fitness = compute_c5(best_particle, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(population_size):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (particle - particle)
            social = social_coeff * r2 * (best_particle - particle)
            velocity = inertia_weight * velocity + cognitive + social
            
            # Update particle
            particle += velocity
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Store updated particle and velocity
            particles[i] = particle
            velocities[i] = velocity

        # Evaluate the swarm and update global best
        for i in range(population_size):
            particle = particles[i]
            fitness = compute_c5(particle, dx)
            if fitness < best_fitness:
                best_fitness = fitness
                best_particle = particle.copy()
                iter_no_improve = 0
            else:
                iter_no_improve += 1

        # Early termination check
        if best_fitness <= 0.3807:
            break

        # Adjust inertia weight
        if iteration % 10 == 0 and iteration > 0:
            inertia_weight *= 0.98

        if iter_no_improve > max_no_improve:
            # increase mutation strength
            for i in range(population_size):
                mutation = np.random.normal(0, 0.05, n_points)
                particles[i] = np.clip(particles[i] + mutation, 0, 1)
                particles[i] = normalize_h(particles[i], target_sum)
            iter_no_improve = 0

    # Final local refinement with constraint-aware optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Try multiple local refinement attempts with different starting points
    current_best = best_particle.copy()
    current_best_c5 = best_fitness
    
    for _ in range(10):  # Up to 10 local refinement attempts
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use current best as starting point for local refinement
        result = minimize(objective, current_best.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            refined = np.clip(result.x, 0, 1)
            refined = normalize_h(refined, target_sum)
            refined_c5 = compute_c5(refined, dx)
            
            if refined_c5 < current_best_c5:
                current_best = refined
                current_best_c5 = refined_c5
        else:
            break

    # Ensure final solution is valid
    final_c5 = compute_c5(current_best, dx)
    
    return (current_best.tolist(), final_c5, n_points)
