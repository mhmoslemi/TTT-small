import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to fix the sum exactly
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Re-adjust to exact sum
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Spread the difference fairly
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create a simple initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some small randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    
    # For coordinate-wise optimization, we'll keep the values in relaxed space
    h_relaxed = h.copy()
    
    # Optimizer parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    reg_lambda = 1e-4
    
    # Moving averages for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    no_improvement_count = 0
    
    # Main loop
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute current gradient
        # Approximate gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        base_c5 = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_to_constraint(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        # Apply Adam update on h_relaxed
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter_count)
        v_hat = v / (1 - beta2 ** iter_count)
        
        h_relaxed = h_relaxed - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Regularization term in the update
        h_relaxed = h_relaxed - lr * reg_lambda * (h_relaxed - h)
        
        # Project to feasible region
        h = np.clip(h_relaxed, 0, 1)
        h = project_to_constraint(h, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate and regularization
        if no_improvement_count >= 100:
            lr *= 0.9
            no_improvement_count = 0
            
        if new_c5 < 0.3807:
            break
        
        # Adjust regularization strength
        if abs(new_c5 - base_c5) / (base_c5 + 1e-10) < 1e-6:
            reg_lambda *= 1.2
        else:
            reg_lambda *= 0.8
        reg_lambda = max(1e-6, min(1e-1, reg_lambda))
    
    return (best_h.tolist(), best_c5, n_points)
