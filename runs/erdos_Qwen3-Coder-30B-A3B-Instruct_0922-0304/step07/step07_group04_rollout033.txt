import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 1000
    tournament_size = max(1, pop_size // 10)
    mutation_rate = 0.1
    crossover_blend_min = 0.2
    crossover_blend_max = 0.8
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        new_h = np.random.rand(n_points)
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        population.append(new_h)
    
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            c5 = evaluate_overlap(h, dx)
            fitness.append(c5)
        return fitness
    
    # Evaluate initial population
    fitness = evaluate_population(population)
    
    # Main GA loop
    no_improvement_count = 0
    max_no_improvement = 20
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        next_pop = [h.copy() for h in selected[:1]]  # Keep best
        while len(next_pop) < pop_size:
            parent1, parent2 = np.random.choice(pop_size, 2, replace=False)
            p1, p2 = selected[parent1], selected[parent2]
            
            # Blend crossover
            alpha = np.random.uniform(crossover_blend_min, crossover_blend_max)
            child = alpha * p1 + (1 - alpha) * p2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.1
                child += np.random.normal(0, 0.05, n_points) * mask
                
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            next_pop.append(child)
        
        # Evaluate new population
        population = next_pop
        fitness = evaluate_population(population)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive mutation rate
        if no_improvement_count > 5:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        else:
            mutation_rate = max(0.01, mutation_rate * 0.99)
        
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final local refinement using SLSQP
    def objective(h_in):
        return evaluate_overlap(h_in, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Refine best solution
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
