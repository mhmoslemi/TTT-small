import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Adaptive Gaussian Mixture Algorithm
    def compute_c5(h_val, dx_val):
        """Compute the C5 value for given h."""
        overlap = correlate(h_val, 1 - h_val, mode="full")
        return np.max(overlap) * dx_val

    def project_to_constraints(h_val, target_sum_val):
        """Project h to satisfy the sum constraint while maintaining bounds."""
        h_val = np.clip(h_val, 0, 1)
        current_sum = np.sum(h_val)
        if abs(current_sum - target_sum_val) > 1e-10:
            h_val = h_val * (target_sum_val / current_sum)
        return h_val

    # Parameters
    n_components = 5
    population_size = 100
    elite_size = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize mixture components
    components = []
    for i in range(n_components):
        if 'initial_h_values' in globals() and i == 0:
            mean = np.array(initial_h_values)
            mean = np.clip(mean, 0, 1)
            mean = mean * (target_sum / np.sum(mean))
        else:
            mean = np.random.rand(n_points) * 0.5
            mean = np.clip(mean, 0, 1)
            mean = mean * (target_sum / np.sum(mean))
        cov = np.eye(n_points) * initial_sigma
        weight = 1.0 / n_components
        components.append({
            'mean': mean.copy(),
            'cov': cov.copy(),
            'weight': weight
        })
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main optimization loop
    iterations = 0
    max_iterations = 6000  # Ensures we have enough time budget
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        # Sample candidates from mixture
        candidates = []
        candidate_values = []
        
        for _ in range(population_size):
            # Choose component based on weights
            component_idx = np.random.choice(n_components, p=[c['weight'] for c in components])
            component = components[component_idx]
            
            # Sample from component
            try:
                sample = np.random.multivariate_normal(component['mean'], component['cov'])
            except:
                sample = component['mean'].copy()
            
            # Apply constraints
            sample = project_to_constraints(sample, target_sum)
            
            # Evaluate
            c5_val = compute_c5(sample, dx)
            
            candidates.append(sample.copy())
            candidate_values.append(c5_val)
        
        # Selection: keep top elite
        elite_indices = np.argsort(candidate_values)[:elite_size]
        elite_candidates = [candidates[i] for i in elite_indices]
        elite_values = [candidate_values[i] for i in elite_indices]
        
        # Update mixture components
        for i, component in enumerate(components):
            # Update weight
            component['weight'] = 0.0
            for j, candidate in enumerate(elite_candidates):
                if np.allclose(candidate, component['mean'], atol=1e-6):
                    component['weight'] += 1.0 / elite_size
            
            # Update mean
            w_sum = 0
            new_mean = np.zeros(n_points)
            for j, candidate in enumerate(elite_candidates):
                weight = 1.0  # Equal weighting for simplicity
                new_mean += weight * candidate
                w_sum += weight
            if w_sum > 0:
                component['mean'] = new_mean / w_sum
            
            # Update covariance
            if len(elite_candidates) > 1:
                # Compute covariance
                centered = np.array(elite_candidates) - component['mean']
                cov = np.zeros((n_points, n_points))
                for j in range(len(elite_candidates)):
                    diff = centered[j:j+1].T
                    cov += diff @ diff.T
                cov /= len(elite_candidates)
                
                # Regularize to prevent degeneracy
                cov = cov + 1e-8 * np.eye(n_points)
                
                # Store updated covariance
                component['cov'] = cov
                
            # Decay covariance
            component['cov'] *= covariance_decay
            
            # Normalize weights to sum to 1
            total_weight = sum(c['weight'] for c in components)
            if total_weight > 0:
                for c in components:
                    c['weight'] /= total_weight
        
        # Update best solution
        for i in range(len(elite_values)):
            if elite_values[i] < best_c5:
                best_c5 = elite_values[i]
                best_h = elite_candidates[i].copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
