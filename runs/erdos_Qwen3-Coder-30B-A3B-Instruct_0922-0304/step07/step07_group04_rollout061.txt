import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.999
    learning_rate = 0.01
    max_iter = int(budget_s * 1000)
    patience = 1000
    patience_counter = 0
    best_improvement = 0.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the solution
        h_new = h.copy()
        perturbation = np.random.normal(0, learning_rate * T, n_points)
        h_new += perturbation
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
                best_improvement = 0.0
                patience_counter = 0
            else:
                patience_counter += 1
                best_improvement += delta
        else:
            patience_counter += 1
            
        # Cooling and learning rate adjustment
        T *= alpha
        if T < 1e-6:
            T = 1e-6
        if patience_counter > patience:
            patience_counter = 0
            T = min(T * 1.1, 1.0)  # Reset temperature if stuck
            
    # Local refinement with gradient descent
    if time.time() - start_time < budget_s - 5:
        # Compute approximate gradients
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
            
        # Add small perturbations to estimate gradients
        eps = 1e-4
        gradient = np.zeros(n_points)
        base_c5 = objective(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
                
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum > 0:
                h_minus = h_minus * (target_sum / h_minus_sum)
                
            grad_i = (objective(h_plus) - objective(h_minus)) / (2 * eps)
            gradient[i] = grad_i
            
        # Gradient descent step
        for step in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute gradient with respect to objective
            h = h - 0.01 * gradient
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (target_sum / h_sum)
                
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
