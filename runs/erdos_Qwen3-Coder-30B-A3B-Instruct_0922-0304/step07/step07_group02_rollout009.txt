import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_sum_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
        
    # PGD parameters
    alpha = 0.1
    beta = 0.9
    momentum = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stagnation_count = 0
    max_stagnation = 500
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < 3000:
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient ∇h C₅ = (1 - h) shifted by k
        grad = np.zeros(n_points)
        if max_k < n_points:
            # h shifted right by max_k
            grad[max_k:] = 1 - h[:-max_k]
        else:
            # max_k >= n_points, so h shifted left by (max_k - n_points + 1)
            k_shift = max_k - n_points + 1
            grad[:k_shift] = 1 - h[k_shift:]
            
        # Nesterov momentum update
        momentum = beta * momentum + grad
        h_new = h - alpha * momentum
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = project_to_sum_constraint(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        h = h_new
        
        # Adaptive learning rate
        if stagnation_count > 100:
            alpha *= 0.95
            alpha = max(alpha, 1e-5)
            stagnation_count = 0
            
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
