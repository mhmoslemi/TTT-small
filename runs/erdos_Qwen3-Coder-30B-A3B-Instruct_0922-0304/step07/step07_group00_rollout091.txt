import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Ensure it sums to target
    h = h / np.sum(h) * target_sum
    
    # Use logit transform for optimization
    w = np.log(h / (1 - h + 1e-12))
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iterations = int(1e5)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Compute current overlap to find max shift
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        g = np.zeros_like(w)
        for i in range(n_points):
            if i + k_max < n_points:
                # Gradient contribution from this overlap term
                g[i] += (1 - 2 * h[i]) * (1 - h[i + k_max])
            if i - k_max >= 0:
                g[i] += (1 - 2 * h[i]) * (1 - h[i - k_max])
        
        # Update parameters with Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        w = w - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply thresholding to maintain feasibility
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-12))
        
        # Check and update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive learning rate decay
        if iteration % 1000 == 0 and iteration > 0:
            lr *= 0.999
            
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
