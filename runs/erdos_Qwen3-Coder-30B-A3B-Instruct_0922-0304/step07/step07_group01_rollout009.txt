import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_bounds(h):
    return np.clip(h, 0, 1)

def project_to_sum(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Stage 1: convex relaxation using CVXPY
    if time.time() - start_time < budget_s - 100:  # Reserve time for stage 2
        # Convex relaxation optimization
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        objective = cp.Minimize(C)
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
        ]
        
        # For each shift k, add constraint that the overlap is bounded by C
        for k in range(2 * n_points - 1):
            # Compute the overlap at shift k
            shift = k - (n_points - 1)
            if shift >= 0:
                overlap_part = h_var[shift:] * (1 - h_var[:n_points-shift])
            else:
                overlap_part = h_var[:n_points+shift] * (1 - h_var[-shift:])
            
            # Add constraint on the sum of overlap
            if len(overlap_part) > 0:
                constraints.append(cp.sum(overlap_part) <= C)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_var.value
                # Convert to step function by thresholding
                h = project_to_bounds(h_relaxed)
                h = project_to_sum(h, target_sum)
                # Update dx as it may have changed due to different resolution
                dx = 2.0 / n_points
        except:
            pass  # Fall back to current h if optimization fails
    
    # Stage 2: Gradient descent refinement
    max_iter = 1000
    learning_rate = 0.01
    decay_rate = 0.995
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        
        # Compute gradients with respect to h
        # This is a simplified gradient approximated with finite differences for simplicity
        # In practice, one would compute analytical gradient or more sophisticated approximation
        h_grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_bounds(h_plus)
            h_plus = project_to_sum(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_grad[i] = (c5_plus - current_c5) / eps
        
        # Update h with gradient descent
        h = h - learning_rate * h_grad
        
        # Project to constraints
        h = project_to_bounds(h)
        h = project_to_sum(h, target_sum)
        
        # Update learning rate
        learning_rate *= decay_rate
        
        # Early termination if target achieved
        if current_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
