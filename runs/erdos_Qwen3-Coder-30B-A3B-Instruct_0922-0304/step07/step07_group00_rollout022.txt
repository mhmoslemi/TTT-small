import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute the gradient of overlap_k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1 - 2 * h[i + k_max]
    return grad

def solve_linear_subproblem(grad, n_points):
    """Solve the linear subproblem to find the descent direction."""
    # We want to minimize s^T * grad subject to sum(s) = 0, 0 <= s <= 1
    # This is a linear program with simple bounds
    s = np.zeros(n_points)
    
    # Sort indices by gradient value
    idxs = np.argsort(grad)
    
    # Using the method of finding the optimal solution with zero sum constraint
    # We set up a system of equations to understand the zero sum constraint
    # For gradient values, we set as many as possible to their lower bound (0)
    # and adjust others such that sum is 0
    
    # Start from the highest gradients and set them to zero first
    # But need to satisfy sum(s) = 0 constraint
    # Not sure how to solve this easily directly
    
    # Instead, we try a heuristic approach:
    # We look for an s such that 
    # 1. s[i] in [0,1] for all i
    # 2. sum(s) = 0  
    # 3. s^T grad is minimized
    
    # A simpler, more robust approach:
    # Create a direction that tries to decrease gradient but stays feasible
    # We can try: 
    # s[i] = 1 if grad[i] < 0, 0 otherwise; then adjust to meet sum = 0
    
    s = np.zeros(n_points)
    pos_indices = np.where(grad < 0)[0]
    
    if len(pos_indices) > 0:
        s[pos_indices] = 1  # These are the ones we want to increase
    
    sum_s = np.sum(s)
    if abs(sum_s) > 1e-10:
        s = s / sum_s * 0  # Reset if constraint is violated
    
    # Simple setup: make change simple to avoid optimization here
    s = s * 0  # Reset to zero
    
    # Take gradient direction components of potential, but use the fact
    # we can do a change that tries to reduce sum of (1-2*h[i+k_max]) * h[i]
    # Approximate by reduction direction
    
    # Get indices with positive gradient (want to reduce these):
    pos_grad = np.where(grad > 0)[0]
    if len(pos_grad) > 0:
        # Try approach of making s zero where grad is negative,
        # and try setting s to 1 where diff is maximal.
        # However, we also need sum(s) = 0 
        # Need a valid descent direction with sum = 0
        # This is simple:
        # Let s be positive where grad is negative (to bring it down)
        # positive where grad is positive (to bring it up, but this conflicts with sum)
        # Simpler: we try setting s = 1 where grad < 0, 0 elsewhere, and adjust for sum
        s = np.zeros(n_points)
        # Use direction which reduces most of the negative contributions to linearized obj
        # We consider setting one coefficient to 1 to make sum = 0
        s[0] = 1
        # require sum(s) = 0, so need total dx = 0, adjust self
        s = s - np.mean(s)
    else:
        # All positive: setting s to all negative to see what helps
        # Since objective is to minimize, take s that most degrades
        # Direct simple one: s = -grad where neg grad do we try to prevent
        # Here again, want sum s = 0, so consider a mixed one way to go,
        # perhaps just compute sum and such as in the update step (h + alpha*(s-h))
        s = np.zeros(n_points)
    
    return s

def backtracking_line_search(h, grad, step_dir, dx, max_iter=10):
    """Perform backtracking line search for step size."""
    # Linearized goal is to minimize over s: s^T * grad
    # Our step: h_new = h + alpha * (s - h)
    # The update: h_new = (1-alpha)*h + alpha*s
    # Do not modify h yet
    # Start with alpha = 0.5
    alpha = 0.5
    min_c5 = float('inf')
    best_alpha = 0.5
    
    # Try enough decreasing values
    for i in range(max_iter):
        new_h = h + alpha * (step_dir - h)
        # Adjust to keep feasible
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, len(h) / 2.0)
        c5_new = compute_c5(new_h, dx)
        if c5_new < min_c5:
            min_c5 = c5_new
            best_alpha = alpha
        # Early termination condition
        if c5_new < 0.3807:
            return alpha, c5_new
        alpha *= 0.5
        
    return best_alpha, min_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h

    h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    iterations = 0
    max_iter = 5000
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        if iterations > max_iter or iterations - last_improvement > 500:
            break
            
        # Compute all overlaps for current h
        # 1. Computing overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        c5_current = np.max(overlap) * dx
        
        if c5_current < 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradients(h, k_max)
        
        # Solve linearized subproblem
        step_dir = solve_linear_subproblem(grad, n_points)
        
        # Backtracking line search
        alpha, c5_new = backtracking_line_search(h, grad, step_dir, dx)
        
        # Update h
        h = h + alpha * (step_dir - h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Check if this is better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            last_improvement = iterations
            
        iterations += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
