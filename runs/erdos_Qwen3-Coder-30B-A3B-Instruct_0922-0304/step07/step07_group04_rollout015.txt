import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use a more sophisticated gradient-based approach with adaptive steps
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Gradient descent with momentum and adaptive learning rate
    max_iter = 1000
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    lr = 0.01
    min_lr = 1e-6
    patience = 10
    best_improvement = 1e10
    patience_counter = 0
    
    # Use first-order approximation for gradient-like behavior
    def finite_diff_grad(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        base_val = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute approximate gradient
        grad = finite_diff_grad(best_h)
        
        # Apply constraints in gradient update
        grad = np.clip(grad, -10, 10)  # Clip gradients to avoid instability
        
        # Update with momentum
        velocity = momentum * prev_grad - lr * grad
        new_h = best_h + velocity
        
        # Project back to feasible set
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            best_improvement = 1e10
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience:
                if lr > min_lr:
                    lr *= 0.5
                    patience_counter = 0
        
        prev_grad = velocity
        
        # Early stopping based on small improvement
        if abs(new_c5 - best_c5) < 1e-8:
            break
    
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
