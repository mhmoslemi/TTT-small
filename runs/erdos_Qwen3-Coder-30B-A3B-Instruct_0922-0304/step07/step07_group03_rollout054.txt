import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    population_size = 20
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with initial solution
    population.append(h.copy())
    fitness.append(compute_c5(h, dx))
    
    # Generate random population
    for _ in range(population_size - 1):
        h_candidate = np.random.uniform(0, 1, n_points)
        h_candidate = project_to_constraints(h_candidate, target_sum)
        population.append(h_candidate)
        fitness.append(compute_c5(h_candidate, dx))
    
    # Main loop
    max_iter = 3000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we've met target
        if fitness[0] <= 0.3807:
            break
            
        # Adaptive mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Create new population
        new_population = population[:elite_size]  # Keep elite
        
        while len(new_population) < population_size:
            # Select parent (tournament selection)
            parent_idx = np.random.randint(0, elite_size)
            parent = population[parent_idx]
            
            # Mutate parent
            mutation = np.random.normal(0, mutation_rate, n_points)
            child = parent + mutation
            child = project_to_constraints(child, target_sum)
            
            new_population.append(child)
            if len(new_population) < population_size:  # Don't compute fitness for replacement candidate
                fitness.append(compute_c5(child, dx))
        
        population = new_population
        fitness = [compute_c5(p, dx) for p in population]
    
    # Return best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
