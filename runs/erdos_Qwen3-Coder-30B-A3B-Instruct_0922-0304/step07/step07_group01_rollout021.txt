import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple initial guess
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h to have correct sum
    h = project_to_constraints(h, target_sum)
    
    # Parameters for optimization
    max_iter = 500
    step_size = 0.01
    
    # Optimizer state
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Gradient-based local refinement
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximate via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_constraints(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        h_new = h - step_size * grad
        h_new = project_to_constraints(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Accept update if better
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Early stopping if target reached
        if current_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
