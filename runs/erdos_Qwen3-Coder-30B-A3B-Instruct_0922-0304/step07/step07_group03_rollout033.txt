import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, maxiter=1000):
    """Local optimization using L-BFGS-B."""
    def objective(x):
        x = project_to_constraints(x, target_sum)
        return compute_c5(x, dx)
    
    def jac(x):
        # Simple finite difference gradient estimation
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad
    
    bounds = [(0, 1) for _ in range(len(h))]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      jac=jac, options={'maxiter': maxiter, 'ftol': 1e-9})
    return project_to_constraints(result.x, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # ES parameters
    population_size = 20
    sigma_initial = 0.1
    sigma_final = 0.01
    max_generations = 2000
    elite_ratio = 0.5
    
    # Initialize population
    population = []
    fitness = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual += np.random.normal(0, sigma_initial, n_points)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = project_to_constraints(individual, target_sum)
        population.append(individual)
        fitness.append(compute_c5(individual, dx))
    
    best_fitness = min(fitness)
    best_individual = population[fitness.index(best_fitness)].copy()
    
    # Evolutionary loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Sort population
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Check if we should do local optimization
        if fitness[0] < 0.381:
            # Do local optimization on best
            better_h = local_optimize(population[0], target_sum, dx)
            better_c5 = compute_c5(better_h, dx)
            if better_c5 < fitness[0]:
                population[0] = better_h
                fitness[0] = better_c5
        
        # Update best solution
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_individual = population[0].copy()
        
        # Elitism
        elite_count = int(elite_ratio * population_size)
        elite = population[:elite_count]
        elite_fitness = fitness[:elite_count]
        
        # New generation
        new_population = elite.copy()
        new_fitness = elite_fitness.copy()
        
        # Generate offspring with Gaussian mutation
        current_sigma = sigma_final + (sigma_initial - sigma_final) * (1 - generation / max_generations)
        
        while len(new_population) < population_size:
            # Tournament selection
            parent_idx = np.random.randint(0, elite_count)
            parent = elite[parent_idx]
            
            # Mutation
            mutant = parent + np.random.normal(0, current_sigma, n_points)
            mutant = project_to_constraints(mutant, target_sum)
            
            new_population.append(mutant)
            new_fitness.append(compute_c5(mutant, dx))
        
        population = new_population
        fitness = new_fitness
        
        # Early stopping
        if best_fitness <= 0.3807:
            break
    
    return (best_individual.tolist(), best_fitness, n_points)
