import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Local search with simulated annealing
    def perturb(h, temperature):
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05 * temperature)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        return h_new
    
    # Simulated Annealing
    temperature = 0.1
    cooling_rate = 0.999
    max_iterations = 10000
    best_temp_c5 = best_c5
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = perturb(best_h, temperature)
        c5_new = evaluate_overlap(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_temp_c5 = c5_new
        elif np.random.random() < np.exp(-(c5_new - best_c5) / temperature):
            best_h = h_new
            best_c5 = c5_new
            best_temp_c5 = c5_new
            
        temperature *= cooling_rate
        
        if i % 1000 == 0:
            # Check if we're making progress
            if best_c5 < 0.38085:  # Early exit if target reached
                break
    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
