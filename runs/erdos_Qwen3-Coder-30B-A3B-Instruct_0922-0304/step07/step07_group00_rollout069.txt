import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    """Sigmoid function to map reals to (0,1)"""
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Use logit (inverse sigmoid) for optimization
    w = np.log(h / (1 - h + 1e-10))
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Moments for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute gradient 
        # For simplicity, use finite differences for gradient estimation
        eps = 1e-4
        grad = np.zeros(n_points)
        
        # Estimate gradient by finite differences
        for i in range(n_points):
            w_plus = w.copy()
            w_plus[i] += eps
            h_plus = sigmoid(w_plus)
            h_plus = h_plus / np.sum(h_plus) * target_sum
            c5_plus = compute_c5(h_plus, dx)
            
            w_minus = w.copy()
            w_minus[i] -= eps
            h_minus = sigmoid(w_minus)
            h_minus = h_minus / np.sum(h_minus) * target_sum
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project back to feasible region
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-10))
        
        # Update best solution if improved
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1

    return (best_h.tolist(), best_c5, n_points)
