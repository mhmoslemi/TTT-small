import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimize using Adam
    max_iter = int(1e5)
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # For numerical stability, compute overlaps manually
        overlap_values = []
        for k in range(n_points):
            overlap = 0
            for i in range(n_points - k):
                overlap += h[i] * (1 - h[i+k])
            overlap_values.append(overlap)
        
        max_k = np.argmax(overlap_values)
        max_overlap = overlap_values[max_k]
        
        # Gradient with respect to w
        g = np.zeros(n_points)
        for i in range(n_points - max_k):
            # derivative of h[i] * (1 - h[i + max_k]) wrt w[i]
            # = h[i] * (-h'[i + max_k]) + (1 - h[i + max_k]) * h'[i]
            # h'[i] = h[i] * (1 - h[i])
            h_prime_i = h[i] * (1 - h[i])
            h_prime_i_plus_k = h[i + max_k] * (1 - h[i + max_k])
            g[i] += h_prime_i_plus_k - h[i] * h_prime_i
            g[i + max_k] -= h_prime_i_plus_k
            
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h
        h = sigmoid(w)
        # Scale to correct sum
        h = h / np.sum(h) * target_sum
        # Clip to stay feasible
        h = np.clip(h, 1e-8, 1 - 1e-8)
        # Transform back to w
        w = np.log(h / (1 - h + 1e-8))
        
        # Evaluate the new solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
