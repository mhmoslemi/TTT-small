import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with both initial solution and random
    population_size = 100
    population = []
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        # Create a random initial solution
        h_init = np.random.rand(n_points)
        h_init = project_to_feasible(h_init, target_sum)
    
    # Fill population with slight variations of initial solution
    for _ in range(population_size // 5):
        h = h_init + np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    # Fill the rest with random feasible solutions
    for _ in range(population_size - population_size // 5):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = population[:]
    velocities = [np.random.rand(n_points) * v_max for _ in range(population_size)]
    
    # Best positions
    p_best = particles[:]
    p_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]
    
    # PSO iterations
    max_iters = 60
    stagnation_count = 0
    stagnation_limit = 20
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        # Update each particle
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = c1 * r1 * (p_best[i] - particles[i])
            social = c2 * r2 * (g_best - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Early termination if no improvement
        if stagnation_count >= stagnation_limit:
            break
    
    # Final gradient descent refinement
    best_h = g_best.copy()
    best_c5 = g_best_fitness
    
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(20):  # Run gradient descent for refinement
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            new_c5 = compute_c5(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                break
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
