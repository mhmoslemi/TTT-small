import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute gradient of overlap_k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # Gradient is (1 - 2*h) shifted by k_max
    for i in range(n - k_max):
        grad[i] += (1 - 2 * h[i + k_max])
    return grad

def line_search(h, grad, s, dx, alpha_max=1.0, c=0.5):
    """Perform backtracking line search."""
    alpha = alpha_max
    current_c5 = compute_c5(h, dx)
    
    for _ in range(10):  # max backtracking steps
        h_new = h + alpha * (s - h)
        h_new = normalize_h(h_new, np.sum(h))  # predict target
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, np.sum(h))  # final normalization
        
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 <= current_c5 - c * alpha * np.dot(grad, s - h):
            return alpha, h_new
        
        alpha *= 0.5
    return alpha, h
