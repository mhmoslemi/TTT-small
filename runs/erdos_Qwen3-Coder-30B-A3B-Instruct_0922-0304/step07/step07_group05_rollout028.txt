import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def evaluate_individual(h, dx, target_sum):
    """Evaluate an individual and ensure feasibility."""
    h = project_to_feasible(h, target_sum)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize population
    population_size = 100
    population = []
    
    # Use initial construction if available, otherwise create random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        h_init = np.random.rand(n_points)
        h_init = normalize_h(h_init, target_sum)
    
    # Create diverse initial population
    population = []
    for _ in range(int(0.2 * population_size)):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particle positions and velocities
    positions = np.array(population)
    velocities = np.random.uniform(-v_max, v_max, (population_size, n_points))
    personal_best_positions = positions.copy()
    personal_best_scores = [evaluate_individual(p, dx, target_sum) for p in positions]
    
    # Find global best
    best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]
    
    # PSO iterations
    max_iter = 60
    stagnation_count = 0
    best_improvement = 0.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            cognitive = c1 * r1 * (personal_best_positions[i] - positions[i])
            social = c2 * r2 * (global_best_position - positions[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            positions[i] += velocities[i]
            
            # Projection to feasible region
            positions[i] = project_to_feasible(positions[i], target_sum)
            
            # Evaluate
            score = evaluate_individual(positions[i], dx, target_sum)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = positions[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = positions[i].copy()
                    stagnation_count = 0
                    best_improvement = 0.0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        if stagnation_count > 20 or global_best_score <= 0.3807:
            break
    
    # Final gradient refinement on the best solution
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Refine best solution with SLSQP
    refined_solution = project_to_feasible(global_best_position, target_sum)
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, refined_solution, method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            refined_solution = result.x
            refined_solution = project_to_feasible(refined_solution, target_sum)
        else:
            break
    
    # Evaluate final solution
    final_c5 = compute_c5(refined_solution, dx)
    
    # Ensure we return valid solution if no progress
    if np.isnan(final_c5) or np.isinf(final_c5) or final_c5 > 1:
        final_c5 = global_best_score
        refined_solution = global_best_position
        
    return (refined_solution.tolist(), final_c5, n_points)
