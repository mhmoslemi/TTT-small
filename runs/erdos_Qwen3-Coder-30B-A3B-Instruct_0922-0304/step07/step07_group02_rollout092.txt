import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    # Re-initialize for coordinate-wise approach
    h_relaxed = h.copy()
    h_relaxed += np.random.normal(0, 0.5, n_points)
    h_relaxed = np.clip(h_relaxed, 0, 1)

    # Learning parameters
    alpha = 0.1
    beta = 0.9
    lambda_reg = 1e-4
    momentum = np.zeros(n_points)
    velocity = np.zeros(n_points)

    # Variables for convergence monitoring
    best_h = h_relaxed.copy()
    best_c5 = compute_c5(h_relaxed, dx)
    no_improvement_count = 0

    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < 3000:
        # Compute current C5
        current_c5 = compute_c5(h_relaxed, dx)

        # Update with gradient-based approach + momentum
        grad = np.zeros(n_points)
        # Manually compute gradient for each variable using finite diff
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_to_constraint(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)

            h_minus = h_relaxed.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)

            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2e-5)

        # Update momentum and velocity
        momentum = beta * momentum + (1 - beta) * grad

        # Update h_relaxed
        update = alpha * momentum / (np.sqrt(velocity) + 1e-8)
        h_relaxed = h_relaxed - update
