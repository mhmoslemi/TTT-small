import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create random initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_iter = 5000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        candidate = h + noise
        candidate = project_to_constraints(candidate, target_sum)
        population.append(candidate)
    
    # Evaluate initial population
    fitness = [compute_c5(individual, dx) for individual in population]
    
    # Best solution found
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Evolution loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * (iter_num / max_iter)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        rest = [population[i] for i in sorted_indices[elite_size:]]
        
        # Generate offspring using mutation
        offspring = []
        for individual in elite:
            noise = np.random.normal(0, mutation_rate, n_points)
            mutated = individual + noise
            mutated = project_to_constraints(mutated, target_sum)
            offspring.append(mutated)
        
        # Replace worst individuals
        population = elite + offspring
        
        # Evaluate new population
        fitness = [compute_c5(individual, dx) for individual in population]
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Early stopping
        if best_fitness <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
