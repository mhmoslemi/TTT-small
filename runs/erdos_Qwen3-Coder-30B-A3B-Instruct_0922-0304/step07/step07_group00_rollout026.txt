import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute the gradient of the objective function."""
    n = len(h)
    grad = np.zeros(n)
    
    # For each position, compute contribution to gradient
    for i in range(n):
        # Compute the sum over all valid k
        for k in range(n):
            # convolution term contribution
            if i + k < n:
                grad[i] += 2 * (h[i] * (1 - h[i + k]) - h[i + k] * (1 - h[i])) / dx
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
        h = normalize_h(h, target_sum)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Adam optimizer parameters
    alpha = 0.01  # learning rate
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update parameters
        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Compute objective value
        c5 = compute_c5(h, dx)
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        if last_improvement > 1000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
