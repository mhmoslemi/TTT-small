import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Lagrangian-based optimization
    max_iter = 1000
    alpha = 0.01
    gamma = 0.1
    tolerance = 1e-6
    last_improvement = 0
    
    h_current = h.copy()
    mu = 1.0
    lambda_plus = np.zeros(n_points)
    lambda_minus = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find maximum overlap shift
        cross_corr = correlate(h_current, 1 - h_current, mode="full")
        max_shift_idx = np.argmax(cross_corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient (stationarity condition)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i - max_shift) % n_points
            gradient[i] = (1 - h_current[j]) - mu + lambda_plus[i] - lambda_minus[i]
            
        # Update h
        h_new = h_current - alpha * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Update multipliers
        mu = max(0, mu + gamma * (np.sum(h_new) - target_sum))
        lambda_plus = np.maximum(0, lambda_plus + gamma * (h_new - 1))
        lambda_minus = np.maximum(0, lambda_minus + gamma * (-h_new))
        
        h_current = h_new
        
        # Check for improvement
        current_c5 = evaluate_overlap(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Check convergence
        if last_improvement > 50:
            break
    
    # Final gradient-based refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
