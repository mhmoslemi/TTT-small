import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def optimize_step_function(h0, n_points, dx):
    # Convert to a step function with optimal positions
    # This is a simplified approach using local search on step positions
    
    def objective(params):
        # params are positions of steps (in normalized space [0,2])
        # Convert to actual step function
        h = np.zeros(n_points)
        step_positions = np.clip(params, 0, 2)
        step_positions = np.sort(step_positions)
        
        # Create step function based on positions
        for i in range(len(step_positions)):
            start = int(step_positions[i] / dx)
            if i == len(step_positions) - 1:
                end = n_points
            else:
                end = int(step_positions[i+1] / dx)
            if start < 0:
                start = 0
            if end > n_points:
                end = n_points
            if end > start:
                h[start:end] = 1
                
        h = project_to_sum_constraint(h, n_points / 2)
        
        # Compute the overlap
        return compute_c5(h, dx)
    
    # Start with initial step positions
    initial_positions = np.linspace(0, 2, len(h0) + 1)[1:-1]
    
    # Optimize using scipy minimize
    res = minimize(objective, initial_positions, method='L-BFGS-B', 
                   bounds=[(0, 2)] * len(initial_positions),
                   options={'maxiter': 500})
    
    # Reconstruct final h
    h_final = np.zeros(n_points)
    step_positions = np.clip(res.x, 0, 2)
    step_positions = np.sort(step_positions)
    
    for i in range(len(step_positions)):
        start = int(step_positions[i] / dx)
        if i == len(step_positions) - 1:
            end = n_points
        else:
            end = int(step_positions[i+1] / dx)
        if start < 0:
            start = 0
        if end > n_points:
            end = n_points
        if end > start:
            h_final[start:end] = 1
    
    h_final = project_to_sum_constraint(h_final, n_points / 2)
    return h_final

    # Make a convex relaxation to get a better starting point
    # Variables for convex optimization
    h_vars = cp.Variable(n_points)
    # Constraints

    # Objective: minimize max convolution
    # We'll approximate the max convolution using a conservative bound
    # Since this is an NP-hard problem, we'll use heuristic assistance
    
    # Reconstruct as a step function alternating (simpler than convex optimization directly)
    h_step = np.zeros(n_points)
    h_step[::2] = 1
    h_step = project_to_sum_constraint(h_step, n_points / 2)
    
    # Local search on step positions
    h_optimized = optimize_step_function(h_step, n_points, dx)
