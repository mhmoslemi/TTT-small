import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        scale = target_sum / current_sum
        h = h * scale
    else:
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, maxiter=1000):
    """Local optimization using L-BFGS-B."""
    def objective(h_vec):
        h_local = h_vec.copy()
        h_local = project_to_constraints(h_local, target_sum)
        return compute_c5(h_local, dx)
    
    def gradient(h_vec):
        # Simple finite difference gradient approximation
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(len(h))]
    
    # Initial h vector
    h_vec = h.copy()
    
    # Optimize
    result = minimize(
        objective,
        h_vec,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'gtol': 1e-9},
        callback=None
    )
    
    if result.success:
        h_opt = result.x
        h_opt = project_to_constraints(h_opt, target_sum)
    else:
        h_opt = h.copy()
    
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # Evolutionary Strategy parameters
    pop_size = 20
    generations = 2000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = normalize_h(individual, target_sum)
        population.append(individual.copy())
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Mutation parameters
    sigma_start = 0.1
    sigma_end = 0.01
    sigma_decay = (sigma_start - sigma_end) / generations
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Threshold for local optimization
        threshold = 0.381
        if fitness[0] < threshold:
            local_h = local_optimize(population[0], target_sum, dx)
            local_c5 = compute_c5(local_h, dx)
            if local_c5 < best_c5:
                best_h = local_h.copy()
                best_c5 = local_c5
        
        # Create new generation
        new_population = []
        
        # Keep top 50%
        top_count = pop_size // 2
        for i in range(top_count):
            new_population.append(population[i].copy())
        
        # Generate mutants
        sigma = sigma_start - gen * sigma_decay
        for _ in range(pop_size - top_count):
            # Select parent
            parent_idx = np.random.randint(0, top_count)
            parent = population[parent_idx].copy()
            
            # Add Gaussian noise
            noise = np.random.normal(0, sigma, n_points)
            mutant = parent + noise
            
            # Apply constraints
            mutant = project_to_constraints(mutant, target_sum)
            new_population.append(mutant)
        
        population = new_population
    
    # Final optimization
    final_h = local_optimize(best_h, target_sum, dx)
    final_c5 = compute_c5(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
