import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a balanced initial step function
        h = np.ones(n_points) * 0.5
        
    # PGD with Nesterov momentum
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    no_improvement_count = 0
    last_best_c5 = best_c5
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient: (1-h) shifted by k_max
        # The full correlation result has length 2*n_points - 1
        # We want the shift k_max to the right, so we shift (1-h) left by k_max
        grad = np.zeros(n_points)
        # The shifting operation (inverse of correlation) perspective
        # Use the element-wise convolution gradient approximation
        shift_vec = 1 - h
        if k_max < n_points:  # Shifting forward
            grad[k_max:] = shift_vec[:n_points - k_max]
        else:  # Shifting backward (or mid overlapping)
            grad[:n_points - (k_max - n_points + 1)] = shift_vec[k_max - n_points + 1:]
            
        # Update velocity with Nesterov momentum
        velocity = momentum * velocity - learning_rate * grad
        new_h = h + velocity
        
        # Apply bounds
        new_h = np.clip(new_h, 0, 1)
        # Apply sum constraint
        new_h = project_to_constraint(new_h, target_sum)
        
        # Evaluate new point
        new_c5 = compute_c5(new_h, dx)
        
        # Update current solution
        h = new_h
        c5 = new_c5
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if no_improvement_count > 100:
            learning_rate *= 0.95
            no_improvement_count = 0
        learning_rate = max(1e-5, learning_rate)
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
