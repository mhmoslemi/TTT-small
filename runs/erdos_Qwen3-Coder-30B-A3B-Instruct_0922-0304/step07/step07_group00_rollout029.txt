import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def gradient_objective(h, n_points, dx):
    """Compute gradient of the objective function approximating C5."""
    # Compute overlap for all lags
    h_extended = np.concatenate([h, np.zeros(n_points)])
    overlap = np.zeros(2 * n_points - 1)
    for k in range(2 * n_points - 1):
        overlap[k] = np.sum(h * (1 - h_extended[k:k+n_points]))
    
    # Approximate max by sum of squares
    max_idx = np.argmax(overlap)
    
    # compute gradient for a single maximizer
    grad = np.zeros(n_points)
    for i in range(n_points):
        for k in range(2 * n_points - 1):
            if i + k < n_points and i + k >= 0:
                # avoid uncertainty in gradient calculation
                if k == max_idx:
                    grad[i] += 2.0 * overlap[k] * dx * (-1 + 2 * h[i+k])
    return grad / (np.linalg.norm(grad) + 1e-12)
