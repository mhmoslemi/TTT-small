import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Use the Lagrangian approach for improvement
    max_iter = 1000
    alpha = 0.01  # Learning rate
    gamma = 0.1   # Multiplier update rate
    epsilon = 1e-5  # Convergence threshold
    
    # Initialize Lagrange multipliers
    mu = 1.0
    lambda_plus = np.zeros(n_points)
    lambda_minus = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find current maximizing shift
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        
        # Compute the gradient: sum over shifts k which maximize overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # The gradient for h[i] is the sum of 1-h[(i-k)%n] over all k that maximize overlap
            if i - k_max >= 0:
                gradient[i] = 1 - h[i - k_max]
            else:
                gradient[i] = 1 - h[i - k_max + n_points]
        
        # Update h using the gradient
        h_new = h - alpha * (gradient - mu + lambda_plus - lambda_minus)
        
        # Project h onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain the sum constraint by renormalizing
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Update Lagrange multipliers
        mu = max(0, mu + gamma * (np.sum(h_new) - target_sum))
        lambda_plus = np.maximum(0, lambda_plus + gamma * (h_new - 1))
        lambda_minus = np.maximum(0, lambda_minus + gamma * (-h_new))
        
        h = h_new
        
        # Evaluate new solution
        c5 = evaluate_overlap(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Final refinement using SLSQP
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
