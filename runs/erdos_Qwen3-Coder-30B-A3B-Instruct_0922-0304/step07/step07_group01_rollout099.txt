import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint
    h = project_to_constraints(h, target_sum)
    
    # Try to improve using convex optimization in frequency domain
    # We use a different optimization approach: setup a convex program
    # to minimize the maximum overlap by constraining the Fourier transform
    
    # Create frequency domain variables
    n_freqs = 2 * n_points - 1  # Full convolution length
    
    # Inputs to optimizer function
    h_opt = cp.Variable(n_points)
    
    # Ensure h_opt is in [0,1]
    constraints = [h_opt >= 0, h_opt <= 1]
    
    # Sum constraint
    constraints.append(cp.sum(h_opt) == target_sum)
    
    # Create the correlation matrix in frequency domain (approximate)
    # In frequency domain, correlation is convolution, which becomes multiplication
    # We are minimizing over the max of the correlation values

    # Let's instead try to perturb the solution using gradient descent with projected updates
    # Start with current solution, optimize locally
    
    # Define a simple local optimization method
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Simple gradient descent approach with projection
    max_iter = 5000

        if time.time() - start_time >= budget_s - 1:

            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Apply gradient descent
        h_new = h - step_size * grad
        h_new = project_to_constraints(h_new, target_sum)
