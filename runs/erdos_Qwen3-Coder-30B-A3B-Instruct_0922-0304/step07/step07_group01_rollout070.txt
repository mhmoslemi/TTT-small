import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint via projection
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust values to enforce sum constraint
        diff = target_sum - current_sum
        h += diff / len(h)
        # Re-project to ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize again to ensure sum constraint
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

        max_idx = np.argmax(overlaps)
