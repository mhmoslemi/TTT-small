import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize to target sum
        h = h * (target_sum / np.sum(h))
    else:
        # Create smooth step-like initial guess
        h = np.zeros(n_points)
        # Make a smooth transition
        for i in range(n_points):
            x = i * dx
            if x < 1.0:
                h[i] = min(1.0, x * 2.0)
            else:
                h[i] = max(0.0, 2.0 - x * 2.0)
        # Normalize to sum constraint
        h = h * (target_sum / np.sum(h))
    
    # Remez-like optimization in frequency domain
    max_iter = 20
    alpha = 0.1
    decay = 0.95
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        
        # Phase inversion using dominant frequency
        # Create a sinusoidal pattern that targets the maximum overlap
        freq_pattern = np.sin(2 * np.pi * max_lag * np.arange(n_points) / n_points)
        
        # Update h using gradient-like step
        h_new = h + alpha * freq_pattern
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Check if this improves our solution
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        alpha *= decay
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with local optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Try a few local refinements
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (target_sum / np.sum(refined_h))
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
