import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust by scaling
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Scale again to meet sum exactly
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    
    # Differential Evolution parameters
    pop_size = 50
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual = np.clip(individual, 0, 1)
            individual = project_to_constraint(individual, target_sum)
        else:
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = project_to_constraint(individual, target_sum)
        population.append(individual)
        fitness.append(compute_c5(individual, dx))
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        if iterations % 100 == 0:
            # Check if we've found a good solution early
            if best_c5 <= 0.3807:
                break
            
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Ensure trial has correct sum
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate trial
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        iterations += 1
        
        # Adaptive mutation factor
        if iterations % 500 == 0 and iterations > 0:
            mutation_factor = max(0.5, mutation_factor * 0.95)
    
    return (best_h.tolist(), best_c5, n_points)
