import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Adaptive Gaussian Mixture Algorithm (AGMA)
    # Parameters
    N = 100  # Population size
    M = 5    # Number of Gaussian components
    P = 20   # Number of elites to keep
    alpha = 0.99  # Covariance decay factor
    sigma = 0.1   # Initial mutation scale
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    # Add one component based on initial solution
    if 'initial_h_values' in globals():
        means.append(h.copy())
    else:
        means.append(np.random.rand(n_points))
    covariances.append(np.eye(n_points) * sigma)
    weights.append(1.0)
    
    # Add random components for diversity
    for _ in range(M - 1):
        means.append(np.random.rand(n_points))
        covariances.append(np.eye(n_points) * sigma)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iter = 6000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample from GMM
        candidates = []
        for _ in range(N):
            # Choose component
            comp_idx = np.random.choice(M, p=weights)
            sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            candidates.append(sample)
        
        # Process candidates
        processed_candidates = []
        for candidate in candidates:
            # Clip to bounds
            candidate = np.clip(candidate, 0, 1)
            # Normalize sum
            candidate = project_to_constraint(candidate, target_sum)
            # Compute C5
            c5 = compute_c5(candidate, dx)
            processed_candidates.append((candidate, c5))
        
        # Sort by C5 and keep elites
        processed_candidates.sort(key=lambda x: x[1])
        elites = processed_candidates[:P]
