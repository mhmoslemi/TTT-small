import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Parameters for simulated annealing
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-8
    learning_rate = 0.01
    max_iterations = int(budget_s * 1000)  # Rough estimate
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Annealing loop
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a perturbation
        h_new = h.copy()
        
        # Perturbation with Gaussian step
        perturbation = np.random.normal(0, 0.01, n_points)
        h_new += perturbation * learning_rate
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to target sum
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate the new solution
        c5_new = evaluate_overlap(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new
            c5_new = evaluate_overlap(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        
        # Update temperature
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        # Adjust learning rate if needed
        if T < 0.1:
            learning_rate = 0.005
    
    # Final refinement with gradient-based method
    if time.time() - start_time < budget_s - 2:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        # Simple local refinement with small steps
        h_refined = best_h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s - 2:
                break
            h_trial = h_refined.copy()
            perturb = np.random.normal(0, 0.005, n_points)
            h_trial += perturb
            h_trial = np.clip(h_trial, 0, 1)
            h_trial_sum = np.sum(h_trial)
            if h_trial_sum > 0:
                h_trial = h_trial * (target_sum / h_trial_sum)
            c5_trial = evaluate_overlap(h_trial, dx)
            if c5_trial < objective(h_refined):
                h_refined = h_trial
                best_c5 = c5_trial
        
        best_h = h_refined
    
    # Ensure final solution is valid
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
