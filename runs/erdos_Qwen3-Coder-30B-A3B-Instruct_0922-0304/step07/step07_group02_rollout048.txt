import numpy as np
from scipy.signal import correlate
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize components
    means = [h.copy()]
    covariances = [np.eye(n_points) * initial_sigma]
    weights = [1.0]
    
    # Add random components for diversity
    for _ in range(n_components - 1):
        comp_h = np.random.rand(n_points)
        comp_h = np.clip(comp_h, 0, 1)
        comp_h = normalize_h(comp_h, target_sum)
        means.append(comp_h)
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # AGMA main loop
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < 6000:
        # Sample from GMM
        samples = []
        sample_weights = []
        
        for i in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            # Sample from component
            sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            # Clip to bounds
            sample = np.clip(sample, 0, 1)
            # Normalize sum
            sample = project_to_constraint(sample, target_sum)
            samples.append(sample)
            sample_weights.append(weights[comp_idx])
        
        # Evaluate samples
        c5_values = []
        for s in samples:
            c5 = compute_c5(s, dx)
            c5_values.append(c5)
        
        # Sort samples by C5 value
        sorted_indices = np.argsort(c5_values)
        top_samples = [samples[i] for i in sorted_indices[:keep_top]]
        top_values = [c5_values[i] for i in sorted_indices[:keep_top]]
        
        # Update best
        for i in range(len(top_samples)):
            if top_values[i] < best_c5:
                best_h = top_samples[i].copy()
                best_c5 = top_values[i]
        
        # Update GMM components
        new_means = []
        new_covariances = []
        new_weights = []
        
        # For each component, fit Gaussian from top samples
        for i in range(n_components):
            # Get random subset of top samples (for diversity)
            subset_indices = np.random.choice(len(top_samples), 
                                            size=min(len(top_samples), 30),
                                            replace=False)
            subset = [top_samples[j] for j in subset_indices]
            
            if len(subset) == 0:
                # Keep same component if no samples
                new_means.append(means[i].copy())
                new_covariances.append(covariances[i].copy())
            else:
                # Compute mean
                subset_array = np.array(subset)
                new_mean = np.mean(subset_array, axis=0)
                # Compute covariance
                centered = subset_array - new_mean
                if len(subset) > 1:
                    new_cov = np.cov(centered.T)
                else:
                    new_cov = np.eye(n_points) * initial_sigma
                # Regularize covariance
                new_cov = new_cov + np.eye(n_points) * 1e-10
                new_means.append(new_mean)
                new_covariances.append(new_cov)
        
        # Update weights (uniform)
        new_weights = [1.0] * n_components
        new_weights = np.array(new_weights)
        new_weights = new_weights / np.sum(new_weights)
        
        # Apply covariance decay
        for i in range(n_components):
            covariances[i] = covariances[i] * covariance_decay
        
        # Update parameters
        means = new_means
        covariances = new_covariances
        weights = new_weights
        
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
