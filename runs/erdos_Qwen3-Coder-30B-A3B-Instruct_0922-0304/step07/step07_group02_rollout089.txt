import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
        population.append(particle)
        fitness.append(compute_c5(particle, dx))
    
    # Find best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]

    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
            
            # Project to constraint
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate trial
            trial_fitness = compute_c5(trial, dx)
            
            # Selection
            if trial_fitness < fitness[i]:
                population[i] = trial
                fitness[i] = trial_fitness
                
                # Update global best
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_h = trial.copy()
        
        iterations += 1
        
        # Periodic budget check
        if iterations % 100 == 0 and best_fitness <= 0.3807:
            break
    
    return (best_h.tolist(), best_fitness, n_points)
