import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Frank-Wolfe style coordinate updates
    max_iter = int(1e6)
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        
        # Find maximum overlap index
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # If we've found our target, return early
        if max_val <= 0.3807:
            break
            
        # Calculate gradient estimates for each coordinate
        # For h[i], gradient is approximately d/dh[i] of overlap at max_idx
        # Using finite differences or estimate based on pattern
        # We'll recompute dC5/dh[i] for candidate points
        gradients = np.zeros(n_points)
        for i in range(n_points):
            tmp_h = h.copy()
            tmp_h[i] = min(1.0, max(0.0, h[i] + 0.01))
            # Renormalize
            tmp_h = normalize_h(tmp_h, target_sum)
            # Clip
            tmp_h = np.clip(tmp_h, 0, 1)
            new_c5 = compute_c5(tmp_h, dx)
            gradients[i] = (new_c5 - max_val) / 0.01
        
        # Select top 10% coordinates to update based on gradient magnitude
        idxs = np.argsort(np.abs(gradients))[::-1][:max(1, n_points//10)]
        
        # Update chosen coordinates
        for i in idxs:
            # Estimate step size
            grad = gradients[i]
            if abs(grad) > 1e-10:
                # Approximate Lipschitz constant from earlier changes
                # For simplicity, just use a reasonable adaptive step size
                alpha = -grad / (1.0 + abs(grad) * 0.1)
                # Ensure alpha keeps h[i] within [0,1]
                alpha = np.clip(alpha, -h[i], 1-h[i])
                
                # Update
                h[i] += alpha
                # Keep within bounds
                h[i] = np.clip(h[i], 0, 1)
                
                # Renormalize
                h = normalize_h(h, target_sum)
                
                # Check if this improved the solution
                current_c5 = compute_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            
        # Early termination if no improvement for too many iterations
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
