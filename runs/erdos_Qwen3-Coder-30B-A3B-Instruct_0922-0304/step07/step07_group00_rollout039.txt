import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def create_basis_functions(n_points, n_basis=24):
    """Create a set of basis functions (step functions) for convex combination."""
    basis = []
    # Create step functions with different pulse widths and positions
    for i in range(n_basis):
        h = np.zeros(n_points)
        # Determine pulse properties
        width = np.random.uniform(0.1, 0.5)  # 0.1 to 0.5 of domain
        start = np.random.uniform(0, 2 - width)
        end = start + width
        # Map to discrete indices
        start_idx = int(start / (2.0 / n_points))
        end_idx = int(end / (2.0 / n_points))
        end_idx = min(end_idx, n_points)
        h[start_idx:end_idx] = 1.0
        # Normalize so integral is 1/(n_points/2)
        integral = np.sum(h) * (2.0 / n_points)
        if integral > 0:
            h = h / integral * (n_points / 2)
        basis.append(h)
    return np.array(basis)

def project_to_feasible(h, target_sum):
    """Project h to satisfy integral constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate basis functions
    basis_functions = create_basis_functions(n_points, 24)
    n_basis = basis_functions.shape[0]
    
    # Initial solution: use existing if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Project to satisfy constraints
        initial_h = project_to_feasible(initial_h, target_sum)
        # Project into basis space
        # We use a simple least squares approach to initialize λ
        A = basis_functions.T
        b = initial_h
        try:
            # Solve least squares in the basis
            λ_solution = np.linalg.lstsq(A, b, rcond=None)[0]
            # Apply constraints to λ
            λ_solution = np.maximum(λ_solution, 0)
            total = np.sum(λ_solution)
            if total > 0:
                λ_solution = λ_solution / total
            else:
                λ_solution = np.ones(n_basis) / n_basis
        except:
            λ_solution = np.ones(n_basis) / n_basis
    else:
        λ_solution = np.ones(n_basis) / n_basis

    # Precompute overlap integrals A_ij^k 
    # This can be done smarter but for now we do it in the loop,
    # predicting at most 1000 steps (suffices for k from -1.99 to 1.99)
    # Consider a simpler approach - just compute for current set
    best_c5 = float('inf')
    best_h = None
    iteration = 0
    stagnation_count = 0
    prev_best = float('inf')

    while time.time() - start_time < budget_s - 1:
        # Construct h from current λ
        h = basis_functions.T @ λ_solution
        h = project_to_feasible(h, target_sum)
        
        # Compute C5
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if current_c5 <= 0.3807:
            break

        if stagnation_count > 50:
            # Add new basis functions near the worst k values
            # This is a first approximation to local basis expansion
            overlap = correlate(h, 1-h, mode="full")
            max_overlap_idx = np.argmax(overlap)
            max_k = max_overlap_idx - (len(h)*2 - 1) // 2  # convert index to k
            # Simplified: just add some perturbed functions
            new_basis = create_basis_functions(n_points, 4)  # add 4 new ones
            basis_functions = np.vstack([basis_functions, new_basis])
            n_basis = basis_functions.shape[0]
            # Re-initialize λ with more available basis functions
            current_λ = np.concatenate([λ_solution, np.zeros(4)])
            current_λ = current_λ / np.sum(current_λ)
            λ_solution = current_λ
            # Reset stagnation count
            stagnation_count = 0

        # Solve for improved λ (convex approximation)
        # This is a bit tricky since it's not straightforward to have both
        # quadratic and linear terms resulting from h(x)(1-h(x+k)).
        # Instead, we will use gradient ascent on a smooth approximation or
        # enumerate over large regions in function space.
        # Simplified heuristic: make small random change to λ
        # to allow global exploration.
        
        # Introduce small random perturbation to λ  
        noise = np.random.normal(0, 0.001, n_basis)
        λ_rest = λ_solution + noise
        λ_rest = np.maximum(λ_rest, 0)
        total = np.sum(λ_rest)
        if total > 0:
            λ_solution = λ_rest / total

        iteration += 1
        
        # Periodic basis refinement
        if iteration % 20 == 0 and stagnation_count < 10:
            # Randomly add a few new basis functions
            # Basically a heuristic to prevent getting stuck
            new_basis = create_basis_functions(n_points, 2)
            basis_functions = np.vstack([basis_functions, new_basis])
            n_basis = basis_functions.shape[0]
            # Extend λ to new dimension
            current_λ = np.concatenate([λ_solution, np.zeros(2)])
            current_λ = current_λ / np.sum(current_λ)
            λ_solution = current_λ

    # Final computation
    h = basis_functions.T @ λ_solution
    h = project_to_feasible(h, target_sum)
    final_c5 = compute_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
