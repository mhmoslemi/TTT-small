import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, max_iter=1000):
    """Perform local optimization using L-BFGS-B."""
    def objective(x):
        # Renormalize
        x = normalize_h(x, target_sum)
        return compute_c5(x, dx)
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    # Ensure bounds
    bounds = [(0, 1) for _ in range(len(h))]
    
    # Add constraint
    constraints = {'type': 'eq', 'fun': constraint_fun}
    
    # Optimize
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      constraints=constraints, options={'maxiter': max_iter})
    
    # Ensure feasibility
    h_new = normalize_h(result.x, target_sum)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # Parameters for Evolutionary Strategy
    population_size = 20
    elite_size = population_size // 2
    sigma_start = 0.1
    sigma_end = 0.01
    max_generations = 2000
    
    # Initialize population
    population = [h.copy()]
    for i in range(population_size - 1):
        pert = np.random.normal(0, sigma_start, n_points)
        new_h = h + pert
        new_h = project_to_constraints(new_h, target_sum)
        population.append(new_h)
    
    # Main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            fit = compute_c5(individual, dx)
            fitness.append(fit)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        best_idx = 0
        best_solution = population[best_idx]
        best_fitness = fitness[best_idx]
        
        # Early termination
        if best_fitness <= 0.3807:
            break
            
        # Local optimization for elite
        for i in range(elite_size):
            if fitness[i] < 0.381:
                optimized = local_optimize(population[i], target_sum, dx)
                fit_opt = compute_c5(optimized, dx)
                if fit_opt < fitness[i]:
                    population[i] = optimized
                    fitness[i] = fit_opt
        
        # Create new generation with mutation
        new_population = [population[0].copy()]
        sigma = sigma_start - (sigma_start - sigma_end) * generation / max_generations
        
        for i in range(1, population_size):
            # Select parent from top half
            parent_idx = np.random.randint(0, elite_size)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, sigma, n_points)
            child = parent + noise
            
            # Project to constraints
            child = project_to_constraints(child, target_sum)
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    final_fitness = compute_c5(best_solution, dx)
    
    return (best_solution.tolist(), final_fitness, n_points)
