import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Lagrangian-based optimization
    max_iter = 1000
    alpha = 0.01
    gamma = 0.1
    tol = 1e-6
    stagnation_count = 0
    max_stagnation = 20
    
    h_current = h.copy()
    mu = 1.0
    lambda_plus = np.zeros(n_points)
    lambda_minus = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify maximizing shift
        overlap_vals = np.zeros(2*n_points - 1)
        for k in range(1 - n_points, n_points):
            if k >= 0:
                shifted_h = np.concatenate([h_current[k:], h_current[:k]])
            else:
                shifted_h = np.concatenate([h_current[-k:], h_current[:-k]])
            overlap_vals[k + n_points - 1] = np.sum(h_current * (1 - shifted_h))
        
        max_k = np.argmax(overlap_vals)
        max_shift = max_k - n_points + 1
        
        # Compute gradient
        if max_shift >= 0:
            shifted_h = np.concatenate([h_current[max_shift:], h_current[:max_shift]])
        else:
            shifted_h = np.concatenate([h_current[-max_shift:], h_current[:-max_shift]])
            
        # Gradient components
        grad = (1 - shifted_h) - mu + lambda_plus - lambda_minus
        
        # Update h
        h_new = h_current - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Update Lagrange multipliers
        mu = max(0, mu + gamma * (np.sum(h_new) - target_sum))
        lambda_plus = np.maximum(0, lambda_plus + gamma * (h_new - 1))
        lambda_minus = np.maximum(0, lambda_minus + gamma * (-h_new))
        
        # Normalize
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Check convergence
        diff = np.sum(np.abs(h_new - h_current))
        if diff < tol:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count >= max_stagnation:
            break
            
        h_current = h_new
        current_c5 = evaluate_overlap(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
        
        # Adaptive step size
        if diff > 0.01:
            alpha = min(0.1, alpha * 1.1)
        elif diff < 1e-4:
            alpha *= 0.9
    
    # Refinement with local optimization
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
