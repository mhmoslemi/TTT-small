import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def threshold_step_function(h_continuous, threshold=0.5):
    # Simple thresholding to get step function
    h_step = (h_continuous > threshold).astype(float)
    return h_step

def gradient_descent_refinement(h, target_sum, dx, max_iter=1000, lr=0.01):
    h = h.copy()
    n_points = len(h)
    for i in range(max_iter):
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx]
        
        # approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[j] = (overlap_plus[max_overlap_idx] - max_overlap) / eps

        # Update using gradient
        h -= lr * grad
        h = project_to_constraints(h, target_sum)
        
        if i % 100 == 0:
            c5 = compute_c5(h, dx)
            if c5 <= 0.3807:
                break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_constraints(h, target_sum)
    else:
        h = np.zeros(n_points)
        # Set up a step-like function
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation: formulate as problem with continuous h
    h_cont = cp.Variable(n_points)
    C = cp.Variable()
    
    constraints = [
        0 <= h_cont,
        h_cont <= 1,
        cp.sum(h_cont) == target_sum,
        C >= 0
    ]
    
    # Add constraints for maximum overlap in the convex form
    # We'll do this by discretizing the full overlap computation in the convex solver
    
    # Materialize overlaps and construct convex constraints
    # Constraint over a limited subset of shifts for efficiency
    shifts_to_check = list(range(1, n_points))  # We only care about non-zero shifts
    for shift in shifts_to_check:
        # For shift k, consider the correlation
        h_shifted = cp.vstack([cp.Constant(0) for _ in range(shift)] + [h_cont[:n_points-shift]]).flatten()
        h_convol = cp.multiply(h_cont, 1 - h_shifted)
        if shift <= n_points:
            max_overlap_idx = shift + n_points - 2
            # Add constraint max(corr) <= C
            constraints.append(cp.sum(h_convol) <= C)

    # Solve convex relaxation
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve the convex relaxation
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_relaxed = h_cont.value
            h_new = threshold_step_function(h_relaxed)
            h_new = project_to_constraints(h_new, target_sum)
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
    except:
        pass

    # Gradient descent refinement on the best solution
    refitted_h = gradient_descent_refinement(best_h, target_sum, dx, max_iter=500, lr=0.005)
    refitted_c5 = compute_c5(refitted_h, dx)
    
    if refitted_c5 < best_c5:
        best_h = refitted_h
        best_c5 = refitted_c5

    # Final resort to local search if budget allows
    if time.time() - start_time < budget_s - 10:
        # Simple local search with random perturbations
        current_h = best_h.copy()
        current_c5 = best_c5
        for i in range(2000):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Perturb a random index
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_trial = current_h.copy()
            h_trial[idx] += delta
            h_trial = project_to_constraints(h_trial, target_sum)
            trial_c5 = compute_c5(h_trial, dx)
            if trial_c5 < current_c5:
                current_h = h_trial
                current_c5 = trial_c5
                if current_c5 <= 0.3807:
                    break

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
