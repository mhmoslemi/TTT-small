import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use existing initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create diverse initial population
        population.append(h.copy())
        for _ in range(population_size // 5):
            variation = h + np.random.normal(0, 0.02, n_points)
            population.append(project_to_feasible(variation, target_sum))
    else:
        # Create random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, target_sum)
            population.append(h.copy())
    
    # Constraint: sum = target_sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # PSO parameters
    inertia = 0.8
    c1 = 1.5
    c2 = 1.5
    w_decay = 0.98
    
    # Initialize particles
    particles = population.copy()
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(population_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    max_iter = 60
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate new fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Decay inertia
        inertia *= w_decay
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
            
        if stagnation_count >= max_stagnation:
            break
    
    # Final refinement using optimization
    def objective(h):
        return compute_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    best_h = global_best.copy()
    
    for i in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
