import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def gradient_step(h, dx, target_sum, step_size=0.01):
    # Compute gradient using finite differences
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    
    # Compute gradient for each point
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_to_constraints(h_plus, target_sum)
        h_minus = project_to_constraints(h_minus, target_sum)
        grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
    
    # Perform gradient step
    h_new = h - step_size * grad
    h_new = project_to_constraints(h_new, target_sum)
    
    return h_new
