import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add variations of initial solution
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h)
    
    # Fill remaining population with random solutions
    while len(particles) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h)
    
    # PSO parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iter = 60
    best_global = None
    best_global_value = float('inf')
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    best_personal = particles[:]
    best_personal_values = [compute_c5(p, dx) for p in particles]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (best_personal[i] - particle)
            social = social_coeff * r2 * (best_global - particle if best_global is not None else np.zeros(n_points))
            velocity = inertia_weight * velocity + cognitive + social
            
            # Update position
            particle = particle + velocity
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Evaluate
            fitness = compute_c5(particle, dx)
            
            # Update personal best
            if fitness < best_personal_values[i]:
                best_personal[i] = particle.copy()
                best_personal_values[i] = fitness
                
                if fitness < best_global_value:
                    best_global = particle.copy()
                    best_global_value = fitness
            
            # Store updated particle and velocity
            particles[i] = particle
            velocities[i] = velocity
        
        # Update inertia weight
        inertia_weight *= 0.98
        
        # Early termination
        if best_global_value <= 0.3807:
            break
    
    # If no best was found, fallback to the initial solution
    if best_global is None:
        best_h = particles[0].copy()
        best_c5 = compute_c5(best_h, dx)
    else:
        best_h = best_global
        best_c5 = best_global_value
    
    # Local refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            current_c5 = compute_c5(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
