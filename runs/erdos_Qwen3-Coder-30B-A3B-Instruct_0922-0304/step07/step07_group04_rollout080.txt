import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Custom genetic algorithm
    pop_size = 100
    max_gens = 1000
    tournament_size = max(2, pop_size // 10)
    mutation_rate = 0.1
    max_no_improvement = 50
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add initial solution
    population.append(h.copy())
    population_fitness.append(best_c5)
    
    # Add random solutions
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand_sum = np.sum(h_rand)
        if h_rand_sum > 0:
            h_rand = h_rand * (target_sum / h_rand_sum)
        population.append(h_rand)
        population_fitness.append(evaluate_overlap(h_rand, dx))
    
    # GA loop
    gen = 0
    no_improvement = 0
    last_best = best_c5
    
    while gen < max_gens and time.time() - start_time < budget_s - 5:
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best solution
        if population_fitness[0] < best_c5:
            best_c5 = population_fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= max_no_improvement:
            break
            
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        new_population_fitness = [population_fitness[0]]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            selected_indices = np.random.choice(pop_size, tournament_size, replace=False)
            selected_fitness = [population_fitness[i] for i in selected_indices]
            parent1_idx = selected_indices[np.argmin(selected_fitness)]
            
            selected_indices = np.random.choice(pop_size, tournament_size, replace=False)
            selected_fitness = [population_fitness[i] for i in selected_indices]
            parent2_idx = selected_indices[np.argmin(selected_fitness)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Blend crossover
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            
            # Normalize sum
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.1
                delta = np.random.normal(0, 0.05, n_points)
                child += delta * mask
                child = np.clip(child, 0, 1)
                
                # Re-normalize sum
                child_sum = np.sum(child)
                if child_sum > 0:
                    child = child * (target_sum / child_sum)
            
            # Evaluate child
            child_fitness = evaluate_overlap(child, dx)
            new_population.append(child)
            new_population_fitness.append(child_fitness)
        
        population = new_population
        population_fitness = new_population_fitness
        
        gen += 1
        
        # Adaptive mutation
        if gen % 100 == 0 and mutation_rate > 0.01:
            mutation_rate *= 0.95
        
        if gen % 100 == 0 and time.time() - start_time > budget_s * 0.8:
            break
    
    # Local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8},
                     callback=lambda x: None)
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
