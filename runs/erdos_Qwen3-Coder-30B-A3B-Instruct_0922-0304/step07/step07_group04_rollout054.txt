import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based optimization
    def compute_gradient(h, dx):
        """Compute gradient of overlap with respect to h."""
        # This is a simplified gradient approximation
        # In practice, we'd compute exact derivatives
        eps = 1e-6
        grad = np.zeros_like(h)
        base_overlap = evaluate_overlap(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum > 0:
                h_minus = h_minus * (target_sum / h_minus_sum)
                
            grad[i] = (evaluate_overlap(h_plus, dx) - evaluate_overlap(h_minus, dx)) / (2 * eps)
        
        return grad
    
    # Main optimization loop
    max_iter = 10000
    learning_rate = 1e-3
    momentum = 0.9
    velocity = np.zeros(n_points)
    prev_grad_norm = 1.0
    stable_count = 0
    lambda_reg = 1e-4
    beta = 0.999
    
    for iteration in range(max_iter):

        # Compute gradient
        grad = compute_gradient(h, dx)
        grad_norm = np.linalg.norm(grad)
        
        # Check for convergence
        if grad_norm < 1e-6:
            break
            
        # Update velocity with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h + velocity
        
        # Apply regularization and project onto feasible set
        h_new = h_new + lambda_reg * (target_sum - np.sum(h_new))
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Accept move if it improves the objective
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            stable_count = 0
            # Increase learning rate slightly if we're good
            learning_rate = min(learning_rate * 1.01, 1e-1)
        else:
            stable_count += 1
            # Reduce learning rate if no progress
            learning_rate *= 0.99
            
        # Update regularization

            lambda_reg = min(lambda_reg * 1.1, 1e-1)
            
        # If stability persists, slow down
        if stable_count > 30:
            break
            
    # Final validation and return
    final_h = np.clip(h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    # Return best so far
    return (final_h.tolist(), final_c5, n_points)
