import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust values to meet sum constraint
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    
    # Scale to maintain exact sum
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, target_sum)
    
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    stagnation_counter = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 3000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        c5_value = max_overlap * dx
        
        # Compute gradient as (1-h) shifted by k_max
        gradient = np.zeros_like(h)
        if k_max < len(h):
            # shift 1-h to the left by k_max positions
            gradient[k_max:] = 1 - h[:-k_max] if k_max > 0 else 1 - h
        else:
            # shift 1-h to the right by (k_max - len(h)) positions
            shift = k_max - len(h) + 1
            gradient[:shift] = 1 - h[shift:]

        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Project to maintain sum
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        new_c5 = compute_c5(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Adapt learning rate if stagnating
        if stagnation_counter > 100:
            learning_rate *= 0.95
            learning_rate = max(1e-5, learning_rate)
            stagnation_counter = 0
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final validate and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
