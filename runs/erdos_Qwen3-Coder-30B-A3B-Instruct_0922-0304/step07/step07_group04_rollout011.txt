import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 100000
    T0 = 1.0
    alpha = 0.999
    eta0 = 0.01
    lambda1 = 100.0
    lambda2 = 100.0
    
    # Initialize from initial_h_values if available

    else:
        h = np.random.uniform(0, 0.5, n_points)

    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Evaluate current solution
    current_c5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing
    T = T0
    eta = eta0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h with Gaussian noise
        h_new = h + np.random.normal(0, eta, n_points)
        
        # Apply bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        new_sum = np.sum(h_new)
        if new_sum > 0:
            h_new = h_new * (target_sum / new_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Compute energy components
        penalty1 = np.sum(np.square(np.clip(h_new, 0, 1) - h_new))
        penalty2 = (np.sum(h_new) - target_sum) ** 2
        E_new = new_c5 + lambda1 * penalty1 + lambda2 * penalty2
        
        penalty1_old = np.sum(np.square(np.clip(h, 0, 1) - h))
        penalty2_old = (np.sum(h) - target_sum) ** 2
        E_old = current_c5 + lambda1 * penalty1_old + lambda2 * penalty2_old
        
        delta_E = E_new - E_old
        
        # Accept or reject
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        # Cooling and update
        T *= alpha
        eta = eta0 * T / T0
    
    # Final gradient-based refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
            
        from scipy.optimize import minimize
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Using the best solution found so far as starting point
        h_refined = best_h.copy()
        
        # Gradient-based optimization
        result = minimize(objective, h_refined, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            refined_sum = np.sum(h_refined)
            if refined_sum > 0:

    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
