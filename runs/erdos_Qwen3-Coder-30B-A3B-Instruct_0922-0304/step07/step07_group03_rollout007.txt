import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, maxiter=1000):
    """Use L-BFGS-B for local optimization."""
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_fun}
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                     constraints=cons, options={'maxiter': maxiter})
    if result.success or result.fun < compute_c5(h, dx):
        return project_to_constraints(result.x, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # ES parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_iter = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected = population[:pop_size//2]
        
        # Update sigma
        sigma = sigma_init - (sigma_init - sigma_final) * iter_num / max_iter
        
        # Generate new population
        new_population = selected.copy()
        for _ in range(pop_size - len(selected)):
            # Pick a parent
            parent = selected[np.random.randint(0, len(selected))]
            # Mutation
            noise = np.random.normal(0, sigma, n_points)
            mutant = parent + noise
            # Project to constraints
            mutant = project_to_constraints(mutant, target_sum)
            # Add to population
            new_population.append(mutant)
            
        # Local optimization for best few candidates
        for i in range(min(5, len(new_population))):
            if fitness[i] < 0.381:
                optimized = local_optimize(new_population[i], target_sum, dx)
                if compute_c5(optimized, dx) < fitness[i]:
                    new_population[i] = optimized
                    # Update best if improved
                    new_c5 = compute_c5(optimized, dx)
                    if new_c5 < best_c5:
                        best_h = optimized.copy()
                        best_c5 = new_c5

        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)
