import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Parameters for adaptive optimization
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize mixture components
    if 'initial_h_values' in globals():
        means = [np.array(initial_h_values)]
        means[0] = np.clip(means[0], 0, 1)
        means[0] = normalize_h(means[0], target_sum)
    else:
        means = [np.random.rand(n_points) for _ in range(n_components)]
        for mean in means:
            mean = np.clip(mean, 0, 1)
            mean = normalize_h(mean, target_sum)
    
    covariances = [np.eye(n_points) * initial_sigma for _ in range(n_components)]
    weights = np.ones(n_components) / n_components
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 0
    max_iter = 6000  # Conservative estimate
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Sample from mixture
        samples = []
        sample_weights = []
        
        for i in range(population_size):
            # Select component
            comp_idx = np.random.choice(n_components, p=weights)
            mean = means[comp_idx]
            cov = covariances[comp_idx]
            
            # Sample from Gaussian
            sample = np.random.multivariate_normal(mean, cov)
            samples.append(sample)
            
        # Process samples
        processed_samples = []
        sample_c5_values = []
        
        for sample in samples:
            # Apply constraints
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            c5 = compute_c5(sample, dx)
            processed_samples.append(sample)
            sample_c5_values.append(c5)
        
        # Select top samples
        top_indices = np.argsort(sample_c5_values)[:keep_top]
        top_samples = [processed_samples[i] for i in top_indices]
        top_c5_values = [sample_c5_values[i] for i in top_indices]
        
        # Update mixture
        # Update means
        new_means = []
        for sample in top_samples:
            new_means.append(sample)
        
        # Blend means and update with weighted average
        means = []
        for i in range(n_components):
            if i < len(top_samples):
                means.append(top_samples[i])
            else:
                # Random component
                means.append(np.random.rand(n_points) * 0.5)
                means[i] = np.clip(means[i], 0, 1)
                means[i] = normalize_h(means[i], target_sum)
        
        # Update covariances using outer products of deviations
        new_covariances = []
        for mean in means:
            # Compute deviations from mean
            deviations = []
            for sample in top_samples:
                deviations.append(sample - mean)
            if len(deviations) == 0:
                # No deviations, use identity with small scale
                cov = np.eye(n_points) * initial_sigma
            else:
                # Compute average outer product
                cov = np.zeros((n_points, n_points))
                for dev in deviations:
                    cov += np.outer(dev, dev)
                cov = cov / len(deviations)
                # Regularize
                cov += np.eye(n_points) * 1e-6
            new_covariances.append(cov)
        
        # Update weights
        new_weights = np.array([np.exp(-c5) for c5 in top_c5_values])
        new_weights = new_weights / np.sum(new_weights)
        
        # Store new parameters
        means = new_means
        covariances = new_covariances
        weights = new_weights
        
        # Apply decay to covariances
        for i in range(len(covariances)):
            covariances[i] *= covariance_decay
        
        # Update best solution
        for i in range(len(top_samples)):
            if top_c5_values[i] < best_c5:
                best_c5 = top_c5_values[i]
                best_h = top_samples[i].copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iterations += 1

    return (best_h.tolist(), best_c5, n_points)
