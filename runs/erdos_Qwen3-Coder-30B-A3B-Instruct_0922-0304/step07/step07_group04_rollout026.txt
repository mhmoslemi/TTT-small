import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    # Precompute indices for correlation
    lags = np.arange(-n_points + 1, n_points)
    shift_indices = []
    for k in lags:
        indices = (np.arange(n_points) - k) % n_points
        shift_indices.append(indices)
    
    # Gradient-based optimization with adaptive regularization
    def compute_gradient(h_vec, c5_val):
        grad = np.zeros_like(h_vec)
        for k, indices in zip(lags, shift_indices):
            # Compute gradient of h[i] * (1 - h[(i-k)%n]) w.r.t. h[i]
            # For each i, this contributes -h[(i-k)%n] to grad[i]
            # We sum over all valid shifts
            for i in range(n_points):
                grad[i] -= h_vec[indices[i]]
        return grad * dx
    
    def objective(h_vec):
        return np.max(correlate(h_vec, 1 - h_vec, mode="full")) * dx
    
    # Adaptive gradient descent
    eta = 1e-3  # initial learning rate
    momentum = 0.9
    v = np.zeros(n_points)
    lambda_reg = 1e-4
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        c5_current = objective(h)
        
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Compute gradient
        grad = compute_gradient(h, c5_current)
        
        # Update with momentum
        v = momentum * v - eta * grad
        h_new = h + v
        
        # Project onto feasible set [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Adaptive regularization to prevent constraint violations
        if np.any(h_new < 0) or np.any(h_new > 1) or abs(np.sum(h_new) - target_sum) > 1e-6:
            lambda_reg *= 10
            # Apply penalty to prevent violations
            penalty = lambda_reg * (np.sum(np.maximum(0, h_new - 1)) + 
                                  np.sum(np.maximum(0, -h_new)) + 
                                  abs(np.sum(h_new) - target_sum))
            grad = grad + penalty / (dx * n_points)
            h_new = h - eta * grad
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            if current_sum > 0:
                h_new = h_new * (target_sum / current_sum)
        
        # Update h
        h = h_new
        
        # Anneal learning rate
        if iteration % 100 == 0:
            eta *= 0.95
        
        # Stop if no improvement in a while
        if iteration % 50 == 0:
            if abs(c5_current - objective(h)) < 1e-8:
                break
    
    # Final validation
    final_h = best_h.copy()
    final_c5 = best_c5
    # Normalize one more time to ensure constraint
    if np.sum(final_h) > 0:
        final_h = final_h * (target_sum / np.sum(final_h))
    
    final_c5 = np.max(correlate(final_h, 1 - final_h, mode="full")) * dx
    
    return (final_h.tolist(), final_c5, n_points)
