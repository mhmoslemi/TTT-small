import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Optimization parameters
    alpha = 0.1  # learning rate
    beta = 0.9   # momentum
    lambda_reg = 1e-4  # regularization
    max_iter = 3000
    
    # Adam-style optimization
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_since_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using finite differences
        epsilon = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_to_constraint(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = project_to_constraint(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
            
        # Adam update
        m = beta * m + (1 - beta) * grad
        v = beta * v + (1 - beta) * grad**2
        
        # Correct bias
        m_hat = m / (1 - beta**(iteration + 1))
        v_hat = v / (1 - beta**(iteration + 1))
        
        # Update h_relaxed (without bounds)
        h_relaxed = h - alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Add regularization term
        h_relaxed = h_relaxed - alpha * lambda_reg * (h_relaxed - h)
        
        # Project to feasible region
        h = np.clip(h_relaxed, 0, 1)
        h = project_to_constraint(h, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Adaptive learning rate and regularization
        if iter_since_improvement > 50:
            alpha *= 0.9
            iter_since_improvement = 0
            
        if iter_since_improvement > 100:
            lambda_reg *= 1.2
            
        if iter_since_improvement > 500:
            lambda_reg *= 0.8
            
        if new_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
