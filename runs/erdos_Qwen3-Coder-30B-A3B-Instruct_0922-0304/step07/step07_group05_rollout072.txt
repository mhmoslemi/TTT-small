import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a triangular pulse
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / (mid - 1)
            h[n_points - 1 - i] = i / (mid - 1)
        h[mid] = 1.0
    
    # Normalize to desired sum
    h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iterative quadratic approximation approach
    max_iterations = 60
    patience = 10
    patience_counter = 0
    prev_best_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation
        # For each point, compute gradient direction by finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        # Evaluate at current point
        current_c5 = compute_c5(h, dx)
        
        # Compute gradient using finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure feasibility
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Compute Hessian approximation (simplified: using diagonal)
        hessian_diag = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure feasibility
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            c5_current = current_c5
            
            hessian_diag[i] = (c5_plus + c5_minus - 2 * c5_current) / (eps**2)
        
        # Quadratic programming: minimize 0.5 * d^T * H * d + grad^T * d
        # subject to: sum(d) = 0, 0 <= h + d <= 1, 0 <= h <= 1
        
        d = cp.Variable(n_points)
        
        # Quadratic objective
        objective = cp.Minimize(0.5 * cp.quad_form(d, np.diag(hessian_diag)) + grad.T @ d)
        
        # Constraints
        constraints = [
            cp.sum(d) == 0,
        ]
        
        # Box constraints
        for i in range(n_points):
            constraints.append(d[i] >= -h[i])
            constraints.append(d[i] <= 1.0 - h[i])
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                update_direction = d.value
            else:
                # Fallback to gradient descent
                update_direction = -grad
        except:
            # Fallback to gradient descent
            update_direction = -grad
            
        # Line search to find step size
        alpha = 1.0
        max_alpha = 1.0
        min_alpha = 1e-10
        
        # Try different step sizes for descent direction
        for _ in range(20):
            candidate_h = h + alpha * update_direction
            candidate_h = project_to_feasible(candidate_h, target_sum)
            candidate_c5 = compute_c5(candidate_h, dx)
            
            if candidate_c5 < current_c5:
                h = candidate_h
                current_c5 = candidate_c5
                break
            else:
                alpha *= 0.5
                if alpha < min_alpha:
                    break
        
        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if patience_counter >= patience:
            break
    
    # Final refinement using local optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    result = minimize(objective, best_h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 50})
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = project_to_feasible(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
