import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a smooth step-like function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, min(1, 2 * (i - mid + 10) / 20.0))
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    
    # Iterative refinement with frequency-domain analysis
    max_iterations = 20
    alpha = 0.1  # Update strength
    decay = 0.95  # Decay factor for alpha
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(best_h, 1 - best_h, mode="full")
        
        # Find the lag with maximum overlap
        max_lag_idx = np.argmax(overlap)
        max_lag = max_lag_idx - (len(overlap) // 2)
        
        # Create perturbation signal (sinusoidal)
        if max_lag != 0:
            freq = 2 * np.pi * max_lag / n_points
            perturbation = np.sin(freq * np.arange(n_points))
        else:
            # If no dominant lag, use a simple perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
        
        # Update h
        new_h = best_h + alpha * perturbation
        new_h = project_to_feasible(new_h, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(new_h, dx)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        
        # Reduce update strength
        alpha *= decay
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
