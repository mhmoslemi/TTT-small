import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Simple projection: adjust all values equally
    h = np.clip(h + diff / len(h), 0, 1)
    # Re-scale to fix sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 3000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Initialize with uniform values
        h = np.ones(n_points) * 0.5
        h = project_to_constraints(h, target_sum)
    
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Track best solution
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    iter_since_improvement = 0
    adaptive_lr = learning_rate
    
    for iteration in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts and find the maximizing one
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient: (1 - h) shifted by k_max
        # In the correlation, we're computing h(x) * (1 - h(x+k_max))
        # The gradient of this would be (1 - h) shifted by k_max
        grad_h = np.zeros_like(h)
        if k_max < len(h):
            grad_h[k_max:] = 1 - h[:-k_max]
        else:
            grad_h[:len(h) - (k_max - len(h) + 1)] = 1 - h[k_max - len(h) + 1:]
            
        # Update with momentum
        velocity = momentum * velocity - adaptive_lr * grad_h
        h += velocity
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Re-project to sum constraint
        h = project_to_constraints(h, target_sum)
        
        # Evaluate
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Adjust learning rate
        if iter_since_improvement > 100:
            adaptive_lr *= 0.95
            if adaptive_lr < 1e-5:
                adaptive_lr = 1e-5
            iter_since_improvement = 0
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
