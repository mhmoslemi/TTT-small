import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_basis_functions(n_points, n_basis=20):
    """Create basis functions for representing h(x)"""
    basis = []
    # Create step functions with various widths and positions
    for i in range(n_basis):
        # Random width and position
        width = np.random.uniform(0.1, 0.5)
        pos = np.random.uniform(0, 2 - width)
        h = np.zeros(n_points)
        start_idx = int(pos / (2.0 / n_points))
        end_idx = int((pos + width) / (2.0 / n_points))
        if end_idx > start_idx:
            h[start_idx:end_idx] = 1.0
        basis.append(h)
    
    # Normalize each basis function to have same integral
    total_integral = np.sum(basis[0]) * (2.0 / n_points)
    for i in range(len(basis)):
        basis[i] = normalize_h(basis[i], total_integral)
    
    return np.array(basis)

def compute_overlap_integrals(basis):
    """Precompute overlap integrals A_ij^k"""
    n_basis = len(basis)
    n_points = len(basis[0])
    dx = 2.0 / n_points
    
    # For each basis function pair, compute all k overlaps
    A = np.zeros((n_basis, n_basis, 2 * n_points - 1))
    
    for i in range(n_basis):
        for j in range(n_basis):
            # Convolve basis[i] with 1 - basis[j]
            conv_result = correlate(basis[i], 1 - basis[j], mode="full")
            A[i, j] = conv_result * dx
            
    return A

    # Let's compute key terms and add constraints one at a time
    
    return lam, constraints

def refit_to_basis(h, basis, target_sum):
    """Refit h to the basis representation"""
    dx = 2.0 / len(h)
    n_points = len(h)
    
    # Solve basis representation
    lam = cp.Variable(len(basis))
    constraints = [cp.sum(lam) == 1, lam >= 0]
    
    # Objective: minimize || basis * lam - h ||_2^2
    objective = cp.Minimize(cp.sum_squares(basis @ lam - h))
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return lam.value, True
    except:
        pass
    
    # Fall back to simple projection
    weights = np.dot(basis, h) / np.sum(basis * basis, axis=1)
    weights = np.maximum(weights, 0)
    weights = weights / np.sum(weights)
    return weights, False

def refine_solution(h, basis, A, target_sum, dx):

    objective_coeffs = []

    sample_ks = np.linspace(0, n_overlaps-1, min(50, n_overlaps), dtype=int)
