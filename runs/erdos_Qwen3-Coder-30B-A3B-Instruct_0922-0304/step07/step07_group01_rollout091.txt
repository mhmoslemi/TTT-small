import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided starting point or fallback
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_constraints(h, target_sum)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Initialize with current best if available
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simple heuristic optimization: random search with local perturbations
    n_iter = 0
    max_iter = 10000
    alpha = 0.05  # perturbation magnitude
    
    while time.time() - start_time < budget_s - 1 and n_iter < max_iter:
        # Try a random perturbation
        h_pert = h + np.random.normal(0, alpha, n_points)
        h_pert = project_to_constraints(h_pert, target_sum)
        
        c5_new = compute_c5(h_pert, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_pert.copy()
            h = h_pert.copy()
            
        # Occasionally reduce perturbation
        if n_iter % 100 == 0:
            alpha *= 0.995
            
        # Early termination if target hit
        if best_c5 <= 0.3807:
            break
            
        n_iter += 1
        
    return (best_h.tolist(), best_c5, n_points)
