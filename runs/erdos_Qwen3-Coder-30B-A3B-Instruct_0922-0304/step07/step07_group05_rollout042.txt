import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    else:
        # Create smooth step function as initial guess
        h = np.zeros(n_points)
        # Model a step function with smooth transition
        mid = n_points // 2
        h[:mid] = 1
        # Add some smoothness to transition
        for i in range(mid, min(mid + 10, n_points)):
            h[i] = 1.0 - (i - mid + 1) / 10.0 if i - mid + 1 <= 10 else 0
        # Normalize to sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Phase inversion method with frequency domain updates
    max_iterations = 50
    alpha = 0.1
    decay = 0.95
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_lag = max_idx - (len(h) - 1)
        
        # Create frequency domain representation
        H = np.fft.fft(h)
        # Create a sinusoidal phase modification to reduce the maximum
        freq_domain_mod = np.zeros_like(H)
        # Find dominant frequency (approximate)
        if abs(max_lag) < len(h):
            k = max_lag
            # Apply inverse phase modification
            freq_domain_mod = np.exp(1j * 2 * np.pi * k * np.arange(len(h)) / len(h))
        
        # Update h in frequency domain
        H_mod = H * freq_domain_mod
        h_new = np.real(np.fft.ifft(H_mod))
        
        # Adjust by scaling to ensure the target sum
        h_new = h_new + alpha * (1 - np.mean(h_new)) * (h_new > 0.5) - alpha * (1 - np.mean(h_new)) * (h_new <= 0.5)
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize to sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate new solution
        candidate_c5 = compute_c5(h_new, dx)
        
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h_new.copy()
            
            # Early stopping if target is met
            if best_c5 <= 0.3807:
                break
        
        h = h_new.copy()
        alpha *= decay
    
    return (best_h.tolist(), best_c5, n_points)
