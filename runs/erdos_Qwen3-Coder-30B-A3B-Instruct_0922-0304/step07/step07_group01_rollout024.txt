import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Optimization using scipy minimize
    def objective(h_vec):
        h = h_vec.copy()
        h = project_to_constraints(h, target_sum)
        return compute_c5(h, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def constraint_bounds(h_vec):
        return np.hstack([h_vec, 1 - h_vec])
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try multiple random starting points
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for i in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Localize around current best
        h_trial = best_h + np.random.normal(0, 0.02, n_points)
        h_trial = project_to_constraints(h_trial, target_sum)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 100})
            if not res.success:
                continue
            c5_trial = compute_c5(res.x, dx)
            if c5_trial < best_c5:
                best_h = res.x.copy()
                best_c5 = c5_trial
        except:
            continue
    
    # Additional refinement using gradient descent
    h_refined = best_h.copy()
    lr = 0.01
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient estimation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_to_constraints(h_minus, target_sum)
            
            grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
        
        # Update with gradient descent
        h_refined -= lr * grad
        h_refined = project_to_constraints(h_refined, target_sum)
        
        # Adaptive learning rate
        lr *= 0.9995
    
    final_c5 = compute_c5(h_refined, dx)
    
    return (h_refined.tolist(), final_c5, n_points)
