import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        scale = target_sum / current_sum
        h = h * scale
    else:
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # ES parameters
    population_size = 20
    sigma_initial = 0.1
    sigma_final = 0.01
    max_generations = 2000
    
    # Generate initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, sigma_initial, n_points)
        candidate = h + perturbation
        candidate = project_to_constraints(candidate, target_sum)
        population.append(candidate)
    
    best_individual = h.copy()
    best_c5 = compute_c5(best_individual, dx)
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Selection (keep top half)
        sorted_indices = np.argsort(fitness)
        selected_indices = sorted_indices[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        selected_fitness = [fitness[i] for i in selected_indices]
        
        # Update best
        current_best_idx = np.argmin(selected_fitness)
        if selected_fitness[current_best_idx] < best_c5:
            best_individual = selected_population[current_best_idx].copy()
            best_c5 = selected_fitness[current_best_idx]
        
        # Local optimization phase
        if best_c5 <= 0.381:
            def objective(x):
                return compute_c5(x, dx)
            
            def constraint_fun(x):
                return np.sum(x) - target_sum
            
            cons = {'type': 'eq', 'fun': constraint_fun}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                result = minimize(objective, best_individual, method='L-BFGS-B', 
                                bounds=bounds, constraints=cons, options={'maxiter': 1000})
                if result.success and result.fun < best_c5:
                    best_individual = result.x.copy()
                    best_c5 = result.fun
            except:
                pass
        
        # Mutation
        new_population = selected_population.copy()
        sigma = sigma_initial - (sigma_initial - sigma_final) * generation / max_generations
        for _ in range(population_size - len(selected_population)):
            parent = selected_population[np.random.choice(len(selected_population))]
            perturbation = np.random.normal(0, sigma, n_points)
            mutant = parent + perturbation
            mutant = project_to_constraints(mutant, target_sum)
            new_population.append(mutant)
        
        population = new_population
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
