import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_constraints(h, target_sum)
    else:
        # Create a block step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Simulated Annealing
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # SA parameters
    initial_temp = 0.1 * current_c5
    temperature = initial_temp
    cooling_rate = 0.99
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Generate a new candidate solution
        # Perturb
        perturbation = np.random.uniform(-temperature, temperature, n_points)
        candidate_h = h + perturbation
        # Project to constraints
        candidate_h = project_to_constraints(candidate_h, target_sum)
        
        # Compute objective for candidate
        candidate_c5 = compute_c5(candidate_h, dx)
        
        # Accept or reject the candidate
        if candidate_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - candidate_c5) / temperature):
            h = candidate_h
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= cooling_rate
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
