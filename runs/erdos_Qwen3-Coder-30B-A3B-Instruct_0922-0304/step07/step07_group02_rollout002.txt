import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        
    # CVXPY approach
    # Variables
    h_vars = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == target_sum
    ]
    
    # Objective: minimize max_k (h ⋅ shift_k(1 - h))
    # We can approximate by adding constraints for a few key shifts
    # Using a more elaborate convex approximation approach
    
    # Try to make convex optimization work
    # We will optimize over a relaxed version and then refine
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try gradient-based refinement
    max_iter = 1000
    step_size = 0.01
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # compute gradient directions
        h_current = best_h.copy()
        c5_current = compute_c5(h_current, dx)
        
        # Simple gradient descent step: perturb h slightly and see improvement
        grad_h = np.zeros(n_points)
        
        # finite diff approximation of gradient (simplified approach)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_current.copy()

            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)

        # apply descent step
        h_new = h_current - step_size * grad_h
        # clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        current_sum_new = np.sum(h_new)
        if current_sum_new > 0:
            h_new = h_new * (target_sum / current_sum_new)
            
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
                
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
