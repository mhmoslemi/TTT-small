import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Distribute the difference evenly
    h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a reasonable initial step function
        h = np.ones(n_points) * 0.5
        
    # PGD parameters
    lr = 0.1
    beta = 0.9
    momentum = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 3000:
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Compute gradient (shifted 1-h)
        # The derivative of the overlap with respect to h at k is (1 - h) shifted by k
        # Let's compute the actual gradient vector
        grad = np.zeros(n_points)
        
        # Extract the relevant part of 1 - h shifted by max_k
        if max_k < n_points:
            # 1-h shifted by max_k position
            shifted = np.concatenate([np.zeros(max_k), 1 - h])
        else:
            # max_k >= n_points, so we shift forward (in the right tail)
            shifted = np.concatenate([1 - h, np.zeros(max_k - n_points)])
            
        shifted = shifted[:n_points]
        grad = shifted
        
        # Update momentum
        momentum = beta * momentum - lr * grad
        # Update h using Nesterov momentum
        h = h + momentum
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        # Project to sum constraint
        h = project_to_sum_constraint(h, target_sum)
        
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if no_improvement_count > 50:
            lr *= 0.95
            lr = max(lr, 1e-5)
            no_improvement_count = 0
            
        iteration += 1
        
        # Early stopping if we hit the target
        if best_c5 <= 0.3807:
            break
    
    # Ensure we return the best solution found
    return (best_h.tolist(), best_c5, n_points)
