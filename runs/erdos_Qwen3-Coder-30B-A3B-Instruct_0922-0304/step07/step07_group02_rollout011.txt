import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values to fix the sum
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values - distribute evenly
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Ensure exact sum by final scaling
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Initialize with a simple step function
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = project_to_constraint(h, target_sum)
    
    # Differential Evolution parameters
    pop_size = 60
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            candidate = np.array(initial_h_values)
            candidate = np.clip(candidate, 0, 1)
            candidate = project_to_constraint(candidate, target_sum)
        else:
            candidate = np.random.rand(n_points)
            candidate = np.clip(candidate, 0, 1)
            candidate = project_to_constraint(candidate, target_sum)
        population.append(candidate)
    
    # Evaluate initial population
    fitness = [compute_c5(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    gbest = population[best_idx].copy()
    gbest_value = fitness[best_idx]
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        for i in range(pop_size):
            # Select three random individuals different from i
            indices = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
            
            # Ensure sum constraint
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate trial
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                population[i] = trial
                fitness[i] = trial_c5
                
                # Update global best if improved
                if trial_c5 < gbest_value:
                    gbest_value = trial_c5
                    gbest = trial.copy()
        
        iterations += 1
        
        # Periodic check for early termination
        if gbest_value <= 0.3807:
            break

    return (gbest.tolist(), gbest_value, n_points)
