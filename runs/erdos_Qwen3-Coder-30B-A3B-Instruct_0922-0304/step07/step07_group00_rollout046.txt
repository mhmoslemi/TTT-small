import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Frank-Wolfe optimization
    max_iter = 5000
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    k_max = -1
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience_counter > patience:
            # Early termination if no improvement
            break
            
        # Build gradient for the critical overlap
        # overlap_k = sum_{i} h[i] * (1 - h[i+k])
        # grad_i = (1 - 2*h[i+k]) if i+k is valid else 0
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = i + k_max
            if 0 <= j < len(h):
                gradient[i] = 1 - 2 * h[j]
        
        # Solve linear program to minimize gradient^T * s
        # subject to: sum(s) = 0, 0 <= s[i] <= 1
        # Use the fact that the optimal s is obtained by setting those components
        # that have the most negative gradient to 1 and others to 0, then adjusting
        # to satisfy the equality constraint
        
        # This is actually a simpler form: choose s that is opposite to gradient
        # but honors constraints. 
        # To satisfy sum(s) = 0 while minimizing gradient^T * s, we want 
        # s = argmin sum_i gradient_i * s_i subject to sum(s) = 0, 0 <= s_i <= 1
        # Simplified: s_i = 1 if gradient_i < 0, and 0 otherwise, 
        # then scale to satisfy sum(s) = 0.
        
        # But to do this properly we'll iterate a bit more over simple analytical method:
        # Start with s_i = 1 if gradient_i < 0, 0 else, adjust to meet sum constraint.
        s = np.zeros_like(h)
        s[gradient < 0] = 1.0
        
        # scale such that sum(s) = 0
        # But for Frank-Wolfe approach, we want direction from h to s
        # so that s = h + alpha * (s - h) where s is the step direction
        # We want to solve: s = argmin_{s} <grad, s> s.t. sum(s) = 0 and 0 <= s <= 1
        # For now let's just explore simple heuristic
        
        # Let's solve the linear subproblem directly:
        # Compute s that minimizes gradient^T * s subject to sum(s) = 0, 
        # and 0 <= s[i] <= 1 for all i.
        # Since sum(s) = 0, it's a constrained optimization. 
        # Best s is to assign 1 to most negative gradient components and 0 to others
        # but we must comply with sum=0.

        # Let's look for s = argmin <gradient, s> subject to sum(s) = 0, 0 <= s[i] <= 1

        # Set s_i = 1 if gradient[i] < 0 and 0 if gradient[i] > 0.
        # This direction minimizes the inner product.

        # h_new = (1 - alpha)*h + alpha * s where s solves linear problem.

        # We must find s to minimize <grad, s> s.t. sum(s) = 0 and 0 <= s <= 1

        s = np.zeros_like(h)
        s[gradient < 0] = 1.0

        # Since all entries are 0 or 1, we compute sum(s)
        current_sum_s = np.sum(s)
        if current_sum_s == 0:
            continue  # no progress
            
        # we want sum(s) = 0. 
        # What if we define the step direction in terms of dual variable?
        # The true Frank Wolfe approach would use either exact line search or backtracking.
        # Let’s perform a simple one-dim line search instead of exact solve:
        alpha = 0.5
        max_alpha = 1.0
        min_alpha = 1e-6
        
        # Try line search:
        best_alpha = 0.0
        best_c5_new = best_c5
        
        for _ in range(10): # shaft search steps
            # Step: h_new = h + alpha * (s - h) 
            h_new = h + alpha * (s - h)
            # Maintain bounds
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_alpha = alpha
            if alpha < min_alpha:
                break
            alpha /= 2.0
            
        alpha = best_alpha
        if alpha > 0:
            h = h + alpha * (s - h)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
    
    # Refine best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
