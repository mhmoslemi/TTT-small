import numpy as np
import time
from scipy.signal import correlate

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available, else create a smooth step
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize to meet constraint
        h = h * (target_sum / np.sum(h))
    else:
        # Create a smooth triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h[i] = i / mid
            else:
                h[i] = 1 - (i - mid) / (n_points - mid)
        # Add small noise for diversity
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iterative refinement using frequency domain analysis
    max_iterations = 50
    alpha = 0.1
    decay_rate = 0.95
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        
        # Use the maximum overlap lag to determine the dominant frequency
        # Create a phase-inversion pattern to reduce that peak
        pattern = np.zeros(n_points)
        # Use the pattern around the max_lag position to identify frequency
        # For simplicity, use a sinusoidal pattern whose frequency corresponds
        # to the structure at max_lag
        if max_lag < n_points:
            # Peak happens for a positive lag
            period = n_points - max_lag
        else:
            # Peak happens at a negative lag, or for zero lag
            period = 1
        
        # Generate sinusoidal pattern for this frequency
        for i in range(n_points):
            pattern[i] = np.sin(2 * np.pi * i / period)
        
        # Invert the pattern if needed to reduce the peak
        if overlap[max_lag] > 0:
            pattern *= 1  # No inversion
        else:
            pattern *= -1  # Invert to reduce peak
        
        # Update h directionally
        h_new = h + alpha * pattern
        
        # Project back to [0,1] and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
        h = h_new.copy()
        alpha *= decay_rate
    
    # Final refinement using local optimization
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Run one final optimization
    try:
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (target_sum / np.sum(final_h))
            final_c5 = compute_c5(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
