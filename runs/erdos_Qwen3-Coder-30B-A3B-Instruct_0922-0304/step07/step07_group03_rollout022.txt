import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit
    deficit = target_sum - current_sum
    if deficit > 0:
        # Increase values
        h += deficit * h / (current_sum + 1e-12)
    else:
        # Decrease values
        h *= target_sum / (current_sum + 1e-12)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Structured initialization
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # AM parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        c = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(c)
        
        # Compute gradient direction: we want to reduce max overlap
        # Gradient is approximately - (1 - h) shifted to align with max overlap
        grad_h = np.zeros_like(h)
        for i in range(len(h)):
            # Simulate contribution to max overlap
            if i < len(h):
                grad_h[i] = -(1 - h[i])
        
        # Update using gradient descent
        h_candidate = h - alpha * grad_h
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Add random perturbation
        perturbation = epsilon * (2 * np.random.rand(n_points) - 1)
        h_candidate += perturbation
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        # Update if better
        if candidate_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = candidate_c5
        
        # Update h
        h = h_candidate.copy()
        
        # Update learning rates
        alpha = 0.2 / (1 + iter_num / tau)
        epsilon = 0.1 / (1 + iter_num / tau)
        
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
