import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial_h_values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Project h to satisfy constraints
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Initial c5 value
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Try different optimizations using cvxpy
    # We'll try a simple convex relaxation approach for better convergence
    try:
        # Create variable for h
        h_var = cp.Variable(n_points)
        
        # Define the constraints
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # For the objective, we need to approximate the maximum correlation
        # We will use a relaxation approach where we bound the maximum
        # We are minimizing max_k (h * shift_k(1-h)) over all shifts k
        # In practice, we take a sample of possible values and optimize a tight upper bound
        
        # Instead of full optimization, we'll do a more guided local search
        # starting from a better configuration
        
        # Try gradient-based optimization with bounds
        from scipy.optimize import minimize
        
        # Objective function to minimize
        def obj_func(h_vec):
            return compute_c5(h_vec, dx)
        
        # Constraints for scipy optimization
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B for bounded optimization
        result = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                         options={'maxiter': 10000, 'ftol': 1e-12})
        
        if not result.success:
            # Fall back to simpler local search if this fails
            h = h.copy()
        else:
            h = result.x
            current_c5 = obj_func(h)
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Allow more exploration for a bit
        # Let's do one final local improvement using gradient descent steps
        # but in small steps to avoid getting stuck in local minima
        step_size = 0.01
        max_iter = 10000
        for i in range(max_iter):
            if time.time() - start_time >= budget_s - 2:
                break
            # Compute gradient direction (approximately)
            grad = np.zeros(n_points)
            # Simple numerical gradient with finite differences
            eps = 1e-6
            current_val = obj_func(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                perturbed_val = obj_func(h_plus)
                grad[j] = (perturbed_val - current_val) / eps
            
            # Update step with projection
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            # Accept step if it improves
            new_val = obj_func(h_new)
            if new_val < current_val:
                h = h_new
                current_val = new_val
                if new_val < best_c5:
                    best_c5 = new_val
                    best_h = h.copy()

    except Exception as e:
        # fallback to simple approach if cvxpy fails
        pass
    
    return (best_h.tolist(), best_c5, n_points)
