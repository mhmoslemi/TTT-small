import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def update_h(h, i, alpha, target_sum):
    """Update h at index i with step size alpha, maintaining sum constraint."""
    h_old = h[i]
    h[i] = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
    scaling = target_sum / np.sum(h)
    h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 100000
    iter_count = 0
    
    # Initialize h
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Use coordinate-wise optimization
    # Sample a small subset of coordinates each iteration
    update_fraction = 0.1
    max_updates = max(1, int(update_fraction * n_points))
    
    # Stopping criteria
    patience = 0
    min_improvement = 1e-6
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        current_max = np.max(overlap)
        
        # Identify indices to update - highest gradient magnitude
        # Gradient in terms of overlap: Δ[i] ≈ overlap[i] * (1 - 2*h[i])
        gradients = overlap * (1 - 2 * h)
        
        # Select indices with highest |gradient| values
        grad_abs = np.abs(gradients)
        update_indices = np.argsort(grad_abs)[-max_updates:]
        
        # Try updating each selected index
        new_h = h.copy()
        improved = False
        
        for i in update_indices:
            # For coordinate i, compute step size to improve overlap
            # Using adaptive learning rate (approximated)
            grad_i = gradients[i]
            if abs(grad_i) < 1e-10:
                continue
                
            # Try a small step in direction of gradient descent
            alpha = min(1.0, max(0, -grad_i / (2.0 + 1e-6)))
            if alpha <= 0:
                continue
            new_h_i = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
            
            # Adjust for sum constraint
            h_delta = new_h_i - h[i]
            new_h = h.copy()
            new_h[i] = new_h_i
            
            # Normalize sum
            scale = target_sum / np.sum(new_h)
            new_h = new_h * scale
            
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < current_max - min_improvement:
                h = new_h.copy()
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                break
        
        if not improved:
            patience += 1
            if patience > 1000:
                break
        else:
            patience = 0
            
        iter_count += 1

            break
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
