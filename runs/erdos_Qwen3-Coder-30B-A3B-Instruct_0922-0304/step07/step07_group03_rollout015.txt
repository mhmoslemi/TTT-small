import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.abs(np.sin(np.pi * np.arange(n_points) / n_points))**2
        h = project_to_constraints(h, target_sum)
    
    # Greedy optimization with adaptive perturbations
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Adaptive parameters
    epsilon_initial = 0.2
    epsilon = epsilon_initial
    decay_factor = 0.99
    min_epsilon = 0.001
    decay_interval = 100
    
    iter_num = 0
    while time.time() - start_time < budget_s - 1:
        if iter_num % decay_interval == 0 and epsilon > min_epsilon:
            epsilon *= decay_factor
        
        # Generate a random perturbation
        direction = np.random.randn(n_points)
        direction = direction / np.linalg.norm(direction)
        
        # Apply perturbation
        h_candidate = best_h + epsilon * direction
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate the candidate
        c5_candidate = compute_c5(h_candidate, dx)
        
        # Accept the candidate if it improves
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            # Reduce epsilon to encourage fine-tuning
            epsilon *= 0.98
        
        iter_num += 1
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
