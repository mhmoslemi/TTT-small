import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Adjust by distributing the difference
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.clip(h, 0, 1)

        # Create a more sophisticated piecewise constant function
        # We'll try to space the steps to reduce overlap
        h[:n_points//3] = 1
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0
        # Add small noise for variety
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Differential Evolution Parameters
    population_size = 60
    mutation_factor = 0.8
    crossover_probability = 0.7
    max_iterations = 3000
    
    # Initialize population
    population = []
    for i in range(population_size):
        if 'initial_h_values' in globals() and i == 0:
            candidate = np.array(initial_h_values)
            candidate = np.clip(candidate, 0, 1)
            candidate = normalize_h(candidate, target_sum)
        else:
            candidate = np.random.rand(n_points)
            candidate = np.clip(candidate, 0, 1)
            candidate = normalize_h(candidate, target_sum)
        population.append(candidate)
    
    # Evaluate initial population
    fitness = []
    for candidate in population:
        fitness.append(compute_c5(candidate, dx))
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    iterations = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        for i in range(population_size):
            # Select three distinct candidates
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_probability, 
                             mutant, population[i])
            
            # Project to constraint
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                population[i] = trial
                fitness[i] = trial_c5

                    best_c5 = trial_c5
                    best_h = trial.copy()

        # Early stopping if no improvement for a while
        if iterations - last_improvement > 500 and best_c5 <= 0.3807:
