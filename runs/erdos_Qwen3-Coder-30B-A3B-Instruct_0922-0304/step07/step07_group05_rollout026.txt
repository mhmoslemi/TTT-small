import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm
    n_particles = 100
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
    else:
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
        h0 += np.random.normal(0, 0.02, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
    
    # Create initial particles
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 60
    stagnation_count = 0
    best_global = None
    best_global_c5 = float('inf')
    
    # Store history
    best_history = []
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i, particle in enumerate(particles):
            # Evaluate current particle
            c5 = compute_c5(particle, dx)
            
            # Update personal best
            if c5 < best_global_c5:
                best_global = particle.copy()
                best_global_c5 = c5
                stagnation_count = 0
                best_history.append((best_global.copy(), best_global_c5))
            else:
                stagnation_count += 1
                
            # Update swarm best if needed
            if len(best_history) > 0:
                if best_global_c5 <= 0.3807:
                    break
            
            # Update particle velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            if best_global is not None:
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (best_global - particle) + 
                                c2 * r2 * (best_global - particle))
                
                particles[i] = particle + velocities[i]
                particles[i] = project_to_feasible(particles[i], target_sum)
        
        # Decay inertia
        w *= 0.98
        
        # Early termination
        if stagnation_count >= 20:
            break
    
    # Final refinement
    best_refined = best_global.copy() if best_global is not None else particles[0]
    best_c5 = compute_c5(best_refined, dx)
    
    # Gradient descent refinement
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_refined.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_refined = result.x
            best_refined = np.clip(best_refined, 0, 1)
            best_refined = normalize_h(best_refined, target_sum)
            new_c5 = compute_c5(best_refined, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break
    
    return (best_refined.tolist(), best_c5, n_points)
