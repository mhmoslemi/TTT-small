import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values with a structured pattern
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    decay_tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlation
        c = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(c)
        
        # Gradient step based on the maximum overlap region
        # Simulate gradient: - (1 - h)[shifted]
        h_candidate = h.copy()
        
        # Create negative gradient w.r.t. the most problematic k
        if max_k >= n_points:
            # At the tail of the correlation, adjust based on 1-h
            shift = max_k - (n_points - 1)
            gradient_term = np.zeros_like(h)
            gradient_term[shift:] = 1 - h[:-shift] 
        else:
            # At the beginning
            shift = max_k
            gradient_term = np.zeros_like(h)
            gradient_term[:n_points-shift] = 1 - h[shift:]
            
        h_candidate = h_candidate - alpha * gradient_term
        
        # Add perturbation
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to constraints
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        # Update if improved
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            # Decrease learning rate
            alpha = alpha / (1 + i / decay_tau)
            epsilon = epsilon / (1 + i / decay_tau)
        else:
            # Revert and decrease learning rate
            alpha = alpha * 0.99
            epsilon = epsilon * 0.99
        
        # Early exit if tight bound achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
