import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Use initial solution as base for some particles
        particles.append(h.copy())
        for _ in range(population_size // 5):
            perturbed = h + np.random.normal(0, 0.01, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = normalize_h(perturbed, target_sum)
            particles.append(perturbed)
        # Fill remaining with random feasible solutions
        while len(particles) < population_size:
            h_rand = np.random.rand(n_points)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand)
    else:
        # Create random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles.append(h)

    # PSO parameters
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_iterations = 60
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (personal_best[i] - particles[i]) + 
                            social * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    
        # Update inertia
        inertia *= 0.98
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Gradient descent refinement
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h, dx)
    
    # Define constraint function
    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Run a few rounds of local refinement
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            new_c5 = compute_c5(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
