import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with diverse candidates
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        # Add diversity to initial solution
        for _ in range(population_size // 5):
            h = base_h + np.random.normal(0, 0.02, n_points)
            h = project_to_feasible(h, target_sum)
            population.append(h)
    else:
        # Create base function
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1
        base_h += np.random.normal(0, 0.02, n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        # Add diverse individuals
        for _ in range(population_size):
            h = base_h + np.random.normal(0, 0.03, n_points)
            h = project_to_feasible(h, target_sum)
            population.append(h)
    
    # Ensure population has exactly population_size individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    population = population[:population_size]
    
    best_h = None
    best_c5 = float('inf')
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    inertia_weight = w_start
    
    # Initialize PSO
    particles = np.array(population)
    velocities = np.random.rand(population_size, n_points) * 0.01
    
    pbest_positions = particles.copy()
    pbest_values = [compute_c5(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    if gbest_value < best_c5:
        best_c5 = gbest_value
        best_h = gbest_position.copy()
    
    # PSO iterations
    max_iterations = 50
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        inertia_weight *= w_decay
        
        for i in range(population_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) + 
                            c2 * r2 * (gbest_position - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project particles to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if fitness < gbest_value:
                    gbest_value = fitness
                    gbest_position = particles[i].copy()
                    stagnation_count = 0
                    
                    if gbest_value < best_c5:
                        best_c5 = gbest_value
                        best_h = gbest_position.copy()
            
            # Early termination
            if gbest_value <= 0.3807:
                break
                
        stagnation_count += 1
        if stagnation_count >= max_stagnation:
            break
    
    # Further refinement with local search
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    if best_h is None:
        # Fallback to best in population
        best_h = population[np.argmin(pbest_values)].copy()
    
    # Apply limited gradient-based refinement
    for _ in range(20):
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50, 'ftol': 1e-6})
        if result.success or not np.isnan(result.fun):
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            current_c5 = compute_c5(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break
    
    # Final validation
    best_h = project_to_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    # Ensure we return a valid result
    if final_c5 >= best_c5:
        best_c5 = final_c5
    else:
        best_h = project_to_feasible(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
