import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # 40% of particles are variations of initial solution
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
        # Fill remaining with random feasible solutions
        for _ in range(population_size - len(particles)):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # 40% of particles are variations of initial solution
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
        # Fill remaining with random feasible solutions
        for _ in range(population_size - len(particles)):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand)

    # PSO Parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iterations = 60

    # Initialize best positions
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]

    # PSO main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_coeff * r1 * (personal_best_positions[i] - particles[i]) +
                social_coeff * r2 * (global_best_position - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            score = compute_c5(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
        
        # Decrease inertia weight
        inertia_weight *= 0.98
        
        # Early termination
        if global_best_score <= 0.3807:
            break

    # Use the best particle as a starting point for local refinement
    best_h = global_best_position.copy()
    
    # Local refinement with gradient descent in a loop
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Simultaneously try multiple local optimization attempts
    best_local_h = best_h.copy()
    best_local_c5 = global_best_score
    
    for attempt in range(3):
        # Add small noise to break symmetry
        if attempt > 0:
            h_noisy = best_h + np.random.normal(0, 0.005, n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
        else:
            h_noisy = best_h.copy()
            
        for _ in range(10):
            result = minimize(objective, h_noisy.copy(), method='SLSQP',
                             bounds=bounds, constraints=constraint,
                             options={'maxiter': 50})
            if result.success:
                h_noisy = result.x
                h_noisy = np.clip(h_noisy, 0, 1)
                h_noisy = normalize_h(h_noisy, target_sum)
                new_score = compute_c5(h_noisy, dx)
                if new_score < best_local_c5:
                    best_local_c5 = new_score
                    best_local_h = h_noisy.copy()
            else:
                break
    
    return (best_local_h.tolist(), best_local_c5, n_points)
