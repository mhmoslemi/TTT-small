import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Try different optimization approaches
    
    # 1. Gradient-based optimization with CVXPY if feasible
    try:
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # We'll relax the max constraint by adding a scalar variable
        # and constraining each overlap term
        k_max = 2 * n_points - 1  # full correlation length
        max_overlap = cp.Variable()
        
        # Define all the correlation terms as constraints
        correlations = []
        for k in range(k_max):
            # For circular shift
            shifted = np.roll(1 - h_var, k)
            corr = cp.sum(cp.multiply(h_var, shifted))
            correlations.append(corr)
        
        # Now add constraint that max_overlap >= all correlations
        for corr in correlations:
            constraints.append(max_overlap >= corr)
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        # Try solving with ECOS, fallback to MOSEK if available
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6, reltol=1e-6)
        except:
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
            except:
                # fallback to other approach if no solver works
                pass
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            h_result = h_var.value
            if h_result is not None:
                h_result = np.clip(h_result, 0, 1)
                h_result = h_result * (target_sum / np.sum(h_result))
                c5_result = compute_c5(h_result, dx)
                if c5_result < best_c5:
                    best_c5 = c5_result
                    best_h = h_result.copy()
    except Exception as e:
        pass  # Continue with other methods if CVXPY fails
    
    # 2. Local optimization with L-BFGS using scipy
    try:
        from scipy.optimize import minimize
        
        def objective(h_vec):
            # Reshape to be consistent
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (target_sum / np.sum(h_vec))
            return compute_c5(h_vec, dx)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - target_sum
        
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # This will be an expensive approach but we'll do a few iterations
        # Let's limit iterations in case it's costly
        if time.time() - start_time < budget_s - 2:
            # Use the best solution so far as a starting point
            x0 = best_h.copy()
            
            # Create a simple local search
            for _ in range(50):
                if time.time() - start_time >= budget_s - 2:
                    break
                try:
                    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-8})
                    if result.success:
                        x0 = result.x
                        c5_result = compute_c5(x0, dx)
                        if c5_result < best_c5:
                            best_c5 = c5_result
                            best_h = x0.copy()
                    break  # Only one optimization attempt to keep things fast
                except:
                    break
                    
    except Exception as e:
        pass  # Continue if L-BFGS fails
    
    # 3. Gradient ascent on worst-case correlation 
    # We'll do a local search if time allows
    if time.time() - start_time < budget_s - 2:
        # Perturb the best solution and check if it improves
        best_h = best_h.copy()
        for iter_num in range(1000):
            if time.time() - start_time >= budget_s - 2:
                break
                
            # Random perturbation with small amplitude
            perturb = np.random.normal(0, 0.01, n_points)
            h_new = best_h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            # Early stopping condition
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
