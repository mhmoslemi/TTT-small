import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing approach for global optimization
    current_h = h.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temperature = 1e-6
    cooling_rate = 0.999
    max_iterations = 10000
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 5 and temperature > min_temperature:
        # Create neighbor solution
        neighbor_h = current_h.copy()
        
        # Perturb randomly
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        
        # Clip and normalize
        neighbor_h = np.clip(neighbor_h, 0, 1)
        neighbor_sum = np.sum(neighbor_h)
        if neighbor_sum > 0:
            neighbor_h = neighbor_h * (target_sum / neighbor_sum)
        
        # Evaluate
        neighbor_c5 = evaluate_overlap(neighbor_h, dx)
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
            no_improvement_count = 0
        else:
            # Accept with probability based on temperature
            delta_e = neighbor_c5 - current_c5
            acceptance_prob = np.exp(-delta_e / temperature) 
            if np.random.random() < acceptance_prob:
                current_h = neighbor_h
                current_c5 = neighbor_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        
        # Decrease temperature
        temperature *= cooling_rate
        
        # Early stopping if no improvement
        if no_improvement_count > max_no_improvement:
            break
            
        iteration += 1
    
    # Final refinement with local search
    if time.time() - start_time < budget_s - 5:
        # Use gradient-free optimizer to refine
        try:
            from scipy.optimize import differential_evolution
            
            def objective(h_vec):
                h = np.clip(h_vec, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (target_sum / h_sum)
                return evaluate_overlap(h, dx)
            
            # Define bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Run differential evolution
            result = differential_evolution(
                objective, 
                bounds, 
                maxiter=100, 
                popsize=15,
                disp=False,
                seed=seed
            )
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
                    
        except:
            pass
            
    # Ensure solution is valid
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
