import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use a simple local search with adaptive steps
    max_evals = 10000
    eval_count = 0
    
    # Generate candidate solutions with perturbations
    while eval_count < max_evals and (time.time() - start_time) < budget_s - 5:
        # Create a perturbed version
        h_new = h.copy()
        # Select a few random indices to modify
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            # Small random change to h[i]
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate
        c5_new = evaluate_overlap(h_new, dx)
        eval_count += 1
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Accept with some probability even if worse (simulated annealing)
        if c5_new < best_c5 or np.random.rand() < 0.01:
            h = h_new.copy()
    
    # Do one final optimization with scipy
    if (time.time() - start_time) < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try a few random starting points
        best_final = best_h.copy()
        best_final_c5 = best_c5
        
        for _ in range(5):
            # Random initialization within bounds
            x0 = np.random.rand(n_points) * 0.5
            x0 = np.clip(x0, 0, 1)
            x0_sum = np.sum(x0)
            if x0_sum > 0:
                x0 = x0 * (target_sum / x0_sum)
            
            result = minimize(objective, x0, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-8})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_final_c5:
                    best_final = refined_h
                    best_final_c5 = final_c5
        
        if best_final_c5 < best_c5:
            best_h = best_final
            best_c5 = best_final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
