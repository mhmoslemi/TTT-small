import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the solution using the provided reference evaluator."""
    # This is the reference implementation that computes C5
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize based on initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a triangular pulse centered with a bit of random noise
        h = np.zeros(n_points)
        center = n_points // 2
        width = n_points // 3
        for i in range(max(0, center - width//2), min(n_points, center + width//2)):
            h[i] = 1.0 - abs(i - center) / (width/2)
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a modified projected gradient method with quadratic approximation
    patience = 0
    max_patience = 10
    max_iterations = 60
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the current maximum overlap index
        overlap = correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Compute gradient using finite differences
        grad = np.zeros_like(best_h)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Normalize
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            # Compute overlap gradient approximation
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_idx]
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_idx]
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        
        # Compute Hessian approximation using finite differences
        # Simple diagonal approximation
        hess_diag = np.zeros_like(best_h)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_idx]
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_idx]
            overlap_center = correlate(best_h, 1 - best_h, mode="full")[max_idx]
            
            hess_diag[i] = (overlap_plus + overlap_minus - 2 * overlap_center) / (epsilon**2)
        
        hess_diag = np.clip(hess_diag, 0.01, 10)
        
        # Formulate and solve QP
        try:
            # Define QP problem
            new_h = cp.Variable(n_points)
            
            # Quadratic term: 0.5 * x^T * H * x
            quad_term = 0.5 * cp.quad_form(new_h, np.diag(hess_diag))
            
            # Linear term: g^T * x
            linear_term = grad @ new_h
            
            # Objective
            objective = cp.Minimize(quad_term + linear_term)
            
            # Constraints
            constraints = [
                cp.sum(new_h) == target_sum,
                new_h >= 0,
                new_h <= 1
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                step = new_h.value - best_h
                
                # Use line search to find the best step size
                alpha = 1.0
                max_alpha = 10.0
                while alpha > 1e-6:
                    candidate_h = best_h + alpha * step
                    # Project back to feasible region
                    candidate_h = np.clip(candidate_h, 0, 1)
                    candidate_h = candidate_h * (target_sum / np.sum(candidate_h))
                    
                    candidate_c5 = evaluate_erdos_solution(candidate_h)
                    
                    if candidate_c5 < best_c5:
                        best_h = candidate_h
                        best_c5 = candidate_c5
                        patience = 0
                        break
                    alpha *= 0.5
                else:
                    patience += 1
            else:
                patience += 1
        except Exception:
            patience += 1
            
        # Early termination
        if patience >= max_patience or best_c5 <= 0.3807:
            break
            
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
