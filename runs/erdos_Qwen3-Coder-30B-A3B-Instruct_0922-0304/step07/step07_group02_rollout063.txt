import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values to meet target sum
    diff = target_sum - current_sum
    # Distribute the difference evenly
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    # Scale to exactly match target sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
        
    # Differential Evolution parameters
    pop_size = 60
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual = np.clip(individual, 0, 1)
            individual = project_to_constraint(individual, target_sum)
        else:
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = project_to_constraint(individual, target_sum)
        population.append(individual)
    
    # Evaluate initial population
    fitness = []
    for individual in population:
        fitness.append(compute_c5(individual, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main evolution loop
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            candidates = list(range(pop_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
            
            # Project to constraint
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
                
                # Update best if needed
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
        population = new_population
        fitness = new_fitness
        
        # Adaptive mutation factor based on progress
        if iterations % 100 == 0 and iterations > 0:
            improvement = np.mean(fitness[:pop_size//2]) - np.mean(fitness[pop_size//2:])
            if improvement < 0.0001:
                mutation_factor = min(1.0, mutation_factor * 1.1)
            else:
                mutation_factor = max(0.5, mutation_factor * 0.95)
                
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
