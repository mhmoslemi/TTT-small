import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try several random restarts
    n_restarts = 10
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:

        # Random perturbation
        h = best_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)

        # Simple gradient descent
        for iter in range(500):
            if time.time() - start_time > budget_s - 1:

            c5 = compute_c5(h, dx)
            # Compute gradient approximations
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):

                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
