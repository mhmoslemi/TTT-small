```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize Gaussians
    means = []
    covariances = []
    weights = []
    
    # First component from initial solution
    if 'initial_h_values' in globals():
        means.append(np.array(initial_h_values))
    else:
        means.append(np.random.rand(n_points))
    covariances.append(np.eye(n_points) * initial_sigma)
    weights.append(1.0)
    
    # Additional random components
    for _ in range(n_components - 1):
        means.append(np.random.rand(n_points))
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
        
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 0
    max_iter = 6000  # Safe limit for 600s budget
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Sample from the mixture
        candidates = []
        for _ in range(population_size):
            # Select component
            comp_idx = np.random.choice(n_components, p=weights)
            # Sample from this component
            sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            # Clip to [0,1]
            sample = np.clip(sample, 0, 1)
            # Project to sum constraint
            sample = project_to_constraint(sample, target_sum)
            candidates.append(sample)
            
        # Evaluate all candidates
        candidate_scores = []
        for candidate in candidates:
            score = compute_c5(candidate, dx)
            candidate_scores.append((candidate, score))
            
        # Sort by score
        candidate_scores.sort(key=lambda x: x[1])
        
        # Keep top candidates
        top_candidates = candidate_scores[:keep_top]
        
        # Update GMM parameters
        if len(top_candidates) > 0:
            # Update means
            new_means = []
            new_covariances = []
            new_weights = []
            
            # Calculate weighted mean of top candidates
            top_h_values = [c[0] for c in top_candidates]
            top_scores = [c[1] for c in top_candidates]
            
            # Use scores as inverse weights (lower is better)
            weights_vec = np.array([1.0 / (1e-10 + s) for s in top_scores])
            weights_vec = weights_vec / np.sum(weights_vec)
            
            # Update means
            mean_update = np.average(top_h_values, axis=0, weights=weights_vec)
            new_means.append(mean_update)
            
            # Update covariance (regularized with minimum)
            if len(top_h_values) > 1:
                # Centered data
                centered_data = np.array(top_h_values) - mean_update
                # For numerical stability
                cov = np.cov(centered_data.T)
                # Regularize
                cov = np.maximum(cov, 1e-8 * np.eye(n_points))
            else:
                cov = np.eye(n_points) * initial_sigma
            # Apply decay
            cov = cov * covariance_decay
            new_covariances.append(cov)
            
            # New weights (uniform) if less than 1 component
            if n_components > 1:
                new_weights.append(1.0)
            else:
                new_weights.append(1.0)
                
            # Add remaining components if needed
            while len(new_means) < n_components:
                new_means.append(np.random.rand(n_points))
                new_covariances.append(np.eye(n_points) * initial_sigma)
                new_weights.append(1.0)
                
            # Normalize weights again
            new_weights = np.array(new_weights)
            new_weights = new_weights / np.sum(new_weights)
            
            means = new_means
            covariances = new_covariances
            weights = new_weights
            
            # Update best if we found a better one
            for h_candidate, score in top_candidates:
                if score < best_c5:
                    best_c5 = score
                    best_h = h_candidate.copy()
        
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
```
