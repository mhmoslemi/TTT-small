import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create structured initial values
        h = np.zeros(n_points)
        # Use a sawtooth-like pattern with gradually decreasing amplitude
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.cos(2 * np.pi * i / n_points))
        h = project_to_constraints(h, target_sum)
    
    # Convex relaxation parameters
    max_iter = 20000
    alpha0 = 0.1
    tau = 1000
    
    # Initialize tracking of best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop with convex relaxation
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Projected gradient step with surrogate objective
        # Approximate gradient: compute descent direction based on current overlap
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Approximate gradient via finite differences of overlap
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = project_to_constraints(h_plus, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_plus_max = np.max(corr_plus)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            h_minus = project_to_constraints(h_minus, target_sum)
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            corr_minus_max = np.max(corr_minus)
            
            grad[i] = (corr_plus_max - corr_minus_max) / (2e-6)
        
        # Gradient descent step
        alpha_t = alpha0 / (1 + t / tau)
        h = h - alpha_t * grad
        
        # Project back to feasible region
        h = project_to_constraints(h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Multi-resolution refinement
    n_refine = 1000
    for i in range(n_refine):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Interpolate to higher resolution
        n_high = n_points * 2
        h_high = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n_points), best_h)
        h_high = project_to_constraints(h_high, n_high / 2)
        
        # Optimization step
        corr = correlate(h_high, 1 - h_high, mode="full")
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        grad = np.zeros_like(h_high)
        for j in range(len(h_high)):
            h_plus = h_high.copy()
            h_plus[j] += 1e-6
            h_plus = project_to_constraints(h_plus, n_high / 2)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_plus_max = np.max(corr_plus)
            
            h_minus = h_high.copy()
            h_minus[j] -= 1e-6
            h_minus = project_to_constraints(h_minus, n_high / 2)
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            corr_minus_max = np.max(corr_minus)
            
            grad[j] = (corr_plus_max - corr_minus_max) / (2e-6)
        
        # Gradient descent
        alpha_t = alpha0 / (1 + i / tau)
        h_high = h_high - alpha_t * grad
        h_high = project_to_constraints(h_high, n_high / 2)
        
        # Downsample and update
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_high), h_high)
        best_h = project_to_constraints(best_h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
