import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Fallback: create a reasonable starting function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Iterative Lagrangian approach
    max_iter = 1000
    alpha = 0.01
    gamma = 0.1
    tol = 1e-6
    
    # For simplicity in this context, use a hybrid approach:
    # Start with a good initialization (from parent or basic construction)
    # Then use gradient-like refinement coupled with constraint handling
    
    h_current = h.copy()
    previous_c5 = best_c5
    
    # Main local optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current overlap matrix for all shifts
        # We'll work with the key insight that the gradient in the
        # Lagrangian sense is effectively the derivative of overlaps
        # But we will take a different route by working with shifts explicitly
        
        # Identify current maximizing shift
        overlap_full = correlate(h_current, 1 - h_current, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k] * dx
        
        # As a simplified approximation, we construct a gradient step
        # Take the gradient with respect to the maximizing shift k*
        # The full overlap is sum_i h_i * (1 - h_{(i-k)%n})
        # So the derivative w.r.t. h_i is (1 - h_{(i-k)%n}) for the maximizing k
        
        gradient = np.zeros(n_points)
        if max_k < n_points:
            # Padding case in full mode: h_i * (1 - h_{i-k})
            for i in range(n_points):
                j = (i - max_k) % n_points
                gradient[i] = 1 - h_current[j]
        else:
            # Normal shift case: h_{i+k} * (1 - h_i) 
            # We reverse that for the derivative (diff with respect to h_i)
            for i in range(n_points):
                j = (i + max_k - n_points + 1) % n_points
                gradient[i] = 1 - h_current[j]
        
        # Lagrange multiplier-like update step with penalty for constraints
        h_new = h_current - alpha * gradient
        
        # Project onto feasible set: [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust for sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Evaluate the new candidate
        new_c5 = evaluate_overlap(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Adaptive step size
        if abs(new_c5 - previous_c5) < tol:
            alpha *= 0.5
        elif abs(new_c5 - previous_c5) < 0.01:
            alpha *= 1.1
            
        h_current = h_new
        previous_c5 = new_c5
        
        # Early stopping
        if abs(new_c5 - best_c5) < 1e-8:
            break
    
    # Final gradient refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-9})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final normalization check
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
