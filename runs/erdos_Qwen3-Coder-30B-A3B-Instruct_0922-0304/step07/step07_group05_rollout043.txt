import numpy as np
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a smooth step function as initial guess
        h = np.zeros(n_points)
        # Create a smooth transition
        x = np.linspace(0, 2, n_points)
        h = 1.0 / (1.0 + np.exp(-20 * (x - 1.0)))
        # Normalize to target sum
        h = h * (target_sum / np.sum(h))
    
    # Zero-pad for better convolution
    padded_length = 2 * n_points
    h_padded = np.pad(h, (0, padded_length - n_points), 'constant')
    
    max_iter = 20
    alpha = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation h * (1-h)
        h_conj = 1 - h_padded
        corr = np.correlate(h_padded, h_conj, mode='full')
        
        # Find the maximum overlap
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Frequency-domain analysis
        H = fft(h_padded)
        H_conj = np.conj(H)
        R = H * H_conj
        R_ifft = ifft(R).real
        
        # Find dominant frequency
        freq_domain = fft(h_padded)
        dominant_freq = np.argmax(np.abs(freq_domain[1:n_points//2]))
        
        # Create phase inversion update
        phase_update = np.sin(2 * np.pi * dominant_freq * np.arange(padded_length) / padded_length)
        
        # Update h
        h_new = h_padded + alpha * phase_update
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to sum constraint
        h_new = h_new * (target_sum / np.sum(h_new[:n_points]))
        
        # Update h
        h_padded = h_new
        h = h_new[:n_points]
        
        # Reduce learning rate
        alpha *= 0.95
    
    # Final evaluation
    corr = np.correlate(h, 1-h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h.tolist(), c5_bound, n_points)
