import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with PSO-style particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Generate 40% of particles from initial solution with small variations
        for _ in range(int(0.4 * population_size)):
            variation = np.random.normal(0, 0.01, n_points)
            particle = np.clip(h + variation, 0, 1)
            particle = normalize_h(particle, target_sum)
            particles.append(particle)
        # Fill remaining particles with random feasible solutions
        for _ in range(population_size - len(particles)):
            particle = np.random.rand(n_points)
            particle = normalize_h(particle, target_sum)
            particles.append(particle)
    else:
        # Initialize entirely with random feasible solutions
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = normalize_h(particle, target_sum)
            particles.append(particle)
    
    # Initialize PSO parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # Track personal bests and swarm best
    personal_bests = [particle.copy() for particle in particles]
    personal_best_fitnesses = [compute_c5(particle, dx) for particle in particles]
    swarm_best_idx = np.argmin(personal_best_fitnesses)
    swarm_best = personal_bests[swarm_best_idx].copy()
    swarm_best_fitness = personal_best_fitnesses[swarm_best_idx]
    
    # PSO main loop
    max_iterations = 100
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(population_size):
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (inertia_weight * velocities[i] + 
                            cognitive_coeff * r1 * (personal_bests[i] - particles[i]) +
                            social_coeff * r2 * (swarm_best - particles[i]))
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Update personal best if improvement
            fitness = compute_c5(particles[i], dx)
            if fitness < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fitness
                personal_bests[i] = particles[i].copy()
                
                # Update swarm best if needed
                if fitness < swarm_best_fitness:
                    swarm_best_fitness = fitness
                    swarm_best = particles[i].copy()
        
        # Adjust PSO parameters dynamically
        if iteration % 10 == 0:
            inertia_weight *= 0.98
        if iteration > 20 and swarm_best_fitness > personal_best_fitnesses[swarm_best_idx]:
            # Increase variation if stagnated
            for i in range(population_size):
                variation = np.random.normal(0, 0.05, n_points)
                particles[i] = np.clip(particles[i] + variation, 0, 1)
                particles[i] = normalize_h(particles[i], target_sum)
        
        # Early termination
        if swarm_best_fitness <= 0.3807:
            break
            
    # Enhanced refinement with local optimization
    best_h = swarm_best.copy()
    best_c5 = swarm_best_fitness
    
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Local optimization for best solution
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
            else:
                break
        else:
            break
            
    # Final check for best solution
    c5_bound = compute_c5(best_h, dx)
    return (best_h.tolist(), c5_bound, n_points)
