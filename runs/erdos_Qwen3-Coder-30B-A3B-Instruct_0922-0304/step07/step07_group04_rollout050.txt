import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with Nesterov momentum and adaptive regularization
    def compute_gradient(h, dx):
        # Compute gradient of overlap w.r.t h
        # This is an approximation using finite differences
        n = len(h)
        grad = np.zeros(n)
        eps = 1e-6
        
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            overlap_plus = evaluate_overlap(h_plus, dx)
            overlap_minus = evaluate_overlap(h_minus, dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        return grad
    
    # Nesterov momentum parameters
    momentum = 0.9
    learning_rate = 1e-3
    lambda_reg = 1e-4  # adaptive regularization
    
    # For Nesterov momentum
    v = np.zeros_like(h)
    h_nesterov = h.copy()
    
    # Optimization loop
    iteration = 0
    max_iter = 10000
    patience = 50
    no_improvement = 0
    min_improvement = 1e-8
    
    while time.time() - start_time < budget_s - 5:
        # Adaptive regularization
        lambda_reg = min(lambda_reg * 1.01, 1e-1)  # Increase slowly
        if np.any(h < 0) or np.any(h > 1):
            lambda_reg = min(lambda_reg * 10, 1e0)  # Increase if violated
        
        # Update using Nesterov momentum
        # First compute gradient
        grad = compute_gradient(h_nesterov, dx)
        
        # Add regularization for sum constraint
        sum_error = np.sum(h_nesterov) - target_sum
        grad -= lambda_reg * sum_error  # Lagrange multiplier term
        
        # Update velocity
        v = momentum * v - learning_rate * grad
        # Update position
        h_new = h_nesterov + v
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > patience:
                break
        
        h_nesterov = h_new
        iteration += 1
        
        if no_improvement > patience or iteration > max_iter:
            break
    
    # Final validation and return
    final_h = best_h.copy()
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
