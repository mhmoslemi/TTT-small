import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use larger n_points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

        h = np.random.uniform(0, 1, n_points)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Evolution Strategy parameters
    pop_size = 20
    elite_size = 5
    max_iter = 20000
    
    # Initial population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            candidate = np.array(initial_h_values)
        else:
            candidate = np.random.uniform(0, 1, n_points)
        candidate = candidate * (target_sum / np.sum(candidate))
        population.append(candidate)
    
    # Main ES loop
    best_c5 = float('inf')
    best_h = None
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of population
        fitness = []
        for candidate in population:
            c5 = compute_c5(candidate, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        rest = [population[i] for i in sorted_indices[elite_size:]]
        
        # Update learning rate (decreasing over time)
        lr = max(0.01, 0.1 * (1 - iter_num / max_iter))
        
        # Generate offspring
        offspring = []
        for i in range(pop_size - elite_size):
            # Select two parents from elite
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Crossover (average)
            child = (parent1 + parent2) / 2
            
            # Mutation with Gaussian noise
            noise = np.random.normal(0, lr, n_points)
            child += noise
            
            # Project to constraints
            child = project_to_constraints(child, target_sum)
            
            offspring.append(child)
        
        # Update population
        population = elite + offspring
        
    # Final check
    if best_h is None:
        final_fitness = compute_c5(h, dx)
        best_h = h
        best_c5 = final_fitness
    else:
        final_fitness = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_fitness, n_points)
