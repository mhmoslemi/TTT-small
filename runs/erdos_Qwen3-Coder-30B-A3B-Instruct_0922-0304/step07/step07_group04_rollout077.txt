import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based optimization with adaptive regularization
    def compute_gradient(h_vec, dx, c5_val):
        """Compute gradient of max overlap w.r.t h."""
        # For all shifts, compute gradient contributions
        n = len(h_vec)
        grad = np.zeros(n)
        # Compute all valid shifted indices
        for k in range(-n+1, n):
            indices = np.arange(n)
            shifted_indices = (indices - k) % n
            # Compute contribution: derivative of h[i] * (1 - h[i-k])
            # For h[i]: contribution is (1 - h[i-k]) if i-k is valid and in range
            # For h[i-k]: contribution is -h[i] if i is valid and in range
            for i in range(n):
                j = shifted_indices[i]
                if 0 <= j < n:
                    grad[i] += (1 - h_vec[j]) * dx
                    grad[j] -= h_vec[i] * dx
        return grad / (c5_val + 1e-12)  # Normalize to avoid division by zero
    
    # Use adaptive stepsize and gradient descent
    max_iter = 2000
    learning_rate = 1e-3
    momentum = 0.9
    regularization = 1e-4
    velocity = np.zeros_like(h)
    
    current_h = h.copy()
    current_c5 = best_c5
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive regularization based on constraint violation
        h_sum_check = np.sum(current_h)
        violation = abs(h_sum_check - target_sum)
        if violation > 1e-4:
            regularization = min(regularization * 1.1, 1e-1)
        else:
            regularization = max(regularization * 0.99, 1e-6)
        
        # Compute gradient
        c5_eval = evaluate_overlap(current_h, dx)
        grad = compute_gradient(current_h, dx, c5_eval)
        
        # Gradient with regularization term to penalize constraint violations
        reg_term = current_h - (target_sum / len(current_h))
        grad += regularization * reg_term
        
        # Nesterov momentum
        velocity = momentum * velocity - learning_rate * grad
        current_h += momentum * velocity - learning_rate * grad
        
        # Project h onto feasible set: [0,1] and sum = target_sum
        current_h = np.clip(current_h, 0, 1)
        current_h_sum = np.sum(current_h)
        if current_h_sum > 0:
            current_h = current_h * (target_sum / current_h_sum)
            
        # Evaluate new value
        new_c5 = evaluate_overlap(current_h, dx)
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                last_improvement = iteration
        else:
            if iteration - last_improvement > 50:
                # Reduce learning rate if no improvement
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
