import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and 0 <= h[i] <= 1"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Adjust to target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def gradient_descent_step(h, dx, target_sum, learning_rate=1e-3):
    """Perform one gradient descent step on h"""
    # Compute overlap and its gradient using finite differences
    overlap = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(overlap)
    max_val = overlap[max_k]
    
    # Approximate gradient via finite differences
    epsilon = 1e-4
    grad = np.zeros_like(h)
    h_plus = h.copy()
    h_minus = h.copy()
    
    for i in range(len(h)):
        h_plus[i] += epsilon
        h_minus[i] -= epsilon
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_k]
        overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_k]
        grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        h_plus[i] = h_minus[i] = h[i]
    
    # Scale gradient
    grad = grad / (np.linalg.norm(grad) + 1e-10)
    
    # Update step
    h_new = h - learning_rate * grad
    
    # Project to constraints
    h_new = project_to_constraints(h_new, target_sum)
    return h_new
