import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit proportionally to remaining space
    deficit = target_sum - current_sum
    max_incr = 1 - h
    total_room = np.sum(max_incr)
    if total_room > 0:
        h += deficit * max_incr / total_room
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Random initialization
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    
    # Evolution Strategy parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    
    # Initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        candidate = h + noise
        candidate = project_to_constraints(candidate, target_sum)
        population.append(candidate)
    
    # Evaluate initial population
    fitnesses = [compute_c5(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_individual = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Evolution loop
    max_iter = 20000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update mutation rate
        t = iter_num / max_iter
        mutation_rate = mutation_rate_start + (mutation_rate_end - mutation_rate_start) * t
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        rest = [population[i] for i in sorted_indices[elite_size:]]
        
        # Generate new population
        new_population = elite.copy()
        
        # Mutate elite members
        for i in range(elite_size):
            parent = elite[i]
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            new_population.append(child)
        
        # Fill with mutated rest individuals
        for i in range(elite_size, population_size-1):
            parent = rest[np.random.randint(0, len(rest))]
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitnesses = [compute_c5(ind, dx) for ind in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_individual = population[current_best_idx].copy()
            best_fitness = fitnesses[current_best_idx]
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    # Final validation
    final_fitness = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
