import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:

        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # ES parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_generations = 2000
    elite_size = pop_size // 2
    
    # Initialize population
    population = []
    fitnesses = []
    
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
        fitnesses.append(compute_c5(individual, dx))
    
    # Best solution tracking
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        elite_fitness = [fitnesses[i] for i in sorted_indices[:elite_size]]
        
        # Update best solution
        if elite_fitness[0] < best_fitness:
            best_h = elite[0].copy()
            best_fitness = elite_fitness[0]
            
        # Check for early termination
        if best_fitness <= 0.3807:
            break
            
        # Create new population
        new_population = elite.copy()
        new_fitnesses = elite_fitness.copy()
        
        # Mutation
        sigma = sigma_init - (sigma_init - sigma_final) * gen / max_generations
        for _ in range(pop_size - elite_size):
            parent = elite[np.random.randint(0, elite_size)]
            mutant = parent + np.random.normal(0, sigma, n_points)
            mutant = project_to_constraints(mutant, target_sum)
            new_population.append(mutant)
            new_fitnesses.append(compute_c5(mutant, dx))
        
        population = new_population
        fitnesses = new_fitnesses
    
    # Local optimization for best solution
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_fun(h):
        return np.sum(h) - target_sum

    # Try local optimization on best solution
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 1000})
        if res.success:
            h_local = res.x
            h_local = project_to_constraints(h_local, target_sum)
            local_c5 = compute_c5(h_local, dx)
            if local_c5 < best_fitness:
                best_h = h_local
                best_fitness = local_c5
    except:
        pass
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
