import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing
    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-6
    max_iterations = int(budget_s * 1000)
    learning_rate = 0.01
    penalty_weight1 = 100.0
    penalty_weight2 = 100.0

    def compute_energy(h_vec, c5_val):
        # Penalty for bounds violation
        bound_penalty = np.sum(np.square(np.clip(h_vec, 0, 1) - h_vec))
        # Penalty for sum constraint
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return c5_val + penalty_weight1 * bound_penalty + penalty_weight2 * sum_penalty

    def perturb_h(h_vec, lr):
        # Gaussian perturbation
        pert = np.random.normal(0, lr, n_points)
        new_h = h_vec + pert
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        # Renormalize
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        return new_h

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h
        new_h = perturb_h(h, learning_rate)
        new_c5 = evaluate_overlap(new_h, dx)
        new_energy = compute_energy(new_h, new_c5)
        old_energy = compute_energy(h, best_c5)
        
        # Accept or reject
        if new_energy < old_energy or np.random.rand() < np.exp(-(new_energy - old_energy) / temperature):
            h = new_h
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
        # Cool down
        temperature *= cooling_rate
        if temperature < min_temperature:
            temperature = min_temperature
            
        # Adjust learning rate based on temperature
        learning_rate = 0.01 * (temperature / 1.0)
    
    # Local refinement with gradient descent
    if time.time() - start_time < budget_s - 5:
        # Use a simple gradient approximation to refine
        def gradient_descent_refinement(h_init, max_iter=200):
            h_current = h_init.copy()
            lr = 0.001
            
            for _ in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Approximate gradient from finite differences
                eps = 1e-4
                grad = np.zeros(n_points)
                base_val = evaluate_overlap(h_current, dx)
                
                for i in range(n_points):
                    h_plus = h_current.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus_sum = np.sum(h_plus)
                    if h_plus_sum > 0:
                        h_plus = h_plus * (target_sum / h_plus_sum)
                    val_plus = evaluate_overlap(h_plus, dx)
                    grad[i] = (val_plus - base_val) / eps
                
                # Gradient step
                h_current -= lr * grad
                # Project onto feasible region
                h_current = np.clip(h_current, 0, 1)
                h_current_sum = np.sum(h_current)
                if h_current_sum > 0:
                    h_current = h_current * (target_sum / h_current_sum)
                
                # Evaluate performance
                current_c5 = evaluate_overlap(h_current, dx)
                if current_c5 < best_c5:
                    best_h[:] = h_current[:]
                    best_c5 = current_c5
                    
                lr *= 0.999  # Decreasing learning rate
                
            return h_current
        
        refined_h = gradient_descent_refinement(best_h)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final normalization
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
