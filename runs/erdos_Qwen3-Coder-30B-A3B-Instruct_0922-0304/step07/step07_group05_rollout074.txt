import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Start with a triangular pulse
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = i / mid
            else:
                h[i] = 1 - (i - mid) / (n_points - mid)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Use a quadratic approximation approach
    patience = 0
    max_patience = 10
    iteration = 0
    max_iterations = 60
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        if patience >= max_patience:
            break
            
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Compute gradient numerically
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            
            grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
        
        # Solve QP to find descent direction using cvxpy
        # Minimize: 0.5 * d^T * H * d + grad^T * d
        # Subject to: 0 <= d <= 1, sum(d) = 0
        
        # Quadratic approximation of the objective around h
        # We can avoid solving the full Hessian
        # Instead, we'll use a simpler SGD-like approach but with projection
        step_size = 0.01
        descent_direction = -grad
        
        # Line search
        alpha = 0.5
        max_line_search = 20
        for _ in range(max_line_search):
            new_h = h + alpha * descent_direction
            new_h = project_to_feasible(new_h, target_sum)
            new_c5 = compute_c5(new_h, dx)
            
            if new_c5 < current_c5:
                h = new_h
                break
            alpha *= 0.5
            
        iteration += 1
        
    # Final refinement
    if time.time() - start_time < budget_s - 1:
        # Try some local search refinement
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            # Random perturb & project
            pert = np.random.normal(0, 0.001, n_points)
            h_try = h + pert
            h_try = project_to_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    return (best_h.tolist(), best_c5, n_points)
