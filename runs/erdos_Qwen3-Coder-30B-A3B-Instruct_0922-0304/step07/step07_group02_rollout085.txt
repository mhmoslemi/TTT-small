import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Simple approach: add uniform correction
    h = h + diff / len(h)
    # Clip to bounds and rescale
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a smooth initial guess, then discretize
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        # A piece-wise constant approximation to a smooth function
        h[x < 1] = 1.0
        h[x >= 1] = 0.0
        h = normalize_h(h, target_sum)
    
    # DE parameters
    population_size = 60
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_iter = 3000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
        population.append(particle)
    
    # Initialize best
    fitnesses = [compute_c5(p, dx) for p in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        new_population = []
        
        for i in range(population_size):
            # Select three distinct random indices
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
            
            # Project to constraints
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitnesses[i]:
                new_population.append(trial)
                fitnesses[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
        
        population = new_population
        iter_count += 1
        
        # Periodic check for early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
