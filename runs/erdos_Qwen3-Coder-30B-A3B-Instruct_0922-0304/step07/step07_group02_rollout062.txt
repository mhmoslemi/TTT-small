import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add small perturbations
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.8
    crossover_probability = 0.7
    max_iterations = 3000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
        population.append(particle)
    
    # Evaluate initial population
    fitness = []
    for particle in population:
        fitness.append(compute_c5(particle, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        if iterations % 100 == 0 and iterations > 0:
            # Early stopping if we've found a good solution
            if best_c5 <= 0.3807:
                break
                
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct random individuals
            indices = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = x1 + mutation_factor * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_probability:
                    trial[j] = mutant[j]
            
            # Project to constraint
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
