import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def solve_convex_relaxation(n_points, target_sum):
    h = cp.Variable(n_points)
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # We'll use a constraint that all convolution values are bounded by C
    # For now, we'll do a simpler approach: minimize the maximum convolution
    C = cp.Variable()
    # Add constraints: each convolution value <= C
    # We'll do full convolution estimation and make it a constraint
    
    # Instead, get the first and last few values as proxy
    # This is a relaxation, not exact
    h_constrained = cp.hstack([h, np.zeros(n_points-1)])
    h_flipped = cp.hstack([1-h, np.zeros(n_points-1)])
    conv = cp.convolve(h_constrained, h_flipped, mode='full')
    # Use a few key entries
    for i in range(n_points*2 - 1):
        constraints.append(conv[i] <= C)
    
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h.value
    except:
        pass
    return None

def generate_step_function(h_cont, n_points, target_sum):
    # Threshold to get step function
    h = np.zeros(n_points)
    # Use median or mean to choose threshold
    threshold = np.mean(h_cont)
    for i in range(n_points):
        if h_cont[i] >= threshold:
            h[i] = 1
        else:
            h[i] = 0

    # Apply exact sum normalization
    h = project_to_constraints(h, target_sum)
    return h

def local_refinement(h, target_sum, dx, max_iter=50):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    n_points = len(h)
    step_positions = np.where(h > 0.5)[0]
    if len(step_positions) == 0:
        # If no steps, initialize some
        step_positions = np.array([n_points//4, 3*n_points//4])
    
    for iter_count in range(max_iter):
        # Perturb step positions 
        new_positions = step_positions.copy()
        for _ in range(5):
            idx = np.random.randint(len(new_positions))
            pert = np.random.randint(-5, 6)
            new_positions[idx] += pert
            new_positions[idx] = max(0, min(n_points-1, new_positions[idx]))
        
        # Put back into step function
        h_new = np.zeros(n_points)
        for pos in new_positions:
            if 0 <= pos < n_points:
                h_new[pos] = 1
                
        h_new = project_to_constraints(h_new, target_sum)

        current_c5 = compute_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h = best_h.copy()
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

        h_cont = solve_convex_relaxation(n_points, target_sum)
        if h_cont is not None:
            h = generate_step_function(h_cont, n_points, target_sum)
        else:
            # Fall-back initialization
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            h = project_to_constraints(h, target_sum)
    
    # Local refinement
    best_h, best_c5 = local_refinement(h, target_sum, dx, 50)
    
    # Iterative improvement
    for i in range(100):
        if time.time() - start_time > budget_s - 5:

        # Improve h with local search
        h_new, c5_new = local_refinement(best_h, target_sum, dx, 10)
