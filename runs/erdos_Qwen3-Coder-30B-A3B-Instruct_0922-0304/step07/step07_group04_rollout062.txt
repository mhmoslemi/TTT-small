import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing approach
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling rate
    eta = 0.01  # Learning rate
    lambda1 = 100.0  # Penalty weight for bounds
    lambda2 = 100.0  # Penalty weight for sum
    
    def compute_energy(h_vec, c5_val):
        # Constraint penalties
        bound_penalty = np.sum(np.square(np.clip(h_vec, 0, 1) - h_vec))
        sum_penalty = (np.sum(h_vec) - target_sum) ** 2
        return c5_val + lambda1 * bound_penalty + lambda2 * sum_penalty
    
    def perturb_h(h_vec, T_val):
        # Perturb using Gaussian noise scaled by temperature
        noise = np.random.normal(0, T_val * 0.1, n_points)
        new_h = h_vec + noise
        # Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Normalize
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * (target_sum / current_sum)
        return new_h
    
    iterations = 0
    max_iter = 100000
    
    while iterations < max_iter and (time.time() - start_time) < budget_s - 5:
        # Perturb
        new_h = perturb_h(h, T)
        
        # Evaluate
        new_c5 = evaluate_overlap(new_h, dx)
        old_c5 = evaluate_overlap(h, dx)
        
        # Energy values
        old_energy = compute_energy(h, old_c5)
        new_energy = compute_energy(new_h, new_c5)
        
        # Accept or reject
        delta_energy = new_energy - old_energy
        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):
            h = new_h
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        
        # Cool
        T *= alpha
        iterations += 1
        
        # Occasionally adjust learning rate
        if iterations % 1000 == 0:
            eta *= 0.99
        
        # Prevent complete stagnation
        if iterations % 5000 == 0 and T < 0.001:
            T = 1.0
    
    # Gradient refinement at the end (if time allows)
    if (time.time() - start_time) < budget_s - 2:
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        def gradient(h_vec):
            # Simple finite difference gradient estimation
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
            return grad
        
        # Gradient descent steps
        steps = 200
        for _ in range(steps):
            if (time.time() - start_time) >= budget_s - 1:
                break
            grad = gradient(best_h)
            best_h -= eta * grad
            # Clip and normalize
            best_h = np.clip(best_h, 0, 1)
            current_sum = np.sum(best_h)
            if current_sum > 0:
                best_h = best_h * (target_sum / current_sum)
            # Re-evaluate
            new_c5 = evaluate_overlap(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
