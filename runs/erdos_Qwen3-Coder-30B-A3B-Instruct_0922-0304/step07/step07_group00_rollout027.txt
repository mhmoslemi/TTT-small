import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the max overlap with respect to h."""
    # We compute approximate gradient using finite differences for max overlap
    # This gives us direction to move for reducing maximum overlap
    grad = np.zeros_like(h)
    # For each position i, compute gradient of max overlap
    # This is a simplified gradient vector that will encourage reduction
    # We compute overlap with 1-h at different shifts k (0 to 2*n_points)
    overlap_full = correlate(h, 1 - h, mode="full")
    
    # Take approximate gradient for max element
    # We compute an approximate gradient based on influence of each h[i]
    # Approximate as derivative of max overlap term w.r.t h[i]
    for i in range(n_points):
        grad[i] = 0.0
        # The influence is sum over k of 1-2*h[i+k] terms, weighted by their overlap influence
        # Simplified as we have indices wrapping (circular)
        for k in range(n_points):
            idx = (i + k) % n_points
            if idx < n_points:
                grad[i] += (1.0 - 2.0 * h[idx]) * overlap_full[n_points - 1 + k]
    
    # Normalize the gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    return grad

        grad = compute_gradient(h, dx, n_points)

        v = 0.9 * v + 0.1 * grad
        h = h - learning_rate * v
        
        # Clip h to [0, 1]

        h = normalize_h(h, target_sum)
