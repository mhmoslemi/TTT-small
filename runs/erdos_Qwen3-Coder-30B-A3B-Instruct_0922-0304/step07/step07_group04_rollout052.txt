import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with adaptive regularization
    max_iter = 10000
    lr = 1e-3
    momentum = 0.9
    reg_lambda = 1e-4
    tolerance = 1e-8
    backoff_factor = 0.5
    
    # Precompute shift indices for correlation
    shift_indices = []
    for k in range(-n_points + 1, n_points):
        indices = np.arange(n_points)
        shifted = (indices - k) % n_points
        shift_indices.append(shifted)
    
    # Initialize gradient tracking
    v = np.zeros_like(h)
    h_prev = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlap
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Estimate gradient
        grad_h = np.zeros_like(h)
        for k, indices in enumerate(shift_indices):
            # Calculate gradient contribution for h[i] from overlap with h[i-k]
            # We approximate the derivative of max over all shifts
            # For computational efficiency, we sample the gradient somewhat 
            # arbitrarily (but in a way that's representative of the dominant 
            # derivatives)
            if k == 0:  # Skip self-correlation which doesn't contribute to gradient
                continue
            # Simple gradient: assume change in overlap is proportional to 
            # the difference h[i] and h[i-k], adjusted for overlap weights
            # This is a rough approximation of actual derivative for C5
            grad_h += (h[indices] - h)  # This is simplified
        
        # Add regularization 
        penalty = reg_lambda * h * (h - 1)
        grad_h += penalty
        
        # Momentum update
        v = momentum * v - lr * grad_h
        h_new = h_prev + v
        
        # Projection onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (target_sum / h_sum)
            
        # Update for next iteration
        h = h_new
        h_prev = h_new
        
        # Adapt learning rate and regularization
        if iteration % 100 == 0:
            # Reduce learning rate if not improving
            if iteration > 0:
                reg_lambda = min(1e-1, reg_lambda * 1.1)
        
        # Check termination
        if np.linalg.norm(grad_h) < tolerance:
            break
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
