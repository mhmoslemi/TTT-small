import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    
    # Initialize relaxed version
    h_relaxed = h + np.random.normal(0, 0.05, n_points)
    
    # Optimizer parameters
    learning_rate = 0.05
    momentum = 0.9
    reg_lambda = 1e-4
    
    # Velocity storage for momentum
    velocity = np.zeros_like(h_relaxed)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and iterations < 3000:
        # Get current c5
        current_c5 = compute_c5(h, dx)
        
        # Compute gradients using finite differences
        grad = np.zeros_like(h_relaxed)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus_clipped = np.clip(h_plus, 0, 1)
            h_plus_clipped = project_to_constraint(h_plus_clipped, target_sum)
            c5_plus = compute_c5(h_plus_clipped, dx)
            
            h_minus = h_relaxed.copy()
            h_minus[i] -= eps
            h_minus_clipped = np.clip(h_minus, 0, 1)
            h_minus_clipped = project_to_constraint(h_minus_clipped, target_sum)
            c5_minus = compute_c5(h_minus_clipped, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_relaxed += velocity
        
        # Regularize to prevent excessive drift from feasible region
        regularization = reg_lambda * (h_relaxed - h)
        h_relaxed -= regularization
        
        # Project back to feasible region and satisfy sum constraint
        h = np.clip(h_relaxed, 0, 1)
        h = project_to_constraint(h, target_sum)

        new_c5 = compute_c5(h, dx)
