import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def project_to_bounds(h):
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_bounds(h)
        h = project_to_sum_constraint(h, target_sum)
    else:
        # Initialize with a step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_sum_constraint(h, target_sum)
    
    # Convex relaxation approach
    # Create variables
    h_var = cp.Variable(n_points)
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Create variables to bound the max correlation
    C = cp.Variable()
    
    # For simplicity, we'll use a constraint-based approach to implicit maximization
    # This is done via a relaxation approach in CVXPY
    # We'll try constrained optimization with cvxpy to get a continuous solution
    
    # Try to relax this to convex problem with an explicit constraint
    # We'll form the correlation as a matrix and bound it
    # The relaxation approach via CVXPY on a bounded continuous h
    
    # Compute the convolution constraints
    # Let's compute the integral of the convolution in terms of constraints
    
    # We'll use convex relaxation for initial continuous h
    # Then we'll quantize to a step function
    
    # Try direct convex solving for continuous h
    try:
        prob = cp.Problem(
            cp.Minimize(C),
            [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        )
        prob.solve(solver=cp.CVXOPT, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_cont = h_var.value
            h = project_to_bounds(h_cont)
            h = project_to_sum_constraint(h, target_sum)
    except:
        pass
    
    # Now refine with local optimization
    # Try a few simple local adjustments
    
    def local_refine(h, dx, target_sum):
        # Try different step positions and heights to improve C5
        best_h = h.copy()
        best_c5 = compute_c5(h, dx)
        
        # Perturb the step function
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            
            h_new = h.copy()
            
            # Change one element
            idx = np.random.randint(0, n_points)
            step_change = np.random.normal(0, 0.05)
            h_new[idx] += step_change
            h_new = project_to_bounds(h_new)
            h_new = project_to_sum_constraint(h_new, target_sum)
            
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
        return best_h, best_c5
    
    h, c5 = local_refine(h, dx, target_sum)
    
    # Try more precise local adjustments
    max_iterations = 1000
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the positions
        idxs = np.random.choice(n_points, size=4, replace=False)
        h_new = h.copy()
        
        # Adjust a few values
        for idx in idxs:
            change = np.random.normal(0, 0.02)
            h_new[idx] += change
        h_new = project_to_bounds(h_new)
        h_new = project_to_sum_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    
    return (h.tolist(), c5, n_points)
