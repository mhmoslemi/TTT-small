import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = int(1e5)
    lr = 0.01
    lambda1 = 100.0
    lambda2 = 100.0
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the function
        h_new = h + np.random.normal(0, lr * T, n_points)
        
        # Apply bounds and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Compute current and new energies
        c5_old = evaluate_overlap(h, dx)
        c5_new = evaluate_overlap(h_new, dx)
        
        # Constraint penalties
        bound_penalty_old = np.sum(np.square(h - np.clip(h, 0, 1)))
        bound_penalty_new = np.sum(np.square(h_new - np.clip(h_new, 0, 1)))
        sum_penalty_old = (np.sum(h) - target_sum)**2
        sum_penalty_new = (np.sum(h_new) - target_sum)**2
        
        E_old = c5_old + lambda1 * bound_penalty_old + lambda2 * sum_penalty_old
        E_new = c5_new + lambda1 * bound_penalty_new + lambda2 * sum_penalty_new
        
        # Metropolis criterion
        delta_E = E_new - E_old
        if delta_E < 0 or np.exp(-delta_E / T) > np.random.random():
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Cooling schedule
        T = max(1e-6, T * alpha)
    
    # Final gradient-based refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
