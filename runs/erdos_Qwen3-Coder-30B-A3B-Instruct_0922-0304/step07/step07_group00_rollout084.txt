import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def update_h(h, i, alpha, target_sum):
    h_old = h[i]
    h[i] = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
    scaling = target_sum / np.sum(h)
    h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Iteration parameters
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Adaptive step size
    alpha_min = 0.001
    alpha_max = 0.1
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Fast track if targets met
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient-like components
        # Approximate gradient component at each point
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Derivative of overlap with respect to h[i]
            # For overlap at position k = i, we have:
            # overlap[k] = sum over j: h[j] * (1 - h[j+k]) 
            # df/dh[i] = (1 - h[i+k]) - h[i+k]
            # But we only compute a simplified approximation:
            grad[i] = -2 * h[i]
            
        # Find indices with highest absolute gradient
        sorted_indices = np.argsort(np.abs(grad))[::-1]
        top_indices = sorted_indices[:max(1, n_points // 10)]
        
        # Try update along top indices
        for i in top_indices:
            # Compute the effect of increasing h[i]
            old_val = h[i]
            # Small increment in h[i] direction
            step = 0.01
            h_test = h.copy()
            h_test[i] = np.clip(h_test[i] + step, 0, 1)
            h_test = normalize_h(h_test, target_sum)
            
            # Check if this improves C5
            test_c5 = compute_c5(h_test, dx)
            if test_c5 < current_max:
                # Do actual update
                new_h = update_h(h.copy(), i, step, target_sum)
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                h = new_h.copy()
                break  # Update one point per iteration
        
        iter_count += 1
        if current_max <= best_c5:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > max_no_improvement:
            break
            
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
