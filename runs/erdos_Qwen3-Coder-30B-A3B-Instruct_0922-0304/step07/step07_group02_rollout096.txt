import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Simple projection: adjust all values equally
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
        
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    mutation_scale = 0.1
    covariance_decay = 0.99
    min_covariance = 1e-6
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    # Main component from initial solution
    means.append(h.copy())
    covariances.append(np.eye(n_points) * mutation_scale)
    weights.append(1.0)
    
    # Add random components
    for _ in range(n_components - 1):
        initial_guess = np.random.rand(n_points)
        initial_guess = np.clip(initial_guess, 0, 1)
        initial_guess = project_to_constraint(initial_guess, target_sum)
        means.append(initial_guess)
        covariances.append(np.eye(n_points) * mutation_scale)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iter = 6000  # Safe limit accounting for overhead
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample from GMM
        candidates = []
        candidate_c5s = []
        
        for _ in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            mean = means[comp_idx]
            cov = covariances[comp_idx]
            
            # Sample from Gaussian
            sample = np.random.multivariate_normal(mean, cov)
            
            # Apply bounds and project
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            
            # Evaluate
            c5 = compute_c5(sample, dx)
            candidates.append(sample)
            candidate_c5s.append(c5)
        
        # Select best candidates
        sorted_indices = np.argsort(candidate_c5s)
        top_candidates = [candidates[i] for i in sorted_indices[:keep_top]]
        top_c5s = [candidate_c5s[i] for i in sorted_indices[:keep_top]]
        
        # Update best
        min_c5_idx = np.argmin(top_c5s)
        if top_c5s[min_c5_idx] < best_c5:
            best_c5 = top_c5s[min_c5_idx]
            best_h = top_candidates[min_c5_idx].copy()
            
        # Update GMM
        if len(top_candidates) > 0:
            # Compute new means
            new_means = []
            new_covariances = []
            
            # Update each component
            for i in range(n_components):
                if i < len(top_candidates):
                    # Use top candidate with probability weights
                    top_idx = i % len(top_candidates)
                    new_means.append(top_candidates[top_idx].copy())
                    
                    # Compute weighted covariance of top candidates
                    if i == 0:  # Best component
                        # Use sample covariance scaled by top candidates
                        x_centered = np.array(top_candidates) - means[i]
                        if x_centered.shape[0] > 0:
                            new_cov = np.cov(x_centered.T) + np.eye(n_points) * min_covariance
                        else:
                            new_cov = covariances[i].copy()
                    else:
                        # For other components, use a mix
                        if len(top_candidates) > 1:
                            new_cov = np.cov(np.array(top_candidates).T) + np.eye(n_points) * min_covariance
                        else:
                            new_cov = covariances[i].copy()
                    
                    # Decay covariance
                    new_cov = new_cov * covariance_decay
                    # Ensure positive definiteness
                    new_cov = new_cov + np.eye(n_points) * min_covariance
                    
                    new_covariances.append(new_cov)
                else:
                    # Keep existing component
                    new_means.append(means[i].copy())
                    new_cov = covariances[i].copy()
                    new_cov = new_cov * covariance_decay
                    new_cov = new_cov + np.eye(n_points) * min_covariance
                    new_covariances.append(new_cov)
            
            # Ensure we have correct number of components
            means = new_means[:n_components]
            covariances = new_covariances[:n_components]
            
            # Update weights - simple approach
            weights = np.ones(n_components) / n_components
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        iteration += 1
    
    # Final validation
    final_h = project_to_constraint(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
