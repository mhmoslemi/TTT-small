import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Custom genetic algorithm
    population_size = 100
    num_generations = 1000
    tournament_size = population_size // 10
    mutation_rate = 0.1
    max_no_improvement = 50
    
    # Initialize population
    def initialize_population():
        population = []
        # Start with current solution
        population.append(h.copy())
        for _ in range(population_size - 1):
            individual = np.random.uniform(0, 1, n_points)
            individual = np.clip(individual, 0, 1)
            current_sum = np.sum(individual)
            if current_sum > 0:
                individual = individual * (target_sum / current_sum)
            population.append(individual)
        return population
    
    def fitness(individual):
        return evaluate_overlap(individual, dx)
    
    # Selection
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(len(population) // 2):
            tournament_indices = np.random.choice(len(population), 
                                                size=tournament_size, 
                                                replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        alpha = np.random.uniform(0.2, 0.8)
        child = alpha * parent1 + (1 - alpha) * parent2
        child = np.clip(child, 0, 1)
        current_sum = np.sum(child)
        if current_sum > 0:
            child = child * (target_sum / current_sum)
        return child
    
    # Mutation
    def mutate(individual, mutation_rate, generation, max_generations):
        # Adaptive mutation rate
        adaptive_rate = mutation_rate * (1 - generation / max_generations)
        mask = np.random.binomial(1, adaptive_rate, n_points)
        noise = np.random.normal(0, 0.05, n_points)
        individual += mask * noise
        individual = np.clip(individual, 0, 1)
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (target_sum / current_sum)
        return individual
    
    # Initialize population
    population = initialize_population()
    
    generation = 0
    no_improvement_count = 0
    last_best = best_c5
    
    while generation < num_generations and (time.time() - start_time) < budget_s - 5:
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Track best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            new_population.extend([child1, child2])
        
        # Mutate
        for i in range(len(new_population)):
            new_population[i] = mutate(new_population[i], mutation_rate, generation, num_generations)
        
        # Keep some of the best from previous generation
        elite_size = population_size // 10
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in range(elite_size):
            new_population[i] = population[elite_indices[i]].copy()
        
        population = new_population[:population_size]
        generation += 1
    
    # Apply local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
