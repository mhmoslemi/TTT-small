import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Read initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with adaptive regularization
    max_iter = 10000
    lr = 1e-3
    momentum = 0.9
    reg_lambda = 1e-4
    beta = 0.99
    grad_norm_threshold = 1e-8
    
    # Precompute shift indices
    all_shifts = []
    for k in range(-n_points + 1, n_points):
        shift_idx = np.arange(n_points)
        shifted_idx = (shift_idx - k) % n_points
        all_shifts.append(shifted_idx)
    
    # Initialize momentum
    v = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences for numerical stability
        current_c5 = evaluate_overlap(h, dx)
        eps = 1e-6
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
            c5_plus = evaluate_overlap(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum > 0:
                h_minus = h_minus * (target_sum / h_minus_sum)
            c5_minus = evaluate_overlap(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with momentum
        v = momentum * v - lr * grad
        h_new = h + v
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (target_sum / h_sum)
        
        # Adaptive regularization
        reg_term = reg_lambda * (np.sum(h_new) - target_sum)
        h_new = h_new - reg_term / (2 * lr)
        h_new = np.clip(h_new, 0, 1)
        
        # Project again
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Compute new overlap
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Update if better
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        # Update for next iteration
        h = h_new.copy()
        
        # Check for convergence
        if np.linalg.norm(grad) < grad_norm_threshold:
            break
            
        # Adjust regularization
        if np.abs(np.sum(h) - target_sum) > 1e-3:
            reg_lambda *= beta
        else:
            reg_lambda *= 1.01
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
