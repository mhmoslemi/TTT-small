import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    no_improvement = 0
    max_no_improvement = 1000
    step_size = 0.1
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        # Compute overlap and find which indices contribute to max overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if best_c5 <= 0.3807:
            break

        # Select top 10% of coordinates based on gradient magnitude
        # Gradient is approximately the difference in overlap when changing h[i]
        grad_magnitude = np.abs(overlap - np.mean(overlap))
        idxs = np.argsort(grad_magnitude)[-max(1, n_points // 10):][::-1]
        
        # Update selected coordinates
        for i in idxs:
            # Compute potential improvement by adjusting this coordinate
            old_val = h[i]
            h[i] = np.clip(old_val + step_size * (1 - 2 * old_val), 0, 1)
            
            # Normalize to maintain sum constraint
            h = normalize_h(h, target_sum)
            
            # Check if improvement
            new_c5 = compute_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                # Revert the update
                h[i] = old_val
                h = normalize_h(h, target_sum)
                
        iteration += 1
        # Reduce step size to encourage convergence
        step_size *= 0.9995

    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
