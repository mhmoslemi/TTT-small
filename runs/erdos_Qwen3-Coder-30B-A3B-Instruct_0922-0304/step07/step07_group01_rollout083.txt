import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h to meet sum constraint
    h = project_to_constraints(h, target_sum)
    
    # Stage 1: Convex relaxation using CVXPY
    # Create variables for continuous relaxation
    h_cont = cp.Variable(n_points)
    C = cp.Variable()
    
    # Define constraints
    constraints = [
        h_cont >= 0,
        h_cont <= 1,

        C >= 0
    ]

    full_overlap = correlate(h_cont.value, 1 - h_cont.value, mode="full")
    max_lag = 2 * n_points - 1
    for i in range(max_lag):
        shift = np.zeros(max_lag)
        shift[i] = 1
        shifted_h = correlate(h_cont, shift, mode="full")[:n_points]
        scaled_overlap = cp.multiply(shifted_h, 1 - h_cont) 
        # Direct constraint on each entry of overlap
        # But we'll simplify to use the maximum principal
        # We'll evaluate marginally via gradient descent next
    
    # Direct approach: Use convex relaxation for just the sum constraint and bounds
    # Just do a refinement with gradient-based method
    
    # Stage 2: Projected gradient descent
    def gradient_step(h, dx, lr=0.01):
        """Simple gradient-based update for h."""
        # Compute current overlap integral as a vector of k values
        # This is a simplified gradient approximation - we can't compute true gradient
        # So we will do a local search
        current_c5 = compute_c5(h, dx)

        perturbations = [0.01, -0.01, 0.005, -0.005]
        best_h = h.copy()
        best_c5 = current_c5

            # Try very small changes to individual elements
            h_trial = h.copy()

            delta = np.random.choice(perturbations)
            h_trial[i] += delta
            
            # Project
            h_trial = project_to_constraints(h_trial, target_sum)
            
            trial_c5 = compute_c5(h_trial, dx)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = h_trial
                
        return best_h, best_c5
    
    # Main optimization loop
    best_h = h.copy()
