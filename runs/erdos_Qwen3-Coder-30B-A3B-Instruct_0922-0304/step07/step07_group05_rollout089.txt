```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm with particles
    population_size = 100
    population = []
    velocities = []
    
    # Set up initial particles
    if 'initial_h_values' in globals():
        # Use the initial construction
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create 40% of particles based on this initial solution with small variations
        for i in range(int(0.4 * population_size)):
            variation = np.random.normal(0, 0.01, n_points)
            particle = np.clip(h + variation, 0, 1)
            particle = normalize_h(particle, target_sum)
            population.append(particle)
            velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    else:
        # No initial construction, just random population
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        for _ in range(population_size):
            variation = np.random.normal(0, 0.01, n_points)
            particle = np.clip(h + variation, 0, 1)
            particle = normalize_h(particle, target_sum)
            population.append(particle)
            velocities.append(np.random.uniform(-0.01, 0.01, n_points))
            
    # Fill up to population_size
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Particle Swarm Optimization parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iterations = 60
    
    # Track personal bests
    personal_bests = [p.copy() for p in population]
    personal_best_fitnesses = [compute_c5(p, dx) for p in population]
    
    # Global best
    global_best_idx = np.argmin(personal_best_fitnesses)
    global_best = personal_bests[global_best_idx].copy()
    global_best_fitness = personal_best_fitnesses[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        inertia_weight *= 0.98
        if iteration % 10 == 0 and iteration > 0:
            inertia_weight *= 0.9
        
        # Iterate through all particles
        for i in range(population_size):
            # Update velocity: v[i] = w*v[i] + c1*r1*(pbest[i] - pos[i]) + c2*r2*(gbest - pos[i])
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive_component = cognitive_coeff * r1 * (personal_bests[i] - population[i])
            social_component = social_coeff * r2 * (global_best - population[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive_component + social_component
            
            # Update position
            population[i] += velocities[i]
            
            # Apply constraints to position
            population[i] = np.clip(population[i], 0, 1)
            population[i] = normalize_h(population[i], target_sum)
            
            # Evaluate new fitness
            fit = compute_c5(population[i], dx)
            
            # Update personal best
            if fit < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fit
                personal_bests[i] = population[i].copy()
            
            # Update global best if needed
            if fit < global_best_fitness:
                global_best_fitness = fit
                global_best = population[i].copy()
                
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Refinement with gradient descent
    best_h = global_best.copy()
    
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    result = minimize(objective, best_h, method='SLSQP',
                     bounds=bounds, constraints=constraint,
                     options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6})
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
