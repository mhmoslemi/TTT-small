import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function from a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = int(1e5)
    eta = 0.01  # Learning rate for gradient step
    lambda1 = 100.0
    lambda2 = 100.0
    
    # Precompute indices
    lags = np.arange(-n_points + 1, n_points)
    
    def compute_energy(h_vec):
        C5 = evaluate_overlap(h_vec, dx)
        # Penalty terms
        bound_penalty = np.sum(np.maximum(0, h_vec - 1)**2 + np.maximum(0, -h_vec)**2)
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return C5 + lambda1 * bound_penalty + lambda2 * sum_penalty
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current energy
        E_current = compute_energy(h)
        
        # Perturb h with Gaussian noise
        h_new = h + np.random.normal(0, eta, n_points)
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Compute new energy
        E_new = compute_energy(h_new)
        
        # Accept or reject
        delta_E = E_new - E_current
        if delta_E < 0 or np.random.random() < np.exp(-delta_E / T):
            h = h_new
            if E_new < best_c5:
                best_c5 = E_new
                best_h = h.copy()
        
        # Cool down
        T = max(T * cooling_rate, min_temp)
    
    # Gradient-based refinement
    if time.time() - start_time < budget_s - 5:
        def gradient_step(h_vec):
            # Compute gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h_vec)
            base_energy = compute_energy(h_vec)
            
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (target_sum / h_plus_sum)
                energy_plus = compute_energy(h_plus)
                grad[i] = (energy_plus - base_energy) / eps
            
            return grad
        
        # Perform gradient descent steps
        for step in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = gradient_step(best_h)
            best_h = best_h - 0.001 * grad
            best_h = np.clip(best_h, 0, 1)
            
            # Renormalize
            best_h_sum = np.sum(best_h)
            if best_h_sum > 0:
                best_h = best_h * (target_sum / best_h_sum)
            
            # Track best
            current_c5 = evaluate_overlap(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
