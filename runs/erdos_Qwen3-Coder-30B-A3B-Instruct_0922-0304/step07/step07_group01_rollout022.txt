import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint exactly
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
    
    h = np.clip(h, 0, 1)
    h = project_to_constraints(h, target_sum)

    # Create frequency domain optimization problem
    # Use CVXPY for convex optimization in frequency domain
    
    # Fourier transform
    H = np.fft.fft(h)
    
    # Variables to optimize
    H_real = cp.Variable(n_points)
    H_imag = cp.Variable(n_points)
    
    # Objective: minimize maximum overlap in frequency domain
    # C = max_k [ (h ⋆ (1 - h))[k] * dx ]
    # Using convolution theorem: h ⋆ (1 - h) = F^{-1}(H * (1 - H))
    
    # Set up problem with constraints
    # Define frequency domain variables
    H_var = H_real + 1j * H_imag
    
    # Objective: minimize max overlap
    obj = cp.Minimize(cp.max(cp.abs(cp.fft.ifft(H_var * (1 - cp.fft.ifft(H_var)))))
    
    # Constraints: 
    # 1. Sum constraint in time domain
    # 2. Bounds on h values
    
    # Build constraints 
    constraints = []
    
    # Real part constraints for H (approximate: we want the h to be real)
    # Common to use:
    
    # Reconstruct h from frequency variables
    h_reconstructed = cp.real(cp.fft.ifft(H_var))
    
    # Sum constraint
    constraints.append(cp.sum(h_reconstructed) == target_sum)
    
    # Bound constraints
    for i in range(n_points):
        constraints.append(h_reconstructed[i] >= 0)
        constraints.append(h_reconstructed[i] <= 1)
    
    # Use only part of the problem as a quick heuristic:
    # Take the better of two strategies:
    # 1. Local refinement with projected gradient descent (faster, but may get stuck)
    # 2. Basic convex approach
    
    # Strategy 1: Gradient descent refinement
    h_best = h.copy()
    c5_best = compute_c5(h_best, dx)
    
    # Simple local search, similar to previous code but with better init
    for iteration in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbations with constraint projection
        h_new = h_best + np.random.normal(0, 0.01, n_points) 
        h_new = project_to_constraints(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            
        if c5_best <= 0.3807:
            break
    
    # Ensure final value is valid
    h_final = project_to_constraints(h_best, target_sum)
    c5_final = compute_c5(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
