import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    max_iterations = 10000
    decay_factor = 0.999
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        g = np.zeros(n_points)
        # Gradient computation via finite difference (simplified)
        for i in range(n_points):
            # Direct computation of gradient wrt h[i]
            grad_val = 0.0
            for k in range(n_points):
                if i + k < n_points:
                    # \frac{d}{dh_i} h[i](1-h[i+k]) = (1-h[i+k]) if k=0, -h[i] if k=i
                    # More precisely, for each shift k, contribution to overlap_k is h[i](1-h[i+k])
                    # If k is such that i+k < n_points, this contributes 
                    # for i=0, k=0 contributes 1-h[0] to dC5/dh[0]
                    # for i=1, k=1 contributes 1-h[2] to dC5/dh[1]
                    # But we want the actual gradient of max_k over all shifts
                    # Direct computation: finite difference approximation
                    if i == k_max:
                        # for the current max shift k_max
                        if i < n_points - k_max:
                            # Only if i+k_max is valid, then contribution to gradient
                            grad_val += (1 - h[i + k_max]) * dx
                        else:
                            # This is approximation, treat as zero or negligible
                            pass
                    elif i + k_max < n_points and k_max == 0:
                        # This case might not be exact, but estimate
                        grad_val += -h[i] * dx
                    else:
                        # Simplified: gradient of the maximum element for stochastic
                        # attending to range of possible contributions
                        if i + k_max < n_points: 
                            # Approximate gradient toward reducing the overlap
                            grad_val += (1-h[i+k_max]) * dx
            g[i] = grad_val
            
        # Update with Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iter_count+1))
        v_hat = v / (1 - beta2**(iter_count+1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h, ensuring feasibility
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-8))
        
        # Check for improvement
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adapt learning rate
        if iter_count % 1000 == 0:
            lr *= decay_factor
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
