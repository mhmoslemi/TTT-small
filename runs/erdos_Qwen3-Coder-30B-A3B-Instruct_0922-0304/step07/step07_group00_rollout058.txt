import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def update_h(h, i, alpha, target_sum):
    """Update h by moving along coordinate i."""
    h_old = h[i]
    h[i] = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
    scaling = target_sum / np.sum(h)
    h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Track optimization progress
    last_improvement = 0
    iteration = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlap and current maximum
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Find indices of maximum overlap
        max_indices = np.where(np.abs(overlap - current_max) < 1e-10)[0]
        
        # Select top 10% of coordinates by gradient magnitude
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Approximate gradient estimate (we use a simplified variant)
            grad[i] = overlap[i] * (1 - 2 * h[i])
        
        # Sort indices by absolute gradient magnitude
        sorted_indices = np.argsort(np.abs(grad))[::-1]
        top_indices = sorted_indices[:max(1, n_points // 10)]
        
        # Choose best improvement direction
        best_direction = None
        best_alpha = 0.0
        best_h_candidate = h.copy()
        best_new_c5 = best_c5
        
        for i in top_indices:
            # Estimate gradient direction
            if np.abs(grad[i]) > 1e-10:
                # Maximize reduction in overlap
                alpha_candidate = min(1.0, max(0.0, -grad[i] * 0.1))
                
                if alpha_candidate > 0.0:
                    h_candidate = h.copy()
                    h_candidate = update_h(h_candidate, i, alpha_candidate, target_sum)
                    new_c5 = compute_c5(h_candidate, dx)
                    
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_direction = i
                        best_alpha = alpha_candidate
                        best_h_candidate = h_candidate.copy()
        
        # Apply the improvement if found
        if best_direction is not None:
            h = best_h_candidate.copy()
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
                last_improvement = 0
        else:
            # No improvement found, reduce step size or try random search
            last_improvement += 1
            
        # Early termination on target
        if best_c5 <= 0.3807:
            break
            
        # Steady state detection
        if last_improvement > max_no_improvement:
            break
            
        iteration += 1
        
        # Periodic full recomputation of overlaps for better precision
        if iteration % 100 == 0:
            overlap = correlate(h, 1 - h, mode="full") * dx
            current_max = np.max(overlap)
    
    return (best_h.tolist(), best_c5, n_points)
