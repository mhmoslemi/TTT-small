import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Problem, Minimize, Variable, sum as cp_sum, norm
from scipy.optimize import minimize

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_bounds(h):
    return np.clip(h, 0, 1)

def project_to_sum(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        h += diff / len(h)
        h = project_to_bounds(h)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def convex_relaxation(n_points):
    h = Variable(n_points)
    target_sum = n_points / 2
    C = Variable()
    
    # Objective: minimize C
    objective = Minimize(C)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp_sum(h) == target_sum
    ]
    
    # Add convolution constraints
    # Since this is a complex non-convex constraint, we simplify to:
    # Minimize max of convolutions - but this is still non-convex.
    # For now, we simplify and solve a reasonable approximation
    # with iterative refinement.
    
    problem = Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, verbose=False)
    
    if problem.status == 'optimal':
        return h.value
    else:
        return None

def find_best_step_function(h_continuous, n_points, target_sum):
    # Convert continuous solution to step function
    adjusted_h = project_to_bounds(h_continuous)
    adjusted_h = project_to_sum(adjusted_h, target_sum)
    return adjusted_h

    # Perturb step locations using gradient descent
    h = h.copy()

    # Simple shape-based optimization
    # Try moving local maxima or change points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    improvement = True
    
    iterations = 0
    while improvement and iterations < 20:
        improvement = False
        for i in range(1, n_points-1):
            # Try shifting the step at position i
            h_trial = h.copy()
            h_trial[i] = max(0, min(1, h[i] + (np.random.rand() - 0.5)*0.05))
            h_trial = project_to_sum(h_trial, target_sum)
            c5_new = compute_c5(h_trial, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_trial.copy()

        h = best_h.copy()

    h_convex = convex_relaxation(n_points)

    # Local refinement
    h_refined, current_c5 = local_refinement(h, n_points, target_sum, dx)
