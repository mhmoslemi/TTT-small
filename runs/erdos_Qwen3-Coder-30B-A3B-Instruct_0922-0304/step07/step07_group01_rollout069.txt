import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Read initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = project_to_constraints(h, target_sum)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Trust region parameters
    trust_radius = 0.1
    trust_radius_min = 1e-6
    trust_radius_max = 0.5
    
    # Feasibility search parameters
    iter_count = 0
    max_iter = 1000
    
    # Dynamic constraint tightening
    alpha = 0.1
    alpha_decay = 0.95
    alpha_growth = 1.05
    alpha_max = 0.5
    
    # Gradient computation with finite differences
    def compute_gradient(h, dx, c5_target):
        eps = 1e-5
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h, dx)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    def feasible_search(h, c5_target, dx, alpha):
        # Perform a few iterations of gradient descent with trust region
        current_h = h.copy()
        prev_c5 = compute_c5(current_h, dx)
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute gradient
            grad = compute_gradient(current_h, dx, c5_target)
            # Update with trust region step
            step = -grad * 0.01
            step_norm = np.linalg.norm(step)
            if step_norm > trust_radius:
                step = step * trust_radius / step_norm
            # Try the update
            new_h = current_h + step
            new_h = project_to_constraints(new_h, target_sum)
            new_c5 = compute_c5(new_h, dx)
            
            # Accept step if improves constraint satisfaction
            if new_c5 <= c5_target * (1 + alpha):
                current_h = new_h
                prev_c5 = new_c5
                # Increase trust radius if improvement
                trust_radius = min(trust_radius_max, trust_radius * 1.1)
            else:
                # Decrease trust radius
                trust_radius = max(trust_radius_min, trust_radius * 0.5)
        
        return current_h, prev_c5
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Binary search approach with dynamic tightness
        left = 0
        right = best_c5 * 2.0
        best_feasible_c5 = right
        
        # Binary search for the smallest feasible C
        for _ in range(20):
            mid = (left + right) / 2
            test_h, test_c5 = feasible_search(best_h, mid, dx, alpha)
            
            if test_c5 <= mid * (1 + alpha):
                best_feasible_c5 = test_c5
                right = mid
                # Tighten alpha
                alpha = max(1e-4, alpha * alpha_decay)
            else:
                left = mid
                # Relax alpha
                alpha = min(alpha_max, alpha * alpha_growth)
            
            # Check for early termination
            if best_feasible_c5 <= 0.3807:
                break
                
        # Accept the improved solution if found
        if best_feasible_c5 < best_c5 and best_feasible_c5 <= 0.3807:

            best_c5 = best_feasible_c5
            # Reset trust radius
            trust_radius = 0.1

            best_h = test_h
            best_c5 = best_feasible_c5
            # Reset trust radius
            trust_radius = 0.1

            perturb = np.random.normal(0, 0.01, n_points)
            test_h = best_h + perturb

            test_c5 = compute_c5(test_h, dx)

                best_h = test_h
                best_c5 = test_c5

            current_h = project_to_constraints(best_h, target_sum)
            best_c5 = compute_c5(current_h, dx)
            best_h = current_h
            
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
