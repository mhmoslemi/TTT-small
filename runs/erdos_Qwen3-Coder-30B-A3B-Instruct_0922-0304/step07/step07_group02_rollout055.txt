import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Distribute the difference evenly
    h = h + diff / len(h)
    # Clamp and renormalize to preserve constraints
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start from initial_h_values if available, otherwise use a good heuristic
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Heuristic initial guess: step function with good overlap characteristics
        h = np.zeros(n_points)
        # Place a step at the middle with half the interval
        h[:n_points//2] = 1
        # Add noise for local search
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # PGD with momentum
    momentum = 0.9
    learning_rate = 0.1
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    last_improvement_iter = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute all overlaps for all possible shifts
        overlaps = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlaps)
        max_k = max_idx - (len(h) - 1)  # Convert index to shift value
        # Compute gradient: (1 - h) shifted by max_k
        h_shifted = np.roll(1 - h, max_k)
        grad = h_shifted
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Enforce bounds and sum constraint
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
        
        # Check for improvement
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement_iter = iteration

        # Adaptive learning rate
        if iteration - last_improvement_iter > 100:
            learning_rate *= 0.95
            if learning_rate < 1e-5:
                learning_rate = 1e-5
            last_improvement_iter = iteration
            
        iteration += 1
        
        # Early stopping if we've reached the target
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
