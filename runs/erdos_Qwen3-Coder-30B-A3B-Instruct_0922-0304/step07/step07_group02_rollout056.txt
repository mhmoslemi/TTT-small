```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Simply adjust all values equally to balance
    h = np.clip(h + diff / len(h), 0, 1)
    # Scale to ensure exact sum
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Try to use initial_h_values if available, otherwise fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Initialize with something reasonable
        h = np.ones(n_points) * 0.5
        h = project_to_constraint(h, target_sum)
    
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    max_iter = 3000
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # For momentum
    velocity = np.zeros_like(h)
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # If the maximum shift is in the middle region, it's the best one
        # Compute the gradient for that shift
        # The gradient is (1 - h) shifted by max_k, but we need to apply
        # the shifting in the correct way.
        # The overlap is computed as sum(h[i] * (1 - h[i + k]))
        # Its gradient w.r.t h[j] touches the indices involved in overlap with shift k
        
        # We compute the gradient here simply via finite difference:
        # ∇C5 = - (1 - h) shifted by k
        gradient = np.zeros_like(h)
        
        # The gradient at each i is the sum of negative terms from the shift k
        # That is for overlap at shift k, the gradient contribution at i is:
        # w.r.t. h[i], it is - (1 - h[i + k])
        # But more accurately, since we are differentiating sum(h[i] * (1 - h[i + k])):
        # d/dh[i] = 1 - h[i + k] in full convolution
                
        # This implements: ∇C5[i] = - (1 - h[i + k]) 
        # So we need to compute shift_k(1 - h) partial
        #Overlap takes 1-h as second argument, so we ask for convolution of h with 1-h
        # The gradient w.r.t. h is the convolution of (1-h) with shifted impulses
        # Thus: ∇C5 = 1 - h_k where h_k means h shifted by k
        # i.e. ∇C5[i] = - (1 - h[i+k]) 
        # But we want gradient of sum(h[i]**2 - h[i]*h[i+k]) 
        # But this is really the gradient of ∫ h(x) (1 - h(x+k)) ...
        # For continuous derivation, this would equate to (1 - h(x+k))
        # But since we're aligning it spatially:
        # d/dh[i] (h[i] * (1 - h[i + k])) = (1 - h[i + k])
        # But since we're up to sign, and we want to minimize, it's + gradient.

        # More accurately:  
        # Given: integral of h(x) * (1 - h(x + k))
        # d/dh[i] (integrand on i-th sample) = (1 - h[i + k])
        # So: ∇C5[i] = -(1 - h[i + k]) 
        # But truncate for all but those that are valid with correct indexing
        
        grad_vec = np.zeros_like(h)
        
        # This is the right way for a convolution: gradient of overlap calculation
        # For overlap sum from i of h[i] * (1 - h[i+k]), we have:
        # d/dh_j (h[i] * (1 - h[i+k])) = 0 if i != j  
        # d/dh_j (h[i] * (1 - h[i+k])) = (1 - h[i+k]) if i = j and j is valid
        # i.e. it makes sense if we go through the array and compute for each k, 
        # particular:  for overlap at shift k, we add at index i the term (1 - h[i + k])
        # ... so gradient at i is sum over all k, etc.
        
        # target k is max_k
        # shift operation: h_{k}[i] = h[i-k] for positive k
        # so we need to accumulate contribution of 1-h from i+k into grad_vec[i]
        # i.e. grad_vec[i+k] += 1 - h[i] 
        # unless this is out of bounds
        
        # Let's just compute shifted version of (1 - h) 
        shifted_h = 1 - h
        shifted_h = np.roll(shifted_h, max_k)
        
        # The gradient should be the negative of that
        gradient = -shifted_h
        
        # Apply momentum
        velocity = momentum * velocity + gradient
        h = h - learning_rate * velocity
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Project to correct sum
        h = project_to_constraint(h, target_sum)
        
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-5:
                break
            patience_counter = 0
        
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
```
