import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a good starting point
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use evolutionary strategy (CMA-ES inspired) to improve
    # We'll do a simple mutation approach
    
    iteration = 0
    no_improvement_count = 0
    max_no_improve = 50
    
    while time.time() - start_time < budget_s - 5 and no_improvement_count < max_no_improve:
        # Create a new candidate by mutating current best
        h_new = h.copy()
        
        # Randomly perturb some entries
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            # Add small random perturbation
            perturb = np.random.normal(0, 0.05)
            h_new[i] += perturb
            
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        new_sum = np.sum(h_new)
        if new_sum > 0:
            h_new = h_new * (target_sum / new_sum)
        
        # Evaluate
        new_c5 = evaluate_overlap(h_new, dx)
        
        # Accept if better (or occasionally if worse to escape local minima)
        if new_c5 < best_c5 or np.random.random() < 0.05:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        iteration += 1
    
    # Final refinement with local search
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try different starting points
        best_local_h = h.copy()
        best_local_c5 = best_c5
        
        for _ in range(5):
            # Start from current best with some noise
            start_h = h + np.random.normal(0, 0.01, n_points)
            start_h = np.clip(start_h, 0, 1)
            start_sum = np.sum(start_h)
            if start_sum > 0:
                start_h = start_h * (target_sum / start_sum)
            
            result = minimize(objective, start_h, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-8})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_local_c5:
                    best_local_h = refined_h
                    best_local_c5 = final_c5
        
        if best_local_c5 < best_c5:
            best_h = best_local_h
            best_c5 = best_local_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
