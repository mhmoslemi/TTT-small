import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = int(1e5)
    
    # Penalty weights
    lambda1 = 100.0
    lambda2 = 100.0
    
    # Learning rate for gradient descent refinement
    learning_rate = 0.01
    
    # Energy calculation function
    def compute_energy(h_vec):
        # Normalize
        h_norm = h_vec * (target_sum / np.sum(h_vec)) if np.sum(h_vec) > 0 else h_vec
        
        # Compute overlap
        overlap = correlate(h_norm, 1 - h_norm, mode="full")
        c5_val = np.max(overlap) * dx
        
        # Penalties
        bound_penalty = np.sum(np.maximum(0, h_norm - 1)**2 + np.maximum(0, -h_norm)**2)
        sum_penalty = (np.sum(h_norm) - target_sum)**2
        
        # Total energy
        energy = c5_val + lambda1 * bound_penalty + lambda2 * sum_penalty
        return energy, c5_val
    
    # Perturbation with gradient guidance
    def perturb(h_vec, T_val, step_size=0.02):
        # Compute gradient (approximate)
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        base_energy, _ = compute_energy(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            energy_plus, _ = compute_energy(h_plus)
            grad[i] = (energy_plus - base_energy) / eps
        
        # Apply gradient-guided perturbation
        noise = np.random.normal(0, step_size, len(h_vec))
        perturbed = h_vec - learning_rate * grad + noise * T_val
        
        # Clip and renormalize
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 0:
            perturbed = perturbed * (target_sum / perturbed_sum)
        
        return perturbed
    
    # Main annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Get new candidate solution
        h_new = perturb(h, T)
        
        # Evaluate energies
        old_energy, old_c5 = compute_energy(h)
        new_energy, new_c5 = compute_energy(h_new)
        
        # Accept or reject
        delta_E = new_energy - old_energy
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new.copy()
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Update temperature
        T = max(T * cooling_rate, min_temp)
        
        # Occasionally lower learning rate
        if i % 1000 == 0 and learning_rate > 1e-4:
            learning_rate *= 0.95
    
    # Gradient descent refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        # Gradient descent for final polishing
        for i in range(200):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute finite differences approximation to gradient
            eps = 1e-4
            grad = np.zeros_like(h)
            base_val = objective(h)
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (target_sum / h_plus_sum)
                val_plus = objective(h_plus)
                grad[j] = (val_plus - base_val) / eps
            
            # Step
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (target_sum / h_sum)
            
            # Check for improvement
            curr_val = objective(h)
            if curr_val < best_c5:
                best_h = h.copy()
                best_c5 = curr_val
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
