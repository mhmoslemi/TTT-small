import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, max_iter=1000):
    """Use L-BFGS-B for local optimization."""
    def objective(x):
        return compute_c5(x, dx)
    
    def grad(x):
        # Numerical gradient approximation
        eps = 1e-6
        grad_val = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad_val[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad_val
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    def jacobian(x):
        return np.ones_like(x)
    
    # Bounds
    bounds = [(0, 1) for _ in range(len(h))]
    
    #_constraints = {'type': 'eq', 'fun': constraint, 'jac': jacobian}
    
    try:
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            #constraints=constraints,
            jac=grad,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if result.success:
            return project_to_constraints(result.x, target_sum)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # ES parameters
    pop_size = 20
    num_generations = 2000
    sigma_initial = 0.1
    sigma_final = 0.01
    elite_size = pop_size // 2
    local_opt_threshold = 0.381
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
            ind += np.random.normal(0, sigma_initial, n_points)
        else:
            ind = np.random.normal(0.5, 0.2, n_points)
        ind = np.clip(ind, 0, 1)
        ind = normalize_h(ind, target_sum)
        population.append(ind)
    
    # Evaluate initial population
    fitnesses = [compute_c5(ind, dx) for ind in population]
    
    # Evolution loop
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check if we've reached target
        if fitnesses[0] <= 0.3807:
            break
        
        # Keep elite
        elites = population[:elite_size]
        
        # Generate offspring through mutation
        offspring = []
        sigma = sigma_initial - (sigma_initial - sigma_final) * gen / num_generations
        
        for _ in range(pop_size - elite_size):
            parent = elites[np.random.randint(0, elite_size)]
            mutant = parent + np.random.normal(0, sigma, n_points)
            mutant = np.clip(mutant, 0, 1)
            mutant = normalize_h(mutant, target_sum)
            offspring.append(mutant)
        
        # Replace worst individuals
        population = elites + offspring
        
        # Evaluate offspring
        fitnesses = [compute_c5(ind, dx) for ind in population]
        
        # Local optimization for best individuals
        if gen % 10 == 0:  # Check every 10 generations
            for i, ind in enumerate(population):
                if fitnesses[i] < local_opt_threshold:
                    optimized = local_optimize(ind, target_sum, dx)
                    optimized_fitness = compute_c5(optimized, dx)
                    if optimized_fitness < fitnesses[i]:
                        population[i] = optimized
                        fitnesses[i] = optimized_fitness
    
    # Final best solution
    final_fitnesses = [compute_c5(ind, dx) for ind in population]
    best_idx = np.argmin(final_fitnesses)
    best_solution = population[best_idx]
    best_fitness = final_fitnesses[best_idx]
    
    return (best_solution.tolist(), best_fitness, n_points)
