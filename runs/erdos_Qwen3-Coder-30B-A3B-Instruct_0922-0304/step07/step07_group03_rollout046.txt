import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:

        h = np.full(n_points, 0.5)  # Initial simple pattern
        h = project_to_constraints(h, target_sum)

    # AM parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    tau = 1000
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlation
        c = correlate(h, 1 - h, mode="full")
        current_max = np.max(c)
        
        # Calculate gradient direction
        # Gradient of h[i] = - (1 - h[i-1]) + (1 - h[i+1]) part
        # We consider a slightly simplified version for gradients
        # Approximate gradient based on correlation and shifts
        grad = np.zeros_like(h)
        for i in range(n_points):
            left_sum = 0
            right_sum = 0
            for j in range(n_points):
                k = i - j
                if 0 <= k < len(c):
                    # The correlation term include 1-h at shifted positions
                    if 0 <= j < n_points:
                        left_sum += (1 - h[j]) * c[k]
            grad[i] = -left_sum * dx
            
        # Update using gradient descent
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, target_sum)
        
        # Add random perturbations
        perturbation = epsilon * (2 * np.random.rand(n_points) - 1)
        h_candidate = h_new + perturbation
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate the candidate
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Reduce learning parameters
            alpha = alpha * 0.999
            epsilon = epsilon * 0.999
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Apply decay to hyperparameters
        alpha = max(0.01, alpha / (1 + iteration / tau))
        epsilon = max(0.001, epsilon / (1 + iteration / tau))

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
