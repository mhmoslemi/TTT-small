import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        h[:n_points//2] = 1
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize the initial function to satisfy sum constraint
    h = project_to_constraints(h, target_sum)
    
    # Stage 1: convex relaxation with CVXPY
    # Set up optimization variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Define constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
        C >= 0
    ]
    
    # Add constraints for all lags (this is a simplified approach)
    # Let's build a moderate approximation: we'll take every 3rd lag to speed up
    overlap_cons = []
    for k in range(0, 2*n_points-1, 3):
        if k < n_points:
            # Real part for first half of h
            if k < n_points:
                shift_h = np.concatenate([np.zeros(k), h[:n_points-k]])
            else:
                shift_h = np.zeros(n_points)
        else:
            # Second part of full correlation
            shift_h = np.concatenate([h[k-n_points+1:], np.zeros(n_points - (k-n_points+1))])
        
        # Constraint that the convolution at this lag is bounded by C
        if k < n_points:
            overlap = cp.sum(h_var * (1 - shift_h))
        else:
            overlap = cp.sum(h_var * (1 - shift_h))
        constraints.append(overlap <= C)
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve the convex relaxation (this is a reduced version for approximating the problem)
    try:
        prob.solve(solver=cp.SCS, verbose=False)

            h_relaxed = h_var.value
            # Convert relaxed solution to step function by thresholding
            h = project_to_constraints(h_relaxed, target_sum)
    except:
        pass  # If CVXPY optimization fails, proceed with gradient descent
        
    # Stage 2: gradient descent refinement
    max_iter = 1000
    learning_rate = 0.01
    decay_rate = 0.995
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Inner loop for gradient descent
    for iter_num in range(max_iter):
        # Compute overlap
        overlap_full = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(overlap_full)
        
        # Compute gradient
        grad = np.zeros_like(h)

            h_plus = h.copy()

            h_plus = project_to_constraints(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
