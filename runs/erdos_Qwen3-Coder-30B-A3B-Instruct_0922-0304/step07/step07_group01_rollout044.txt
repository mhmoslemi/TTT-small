import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

    h = project_to_constraints(h, target_sum)
    
    # Start with the current solution as baseline
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent setup
    max_iter = 1000
    lr = 0.01
    max_trust_radius = 0.1
    min_trust_radius = 1e-6
    trust_radius = max_trust_radius
    patience = 10
    patience_counter = 0

    def compute_gradient(h, dx):

        grad = np.zeros_like(h)

            c5_plus = compute_c5(h_plus, dx)

            h_plus[i] = h_minus[i] = h[i]  # Reset
