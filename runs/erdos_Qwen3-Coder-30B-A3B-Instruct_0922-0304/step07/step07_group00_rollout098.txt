def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k):
    """Compute the gradient of overlap_k at h."""
    n = len(h)
    grad = np.zeros(n)
    # For overlap_k = sum(h[i] * (1 - h[i+k])), gradient w.r.t. h[i] is (1 - 2*h[i+k]) if i+k < n
    for i in range(n):
        if i + k < n:
            grad[i] = 1 - 2 * h[i + k]
    return grad

def backtracking_line_search(h, k_max, grad, step, max_iter=10):
    """Backtrack to find a good step size."""

    new_h = np.clip(new_h, 0, 1)
    new_h = normalize_h(new_h, target_sum)
    new_c5 = compute_c5(new_h, dx)
    
    for _ in range(max_iter):
        if new_c5 < old_c5:
            return alpha, new_h
        alpha *= 1/2
        new_h = h + alpha * (step - h)

        new_h = normalize_h(new_h, target_sum)
        new_c5 = compute_c5(new_h, dx)
    
    return alpha, new_h
