import numpy as np
from scipy.signal import correlate
import time
from scipy.stats import multivariate_normal

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    mutation_scale = 0.1
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    # Add the initial solution
    means.append(h.copy())
    covariances.append(np.eye(n_points) * mutation_scale**2)
    weights.append(1.0)
    
    # Add random initial components
    for _ in range(n_components - 1):
        init_h = np.random.rand(n_points)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, target_sum)
        means.append(init_h.copy())
        covariances.append(np.eye(n_points) * mutation_scale**2)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iter = 6000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample new candidates
        candidates = []
        candidate_c5s = []
        
        for _ in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            mean = means[comp_idx]
            cov = covariances[comp_idx]
            sample = np.random.multivariate_normal(mean, cov)
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            c5 = compute_c5(sample, dx)
            candidates.append(sample)
            candidate_c5s.append(c5)
        
        # Sort and keep best
        sorted_indices = np.argsort(candidate_c5s)
        top_candidates = [candidates[i] for i in sorted_indices[:keep_top]]
        top_c5s = [candidate_c5s[i] for i in sorted_indices[:keep_top]]
        
        # Update best
        if top_c5s[0] < best_c5:
            best_c5 = top_c5s[0]
            best_h = top_candidates[0].copy()
        
        # Update GMM
        new_means = []
        new_covariances = []
        new_weights = []
        
        for i in range(n_components):
            if i < len(top_candidates):
                # Use top candidate to update component
                new_means.append(top_candidates[i])
                # Update covariance
                diff = np.array(top_candidates) - means[i]
                new_cov = np.cov(diff.T) + np.eye(n_points) * 1e-8  # Regularize
                new_cov = covariance_decay * covariances[i] + (1 - covariance_decay) * new_cov
                new_covariances.append(new_cov)
                new_weights.append(1.0)
            else:
                # Keep old component if no update
                new_means.append(means[i])
                new_covariances.append(covariances[i])
                new_weights.append(0.0)
        
        # Normalize weights
        total_weight = sum(new_weights)
        if total_weight > 0:
            new_weights = [w / total_weight for w in new_weights]
        else:
            new_weights = [1.0/len(new_weights)] * len(new_weights)
        
        means = new_means
        covariances = new_covariances
        weights = np.array(new_weights)
        
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
