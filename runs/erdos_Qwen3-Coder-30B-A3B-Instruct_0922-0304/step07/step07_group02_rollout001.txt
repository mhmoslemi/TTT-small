import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Gradient-based optimization with projected gradients
    max_iter = 5000
    learning_rate = 0.01
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using finite differences
        grad = np.zeros(n_points)
        epsilon = 1e-5
        
        current_c5 = compute_c5(h, dx)
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += epsilon
            h_eps = np.clip(h_eps, 0, 1)
            h_eps = project_to_constraint(h_eps, target_sum)
            
            c5_eps = compute_c5(h_eps, dx)
            grad[i] = (c5_eps - current_c5) / epsilon
        
        # Update h with gradient descent
        h_new = h - learning_rate * grad
        
        # Project back to valid region
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraint(h_new, target_sum)
        
        # Update h
        h = h_new
        
        # Update best solution if improved
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Check for convergence
        if abs(new_c5 - current_c5) < tolerance:
            break
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
