import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def local_refinement(h, dx, target_sum, max_iter=50):
    def objective(x):
        h_temp = x.copy()
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        return compute_c5(h_temp, dx)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    # Use scipy minimize with SLSQP for local refinement
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options={'maxiter': max_iter})
    
    return project_to_constraints(result.x, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a reasonable initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Convex relaxation to get starting point
    h_cvx = cp.Variable(n_points)
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == target_sum]
    C = cp.Variable()
    # Add constraint for convolution
    # For simplicity, use a heuristic approach to estimate the maximum
    
    # Since we're using CVXPY, avoid full convolution in constraint form for now
    # Instead solve a simpler: minimize max values (we'll evaluate it separately)
    
    # As a workaround, we'll still do just a local refinement 
    # But initialize with something more structured
    
    # Instead let's do a hybrid with structured search
    # Try different step positions
    for n_steps in [2, 3, 4]:
        if time.time() - start_time > budget_s - 5:
            break
            
        step_positions = np.linspace(0, n_points, n_steps+1, dtype=int)
        step_widths = np.diff(step_positions)
        max_width = max(step_widths) if len(step_widths) > 0 else 1
        
        # Try different heights for each step
        for height in np.linspace(0.1, 1.0, 10):
            h_temp = np.zeros(n_points)
            if n_steps == 1:
                h_temp[:] = height
            else:
                for i in range(n_steps):
                    start_idx = step_positions[i]
                    end_idx = step_positions[i+1]
                    h_temp[start_idx:end_idx] = height
                
            h_temp = project_to_constraints(h_temp, target_sum)
            c5 = compute_c5(h_temp, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_temp.copy()
    
    # Local refinement
    refinement_h = local_refinement(best_h, dx, target_sum)
    refinement_c5 = compute_c5(refinement_h, dx)
    if refinement_c5 < best_c5:
        best_c5 = refinement_c5
        best_h = refinement_h

    # Fine tune with particle swarm optimization for remaining time
    n_particles = 30
    inertia_weight_start = 0.9
    inertia_weight_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.05

    particles = np.array([best_h.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-velocity_max, velocity_max, (n_particles, n_points))
    pbest_positions = particles.copy()
    pbest_values = np.array([compute_c5(p, dx) for p in particles])

    gbest_index = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_index].copy()
    gbest_value = pbest_values[gbest_index]

    iterations = 0
    max_iter = 1000

    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (iterations / max_iter)

        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] +
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (gbest_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            particles[i] += velocities[i]
            particles[i] = project_to_constraints(particles[i], target_sum)
            
            current_c5 = compute_c5(particles[i], dx)
            
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
                
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = particles[i].copy()
        
        if gbest_value <= 0.3807:
            break
            
        iterations += 1

    # Final evaluation
    final_c5 = compute_c5(gbest_position, dx)
    return (gbest_position.tolist(), final_c5, n_points)
