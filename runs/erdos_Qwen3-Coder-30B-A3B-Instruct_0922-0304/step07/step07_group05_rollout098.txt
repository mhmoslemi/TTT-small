import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from available construction or create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Triangular pulse centered at 1
        h = np.zeros(n_points)
        midpoint = n_points // 2
        for i in range(n_points):
            dist = abs(i - midpoint)
            h[i] = max(0, 1 - 2 * dist / n_points)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    patience = 0
    max_patience = 10
    max_iterations = 60
    iteration = 0
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        if patience >= max_patience:
            break
            
        # Formulate and solve quadratic program to find descent direction
        # Define QP variables
        h_var = cp.Variable(n_points)
        # Objective: approximate C5 at current point
        # For simplicity, we'll use a more direct approach for optimization
        
        # Calculate gradient approximations using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            val_plus = compute_c5(h_plus, dx)
            val_minus = compute_c5(h_minus, dx)
            grad[i] = (val_plus - val_minus) / (2 * eps)
        
        # Form QP: minimize 0.5 * d^T * H * d + grad^T * d
        # We'll approximate as a simple descent direction using gradient
        
        # Standard gradient descent direction
        direction = -grad
        
        # Line search to determine step size
        step_size = 1.0
        max_linesearch = 20
        for _ in range(max_linesearch):
            new_h = h + step_size * direction
            new_h = project_to_feasible(new_h, target_sum)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                h = new_h
                c5 = new_c5
                patience = 0
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            step_size *= 0.5
        else:
            # Even if no improvement in linesearch, try a small step
            new_h = h + 0.1 * direction
            new_h = project_to_feasible(new_h, target_sum)
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                
        iteration += 1
    
    # Final refinement with a few iterations of local optimization
    # Using scipy minimize with constraints
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    h = best_h.copy()
    for _ in range(5):  # few rounds of local refinement
        result = minimize(objective, h, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            h = result.x
            h = project_to_feasible(h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(h, dx)
    if final_c5 < best_c5:
        best_h = h.copy()
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
