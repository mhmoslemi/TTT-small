import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use local gradient-based optimization for refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Try multiple random restarts with SLSQP
    for _ in range(10):
        if time.time() - start_time > budget_s - 5:
            break
        # Random warmup
        h_restart = np.random.rand(n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart_sum = np.sum(h_restart)
        if h_restart_sum > 0:
            h_restart = h_restart * (target_sum / h_restart_sum)
        
        result = minimize(objective, h_restart, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Try a more aggressive refinement with L-BFGS-B if time permits
    if time.time() - start_time < budget_s - 5:
        h_restart = best_h.copy()
        result = minimize(objective, h_restart, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 200})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
