import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to sum to n_points/2
    h = h / np.sum(h) * (n_points / 2)
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current C5 and gradient
        h = sigmoid(w)
        h = h / np.sum(h) * (n_points / 2)
        
        # Clip to prevent numerical issues
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h))
        
        # Compute overlap and find max shift
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient contribution
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                g[i] += (1 - 2 * h[i]) * (1 - h[i + k_max])
        
        g *= dx
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply thresholding and normalization
        h = sigmoid(w)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h))
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
