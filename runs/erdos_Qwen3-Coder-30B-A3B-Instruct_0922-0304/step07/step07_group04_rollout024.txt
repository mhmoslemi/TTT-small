import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try different approaches to improve solution
    
    # 1. Local refinement with different algorithms
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try multiple local search strategies
        methods = ['SLSQP', 'L-BFGS-B', 'TNC']
        for method in methods:
            try:
                result = minimize(objective, best_h.copy(), method=method, 
                                bounds=bounds, constraints=constraints, 
                                options={'maxiter': 100, 'ftol': 1e-8})
                if result.success:
                    refined_h = np.clip(result.x, 0, 1)
                    refined_sum = np.sum(refined_h)
                    if refined_sum > 0:
                        refined_h = refined_h * (target_sum / refined_sum)
                    final_c5 = evaluate_overlap(refined_h, dx)
                    if final_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = final_c5
            except:
                continue
    
    # 2. Simple grid search to find promising regions
    if time.time() - start_time < budget_s - 5:
        # Do local search around promising points
        for _ in range(20):
            # Create perturbed version
            pert_h = best_h.copy() + np.random.normal(0, 0.01, n_points)
            pert_h = np.clip(pert_h, 0, 1)
            # Renormalize
            pert_sum = np.sum(pert_h)
            if pert_sum > 0:
                pert_h = pert_h * (target_sum / pert_sum)
            pert_c5 = evaluate_overlap(pert_h, dx)
            if pert_c5 < best_c5:
                best_h = pert_h
                best_c5 = pert_c5
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
