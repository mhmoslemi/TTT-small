import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial uniform step function
        h = np.full(n_points, 0.5)
        h = project_to_constraint(h, target_sum)
    
    # PGD parameters
    lr = 0.1
    beta = 0.9
    max_iter = 3000
    patience = 100
    patience_counter = 0
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Momentum
    velocity = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient: (1 - h) shifted by max_k
        grad = np.zeros(n_points)
        if max_k < n_points:
            grad[max_k:] = 1 - h[:-max_k]
        else:
            grad[:n_points - max_k] = 1 - h[max_k:]
        
        # Update with momentum
        velocity = beta * velocity - lr * grad
        h = h + velocity
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Project to sum constraint
        h = project_to_constraint(h, target_sum)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter >= patience:
            lr *= 0.95
            lr = max(lr, 1e-5)
            patience_counter = 0
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
