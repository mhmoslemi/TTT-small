import numpy as np
import time

def compute_c5(h, dx):
    # Compute the maximum overlap integral
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx

def project_to_constraints(h, target_sum):
    # Clip to bounds [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to ensure sum equals target_sum
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h
