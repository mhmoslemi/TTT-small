import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum(h, target_sum):
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_sum(h, target_sum)
    else:
        # Initialize with a symmetric step
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        h = project_to_sum(h, target_sum)
    
    # Start with convex relaxation to get a continuous solution
    h_cvx = Variable(n_points)
    constraints = [0 <= h_cvx, h_cvx <= 1, sum(h_cvx) == target_sum]
    
    # Use a slack variable to handle max constraint in convex form
    C = Variable()
    # This is an approximation to the quadratic constraint
    # A convex formulation not easily expressible; so we proceed with taking a stable 
    # solution from a dummy first convex solve and reproject for local search
    
    # Simple discrete relaxation approach using a Greedy ad hoc starting construction
    h = np.clip(h, 0, 1)
    h = project_to_sum(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    current_c5 = best_c5
    
    # Local search with gradient-based refinement using scipy.optimize
    # Using projected gradient descent or similar approach
    
    # Generate candidate solutions using iterative refinement
    max_iter = 1000
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Slightly perturb each sample point within bounds
        perturbed = h.copy() + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = project_to_sum(perturbed, target_sum)
        
        c5 = compute_c5(perturbed, dx)
        if c5 < current_c5:
            h = perturbed
            current_c5 = c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
