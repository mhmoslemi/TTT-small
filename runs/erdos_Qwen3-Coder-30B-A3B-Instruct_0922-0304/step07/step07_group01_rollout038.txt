import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    # Simulated Annealing parameters
    initial_temp = 0.1 * compute_c5(h, dx)
    temperature = initial_temp
    cooling_rate = 0.99
    max_iter = 10000
    iteration = 0
    
    # Current and best solutions
    current_h = h.copy()
    current_c5 = compute_c5(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Generate a candidate solution by perturbing current solution
        candidate_h = current_h + np.random.uniform(-temperature, temperature, n_points)
        candidate_h = project_to_constraints(candidate_h, target_sum)
        
        # Evaluate candidate
        candidate_c5 = compute_c5(candidate_h, dx)
        
        # Accept or reject candidate
        if candidate_c5 < current_c5 or np.random.random() < np.exp((current_c5 - candidate_c5) / temperature):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best solution if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down temperature
        temperature *= cooling_rate
        iteration += 1
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
