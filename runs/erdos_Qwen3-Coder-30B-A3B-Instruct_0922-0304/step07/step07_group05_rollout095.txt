import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum = target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def gradient_approx(h, dx):
    """Approximate gradient of C5 with finite differences."""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    base_c5 = compute_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h)/2)
        c5_plus = compute_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, len(h)/2)
        c5_minus = compute_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def line_search(h, d, dx, c5_current):
    """Perform line search to find optimal step size."""
    def phi(alpha):
        new_h = h + alpha * d
        new_h = project_to_feasible(new_h, len(h)/2)
        return compute_c5(new_h, dx)
    
    # Find approximate minimum using golden section search
    res = minimize_scalar(phi, bounds=(0, 10), method='bounded')
    alpha = res.x
    
    return alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a more structured initial function
        h = np.zeros(n_points)
        center = n_points // 2
        width = n_points // 4
        h[max(0, center - width//2):min(n_points, center + width//2)] = 1
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    patience = 0
    max_patience = 10
    
    for iteration in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient approximation
        grad = gradient_approx(h, dx)
        
        # Solve QP to find search direction
        # Minimize: 0.5 * d^T * H * d + grad^T * d
        # Subject to: sum(d) = 0, 0 <= h + d <= 1
        
        # For simplicity, we'll do a gradient descent step with line search
        d = -grad
        step_size = line_search(h, d, dx, best_c5)
        h = h + step_size * d
        
        # Project back to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if best_c5 <= 0.3807:
            break
            
        if patience >= max_patience:
            break
            
        # Occasionally use a different search strategy
        if iteration % 10 == 0:
            # Small random perturbation
            h += np.random.normal(0, 0.01, n_points)
            h = project_to_feasible(h, target_sum)
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    from scipy.optimize import minimize
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        
        result = minimize(objective, best_h.copy(), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 20})
        if result.success:
            best_h = result.x
            best_h = project_to_feasible(best_h, target_sum)
        else:
            break
            
        new_c5 = compute_c5(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            break
            
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
