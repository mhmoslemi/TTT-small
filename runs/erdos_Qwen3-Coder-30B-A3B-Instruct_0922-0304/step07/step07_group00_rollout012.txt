import numpy as np
from scipy.signal import correlate
from cvxpy import Variable, Minimize, Problem, sum_entries, multiply, norm
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def create_basis_functions(n_points, n_basis):
    """Create a set of basis functions - step functions with varying pulse widths."""
    basis = []
    for i in range(n_basis):
        # Divide the interval [0,2] into 4-8 segments
        segments = np.random.randint(4, 9)
        b = np.zeros(n_points)
        # Randomly place pulses
        for j in range(segments):
            start = np.random.randint(0, n_points)
            width = np.random.randint(1, n_points//8 + 1)
            end = min(start + width, n_points)
            b[start:end] = 1
        # Normalize so total integral equals 1
        b = b * (n_points / (2 * np.sum(b)))
        basis.append(b)
    return basis

def precompute_overlap_integrals(basis):
    """Precompute overlap integrals A_ij^k."""
    n_basis = len(basis)
    n_points = len(basis[0])
    A = np.zeros((n_basis, n_basis, 2*n_points - 1))
    for i in range(n_basis):
        for j in range(n_basis):
            A[i, j] = correlate(basis[i], basis[j], mode="full")
    return A

def decompose_initial_h_into_basis(initial_h, basis, dx, target_sum):
    """Decompose initial_h into the basis space using least squares."""
    n_basis = len(basis)
    # Build matrix
    Phi = np.column_stack(basis)
    # Add regularization to avoid ill-conditioning
    Phi = Phi + 1e-8 * np.random.rand(*Phi.shape)
    # Solve least squares problem min ||Phi lambda - h||_2^2
    # subject to sum(lambda) = 1, lambda >= 0
    # We normalize h to have the correct integral
    h = np.array(initial_h) * (target_sum / np.sum(initial_h))
    # Direct solution using least squares with constraints
    # Formulate as: min (Phi * L - h)^2 such that sum(L) = 1, L >= 0
    # Since not available directly in numpy, we'll use cvxpy for exact solution
    L = Variable(n_basis)
    cost = sum_entries(multiply(Phi.T, Phi) * L) - 2 * L.T @ (Phi.T @ h)
    constraints = [sum_entries(L) == 1, L >= 0]
    prob = Problem(Minimize(cost), constraints)
    prob.solve(solver='ECOS', verbose=False)
    return L.value
