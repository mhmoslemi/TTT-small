import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Transform to log-odds space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all k
        overlap_vals = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_vals)
        
        # Compute gradient w.r.t. h: contribution from the maximal overlap
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                g[i] += (1 - 2 * h[i]) * (1 - h[i + k_max])
            if i - k_max >= 0:
                g[i + k_max] += (1 - 2 * h[i + k_max]) * (1 - h[i])
        
        # Apply gradient to w (since h = sigmoid(w))
        g_w = g * h * (1 - h)  # derivative of sigmoid
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_w
        v = beta2 * v + (1 - beta2) * g_w**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h))
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
