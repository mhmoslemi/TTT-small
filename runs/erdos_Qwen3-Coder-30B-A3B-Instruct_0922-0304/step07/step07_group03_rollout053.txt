import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # params
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 20
    max_iter = 5000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (target_sum / np.sum(h0))
    else:
        h0 = np.random.uniform(0, 1, n_points)
        h0 = h0 * (target_sum / np.sum(h0))
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            individual = h0.copy()
        else:
            individual = h0 + np.random.normal(0, 0.05, n_points)
        individual = project_to_constraints(individual, target_sum)
        population.append(individual)
    
    # Evaluate fitness
    fitness = [compute_c5(ind, dx) for ind in population]
    
    # Best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolutionary process
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation (adaptive)
            # Decay mutation rate
            mut_rate = 0.1 * (1.0 - iter_num/max_iter) + 0.01
            noise1 = np.random.normal(0, mut_rate, n_points)
            child1 += noise1
            child1 = project_to_constraints(child1, target_sum)
            
            noise2 = np.random.normal(0, mut_rate, n_points)
            child2 += noise2
            child2 = project_to_constraints(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:pop_size]
        
        # Re-evaluate fitness
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Update best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_h = population[new_best_idx].copy()
            best_c5 = fitness[new_best_idx]
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
