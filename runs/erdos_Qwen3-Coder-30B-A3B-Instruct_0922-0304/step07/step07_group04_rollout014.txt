import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 100000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Setup for simulated annealing
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling rate
    eta = 0.01  # Learning rate
    lambda1 = 100  # Penalty weight for bounds
    lambda2 = 100  # Penalty weight for sum
    
    def compute_energy(h_vec):
        c5 = evaluate_overlap(h_vec, dx)
        # Constraint penalties
        bound_penalty = np.sum(np.square(np.clip(h_vec, 0, 1) - h_vec))
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return c5 + lambda1 * bound_penalty + lambda2 * sum_penalty
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h
        h_new = h.copy()
        perturbation = np.random.normal(0, eta * T, n_points)
        h_new += perturbation
        
        # Apply constraint penalties
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
            
        # Energy computation
        e_old = compute_energy(h)
        e_new = compute_energy(h_new)
        
        # Accept or reject
        delta_e = e_new - e_old
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h = h_new
            if e_new < compute_energy(best_h):
                best_h = h.copy()
                best_c5 = evaluate_overlap(h, dx)
        
        # Update temperature
        T *= alpha
        if T < 1e-8:
            T = 1e-8
    
    # Refinement step with gradient descent
    def objective(h_vec):
        return evaluate_overlap(h_vec, dx)
    
    # Simple gradient estimate
    def gradient(h_vec):
        # Approximate gradient by finite differences
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # Normalize
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum > 0:
                h_minus = h_minus * (target_sum / h_minus_sum)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    for i in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        grad = gradient(h_refined)
        h_refined -= 0.001 * grad
        # Enforce constraints
        h_refined = np.clip(h_refined, 0, 1)
        h_refined_sum = np.sum(h_refined)
        if h_refined_sum > 0:
            h_refined = h_refined * (target_sum / h_refined_sum)
        
        c5_refined = evaluate_overlap(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Final validation
    final_h = best_h.copy()
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
