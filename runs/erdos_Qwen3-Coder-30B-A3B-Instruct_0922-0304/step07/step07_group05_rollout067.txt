import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a smooth step-like function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        # Add small smoothing
        h = np.convolve(h, np.ones(5)/5, mode='same')
        
    # Ensure constraint ∫h(x)dx = 1
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Phase inversion algorithm inspired by Remez exchange
    max_iter = 15
    alpha = 0.1
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap
        overlap = correlate(h, 1 - h, mode="full")
        
        # Find the maximum overlap location
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Compute the frequency domain to identify dominant frequency
        # Transform to frequency domain
        H = np.fft.fft(h)
        H_conj = np.conj(H)
        corr_freq = H * H_conj
        freq_domain = np.fft.ifft(corr_freq)
        
        # Use the dominant frequency for phase inversion
        # Find dominant frequency by looking at peak in power spectrum
        powers = np.abs(corr_freq)
        dominant_freq = np.argmax(powers[1:len(powers)//2]) + 1  # Exclude DC
        
        # Create phase inversion signal
        x = np.arange(n_points)
        phase_signal = np.sin(2 * np.pi * dominant_freq * x / n_points)
        
        # Update h with phase inversion
        h_new = h + alpha * phase_signal
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Check improvement
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        h = h_new.copy()
        alpha *= 0.95  # Decay learning rate
    
    # Final local refinement using bounded optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    result = minimize(objective, best_h.copy(), method='SLSQP',
                      bounds=bounds, constraints=constraint,
                      options={'maxiter': 100})
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        final_c5 = compute_c5(best_h, dx)
    else:
        final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
