import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Adam-style optimization with momentum
    alpha = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Momentum and velocity terms
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Gradient computation function
    def compute_gradient(h_vals):
        # Compute overlap matrix for all shifts
        h_rev = np.flip(h_vals)
        overlap_full = np.convolve(h_vals, h_rev, mode='full')
        overlap_full = overlap_full[len(h_vals)-1:]  # Get valid shifts
        # Shifts are in range -n+1 to n-1, so overlap_full index maps to shift k = i - n + 1
        # We're interested in sum h(i) * (1 - h(i-k)) 
        # which is sum h(i) - sum h(i) * h(i-k)
        #   = sum h(i) - conv(h(i), h(i-k)) for all k.
        # Wait, let's recompute this correctly using the actual integral form.
        # For fixed k, overlap = sum h(i) * (1 - h(i-k)) 
        # i.e. sum h(i) - sum h(i) * h(i-k)
        
        grad = np.zeros_like(h_vals)
        for k in range(-n_points+1, n_points): # equivalent to range(0, 2*n_points-1) but offset
            if k >= 0:
                # h(i-k) for i=0..n-1 where i-k >= 0
                shifted = np.zeros_like(h_vals)
                shifted[k:] = h_vals[:-k] if k > 0 else h_vals
            else:
                # h(i-k) for i=0..n-1 where i-k < 0
                shifted = np.zeros_like(h_vals)
                shifted[:k] = h_vals[-k:] if k < 0 else h_vals
                
            # Sum h(i) * h(i-k) over i
            conv = np.sum(h_vals * shifted)
            # And the gradient w.r.t. h(i) will be (1 - shifted(i))
            # But we want to minimize this so gradient is negative:
            # Sum (1 - h(i-k)) for i + Sum (1 - h(i-k)) for i-k>=0
            # ... 
            # Simpler way:
            # For fixed k, the overlap integral is \int h(x) * (1-h(x+k)) dx
            # Discretized:
            # sum h(i) * (1 - h(i + k)) for all i where i + k is within range
            # Gradient w.r.t. h(i) is:
            # -h((i-k)%n) for shifts where (i-k)%n = j for any valid j
            # and 1 for shifts for which i-k = j
            
            # Let's directly compute the gradient for shift k to build up gradient for each h(i)
            
            # This is bit messier. Let's do matrix approach:
            # Jacobian shape n x (2n-1) - each overlap function has one parameter
        
        # Do a matrix-based approach for computing gradient of the max overlap w.r.t h_i
        # Would requiretworzyć large arrays, so use simplified way:
        # compute gradient using sophisticated gradients + alternate methods later
        
        # For now, compute an approximate gradient

        # To get a feeling of the direction
        k_max = np.argmax(overlap_full)

        # and compute \grad h(i) for max overlap

        max_shift = k_max - (n_points-1)

            shifted = np.zeros_like(h_vals)

        # f_k = \sum h[i] * (1 - h[(i-k)%n])

        # 
        # Max overlap is some f_k*, so when k* is fixed
        # we want to minimize f_k* = \sum h[i] * (1 - h[(i-k*)%n])
        # \partial f_k*/\partial h[i] = 1 - h[(i-k*)%n]
        
        # We want gradient to decrease the objective (i.e., make sum smaller),

        if max_shift < 0:
            shift_backward = np.roll(h_vals, -max_shift)
        else:
            shift_backward = np.roll(h_vals, max_shift)
        
        grad = shift_backward - 1.0
        return grad

        # Compute gradient
        grad = compute_gradient(h)
        
        # Update momentum and velocity
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update parameters
        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6 and current_sum > 0:
            h = h * (target_sum / current_sum)
        
        # Evaluate function
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Stopping condition
        if t > 1000 and c5 < 0.3807:  # Early exit if target achieved
            break
    
    # Final polishing with scipy
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
