import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():

    else:

    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Forward pass - compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient with respect to w
        grad = np.zeros(n_points)
        # Compute the gradient more carefully
        contrib = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                contrib[i] = (1 - 2 * h[i]) * (1 - h[i + k_max])
        
        grad = contrib
        
        # Update momentum
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_count+1))
        v_hat = v / (1 - beta2**(iter_count+1))
        
        # Update parameters
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h
        h = sigmoid(w)
        
        # Normalize to sum constraint
        h = h / np.sum(h) * target_sum
        
        # Clip to prevent numerical issues
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Transform back to w for next iteration
        w = np.log(h / (1 - h + 1e-8))
        
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
