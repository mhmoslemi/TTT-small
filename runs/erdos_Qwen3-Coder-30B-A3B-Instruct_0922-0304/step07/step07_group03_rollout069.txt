import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

        h = np.sin(np.pi * np.arange(n_points) / n_points) ** 2
        h = project_to_constraints(h, target_sum)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Greedy optimization parameters
    epsilon_initial = 0.2
    epsilon_min = 0.001
    decay_rate = 0.99
    decay_interval = 100
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current epsilon
        epsilon = max(epsilon_min, epsilon_initial / (1 + 0.001 * iteration))
        
        # Generate random perturbation
        v = np.random.randn(n_points)
        v = v / np.linalg.norm(v)
        mutation = epsilon * v
        
        # Apply mutation
        h_candidate = h + mutation
        
        # Project to constraints
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Compute C5 for candidate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        # Accept if better
        if candidate_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = candidate_c5
            best_h = h.copy()
        
        # Adapt epsilon
        if iteration % decay_interval == 0:
            epsilon_initial *= decay_rate
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
