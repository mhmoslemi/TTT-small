import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    # PSO parameters
    n_particles = 100
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    
    # Create initial population with diversity
    particles.append(h.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate 40% of particles as small variations of initial solution
    for _ in range(int(0.4 * (n_particles - 1))):
        variation = np.random.normal(0, 0.01, n_points)
        new_h = h + variation
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, target_sum)
        particles.append(new_h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Fill the rest randomly
    for _ in range(n_particles - len(particles)):
        new_h = np.random.rand(n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, target_sum)
        particles.append(new_h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best initialization
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    max_iter = 60
    w = w_start
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary conditions
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Update personal best
            fitness = compute_c5(particles[i], dx)
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
                
        # Early termination
        if global_best_fitness <= 0.3807:
            break
        
        # Update inertia weight
        if iteration % 10 == 0:
            w *= w_decay
    
    # Gradient refinement of best solution
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    refined_solution = global_best.copy()
    
    # Multiple refinement rounds to avoid local minima
    for _ in range(10):
        result = minimize(objective, refined_solution, method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            refined_solution = result.x.copy()
            refined_solution = np.clip(refined_solution, 0, 1)
            refined_solution = normalize_h(refined_solution, target_sum)
        else:
            break
    
    final_c5 = compute_c5(refined_solution, dx)
    
    return (refined_solution.tolist(), final_c5, n_points)
