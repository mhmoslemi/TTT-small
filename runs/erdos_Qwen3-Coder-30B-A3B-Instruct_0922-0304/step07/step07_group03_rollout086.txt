import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to meet target sum
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values with a structured approach
        h = np.zeros(n_points)
        # Start with a periodic pattern and adjust
        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Convex relaxation parameters
    max_iter = 20000
    learning_rate = 0.1
    decay_rate = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        c = correlate(h, 1 - h, mode="full")
        
        # Compute gradient of surrogate objective: 2 * c
        gradient = 2 * c * dx**2
        
        # Update h in direction of negative gradient
        h = h - learning_rate * gradient[:len(h)]
        
        # Project to constraints
        h = project_to_constraints(h, target_sum)
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best if better
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive learning rate
        learning_rate = 0.1 / (1 + iter_num / decay_rate)
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Multi-resolution refinement
    # Start from a coarse resolution and refine
    coarse_points = 10
    coarse_dx = 2.0 / coarse_points
    
    if 'initial_h_values' in globals():
        coarse_h = np.array(initial_h_values[:coarse_points])
    else:
        coarse_h = np.ones(coarse_points) * 0.5
    coarse_h = project_to_constraints(coarse_h, coarse_points / 2)
    
    for _ in range(5):  # 5 refinement steps
        # Interpolate to higher resolution
        fine_h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, coarse_points), coarse_h)
        fine_h = project_to_constraints(fine_h, target_sum)
        
        # Fine tuning
        for tune_iter in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            c = correlate(fine_h, 1 - fine_h, mode="full")
            gradient = 2 * c * dx**2
            fine_h = fine_h - 0.05 * gradient[:len(fine_h)]
            fine_h = project_to_constraints(fine_h, target_sum)
            current_c5 = compute_c5(fine_h, dx)
            if current_c5 < best_c5:
                best_h = fine_h.copy()
                best_c5 = current_c5
                
        coarse_h = np.interp(np.linspace(0, 2, coarse_points), 
                            np.linspace(0, 2, n_points), fine_h)
        
        coarse_points = min(coarse_points * 2, 70)
        coarse_dx = 2.0 / coarse_points
        if coarse_points >= 70:
            break
    
    return (best_h.tolist(), best_c5, n_points)
