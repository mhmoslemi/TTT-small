import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def gradient_step(h, dx, target_sum, c_bound, alpha=0.1):

    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    max_val = overlap[max_idx]

    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_to_constraints(h_plus, target_sum)
        h_minus = project_to_constraints(h_minus, target_sum)
        
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
        overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
        
        grad[i] = (overlap_plus[max_idx] - overlap_minus[max_idx]) / (2 * eps)
    
    # Perform step
    step_size = 0.01
    new_h = h - step_size * grad
    
    # Project back
    new_h = project_to_constraints(new_h, target_sum)
    
    # Check constraint satisfaction with slack
    new_overlap = correlate(new_h, 1 - new_h, mode="full")
    new_max_val = np.max(new_overlap)
    
    # If we're improving or within bounds with slack
    if new_max_val <= c_bound * (1 + alpha):
        return new_h

        # Try smaller step
        for step in [0.005, 0.001]:

            temp_overlap = correlate(temp_h, 1 - temp_h, mode="full")
            temp_max_val = np.max(temp_overlap)
            if temp_max_val <= c_bound * (1 + alpha):
                return temp_h
        return h
