import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx, target_sum):
    """Evaluate individual with constraint checking."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with both provided values and random
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add variations of the initial solution
        for _ in range(population_size // 5):
            variation = h + np.random.normal(0, 0.02, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            population.append(variation)
        # Fill with random individuals
        for _ in range(population_size - population_size // 5):
            h_random = np.random.rand(n_points)
            h_random = normalize_h(h_random, target_sum)
            population.append(h_random)
    else:
        # Create random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    # PSO parameters
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_iterations = 60
    
    # Initialize particles
    particles = population[:]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(population_size)]
    
    # Find best individual so far
    best_fits = [evaluate_individual(p, dx, target_sum) for p in particles]
    best_idx = np.argmin(best_fits)
    global_best = particles[best_idx].copy()
    global_best_fit = best_fits[best_idx]
    
    # Local search parameters
    max_local_iter = 10
    
    # PSO iteration
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (particles[i] - particles[i]) +
                social * r2 * (global_best - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate
            fit = evaluate_individual(particles[i], dx, target_sum)
            
            # Update personal best
            if fit < best_fits[i]:
                best_fits[i] = fit
                
            # Update global best
            if fit < global_best_fit:
                global_best = particles[i].copy()
                global_best_fit = fit
        
        # Reduce inertia
        inertia *= 0.98
        
        # Early termination check
        if global_best_fit <= 0.3807:
            break
    
    # Refinement with local optimization
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        return compute_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Apply optimization on global best
    result = minimize(objective, global_best, method='SLSQP',
                     bounds=bounds, options={'maxiter': 50})
    
    if result.success:
        global_best = np.clip(result.x, 0, 1)
        global_best = normalize_h(global_best, target_sum)
    
    final_c5 = compute_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
