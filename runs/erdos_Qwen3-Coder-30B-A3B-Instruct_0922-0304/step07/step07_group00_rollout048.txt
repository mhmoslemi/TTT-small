import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def get_critical_shift(h, dx):
    """Identify the shift that currently maximizes overlap."""
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    # Adjust index to get the shift value k
    # Full convolution has length 2*n-1
    k = max_idx - (len(h) - 1)
    return k, overlap[max_idx] * dx

def line_search(h, s, k_max, dx, alpha_max=1.0, tol=1e-6):
    """Perform backtracking line search to find optimal alpha."""
    # Ensure s is from the same space as h
    current_c5 = compute_c5(h, dx)
    # Compute gradient at k_max
    # For overlap_k = h * (1-h) shift_k, the gradient w.r.t. h_i is:
    # (1 - 2*h_i) if i+k_max is valid
    # We use finite differences to approximate more accurately
    grad_h = (1 - 2 * h)
    # Compute gradient for the shift k_max
    # When computing overlap_k(h) = sum_i h[i] * (1 - h[i+k_max]), the partial
    # derivative of that sum w.r.t h_i is (1 - h[i+k_max]) if i+k_max is valid.
    # Use a more precise approach to get gradient
    
    # Instead of approximating the gradient, we compute the contribution
    # of each element to the final overlap, which is the gradient we need
    
    for alpha in [alpha_max * (0.5 ** i) for i in range(10)]:
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, np.sum(h))
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5 - tol:
            return alpha
    return alpha_max
