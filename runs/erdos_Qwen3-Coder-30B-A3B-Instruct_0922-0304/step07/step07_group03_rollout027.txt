import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
    
    # Project to correct sum
    h = project_to_constraints(h, target_sum)
    
    # Evolutionary Strategy with adaptive mutation
    pop_size = 20
    elite_size = pop_size // 2
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = project_to_constraints(individual, target_sum)
        population.append(individual)
    
    # Main ES loop
    max_iter = 20000
    mutation_rate = 0.1
    mutation_rate_decay = 0.9999
    
    best_overall = None
    best_overall_value = float('inf')
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            fit = compute_c5(individual, dx)
            fitnesses.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best
        if fitnesses[0] < best_overall_value:
            best_overall = population[0].copy()
            best_overall_value = fitnesses[0]
        
        # Early termination
        if best_overall_value <= 0.3807:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Generate offspring through mutation
        offspring = []
        for parent in elite:
            # Mutate
            noise = np.random.randn(n_points) * mutation_rate
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            offspring.append(child)
        
        # Replace worst with offspring
        population[-len(offspring):] = offspring
        
        # Adapt mutation rate
        mutation_rate *= mutation_rate_decay
    
    # Final check
    final_c5 = compute_c5(best_overall, dx) if best_overall is not None else float('inf')
    
    return (best_overall.tolist() if best_overall is not None else [0]*n_points, final_c5, n_points)
