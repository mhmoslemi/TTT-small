import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a block-like step function and add some randomness
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Stage 1: Use convex relaxation with CVXPY
    if time.time() - start_time < budget_s - 100:
        try:
            # Create a continuous relaxation of h
            h_var = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == target_sum,
                # Relax the overlap constraint
            ]
            
            # For all possible shifts k:
            # h * (1 - h_shifted) <= C
            # We will implement a simplified approximation or bound
            
            # Solve problem
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            
            # Try different solvers
            solver_attempts = [cp.CVXOPT, cp.GLPK, cp.SCS, cp.ECOS]
            for solver in solver_attempts:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                        h_relaxed = h_var.value
                        break
                except:
                    continue
            else:
                # Fallback to simpler approach if CVXPY fails
                h_relaxed = h.copy()
                
            # Convert to step function
            # Truncate to step function based on a threshold
            threshold = 0.5
            h = (h_relaxed > threshold).astype(float)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        except:
            # If CVXPY fails, fall back to just using the last h from PSO
            pass
    
    # Stage 2: Gradient descent refinement
    max_iter = 1000
    learning_rate = 0.01
    decay_factor = 0.95
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        h = project_to_constraints(h, target_sum)
        current_c5 = compute_c5(h, dx)
        
        # Simulate gradient direction (approximate)
        # Use a small perturbation approach to get a notion of gradient
        eps = 1e-4
        grad_suggestion = np.zeros_like(h)
        
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += eps
            h_pert = project_to_constraints(h_pert, target_sum)
            c5_plus = compute_c5(h_pert, dx)
            
            h_pert[i] -= 2*eps
            h_pert = project_to_constraints(h_pert, target_sum)
            c5_minus = compute_c5(h_pert, dx)
            
            grad_suggestion[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update using gradient descent with projection
        h_new = h - learning_rate * grad_suggestion
        h_new = project_to_constraints(h_new, target_sum)
        
        # Accept improvement or accept with probability based on cooling
        current_c5 = compute_c5(h, dx)
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability
            prob = np.exp(- (new_c5 - current_c5) / (1e-3 + current_c5))
            if np.random.rand() < prob:
                h = h_new
                
        # Apply learning rate decay
        learning_rate *= decay_factor
        
        # Early termination
        if current_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
