import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, maxiter=1000):
    """Local optimization using L-BFGS-B."""
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_fun(x):
        return np.sum(x) - target_sum

    cons = {'type': 'eq', 'fun': constraint_fun}
    bounds = [(0, 1) for _ in range(len(h))]
    
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                   constraints=cons, options={'maxiter': maxiter})
    
    if res.success:
        return project_to_constraints(res.x, target_sum)
    else:
        return project_to_constraints(h, target_sum)

    # ES parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_iter = 5000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            candidate = np.array(initial_h_values)
        else:
            candidate = np.random.uniform(0, 1, n_points)
        candidate = normalize_h(candidate, target_sum)
        population.append(candidate)
    
    # Evaluate initial population
    fitness = []
    for p in population:
        fitness.append(compute_c5(p, dx))
    
    # Best solution so far
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolutionary loop
    for gen in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Update sigma (linear decay)
        sigma = sigma_init - (sigma_init - sigma_final) * gen / max_iter
        
        # Selection (keep top 50%)
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:pop_size//2]]
        selected_fitness = [fitness[i] for i in sorted_indices[:pop_size//2]]
        
        # Create offspring
        offspring = []

            # Add Gaussian noise
            noise = np.random.normal(0, sigma, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            offspring.append(child)
        
        # Replace worst individuals
        worst_indices = sorted_indices[pop_size//2:]
        for i, idx in enumerate(worst_indices):
            population[idx] = offspring[i]

        # Check for improvement
