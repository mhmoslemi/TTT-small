import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with momentum and adaptivity
    def compute_gradient(h_vec, dx):
        """Compute gradient of overlap w.r.t. h."""
        # For each position i, compute partial derivative of max_k h[i](1-h[i+k])
        # This is a complex derivative; we'll compute a rough finite difference
        # or approximate it using correlation of h and 1-h
        
        # Use correlation to help approximate gradient direction
        # The actual gradient is complex to derive exactly, so we use a simplified
        # estimation approach
        
        # Simple finite difference approximation
        eps = 1e-6
        
        grad = np.zeros_like(h_vec)
        
        # Compute current overlap value
        current_overlap = evaluate_overlap(h_vec, dx)
        
        # For each variable, compute finite difference
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus = h_vec.copy()
            h_minus[i] = max(0.0, h_minus[i] - eps)
            
            # Enforce sum constraint in both cases
            if i < len(h_vec) - 1:
                # Adjust other elements to match sum
                diff_plus = np.sum(h_plus) - target_sum
                diff_minus = np.sum(h_minus) - target_sum
                h_plus[:-1] -= diff_plus/(len(h_plus)-1)
                h_minus[:-1] -= diff_minus/(len(h_minus)-1)
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
            
            val_plus = evaluate_overlap(h_plus, dx)
            val_minus = evaluate_overlap(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
        
        return grad
    
    # Local optimization with adaptive line search
    max_iter = 1000
    learning_rate = 1e-3
    momentum = 0.9
    v = np.zeros_like(h)
    lambda_reg = 1e-4
    
    # Precompute kernel for correlation
    correlations = []
    for i in range(n_points):
        # For fixed shift, compute overlap contributions
        corr = np.zeros(2*n_points - 1)
        for j in range(n_points):
            if 0 <= j+i < n_points:
                corr[n_points-1+i-j] += h[j] * (1 - h[j+i])
        correlations.append(corr)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current objective and gradient
        current_c5 = evaluate_overlap(h, dx)
        
        # For gradient computation, we use heuristic
        # Gradient w.r.t. each h[i] is approximately
        # sum_{k} (h[i](1-h[i+k]) + h[i+k](1-h[i]))  
        # But this is only appx; using finite diff for stability
        grad = compute_gradient(h, dx)
        
        # Add regularization term to enforce constraints (sum = target_sum)
        grad += lambda_reg * (np.sum(h) - target_sum)
        
        # Nesterov momentum update
        v = momentum * v - learning_rate * grad
        h_new = h + momentum * v - learning_rate * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust h_new to meet sum constraint via simple rescaling
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (target_sum / h_sum)
        
        # Backtracking line search
        # Evaluate candidate point
        c5_new = evaluate_overlap(h_new, dx)
        
        if c5_new < current_c5:
            h = h_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            # Update learning rate
            learning_rate *= 0.9
            if learning_rate < 1e-8:
                learning_rate = 1e-8
            
            # Try again with reduced learning rate
            v = momentum * v - learning_rate * grad
            h_new_2 = h + momentum * v - learning_rate * grad
            h_new_2 = np.clip(h_new_2, 0, 1)
            h_sum = np.sum(h_new_2)
            if h_sum > 0:
                h_new_2 = h_new_2 * (target_sum / h_sum)
            c5_new_2 = evaluate_overlap(h_new_2, dx)
            
            if c5_new_2 < best_c5:
                h = h_new_2
                best_c5 = c5_new_2
                best_h = h.copy()
            
        # Adjust lambda_reg
        if np.abs(np.sum(h) - target_sum) < 1e-4:
            lambda_reg *= 0.9
        else:
            lambda_reg = max(1e-6, lambda_reg * 1.1)
    
    # Final normalization and validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
