import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        if diff > 0:
            h += diff / len(h)
        else:
            h += diff / len(h)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    h = project_to_constraints(h, target_sum)
    
    # Convex relaxation
    h_cvx = cp.Variable(n_points)
    C = cp.Variable()
    
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == target_sum,
        C >= 0
    ]
    
    # Compute the correlation constraint as a series of inequalities
    # For each lag k, we form the overlap sum and add a constraint
    overlaps = []
    for k in range(2 * n_points - 1):
        lag = k - (n_points - 1)
        if lag <= 0:
            corr = cp.sum(cp.multiply(
                h_cvx[:n_points + lag],
                1 - h_cvx[-lag:]
            )) * dx
        else:
            corr = cp.sum(cp.multiply(
                h_cvx[lag:],
                1 - h_cvx[:-lag]
            )) * dx
        constraints.append(corr <= C)
    
    # Minimize C
    prob = cp.Problem(cp.Minimize(C), constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_relaxed = h_cvx.value
            # Threshold to make it a step function
            h = np.clip(h_relaxed, 0, 1)
            h = project_to_constraints(h, target_sum)
    except:
        # Fall back to previous method if convex relaxation fails
        h = project_to_constraints(h, target_sum)
    
    # Local refinement with projected gradient descent
    # Gradient step size
    lr = 0.01
    iters = 0
    max_iter = 1000
    
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    for it in range(max_iter):
        iters += 1
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        max_position = np.argmax(overlaps)
        max_overlap = overlaps[max_position] * dx
        
        # Gradient w.r.t. h (approximate)
        # Compute the derivative using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            pert_h = h.copy()
            pert_h[i] += 1e-6
            pert_h = project_to_constraints(pert_h, target_sum)
            pert_overlap = correlate(pert_h, 1 - pert_h, mode="full")
            grad[i] = (pert_overlap[max_position] * dx - max_overlap) / 1e-6
        
        # Gradient descent step
        h = h - lr * grad
        h = project_to_constraints(h, target_sum)
        
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Adaptive learning rate
        lr *= 0.999
    
    return (best_h.tolist(), best_c5, n_points)
