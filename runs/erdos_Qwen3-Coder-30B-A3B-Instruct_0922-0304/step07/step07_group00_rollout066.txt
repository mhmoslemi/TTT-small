```python
import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the parent construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute gradient
        # Direct gradient computation through overlaps
        h_sigmoid = sigmoid(w)
        h_sigmoid = h_sigmoid / np.sum(h_sigmoid) * target_sum
        
        # Compute all overlaps
        overlaps = correlate(h_sigmoid, 1 - h_sigmoid, mode="full")
        # The full convolution is of length 2*n_points - 1
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Gradient of the overlap function w.r.t. w
        grad = np.zeros(n_points)
        # For max_k, we compute the gradient
        # The overlap at k is sum_i h[i] * (1 - h[i+k])
        # So ∂overlap_k/∂h[i] = (1 - h[i+k]) if i+k < n_points, 0 otherwise
        # and ∂overlap_k/∂h[i+k] = -h[i] if i < n_points 
        # But we need to handle the constraint and the normalization
        # This is a bit tricky, so let's compute the discrete gradient
        
        # Change in overlap when varying h[i]
        # Using finite differences or direct chain rule
        # With the form overlaps[k] = sum_{i} h[i] * (1 - h[i+k])
        # ∂/∂w_i of h[i] = h[i] * (1 - h[i])
        h_grad = h_sigmoid * (1 - h_sigmoid)
        
        # Let's instead compute the critical shift directly
        k_max = max_k - (n_points - 1)  # Convert to actual shift
        # The shift is from 0 to 2*n_points - 2, we want k from -(n_points - 1) to n_points - 1
        # Which is a mapping to indexes 0..2*n_points-2 -> -(n_points-1) to n_points-1
        
        # Reconstruct the gradient more carefully
        # ∂overlaps[k]/∂h[i] = 1-h[i+k] if index i+k valid, -h[i-k] if index i-k valid
        grad = np.zeros(n_points)
        
        # For k maximal:
        # If k=0, then overlap is sum h[i](1-h[i]), gradient wrt h[i] is (1-2*h[i])
        # If k=1, overlap is sum h[i](1-h[i+1]), gradient is (1-h[i+1]) for h[i] and -h[i-1] for h[i+1]
        # In general, overlapping terms contribute
        # if i-k >=0 and i < n_points, then the h[i-k] contributes -h[i] to its gradient
        # if i+k < n_points and i >= 0, then the h[i] contributes (1-h[i+k]) to its gradient 
        
        # Let's compute manually here
        if k_max < 0:
            # Shift is to the left
            for i in range(n_points):
                if i + (-k_max) < n_points:
                    # h[i] affects overlap_k: it contributes (1-h[i + (-k_max)])
                    grad[i] += 1 - h_sigmoid[i + (-k_max)]
                if i - k_max >= 0:
                    # h[i - k_max] affects overlap_k: it contributes -h[i]
                    grad[i - k_max] += -h_sigmoid[i]
        else:
            # Shift is to the right
            for i in range(n_points):
                if i + k_max < n_points:
                    # h[i] affects overlap_k: it contributes (1-h[i + k_max])
                    grad[i] += 1 - h_sigmoid[i + k_max]
                if i - k_max >= 0:
                    # h[i - k_max] affects overlap_k: it contributes -h[i]
                    grad[i - k_max] += -h_sigmoid[i]

        # Capitalize: gradient wrt w[i] = gradient wrt h[i] * dh[i]/dw[i]
        # dh[i]/dw[i] = h[i] * (1 - h[i])
        grad = grad * h_grad
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Update h from w
        h_sigmoid = sigmoid(w)
        h = h_sigmoid / np.sum(h_sigmoid) * target_sum
        # Clip and re-transform
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h))
        
        # Check if we improved
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
