import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Generate 40% as variations of initial
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
        # Fill remaining with random
        while len(particles) < population_size:
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand)
    else:
        # Generate random feasible population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            particles.append(h)
    
    # PSO parameters
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    velocity = [np.zeros(n_points) for _ in range(population_size)]
    
    # Initialize personal bests and global best
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    max_iterations = 60
    stagnation_count = 0
    stagnation_threshold = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            particle = particles[i]
            v = velocity[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            v = (inertia * v + 
                 cognitive * r1 * (personal_best[i] - particle) +
                 social * r2 * (global_best - particle))
            
            # Update position
            particle = particle + v
            
            # Apply constraints
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            particles[i] = particle
            velocity[i] = v
            
            # Evaluate particle
            fitness = compute_c5(particle, dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particle.copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particle.copy()
                    global_best_fitness = fitness
                    stagnation_count = 0
                    
        # Update inertia
        if iteration % 10 == 0:
            inertia *= 0.98
            
        # Check for stagnation
        stagnation_count += 1
        if stagnation_count >= stagnation_threshold:
            # Increase mutation strength
            for i in range(population_size):
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = np.clip(particles[i] + noise, 0, 1)
                particles[i] = normalize_h(particles[i], target_sum)
            stagnation_count = 0
            
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Local refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    # Run gradient descent for a few times
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            current_c5 = compute_c5(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
