import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Convex relaxation approach
    # Create the CVXPY problem
    h_cvx = Variable(n_points)
    C = Variable()
    
    # Construct constraints
    constraints = [0 <= h_cvx, h_cvx <= 1, sum(h_cvx) == target_sum]
    
    # Add constraints for the convolution maximum
    # This approach approximates C by constraining all convolution values
    # But that would be non-convex, so we use a surrogate - the maximum of a few
    # key sampled points or use the worst case look
    
    # Convolution constraint (this is approximate and not strict)
    # We'll later do a local search for more precise improvement
    
    # Solve convex optimization problem
    objective = Minimize(C)
    problem = Problem(objective, constraints)
    
    try:
        problem.solve(solver=ECOS, verbose=False)
        h_solution = h_cvx.value
        h_solution = np.clip(h_solution, 0, 1)
        # Normalize ensuring exact sum constraint
        h_solution = h_solution * (target_sum / np.sum(h_solution))
        
        # Better initial for local search
        h = h_solution.copy()
    except:
        pass  # Fall back to previous method if needed
    
    # Local optimization: adjust step positions
    max_iter = 500
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Perturb step positions to get better arrangement
        # Use a greedy local search based on small shifts
        h_trial = h.copy()
        # Try to shift one position at a time
        i = np.random.randint(0, n_points)
        # Back up current value
        current_val = h_trial[i]
        # Try moving it to nearby positions
        max_shift = min(2, n_points - 1)
        for shift in range(-max_shift, max_shift + 1):
            if shift == 0:
                continue
            pos = i + shift
            # Ensure valid index
            if pos < 0 or pos >= n_points:
                continue
            # Swap
            h_trial[i], h_trial[pos] = h_trial[pos], h_trial[i]
            
            # Adjust to satisfy sum constraint exactly
            total = np.sum(h_trial)
            if abs(total - target_sum) > 1e-10:
                diff = target_sum - total
                # Distribute across all points
                h_trial += diff / n_points
                h_trial = np.clip(h_trial, 0, 1)
                # Ensure exact sum
                h_trial = h_trial * (target_sum / np.sum(h_trial))
            
            new_c5 = compute_c5(h_trial, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_trial.copy()
                
            h_trial[i], h_trial[pos] = h_trial[pos], h_trial[i]  # Revert
        
        # A block-approximation approach (uniform step functions)
        if iter_count % 10 == 0:
            h_block = np.zeros(n_points)
            mid = n_points // 2
            h_block[:mid] = 1
            h_block = project_to_constraints(h_block, target_sum)
            c5_block = compute_c5(h_block, dx)
            if c5_block < best_c5:
                best_c5 = c5_block
                best_h = h_block.copy()
        
        # Small random perturbations to escape local minimum
        h_pert = h.copy()
        noise = np.random.normal(0, 0.005, n_points)
        h_pert += noise
        h_pert = project_to_constraints(h_pert, target_sum)
        c5_pert = compute_c5(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        
        h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
