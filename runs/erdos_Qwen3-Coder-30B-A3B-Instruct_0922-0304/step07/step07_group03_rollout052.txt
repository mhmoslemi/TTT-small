import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    
    # Evolution Strategy parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = individual * (target_sum / np.sum(individual))
        population.append(individual.copy())
    
    # Evaluate initial population
    fitnesses = []
    for individual in population:
        fitness = compute_c5(individual, dx)
        fitnesses.append(fitness)
    
    best_fitness = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    # Evolution iteration
    max_iter = 3000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        elite_fitnesses = fitnesses[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Mutate elite members to create offspring
        for i in range(population_size - elite_size):
            parent = elite[i % elite_size]
            # Gaussian mutation
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitnesses = []
        for individual in population:
            fitness = compute_c5(individual, dx)
            fitnesses.append(fitness)
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
