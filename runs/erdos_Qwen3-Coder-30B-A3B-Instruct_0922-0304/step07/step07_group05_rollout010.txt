import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Create diverse particles starting from initial solution
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        
        # Add 40% variations of initial solution
        for _ in range(int(0.4 * population_size) - 1):
            variation = h + np.random.normal(0, 0.02, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation.copy())
            velocities.append(np.zeros(n_points))
        
        # Fill the rest with random feasible solutions
        for _ in range(population_size - len(particles)):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    else:
        # Generate random feasible population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points))
    
    # Initialize personal bests and global best
    personal_bests = [p.copy() for p in particles]
    personal_best_fitnesses = [compute_c5(p, dx) for p in particles]
    
    global_best_idx = np.argmin(personal_best_fitnesses)
    global_best = personal_bests[global_best_idx].copy()
    global_best_fitness = personal_best_fitnesses[global_best_idx]
    
    # PSO parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iterations = 60
    stagnation_count = 0
    stagnation_limit = 20
    
    # PSO iteration loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (personal_bests[i] - particles[i])
            social = social_coeff * r2 * (global_best - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fitness
                personal_bests[i] = particles[i].copy()
                
                # Check if new global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    stagnation_count = 0  # Reset stagnation
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Adapt inertia weight based on stagnation
        if stagnation_count > stagnation_limit:
            inertia_weight = max(0.1, inertia_weight * 0.95)
            stagnation_count = 0
        
        # Decay inertia weight
        inertia_weight *= 0.98
            
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    result = minimize(objective, global_best.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 20})
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        final_c5 = compute_c5(refined_h, dx)
        if final_c5 < global_best_fitness:
            global_best = refined_h
            global_best_fitness = final_c5
    
    return (global_best.tolist(), global_best_fitness, n_points)
