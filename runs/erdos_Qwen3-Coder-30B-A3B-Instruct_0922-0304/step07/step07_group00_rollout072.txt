import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max, n_points):
    """Compute gradient for the k_max overlap."""
    grad = np.zeros_like(h)
    # Compute gradient for overlap_k_max
    # For the overlap at shift k_max, we have h[i] * (1 - h[i+k_max])
    # So the gradient w.r.t. h[i] is (1 - 2*h[i+k_max]) if i < n_points - k_max
    # and 0 otherwise.
    if k_max < n_points:
        grad[:-k_max] = 1 - 2 * h[k_max:]
    return grad

def find_max_overlap_shift(h, n_points):
    """Find the shift that maximizes overlap."""
    # Compute overlaps for all shifts using convolution
    # But compute only the maximum point from the full convolution
    overlap = correlate(h, 1 - h, mode="full")
    max_shift = np.argmax(overlap)
    # Shift values in overlap go from 0 to 2*n_points - 2
    # We want the shift within the constraint [-1, 1] (in terms of the shift)
    # But the actual shift is max_shift - (n_points - 1)
    return max_shift - (n_points - 1)

def line_search(h, grad, target_sum, dx, max_iterations=10):
    """Perform backtracking line search to find optimal step size."""
    alpha = 0.5
    step_size = 0.5
    computed_c5 = compute_c5(h, dx)
    epsilon = 1e-12
    for i in range(max_iterations):
        # Try step s = -grad (or rather, the descent direction)
        s = -grad
        # Project to feasible set (maintain sum = target_sum)
        s_proj = s - np.mean(s)  # center the step
        s_proj = s_proj * (target_sum / (np.sum(s_proj) + epsilon))
        # Compute new solution
        h_new = h + alpha * s_proj
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Check if the new value is better
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < computed_c5 - 1e-10:
            # Accept the step
            return alpha, h_new
        
        # Reduce step size
        alpha *= 0.5
    return alpha, h

def frank_wolfe_step(h, n_points, target_sum, dx, k_max=None):
    """Perform a single Frank-Wolfe step."""
    if k_max is None:
        # Find the k that maximizes the overlap
        k_max = find_max_overlap_shift(h, n_points)
        k_max = max(0, k_max)
    
    # Compute the gradient of overlap at that shift
    grad = compute_gradients(h, k_max, n_points)
    
    # Find the optimal step using line search
    _, h_new = line_search(h, grad, target_sum, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iteration loop
    iterations = 0
    no_improvement_count = 0
    last_c5 = float('inf')
    
    max_iter = 5000
    while (time.time() - start_time < budget_s - 1 and 
           iterations < max_iter and 
           no_improvement_count < 500):
        
        # Find max overlap shift
        k_max = find_max_overlap_shift(h, n_points)
        
        # Perform Frank-Wolfe step
        h = frank_wolfe_step(h, n_points, target_sum, dx, k_max)
        
        # Evaluate C5
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Termination condition
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
