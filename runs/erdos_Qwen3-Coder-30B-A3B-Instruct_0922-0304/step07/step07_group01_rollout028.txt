import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def solve_convex_relaxation(n_points, target_sum):
    h = cp.Variable(n_points)
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    # Convex relaxation constraint: max convolution <= C
    C = cp.Variable()
    # We build a temporary linearized constraint
    # This is a simplified version to get a continuous solution
    # The full convolution constraints are hard to express in CVXPY directly
    # So we solve a simpler version
    
    # We'll use a basic approach for the relaxation:
    # Try to minimize some upper bound on C
    
    # Instead of doing full convolution as a constraint (hard in CVXPY),
    # we'll solve a relaxed version based on sparse constraints
    # This ignores full convolution but provides good initial solution
    
    # For the relaxation we can consider this:
    # Try to minimize single peak value of convolution
    # Which we approximate as: minimize max h_i * (1 - h_{i+1})
    
    # Let's just solve with a simpler constraint that is being aggressive
    # Use basic convex relaxation approach
    objective = cp.Minimize(C)
    
    # Try a basic scheme: ensure we try to bound key correlations
    # Build one such constraint
    # This will use an optimization approach to find a candidate 
    # solution and then proceed to refinement
    
    # For now run basic consistent problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
    
    if prob.status == cp.OPTIMAL:
        return h.value
    else:
        # Fallback to feasible solution
        return np.ones(n_points) * (target_sum / n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():

    else:
        # Initialize with a blocky structure
        mid = n_points // 2
        h[:mid] = 1
        h = project_to_constraints(h, target_sum)
    
    # Try convex relaxation first  
    try:
        relaxed_h = solve_convex_relaxation(n_points, target_sum)
        if relaxed_h is not None:
            h = np.clip(relaxed_h, 0, 1)
            h = project_to_constraints(h, target_sum)
    except:
        pass  # continue with fallback
    
    # Greedy local refinement based on perturbing some steps
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 500
    iteration = 0
    
    # Helper function to compute gradient-like updates
    def compute_step_jacobian(h, dx):
        # Simple finite difference approximation
        jacobian = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = project_to_constraints(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            h_minus = project_to_constraints(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            jacobian[:, i] = (c5_plus - c5_minus) / (2e-6)
        return jacobian
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Attempt small random adjustments in step positions
        # This assumes h is a step function with a few distinct values
        
        # Randomly tweak some values
        for _ in range(3):
            idx = np.random.randint(0, n_points)

            h[idx] += delta
            h = project_to_constraints(h, target_sum)
        
        # Simple greedy local search with bounds
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if best_c5 <= 0.3807:

        iteration += 1
    
    # Final flip to ensure correct measure
    final_c5 = compute_c5(best_h, dx)
