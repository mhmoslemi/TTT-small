import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Frank-Wolfe inspired approach: coordinate-wise updates
    max_iter = 100000
    iter_count = 0
    stale_count = 0
    min_improvement = 1e-8
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Find coordinates that influence the maximum overlap
        # Use the full overlap to find indices that matter
        max_indices = np.where(np.abs(overlap - current_max) < 1e-10)[0]
        
        # Create gradient-like quantities for each coordinate
        # Simplified gradient: we don't compute formal gradients but 
        # estimate direction from overlap structure
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Simulate a gradient on h[i]
            # If we increase h[i] by small amount, how does overlap change?
            # For simplicity, small change in h[i]
            temp_h = h.copy()
            delta = 1e-6
            temp_h[i] = np.clip(h[i] + delta, 0, 1)
            temp_overlap = correlate(temp_h, 1 - temp_h, mode="full") * dx
            delta_overlap = np.max(temp_overlap) - current_max
            
            # Approximate derivative
            grad[i] = delta_overlap / delta
        
        # Select most impactful coordinates
        # Take top 10% based on gradient magnitude
        idxs = np.argsort(np.abs(grad))[::-1][:max(1, n_points//10)]
        
        # Update selected coordinates with approximate Frank-Wolfe step
        for i in idxs:
            # Estimate direction
            direction = 1 - 2 * h[i]  # This is the direction that improves overlap
            
            # Step size: try few variations and pick the best
            best_h_candidate = None
            best_candidate_c5 = current_max
            
            # Try a few alpha values
            alphas = np.linspace(0, 1, 10)
            for alpha in alphas:
                h_trial = h.copy()
                h_trial[i] = np.clip(h[i] + alpha * direction, 0, 1)
                
                # Renormalize
                h_trial = normalize_h(h_trial, target_sum)
                
                # Compute new overlap
                trial_overlap = correlate(h_trial, 1 - h_trial, mode="full") * dx
                trial_max = np.max(trial_overlap)
                
                if trial_max < best_candidate_c5:
                    best_candidate_c5 = trial_max
                    best_h_candidate = h_trial.copy()
            
            # If found a better candidate, accept it
            if best_h_candidate is not None and best_candidate_c5 < current_max:
                h = best_h_candidate
                current_max = best_candidate_c5
                stale_count = 0
            else:
                stale_count += 1
                
        iter_count += 1
        
        if stale_count > 1000:
            # Add some randomness to escape local minimum
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            stale_count = 0
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
