import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def clip_and_normalize(h, target_sum):
    """Clip h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = clip_and_normalize(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = clip_and_normalize(h, target_sum)
    
    # Create initial population with some variation
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.01, n_points)
        individual = clip_and_normalize(individual, target_sum)
        population.append(individual)
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_iter = 60
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    
    for _ in range(n_particles):
        # Start with population members
        p = population[np.random.randint(0, len(population))].copy()
        p = clip_and_normalize(p, target_sum)
        particles.append(p)
        velocities.append(np.zeros(n_dims))
        personal_best.append(p.copy())
        personal_best_fitness.append(compute_c5(p, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_dims), np.random.rand(n_dims)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best[i] - particles[i]) + 
                           social * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = clip_and_normalize(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Update inertia
        inertia *= 0.98
        
        # Early termination on stagnation
        if stagnation_count > 20:
            break
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    # Run a few rounds of local refinement
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 20})
        if result.success:
            best_h = result.x
            best_h = clip_and_normalize(best_h, target_sum)
            new_c5 = compute_c5(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break
    
    # Final check
    best_h = clip_and_normalize(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
