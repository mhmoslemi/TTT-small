import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a smooth step-like function
        h = np.zeros(n_points)
        mid = n_points // 2
        # Create a smooth transition
        for i in range(n_points):
            if i < mid:
                h[i] = 1.0
            else:
                h[i] = 1.0 - (i - mid) / (n_points - mid)
        # Normalize
        h = h * (target_sum / np.sum(h))
    
    # Iterative refinement using a frequency-domain approach
    max_iter = 20
    alpha = 0.1
    
    best_h = h.copy()
    best_c5 = np.max(signal.correlate(h, 1 - h, mode="full")) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx] * dx
        
        # Stop if we've found a good enough solution
        if max_corr <= 0.3807:
            break
            
        # Invert phase to reduce maximum correlation
        # We simulate a simpler update using a phase shift in frequency domain
        # by using sinewave modulation in time domain
        k = max_idx - len(h) + 1  # lag of max correlation
        
        # Use a sinusoidal perturbation with dominant frequency
        perturbation = np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
        h_new = h + alpha * perturbation
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate
        new_c5 = np.max(signal.correlate(h_new, 1 - h_new, mode="full")) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        alpha *= 0.95  # decay learning rate
    
    # Final local optimization
    def objective(h):
        return np.max(signal.correlate(h, 1 - h, mode="full")) * dx

    from scipy.optimize import minimize
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run local optimization
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 100})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target_sum / np.sum(best_h))
            best_c5 = objective(best_h)
        else:
            break
    
    # Final check
    final_c5 = np.max(signal.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
