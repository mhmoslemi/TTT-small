import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit proportionally to current values
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    else:
        h.fill(target_sum / len(h))
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Structured initialization
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # AM parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute overlap and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Initialize gradient
        gradient = np.zeros(n_points)
        
        # Compute gradient direction
        for k in range(len(overlap)):
            if abs(overlap[k] - overlap[max_k]) < 1e-6:
                # Approximate gradient based on shift
                shift = k - (len(h) - 1)
                if shift >= 0:
                    gradient[:-shift] -= h[shift:]
                else:
                    gradient[-shift:] -= h[:shift]
        
        # Update h
        h_candidate = h - alpha * gradient
        
        # Add random perturbation
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to feasible region
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        # Update if better
        if new_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = new_c5
            
        # Update current h
        h = h_candidate.copy()
        
        # Update step sizes
        alpha = 0.2 / (1 + iter_num / tau)
        epsilon = 0.1 / (1 + iter_num / tau)
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
