import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, dx, target_sum):
    """Apply local optimization using L-BFGS-B to refine h."""
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_func(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      constraints=constraints, options={'maxiter': 1000})
    
    if result.success:
        return project_to_constraints(result.x, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # Evolution Strategies parameters
    population_size = 20
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    generations = 2000
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
        fitness.append(compute_c5(individual, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update mutation rate (linear decay)
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * gen / generations
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Select top 50% for breeding
        elite_size = population_size // 2
        elite = population[:elite_size]
        
        # Generate new offspring
        new_population = elite.copy()
        new_fitness = [fitness[i] for i in range(elite_size)]
        
        for i in range(population_size - elite_size):
            # Select parent from elite
            parent = elite[np.random.randint(0, elite_size)]
            
            # Mutate
            mutation = np.random.normal(0, mutation_rate, n_points)
            child = parent + mutation
            child = project_to_constraints(child, target_sum)
            
            # Local optimization if promising
            if compute_c5(child, dx) < 0.381:
                child = local_optimize(child, dx, target_sum)
            
            new_population.append(child)
            new_fitness.append(compute_c5(child, dx))
        
        population = new_population
        fitness = new_fitness
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Final local optimization on best solution
    final_h = local_optimize(best_h, dx, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
