import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Distribute the difference evenly
    h = h + diff / len(h)
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to exact sum
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_sum_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
        h = project_to_sum_constraint(h, target_sum)
        
    # PGD parameters
    alpha = 0.1
    beta = 0.9
    momentum = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement_count = 0
    max_no_improvement = 100
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < 3000:
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # Find the shift k that maximizes overlap
        k_opt = np.argmax(overlap) - (n_points - 1)
        # Compute gradient: (1-h) shifted by k_opt
        if k_opt == 0:
            grad = 1 - h
        elif k_opt > 0:
            grad = np.concatenate([np.zeros(k_opt), 1 - h[:-k_opt]])
        else:
            grad = np.concatenate([1 - h[-k_opt:], np.zeros(-k_opt)])
        
        # Nesterov momentum update
        momentum = beta * momentum - alpha * grad
        h = h + momentum
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Project to sum constraint
        h = project_to_sum_constraint(h, target_sum)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Learning rate adaptation
        if no_improvement_count > 100:
            alpha *= 0.95
            alpha = max(alpha, 1e-5)
            no_improvement_count = 0
            
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
