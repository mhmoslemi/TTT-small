import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 100
    max_generations = 1000
    tournament_size = population_size // 10
    mutation_rate = 0.1
    blending_min, blending_max = 0.2, 0.8
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand_sum = np.sum(h_rand)
        if h_rand_sum > 0:
            h_rand = h_rand * (target_sum / h_rand_sum)
        population.append(h_rand)
    
    def fitness(individual):
        return evaluate_overlap(individual, dx)
    
    # Track best solution
    best_individual = h.copy()
    best_fitness = fitness(best_individual)
    
    # Main GA loop
    no_improvement_count = 0
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all individuals in population
        fitnesses = [fitness(ind) for ind in population]
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_fitness:
            best_fitness = fitnesses[0]
            best_individual = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if no_improvement_count > 50:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            best_in_tournament = min(tournament, key=lambda i: fitnesses[i])
            selected.append(population[best_in_tournament])
        
        # Crossover and mutation
        new_population = [best_individual.copy()]
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, len(selected)-1)]
            
            # Blending crossover
            alpha = np.random.uniform(blending_min, blending_max)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            
            # Normalize sum
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.1
                noise = np.random.normal(0, 0.1, n_points)
                child += noise * mask
                child = np.clip(child, 0, 1)
                
                # Normalize sum
                child_sum = np.sum(child)
                if child_sum > 0:
                    child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        # Ensure population size
        while len(new_population) < population_size:
            new_population.append(np.random.rand(n_points))
            new_population[-1] = np.clip(new_population[-1], 0, 1)
            new_population[-1] = new_population[-1] * (target_sum / np.sum(new_population[-1]))
            
        population = new_population[:population_size]
        
        # Adaptive mutation rate
        mutation_rate = max(0.01, mutation_rate * 0.99)
    
    # Local refinement with gradient descent
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_individual, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 100, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_fitness:
            best_individual = refined_h
            best_fitness = final_c5
    
    # Final validation and return
    final_h = np.clip(best_individual, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
