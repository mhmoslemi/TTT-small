import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def update_h(h, i, alpha, target_sum):
    """Update h at index i with step size alpha."""
    h_old = h[i]
    h[i] = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
    scaling = target_sum / np.sum(h)
    h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    
    # Try to make meaningful improvements
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Find the indices that contribute to the maximum overlap
        max_indices = np.where(np.abs(overlap - current_max) < 1e-10)[0]
        
        # Create gradients for each point
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Approximate gradient calculation
            gradients[i] = overlap[i] * (1 - 2 * h[i])
        
        # Select the top 10% points to update
        top_k = max(1, n_points // 10)
        top_indices = np.argsort(np.abs(gradients))[::-1][:top_k]
        
        # Update selected indices
        for i in top_indices:
            # Estimate step size
            if abs(1 - 2 * h[i]) > 1e-10:
                alpha = -gradients[i] / (1 - 2 * h[i])
            else:
                alpha = 0.1 if gradients[i] < 0 else -0.1
            alpha = np.clip(alpha, -1, 1)
            
            # Try small steps to check improvement
            best_alpha = 0.0
            best_h_try = h.copy()
            
            # Try a few step sizes
            for step in np.linspace(0, 1, 10):
                temp_h = update_h(h.copy(), i, step, target_sum)
                temp_c5 = compute_c5(temp_h, dx)
                if temp_c5 < best_c5:
                    best_c5 = temp_c5
                    best_h = temp_h.copy()
                    best_alpha = step
                    best_h_try = temp_h.copy()
            
            if best_alpha != 0:
                h = best_h_try.copy()
        
        # Check for convergence
        iter_count += 1
        if iter_count % 1000 == 0:
            current_c5 = compute_c5(h, dx)
            if abs(current_c5 - best_c5) < 1e-6:
                no_improvement_count += 1
            else:
                no_improvement_count = 0
            if no_improvement_count > 5:
                break
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
