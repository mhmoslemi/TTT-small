import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * target_sum / np.sum(h)
    return h

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    # Initialize relaxed version
    h_relaxed = h + np.random.normal(0, 0.05, n_points)
    
    # Optimizer parameters
    learning_rate = 0.1
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lambda_reg = 1e-4
    m = np.zeros_like(h_relaxed)
    v = np.zeros_like(h_relaxed)
    
    # For adaptive learning rate
    last_improvement = 0
    no_improvement_count = 0
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current c5
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive learning rate and regularization
        if current_c5 < best_c5:
            no_improvement_count = 0
        else:

        if no_improvement_count > 100:
            learning_rate *= 0.9

        # Compute gradient approximation
        grad = np.zeros_like(h_relaxed)

        h_plus = project_to_constraint(h_plus, target_sum)
        h_minus = project_to_constraint(h_minus, target_sum)
        
        c5_plus = compute_c5(h_plus, dx)
        c5_minus = compute_c5(h_minus, dx)
        
        # Approximate gradient
        for i in range(n_points):

        # Add regularization term to gradient
        reg_term = lambda_reg * (h_relaxed - h)
        grad += reg_term
