import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment for sum
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
    
    # Normalize sum
    h = project_to_constraints(h, target_sum)
    
    # Evolution Strategies parameters
    population_size = 20
    elite_size = 8
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_iter = 20000
    
    # Initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        # Perturb initial vector with noise
        noise = np.random.normal(0, 0.05, n_points)
        candidate = h + noise
        candidate = project_to_constraints(candidate, target_sum)
        population.append(candidate)
    
    # Evaluate initial population
    fitnesses = [compute_c5(individual, dx) for individual in population]
    
    # Tracking best solution
    best_idx = np.argmin(fitnesses)
    best_individual = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # ES Loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate adaptive mutation rate
        t = iter_num / max_iter
        current_mutation_rate = mutation_rate_start + (mutation_rate_end - mutation_rate_start) * t
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        top_elite = [population[i] for i in sorted_indices[:elite_size]]
        rest = [population[i] for i in sorted_indices[elite_size:]]
        
        # Create new population
        new_population = top_elite.copy()
        new_fitnesses = [fitnesses[i] for i in sorted_indices[:elite_size]]
        
        # Mutate elite members
        for i in range(len(top_elite)):
            # Create mutated version
            noise = np.random.normal(0, current_mutation_rate, n_points)
            mutated = top_elite[i] + noise
            mutated = project_to_constraints(mutated, target_sum)
            new_population.append(mutated)
            new_fitnesses.append(compute_c5(mutated, dx))
        
        # If population size is not correct, we fill with random individuals
        while len(new_population) < population_size:
            individual = np.random.uniform(0, 1, n_points)
            individual = project_to_constraints(individual, target_sum)
            new_population.append(individual)
            new_fitnesses.append(compute_c5(individual, dx))
        
        # Update population and fitnesses
        population = new_population
        fitnesses = new_fitnesses
        
        # Update global best
        new_best_idx = np.argmin(fitnesses)
        if fitnesses[new_best_idx] < best_fitness:
            best_individual = population[new_best_idx].copy()
            best_fitness = fitnesses[new_best_idx]
            
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
