import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

    def objective(freq_coeffs):
        # Reconstruct h from frequency coefficients
        # freq_coeffs is real-valued as we want a real h
        # As h is real and even in our formulation, we can do:
        h_reconstructed = ifft(freq_coeffs).real
        
        # Project to [0,1] and re-normalize
        h_projected = np.clip(h_reconstructed, 0, 1)
        h_projected = h_projected * (target_sum / np.sum(h_projected))
        
        # Compute C5
        c5 = compute_c5(h_projected, dx)
        return c5

        h_projected = np.clip(h_reconstructed, 0, 1)

    # Try to optimize in frequency domain - use scipy minimize directly on h
    result = minimize(objective, h0, method='L-BFGS-B', 
                      bounds=[(0, 1) for _ in range(n)], 
                      options={'maxiter': max_iter, 'ftol': 1e-8})
    
    final_h = result.x
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (target_sum / np.sum(final_h))
    
    return final_h
