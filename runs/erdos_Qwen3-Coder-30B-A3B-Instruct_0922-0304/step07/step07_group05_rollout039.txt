import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add some random variation to the initial solution
        for _ in range(int(0.4 * population_size)):
            perturbation = np.random.normal(0, 0.01, n_points)
            particle = np.clip(h + perturbation, 0, 1)
            particle = normalize_h(particle, target_sum)
            particles.append(particle)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h)
    
    # Fill remaining particles
    while len(particles) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h)
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    
    # Track best solutions
    personal_best = particles.copy()
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO parameters
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 60
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
        
        # Update inertia weight
        w *= 0.98
        
        # Early termination check
        if global_best_fitness <= 0.3807:
            break
    
    # Final gradient descent refinement of the best particle
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    result = minimize(objective, global_best.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 50})

    if result.success:
        global_best = result.x
        global_best = np.clip(global_best, 0, 1)
        global_best = normalize_h(global_best, target_sum)
    
    final_c5 = compute_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
