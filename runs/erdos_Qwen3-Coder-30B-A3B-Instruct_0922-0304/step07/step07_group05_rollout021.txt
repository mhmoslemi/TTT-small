import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values):
    """Reference evaluation function."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and bounds [0,1]."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to have the correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Simple triangular pulse
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h = project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Iterative refinement with quadratic programming
    max_iters = 100
    patience = 10
    no_improvement_count = 0
    
    # For each iteration, we will find a descent direction using quadratic programming
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience or best_c5 <= 0.3807:
            break
            
        # Build quadratic approximation at current point
        # For simplicity use a finite difference approach for gradient
        eps = 1e-4
        grad = np.zeros(n_points)
        h_plus = best_h.copy()
        h_minus = best_h.copy()
        
        for i in range(n_points):
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure they are feasible
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            
            val_plus = compute_c5(h_plus, dx)
            val_minus = compute_c5(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
            
            # Restore
            h_plus[i] = best_h[i]
            h_minus[i] = best_h[i]
            
        # At this point, we have a gradient estimate
        # Try to solve a QP: min 0.5 * d^T * H * d + grad^T * d, s.t. constraints
        # For simplicity, we reconstruct a simpler QP
        # Just project the negative gradient direction onto constraints, like a simple line search
        
        # Try with negative gradient direction (approximate descent)
        d = -grad
        # Ensure we're taking a step that's not too large
        step_size = 0.01
        h_trial = best_h + step_size * d
        
        # Project onto feasible set 
        h_trial = project_to_feasible(h_trial, target_sum)
        
        # Simple backtracking line search
        c5_trial = compute_c5(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial
        else:
            # Try a smaller step size
            for _ in range(10):
                step_size *= 0.5
                h_trial = best_h + step_size * d
                h_trial = project_to_feasible(h_trial, target_sum)
                c5_trial = compute_c5(h_trial, dx)
                if c5_trial < best_c5:
                    best_h = h_trial
                    break
                    
    # Final check and refinement
    best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
