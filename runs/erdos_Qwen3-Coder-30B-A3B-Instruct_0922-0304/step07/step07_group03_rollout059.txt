import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

    else:

        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # Evolution Strategy parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_generations = 2000
    
    # Initialize population
    population = [h.copy()]
    fitness = [compute_c5(h, dx)]
    
    # Create initial population
    for _ in range(pop_size - 1):
        perturbed = h + np.random.normal(0, sigma_init, n_points)
        perturbed = project_to_constraints(perturbed, target_sum)
        population.append(perturbed)
        fitness.append(compute_c5(perturbed, dx))
    
    # Track best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # ES iteration
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update sigma
        sigma = sigma_init - (sigma_init - sigma_final) * gen / max_generations
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% + 1 elite
        elite_size = pop_size // 2 + 1
        elite_population = population[:elite_size]
        elite_fitness = fitness[:elite_size]
        
        # Generate new population by mutation
        new_population = []
        for i in range(pop_size - elite_size):
            parent = elite_population[np.random.randint(elite_size)]
            mutant = parent + np.random.normal(0, sigma, n_points)
            mutant = project_to_constraints(mutant, target_sum)
            new_population.append(mutant)
        
        # Combine elites and new population
        population = elite_population + new_population
        fitness = elite_fitness + [compute_c5(p, dx) for p in new_population]
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Local optimization for good candidates
        if best_c5 < 0.381:
            # Find candidates that are good enough
            good_indices = [i for i in range(len(fitness)) if fitness[i] < 0.381]
            for idx in good_indices:
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Apply local optimization with L-BFGS-B
                def objective(h_local):
                    return compute_c5(h_local, dx)
                
                def constraint_sum(h_local):
                    return np.sum(h_local) - target_sum
                
                # Setup bounds
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Minimize with bounds
                try:
                    result = minimize(
                        objective,
                        population[idx].copy(),
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 1000}
                    )
                    if result.success:
                        optimized_h = result.x
                        optimized_h = project_to_constraints(optimized_h, target_sum)
                        optimized_c5 = compute_c5(optimized_h, dx)
                        
                        if optimized_c5 < best_c5:
                            best_h = optimized_h.copy()
                            best_c5 = optimized_c5
                except:
                    pass
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
