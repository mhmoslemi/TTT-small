import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Start with simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try different heuristic approaches
    # 1. Gradient-based refinement using SLSQP
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try with different starting points
        for _ in range(3):
            if time.time() - start_time > budget_s - 5:
                break
            # Random perturbation
            h_try = h + np.random.normal(0, 0.01, n_points)
            h_try = np.clip(h_try, 0, 1)
            h_try_sum = np.sum(h_try)
            if h_try_sum > 0:
                h_try = h_try * (target_sum / h_try_sum)
            
            result = minimize(objective, h_try, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-8})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
    
    # 2. Genetic algorithm approach
    if time.time() - start_time < budget_s - 10:
        # Simple genetic algorithm
        population_size = 20
        generations = 50
        mutation_rate = 0.1
        
        population = []
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual_sum = np.sum(individual)
            if individual_sum > 0:
                individual = individual * (target_sum / individual_sum)
            population.append(individual)
        
        for gen in range(generations):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Evaluate fitness
            fitness_scores = []
            for individual in population:
                score = evaluate_overlap(individual, dx)
                fitness_scores.append(1.0 / (1.0 + score))  # Lower overlap = higher fitness
            
            # Keep top 50%
            sorted_indices = np.argsort(fitness_scores)[::-1]
            population = [population[i] for i in sorted_indices[:population_size//2]]
            
            # Create new population
            new_population = population.copy()
            for _ in range(population_size - len(population)):
                parent1, parent2 = np.random.choice(population, size=2, replace=False)
                # Crossover
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
                
                # Normalize
                child_sum = np.sum(child)
                if child_sum > 0:
                    child = child * (target_sum / child_sum)
                
                new_population.append(child)
            
            population = new_population
            
            # Check for best solution in this generation
            for individual in population:
                c5 = evaluate_overlap(individual, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = individual.copy()
    
    # 3. Pattern refinement with local search
    if time.time() - start_time < budget_s - 5:
        # Try small local perturbations around best solution
        current_h = best_h.copy()
        current_c5 = best_c5
        
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Make small random changes
            new_h = current_h.copy()
            i = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            new_h[i] += delta
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * (target_sum / new_sum)
            
            new_c5 = evaluate_overlap(new_h, dx)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
