import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, maxiter=1000):
    """Perform local optimization using L-BFGS-B."""
    def objective(x):
        # Reshape and ensure constraints
        h_temp = project_to_constraints(x, target_sum)
        return compute_c5(h_temp, dx)
    
    def jac(x):
        # Simple finite difference gradient approximation
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad
    
    bounds = [(0, 1) for _ in range(len(h))]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      jac=jac, options={'maxiter': maxiter}, 
                      callback=None)
    return project_to_constraints(result.x, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # ES parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_generations = 2000
    
    # Population
    population = [h.copy()]
    fitness = [compute_c5(h, dx)]
    
    # Generate initial population
    for _ in range(pop_size - 1):
        h_temp = h + np.random.normal(0, sigma_init, n_points)
        h_temp = project_to_constraints(h_temp, target_sum)
        population.append(h_temp)
        fitness.append(compute_c5(h_temp, dx))
    
    # Evolution loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% 
        keep_count = pop_size // 2
        population = population[:keep_count]
        fitness = fitness[:keep_count]
        
        # Update sigma
        sigma = sigma_init - (sigma_init - sigma_final) * gen / max_generations
        
        # Generate new population
        new_population = population.copy()
        new_fitness = fitness.copy()
        
        for _ in range(pop_size - keep_count):
            # Selection: Tournament
            idx1, idx2 = np.random.choice(keep_count, 2, replace=False)
            parent_idx = idx1 if fitness[idx1] < fitness[idx2] else idx2
            
            # Mutation
            child = population[parent_idx] + np.random.normal(0, sigma, n_points)
            child = project_to_constraints(child, target_sum)
            
            # Local optimization if good candidate
            if fitness[parent_idx] < 0.381:
                child = local_optimize(child, target_sum, dx)
            
            new_population.append(child)
            new_fitness.append(compute_c5(child, dx))
        
        population = new_population
        fitness = new_fitness
        
        # Check if we have improved solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < 0.3807:
            break
    
    # Return best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
