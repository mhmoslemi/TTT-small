import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_gradient(h, k_max):
    """Evaluate the gradient at shift k_max for linearization."""
    n = len(h)
    grad = np.zeros(n)
    # For overlap_k_max = sum_{i=0}^{n-k_max-1} h[i] * (1 - h[i + k_max])
    # Gradient w.r.t. h[i] = 1 - 2*h[i+k_max] for i in [0,n-k_max-1]
    for i in range(n - k_max):
        grad[i] += 1 - 2 * h[i + k_max]
    return grad

def solve_lp_direction(grad, n_points):

    # subject to sum(s) = 0, 0 <= s[i] <= 1
    # This is a simple LP: just choose s[i] such that <grad, s> is minimized
    
    # Sort indices by gradient value (ascending)

    # Set s to 0 or 1 based on which element should contribute to minimum cost
    s = np.zeros(n_points)

    # This is a bit tricky since we constrain sum(s) = 0
    # We'll fix sum(s) = 0 by modifying the signs
    
    # Simpler: just try setting parts to 0 or 1 to make objective small
    # Set s = 1 where gradient is negative (more negative = better to decrease)
    # and s = 0 where gradient is positive

        s[neg_indices] = 1  # Set all negative ones to 1

def backtracking_line_search(h, grad, s, k_max, dx, max_alpha=1.0):
    """Perform backtracking line search to find good step size."""
    alpha = max_alpha
    base_c5 = compute_c5(h, dx)
    for _ in range(10):
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, len(h_new) / 2.0)
        new_c5 = compute_c5(h_new, dx)
        # Check sufficient decrease (Armijo condition)
        if new_c5 < base_c5 - 1e-10 * alpha * np.dot(grad, s - h):
            return alpha, h_new
        alpha *= 0.5
    return alpha, h_new

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    patience = 0
    
    iterations = 0
    last_c5 = best_c5 + 1000
    
    while time.time() - start_time < budget_s - 1 and patience < 500:
        # Compute all overlaps to find worst case shift
        overlaps = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps)
        overlap_value = overlaps[k_max] * dx
        
        if overlap_value < best_c5:

        # Early termination if target achieved

            s = s + np.sum(s) / len(s) * (-1)  # Make sum zero

        s = np.clip(s, 0, 1)

        if last_c5 - best_c5 < 1e-8:
            patience += 1
        else:
            patience = 0
        last_c5 = best_c5
        
        iterations += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
