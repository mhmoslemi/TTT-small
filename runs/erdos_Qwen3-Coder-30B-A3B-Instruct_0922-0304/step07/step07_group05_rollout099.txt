```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create base triangular shape, then add noise
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h += np.random.normal(0, 0.01, n_points)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # For 600s budget, use around 60 iterations
    max_iterations = 60
    patience = 0
    patience_limit = 10
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Approximate gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        c5_base = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_base) / eps
        
        # Formulate and solve QP
        try:
            # Create symbolic variables for CVXPY
            h_var = Variable(n_points)
            
            # Objective: minimize quadratic approximation 
            # QP: min 0.5 * d^T * H * d + grad^T * d 
            # Since we're approximating the gradient, we use it directly
            
            # Linear objective (gradient term)
            objective = Minimize(grad @ h_var)
            
            # Constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                sum(h_var) == target_sum
            ]
            
            # Solve QP
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                d = h_var.value
                
                # Line search to find good step size
                step_size = 1.0
                for _ in range(20):
                    h_new = h - step_size * d
                    h_new = project_to_feasible(h_new, target_sum)
                    c5_new = compute_c5(h_new, dx)
                    
                    if c5_new < c5_base:
                        h = h_new
                        c5_base = c5_new
                        break
                    step_size *= 0.5
                    
                if c5_base < best_c5:
                    best_c5 = c5_base
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                # If QP fails, fallback to small gradient descent step
                step_size = 0.01
                h = h - step_size * grad
                h = project_to_feasible(h, target_sum)
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                    
        except Exception:
            # Fallback to simple gradient descent
            step_size = 0.01
            h = h - step_size * grad
            h = project_to_feasible(h, target_sum)
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                
        if patience >= patience_limit:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
