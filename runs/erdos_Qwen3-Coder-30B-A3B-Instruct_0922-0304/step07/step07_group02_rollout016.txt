import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to fix the sum exactly
    h = h * (target_sum / current_sum)
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_relaxed = np.array(initial_h_values)
        h_relaxed = np.clip(h_relaxed, 0, 1)
        h_relaxed = project_to_constraint(h_relaxed, target_sum)
    else:
        # Create an initial step function
        h_relaxed = np.zeros(n_points)
        h_relaxed[:n_points//2] = 1
        # Add randomness
        h_relaxed += np.random.normal(0, 0.02, n_points)
        h_relaxed = np.clip(h_relaxed, 0, 1)
        h_relaxed = project_to_constraint(h_relaxed, target_sum)
    
    # Optimizer parameters
    learning_rate = 0.1
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lambda_reg = 1e-4
    
    # Adam variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # For learning rate adaptation
    last_improvement = 0
    no_improvement_count = 0
    
    # Track best solution
    best_h = h_relaxed.copy()
    best_c5 = compute_c5(h_relaxed, dx)
    
    max_iter = 3000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h_relaxed, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_relaxed.copy()
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if no_improvement_count > 100:
            learning_rate *= 0.9
            no_improvement_count = 0
            if learning_rate < 1e-6:
                break
                
        # Sample gradient via finite difference
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Perturb h_relaxed[i]
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_to_constraint(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        # Update h_relaxed
        h_relaxed = h_relaxed - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Add regularization
        h_relaxed = h_relaxed - lambda_reg * (h_relaxed - best_h)
        
        # Project to feasible region
        h_relaxed = np.clip(h_relaxed, 0, 1)
        h_relaxed = project_to_constraint(h_relaxed, target_sum)
        
        # Adaptive regularization
        if iteration % 10 == 0 and np.any(np.abs(h_relaxed) > 1):
            lambda_reg *= 1.2
        elif iteration % 10 == 0 and lambda_reg > 1e-6:
            lambda_reg *= 0.8
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
