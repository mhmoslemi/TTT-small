import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try using convex optimization approach

        # Using CVXPY to solve with a relaxed formulation
        # Minimize max_k (h * shift_k(1-h)) 
        # Let's construct a matrix representing all shifts

        # Define the variable
        h_var = cp.Variable(n_points)
        
        # Objective: Maximize over all shifts, but in a different approach,
        # this simple convex optimization can only do what it can do

        # Set up problem via optimization using CVXPY
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # Create a correlated sum expression for one shift
