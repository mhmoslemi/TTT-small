import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Iterative improvement using projected gradient method
    max_iter = 2000
    alpha = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradients for all points
        # For each h_i, gradient is sum of (1 - h_{(i-k)%n}) for the maximizing k
        # We also track the current maximum overlap shift k*
        overlap_vals = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_vals)
        max_val = overlap_vals[max_k]
        
        # Compute analytical gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # The derivative w.r.t. h[i]
            shifted_idx = (i - max_k) % n_points
            grad[i] = 1 - h[shifted_idx]
        
        # Update h using gradient descent
        h_new = h - alpha * grad
        
        # Project to [0, 1] and maintain sum constraint
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Check improvement
        new_c5 = evaluate_overlap(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            alpha = min(alpha * 1.05, 0.1)  # Adaptive learning rate
        else:
            alpha *= 0.9  # Reduce learning rate on no improvement
            
        h = h_new
        
        # Convergence check
        if iteration > 10 and abs(best_c5 - new_c5) < tolerance:
            break
    
    # Final refinement with scipy.optimize
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try multiple random initializations for better results
        best_refined_h = best_h.copy()
        best_refined_c5 = best_c5
        
        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            # Perturb current best
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum > 0:
                h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)
            
            result = minimize(objective, h_perturbed, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-7})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_sum = np.sum(refined_h)
                if refined_sum > 0:
                    refined_h = refined_h * (target_sum / refined_sum)
                final_c5 = evaluate_overlap(refined_h, dx)
                if final_c5 < best_refined_c5:
                    best_refined_h = refined_h
                    best_refined_c5 = final_c5
        
        best_h = best_refined_h
        best_c5 = best_refined_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
