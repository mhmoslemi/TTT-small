import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_constraints(h, target_sum)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Fourier-based optimization approach
    fft_size = 2 * n_points - 1
    fft_freqs = np.fft.fftfreq(fft_size, d=dx)
    
    # Define optimization variables in frequency domain
    H_real = cp.Variable(fft_size)
    H_imag = cp.Variable(fft_size)
    
    # Constraints in frequency domain
    # Real part of h must be real and symmetric
    # h = ifft(H), so H must satisfy conjugate symmetry
    H = H_real + 1j * H_imag
    
    # Define the objective in terms of frequency domain
    # Overlap in time domain = correlation h * (1-h)
    # In frequency domain: fft(h) * conj(fft(1-h)) = fft(h) * (1 - fft(h))
    
    # Simplified: Constrain to minimize max of overlap
    # Build product in frequency domain
    def get_overlap_from_freq(H_vec):
        # H_vec: complex vector of length fft_size
        h_freq = H_vec
        complement_freq = 1.0 - H_vec
        # Correlation result in freq domain
        overlap_freq = h_freq * np.conj(complement_freq)
        return np.real(np.fft.ifft(overlap_freq))
    
    # Solve with CVXPY
    # Initialize with current h
    h_init = h.copy()
    H_init = np.fft.fft(h_init, n=fft_size)
    
    # Set up optimization problem
    H_real_init = H_init.real
    H_imag_init = H_init.imag
    
    # Fix real and imaginary parts initially
    H_real.value = H_real_init
    H_imag.value = H_imag_init
    
    # Add regularization and bounds
    # We want to minimize the peak correlation
    
    # Objective: Minimize max of (h * (1 - h)) overlap
    # In frequency: max of real(ifft(H * conj(1 - H))) over all shifts
    max_overlap = cp.Variable()
    
    # Simple gradient descent as fallback
    def refine_h(h_vec, max_iter=50):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            # Compute overlap values, find max
            overlap = correlate(h_new, 1 - h_new, mode="full")
            c5 = np.max(overlap) * dx
            
            # Small perturbations and project
            perturb = np.random.normal(0, 0.001, n_points)
            h_test = h_new + perturb
            h_test = project_to_constraints(h_test, target_sum)
            
            test_overlap = correlate(h_test, 1 - h_test, mode="full")
            test_c5 = np.max(test_overlap) * dx
            
            if test_c5 < c5:
                h_new = h_test
        return h_new

    # Run the main optimization with some refinement and other ideas
    h_final = h.copy()
    best_c5 = compute_c5(h_final, dx)
    
    # Adaptive refinement
    refine_iters = 0
    for i in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        h_test = h_final + np.random.normal(0, 0.005, n_points)
        h_test = project_to_constraints(h_test, target_sum)
        
        c5_new = compute_c5(h_test, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_final = h_test.copy()
        
        # Periodic project to keep feasible and refine
        refine_iters += 1
        if refine_iters > 10:
            h_final = refine_h(h_final)
            best_c5 = compute_c5(h_final, dx)
            refine_iters = 0
            
        if best_c5 <= 0.3807:
            break

    return (h_final.tolist(), best_c5, n_points)
