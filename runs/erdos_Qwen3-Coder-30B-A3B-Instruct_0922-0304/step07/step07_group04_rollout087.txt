import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing
    T0 = 1.0
    alpha = 0.999
    T = T0
    max_iter = 100000
    eta = 0.01
    lambda1 = 100
    lambda2 = 100
    
    def compute_energy(h_vec, c5_val):
        # Constraint penalties
        bound_penalty = np.sum(np.maximum(0, h_vec - 1)**2 + np.maximum(0, -h_vec)**2)
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return c5_val + lambda1 * bound_penalty + lambda2 * sum_penalty
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb h
        h_new = h.copy()
        perturb = np.random.normal(0, eta * T, n_points)
        h_new += perturb
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Compute new energy
        c5_new = evaluate_overlap(h_new, dx)
        energy_new = compute_energy(h_new, c5_new)
        energy_old = compute_energy(h, best_c5) if h is best_h else compute_energy(h, evaluate_overlap(h, dx))
        
        # Accept or reject
        dE = energy_new - energy_old
        if dE < 0 or np.random.rand() < np.exp(-dE / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Cooling
        T = T0 * (alpha ** i)
        
        # Adapt learning rate
        if T < 0.0001:
            eta = 0.005
            
    # Final refinement with gradient methods
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    def gradient(h_vector):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vector)
        base_val = objective(h_vector)
        for i in range(len(h_vector)):
            h_plus = h_vector.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        grad = gradient(h_refined)
        h_refined -= 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined_sum = np.sum(h_refined)
        if h_refined_sum > 0:
            h_refined = h_refined * (target_sum / h_refined_sum)
        new_c5 = evaluate_overlap(h_refined, dx)
        if new_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = new_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
