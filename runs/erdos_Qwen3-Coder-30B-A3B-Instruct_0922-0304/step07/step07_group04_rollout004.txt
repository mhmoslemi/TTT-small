import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient-based optimization
    def compute_gradient(h, dx):
        # Compute gradient of overlap w.r.t. h
        # This is an approximation: we compute gradient of the max overlap with a fixed k
        # For better accuracy, it should consider how each component affects all overlaps
        # But for speed, use a sampling approach
        n = len(h)
        grad = np.zeros(n)
        eps = 1e-6
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            sum_plus = np.sum(h_plus)
            if sum_plus > 0:
                h_plus = h_plus * (target_sum / sum_plus)
            c5_plus = evaluate_overlap(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            sum_minus = np.sum(h_minus)
            if sum_minus > 0:
                h_minus = h_minus * (target_sum / sum_minus)
            c5_minus = evaluate_overlap(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Enhanced gradient descent with momentum and adaptive learning rate
    max_iter = 2000
    eta = 1e-3
    beta = 0.9
    momentum = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Use a local optimization approach with better gradient handling
        # Compute current gradient
        grad = compute_gradient(h, dx)
        
        # Update with momentum
        momentum = beta * momentum + grad
        h_new = h - eta * momentum
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate new solution
        c5_new = evaluate_overlap(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update h
        h = h_new
        
        # Adaptive learning rate
        eta *= 0.9995
        
        # Check for convergence
        if np.linalg.norm(grad) < 1e-8:
            break
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
