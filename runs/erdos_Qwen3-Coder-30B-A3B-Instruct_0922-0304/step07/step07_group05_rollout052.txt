import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with either initial_h_values or a constructed one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 100
    max_iter = 60
    w_start = 0.8
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Slightly perturb the best known solution to diversify
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(compute_c5(particle, dx))
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_end / w_start) ** (iter_count / max_iter)
        
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest[i] - particle) + 
                       c2 * r2 * (gbest - particle))
            
            # Update particle
            particle = particle + velocity
            particle = project_to_feasible(particle, target_sum)
            
            particles[i] = particle
            
            # Evaluate and update personal best
            fitness = compute_c5(particle, dx)
            if fitness < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = fitness
                
                if fitness < gbest_val:
                    gbest = particle.copy()
                    gbest_val = fitness
        
        # Early termination if improvement is minimal
        if gbest_val <= 0.3807:
            break
            
        if iter_count > 10 and all(np.abs(pbest_vals[i] - gbest_val) < 1e-6 for i in range(10)):
            break
    
    # Final refinement with local search
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Try local optimization on the best solution
    refined_h = gbest.copy()
    for _ in range(20):
        result = minimize(objective, refined_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(refined_h, dx)
    return (refined_h.tolist(), final_c5, n_points)
