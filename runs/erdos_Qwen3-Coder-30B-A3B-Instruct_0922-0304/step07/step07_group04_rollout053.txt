import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Enhanced gradient descent with adaptive regularization
    def compute_gradient(h, dx):
        """Compute gradient of overlap w.r.t. h."""
        # For each position, compute how much changing it affects the overlap
        grad = np.zeros_like(h)
        # Compute overlap for all shifts
        full_length = 2 * n_points - 1
        result = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(result)
        # Find the maximum overlap and its corresponding shift
        shift = max_idx - (n_points - 1)
        
        # Simple finite difference gradient approximation
        # For now, use a simpler approach - analytical gradient of the max
        # In practice this would require computing the derivative correctly,
        # but for now we'll use a projected gradient approach
        
        # Gradient of overlap w.r.t. h_i
        # Only contributes for indices that are shifted by some k
        for i in range(n_points):
            # For shift k = shift (the one causing max), 
            # contributing terms are [h[i] * (1 - h[i-shift]), h[i+shift] * (1 - h[i])]
            # But there are 2 directions, need full consideration
            # Focusing on single direction for simplification:
            # Gradients scaling to 1
            if shift >= 0:
                grad[i] += 1 - h[(i-shift) % n_points]
            else:
                grad[i] += 1 - h[(i+shift) % n_points]
                
        return grad
    
    # Adaptive gradient descent with momentum
    max_iter = 10000
    learning_rate = 1e-3
    momentum = 0.9
    regularization_weight = 1e-4
    velocity = np.zeros_like(h)
    c5_history = []
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_overlap(h, dx)
        c5_history.append(current_c5)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute gradient
        try:
            grad = compute_gradient(h, dx)
        except Exception:
            break
            
        # Add regularization term to push towards sum constraint
        sum_diff = np.sum(h) - target_sum
        regularized_grad = grad + regularization_weight * sum_diff
        
        # Nesterov momentum
        velocity = momentum * velocity - learning_rate * regularized_grad
        h_new = h + momentum * velocity - learning_rate * regularized_grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        current_sum_new = np.sum(h_new)
        if current_sum_new > 0:
            h_new = h_new * (target_sum / current_sum_new)
            
        # Line search idea - if too large, try smaller learning rate
        h_trial = h_new.copy()
        trial_c5 = evaluate_overlap(h_trial, dx)
        
        if trial_c5 < current_c5:
            h = h_trial.copy()
        else:
            # Backtrack
            learning_rate *= 0.5
            if learning_rate < 1e-8:
                break
    
    # Final validation and return
    final_h = best_h.copy()
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
