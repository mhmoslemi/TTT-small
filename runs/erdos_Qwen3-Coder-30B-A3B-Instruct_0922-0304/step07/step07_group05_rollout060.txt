import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with PSO
    population_size = 100
    population = []
    velocities = []
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Generate small random variations of the initial solution
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            population.append(variation.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Create random population
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        for _ in range(population_size):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            population.append(variation.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal best positions and fitnesses
    personal_best_positions = [p.copy() for p in population]
    personal_best_fitnesses = [compute_c5(p, dx) for p in population]
    
    # Find global best
    best_idx = np.argmin(personal_best_fitnesses)
    global_best_position = personal_best_positions[best_idx].copy()
    global_best_fitness = personal_best_fitnesses[best_idx]
    
    # PSO parameters
    w = 0.8  # inertia weight
    c1 = 1.5  # cognitive coefficient
    c2 = 1.5  # social coefficient
    max_iter = 60
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity and position
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                          c1 * r1 * (personal_best_positions[i] - population[i]) +
                          c2 * r2 * (global_best_position - population[i]))
            
            # Apply velocity to position
            population[i] += velocities[i]
            
            # Enforce bounds and constraint
            population[i] = np.clip(population[i], 0, 1)
            population[i] = normalize_h(population[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(population[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fitness
                personal_best_positions[i] = population[i].copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_position = population[i].copy()
        
        # Update inertia weight
        w *= 0.98
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    best_h = global_best_position.copy()
    
    # Refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Run several rounds of refinement
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
