import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 100
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Start with a modified version of initial solution
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
        for _ in range(n_particles // 2):
            perturbation = np.random.normal(0, 0.02, n_points)
            new_h = h + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            particles.append(new_h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
        for _ in range(n_particles - 1):
            new_h = np.random.rand(n_points)
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            particles.append(new_h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.8  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    max_iter = 60
    
    # Track global best
    gbest_idx = 0
    gbest_c5 = compute_c5(particles[0], dx)
    gbest_h = particles[0].copy()
    
    # Track stagnation
    stagnation_count = 0
    last_improvement = 0
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update particle velocity
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (particles[i] - particles[i])  # current best
            social = c2 * r2 * (gbest_h - particles[i])
            
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update particle position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if c5 < compute_c5(particles[i], dx):
                pass  # Already using personal best
            
            # Update global best if needed
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_h = particles[i].copy()
                last_improvement = iteration
                
        # Adjust parameters
        if iteration % 10 == 0:
            w *= 0.98
            if w < 0.3:
                w = 0.3
                
        # Early termination
        if gbest_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if iteration - last_improvement > 20:
            # Increase variation and reset
            for i in range(1, n_particles):
                noise = np.random.uniform(-0.05, 0.05, n_points)
                particles[i] = np.clip(particles[i] + noise, 0, 1)
                particles[i] = normalize_h(particles[i], target_sum)
                
            stagnation_count += 1
            if stagnation_count > 1:
                break  # Stop if stagnated too long
    
    # Refinement with local optimization on gbest
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Run refinement
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, gbest_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        if result.success:
            gbest_h = result.x
            gbest_h = np.clip(gbest_h, 0, 1)
            gbest_h = normalize_h(gbest_h, target_sum)
            new_c5 = compute_c5(gbest_h, dx)
            if new_c5 < gbest_c5:
                gbest_c5 = new_c5
        else:
            break
    
    return (gbest_h.tolist(), gbest_c5, n_points)
