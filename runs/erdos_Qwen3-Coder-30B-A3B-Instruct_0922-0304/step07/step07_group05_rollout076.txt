import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Create diverse initial population
    for i in range(population_size):
        if i == 0:
            individual = h.copy()
        elif i < population_size // 5:  # 20% from initial
            individual = h + np.random.normal(0, 0.05, n_points)
        else:  # Rest random
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
        individual = project_feasible(individual, target_sum)
        population.append(individual)
    
    # PSO parameters
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = [pop.copy() for pop in population]
    velocities = [np.random.rand(n_points) * 0.01 for _ in range(population_size)]
    pbest = [p.copy() for p in particles]
    pbest_vals = [compute_c5(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    max_iter = 60
    stagnation_count = 0
    stagnation_limit = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], target_sum)
            
            # Evaluate
            current_val = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Update inertia weight
        w = max(0.3, w * 0.98)
        
        # Early stopping
        if gbest_val <= 0.3807:
            break
            
        if stagnation_count >= stagnation_limit:
            break
    
    # Final refinement with gradient descent method: SLSQP
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    refined_h = gbest.copy()
    for _ in range(20):  # More rounds of local refinement
        result = minimize(objective, refined_h.copy(), method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = project_feasible(refined_h, target_sum)
        else:
            break
            
    final_c5 = compute_c5(refined_h, dx)
    return (refined_h.tolist(), final_c5, n_points)
