import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values with sawtooth pattern
        h = np.sin(np.pi * np.arange(n_points) / n_points) ** 2
        h = project_to_constraints(h, target_sum)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Mutation parameters
    epsilon_initial = 0.2
    epsilon_min = 0.001
    decay_factor = 0.99
    decay_interval = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Adapt epsilon
        epsilon = max(epsilon_min, epsilon_initial / (1.0 + 0.001 * iteration))
        
        # Generate perturbation
        v = np.random.randn(n_points)
        v = v / np.linalg.norm(v)
        mutation = epsilon * v
        
        # Apply and project
        h_candidate = h + mutation
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        c5_candidate = compute_c5(h_candidate, dx)
        
        # Accept improvement
        if c5_candidate < best_c5:
            h = h_candidate.copy()
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
            # Adjust epsilon more aggressively
            epsilon = max(epsilon_min, epsilon * decay_factor)
        else:
            # Keep current solution, dampen epsilon slightly
            epsilon = max(epsilon_min, epsilon * 0.9999)
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        # Periodic logging
        if iteration % 1000 == 0:
            pass
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
