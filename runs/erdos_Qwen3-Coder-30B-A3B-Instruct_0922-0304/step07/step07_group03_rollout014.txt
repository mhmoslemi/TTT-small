import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def initialize_h(n_points):

    else:
        # Create a sawtooth pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
        h = np.clip(h, 0, 1)
        target_sum = n_points / 2
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)

    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    h = initialize_h(n_points)
    
    # Project to constraints
    h = project_to_constraints(h, target_sum)
    
    # Optimization parameters
    alpha_0 = 0.1
    tau = 1000
    max_iter = 10000
    early_stop = False
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient descent with adaptive learning rate
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Generate surrogate gradient (simplified)
        # For simplicity, we compute gradient using finite differences
        gradient = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_corr_plus = np.max(corr_plus)
            gradient[i] = (max_corr_plus - max_corr) / eps
        
        # Update
        alpha_t = alpha_0 / (1 + t / tau)
        h = h - alpha_t * gradient
        h = project_to_constraints(h, target_sum)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
