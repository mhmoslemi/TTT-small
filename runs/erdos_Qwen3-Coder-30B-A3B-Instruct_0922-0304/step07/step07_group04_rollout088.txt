import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing
    T0 = 1.0
    alpha = 0.999
    T = T0
    eta = 0.01
    lambda1 = 100
    lambda2 = 100
    
    def compute_energy(h_vec, c5_val):
        # Constraint penalties
        bound_penalty = np.sum(np.square(np.clip(h_vec, 0, 1) - h_vec))
        sum_penalty = (np.sum(h_vec) - target_sum)**2
        return c5_val + lambda1 * bound_penalty + lambda2 * sum_penalty
    
    def perturb(h_vec, eta_val, T_val):
        # Gaussian perturbation with temperature scaling
        delta = np.random.normal(0, eta_val * T_val, len(h_vec))
        new_h = h_vec + delta
        # Clip to [0,1] and re-normalize
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        return new_h
    
    iterations = 0
    stagnant = 0
    max_stagnant = 1000
    
    while time.time() - start_time < budget_s - 5 and stagnant < max_stagnant:
        # Perturb
        new_h = perturb(h, eta, T)
        new_c5 = evaluate_overlap(new_h, dx)
        new_energy = compute_energy(new_h, new_c5)
        old_energy = compute_energy(h, best_c5)
        
        # Metropolis criterion
        delta = new_energy - old_energy
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                stagnant = 0  # Reset stagnation counter on improvement
            else:
                stagnant += 1
        else:
            stagnant += 1
            
        T *= alpha
        iterations += 1
    
    # Local refinement using gradient
    def objective(h_vec):
        return evaluate_overlap(h_vec, dx)
    
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': 100, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
