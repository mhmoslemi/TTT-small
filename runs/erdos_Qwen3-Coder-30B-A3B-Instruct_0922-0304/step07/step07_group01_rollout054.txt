import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    h_cvx = cp.Variable(n_points)
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == target_sum]
    
    # Use a conservative estimate for C initially
    C = cp.Variable()
    # We build a matrix to represent the convolution constraint
    # As a relaxation, we ensure all values of the convolution are bounded by C
    
    # Instead, we solve with a simplified approach concentrating on key constraints
    # and then refining the solution
    
    # Solve convex relaxation
    try:
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_cvx_sol = h_cvx.value
            h = project_to_constraints(h_cvx_sol, target_sum)
    except:
        pass
    
    # Refinement strategy: local search around best solution   
    def evaluate_and_improve(h_local):
        c5 = compute_c5(h_local, dx)
        return c5

    # Generate initial step function based on continuous solution
    h_initial = h.copy()
    
    # Iterative refinement with hill climbing-style search
    n_iterations = 0
    max_iterations = 1000
    
    # Track best state
    best_h = h_initial.copy()
    best_c5 = evaluate_and_improve(best_h)
    
    # Try adjusting step positions via gradient-like search
    step_positions = np.linspace(0, 2, n_points+1)
    h_refined = h_initial.copy()
    
    while time.time() - start_time < budget_s - 1 and n_iterations < max_iterations:
        h_trial = h_refined.copy()
        
        # Perturb a few points
        perturb_idx = np.random.choice(n_points, min(5, n_points//4), replace=False)
        for i in perturb_idx:
            if np.random.random() > 0.5:
                h_trial[i] = min(1.0, h_trial[i] + 0.02 * np.random.random())
            else:
                h_trial[i] = max(0.0, h_trial[i] - 0.02 * np.random.random())
                
        # Ensure it still satisfies sum constraint
        h_trial = project_to_constraints(h_trial, target_sum)
        
        c_trial = evaluate_and_improve(h_trial)
        if c_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c_trial
            
        # Improve h_refined slightly toward better solution
        h_refined = h_trial.copy()
        
        n_iterations += 1
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
