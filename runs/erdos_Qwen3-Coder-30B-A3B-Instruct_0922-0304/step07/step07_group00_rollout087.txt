import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def create_basis_functions(n_points, n_basis=16):
    """Create basis functions for convex combination."""
    basis = []
    for i in range(n_basis):
        # Create piecewise constant basis functions
        step = n_points // (n_basis + 1)
        h = np.zeros(n_points)
        start = i * step
        end = min(start + step, n_points)
        if start < n_points:
            h[start:end] = 1.0
        basis.append(h)
    return np.array(basis)

def get_basis_overlap_integrals(basis):
    """Precompute overlap integrals A_{ij}^k."""
    n_basis = basis.shape[0]
    n_points = basis.shape[1]
    # For each k, precompute all A_{ij}^k
    overlaps = {}
    for k in range(2 * n_points):
        overlap_k = np.zeros((n_basis, n_basis))
        for i in range(n_basis):
            for j in range(n_basis):
                # Shift basis j by k
                if k == 0:
                    overlap_k[i, j] = np.sum(basis[i] * basis[j])
                else:
                    shifted = np.roll(basis[j], k)
                    overlap_k[i, j] = np.sum(basis[i] * shifted)
        overlaps[k] = overlap_k
    return overlaps

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Create basis functions
    basis = create_basis_functions(n_points, n_basis=16)
    # Normalize basis functions to have integral 1/2 (we're working with dx=2/n)
    for i in range(basis.shape[0]):
        basis[i] *= (target_sum / np.sum(basis[i]))
    
    # Precompute overlapping integrals
    overlaps = get_basis_overlap_integrals(basis)
    
    best_c5 = float('inf')
    best_h = None
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Decompose into basis
        A = basis.T
        # Use least squares to find coefficients
        try:
            coeffs = np.linalg.lstsq(A, initial_h, rcond=None)[0]
            coeffs = np.maximum(coeffs, 0)
            coeffs = coeffs / np.sum(coeffs) * target_sum / np.sum(initial_h)
        except:
            coeffs = np.ones(len(basis)) / len(basis)
        h = np.dot(basis.T, coeffs)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Default initial guess
        coeffs = np.ones(len(basis)) / len(basis)
        h = np.dot(basis.T, coeffs)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Iterative refinement using basis convex combination
    n_iter = 0
    max_iter = 1000
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and n_iter < max_iter:
        n_iter += 1
        # Solve for coefficients using convex optimization
        lambda_coeffs = Variable(len(basis))
        
        # Approximate objective as a convex function of overlaps
        obj = 0
        for k in range(2 * n_points):
            overlap_k = overlaps[k]
            sum_term = 0
            for i in range(len(basis)):
                for j in range(len(basis)):
                    sum_term += lambda_coeffs[i] * lambda_coeffs[j] * (overlap_k[i, j] - overlap_k[i, j] * overlap_k[j, j])
            obj += sum_term

        # Constraints
        constraints = [lambda_coeffs >= 0, sum(lambda_coeffs) == 1]
        
        # Form and solve the problem
        prob = Problem(Minimize(obj), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                coeffs = lambda_coeffs.value
                # Reconstruct h
                h = np.dot(basis.T, coeffs)
                # Renormalize to satisfy integral constraint
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
                
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    last_improvement = n_iter
                    if best_c5 <= 0.3807:
                        break
                        
                # Add some randomness for diversity
                coeffs += np.random.normal(0, 0.01, len(coeffs))
                coeffs = np.maximum(coeffs, 0)
                coeffs = coeffs / np.sum(coeffs) * target_sum / np.sum(h)
                
        except:
            # If optimization fails, fall back to a simple perturbation
            coeffs = np.random.rand(len(basis))
            coeffs = coeffs / np.sum(coeffs)
            
        # Occasionally add new basis functions
        if n_iter % 100 == 0 and len(basis) < 32:
            # Add some variation to the basis
            new_basis = create_basis_functions(n_points, n_basis=1)
            new_basis = new_basis[0] * np.random.rand()
            basis = np.vstack([basis, new_basis])
            # Update overlaps
            overlaps = get_basis_overlap_integrals(basis)
        
        # Early stopping criteria
        if n_iter - last_improvement > 100:
            break

    return (best_h.tolist(), best_c5, n_points)
