import numpy as np
from scipy.fft import fft, ifft
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    # Using numpy's correlate for cross-correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a smooth initial function
        h = np.zeros(n_points)
        # Place a triangular pulse in the middle
        mid = n_points // 2
        half_width = n_points // 4
        for i in range(max(0, mid - half_width), min(n_points, mid + half_width)):
            h[i] = 1.0 - abs(i - mid) / half_width
        # Normalize
        h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Remez-like optimization in frequency domain
    max_iterations = 20
    alpha = 0.1
    decay = 0.95
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the cross-correlation
        overlap = np.correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        
        # The maximum occurs at lag
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute the frequency domain of h
        H = fft(h)
        # Spacing of frequency bins
        freqs = np.fft.fftfreq(len(h), dx)
        
        # Create a phase inversion to reduce maximum overlap
        # This is a heuristic: try to modify h to reduce peak at lag
        # Create a sinusoidal pattern based on the dominant frequency
        # Find dominant frequency by looking at peak in overlap
        phase_inv = np.exp(2j * np.pi * freqs * max_lag * dx)
        
        # Invert the phase to adjust h
        H_new = H * phase_inv
        
        # Inverse transform
        h_new = np.real(ifft(H_new))
        
        # Adjust amplitude based on the required average
        # Apply the change and maintain sum constraint
        h_update = h_new - h
        h = h + alpha * h_update
        
        # Project back to [0,1]
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h = h * (target_sum / np.sum(h))
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        
        # Update best known solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive learning rate
        alpha *= decay
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
