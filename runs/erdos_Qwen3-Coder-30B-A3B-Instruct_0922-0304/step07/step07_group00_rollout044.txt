import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Initialize momentum terms
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if best_c5 <= 0.3807:
            break
            
        # Gradient computation
        # Compute all overlaps
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k]
        
        # Compute gradient for each point
        g = np.zeros(n_points)
        for i in range(n_points):
            # Compute derivative of overlap w.r.t. h[i]
            # For shift k where max overlap occurs
            if max_k < n_points:
                # h[i] contributes to overlap when it's involved
                if i < n_points - max_k:
                    # Specifically: (1 - h[i + max_k]) term
                    g[i] += (1 - h[i + max_k]) * dx
            else:
                # This is the case when max_k >= n_points
                # This is the case of correlation when we wrap around
                # Since convolution is essentially shift & multiply, check:
                # The North-West to South-East diagonals in the correlation matrix
                # But full mode does not wrap, so ignore this edge case
                # Actually, need derivatives for all k and sum them up properly
                
        # Simplified gradient approximation for simple ascent:
        # This will ideally be replaced with exact gradient
        # For now, we will use a simpler but effective approach:
        # The gradient direction is roughly the weight of where we have bumpy parts
        # to alleviate fine structures.
        # Let's just use a basic GD approach with fixed learning rate but fit steps
        
        # Compute gradient approximately via numerical differentiation
        eps = 1e-4
        g = np.zeros(n_points)
        for i in range(n_points):
            # Perturb h[i] 
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 1e-8, 1 - 1e-8)
            h_plus = h_plus / np.sum(h_plus) * target_sum
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 1e-8, 1 - 1e-8)
            h_minus = h_minus / np.sum(h_minus) * target_sum
            c5_minus = compute_c5(h_minus, dx)
            
            g[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update parameters with Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h space and ensure constraints
        h = sigmoid(w)
        # Renormalize
        h = h / np.sum(h) * target_sum
        # Clip to avoid numerical issues
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-8))
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
