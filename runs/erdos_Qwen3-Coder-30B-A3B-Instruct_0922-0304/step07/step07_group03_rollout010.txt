import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

        h = project_to_constraints(h, target_sum)
    
    # Convex relaxation parameters
    max_iter = 10000
    alpha_0 = 0.1
    tau = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex relaxation loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Learning rate decay
        alpha = alpha_0 / (1 + iter_num / tau)

        # surrogate = 0.5 * sum(corr[k]^2 * dx^2)
        # grad = corr * dx^2 (approximating gradient)
        grad = corr * dx**2
        
        # Update h using projected gradient descent
        h_new = h - alpha * grad
        
        # Apply projections
        h = project_to_constraints(h_new, target_sum)

        if best_c5 <= 0.3807:
