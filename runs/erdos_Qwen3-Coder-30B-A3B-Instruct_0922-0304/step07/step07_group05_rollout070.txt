import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a more structured initial function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Track stagnation
    stagnation_count = 0
    
    # Main loop
    max_iterations = 60
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local optimization using quadratic approximation
        def objective(h_vec):
            h = h_vec.copy()
            h = project_to_feasible(h, target_sum)
            return compute_c5(h, dx)
        
        # Compute gradient numerically using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        
        # Try quadratic approximation using a small subset of points
        # Form QP: minimize 0.5 * d^T * H * d + grad^T * d 
        # subject to 0 <= d <= 1, sum(d) = 0 (for change vector)
        
        # For simplicity, use a conjugate gradient like approach with line search
        # Start with a simple gradient descent step, but make it more sophisticated
        
        # Attempt a more sophisticated local search: trust region approach
        # Solve QP for descent direction using CVXPY
        try:
            # Define variables
            d = cp.Variable(n_points)
            
            # Constraint 1: Each change is between -1 and 1 (not pure bounds but since we 
            # work in a small neighborhood, we can approximate)
            # Constraint 2: Sum of changes is 0 (since we only care about relative change)
            constraints = [cp.sum(d) == 0]
            
            # Constraint 3: 0 <= h + d <= 1
            # Make it safe with 1e-6 padding on both sides
            for i in range(n_points):
                constraints.append(d[i] >= -h[i] + 1e-6)
                constraints.append(d[i] <= 1 - h[i] - 1e-6)
                
            # Objective: quadratic approximation around current h
            # We will use finite differences to build Hessian approximation
            # Instead of full Hessian, just set H to diagonal with a small positive value for simplicity
            
            # Quadratic approximation: 0.5*d^T * H * d + grad^T * d
            # Let H be diagonal with 1e-3 for all diagonals
            H = np.eye(n_points) * 1e-3
            
            # Simplify objective to minimize ||d||^2 + grad^T * d
            # Add a small regularization to avoid exact zero Hessian
            objective_func = cp.Minimize(0.5 * cp.quad_form(d, H) + grad.T @ d)
            
            # Solve QP
            prob = cp.Problem(objective_func, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                direction = d.value
                # Ensure we don't have huge steps
                step_size = 0.1
                new_h = h + step_size * direction
                
                # Project back to feasible region
                new_h = project_to_feasible(new_h, target_sum)
                
                # Evaluate
                new_c5 = compute_c5(new_h, dx)
                
                if new_c5 < best_c5:
                    h = new_h
                    best_c5 = new_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                # Fallback to standard step with SGD-like descent
                step_size = 0.05
                step = -grad * step_size
                new_h = h + step
                new_h = project_to_feasible(new_h, target_sum)
                new_c5 = compute_c5(new_h, dx)
                
                if new_c5 < best_c5:
                    h = new_h
                    best_c5 = new_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        except Exception:
            # Fallback to simple gradient descent with line search
            step_size = 0.05
            step = -grad * step_size
            new_h = h + step
            new_h = project_to_feasible(new_h, target_sum)
            new_c5 = compute_c5(new_h, dx)
            
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                stagnation_count = 0
            else:
                stagnation_count += 1

        # Early stopping condition
        if stagnation_count > 10:
            break
            
        if best_c5 <= 0.3807:
            break

    # Final check and refinement
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
