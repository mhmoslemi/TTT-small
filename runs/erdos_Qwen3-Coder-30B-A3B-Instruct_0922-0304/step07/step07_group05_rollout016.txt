import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        # Add some smoothness
        h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
    
    # Normalize to exact sum constraint
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Remez-like iteration with frequency domain adjustment
    max_iter = 30
    alpha = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Identify dominant frequency from correlation
        # Use FFT to get frequency information
        corr_fft = np.fft.fft(overlap)
        freqs = np.fft.fftfreq(len(overlap), d=dx)
        
        # Find dominant frequency (excluding DC)
        dominant_freq_idx = np.argmax(np.abs(corr_fft[1:])) + 1
        
        # Phase inversion adjustment in time domain
        if dominant_freq_idx < len(freqs) // 2:
            # Apply sinusoidal update
            k = freqs[dominant_freq_idx]
            update = alpha * np.sin(2 * np.pi * k * np.arange(n_points) * dx)
            h_new = h + update
            
            # Project back to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            # Update if improvement
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
        # Decay learning rate
        alpha *= 0.95
        
        # Early termination
        if best_c5 <= 0.3807:
            break

    # Final refinement with local search
    def obj(h):
        return compute_c5(h, dx)
    
    # Simple local refinement with projected gradient descent steps
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient approximation
        eps = 1e-4
        grad = np.zeros_like(h)
        base_val = obj(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            grad[i] = (obj(h_plus) - base_val) / eps
        
        # Gradient descent step
        step = 0.01
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        new_val = obj(h_new)
        if new_val < best_c5:
            best_h = h_new.copy()
            best_c5 = new_val
        h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
