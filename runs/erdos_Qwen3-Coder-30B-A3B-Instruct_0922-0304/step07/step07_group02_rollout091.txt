import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    h = np.clip(h + diff / len(h), 0, 1)
    h = normalize_h(h, target_sum)
    return h

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize relaxed variables (can exceed bounds temporarily)
    h_relaxed = h.copy()
    
    # Optimizer parameters
    learning_rate = 0.1
    momentum = 0.9
    regularization = 1e-4
    v = np.zeros(n_points)  # velocity for momentum
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Iteration variables
    iterations = 0
    unchanged_iterations = 0
    max_unchanged = 200
    
    while time.time() - start_time < budget_s - 1 and iterations < 3000:
        # Evaluate current performance
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            unchanged_iterations = 0
        else:
            unchanged_iterations += 1
            
        # Early termination if target is reached
        if best_c5 <= 0.3807:
            break
            
        if unchanged_iterations > max_unchanged:
            # Reduce learning rate if no improvement
            learning_rate *= 0.9
            unchanged_iterations = 0
            
        # Compute gradient approximation
        # Simple finite difference for gradient of overlap
        grad = np.zeros(n_points)
        epsilon = 1e-5
        
        for i in range(n_points):
            # Perturb h_relaxed[i] and compute change in overlap
            h_perturbed = h_relaxed.copy()
            h_perturbed[i] += epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
            
            c5_perturbed = evaluate_c5(h_perturbed, dx)
            grad[i] = (c5_perturbed - current_c5) / epsilon
            
        # Update relaxed variables with momentum
        v = momentum * v - learning_rate * grad
        h_relaxed += v
        
        # Add regularization term to encourage staying near previous good solution
        regularization_term = regularization * (h_relaxed - h)
        h_relaxed += regularization_term
        
        # Project to feasible region [0,1]
        h_relaxed = np.clip(h_relaxed, 0, 1)
        
        # Enforce sum constraint
        h = project_to_constraint(h_relaxed, target_sum)
        
        iterations += 1
        
        # Adjust regularization parameter based on constraint violation
        violation = abs(np.sum(h) - target_sum)
        if violation > 1e-4:
            regularization *= 1.2
        else:
            regularization *= 0.8
            
        # Clamp regularization
        regularization = max(1e-6, min(1e-1, regularization))
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
