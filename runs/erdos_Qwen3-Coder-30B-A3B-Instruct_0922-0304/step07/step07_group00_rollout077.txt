import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the overlap objective."""
    # Compute full correlation of h with (1-h)
    corr_h_1h = correlate(h, 1 - h, mode="full")
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i in range(n_points):
        for k in range(2 * n_points - 1):
            # Convert full correlation index to valid h indices
            j = k - n_points + 1
            if 0 <= i + j < n_points:
                # Get the correlation value at this offset
                corr_val = corr_h_1h[k]
                # Compute gradient component for this position
                grad[i] += 2 * corr_val * (1 - 2 * h[i + j])
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2

    # Normalize to target sum

    # Gradient-based optimization
    lr = 0.01
    decay_rate = 0.999
    min_lr = 1e-6
    tolerance = 1e-8
    max_iter = 10000
    patience = 1000
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    prev_c5 = best_c5 + 1.0
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Update h
        h = h - lr * grad
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h = normalize_h(h, target_sum)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adapt learning rate
        if abs(current_c5 - prev_c5) < tolerance:
            lr = max(min_lr, lr * decay_rate)
        
        prev_c5 = current_c5
        iteration += 1
        
        # Patience check
        if no_improvement_count >= patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
