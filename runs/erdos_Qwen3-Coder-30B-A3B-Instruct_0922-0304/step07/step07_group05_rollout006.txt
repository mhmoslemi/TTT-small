import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # PSO Parameters
    population_size = 100
    max_iterations = 60
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocity_clamp = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create variations of the initial solution
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        for _ in range(population_size // 2 - 1):
            variation = h + np.random.normal(0, 0.02, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
            velocities.append(np.random.uniform(-velocity_clamp, velocity_clamp, n_points))
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            particles.append(h)
            velocities.append(np.random.uniform(-velocity_clamp, velocity_clamp, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO Loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (personal_best[i] - particles[i])
            social = social_coeff * r2 * (global_best - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Apply velocity clamp
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
        
        # Update inertia weight
        if iteration % 10 == 0 and iteration > 0:
            inertia_weight *= 0.98
            
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Local refinement using gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Try to refine globally best particle
    refined_h = global_best.copy()
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, refined_h, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(refined_h, dx)
    
    return (refined_h.tolist(), final_c5, n_points)
