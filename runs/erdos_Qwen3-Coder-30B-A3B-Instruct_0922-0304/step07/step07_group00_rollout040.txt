import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = h / np.sum(h) * target_sum
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap for gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient
        g = np.zeros(n_points)
        for i in range(n_points):
            # Compute the effect of changing h[i] on the maximum overlap
            g[i] = 0
            # For each shift k, we partially compute derivative if relevant
            for k in range(n_points):
                if i + k < n_points:
                    # Change in overlap due to h[i] for shift k
                    g[i] += (1 - 2 * h[i]) * (1 - h[i+k]) * dx
                    
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2

        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        # Update weights
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h and enforce constraints
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h))
        
        # Evaluate and track best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Stopping criterion
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
