import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:
        # Create initial values with some smoothness
        h = np.random.uniform(0.1, 0.9, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    population_size = 20
    mutation_rate = 0.1
    elite_size = population_size // 2
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
        else:
            ind = np.random.uniform(0.1, 0.9, n_points)
        ind = project_to_constraints(ind, target_sum)
        population.append(ind)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fit = compute_c5(ind, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Main Evolution Strategy loop
    max_iter = 20000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we've achieved target
        if fitness[0] <= 0.3807:
            break
            
        # Create offspring via mutation of elite
        offspring = []
        for i in range(elite_size):
            parent = population[i]
            # Adaptive mutation rate
            current_mutation_rate = mutation_rate * (1 - iter_num / max_iter)
            noise = np.random.normal(0, current_mutation_rate, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            offspring.append(child)
        
        # Add some diversity - random individuals
        while len(offspring) < population_size:
            if 'initial_h_values' in globals():
                ind = np.array(initial_h_values)
            else:
                ind = np.random.uniform(0.1, 0.9, n_points)
            ind = project_to_constraints(ind, target_sum)
            offspring.append(ind)
        
        # Evaluate offspring
        offspring_fitness = []
        for ind in offspring:
            fit = compute_c5(ind, dx)
            offspring_fitness.append(fit)
        
        # Update population
        population = offspring
        fitness = offspring_fitness
        
        # Update global best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_fitness:
            best_fitness = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
    
    # Final check
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
