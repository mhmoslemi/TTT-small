import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Lagrangian-based optimization approach
    max_iter = 1000
    alpha = 0.01
    gamma = 0.1
    tolerance = 1e-6
    min_improvement = 1e-8
    
    # Initialize multipliers
    mu = 1.0
    lambda_plus = np.zeros(n_points)
    lambda_minus = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current overlap
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Identify the maximizing shift
        overlap_full = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap_full)
        max_k = max_shift_idx - (n_points - 1)
        
        # Compute gradient for stationarity condition
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            # Compute contribution from the max shift k
            j = (i - max_k) % n_points
            grad_h[i] += (1 - h[j])
        
        # Add Lagrangian terms
        grad_h -= mu
        grad_h += lambda_plus - lambda_minus
        
        # Update h
        h_new = h - alpha * grad_h
        
        # Project back to feasible set [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint with Lagrange multiplier update
        current_sum_new = np.sum(h_new)
        mu = max(0, mu + gamma * (current_sum_new - target_sum))
        
        # Update multipliers for bounds
        lambda_plus = np.maximum(0, lambda_plus + gamma * (h_new - 1))
        lambda_minus = np.maximum(0, lambda_minus + gamma * (-h_new))
        
        # Check convergence
        if np.linalg.norm(h_new - h) < tolerance:
            break
            
        h = h_new
        
        # Adaptive learning rate
        if iteration > 0 and abs(current_c5 - evaluate_overlap(h, dx)) < min_improvement:
            alpha *= 0.95
        elif abs(current_c5 - evaluate_overlap(h, dx)) < 1e-6:
            alpha *= 0.98
    
    # Final gradient-based refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    from scipy.optimize import minimize
    
    if time.time() - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
