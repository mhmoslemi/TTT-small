import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def update_h(h, i, alpha, target_sum):
    """Update h by moving along coordinate i with step size alpha."""
    h_old = h[i]
    h[i] = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
    scaling = target_sum / np.sum(h)
    h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    # Initial evaluation
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    iterations = 0
    max_iter = 100000
    patience = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        iterations += 1
        if iterations % 100 == 0:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = iterations
            if iterations - last_improvement > 1000:
                break
            if best_c5 <= 0.3807:
                break
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Compute gradient components
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Approximate gradient: ∂/∂h[i] (max_k h[i] * (1-h[i+k]))
            # For each k, overlap[k] = sum_{j} h[j] * (1 - h[j+k])
            # ∂ overlap[k] / ∂ h[i] = 1 - 2 * h[i] if i >= k and i-k < n_points
            grad_term = 0.0
            for k in range(n_points):
                if 0 <= i - k < n_points:
                    if i >= k:
                        grad_term += (1 - 2 * h[i])
            grad[i] = grad_term
        
        # Rank by gradient magnitude and choose top 10%
        idxs = np.argsort(np.abs(grad))[::-1][:max(1, n_points // 10)]
        
        # Update selected coordinates
        for i in idxs:
            # Estimate step size
            # Since ∂overlap[i] / ∂h[i] = 1 - 2*h[i] when other h are fixed,
            # and we want to reduce the max overlap
            # α = -sign(gradient) * (current_max - current_max_candidate) 
            grad_i = grad[i]
            
            # Simple line search
            step_size = 0.01
            alpha = 0.0
            if np.abs(grad_i) > 1e-10:
                # We want to minimize overlap, so we move in direction that 
                # reduces gradient term.
                # Let's perform a simple backtracking line search
                alpha = -grad_i * step_size # Simple update without precise line search
                new_h = h.copy()
                new_h = update_h(new_h, i, alpha, target_sum)
                if np.sum(new_h) > 0:
                    new_c5 = compute_c5(new_h, dx)
                    if new_c5 < current_max:
                        h = new_h.copy()
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
