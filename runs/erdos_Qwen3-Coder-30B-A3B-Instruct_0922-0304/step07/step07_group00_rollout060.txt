import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem, Variable, sum as cp_sum, multiply, norm

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_basis(n_points, n_basis=16):
    """Create a set of basis functions for representing h(x)."""
    basis = []
    for i in range(n_basis):
        h = np.zeros(n_points)
        # Create a simple step function basis with random positions and widths
        start = np.random.randint(0, n_points//4)
        end = min(start + np.random.randint(1, n_points//4), n_points)
        h[start:end] = 1.0
        basis.append(h)
    return np.array(basis)

def evaluate_basis_overlaps(basis, n_points):
    """Precompute overlap integrals for all basis functions."""
    n_basis = len(basis)
    A = np.zeros((n_basis, n_basis, 2*n_points - 1))
    for i in range(n_basis):
        for j in range(n_basis):
            A[i, j, :] = correlate(basis[i], basis[j], mode="full")
    return A

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_basis = 16
    
    # Create initial basis
    basis = create_initial_basis(n_points, n_basis)
    A = evaluate_basis_overlaps(basis, n_points)
    
    # Initialize to a good starting solution if available
    if 'initial_h_values' in globals():

        # Decompose into basis representation

        constraints = [cp_sum(lambda_vars) == 1, lambda_vars >= 0]
        prob = Problem(objective, constraints)
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-6)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            initial_lambda = lambda_vars.value
        else:
            initial_lambda = np.ones(n_basis) / n_basis
    else:
        initial_lambda = np.ones(n_basis) / n_basis
    
    best_lambda = initial_lambda.copy()
    best_c5 = float('inf')
    best_h = None
    
    # Iteratively refine with convex optimization
    max_iter = 1000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve for h in terms of basis
        current_lambda = best_lambda.copy()  # current_lambda is fixed in primal
        current_h = basis.T @ current_lambda
        current_h = normalize_h(current_h, target_sum)
        current_c5 = compute_c5(current_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            if best_c5 <= 0.3807:
                break
        
        # Update basis (simple/cheap variant: switch to a new basis if needed)
        # For this simplified version, we'll just optimize the current lambda with CVXPY
        
        # Form and solve the dual problem approximately
        # Calculate the overlap terms for the current configuration
        r = np.zeros(2*n_points - 1)
        for k in range(2*n_points - 1):
            for i in range(n_basis):
                for j in range(n_basis):
                    r[k] += current_lambda[i] * current_lambda[j] * A[i, j, k]
        
        # Simple local Newton-like update - it's more direct
        u = np.ones(2*n_points - 1)
        u *= 0.1  # Small step size for stability
        
        # Do one iteration of projected gradient on lambda
        # This is a simplified approach against a linearization
        
        # Gradient approximation using forward differences
        eps = 1e-4
        grad = np.zeros(n_basis)

            lambda_plus[i] += eps
            if np.sum(lambda_plus) > 0:

            h_plus = basis.T @ lambda_plus
            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            lambda_minus = current_lambda.copy()

            h_minus = normalize_h(h_minus, target_sum)

    # Final cleanup
    final_h = basis.T @ best_lambda
    final_h = normalize_h(final_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
