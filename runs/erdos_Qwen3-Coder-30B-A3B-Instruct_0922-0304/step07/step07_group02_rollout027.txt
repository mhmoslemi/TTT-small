import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        # Add small randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        
    # Normalize to exact sum
    h = h * (target_sum / np.sum(h))
    
    # Try convex optimization approach
    try:
        # Define the optimization problem
        h_var = cp.Variable(n_points)
        # Objective: minimize max_k (h * shift_k(1-h))
        # We can reformulate by introducing auxiliary variables
        # but since it's not convex, we'll use iterative approach
        
        # For better results, let's also refine with local optimization
        best_h = h.copy()
        best_c5 = compute_c5(h, dx)
        
        # Gradient-based minization approach with constraints
        # Use scipy.optimize.minimize with bounds
        
        from scipy.optimize import minimize
        
        def objective(h_vec):
            # Normalize to sum constraint
            h_norm = h_vec * (target_sum / np.sum(h_vec))
            return compute_c5(h_norm, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        # Define bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Add sum constraint as equality constraint
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Try several random restarts
        for i in range(20):
            if time.time() - start_time > budget_s - 2:
                break
            
            # Random initialization within bounds
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init * (target_sum / np.sum(h_init))
            
            result = minimize(
                objective, 
                h_init, 
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            
            if result.success:
                h_opt = result.x * (target_sum / np.sum(result.x))
                c5_val = compute_c5(h_opt, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
            
            if best_c5 <= 0.3807:
                break
                
        h = best_h
        c5_bound = best_c5
        
    except Exception as e:
        # Fallback to simple gradient descent if optimization fails
        # Schedule for local refinement: gradient descent with constraint projection
        h = h.copy()
        lr = 0.01
        decay = 0.995
        
        for i in range(10000):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute gradient numerically (simple finite difference for now)
            c5_current = compute_c5(h, dx)
            
            # Gradient ascent step (we're minimizing C5)
            grad = np.zeros_like(h)

            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps

            # Decay learning rate
            lr *= decay

                if c5_current <= 0.3807:
                    break
            
        c5_bound = compute_c5(h, dx)
    
    return (h.tolist(), c5_bound, n_points)
