import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm
    population_size = 100
    swarm = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Seed with variations of initial solution
        swarm.append(h.copy())
        for _ in range(int(0.4 * population_size) - 1):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            swarm.append(variation)
        # Fill with random solutions
        for _ in range(population_size - len(swarm)):
            h_rand = np.random.rand(n_points)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = normalize_h(h_rand, target_sum)
            swarm.append(h_rand)
    else:
        # Create a diverse initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            swarm.append(h)
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = 1.5
    c2 = 1.5
    
    # Particle velocities
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # Personal best positions and fitnesses
    pbest_positions = [swarm[i].copy() for i in range(population_size)]
    pbest_fitnesses = [compute_c5(swarm[i], dx) for i in range(population_size)]
    
    # Global best
    best_idx = np.argmin(pbest_fitnesses)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_fitness = pbest_fitnesses[best_idx]
    
    # PSO iterations
    max_iterations = 60
    stagnation_count = 0
    stagnation_limit = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - swarm[i]) + 
                           c2 * r2 * (gbest_position - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Apply constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = normalize_h(swarm[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(swarm[i], dx)
            
            # Update personal best
            if fitness < pbest_fitnesses[i]:
                pbest_fitnesses[i] = fitness
                pbest_positions[i] = swarm[i].copy()
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_position = swarm[i].copy()
                    stagnation_count = 0
            
            # Early stopping
            if gbest_fitness <= 0.3807:
                break
        else:
            # Check for stagnation
            stagnation_count += 1
            if stagnation_count >= stagnation_limit:
                # Increase mutation to escape local optima
                for i in range(population_size):
                    swarm[i] += np.random.normal(0, 0.05, n_points)
                    swarm[i] = np.clip(swarm[i], 0, 1)
                    swarm[i] = normalize_h(swarm[i], target_sum)
                stagnation_count = 0
            continue
        break  # Early exit from main loop
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Try several local minimizations with different starts
    best_refined_h = gbest_position.copy()
    best_refined_c5 = gbest_fitness
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start from different particles
        start_pos = swarm[np.random.randint(0, population_size)].copy()
        
        result = minimize(objective, start_pos, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 30})
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            
            if refined_c5 < best_refined_c5:
                best_refined_c5 = refined_c5
                best_refined_h = refined_h.copy()
    
    # Double-check final solution
    final_c5 = compute_c5(best_refined_h, dx)
    
    return (best_refined_h.tolist(), final_c5, n_points)
