import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def create_basis_functions(n_points, n_basis=24):
    """Create a set of basis functions for h(x) representation."""
    basis = []
    # Create step functions with varying positions and widths
    for i in range(n_basis):
        # Create a basis function with a pulse
        h = np.zeros(n_points)
        width = max(2, n_points // 10)
        start = i * (n_points // n_basis)
        end = min(start + width, n_points)
        if end > start:
            h[start:end] = 1.0
        basis.append(h)
    # Normalize so each integrates to 1/(n_points/2)
    for i in range(len(basis)):
        basis[i] = basis[i] * (n_points / 2) / np.sum(basis[i])
    return np.array(basis)

def build_overlap_matrix(basis):
    """Precompute overlap integrals A_{ij}^k."""
    n_basis = basis.shape[0]
    n_points = basis.shape[1]
    # For each pair of basis functions, and each shift k
    overlap_matrix = {}
    for i in range(n_basis):
        for j in range(n_basis):
            overlap_matrix[(i,j)] = []
            for k in range(2 * n_points):  # All possible shifts
                # Convolution of basis[i] and 1-basis[j](x+k)
                shifted_j = np.concatenate([basis[j, n_points-k:], basis[j, :n_points-k]])
                integral = np.dot(basis[i], 1 - shifted_j)
                overlap_matrix[(i,j)].append(integral)
    return overlap_matrix

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_basis = 24
    
    # Create basis functions
    basis = create_basis_functions(n_points, n_basis)
    overlap_matrix = build_overlap_matrix(basis)
    
    # Initialize with initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Decompose into basis coefficients using least squares
        A = basis.T
        b = initial_h
        # Use pseudo-inverse to get coefficients
        coeffs = np.linalg.lstsq(A, b, rcond=None)[0]
        # Ensure non-negativity and normalization
        coeffs = np.maximum(coeffs, 0)
        coeffs = coeffs / np.sum(coeffs) * (n_points / 2) / np.sum(basis @ coeffs)
    else:
        # Random starting solution
        coeffs = np.random.rand(n_basis)
        coeffs = coeffs / np.sum(coeffs)
        
    # Iterate to improve solution
    best_c5 = float('inf')
    best_coeffs = coeffs.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Create h from current coefficients
        h = basis.T @ coeffs
        # Clamp h to [0,1] to ensure feasibility
        h = np.clip(h, 0, 1)
        # Re-normalize to target integral
        h = h * (target_sum / np.sum(h))
        
        # Evaluate C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_coeffs = coeffs.copy()
            
        # Early exits
        if best_c5 <= 0.3807:
            break
            
        # Convex optimization for next coefficients
        # Objective is to minimize max_k \sum_{i,j} coeffs_i * coeffs_j * (overlap_matrix[(i,j)][k] - overlap_matrix[(i,j)][k] * overlap_matrix[(j,j)][k])
        # We solve a simplified version using quadratic approximation
        # This serves as heuristic to guide improvement
        
        # Compute the full overlap matrix for current h
        expand_h = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.array([np.concatenate([expand_h[i:], np.zeros(i)]) for i in range(2 * n_points)])
        h_convol = np.array([np.dot(h, 1 - h_shifted[i, :n_points]) for i in range(2 * n_points)])
        
        # Separate cases for different shifts to use in gradient descent
        # Similar to dual decomposition but using simple gradient descent
        
        # Approximate gradient using finite differences
        grad_coeffs = np.zeros(n_basis)
        eps = 1e-4
        for i in range(n_basis):
            # Perturb coeffs[i]
            coeffs_plus = coeffs.copy()
            coeffs_minus = coeffs.copy()
            coeffs_plus[i] += eps
            coeffs_minus[i] -= eps
            
            # Normalize
            coeffs_plus = coeffs_plus / np.sum(coeffs_plus) * np.sum(coeffs) / np.sum(coeffs_plus)
            coeffs_minus = coeffs_minus / np.sum(coeffs_minus) * np.sum(coeffs) / np.sum(coeffs_minus)
            
            h_plus = basis.T @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            h_minus = basis.T @ coeffs_minus
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            grad_coeffs[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update using gradient (simplest descent)
        learning_rate = 0.01
        coeffs = coeffs - learning_rate * grad_coeffs
        coeffs = np.maximum(coeffs, 0)
        
        # Normalize
        coeffs = coeffs / np.sum(coeffs) * (n_points / 2) / np.sum(basis @ coeffs)
        
        # Optional: Add regularization
        coeffs = coeffs + 1e-6  # Add small constant to avoid degeneracy
        coeffs = coeffs / np.sum(coeffs)
    
    # Final refinement
    h_final = basis.T @ best_coeffs
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    final_c5 = compute_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
