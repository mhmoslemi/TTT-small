import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Create initial population
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.01, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    # PSO parameters
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    inertia_decay = 0.98
    
    # Initialize particles
    positions = np.array(population)
    velocities = np.random.uniform(-0.1, 0.1, (population_size, n_points))
    personal_best = positions.copy()
    personal_best_fitness = np.array([compute_c5(p, dx) for p in positions])
    
    # Global best
    best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[best_idx].copy()
    global_best_fitness = personal_best_fitness[best_idx]
    
    # PSO iterations
    max_iterations = 60
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(population_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - positions[i]) + 
                           c2 * r2 * (global_best - positions[i]))
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = normalize_h(positions[i], target_sum)
            
            # Evaluate and update personal best
            current_fitness = compute_c5(positions[i], dx)
            if current_fitness < personal_best_fitness[i]:
                personal_best[i] = positions[i].copy()
                personal_best_fitness[i] = current_fitness
                
                # Update global best if needed
                if current_fitness < global_best_fitness:
                    global_best = positions[i].copy()
                    global_best_fitness = current_fitness
        
        # Update inertia weight
        w *= inertia_decay
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Final refinement with gradient descent
    best_h = global_best.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Multiple rounds of SLSQP optimization
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
