import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max):
    """Compute gradient of overlap at shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n - k_max):
        grad[i] += 1 - 2 * h[i + k_max]
    return grad

def solve_linear_program(grad, n_points):
    """Solve the linear program to find the descent direction."""
    # Simple heuristic: find the index with max negative gradient
    # This represents the steepest descent direction respecting constraints
    s = np.zeros(n_points)
    max_neg_grad_idx = np.argmin(grad)
    if max_neg_grad_idx >= 0 and max_neg_grad_idx < n_points:
        s[max_neg_grad_idx] = 1.0
    return s

def backtracking_line_search(h, grad, s, dx, max_iter=10):
    """Perform backtracking line search."""
    alpha = 0.5
    c5_current = compute_c5(h, dx)
    step_size = alpha
    for _ in range(max_iter):
        h_new = h + step_size * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_current:
            return step_size
        step_size *= 0.5
    return step_size

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stagnation_count = 0
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and stagnation_count < 500:
        # Compute current overlaps to find max
        overlap = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(overlap)
        
        # Create gradient for this shift
        grad = compute_gradient(h, max_k_idx)
        
        # Find descent direction
        s = solve_linear_program(grad, n_points)
        s = np.clip(s, 0, 1)
        
        # Line search
        alpha = backtracking_line_search(h, grad, s, dx)
        
        # Update
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        h = h_new
        iteration += 1
        
        # Periodic validation
        if iteration % 100 == 0:
            current_c5 = compute_c5(h, dx)
            if abs(current_c5 - best_c5) > 1e-5:
                best_h = h.copy()
                best_c5 = current_c5
        
    return (best_h.tolist(), best_c5, n_points)
