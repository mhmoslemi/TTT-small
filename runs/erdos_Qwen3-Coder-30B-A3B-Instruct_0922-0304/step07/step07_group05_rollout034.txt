import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # PSO parameters
    population_size = 100
    max_iterations = 60
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create diverse initial population
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
        for _ in range(population_size - 1):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Create initial population from random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO main loop
    best_global_pos = None
    best_global_fitness = float('inf')
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < best_global_fitness:
                best_global_fitness = fitness
                best_global_pos = particles[i].copy()
            
            # Update velocities and positions
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            cognitive_coeff * r1 * (particles[i] - particles[i]) + 
                            social_coeff * r2 * (best_global_pos - particles[i]))
            
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Check early termination
            if best_global_fitness <= 0.3807:
                break
                
        # Update inertia weight
        inertia_weight *= 0.98
    
    # Final refinement with gradient descent
    best_h = best_global_pos.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Try local optimization with multiple restarts
    best_refined = best_h.copy()
    best_refined_c5 = compute_c5(best_h, dx)
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_refined_c5:
                best_refined = refined_h
                best_refined_c5 = refined_c5
    
    # Final check to make sure we have a valid solution
    final_c5 = compute_c5(best_refined, dx)
    
    return (best_refined.tolist(), final_c5, n_points)
