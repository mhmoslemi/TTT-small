import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    pop_size = 100
    tournament_size = max(5, pop_size // 20)
    max_generations = 1000
    mutation_rate = 0.1
    crossover_blend_min = 0.2
    crossover_blend_max = 0.8
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        individual = np.random.uniform(0, 1, n_points)
        individual = np.clip(individual, 0, 1)
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (target_sum / current_sum)
        population.append(individual)
    
    # Evaluate initial population
    fitness = [evaluate_overlap(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # GA loop
    stagnation_count = 0
    prev_best = best_c5
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = [best_h.copy()]
        for i in range(0, pop_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1]
            
            # Blend crossover
            alpha = np.random.uniform(crossover_blend_min, crossover_blend_max)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                mut_mask = np.random.random(n_points) < 0.1
                delta = np.random.normal(0, 0.05, n_points)
                child1 += delta * mut_mask
                child1 = np.clip(child1, 0, 1)
                current_sum = np.sum(child1)
                if current_sum > 0:
                    child1 = child1 * (target_sum / current_sum)
            
            if np.random.random() < mutation_rate:
                mut_mask = np.random.random(n_points) < 0.1
                delta = np.random.normal(0, 0.05, n_points)
                child2 += delta * mut_mask
                child2 = np.clip(child2, 0, 1)
                current_sum = np.sum(child2)
                if current_sum > 0:
                    child2 = child2 * (target_sum / current_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        if len(new_population) != pop_size:
            new_population = new_population[:pop_size]
        
        # Evaluate new population
        fitness_new = [evaluate_overlap(ind, dx) for ind in new_population]
        
        # Elitism
        population = new_population
        fitness = fitness_new
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive mutation
        if stagnation_count > 20:
            mutation_rate = min(0.5, mutation_rate * 1.2)
            stagnation_count = 0
            
        if abs(prev_best - best_c5) < 1e-6:
            stagnation_count += 1
        else:
            prev_best = best_c5
            
        if stagnation_count > 50:
            break
    
    # Local refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
