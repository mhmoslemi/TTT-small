import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust values to satisfy sum constraint
        diff = target_sum - current_sum
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        # Small correction to ensure sum exact
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    # Use finite differences to approximate gradient of max_overlap
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    base_c5 = compute_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, len(h)/2)
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def trust_region_step(h, grad, trust_radius, dx):
    # Simple gradient descent step with trust region
    step = -grad
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    new_h = h + step
    return project_to_constraints(new_h, len(h)/2)

        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint exactly
    h = project_to_constraints(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Dynamic bound search
    max_iter = 1000
    alpha = 0.1
    c_low = 0.0
    c_high = best_c5
    trust_radius = 0.05
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute mid point
        c_mid = (c_low + c_high) / 2
        # This is a simplification - we'll directly optimize
        # Instead, we'll do gradient-based optimization
        
        # Perform trust-region descent
        max_steps = 20
        for step in range(max_steps):
            if time.time() - start_time >= budget_s - 1:
                break
            grad = compute_gradient(h, dx)
            h_new = trust_region_step(h, grad, trust_radius, dx)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()

            if c5_new < compute_c5(h, dx):
