import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max):
    """Compute the gradient of overlap_k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For overlap_k_max = sum_{i=0}^{n-k_max-1} h[i] * (1 - h[i+k_max])
    # d/dh[i] overlap_k_max = 1 - 2*h[i+k_max] if i+k_max < n and i < n
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1 - 2 * h[i + k_max]
    return grad

def solve_linear_program(grad, n):
    """Solve the linear program to find the minimizing direction."""
    # We want to minimize grad^T * s subject to sum(s) = 0 and 0 <= s[i] <= 1
    # This is a simple linear programming problem
    s = np.zeros(n)
    
    # Simple heuristic approach: set s so that the gradient is minimized
    # Idea: set components where gradient is positive to 0, negative to 1, 
    # and adjust to meet sum=0 constraint
    indices = np.argsort(grad)
    s[indices[:n//2]] = 1
    s[indices[n//2:]] = 0
    s = s - np.mean(s)  # make sum zero
    
    # Project to feasible region [0,1]^n
    s = np.clip(s, 0, 1)
    
    # Ensure sum is 0
    s = s - np.mean(s)
    
    # Renormalize so that sum is effectively 0, but perhaps not exactly 0
    # This is approximation for the LP, good enough for our purposes
    return s

def backtracking_line_search(h, s, k_max, dx, max_alpha=1.0):
    """Perform backtracking line search for optimal step size."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    grad = compute_gradient(h, k_max)
    while alpha > 1e-6:
        h_new = h + alpha * (s - h)
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Constraint check: sum h_new should be approximately target
        target_sum = len(h) / 2
        h_new = normalize_h(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)
        # Armijo condition: 
        # new_c5 <= current_c5 + 0.1 * alpha * grad^T * (s - h)
        # Simplified as: new_c5 <= current_c5 + 0.1 * alpha * grad^T * (s - h)
        if new_c5 <= current_c5:
            return alpha
        alpha *= 0.5
    return 0.1  # fallback

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Frank-Wolfe iterations
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Early stopping condition
        if max_overlap < 0.3807:
            break
            
        # Compute gradient at k_max
        grad = compute_gradient(h, max_k)
        
        # Solve LP to find direction
        s = solve_linear_program(grad, n_points)
        
        # Line search
        alpha = backtracking_line_search(h, s, max_k, dx)
        
        # Update
        h_new = h + alpha * (s - h)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Check if we improved
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement = iteration
            
        h = h_new.copy()
        
        # Check for early stopping
        if iteration - last_improvement > 500:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
