import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Adaptive quadratic optimization approach
    max_iterations = 60
    patience = 10
    patience_counter = 0
    last_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap values for all lags
        h_ref = best_h
        overlap_full = correlate(h_ref, 1 - h_ref, mode="full")
        max_lag_idx = np.argmax(overlap_full)
        max_overlap = overlap_full[max_lag_idx] * dx
        
        # Generate quadratic approximation for descent direction
        # Approximate gradient and Hessian based on finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        h_plus = h_ref.copy()
        h_minus = h_ref.copy()
        
        for i in range(n_points):
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_lag_idx] * dx
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_lag_idx] * dx
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
            h_plus[i] -= eps
            h_minus[i] += eps
        
        # Formulate and solve quadratic program for descent direction
        h_var = Variable(n_points)
        
        # Define the quadratic approximation around current point
        Q = np.eye(n_points) * 1e-6  # Small regularization
        for i in range(n_points):
            Q[i, i] = 1e-3
        
        P = Q  # Approximate Hessian (identity with small regularization)
        q = grad
        
        # Constraints for QP
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Objective: minimize quadratic approximation
        objective = cp.Minimize(0.5 * cp.quad_form(h_var - h_ref, P) + q.T @ (h_var - h_ref))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                descent_direction = h_var.value - h_ref
            else:
                descent_direction = -grad
                descent_direction = descent_direction / (np.linalg.norm(descent_direction) + 1e-12)
        except:
            descent_direction = -grad
            descent_direction = descent_direction / (np.linalg.norm(descent_direction) + 1e-12)
        
        # Line search for step size
        step_size = 1.0
        max_step = 0.1
        alpha = 0.5
        max_backtrack = 20
        
        for _ in range(max_backtrack):
            h_new = h_ref + step_size * descent_direction
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                patience_counter = 0
                break
            else:
                step_size *= alpha
                if step_size < 1e-8:
                    break
        
        if best_c5 < last_c5:
            last_c5 = best_c5
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
