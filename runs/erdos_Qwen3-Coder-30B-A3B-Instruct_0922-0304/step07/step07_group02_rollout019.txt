import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    
    # Coordinate-wise optimization with adaptive regularization
    h_relaxed = h.copy()
    lambda_reg = 1e-4
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros_like(h_relaxed)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradients via finite differences
        grad = np.zeros_like(h_relaxed)
        eps = 1e-5
        current_c5 = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_to_constraint(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h_relaxed.copy()
            h_minus[i] -= eps

            c5_minus = compute_c5(h_minus, dx)

        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_relaxed = h_relaxed + velocity
        
        # Apply regularization
        h_relaxed = h_relaxed - lambda_reg * (h_relaxed - h)
        
        # Project to [0,1] and enforce sum constraint
        h = np.clip(h_relaxed, 0, 1)
        h = project_to_constraint(h, target_sum)

        new_c5 = compute_c5(h, dx)
        
        if new_c5 < best_c5:
