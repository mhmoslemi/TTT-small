import numpy as np
import time

def compute_c5(h, dx):
    # Compute correlation using numpy's correlate function
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust sum to target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Use proportional adjustment to maintain constraint
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

    # Initial evaluation
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Simulated Annealing parameters
    temp_init = 0.1 * current_c5
    temp = temp_init
    cooling_rate = 0.9995

    step_size = 0.05
    
    # Annealing loop

        # Accept or reject
        if new_c5 < current_c5:

            if new_c5 < best_c5:
                best_h = h_new.copy()

            delta = new_c5 - current_c5
