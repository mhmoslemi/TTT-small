import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing with local refinement
    current_h = h.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temperature = 1e-6
    cooling_rate = 0.9995
    iterations_per_temp = 50
    
    # Main optimization loop
    for iteration in range(int(budget_s * 100)):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a neighbor solution
        neighbor_h = current_h.copy()
        
        # Perturb a random point
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        
        # Apply bounds
        neighbor_h = np.clip(neighbor_h, 0, 1)
        
        # Normalize
        neighbor_sum = np.sum(neighbor_h)
        if neighbor_sum > 0:
            neighbor_h = neighbor_h * (target_sum / neighbor_sum)
        
        # Evaluate the neighbor
        neighbor_c5 = evaluate_overlap(neighbor_h, dx)
        
        # Accept or reject based on simulated annealing
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(neighbor_c5 - current_c5) / temperature):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Decrease temperature
        temperature = max(min_temperature, temperature * cooling_rate)
        
        # Occasionally do local refinement using scipy optimizer
        if iteration % 100 == 0 and iteration > 0:
            try:
                from scipy.optimize import minimize
                
                def objective(h_vector):
                    return evaluate_overlap(h_vector, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
                
                result = minimize(objective, current_h, method='SLSQP', 
                                 bounds=bounds, constraints=constraints, 
                                 options={'maxiter': 50, 'ftol': 1e-6})
                
                if result.success:
                    refined_h = np.clip(result.x, 0, 1)
                    refined_sum = np.sum(refined_h)
                    if refined_sum > 0:
                        refined_h = refined_h * (target_sum / refined_sum)
                    refined_c5 = evaluate_overlap(refined_h, dx)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
                        current_h = refined_h
                        current_c5 = refined_c5
            except:
                pass
                
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
