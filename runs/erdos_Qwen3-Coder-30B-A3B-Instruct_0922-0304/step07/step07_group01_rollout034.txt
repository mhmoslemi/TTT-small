import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        # Distribute the difference
        if diff > 0:
            h += diff / len(h)
        else:
            h += diff / len(h)
        # Re-project to ensure bounds
        h = np.clip(h, 0, 1)
        # Apply small adjustment to maintain sum constraint exactly
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from initial_h_values if available

        n_points = len(h)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1) * 0.5  # Make it more moderate
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)

    # Convex relaxation approach
    h_relaxed = cp.Variable(n_points, name="h")
    C = cp.Variable(name="C")

    # Constraints
    constraints = [
        h_relaxed >= 0,
        h_relaxed <= 1,
        cp.sum(h_relaxed) == target_sum
    ]
    
    # Overlap constraints with upper bound C
    for i in range(2 * n_points - 1):
        if i < n_points:  # First part
            lagged = np.concatenate([h_relaxed[i:], np.zeros(i)])
        else:  # Second part
            lagged = np.concatenate([np.zeros(i - n_points + 1), h_relaxed[:n_points - (i - n_points + 1)]])
        # If there are different formulations that yield the same C, use a consistent constraint logic
        constraint_expr = cp.sum(cp.multiply(h_relaxed, 1 - lagged)) <= C * dx
        constraints.append(constraint_expr)

    # Objective
    objective = cp.Minimize(C)
    problem = cp.Problem(objective, constraints)
    
    # Solve the convex relaxation
    try:
        problem.solve(solver=cp.CBC)
        h_relaxed_val = h_relaxed.value
        h_opt = np.clip(h_relaxed_val, 0, 1)
        # Ensure the sum is correct
        h_opt = normalize_h(h_opt, target_sum)
    except Exception:
        h_opt = h.copy()

    # Gradient descent refinement with projected updates
    learning_rate = 0.01
    max_iter = 5000
    step_size = 0.001
    current_h = h_opt.copy()
    
    best_c5 = compute_c5(current_h, dx)
    best_h = current_h.copy()
    
    # Evaluate different gradient types: gradient wrt. C5
    def gradient_for_c5(h_vals, dx):
        # Efficient direct gradient computation
        n = len(h_vals)
        grad = np.zeros(n)
        c5_val = compute_c5(h_vals, dx)
        # We want to differentiate max_k integral h * (1 - h_{k})
        # For this, we compute each contribution to loss separately
        
        # Compute the correlation
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Find max_index where this occurs
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Grad wrt the overlap contribution
        # Could approximate or use finite difference:
        # For better precision, resample around optimal solution using NumPy's signal package
        # But simpler approach:
        h_copy = h_vals.copy()
        for i in range(n):
            h_copy[i] += step_size
            # Ensure h stays in bounds
            h_copy = np.clip(h_copy, 0, 1)
            h_copy = normalize_h(h_copy, target_sum)
            new_c5 = compute_c5(h_copy, dx)
            
            grad[i] = (new_c5 - c5_val) / step_size
            h_copy[i] = h_vals[i]  # Reset
            
        return grad
    
    # Now perform gradient descent on the whole vector
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        try:
            grad = gradient_for_c5(current_h, dx)
            # Update
            current_h -= learning_rate * grad
            # Project to feasible set
            current_h = project_to_constraints(current_h, target_sum)
            
            # Evaluate new C5
            c5_val = compute_c5(current_h, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = current_h.copy()
                
            # Early stopping
            if best_c5 <= 0.3807:
                break
        except Exception:
            pass  # If optimization fails, continue with backup

    return (best_h.tolist(), best_c5, n_points)
