import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Local refinement using simulated annealing-like approach
    current_h = h.copy()
    current_c5 = best_c5
    
    # Temperature schedule
    T_start = 0.1
    T_end = 0.001
    max_iter = int(1e5)
    decay_rate = (T_end / T_start) ** (1.0 / max_iter)
    
    # Perturbation parameters
    step_size = 0.05
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Simulate a random neighbor
        new_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, step_size)
        new_h[idx] += delta
        
        # Project back to feasible region
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        
        # Evaluate
        new_c5 = evaluate_overlap(new_h, dx)
        
        # Metropolis acceptance criterion
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T_start):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down
        T_start *= decay_rate
        
        # Occasionally re-sample if stuck
        if i % 1000 == 0 and i > 0 and np.isclose(current_c5, best_c5, rtol=1e-6):
            current_h = np.random.rand(n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h_sum = np.sum(current_h)
            if current_h_sum > 0:
                current_h = current_h * (target_sum / current_h_sum)
            current_c5 = evaluate_overlap(current_h, dx)
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
