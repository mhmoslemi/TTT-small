import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize if needed
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Make sure it's feasible
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try to improve with local search
    def local_search(h, max_iter=1000, step_size=0.01):
        current_h = h.copy()
        current_c5 = evaluate_overlap(current_h, dx)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Generate a random perturbation
            pert = np.random.normal(0, step_size, n_points)
            new_h = current_h + pert
            
            # Clip and re-normalize
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h)
            if current_sum > 0:
                new_h = new_h * (target_sum / current_sum)
            
            new_c5 = evaluate_overlap(new_h, dx)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    
        return best_h, best_c5
    
    # Start local search from the initial point or the one refined from SDP
    h_refined, c5_refined = local_search(best_h, max_iter=500, step_size=0.01)
    
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined
    
    # Try another refinement with smaller step size if time allows
    if time.time() - start_time < budget_s - 5:
        h_refined2, c5_refined2 = local_search(best_h, max_iter=500, step_size=0.005)
        if c5_refined2 < best_c5:
            best_h = h_refined2
            best_c5 = c5_refined2
    
    # Final check and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
