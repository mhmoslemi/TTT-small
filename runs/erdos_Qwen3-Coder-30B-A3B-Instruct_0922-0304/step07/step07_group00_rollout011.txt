import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def create_basiss(n_points, n_basiss=20):
    """Create a set of basis functions with varying pulse widths and positions."""
    basis = []
    dx = 2.0 / n_points
    for i in range(n_basiss):
        # Create a piecewise constant basis function
        h = np.zeros(n_points)
        # Randomly position and size a pulse
        start = np.random.randint(0, n_points // 2)
        end = min(start + np.random.randint(2, n_points // 4), n_points)
        h[start:end] = 1.0
        # Ensure normalization
        h = h * (n_points / 2) / np.sum(h)
        basis.append(h)
    return np.array(basis)

def project_to_feasible(h, target_sum):
    """Project h to a feasible solution."""
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h
