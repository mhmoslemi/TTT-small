import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function with some structure
        h = np.zeros(n_points)
        # Place some initial mass in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Local refinement using simulated annealing approach
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Annealing schedule
    max_iter = 10000
    temp_start = 0.1
    temp_end = 0.001
    decay = (temp_end / temp_start) ** (1.0 / max_iter)
    
    temp = temp_start
    current_h = h.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create candidate solution
        candidate_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        # Perturb one element
        delta = np.random.normal(0, temp * 0.1)
        candidate_h[idx] += delta
        candidate_h[idx] = np.clip(candidate_h[idx], 0, 1)
        
        # Re-normalize to sum constraint
        current_sum = np.sum(candidate_h)
        if current_sum > 0:
            candidate_h = candidate_h * (target_sum / current_sum)
        
        # Evaluate
        candidate_c5 = objective(candidate_h)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.random() < np.exp((current_c5 - candidate_c5) / temp):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Decay temperature
        temp *= decay
    
    # Final refinement with local optimization
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective_to_minimize(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        # Direct constraint using sum
        def constraint_fun(h_vector):
            return np.sum(h_vector) - target_sum
            
        constraints = {'type': 'eq', 'fun': constraint_fun}
        
        # Use the best solution as starting point
        result = minimize(objective_to_minimize, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 300, 'ftol': 1e-8}, 
                         callback=lambda x: None if time.time() - start_time < budget_s - 1 else None)
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
