import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    population_size = 20
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    elite_size = population_size // 2
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Include the starting solution and add variation
    population.append(h.copy())
    population_fitness.append(compute_c5(h, dx))
    
    for i in range(population_size - 1):
        # Add small noise to initial solution
        noise = np.random.normal(0, 0.02, n_points)
        candidate = h + noise
        candidate = project_to_constraints(candidate, target_sum)
        population.append(candidate)
        population_fitness.append(compute_c5(candidate, dx))
    
    # Sort population by fitness
    sorted_indices = np.argsort(population_fitness)
    population = [population[i] for i in sorted_indices]
    population_fitness = [population_fitness[i] for i in sorted_indices]
    
    best_solution = population[0].copy()
    best_fitness = population_fitness[0]
    
    # Evolution iteration
    max_iter = 30000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Create offspring using elitism and mutation
        offspring = []
        
        # Keep elite
        for i in range(elite_size):
            offspring.append(population[i].copy())
        
        # Generate new individuals
        for i in range(population_size - elite_size):
            # Select two parents from top half
            parent1 = population[np.random.randint(0, elite_size)]
            parent2 = population[np.random.randint(0, elite_size)]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            mutation = np.random.normal(0, mutation_rate, n_points)
            child += mutation
            
            # Project to constraints
            child = project_to_constraints(child, target_sum)
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = []
        for candidate in offspring:
            fitness = compute_c5(candidate, dx)
            offspring_fitness.append(fitness)
        
        # Combine and select new population
        population.extend(offspring)
        population_fitness.extend(offspring_fitness)
        
        # Sort and select top individuals
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices[:population_size]]
        population_fitness = [population_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update global best
        if population_fitness[0] < best_fitness:
            best_solution = population[0].copy()
            best_fitness = population_fitness[0]
            
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_solution, dx)
    
    return (best_solution.tolist(), final_c5, n_points)
