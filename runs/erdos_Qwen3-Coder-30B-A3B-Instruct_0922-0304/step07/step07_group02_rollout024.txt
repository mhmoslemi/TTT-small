import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    # Clip to [0,1] first
    h = np.clip(h, 0, 1)
    # Normalize sum
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # GMM parameters
    M = 5  # Number of components
    N = 100  # Population size
    P = 20   # Number of best to keep
    alpha = 0.99  # Covariance decay factor
    
    # Initialize GMM
    if 'initial_h_values' in globals():
        # Start with the provided solution
        means = [np.array(initial_h_values)]
        means[0] = np.clip(means[0], 0, 1)
        means[0] = normalize_h(means[0], target_sum)
    else:
        # Start with random solution
        means = [np.random.rand(n_points)]
        means[0] = np.clip(means[0], 0, 1)
        means[0] = normalize_h(means[0], target_sum)
    
    # Add diversity
    for _ in range(M - 1):
        mean = np.random.rand(n_points)
        mean = np.clip(mean, 0, 1)
        mean = normalize_h(mean, target_sum)
        means.append(mean)
    
    covariances = [np.eye(n_points) * 0.1 for _ in range(M)]
    weights = np.ones(M) / M
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    max_iterations = int(budget_s * 10)  # Rough estimate for iterations
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Sample candidates from GMM
        candidates = []
        for _ in range(N):
            # Choose component
            comp_idx = np.random.choice(M, p=weights)
            # Sample from component
            sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            # Apply constraints
            sample = np.clip(sample, 0, 1)
            sample = normalize_h(sample, target_sum)
            candidates.append(sample)
        
        # Evaluate candidates
        candidate_scores = []
        for c in candidates:
            score = compute_c5(c, dx)
            candidate_scores.append(score)
        
        # Sort by score
        sorted_indices = np.argsort(candidate_scores)
        top_candidates = [candidates[i] for i in sorted_indices[:P]]
        top_scores = [candidate_scores[i] for i in sorted_indices[:P]]
        
        # Update best
        if top_scores[0] < best_c5:
            best_c5 = top_scores[0]
            best_h = top_candidates[0].copy()
        
        # Check termination condition
        if best_c5 <= 0.3807:
            break
            
        # Update GMM
        # Compute new means
        new_means = []
        for i in range(M):
            # Select a component
            if i < len(top_candidates):
                new_means.append(top_candidates[i])
            else:
                # Generate new mean by sampling from current GMM
                comp_idx = np.random.choice(M, p=weights)
                new_mean = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
                new_mean = np.clip(new_mean, 0, 1)
                new_mean = normalize_h(new_mean, target_sum)
                new_means.append(new_mean)
        
        # Compute new covariance matrices
        new_covariances = []
        for i in range(M):
            # Get data points for this component
            # For now, we'll simply use the top candidates (weighted) for each component
            # In practice, use specific assignments or expect maximization
            
            # Use weighted deviation from the mean
            if i < len(top_candidates):
                mean = new_means[i]
                # Gather points for this Gaussian
                points = [c for c in top_candidates]
                # Compute scatter matrix
                diff = np.array(points) - mean
                # Add regularization
                scatter = np.dot(diff.T, diff) / len(points) + np.eye(n_points) * 1e-8
                # Apply decay
                new_cov = scatter * alpha
            else:
                # Keep existing but decay
                new_cov = covariances[i] * alpha
                
            new_covariances.append(new_cov)
        
        # Update weights
        # Weighted by score (lower score = more weight)
        weights = np.array(top_scores[:M])
        weights = 1.0 / (weights + 1e-10)  # Avoid division by zero
        weights = weights / np.sum(weights)
        
        # Normalize the means and set updated values
        means = new_means
        covariances = new_covariances
        
        iteration += 1
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
    else:
        # Fall back to initial if nothing was found
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            final_c5 = compute_c5(h, dx)
            best_h = h
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            final_c5 = compute_c5(h, dx)
            best_h = h
    
    return (best_h.tolist(), final_c5, n_points)
