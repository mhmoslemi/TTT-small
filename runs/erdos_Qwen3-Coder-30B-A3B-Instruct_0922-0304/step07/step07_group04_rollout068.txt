import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simple gradient-based approach with adaptive learning rate
    max_iter = 1000
    alpha = 0.01
    tol = 1e-8
    prev_c5 = best_c5 + 1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradients using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        # Calculate current overlap
        current_overlap = evaluate_overlap(h, dx)
        
        # Estimate gradient by perturbing each component
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - eps)
            
            overlap_plus = evaluate_overlap(h_plus, dx)
            overlap_minus = evaluate_overlap(h_minus, dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Approximate gradient of the maximum overlap
        # We use the gradient direction from the maximizing shift
        # This is a simplified approach
        h_new = h - alpha * grad
        
        # Project to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        current_sum_new = np.sum(h_new)
        if current_sum_new > 0:
            h_new = h_new * (target_sum / current_sum_new)
        
        c5_new = evaluate_overlap(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Adaptive learning rate if no significant improvement
        if abs(c5_new - current_overlap) < tol:
            alpha *= 0.9
        elif c5_new > current_overlap:
            alpha *= 0.5
            
        h = h_new.copy()
        
        # Early stopping if we're not improving much
        if abs(prev_c5 - best_c5) < 1e-7:
            break
        prev_c5 = best_c5
        
    # Final refinement with constrained optimization
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try multiple random restarts to avoid local minima
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:

            if result.success:

                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
