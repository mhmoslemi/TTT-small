import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Read initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Initialize with a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        
    # Normalize to required sum
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex optimization part
    # We use cvxpy to solve a relaxed version
    h_cvx = Variable(n_points)
    constraints = [h_cvx >= 0, h_cvx <= 1, sum(h_cvx) == target_sum]
    C = Variable()
    # This is more complex to model due to the convolution, so we'll do a simpler approach
    
    # Use a more direct approach: start with a better-known construction
    # and perform local search for improvement
    
    # Use a simple local search approach with gradient estimation
    
    # Try to improve using a Nelder-Mead-like procedure with small steps
    steps = 1000
    step_size = 0.01
    max_evals = 5000
    
    eval_count = 0
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for step in range(steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        if eval_count >= max_evals:
            break
                
        # Try random perturbations
        test_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, step_size)
        test_h[idx] += delta
        test_h = np.clip(test_h, 0, 1)
        
        # Renormalize to sum constraint
        test_h *= target_sum / np.sum(test_h)
        
        test_c5 = compute_c5(test_h, dx)
        eval_count += 1
        
        if test_c5 < current_c5:
            current_h = test_h
            current_c5 = test_c5
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Adaptive step size
        if eval_count % 100 == 0:
            step_size *= 0.95
            
    # Final refinement using small steps around the best
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        test_h = best_h.copy()
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        if idx1 != idx2:
            diff = np.random.normal(0, 0.005)
            test_h[idx1] += diff
            test_h[idx2] -= diff
            test_h = np.clip(test_h, 0, 1)
            test_h *= target_sum / np.sum(test_h)
            
            test_c5 = compute_c5(test_h, dx)
            if test_c5 < best_c5:
                best_h = test_h
                best_c5 = test_c5
                
    return (best_h.tolist(), best_c5, n_points)
