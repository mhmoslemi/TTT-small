import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy the sum constraint and bounds"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initialize with a block-like function
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h to meet sum constraint
    h = project_to_constraints(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Stage 1: Convex relaxation approach
    if time.time() - start_time < budget_s * 0.25:
        # Create continuous relaxation variable
        h_cont = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_cont >= 0,
            h_cont <= 1,
            cp.sum(h_cont) == target_sum,
            C >= 0
        ]
        
        # Add constraints for all lags
        for k in range(2 * n_points - 1):
            # Get shifted indices
            if k < n_points:
                left = 0
                right = n_points - k
                if right > 0:
                    overlap = cp.sum(cp.multiply(h_cont[left:right], 
                                              1 - h_cont[k:k+right]))
                else:
                    overlap = 0
            else:
                left = k - n_points + 1
                right = n_points - left
                if right > 0:
                    overlap = cp.sum(cp.multiply(h_cont[left:left+right], 
                                              1 - h_cont[0:right]))
                else:
                    overlap = 0
            # Add constraint to ensure overlap is bounded by C
            constraints.append(overlap <= C)
        
        # Objective
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_cont_sol = h_cont.value
                # Convert to step function by thresholding
                h_candidates = []
                for threshold in np.linspace(0.1, 0.9, 9):
                    h_step = (h_cont_sol > threshold).astype(float)
                    h_step = project_to_constraints(h_step, target_sum)
                    h_candidates.append(h_step)
                
                # Select best among candidates
                best_candidate = min(h_candidates, key=lambda x: compute_c5(x, dx))
                h = best_candidate.copy()
                best_c5 = compute_c5(h, dx)
        except:
            pass
    
    # Stage 2: Gradient-based refinement
    if time.time() - start_time < budget_s * 0.8:
        # Gradient descent using projected updates
        max_iter = 1000
        lr = 0.01
        decay_rate = 0.99
        tol = 1e-6
        
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient analytically for overlap
            C_val = compute_c5(h, dx)
            if C_val <= 0.3807:
                break
                
            # Calculate gradient with respect to h (estimation)
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, target_sum)
                C_plus = compute_c5(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_constraints(h_minus, target_sum)
                C_minus = compute_c5(h_minus, dx)
                
                grad[i] = (C_plus - C_minus) / (2 * eps)
            
            # Update step
            h_new = h - lr * grad
            h_new = project_to_constraints(h_new, target_sum)
            
            # Check improvement
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            # Update h
            h = h_new
            
            # Decay learning rate
            lr *= decay_rate
            
            # Early termination
            if abs(C_val - c5_new) < tol:
                break
    
    # Final validation
    final_h = project_to_constraints(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
