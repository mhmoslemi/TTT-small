import numpy as np
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = project_to_constraints(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Mutation parameters
    epsilon_initial = 0.2
    epsilon_min = 0.001
    decay_factor = 0.99
    adaptation_interval = 100
    
    epsilon = epsilon_initial
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate random perturbation
        v = np.random.randn(n_points)
        v = v / np.linalg.norm(v)
        mutation = epsilon * v
        
        # Apply perturbation
        h_candidate = best_h + mutation
        
        # Project to feasible region
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        c5_candidate = compute_c5(h_candidate, dx)
        
        # Accept if better
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            epsilon *= decay_factor  # Reduce mutation strength
        else:
            # Sometimes accept worse to escape local optima
            if np.random.rand() < 0.01:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                epsilon *= 0.95  # Reduce less aggressively
            
        # Adapt epsilon
        if iteration % adaptation_interval == 0:
            epsilon = max(epsilon_min, epsilon * decay_factor)
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
