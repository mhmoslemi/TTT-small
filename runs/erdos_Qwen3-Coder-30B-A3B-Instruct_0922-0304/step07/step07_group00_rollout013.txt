import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (target_sum / np.sum(h_current))
    else:
        h_current = np.ones(n_points) * 0.5
        
    # Initialize dual variables for constraints
    dual_vars = np.zeros(n_points * 2 - 1)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap_idx = np.argmax(overlap)
        
        # Ascent direction from dual variables 
        # Really we should add a convex relaxation over basis functions
        # but let's simpler approach - just gradient descent on objective
        # via perturbing the solution slightly in gradient direction
        
        # Perturb h in a direction that works well empirically
        h_new = h_current.copy()
        
        # Make a small perturbation
        perturb_mag = 0.02 * np.exp(-iteration / 100)
        if iteration > 0:
            # Try to lower the max overlap by smoothing the least important regions
            idxs = np.arange(n_points)
            # Instead of trying derivatives, do simulated annealing-like local search
            for i in range(10):
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-perturb_mag, perturb_mag)
                h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Renormalize
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Check if this improved the solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Use the best one for next iteration
        h_current = h_new.copy()
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with gradient-like search
    h_refined = best_h.copy()
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Simple local gradient ascent on small perturbations
        h_temp = h_refined.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.01)
        h_temp[idx] = np.clip(h_temp[idx] + delta, 0, 1)
        h_temp = h_temp * (target_sum / np.sum(h_temp))
        c5_temp = compute_c5(h_temp, dx)
        if c5_temp < compute_c5(h_refined, dx):
            h_refined = h_temp.copy()
            
    final_c5 = compute_c5(h_refined, dx)
    return (h_refined.tolist(), final_c5, n_points)
