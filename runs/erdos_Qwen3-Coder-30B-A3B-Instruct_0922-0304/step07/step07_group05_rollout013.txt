import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    num_particles = 100
    particles = []
    velocities = []
    best_particle = None
    best_fitness = float('inf')
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Initialize with small variations of the initial solution
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
        for _ in range(num_particles // 5 - 1):
            perturbation = np.random.normal(0, 0.02, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
            particles.append(h_perturbed)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
    
    # Fill the rest of the population
    for _ in range(num_particles - len(particles)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    inertia_w = 0.8
    cognitive_c = 1.5
    social_c = 1.5
    max_iterations = 60
    
    # Particle bests tracking
    particle_best_positions = [p.copy() for p in particles]
    particle_best_fitnesses = [compute_c5(p, dx) for p in particles]
    
    iteration = 0
    stagnation_count = 0
    last_best = best_fitness
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        for i in range(num_particles):
            # Compute current fitness
            current_fitness = particle_best_fitnesses[i]
            if current_fitness < best_fitness:
                best_fitness = current_fitness
                best_particle = particle_best_positions[i].copy()
            
            # Update if new global best
            if current_fitness < particle_best_fitnesses[i]:
                particle_best_fitnesses[i] = current_fitness
                particle_best_positions[i] = particles[i].copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_c * r1 * (particle_best_positions[i] - particles[i])
            social = social_c * r2 * (best_particle - particles[i])
            velocities[i] = (inertia_w * velocities[i] + cognitive + social)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to maintain constraint
            particles[i] = normalize_h(particles[i], target_sum)
        
        # Adjust parameters
        if iteration % 10 == 0:
            inertia_w *= 0.98
        
        # Check for stagnation
        if best_fitness < last_best:
            last_best = best_fitness
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 10:
                # Increase mutation to escape local minima
                for i in range(num_particles):
                    mutation_strength = 0.05
                    noise = np.random.normal(0, mutation_strength, n_points)
                    particles[i] += noise
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = normalize_h(particles[i], target_sum)
                stagnation_count = 0
        
        iteration += 1
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    # Final refinement with gradient descent for best individual
    if best_particle is not None:
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        from scipy.optimize import minimize
        
        refine_iter = 0
        result = minimize(objective, best_particle, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 20, 'ftol': 1e-6})
        
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            best_c5 = compute_c5(best_h, dx)
        else:
            best_h = best_particle
            best_c5 = best_fitness
    else:
        best_h = particles[0]
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
