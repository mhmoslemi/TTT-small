import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_basis_functions(n_points, n_basis=16):
    """Create a set of basis functions, each integrating to 1/(n_points/2)"""
    basis = []
    target_sum = n_points / 2
    
    # Create pulse functions with different widths and positions
    widths = [1, 2, 3, 4, 5]  # widths in points
    for w in widths:
        for i in range(n_points - w):
            phi = np.zeros(n_points)
            phi[i:i+w] = 1.0
            phi = normalize_h(phi, target_sum)
            basis.append(phi)
    
    # Ensure we have exactly n_basis functions (or more if needed)
    while len(basis) < n_basis:
        # Add linear combinations to increase diversity
        i, j = np.random.randint(0, len(basis), 2)
        alpha = np.random.rand()
        new_basis = alpha * basis[i] + (1-alpha) * basis[j]
        new_basis = normalize_h(new_basis, target_sum)
        basis.append(new_basis)
    
    return np.array(basis[:n_basis])

def construct_and_solve_subproblem(basis, dx, target_sum, mu_vals):
    """Solve the subproblem using convex optimization."""
    n_basis = basis.shape[0]
    
    # Create decision variables
    lambdas = Variable(n_basis)
    
    # Objective: minimize sum_k mu_k * sum_{i,j} lambda_i * lambda_j * (A_ij^k - A_ij^k * A_jj^k)
    # We're actually solving: minimize sum_k mu_k * term_k
    
    # However, we're going to compute the full summing approach in a simplified way
    # The actual computation of A_ij^k would be too complex, so we directly work with the functions
    
    # Approximation: Create a simpler energy-based approach
    # Use a quadratic approximation of the overlap for iteratively solving
    
    # Compute dot products for simplified approach
    A = np.zeros((n_basis, n_basis))
    for i in range(n_basis):
        A[i, i] = np.sum(basis[i] * (1 - basis[i]))
        for j in range(i+1, n_basis):
            A[i, j] = np.sum(basis[i] * (1 - basis[j]))
            A[j, i] = A[i, j]
    
    # However this approach is too convoluted for now, we switch to a simpler method:
    # Use direct optimization on coefficients without full convolution matrix
    
    # Instead, we'll create a convex relaxation
    # Just impose: sum(lambda) = 1, lambda >= 0
    
    # Let's approximate the actual C5 with an average of possible combinations
    # For practicality, generate multiple samples from the basis and choose the best
    
    return lambdas

def adapt_basis_functions(basis, h, n_basis=32):
    """Add new basis functions based on current h"""
    new_basis = []
    # Attempt to add functions like individual spikes or pulses near supporting regions
    # Here we just generate 16 more based on different pulse locations
    target_sum = np.sum(basis[0])
    
    # Explore regions with high h values to make more detailed functions
    points = np.where(h > np.quantile(h, 0.7))[0]
    if len(points) > 0 and len(basis) < n_basis:
        for _ in range(16):
            i = np.random.choice(points)
            w = np.random.choice([1, 2, 3])
            phi = np.zeros(len(h))
            start = max(0, i - w//2)
            end = min(len(h), start + w)
            phi[start:end] = 1.0
            phi = normalize_h(phi, target_sum)
            new_basis.append(phi)
    
    # Only return basis if we still need more
    if len(basis) + len(new_basis) <= n_basis:
        return np.vstack([basis, np.array(new_basis)])
    else:
        return basis[:n_basis]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    else:
        # Create a simple initial guess
        initial_h = np.random.rand(n_points)
        initial_h = normalize_h(initial_h, target_sum)
    
    # Initialize basis functions
    basis = create_basis_functions(n_points, n_basis=16)
    current_basis = basis
    
    # Try direct optimization on initial basis 
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Simple gradient-based improvement
    # But start with the basis approach
    for it in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Fit to basis representation
        try:
            # Try to represent initial_h in terms of current basis
            # Least squares fits
            A = current_basis.T @ current_basis
            b = current_basis.T @ initial_h
            lambdas = np.linalg.solve(A + 1e-8 * np.eye(A.shape[0]), b)
            lambdas = np.maximum(lambdas, 0)
            lambdas = lambdas / np.sum(lambdas)  # Normalize
            
            # Create new h from these coefficients
            new_h = np.sum(lambdas[:, None] * current_basis, axis=0)
            new_h = normalize_h(new_h, target_sum)
            new_h = np.clip(new_h, 0, 1)
            
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                
            # Termination if good enough
            if best_c5 <= 0.3807:
                break
                
        except:
            pass
        
        # Occasionally explore new basis if stuck
        if it % 50 == 0 and len(current_basis) < 32:
            current_basis = adapt_basis_functions(current_basis, initial_h, n_basis=32)
        
        # Recompute from basis approach each time a bit for better fit and generalization
        if it % 10 == 0:
            # Reconstruct with adaptive basis function
            try:
                A = current_basis.T @ current_basis + 1e-6 * np.eye(current_basis.shape[0])
                b = current_basis.T @ best_h
                lambdas = np.linalg.solve(A, b)
                lambdas = np.maximum(lambdas, 0)
                if np.sum(lambdas) > 0:
                    lambdas = lambdas / np.sum(lambdas)
                    new_h = np.sum(lambdas[:, None] * current_basis, axis=0)
                    new_h = normalize_h(new_h, target_sum)
                    new_h = np.clip(new_h, 0, 1)
                    new_c5 = compute_c5(new_h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
            except:
                pass

    # Final refinement with grid search on a small neighborhood
    # for candidate improvements in close vicinity
    best_h_final = best_h.copy()
    final_c5 = best_c5
    
    # Simple localized search: 7-pulse structures
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = best_h_final.copy()
        
        # Choice of 1/2 of 5-elements σ (small perturbation)
        # In so-called "reduce-then-solve" or domain randomization mode
        pidx = np.random.randint(0, n_points)
        # Modify bounds
        if h_new[pidx] > 0.1 and h_new[pidx] < 0.9:
            h_new[pidx] = np.clip(h_new[pidx] + np.random.normal(0, 0.01), 0, 1)
            h_new = normalize_h(h_new, target_sum)
        
        test_c5 = compute_c5(h_new, dx)
        if test_c5 < final_c5:
            final_c5 = test_c5
            best_h_final = h_new.copy()
            if final_c5 <= 0.3807:
                break
    
    return (best_h_final.tolist(), final_c5, n_points)
