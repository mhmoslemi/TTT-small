import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            sample = np.array(initial_h_values) + np.random.normal(0, 0.01, n_points)
        else:
            sample = np.random.uniform(0, 1, n_points)
        sample = project_to_constraints(sample, target_sum)
        population.append(sample.copy())
    
    # Evaluate initial population
    fitnesses = []
    for ind in population:
        fitnesses.append(compute_c5(ind, dx))
    
    # Sort by fitness (lower is better)
    indices = np.argsort(fitnesses)
    population = [population[i] for i in indices]
    fitnesses = [fitnesses[i] for i in indices]
    
    # Best solution so far
    best_idx = 0
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Evolution loop
    max_iter = 30000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Adaptive mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Select elite
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        while len(offspring) < population_size - elite_size:
            # Tournament selection
            parent_idx = np.random.randint(0, elite_size)
            parent = elite[parent_idx]
            
            # Mutation
            mutant = parent + np.random.normal(0, mutation_rate, n_points)
            
            # Project to feasible region
            mutant = project_to_constraints(mutant, target_sum)
            
            offspring.append(mutant)
        
        # Add elite individuals
        offspring.extend(elite)
        
        # Evaluate offspring
        offspring_fitnesses = []
        for ind in offspring:
            offspring_fitnesses.append(compute_c5(ind, dx))
        
        # Combine and sort
        population = offspring
        fitnesses = offspring_fitnesses
        indices = np.argsort(fitnesses)
        population = [population[i] for i in indices]
        fitnesses = [fitnesses[i] for i in indices]
        
        # Update best
        if fitnesses[0] < best_fitness:
            best_h = population[0].copy()
            best_fitness = fitnesses[0]
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    return (best_h.tolist(), best_fitness, n_points)
