import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Construct smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        # Create a smooth transition
        for i in range(n_points):
            h[i] = max(0, min(1, 2 * (i - mid/2) / (n_points/2)))
        h = project_to_feasible(h, target_sum)
        
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Remez-like iteration with phase adjustments
    max_iterations = 100
    alpha = 0.1
    decay = 0.95
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        # The overlap with largest value corresponds to the maximum C5
        max_overlap = overlap[max_lag_idx]
        c5_val = max_overlap * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Try to find dominant frequency from correlation
        # Perform FFT on overlap to identify dominant frequency
        fft_overlap = np.fft.fft(overlap)
        # Find frequency with largest magnitude (excluding DC)
        freq_indices = np.argsort(np.abs(fft_overlap))[::-1]
        dominant_freq_idx = freq_indices[1] if freq_indices[0] == 0 else freq_indices[0]
        
        # Generate sinusoidal update based on dominant frequency
        freq = dominant_freq_idx / len(overlap)
        update = np.sin(2 * np.pi * freq * np.arange(len(h)))
        
        # Adjust h in direction of reduction in c5
        h_new = h - alpha * update
        h_new = project_to_feasible(h_new, target_sum)
        
        # Try a second variation if beneficial
        h_new2 = h + alpha * update
        h_new2 = project_to_feasible(h_new2, target_sum)
        
        # Evaluate both updates
        c5_new1 = compute_c5(h_new, dx)
        c5_new2 = compute_c5(h_new2, dx)
        
        if c5_new1 < c5_new2:
            h = h_new
        else:
            h = h_new2
            
        alpha *= decay
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
