import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute the gradient of overlap at k_max."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1 - 2 * h[i + k_max]
    return grad

def solve_lp_direction(grad, n_points):
    """Solve the linear program to find step direction s."""
    # s minimizes grad^T * s subject to sum(s) = 0, 0 <= s <= 1
    s = np.zeros(n_points)
    # Simple approach: set to -grad and project to feasibility
    s = -grad
    # Ensure sum is zero by adjusting
    s = s - np.mean(s)
    s = np.clip(s, 0, 1)
    return s

def line_search(h, grad, k_max, dx, max_alpha=1.0):
    """Perform backtracking line search."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    step_size = 0.5
    
    # Remove small steps
    while alpha > 1e-6:
        candidate = h + alpha * (solve_lp_direction(grad, len(h)) - h)
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, len(h) / 2)
        new_c5 = compute_c5(candidate, dx)
        if new_c5 < current_c5:
            return alpha
        alpha *= step_size
    return alpha

    np.random.seed(seed)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Iteration counter
    iter_count = 0
    stagnation = 0
    
    while time.time() - start_time < budget_s - 1 and stagnation < 500:
        # Compute all overlaps to find k_max
        overlaps = []
        for k in range(n_points):
            # Compute overlap for shift k using correlation
            overlap = np.sum(h[:-k] * (1 - h[k:])) if k > 0 else np.sum(h * (1 - h))
            overlaps.append(overlap)
        
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute gradient
        grad = compute_gradients(h, k_max)
        
        # Solve for step direction and optimize step size
        s = solve_lp_direction(grad, n_points)
        alpha = line_search(h, grad, k_max, dx)
        
        # Update h
        h = h + alpha * (s - h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Evaluate performance
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation = 0
        else:
            stagnation += 1
            
        # Early termination check
        if best_c5 <= 0.3807:
