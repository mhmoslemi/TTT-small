import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 100
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = normalize_h(h_ref, target_sum)
    else:
        h_ref = np.zeros(n_points)
        h_ref[:n_points//2] = 1
        h_ref += np.random.normal(0, 0.02, n_points)
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = normalize_h(h_ref, target_sum)
    
    # Create initial population
    for i in range(n_particles):
        if i < n_particles // 5:  # 20% from reference
            h = h_ref + np.random.normal(0, 0.03, n_points)
        else:  # 80% random
            h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 60
    
    # Track best solution
    gbest_idx = 0
    gbest_c5 = compute_c5(particles[0], dx)
    gbest_h = particles[0].copy()
    
    stagnation_count = 0
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (particles[gbest_idx] - particles[i]) +
                           c2 * r2 * (gbest_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate
            c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_h = particles[i].copy()
                gbest_idx = i
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Update inertia weight
        w = max(0.4, w * 0.98)
        
        # Early termination
        if gbest_c5 <= 0.3807:
            break
            
        if stagnation_count > 20:
            break
    
    # Final refinement with local search
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        try:
            result = minimize(objective, gbest_h.copy(), method='SLSQP', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 50})
            if result.success:
                gbest_h = result.x
                gbest_h = np.clip(gbest_h, 0, 1)
                gbest_h = normalize_h(gbest_h, target_sum)
                gbest_c5 = compute_c5(gbest_h, dx)
            else:
                break
        except:
            break
    
    final_c5 = compute_c5(gbest_h, dx)
    return (gbest_h.tolist(), final_c5, n_points)
