import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient-based optimization with constraints
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    from scipy.optimize import minimize
    
    # Support for bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints: sum equals target_sum
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Iteratively refine solution with multiple strategies
    max_iterations = 1000
    iteration = 0
    prev_c5 = best_c5 + 1.0  # ensure first iteration runs
    
    while time.time() - start_time < budget_s - 20 and iteration < max_iterations:
        iteration += 1
        
        # Run optimization with SLSQP method
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        # Check if optimization was successful
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
                prev_c5 = best_c5 + 1.0  # reset convergence check
        else:
            # If optimization failed, try a random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = best_h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum > 0:
                h_perturbed = h_perturbed * (target_sum / h_perturbed_sum)
            c5_perturbed = evaluate_overlap(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        # Check convergence (small improvement threshold)
        if abs(prev_c5 - best_c5) < 1e-6:
            break
        prev_c5 = best_c5
    
    # Final refinement with SLSQP
    final_result = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=constraints, 
                           options={'maxiter': 200, 'ftol': 1e-9})
    
    if final_result.success:
        final_h = np.clip(final_result.x, 0, 1)
        final_sum = np.sum(final_h)
        if final_sum > 0:
            final_h = final_h * (target_sum / final_sum)
        final_c5 = evaluate_overlap(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
