import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a better initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add small Gaussian noise to break symmetry
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Use coordinate-wise optimization with a modified approach
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Learning parameters
    eta = 0.05  # learning rate
    alpha = 0.9  # momentum
    beta = 0.999  # momentum for RMSprop
    epsilon = 1e-8  # numerical stability
    lambda_reg = 1e-4  # regularization strength
    
    # First order momentum
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # For tracking improvement
    last_improvement_iter = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        overlaps = {}
        for k in range(-n_points + 1, n_points):
            shift = np.roll(1 - h, k)
            if k >= 0:
                overlap = np.sum(h[:n_points-k] * shift[k:])
            else:
                overlap = np.sum(h[-k:] * shift[:-k])
            overlaps[k] = overlap
        
        # Select the maximum overlap
        C5_current = max(overlaps.values()) * dx

        # Compute gradients
        gradient = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            shift = np.roll(1 - h, k)
            # For each point i, calculate the contribution to the gradient with respect to h[i]
            if k >= 0:
                # h[i] * (1-h[i+k]) for i = 0..n_points-k-1
                for i in range(n_points - k):
                    gradient[i] += shift[i+k]  # partial derivative of h[i] in overlap_k
            else:
                # h[i-k] * (1-h[i]) for i = -k to n_points-1
                for i in range(-k, n_points):
                    gradient[i] += shift[i]  # partial derivative of h[i] in overlap_k

        # gradient is actually calculated as the derivative of C5 with respect to h.
        # Since C5 is the maximum of overlaps, we use the gradient with respect to the maximizing shift.
        # We're approximating by one of them, which is a good heuristic if C5 is relatively smooth.
        # This simplifies our estimate to compute a full gradient from one wall of overlaps.
        max_key = max(overlaps, key=overlaps.get)
        shift = np.roll(1 - h, max_key)
        if max_key >= 0:
            gradient = shift[max_key:]  # h[i] contribution is from overlapping with (1-h[i+max_key])
        else:
            gradient = shift[:-max_key]  # h[i+k] contribution is from overlapping with (1-h[i])
        # Ensure proper size of gradient vector for variable the vector
        gradient = np.pad(gradient, (0, n_points - len(gradient)), mode="constant", constant_values=0)
        
        # Add regularization
        gradient += lambda_reg * (h - best_h)
        
        # Momentum and RMSprop Update
        m = alpha * m + (1 - alpha) * gradient
        v = beta * v + (1 - beta) * (gradient * gradient)
        m_hat = m / (1 - alpha ** (iteration + 1))
        v_hat = v / (1 - beta ** (iteration + 1))
        update = eta * m_hat / (np.sqrt(v_hat) + epsilon)

        # Apply update to relaxed version
        h_relaxed = h - update
        
        # Project to feasible region
        h_new = np.clip(h_relaxed, 0, 1)
        h_new = project_to_constraint(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Update best if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement_iter = iteration
        
        # Point updates
        h = h_new.copy()
        
        # Adaptive learning rate
        if iteration - last_improvement_iter > 100:
            eta *= 0.95  # reduce learning rate
            
        # Adapt regularization
        if abs(new_c5 - C5_current) < 1e-5:
            # No significant improvement in objective
            lambda_reg *= 1.3
        else:
            lambda_reg *= 0.95
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
