import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Adaptive Gaussian Mixture Optimization (AGMA)
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize Gaussian components
    means = []
    covariances = []
    weights = []
    
    # Add component from initial solution
    if 'initial_h_values' in globals():
        means.append(h.copy())
    else:
        means.append(np.random.rand(n_points))
        means[-1] = np.clip(means[-1], 0, 1)
        means[-1] = normalize_h(means[-1], target_sum)
    
    # Add random components for diversity
    for i in range(n_components - 1):
        rand_h = np.random.rand(n_points)
        rand_h = np.clip(rand_h, 0, 1)
        rand_h = normalize_h(rand_h, target_sum)
        means.append(rand_h)
    
    # Initialize covariances
    for i in range(n_components):
        covariances.append(initial_sigma * np.eye(n_points))
        
    # Initialize equal weights
    weights = np.ones(n_components) / n_components
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iterations = 6000  # To have buffer for time budget
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Sample from mixture model
        candidates = []
        candidate_scores = []
        
        # Sample from each component according to weights
        for i in range(population_size):
            # Select component
            comp_idx = np.random.choice(n_components, p=weights)
            component_mean = means[comp_idx]
            component_cov = covariances[comp_idx]
            
            # Sample from Gaussian
            sample = np.random.multivariate_normal(component_mean, component_cov)
            # Clip and normalize
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            
            # Evaluate
            c5 = compute_c5(sample, dx)
            candidates.append(sample.copy())
            candidate_scores.append(c5)
            
        # Sort candidates by score
        sorted_indices = np.argsort(candidate_scores)
        top_candidates = [candidates[i] for i in sorted_indices[:keep_top]]
        top_scores = [candidate_scores[i] for i in sorted_indices[:keep_top]]

        if top_scores[0] < best_c5:
            best_c5 = top_scores[0]
            best_h = top_candidates[0].copy()
        
        # Update GMM components
        if len(top_candidates) > 0:
            # Update means
            for i in range(n_components):
                if i < len(top_candidates):
                    means[i] = np.mean(top_candidates[:keep_top], axis=0)
                else:
                    # Keep means unchanged

            # Update covariances using top candidates
            for i in range(n_components):
                if i < len(top_candidates):
                    # Compute covariance matrix for selected candidates
                    selected = np.array(top_candidates[:keep_top])
                    diff = selected - means[i]
                    # Update covariance
                    if len(diff) > 1:
                        covariances[i] = np.cov(diff.T) + 1e-8 * np.eye(n_points)
                    else:
                        covariances[i] = 0.01 * np.eye(n_points) + 1e-8 * np.eye(n_points)
                else:
                    # Keep covariance unchanged
                    pass
            
            # Update weights (simple approach: inverse of scores)
            # Normalize scores
            normalized_scores = np.array(top_scores[:keep_top])
            if np.all(normalized_scores == normalized_scores[0]):

            else:
                # Use inverse of normalized scores
                weights = 1.0 / (normalized_scores + 1e-8)
                weights = weights / np.sum(weights)
        
        # Decay covariances to encourage exploration
        for i in range(n_components):
            covariances[i] *= covariance_decay
            
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
