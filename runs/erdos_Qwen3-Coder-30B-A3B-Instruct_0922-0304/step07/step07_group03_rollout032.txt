import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, target_sum, dx, maxiter=1000):
    """Local optimization using L-BFGS-B."""
    def objective(x):
        return compute_c5(x, dx)
    
    def grad(x):
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = project_to_constraints(x_plus, target_sum)
            x_minus = x.copy()
            x_minus[i] -= eps
            x_minus = project_to_constraints(x_minus, target_sum)
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2*eps)
        return grad
    
    bounds = [(0, 1) for _ in range(len(h))]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      jac=grad, options={'maxiter': maxiter})
    return project_to_constraints(result.x, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # ES parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_generations = 2000
    
    # Initialize population
    population = []
    fitness = []
    
    # Main ES loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        if gen == 0:
            # Evaluate initial population
            fitness = []
            for i in range(pop_size):
                if i == 0:
                    individual = h.copy()
                else:
                    individual = np.random.uniform(0, 1, n_points)
                    individual = normalize_h(individual, target_sum)
                
                individual = project_to_constraints(individual, target_sum)
                fit = compute_c5(individual, dx)
                population.append(individual)
                fitness.append(fit)
        else:
            # Selection: keep top 50%
            sorted_indices = np.argsort(fitness)
            selected_indices = sorted_indices[:pop_size//2]
            selected_pop = [population[i] for i in selected_indices]
            selected_fitness = [fitness[i] for i in selected_indices]
            
            # Create new population by mutation
            new_population = selected_pop.copy()
            
            # Add mutated individuals
            for i in range(pop_size - len(selected_pop)):
                parent = selected_pop[i % len(selected_pop)]
                # Mutation with adaptive sigma
                sigma = sigma_init - (sigma_init - sigma_final) * gen / max_generations
                noise = np.random.normal(0, sigma, n_points)
                child = parent + noise
                child = project_to_constraints(child, target_sum)
                new_population.append(child)
                
            population = new_population
            fitness = []
            for indiv in population:
                fit = compute_c5(indiv, dx)
                fitness.append(fit)
                
        # Local optimization phase
        min_fitness = min(fitness)
        if min_fitness < 0.381:
            # Optimize the best individual
            best_idx = np.argmin(fitness)
            best_individual = population[best_idx].copy()
            optimized = local_optimize(best_individual, target_sum, dx, 1000)
            optimized_fitness = compute_c5(optimized, dx)
            if optimized_fitness < min_fitness:
                population[best_idx] = optimized
                fitness[best_idx] = optimized_fitness
                
        # Early termination
        if min(fitness) <= 0.3807:
            break
    
    # Return best solution
    final_fitness = min(fitness)
    final_idx = np.argmin(fitness)
    final_h = population[final_idx]
    
    return (final_h.tolist(), final_fitness, n_points)
