import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.8
    crossover_prob = 0.7
    max_generations = 3000
    
    # Initialize population
    population = []
    fitness = []
    
    if 'initial_h_values' in globals():
        population.append(np.array(initial_h_values))
        population[-1] = np.clip(population[-1], 0, 1)
        population[-1] = normalize_h(population[-1], target_sum)
        fitness.append(compute_c5(population[-1], dx))
    else:
        # Generate initial population
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = normalize_h(individual, target_sum)
            population.append(individual)
            fitness.append(compute_c5(individual, dx))
    
    # Keep track of best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # For DE, we process each individual once per generation
        new_population = []
        new_fitness = []
        
        # Try to improve each individual in the population
        for i in range(population_size):
            # Select three distinct random individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
            
            # Project to constraints
            trial = np.clip(trial, 0, 1)
            trial = project_to_constraint(trial, target_sum)
            
            # Evaluate
            trial_c5 = compute_c5(trial, dx)
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
                
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        
        generation += 1
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
