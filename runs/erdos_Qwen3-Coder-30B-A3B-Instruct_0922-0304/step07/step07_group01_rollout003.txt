import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

    # Simple alternative: generate a trial solution
    # For now, just optimizing the simplest step pattern that gives good result
    # Try binary-like pattern with sqrt(n) steps
    pattern = np.zeros(n_points)
    steps = int(np.sqrt(n_points))
    step_size = n_points // steps
    for i in range(steps):
        start = i * step_size
        end = min((i + 1) * step_size, n_points)
        pattern[start:end] = 1.0
    return pattern

def local_refine_with_gradient(h, dx, target_sum, n_points):
    # Simple local search around current solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Try simple perturbations

        # Slightly perturb each element

        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            
    return best_h, best_c5
