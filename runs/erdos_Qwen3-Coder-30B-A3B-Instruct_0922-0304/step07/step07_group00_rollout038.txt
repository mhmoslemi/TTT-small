def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_basis_functions(n_points, n_basis=16):

    basis = []

        l = np.zeros(n_points)
        # Determine pulse width and position
        pulse_width = 0.1 + 0.3 * (i % 4) / 3.0  # 0.1 to 0.4
        center = 0.1 + 0.8 * (i / n_basis)  # 0.1 to 0.9
        start = max(0, int((center - pulse_width/2) / dx))
        end = min(n_points, int((center + pulse_width/2) / dx))
        
        # Make sure we have area 1
        if end > start:
            area = (end - start) * dx
            l[start:end] = 1.0 / area
        else:
            # Handle case where pulse width too narrow
            l[0] = 1.0 / dx
        l = normalize_h(l, target_sum)
        basis.append(l)
    
    return basis

def compute_overlap_matrix(basis, dx):
    """Compute overlap integrals A_ij^k for all basis functions."""
    n_basis = len(basis)
    n_points = len(basis[0])
    # Precompute full overlaps for all basis pairs
    A = np.zeros((n_basis, n_basis, 2*n_points-1))
    
    for i in range(n_basis):
        for j in range(n_basis):
            A[i, j] = correlate(basis[i], basis[j], mode="full")
    
    return A
