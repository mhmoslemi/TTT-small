import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h_relaxed, target_sum):
    """Project to feasible region and enforce sum constraint."""
    h = np.clip(h_relaxed, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_relaxed = np.array(initial_h_values)
        h_relaxed = np.clip(h_relaxed, 0, 1)
        h_relaxed = h_relaxed * (target_sum / np.sum(h_relaxed))
    else:
        # Initialize with a step function
        h_relaxed = np.zeros(n_points)
        h_relaxed[:n_points//2] = 1
        # Random perturb
        h_relaxed += np.random.normal(0, 0.1, n_points)
        h_relaxed = project_to_feasible(h_relaxed, target_sum)
    
    # Parameters
    lr = 0.1
    beta = 0.9
    reg_weight = 1e-4
    momentum = np.zeros(n_points)
    velocity = np.zeros(n_points)
    
    best_c5 = float('inf')
    best_h = h_relaxed.copy()
    stagnation_count = 0
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 3000:
        # Compute current C5
        current_c5 = compute_c5(project_to_feasible(h_relaxed, target_sum), dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_relaxed.copy()
            last_improvement = iteration
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        epsilon = 1e-4
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h_relaxed.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update with Adam-like scheme
        momentum = beta * momentum + (1 - beta) * grad
        velocity = beta * velocity + (1 - beta) * grad**2
        adjusted_grad = momentum / (np.sqrt(velocity) + 1e-8)
        
        # Apply update with regularization term
        h_relaxed_new = h_relaxed - lr * (adjusted_grad + reg_weight * (h_relaxed - best_h))
        
        # Apply PGD-style projection step
        h_relaxed = project_to_feasible(h_relaxed_new, target_sum)
        
        # Adjust learning rate and regularization
        if iteration - last_improvement > 100:
            lr *= 0.9
            if lr < 1e-6:
                lr = 1e-6
            last_improvement = iteration
            
        if reg_weight < 1e-2:
            reg_weight *= 1.2
        else:
            reg_weight *= 0.95
            
        if iteration - last_improvement > 200:
            reg_weight = 1e-3
            
        iteration += 1
    
    # Final computation
    final_h = project_to_feasible(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
