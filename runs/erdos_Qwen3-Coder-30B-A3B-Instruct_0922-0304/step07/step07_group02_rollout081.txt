import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_sum_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5
        
    # PGD parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 3000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # Find the shift k that maximizes overlap
        max_k = np.argmax(overlap)
        # Get the gradient: (1 - h) shifted back by k
        gradient = np.zeros(n_points)
        # Compute shift polynomial (this is a full convolution, we need to get back the correct shift)
        shifted = np.zeros(n_points)
        shifted[max_k] = 1
        # Get 1-h shifted
        shifted_1_h = np.zeros(n_points)
        shifted_1_h[max_k] = 1
        # If the shift is 0, no shift needed
        if max_k != 0:
            # For full correlation, max_k corresponds to shift of (max_k - (n_points - 1))
            # or the corresponding shift index in terms of our coordinate system
            shift_index = max_k - (n_points - 1)
            # This is normally a negative shift, so we construct shifted version of (1 - h) for that
            if shift_index < 0:
                shifted_1_h[:shift_index] = 1 - h[-shift_index:]
            elif shift_index > 0:
                shifted_1_h[shift_index:] = 1 - h[:-shift_index]
            else:
                shifted_1_h = 1 - h
        # The gradient is the corresponding shifted 1-h for that max_k:
        gradient = shifted_1_h
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Project to maintain sum constraint
        h = project_to_sum_constraint(h, target_sum)
        
        # Evaluate current
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if no_improvement_count > 100:
            learning_rate *= 0.95
            no_improvement_count = 0
        learning_rate = max(learning_rate, 1e-5)
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
