import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        
    # Ensure bounds and normalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Precompute shift indices for correlation
    shifts = []
    for k in range(-n_points + 1, n_points):
        shifted = np.arange(n_points) - k
        # Wrap around indices
        shifted = shifted % n_points
        shifts.append(shifted)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient-based optimization with adaptive regularization
    max_iter = 10000
    lr = 1e-3
    beta = 0.9
    reg_lambda = 1e-4
    momentum = np.zeros(n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap integral
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            sum_plus = np.sum(h_plus)
            if sum_plus > 0:
                h_plus = h_plus * (target_sum / sum_plus)
            c5_plus = evaluate_overlap(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            sum_minus = np.sum(h_minus)
            if sum_minus > 0:
                h_minus = h_minus * (target_sum / sum_minus)
            c5_minus = evaluate_overlap(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
        # Apply momentum
        momentum = beta * momentum + grad
        h_new = h - lr * momentum
        
        # Apply regularization
        h_new = h_new - reg_lambda * (np.sum(h_new) - target_sum)
        
        # Clamp to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (target_sum / h_new_sum)
        
        # Line search with backtracking
        t = 1.0
        c5_new = evaluate_overlap(h_new, dx)
        
        # Simple backtracking
        while c5_new >= current_c5 and t > 1e-6:
            t *= 0.5
            h_try = h - t * lr * momentum
            h_try = np.clip(h_try, 0, 1) 
            sum_try = np.sum(h_try)
            if sum_try > 0:
                h_try = h_try * (target_sum / sum_try)
            c5_new = evaluate_overlap(h_try, dx)
        
        h = h_new
        # Adjust regularization
        if t < 0.1:
            reg_lambda *= 1.1
        else:
            reg_lambda *= 0.99
            
        # Periodic projection
        if i % 100 == 0:
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (target_sum / h_sum)
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
