import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max):
    """Compute the gradient of the overlap at shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    # Gradient of overlap_k_max = sum_{i=0}^{n-k_max-1} h[i] * (1 - h[i+k_max])
    # For each i, the gradient w.r.t. h[i] is (1 - h[i+k_max])
    # and the gradient w.r.t. h[i+k_max] is -h[i]
    for i in range(n - k_max):
        grad[i] += 1 - h[i + k_max]
        grad[i + k_max] -= h[i]
    return grad

def find_max_overlap_shift(h, dx):
    """Find the shift that maximizes the overlap."""
    overlap = correlate(h, 1 - h, mode="full")
    # Shifts are from -n+1 to n-1, but we're interested in non-negative shifts
    # Top convolution index corresponds to shift 0, so we take only positive shifts
    max_shift = np.argmax(overlap)
    return max_shift - (len(h) - 1)

def line_search(h, s, dx, k_max, alpha_max=1.0, decrease_factor=0.5):
    """Perform backtracking line search to find suitable alpha."""
    alpha = alpha_max
    h_current = h.copy()
    c5_current = compute_c5(h_current, dx)
    grad = compute_gradient(h_current, k_max)
    
    # Evaluate direction
    direction = s - h_current
    # Ensure we don't violate bounds with small steps
    for i in range(len(h_current)):
        if direction[i] > 0:
            # h[i] + alpha * direction[i] <= 1
            max_alpha_pos = (1 - h_current[i]) / direction[i] if direction[i] > 0 else float('inf')
            alpha = min(alpha, max_alpha_pos)
        elif direction[i] < 0:
            # h[i] + alpha * direction[i] >= 0
            max_alpha_neg = h_current[i] / (-direction[i]) if direction[i] < 0 else float('inf')
            alpha = min(alpha, max_alpha_neg)
    
    # Backtracking
    c5_new = c5_current + 1.0  # Initialize to ensure we enter loop
    while alpha > 1e-6 and c5_new >= c5_current:
        h_new = h_current + alpha * direction
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, np.sum(h_current))
        c5_new = compute_c5(h_new, dx)
        if c5_new >= c5_current:
            alpha *= decrease_factor
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    improvement_counter = 0
    best_c5 = float('inf')
    early_stop_counter = 0
    max_early_stop = 500
    
    # Frank-Wolfe iterations
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the maximum overlap
        k_max = find_max_overlap_shift(h, dx)
        if k_max < 0:
            k_max = -k_max  # We focus on positive shifts but interpret correctly
            
        # Compute gradient
        grad = compute_gradient(h, k_max)
        
        # Find direction s that minimizes the gradient
        # s minimizing <grad, s> subject to ||s||_1 <= 1, s>=0, sum(s) = 0
        # Simple approach: take s such that s = argmin <grad, s>
        s = np.zeros(n_points)
        # Same approach as Frank-Wolfe: minimize linearized objective
        # Most negative components get 1, others get 0
        # Actually, we need sum(s) = 0, so we distribute more carefully
        # We want to minimize <grad, s> where s is feasible
        
        # Simple naive approach for s:
        # Let s[i] = 1 if grad[i] is most negative among elements we can change
        # but must also satisfy sum(s) = 0. Best linear setup:
        # We compute the best s that keeps sum(s) = 0, s >= 0, and minimizes <grad, s>
        # That's a linear optimization (minimize <grad, s>) subject to sum(s) = 0, s >= 0.
        # This means we want to pick subset of indices where grad[i] is negative 
        # and reduce them carefully. Instead, let's go to a practical tradeoff:
        
        # Use simple Frank-Wolfe heuristic:
        # Make s by:
        # 1. Pick index i where grad[i] is most negative
        # 2. Pick index j where grad[j] is most positive (for zero sum)
        # 3. Modify so that sum(s) = 0
        
        negative_indices = np.where(grad < 0)[0]
        positive_indices = np.where(grad > 0)[0]
        
        # We might just take:
        s = np.zeros(n_points)
        if len(negative_indices) > 0 and len(positive_indices) > 0:
            # Pick the most negative and most positive and adjust them
            i_neg = negative_indices[np.argmin(grad[negative_indices])]
            i_pos = positive_indices[np.argmax(grad[positive_indices])]
            s[i_neg] = 1.0
            s[i_pos] = -1.0
        elif len(negative_indices) > 0:
            i_neg = negative_indices[np.argmin(grad[negative_indices])]
            s[i_neg] = 1.0
        elif len(positive_indices) > 0:
            i_pos = positive_indices[np.argmax(grad[positive_indices])]
            s[i_pos] = -1.0
        
        # Normalize s to be within feasible range and ensure sum = 0
        # s must be explicitly adjusted to have sum 0, but it's also constrained to [0,1]
        # So it turns out better to think of it as projecting onto constraints:
        # Using projection onto convex set, which in this case is just a line integral
        step = 0.5  # Initial step size
        # Try to keep it feasible, but fall back to some reasonable line search
        temp_h = h + step * (s - h)
        # Clause room to avoid numerical issues:
        temp_h = np.clip(temp_h, 0, 1)
        temp_h = normalize_h(temp_h, target_sum)
        
        # Simple alternate: just do gradient descent
        step = 0.05
        s = -grad * step  # move opposite to gradient
        # Search along line for best point
        temp_h = h + s
        temp_h = np.clip(temp_h, 0, 1)
        temp_h = normalize_h(temp_h, target_sum)
        
        # Apply projected line search and iteration
        c5_before = compute_c5(h, dx)
        alpha, h_next = line_search(h, s, dx, k_max)
        if alpha > 1e-6:
            h = h_next
            
        # Update
        c5_after = compute_c5(h, dx)
        if c5_after < best_c5:
            best_c5 = c5_after
            improvement_counter = 0  # reset counter
        else:
            improvement_counter += 1
        
        if improvement_counter > max_early_stop:
            break
            
        if best_c5 <= 0.3807:
            break
            
        if iteration % 100 == 0:
            # Verify
            _ = compute_c5(h, dx)
            if iteration % 500 == 0:
                print(f"Iteration {iteration}, C5 = {best_c5:.6f}")

    # Final accuracy check and return
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
