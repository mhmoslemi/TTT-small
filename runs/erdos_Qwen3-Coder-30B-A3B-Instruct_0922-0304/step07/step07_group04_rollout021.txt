import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm approach to find better solutions
    def mutate(individual, mutation_rate=0.1):
        child = individual.copy()
        for i in range(len(child)):
            if np.random.random() < mutation_rate:
                # Randomly perturb value
                delta = np.random.normal(0, 0.05)
                child[i] = np.clip(child[i] + delta, 0, 1)
        return child
    
    def crossover(parent1, parent2):
        # Uniform crossover
        child = np.zeros_like(parent1)
        mask = np.random.random(len(parent1)) > 0.5
        child[mask] = parent1[mask]
        child[~mask] = parent2[~mask]
        return child
    
    def fitness(individual):
        # Normalize to target sum
        current_sum = np.sum(individual)
        if current_sum > 0:
            normalized = individual * (target_sum / current_sum)
        else:
            normalized = individual
        
        return evaluate_overlap(normalized, dx)
    
    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.03, n_points)
        individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    # Evolution loop
    max_generations = 500
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection (tournament)
        tournament_size = 3
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population
        new_population = []
        for i in range(0, population_size, 2):
            parent1, parent2 = selected[i], selected[i+1] if i+1 < len(selected) else selected[0]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final refinement with local search
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        # Try multiple starting points for better exploration
        for _ in range(10):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Start from current best + noise
            perturbed_h = best_h + np.random.normal(0, 0.01, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            norms = np.sum(perturbed_h)
            if norms > 0:
                perturbed_h = perturbed_h * (target_sum / norms)
            
            try:
                result = minimize(objective, perturbed_h, method='SLSQP', 
                                bounds=bounds, constraints=constraints, 
                                options={'maxiter': 100, 'ftol': 1e-7})
                
                if result.success:
                    refined_h = np.clip(result.x, 0, 1)
                    refined_sum = np.sum(refined_h)
                    if refined_sum > 0:
                        refined_h = refined_h * (target_sum / refined_sum)
                    final_c5 = evaluate_overlap(refined_h, dx)
                    if final_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = final_c5
            except:
                continue
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
