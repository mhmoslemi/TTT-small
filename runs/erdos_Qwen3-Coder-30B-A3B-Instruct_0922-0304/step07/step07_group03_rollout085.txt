import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values as a smooth sawtooth
        h = np.zeros(n_points)
        for i in range(n_points):
            t = i * dx
            h[i] = 0.5 * (1 - np.cos(2 * np.pi * t))
        h = project_to_constraints(h, target_sum)
    
    # Gradient-based optimization with projected updates
    max_iter = 20000
    learning_rate = 0.1
    decay_factor = 0.999
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap correlation
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Compute gradient approximation for surrogate
        # We approximate gradient of (h ⋆ (1 - h))[k]^2
        # Referenced using the correlation indices
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # For k = 0, h[i] contributes to overlap[0]
            # Calculate gradient with respect to {h[i]}
            for j in range(len(h)):
                if i+j < len(h):
                    grad[i] += 2 * (1 - h[i+j]) * (max_overlap * dx) 
                if i-j >= 0:
                    grad[i] += 2 * (1 - h[i-j]) * (max_overlap * dx)
        
        # Apply gradient with learning rate
        h_new = h - learning_rate * grad
        h_new = project_to_constraints(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        if new_c5 < 0.3807:
            break
        
        h = h_new.copy()
        learning_rate *= decay_factor
        
        if iter_num % 1000 == 0:
            print(f"Iteration {iter_num}, C5 = {best_c5:.6f}")
    
    # Final refinement with smoother approach
    # Perform one more coarse optimization at higher resolution
    h_fine = np.interp(np.linspace(0, 2, 140), np.linspace(0, 2, 70), best_h)
    h_fine = project_to_constraints(h_fine, 70)
    
    max_iter_fine = 5000
    learning_rate_fine = 0.05
    for iter_num in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        overlap = correlate(h_fine, 1 - h_fine, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        grad = np.zeros_like(h_fine)
        for i in range(len(h_fine)):
            for j in range(len(h_fine)):
                if i+j < len(h_fine):
                    grad[i] += 2 * (1 - h_fine[i+j]) * (max_overlap * 0.01) 
                if i-j >= 0:
                    grad[i] += 2 * (1 - h_fine[i-j]) * (max_overlap * 0.01)
        
        h_fine_new = h_fine - learning_rate_fine * grad
        h_fine_new = project_to_constraints(h_fine_new, 70)
        new_c5 = compute_c5(h_fine_new[:70], dx)
        
        if new_c5 < best_c5:
            best_h = h_fine_new[:70].copy()
            best_c5 = new_c5
            
        h_fine = h_fine_new.copy()
        learning_rate_fine *= 0.999
    
    return (best_h.tolist(), best_c5, 70)
