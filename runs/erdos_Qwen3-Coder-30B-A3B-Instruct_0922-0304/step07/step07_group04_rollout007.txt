import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Custom genetic algorithm with hybrid refinement
    population_size = 100
    max_generations = 1000
    tournament_size = max(5, population_size // 20)
    mutation_rate = 0.1
    max_no_improve = 100
    
    # Create initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand_sum = np.sum(h_rand)
        if h_rand_sum > 0:
            h_rand = h_rand * (target_sum / h_rand_sum)
        population.append(h_rand)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fit = evaluate_overlap(ind, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx].copy()
    
    no_improve_count = 0
    
    # Main GA loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, size=tournament_size, replace=False)
            best_in_tournament = np.argmin([fitness[i] for i in tournament])
            selected.append(population[tournament[best_in_tournament]].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_h.copy())
        
        while len(new_population) < population_size:
            # Select parents
            parent1_idx = np.random.randint(0, population_size)
            parent2_idx = np.random.randint(0, population_size)
            
            # Blend crossover
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * selected[parent1_idx] + (1 - alpha) * selected[parent2_idx]
            
            # Mutation
            if np.random.random() < mutation_rate:
                mask = np.random.random(n_points) < 0.3
                noise = np.random.normal(0, 0.05, n_points)
                child += noise * mask
            
            # Ensure bounds and normalization
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        # Evaluate new population
        new_fitness = []
        for ind in new_population:
            fit = evaluate_overlap(ind, dx)
            new_fitness.append(fit)
        
        # Update best
        best_new_idx = np.argmin(new_fitness)
        if new_fitness[best_new_idx] < best_c5:
            best_c5 = new_fitness[best_new_idx]
            best_h = new_population[best_new_idx].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= max_no_improve:
            mutation_rate = min(0.5, mutation_rate * 1.2)
            no_improve_count = 0
            
        # Update population
        population = new_population
        fitness = new_fitness
    
    # Local refinement with gradient-based optimization
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
