import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure constraints on initial h
    h = project_to_constraints(h, target_sum)
    
    c5_bound = compute_c5(h, dx)
    
    # Try convex optimization in frequency domain
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 2 and iter_count < max_iter:
        # Optimization attempt in frequency domain
        # Reconstruct h in time domain from Fourier coefficients
        H = np.fft.fft(h)
        
        # The overlap can be expressed in frequency domain:
        # overlap(k) = sum_j h[j] * (1 - h[j+k])
        # In frequency domain: overlap = h * (1 - h) in convolution
        # So in frequency domain: O(ω) = H(ω) * (1 - H(ω)) (point-wise)
        
        # Let's form a convex relaxation with $H$ real and positive frequencies
        # Minimize max overlap by minimizing relaxation of the frequency response.
        
        # Instead, we will try a simple gradient descent approach to improve bounds. 
        # The overlap is captured by convolution of h and 1-h
        # Let's try gradient descent based on small perturbations
    
        # Gradient-based local search to refine
        step_size = 0.001
        for _ in range(10):  # small number of gradient steps
            # Compute overlap directly
            overlap_full = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap_full)
            
            # Compute gradient estimation (finite diff given small changes)
            grad_estimate = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = project_to_constraints(h_plus, target_sum)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                max_plus = np.max(overlap_plus)
                
                grad_estimate[i] = (max_plus - max_overlap) / eps 
                
            # Update h in gradient direction (negative to minimize)
            h = h - step_size * grad_estimate
            h = project_to_constraints(h, target_sum)
            
        # Update bound
        c5_bound = compute_c5(h, dx)
        
        iter_count += 1
        if c5_bound <= 0.3807:
            break
            
    return (h.tolist(), c5_bound, n_points)
