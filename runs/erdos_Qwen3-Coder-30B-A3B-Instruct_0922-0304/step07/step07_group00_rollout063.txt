import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def generate_basis_functions(n_points, n_basis=20):
    """Generate a set of basis functions for convex combination."""
    basis = []
    for i in range(n_basis):
        # Create piecewise constant basis functions with varying support
        bf = np.zeros(n_points)
        start = int(i * n_points / n_basis)
        end = int((i + 1) * n_points / n_basis)
        bf[start:end] = 1.0
        basis.append(bf)
    return np.array(basis)

def get_basis_overlap_integrals(basis):
    """Precompute the overlap integrals for basis functions."""
    n_basis = len(basis)
    n_points = len(basis[0])
    A = np.zeros((n_basis, n_basis, 2*n_points - 1))
    for i in range(n_basis):
        for j in range(n_basis):
            A[i, j] = correlate(basis[i], basis[j], mode="full")
    return A

def project_to_feasible(h, target_sum):
    """Project h to be feasible within the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70  # Use the same discretization as the parent
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_basis = 24  # Number of basis functions
    
    # Generate or load basis
    basis = generate_basis_functions(n_points, n_basis)
    A = get_basis_overlap_integrals(basis)
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_to_feasible(initial_h, target_sum)
        # Project initial h onto basis space
        x = Variable(n_basis)
        objective = cp.Minimize(cp.sum_squares(basis @ x - initial_h))
        constraints = [cp.sum(x) == 1, x >= 0]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            lambda_init = x.value
        except:
            # Fallback to random initialization
            lambda_init = np.random.rand(n_basis)
            lambda_init = lambda_init / np.sum(lambda_init)
    else:
        lambda_init = np.random.rand(n_basis)
        lambda_init = lambda_init / np.sum(lambda_init)
    
    # Store best
    best_c5 = float('inf')
    best_h = None
    
    # Try a few random initializations for population diversity
    n_solvers = 10
    for run_index in range(n_solvers):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Perturb initial coefficients
        if run_index == 0:
            lambda_current = lambda_init.copy()
        else:
            # Small random perturbation
            lambda_current = lambda_init + np.random.normal(0, 0.01, n_basis)
            lambda_current = np.maximum(lambda_current, 0)
            lambda_current = lambda_current / np.sum(lambda_current)
        
        # Use CVXPY to optimize over lambda dynamically
        max_iter = 500  # max iterations per run
        lambda_k = lambda_current.copy()
        
        # To save time, we will only do dynamics over 200 iterations max
        for iter_k in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Reconstruct h from lambda_k
            h_reconstructed = basis.T @ lambda_k
            h_reconstructed = project_to_feasible(h_reconstructed, target_sum)
            
            # Compute C5
            c5_val = compute_c5(h_reconstructed, dx)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_reconstructed.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Estimate gradient approx by finite differences (simplified)
            # From here is a very basic gradient-free approach - no real gradients used
            # We just sample random perturbations to improve
            perturb = np.random.normal(0, 0.01, n_basis)
            lambda_new = lambda_k + perturb
            lambda_new = np.maximum(lambda_new, 0)
            lambda_new = lambda_new / np.sum(lambda_new)
            
            h_new = basis.T @ lambda_new
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < c5_val:
                lambda_k = lambda_new
                c5_val = c5_new
    
    # Final refinement on the best solution
    if best_h is not None:
        # One final local gradient-free search
        h_test = best_h.copy()
        c5_test = compute_c5(h_test, dx)
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            # Small random perturbations
            noise = np.random.normal(0, 0.005, n_points)
            h_trial = h_test + noise
            h_trial = project_to_feasible(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            
            if c5_trial < c5_test:
                h_test = h_trial
                c5_test = c5_trial
                if c5_test <= 0.3807:
                    break
                    
        best_h = h_test
        best_c5 = c5_test

    return (best_h.tolist(), best_c5, n_points)
