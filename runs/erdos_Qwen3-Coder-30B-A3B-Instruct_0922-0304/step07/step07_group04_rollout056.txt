import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Parameters for genetic algorithm
    population_size = 100
    num_generations = 1000
    tournament_size = max(2, population_size // 10)
    mutation_rate = 0.1
    max_no_improvement = 50
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add the initial solution
    population.append(h.copy())
    population_fitness.append(best_c5)
    
    # Add random solutions
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        current_sum = np.sum(h_rand)
        if current_sum > 0:
            h_rand = h_rand * (target_sum / current_sum)
        population.append(h_rand)
        population_fitness.append(evaluate_overlap(h_rand, dx))
    
    # Track best solution
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # GA loop
    no_improvement_count = 0
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current population
        current_fitness = []
        for ind in population:
            current_fitness.append(evaluate_overlap(ind, dx))
        
        # Selection (tournament)
        selected_indices = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [current_fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_indices.append(winner_idx)
        
        # Create new population via crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1_idx = selected_indices[i]
            parent2_idx = selected_indices[min(i+1, population_size-1)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Blend crossover
            alpha = np.random.uniform(0.2, 0.8)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_strength = 0.1 * (1.0 - generation / num_generations)
                mask = np.random.random(n_points) < 0.1
                child1 += np.random.normal(0, mutation_strength, n_points) * mask
                child1 = np.clip(child1, 0, 1)
                current_sum = np.sum(child1)
                if current_sum > 0:
                    child1 = child1 * (target_sum / current_sum)
            
            if np.random.random() < mutation_rate:
                mutation_strength = 0.1 * (1.0 - generation / num_generations)
                mask = np.random.random(n_points) < 0.1
                child2 += np.random.normal(0, mutation_strength, n_points) * mask
                child2 = np.clip(child2, 0, 1)
                current_sum = np.sum(child2)
                if current_sum > 0:
                    child2 = child2 * (target_sum / current_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        
        # Track best solution
        current_best_idx = np.argmin([evaluate_overlap(ind, dx) for ind in population])
        current_best_c5 = evaluate_overlap(population[current_best_idx], dx)
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive mutation rate
        if no_improvement_count > 10:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        else:
            mutation_rate = max(0.01, mutation_rate * 0.99)
        
        # Early stopping
        if no_improvement_count > max_no_improvement:
            break
    
    # Local refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Return final solution
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
