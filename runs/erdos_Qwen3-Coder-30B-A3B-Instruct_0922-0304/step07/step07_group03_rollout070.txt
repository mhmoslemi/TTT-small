import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit proportionally
    deficit = target_sum - current_sum
    if deficit > 0:
        # Need to increase values
        h += deficit * h / current_sum
    else:
        # Need to decrease values
        h *= (target_sum / current_sum)
    return np.clip(h, 0, 1)

    best_c5 = compute_c5(h, dx)

        # We want to reduce the maximum value in the correlation
        # Rough gradient: for each k, we have that h[i] contributes to (1-h[i+k])

        # Use negative gradient of the overlap
        # Approximate gradient of max overlap

        # Compute what would happen if we slightly reduce the current h
        # Look at the correlation matrix to see where to make changes
        gradient = np.zeros_like(h)

        # Since it's full correlation, max_idx is between 0 and 2*n-2

        # But intuitive: to get k from max_idx in full correl:

    return (best_h.tolist(), best_c5, n_points)
