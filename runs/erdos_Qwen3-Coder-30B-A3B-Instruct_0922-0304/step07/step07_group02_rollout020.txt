import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    
    # Initialize GMM components
    gmm_means = []
    gmm_covs = []
    gmm_weights = []
    
    # Add component centered on initial solution if available
    if 'initial_h_values' in globals():
        gmm_means.append(h.copy())
    else:
        gmm_means.append(np.random.rand(n_points))
    
    # Add additional random components
    for _ in range(n_components - len(gmm_means)):
        gmm_means.append(np.random.rand(n_points))
    
    # Initialize covariances and weights
    for i in range(n_components):
        gmm_covs.append(np.eye(n_points) * 0.1)
        gmm_weights.append(1.0 / n_components)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 6000:
        # Sample new candidates
        candidates = []
        candidate_c5s = []
        
        for _ in range(population_size):
            # Sample from GMM
            comp_idx = np.random.choice(n_components, p=gmm_weights)
            sample = np.random.multivariate_normal(gmm_means[comp_idx], gmm_covs[comp_idx])
            
            # Clip to [0,1]
            sample = np.clip(sample, 0, 1)
            # Project to sum constraint
            sample = project_to_constraint(sample, target_sum)
            
            c5 = compute_c5(sample, dx)
            candidates.append(sample)
            candidate_c5s.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(candidate_c5s)
        top_candidates = [candidates[i] for i in sorted_indices[:keep_top]]
        top_c5s = [candidate_c5s[i] for i in sorted_indices[:keep_top]]
        
        # Update GMM
        # Update means
        new_means = []
        for i in range(n_components):
            if i < len(top_candidates):
                new_means.append(top_candidates[i])
            else:
                new_means.append(gmm_means[i])
        
        # Update covariances
        new_covs = []
        for i in range(n_components):
            if i < len(top_candidates):
                # Compute covariance from top candidates
                selected = np.array(top_candidates[:min(5, len(top_candidates))])
                if len(selected) > 1:
                    cov = np.cov(selected.T)
                else:
                    cov = gmm_covs[i] * 0.5
                # Add regularization
                cov = cov + np.eye(n_points) * 1e-6
                new_covs.append(cov)
            else:
                new_covs.append(gmm_covs[i] * covariance_decay)
        
        # Update weights (simple proportional to inverse of fit)
        if len(top_c5s) > 0:
            inv_fits = 1.0 / (np.array(top_c5s) + 1e-10)
            total = np.sum(inv_fits)
            if total > 0:
                new_weights = inv_fits / total
            else:
                new_weights = [1.0/n_components]*n_components
        else:
            new_weights = gmm_weights
        
        gmm_means = new_means
        gmm_covs = new_covs
        gmm_weights = new_weights
        
        # Update best
        if top_c5s[0] < best_c5:
            best_c5 = top_c5s[0]
            best_h = top_candidates[0].copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
