import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def update_h(h, i, alpha, target_sum):
    """Update h at coordinate i with step size alpha."""
    h_old = h[i]
    h[i] = np.clip(h[i] + alpha * (1 - 2*h[i]), 0, 1)
    scaling = target_sum / np.sum(h)
    h = h * scaling
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(overlap)
        
        # Compute gradients for each coordinate
        # Gradient of overlap w.r.t. h[i] is:
        # - sum of terms from neighboring positions where we subtract h[i+shift]
        #   and add h[i-shift] for the appropriate shifts
        # Simplified gradient: 2 * h[i] - 1
        # We numerically approximate gradients more carefully
        
        # Instead, update coordinates with highest impact
        # Compute approximate derivative for each point
        grads = np.zeros(n_points)
        
        # For each point, determine what happens if we slightly adjust it
        for i in range(n_points):
            # Approximate gradient via finite difference
            h_temp = h.copy()
            delta = 1e-4
            h_temp[i] += delta
            h_temp = normalize_h(h_temp, target_sum)
            h_temp = np.clip(h_temp, 0, 1)
            overlap_plus = correlate(h_temp, 1 - h_temp, mode="full") * dx
            
            h_temp = h.copy()
            h_temp[i] -= delta
            h_temp = normalize_h(h_temp, target_sum)
            h_temp = np.clip(h_temp, 0, 1)
            overlap_minus = correlate(h_temp, 1 - h_temp, mode="full") * dx
            
            # Finite difference
            grad = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * delta)
            grads[i] = grad
        
        # Select top points (top 10%) by absolute gradient magnitude
        top_indices = np.argsort(np.abs(grads))[::-1][:max(1, n_points // 10)]
        
        # Update selected points
        for i in top_indices:
            # Compute step size to improve gradient
            grad = grads[i]
            
            # Update for minimizing overlap
            # Gradient direction: -grad times unit vector
            # But we constrain 0 <= h[i] <= 1, and sum must remain same
            
            # Direct update with limited move size (line search)
            # Search for best alpha in [0,1]
            best_alpha = 0.0
            best_new_c5 = current_max
            
            # Enumerate various step sizes
            for alpha in [0, 0.02, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0]:
                # Try h[i] + alpha
                old_h = h[i]
                h_new = h[i] + alpha * (1 - 2 * h[i])
                h_new = np.clip(h_new, 0, 1)
                
                # Re-scale to maintain sum
                temp_h = h.copy()
                temp_h[i] = h_new
                scaled_h = normalize_h(temp_h, target_sum)
                scaled_h = np.clip(scaled_h, 0, 1)
                
                new_c5 = compute_c5(scaled_h, dx)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                
            # Apply best alpha found
            h_old = h[i]
            h[i] = np.clip(h[i] + best_alpha * (1 - 2 * h[i]), 0, 1)
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
        
        # Check progress
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if best_c5 <= 0.3807:
            break
            
        if stagnation_count > max_stagnation:
            break
    
    return (best_h.tolist(), best_c5, n_points)
