import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to exactly match target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        scale_factor = target_sum / current_sum
        h = h * scale_factor
        h = np.clip(h, 0, 1)
    return h
