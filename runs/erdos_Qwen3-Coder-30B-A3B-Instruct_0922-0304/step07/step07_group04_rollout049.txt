import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing with refinement
    T_init = 0.1
    T_min = 1e-6
    alpha = 0.95
    max_iter = 1000
    
    T = T_init
    current_h = h.copy()
    current_c5 = best_c5
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while T > T_min and time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Create a neighbor solution
        neighbor_h = current_h.copy()
        # Perturb one element
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        
        # Projection to [0,1]
        neighbor_h = np.clip(neighbor_h, 0, 1)
        
        # Normalize
        neighbor_sum = np.sum(neighbor_h)
        if neighbor_sum > 0:
            neighbor_h = neighbor_h * (target_sum / neighbor_sum)
        
        # Evaluate
        neighbor_c5 = evaluate_overlap(neighbor_h, dx)
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
            no_improvement_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(neighbor_c5 - current_c5) / T):
                current_h = neighbor_h
                current_c5 = neighbor_c5
            no_improvement_count += 1
            
        # Cooling
        T *= alpha
        
        # Early stopping
        if no_improvement_count > max_no_improvement:
            break
    
    # Final local refinement
    if time.time() - start_time < budget_s - 2:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-9})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
                
    # Ensure final sum constraint and bounds
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
