import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = project_feasible(h, target_sum)
    
    # Create initial population with some diversity
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = project_feasible(individual, target_sum)
        population.append(individual)
    
    # PSO parameters
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = population.copy()
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(population_size)]
    pbest = particles.copy()
    pbest_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best initialization
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # PSO iterations
    max_iter = 60
    stagnation_count = 0
    best_c5 = gbest_fitness
    best_h = gbest.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    stagnation_count = 0
                    if gbest_fitness < best_c5:
                        best_c5 = gbest_fitness
                        best_h = gbest.copy()
                    
                    if best_c5 <= 0.3807:
                        break
                        
            else:
                stagnation_count += 1

        # Update parameters
        w *= 0.98
        if stagnation_count > 20:
            # Introduce mutation to escape local optima
            for i in range(population_size):
                if np.random.rand() < 0.1:
                    particles[i] = project_feasible(particles[i] + 
                                                   np.random.normal(0, 0.05, n_points), 
                                                   target_sum)
        
        # Early termination for no improvement
        if stagnation_count > 20:
            break
    
    # Final gradient descent refinement
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Try to improve the best solution found with local optimization
    refined_h = project_feasible(best_h.copy(), target_sum)
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, refined_h, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10})
        if result.success:
            refined_h = project_feasible(result.x, target_sum)
            new_c5 = compute_c5(refined_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = refined_h.copy()
            else:
                break
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
