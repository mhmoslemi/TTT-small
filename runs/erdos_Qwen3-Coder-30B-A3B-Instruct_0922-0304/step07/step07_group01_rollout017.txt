import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Initialize with a blocky step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = project_to_constraints(h, target_sum)
    
    # Initialize variables
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Heuristics for gradient descent with trust region
    max_iter = 1000
    trust_radius = 0.1
    min_trust_radius = 1e-6
    max_trust_radius = 0.5
    trust_update_factor = 2.0
    
    # Try different optimization strategies
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_constraints(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Compute step direction (negative gradient)
        step = -grad
        
        # Apply trust region constraint
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        # Try this step
        h_new = h + step
        h_new = project_to_constraints(h_new, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject step based on improvement
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            if best_c5 <= 0.3807:
                break
            trust_radius = min(max_trust_radius, trust_radius * trust_update_factor)
        else:
            trust_radius = max(min_trust_radius, trust_radius / trust_update_factor)
            
        # Occasionally perturb for diversity
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.02, n_points)
            h = project_to_constraints(h + noise, target_sum)
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final validation
    final_h = project_to_constraints(h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    
    return (best_h.tolist(), best_c5, n_points)
