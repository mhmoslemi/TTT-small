import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    h = normalize_h(h, target_sum)
    
    # Frank-Wolfe algorithm
    max_iter = 5000
    tolerance = 1e-6
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap_val = overlap[max_k] * dx
        
        # Update best if improved
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        if patience_counter > patience:
            break
            
        # Compute gradient for the maximizing shift
        # Gradient of overlap_k at h is 1 - 2*h for i+k positions
        # We need to support this in a full vector form
        
        # Linearization around h
        # Finding s to minimize: s^T * gradient
        # gradient_k[i] = 1 - 2 * h[i+k] for all valid i
        # but when k=0, it's 1 - 2*h, etc.
        # For shift k_max, assuming k_max is from full correlation A = 0,...(n-1) so A = 0:2n-2
        # we need to convert k_max to actual index of shift
        shift = max_k - (n_points - 1)  # k corresponds to shift
        gradient = np.zeros(n_points)
        
        # Gradient entries for h[i] (i from 0 to n_points-1):
        # Count how many times h[i] appears in the i+k position
        for i in range(n_points):
            new_idx = i + shift
            if 0 <= new_idx < n_points:
                gradient[i] = 1 - 2 * h[new_idx]
                
        # Frank-Wolfe step: solve min_s s^T * gradient subject to:
        # sum(s) = sum(h), s[i] in [0,1]
        # The dual variable is just the gradient, but we can solve this as a simple LP
        # s* = argmin_s s^T * gradient subject to sum(s) = sum(h)
        # Set s[i] = 0 if gradient[i] > 0, 1 if gradient[i] < 0, then adjust
        s = np.zeros(n_points)
        for i in range(n_points):
            s[i] = 0 if gradient[i] > 0 else (1 if gradient[i] < 0 else 0.5)
        
        # Adjust s to meet constraint sum(s) = sum(h)
        s = normalize_h(s, target_sum)
        
        # Clip to [0,1]
        s = np.clip(s, 0, 1)
        s = normalize_h(s, target_sum)  # Ensure constraint holds after clipping
        
        # Line search for optimal step size
        alpha = 0.5
        min_alpha = 1e-6
        max_alpha = 1.0
        best_alpha = alpha
        
        def test_alpha(alpha):
            new_h = h + alpha * (s - h)
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            return compute_c5(new_h, dx) 
            
        # Backtracking line search
        while alpha > min_alpha:
            new_c5 = test_alpha(alpha)
            if new_c5 < max_overlap_val - 1e-8:
                best_alpha = alpha
                break
            else:
                alpha /= 2.0
                
        # Apply update
        h = h + best_alpha * (s - h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
