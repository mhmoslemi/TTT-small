import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Main optimization loop
    patience = 0
    max_patience = 10
    max_iterations = 60
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute approximate gradient and Hessian using finite differences
        current_c5 = compute_c5(h, dx)
        
        # Approximate the gradient differently - use finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Local descent using QP - solve constrained quadratic approximation
        # Form QP: minimize 0.5 * d^T * H * d + grad^T * d subject to sum(d) = 0, 0 <= h+d <= 1
        # Simplified: ignore second order term for now - gradient descent approx.
        step_size = 0.01
        d = -step_size * grad
        
        # Ensure sum(d) = 0
        d_sum = np.sum(d)
        d = d - d_sum / n_points  # Remove mean to preserve sum
        
        # Compute new h
        h_new = h + d
        h_new = project_to_feasible(h_new, target_sum)
        
        # Line search
        alpha = 1.0
        max_alpha = 10.0
        min_alpha = 1e-6
        while alpha > min_alpha:
            h_candidate = h + alpha * d
            h_candidate = project_to_feasible(h_candidate, target_sum)
            c5_candidate = compute_c5(h_candidate, dx)
            if c5_candidate <= current_c5 * 0.999:
                h = h_candidate
                break
            alpha *= 0.5
        
        # Update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience = 0
        else:
            patience += 1
            
        if patience >= max_patience:
            break
    
    # Final refinement using scipy
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Use the best solution as starting point
    h = best_h.copy()
    result = minimize(objective, h, method='SLSQP', 
                     bounds=bounds, constraints=constraint,
                     options={'maxiter': 50})
    
    if result.success:
        h = result.x
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
        final_c5 = compute_c5(h, dx)
    else:
        final_c5 = best_c5
    
    # Final validation
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
