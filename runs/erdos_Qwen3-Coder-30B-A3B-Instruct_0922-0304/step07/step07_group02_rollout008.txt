import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Add the difference evenly to all elements
    h = h + diff / len(h)
    # Clamp to [0,1] and renormalize
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points) * 0.5  # Uniform initialization
        h = project_to_constraint(h, target_sum)
        
    # PGD parameters
    alpha = 0.1  # learning rate
    beta = 0.9   # momentum
    max_iter = 3000
    last_improvement_iter = 0
    patience = 100
    
    # Initialize momentum
    velocity = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute overlap values for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # Find the shift that maximizes overlap
        k_max = np.argmax(overlap)
        # The gradient is (1 - h) shifted by k_max
        # Since we are using correlate, we compute the shifted version
        # of 1-h to match the convention (temporarily)
        shifted = np.zeros_like(h)
        shifted[k_max:] = 1 - h[:-k_max] if k_max > 0 else 1 - h
        gradient = shifted
        
        # Update with momentum
        velocity = beta * velocity - alpha * gradient
        h = h + velocity
        
        # Clamp h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Project to correct sum
        h = project_to_constraint(h, target_sum)
        
        # Compute C5
        current_c5 = compute_c5(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement_iter = iterations
        
        # Adapt learning rate
        if iterations - last_improvement_iter > patience:
            alpha *= 0.95
            alpha = max(alpha, 1e-5)
            last_improvement_iter = iterations
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
