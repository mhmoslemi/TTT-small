import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5_exact(h, dx):
    """Compute exact C5 value."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute approximate gradient of C5."""
    # Compute overlap values for all shifts
    overlap = correlate(h, 1 - h, mode="full")
    
    # We'll compute the gradient for each position
    grad = np.zeros(n_points)
    
    # Compute gradient components
    for i in range(n_points):
        grad_comp = 0.0
        for k in range(2 * n_points - 1):
            # k corresponds to the shift, overlap_k is at index k
            if k < n_points:
                # Forward shift case
                j = i + k
                if j < n_points:
                    grad_comp += (overlap[k] / dx) * (1 - 2 * h[j])
            else:
                # Backward shift case
                j = i - (2 * n_points - 1 - k)
                if j >= 0:
                    grad_comp += (overlap[k] / dx) * (1 - 2 * h[j])
        
        grad[i] = grad_comp
    
    return grad

def objective_and_gradient(h, dx, n_points, target_sum):
    """Compute objective and gradient."""
    # Normalize h to sum to target_sum
    h = normalize_h(h, target_sum)
    
    # Compute C5
    c5 = compute_c5_exact(h, dx)
    
    # Gradient computation is approximate
    grad = compute_gradient(h, dx, n_points)
    
    # Normalize gradient to prevent wild steps
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    
    # Return negative gradient since we minimize
    return c5, -grad

    best_c5 = compute_c5_exact(best_h, dx)
    
    # Optimization parameters
    lr = 0.01
    decay_rate = 0.999
    patience = 0
    max_patience = 1000
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Each iteration we compute gradient and take a step
        c5, grad = objective_and_gradient(h, dx, n_points, target_sum)
        
        # Update h
        h = h + lr * grad
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Update learning rate according to exponential decay
        lr *= decay_rate
        
        # Early termination
        if best_c5 <= 0.3807:
