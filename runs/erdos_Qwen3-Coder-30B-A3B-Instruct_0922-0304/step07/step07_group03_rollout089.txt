import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

        # Scale to target sum
        target_sum = n_points / 2
        h = project_to_constraints(h, target_sum)

    dx = 2.0 / len(h)
    max_iter = 20000
    alpha_0 = 0.05
    tau = 1000
    target_sum = len(h) / 2
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Compute gradient update using surrogate objective
        c = correlate(h, 1 - h, mode="full")
        # For surrogate: ∇(||h ⋆ (1-h)||^2) = 2 * (h ⋆ (1-h)) * (2/n)
        # But we need just the overlapped values that are maximized
        # simple average gradient approximation of the max
        grad_h = np.zeros_like(h)
        # Approximate gradient from correlation values
        # Approximation: use central differences or simple histogram
        # Here we use direct correlation with known positions
        for i in range(len(h)):

                k_index = i - j + len(h) - 1
                if 0 <= k_index < 2 * len(h) - 1:
                    # Gradient w.r.t. h[i] due to (1 - h[j]) component
                    # Only the correlation at index k_index affects this term
                    grad_h[i] += 2 * (1 - h[j]) * c[k_index]
        
        # Projected gradient descent
        alpha = alpha_0 / (1 + iter_num / tau)
        h = h - alpha * grad_h/len(h)  # Normalize gradient

        # Evaluate

        if current_c5 < best_c5:
