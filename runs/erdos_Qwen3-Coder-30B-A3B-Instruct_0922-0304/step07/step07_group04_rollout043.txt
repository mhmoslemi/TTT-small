import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use iterative Lagrangian method
    max_iter = 1000
    alpha = 0.01
    gamma = 0.1
    tol = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current overlap and find maximizing k
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        max_overlap = overlap_full[k_max] * dx
        
        # Update h using gradient descent on the max overlap term
        # For shift k_max, the gradient of h_i is (1 - h_{i-k_max}) 
        h_grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            h_grad[i] = 1 - h[j]
        
        # Update h with gradient descent
        h_new = h - alpha * h_grad
        
        # Project onto [0,1] constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        current_sum_new = np.sum(h_new)
        if current_sum_new > 0:
            h_new = h_new * (target_sum / current_sum_new)
        
        # Check improvement
        new_c5 = evaluate_overlap(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        h = h_new
    
    # Final gradient refinement with SLSQP
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
