import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Use local refinement with simulated annealing approach
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Start with initial solution
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Simulated annealing parameters
    T_start = 1.0
    T_min = 1e-6
    alpha = 0.995
    max_iter = 10000
    
    T = T_start
    iter_count = 0
    
    while T > T_min and time.time() - start_time < budget_s - 10:
        # Generate neighbor solution
        neighbor_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        # Perturb one element
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        neighbor_h = np.clip(neighbor_h, 0, 1)
        
        # Normalize to constraint
        neighbor_sum = np.sum(neighbor_h)
        if neighbor_sum > 0:
            neighbor_h = neighbor_h * (target_sum / neighbor_sum)
        
        neighbor_c5 = objective(neighbor_h)
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Accept with probability based on temperature
            if np.random.random() < np.exp(-(neighbor_c5 - current_c5) / T):
                current_h = neighbor_h
                current_c5 = neighbor_c5
        
        T *= alpha
        iter_count += 1
        
        if iter_count % 100 == 0:
            # Periodically check if time is running out
            if time.time() - start_time > budget_s - 20:
                break

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
