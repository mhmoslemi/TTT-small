import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategies parameters
    pop_size = 20
    elite_count = pop_size // 2
    mutation_strength = 0.1
    max_iter = 10000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        noise = np.random.normal(0, mutation_strength, n_points)
        new_h = h + noise
        new_h = project_to_constraints(new_h, target_sum)
        population.append(new_h)
    
    # Evaluate initial population
    fitnesses = [compute_c5(p, dx) for p in population]
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best
        if fitnesses[0] < best_fitness:
            best_h = population[0].copy()
            best_fitness = fitnesses[0]
            
        # If target achieved, terminate early
        if best_fitness <= 0.3807:
            break
            
        # Create offspring
        offspring = []
        elite = population[:elite_count]
        
        # Keep elite
        offspring.extend(elite)
        
        # Generate new individuals by mutating elite
        for i in range(pop_size - elite_count):
            parent_idx = np.random.randint(0, elite_count)
            parent = elite[parent_idx]
            
            # Adaptive mutation
            current_time = time.time() - start_time
            adaptive_strength = mutation_strength * (1.0 - current_time / budget_s)
            adaptive_strength = max(adaptive_strength, 0.01)
            
            noise = np.random.normal(0, adaptive_strength, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            offspring.append(child)
            
        # Replace population
        population = offspring
        
        # Evaluate new population
        fitnesses = [compute_c5(p, dx) for p in population]
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
