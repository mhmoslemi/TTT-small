```python
import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to target sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Forward pass
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Compute gradient
        g = np.zeros(n_points)
        
        # For the maximum overlap term
        # d/dw_i of max_overlap term
        # Use finite differences for each term to compute gradient
        eps = 1e-6
        for i in range(n_points):
            # Construct perturbed h
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus / np.sum(h_plus) * target_sum
            h_plus = np.clip(h_plus, 1e-8, 1 - 1e-8)
            
            # Calculate overlap
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_overlap_plus = overlap_plus[max_k]
            
            # Finite difference gradient
            g[i] = (max_overlap_plus - max_overlap) / eps * dx
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Ensure feasibility of h
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-8))
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        iteration += 1
        
        # Adjust learning rate
        if iteration % 1000 == 0 and lr > 1e-5:
            lr *= 0.999
    
    return (best_h.tolist(), best_c5, n_points)
```
