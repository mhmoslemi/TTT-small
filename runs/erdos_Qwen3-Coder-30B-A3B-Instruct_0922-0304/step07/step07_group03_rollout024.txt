import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust proportionally to maintain feasibility
    if current_sum > target_sum:
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Distribute deficit
        deficit = target_sum - current_sum
        # Distribute evenly at first
        h += deficit / len(h)
        # Clip and re-adjust to maintain sum exactly
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Adjust fractionally
            h += diff / len(h)
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Structured initial pattern: sine wave
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # AM parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    decay_tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute overlaps for current h
        overlap_corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap_corr)
        max_indices = np.where(overlap_corr >= 0.99 * max_overlap)[0]
        
        # Gradient step
        # Approximate gradient as derivative of overlap w.r.t. h
        # For k that maximizes overlap, derive gradient from 1-h shifted by k
        grad = np.zeros_like(h)
        for k in max_indices:
            # For full correlation mode, shift of k indicates h[i] * (1-h[i+k])
            # We don't directly compute it, but we approximate gradient as 
            # gradient of sum of (1-h) * h from suitable k
            # This is a simplified approximation of subgradient
            if k < len(h):
                # Reverse correlation - approximating gradient contribution
                for i in range(len(h)):
                    if i + k < len(h):
                        # Influence of h[i] on overlap at k
                        grad[i] += 1 - h[i+k]
        
        # Normalize gradient (small safeguard to prevent overflow)
        if np.linalg.norm(grad) > 1e-10:
            grad = grad / np.linalg.norm(grad)
        
        # Update h with gradient step and perturb
        h_candidate = h - alpha * grad
        
        # Add random perturbation
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to feasible set
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate new C5
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_candidate
            best_c5 = new_c5
            # Reduce learning rate if improvement
            alpha *= 0.99
        else:
            # Reduce learning rate even if no improvement
            alpha *= 0.995
            
        # Apply learning rate decay
        alpha = max(alpha, 1e-5)
        epsilon = max(epsilon * 0.999, 1e-5)
    
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
