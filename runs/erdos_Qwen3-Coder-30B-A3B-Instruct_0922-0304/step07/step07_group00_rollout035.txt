import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def create_basis_functions(n_points, n_basis=20):
    """Create a set of basis functions for representing h(x)."""
    basis = []
    # Create pulse basis functions
    for i in range(n_basis):
        # Create a step function with random width and position
        width = np.random.uniform(0.1, 0.5)
        start = np.random.uniform(0, 2 - width)
        x = np.linspace(0, 2, n_points)
        phi = np.zeros(n_points)
        idx = np.logical_and(x >= start, x <= start + width)
        phi[idx] = 1.0
        basis.append(phi)
    
    return np.array(basis)

def compute_overlap_integrals(basis):
    """Precompute overlap integrals A_{ij}^k."""
    n_basis = len(basis)
    n_points = len(basis[0])
    # Full overlap matrix for all shifts
    A = np.zeros((n_basis, n_basis, 2 * n_points - 1))
    for i in range(n_basis):
        for j in range(n_basis):
            A[i, j] = correlate(basis[i], basis[j], mode="full")
    return A

def evaluate_basis_solution(lambdas, basis, A, dx):
    """Evaluate the C5 for a given lambda combination of basis functions."""
    # Sum over all shifts k
    n_points = len(basis[0])
    max_c5 = 0.0
    for k in range(2 * n_points - 1):
        # Compute sum of lambdas^2 * (A[i,j,k] - A[i,j,k] * A[j,j,k])
        c5_k = 0.0
        for i in range(len(lambdas)):
            for j in range(len(lambdas)):
                if i == j:
                    c5_k += lambdas[i] * lambdas[j] * A[i, j, k] * (1 - A[i, j, k])
                else:
                    c5_k += lambdas[i] * lambdas[j] * A[i, j, k]
        max_c5 = max(max_c5, c5_k)
    
    return max_c5 * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # If initial_h exists, create an initial reasonable lambda set
    # Otherwise, start with a random basis
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    
    # Create basis functions
    basis = create_basis_functions(n_points, 20)
    A = compute_overlap_integrals(basis)
    
    # Normalize basis functions to have integration 1/(n_points/2)
    for i in range(len(basis)):
        basis[i] *= (target_sum / np.sum(basis[i]))
        
    best_lambdas = None
    best_c5 = float('inf')
    iteration = 0
    
    # Use a limited number of iterations with early stopping
    max_iterations = 1000
    patience = 50
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 5:
        # Solve for optimal lambdas subject to the convex relaxation
        # We optimize C5 = max_k sum_{i,j} lambda_i lambda_j * (A_{ij}^k - A_{ij}^k * A_{jj}^k)
        # This is 1 binary quadratic program, we linearize it or use convex relaxation
        
        # Alternative: directly optimize using CVXPY for improvement
        lambdas = Variable(len(basis))
        
        # Objective: minimize max over k of the sliding sum
        objectives = []
        for k in range(2 * n_points - 1):
            obj = 0
            for i in range(len(basis)):
                for j in range(len(basis)):
                    if i == j:
                        obj += lambdas[i] * lambdas[j] * A[i, j, k] * (1 - A[i, j, k])
                    else:
                        obj += lambdas[i] * lambdas[j] * A[i, j, k]
            objectives.append(obj)
            
        # Maximize over all k's objective terms
        max_term = cp.max(objectives)
        
        # Constraints: sum of lambdas = 1 and lambdas >= 0
        constraints = [cp.sum(lambdas) == 1, lambdas >= 0]
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(max_term), constraints)
        
        try:
            prob.solve(solver=SCS, verbose=False)
        except:
            # Fallback to a simplified approach
            prob = cp.Problem(cp.Minimize(cp.max(objectives)), [cp.sum(lambdas) == 1, lambdas >= 0])
            try:
                prob.solve(solver=ECOS, verbose=False)
            except:
                # If no solver works, pick a random solution
                lambdas_val = np.random.rand(len(basis))
                lambdas_val /= np.sum(lambdas_val)
                lambdas_val = np.clip(lambdas_val, 0, 1)
                lambdas_val /= np.sum(lambdas_val)
            else:
                lambdas_val = lambdas.value
        else:
            try:
                lambdas_val = lambdas.value
            except:
                lambdas_val = np.random.rand(len(basis))
                lambdas_val /= np.sum(lambdas_val)
                
        # Ensure valid lambda values
        lambdas_val = np.clip(lambdas_val, 0, 1)
        lambdas_val /= np.sum(lambdas_val)
        
        if np.isnan(np.sum(lambdas_val)):
            import pdb
            pdb.set_trace()
            
        # Evaluate
        c5 = evaluate_basis_solution(lambdas_val, basis, A, dx)
        
        # Convert to actual h function
        h = basis.T @ lambdas_val
        h = np.clip(h, 0, 1)
        # Normalize to correct integral
        h = h * (target_sum / np.sum(h, axis=0))

        if c5 < best_c5:
            best_c5 = c5
            best_lambdas = lambdas_val.copy()
            # Also compute the actual C5 for this h
            actual_c5 = compute_c5(h, dx)
            best_c5 = actual_c5
            
        if best_c5 <= 0.3807:
            break
                
        # Add new basis function if heavily failed on some k
        # This is a simple heuristic - should be more nuanced in full version
        iteration += 1

    # Convert back to original representation
    if best_lambdas is not None:
        h_final = basis.T @ best_lambdas
        h_final = np.clip(h_final, 0, 1)
        # Renormalize to exact integral constraint
        h_final = h_final * (target_sum / np.sum(h_final))
        actual_c5 = compute_c5(h_final, dx)
        return (h_final.tolist(), actual_c5, n_points)
    
    # Fallback if optimization failed
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    c5 = compute_c5(h, dx)
    return (h.tolist(), c5, n_points)
