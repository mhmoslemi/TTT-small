import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and bounds [0,1]."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute approximate gradient of C5 with respect to h."""
    # This is a simplified approximation - in practice could use more sophisticated methods
    n = len(h)
    # For now, we'll use finite differences approximating the gradient
    # This is not a true gradient but a heuristic approach
    grad = np.zeros(n)
    eps = 1e-6
    base_c5 = compute_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, np.sum(h))
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create a good starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a triangular pulse centered at 1.0, normalized
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if abs(i - mid) < n_points // 4:
                h[i] = 1.0 - 2.0 * abs(i - mid) / (n_points // 4)
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    patience = 0
    max_patience = 10
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Try different local search strategies
        new_h = h.copy()
        step_size = 0.01
        
        # Try direct gradient descent with small step sizes to avoid overshooting
        gradient = compute_gradient(h, dx)
        new_h = h - step_size * gradient
        
        # Project to feasible region
        new_h = project_to_feasible(new_h, target_sum)
        
        # Adjust step size using line search
        def line_search_obj(alpha):
            trial_h = h - alpha * gradient
            trial_h = project_to_feasible(trial_h, target_sum)
            return compute_c5(trial_h, dx)
            
        # Find the best step size via line search
        try:
            result = minimize_scalar(line_search_obj, bounds=(0, 1), method='bounded')
            best_alpha = result.x
            trial_h = h - best_alpha * gradient
            trial_h = project_to_feasible(trial_h, target_sum)
            trial_c5 = compute_c5(trial_h, dx)
            
            if trial_c5 < current_c5:
                h = trial_h
                current_c5 = trial_c5
                patience = 0  # Reset patience
            else:
                patience += 1
        except:
            patience += 1
            
        # Update the best solution found
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
        if iteration >= 300:
            break
                    
    # Final optimization with scipy
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, best_h, method='SLSQP',

                         options={'maxiter': 100})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = project_to_feasible(best_h, target_sum)
            best_c5 = compute_c5(best_h, dx)
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
