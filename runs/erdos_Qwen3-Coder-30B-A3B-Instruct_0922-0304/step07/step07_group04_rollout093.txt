import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Initialize Lagrange multipliers
    mu = 1.0
    lambda_plus = np.zeros(n_points)
    lambda_minus = np.zeros(n_points)
    
    # Identify current maximum shift
    def get_max_shift(h_vec):
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        return np.argmax(overlap)
    
    # Lagrangian optimization with projected gradient descent
    max_iter = 500
    alpha = 0.01
    gamma = 0.1
    tol = 1e-6
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find current maximum shift
        max_k = get_max_shift(h)
        
        # Compute the gradient of the support function
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = max_k
            shifted_idx = (j - k) % n_points
            grad[j] = 1 - h[shifted_idx] - mu + lambda_plus[j] - lambda_minus[j]
        
        # Update h with projected gradient descent
        h_new = h - alpha * grad
        
        # Project to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (target_sum / h_sum)
        
        # Update Lagrange multipliers
        mu = max(0, mu + gamma * (np.sum(h_new) - target_sum))
        lambda_plus = np.maximum(0, lambda_plus + gamma * (h_new - 1))
        lambda_minus = np.maximum(0, lambda_minus + gamma * (-h_new))
        
        h = h_new
        
        # Evaluate
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final refinement with scipy
    if time.time() - start_time < budget_s - 10:
        from scipy.optimize import minimize
        
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Return final solution
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
