import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    tournament_size = max(5, population_size // 20)
    max_generations = 1000
    mutation_rate = 0.1
    blending_min, blending_max = 0.2, 0.8
    generation_without_improvement = 0
    max_no_improvement = 50
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Add initial solution
    population.append(h.copy())
    fitnesses.append(best_c5)
    
    # Fill the rest with random solutions
    for i in range(population_size - 1):
        individual = np.random.rand(n_points)
        individual = np.clip(individual, 0, 1)
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (target_sum / current_sum)
        population.append(individual)
        fitnesses.append(evaluate_overlap(individual, dx))
    
    # Main genetic algorithm loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitnesses[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitnesses[0]
            generation_without_improvement = 0
        else:
            generation_without_improvement += 1
            
        if generation_without_improvement > max_no_improvement:
            # Reset mutation rate if no improvement
            mutation_rate = min(0.5, mutation_rate * 1.1)
            generation_without_improvement = 0
        
        # Create new population
        new_population = [sorted_population[0].copy()]  # Elitism
        
        # Generate offspring through tournament selection + crossover + mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.choice(tournament_size)]
            parent2_idx = sorted_indices[np.random.choice(tournament_size)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Blend crossover
            alpha = np.random.uniform(blending_min, blending_max)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            
            # Normalize sum
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < 0.1
                noise = np.random.normal(0, 0.05, n_points)
                child += noise * mask
                child = np.clip(child, 0, 1)
                
                # Re-normalize
                child_sum = np.sum(child)
                if child_sum > 0:
                    child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitnesses = [evaluate_overlap(ind, dx) for ind in population]
    
    # Local refinement with gradient descent
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Try to improve with SLSQP on the best solution
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
