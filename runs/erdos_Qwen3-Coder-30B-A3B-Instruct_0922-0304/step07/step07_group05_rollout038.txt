import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Add 40% varied particles from initial
        for _ in range(int(0.4 * population_size)):
            perturbed = h + np.random.normal(0, 0.01, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = normalize_h(perturbed, target_sum)
            particles.append(perturbed)
        # Fill remaining with random
        for _ in range(population_size - len(particles)):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand)
    else:
        # Default initialization
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            particles.append(h)
    
    # PSO parameters
    inertia = 0.8
    cog_coeff = 1.5
    soc_coeff = 1.5
    
    # Initialize personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_bests[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    max_iter = 60
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i, p in enumerate(particles):
            # Random factors for PSO update
            r1, r2 = np.random.rand(2)
            
            # Update velocity (simplified)
            new_velocity = (inertia * (p - personal_bests[i]) + 
                           cog_coeff * r1 * (personal_bests[i] - p) + 
                           soc_coeff * r2 * (global_best - p))
            
            # Update position
            new_position = p + new_velocity
            
            # Apply constraints
            new_position = np.clip(new_position, 0, 1)
            new_position = normalize_h(new_position, target_sum)
            
            # Update particle
            particles[i] = new_position
            
            # Update personal and global bests
            fitness = compute_c5(new_position, dx)
            if fitness < personal_best_fitness[i]:
                personal_bests[i] = new_position.copy()
                personal_best_fitness[i] = fitness
                if fitness < global_best_fitness:
                    global_best = new_position.copy()
                    global_best_fitness = fitness
        
        # Update inertia
        inertia *= 0.98
        
        # Early termination
        if global_best_fitness <= 0.3807:
            break
    
    # Final refinement using local optimization
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Refine the best particle found
    best_h = global_best.copy()
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h, method='SLSQP',
                         bounds=bounds, constraints=constraint,
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
            
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
