import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to correct sum
    h = project_to_constraints(h, target_sum)
    
    # Stage 1: Convex relaxation to get initial solution
    # Formulate as convex optimization:
    # minimize C
    # subject to 0 <= h[i] <= 1
    #          sum(h) = target_sum
    #          h * (1 - h_shifted) <= C for all lags
    
    # Let's do a basic approach with gradient descent refinement of a good starting point
    
    # Stage 2: Gradient Descent refinement
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Create gradient descent using projected gradient steps
    max_iter = 5000
    learning_rate = 0.01
    decay_rate = 0.995  # Exponential decay for learning rate
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        c5_base = compute_c5(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_base) / eps
        
        # Gradient ascent (we want to minimize)
        new_h = best_h - learning_rate * grad
        new_h = project_to_constraints(new_h, target_sum)
        
        # Update learning rate
        learning_rate *= decay_rate
        
        new_c5 = compute_c5(new_h, dx)

            best_c5 = new_c5
            
            # Early stopping if target achieved
            if best_c5 <= 0.3807:
                break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
