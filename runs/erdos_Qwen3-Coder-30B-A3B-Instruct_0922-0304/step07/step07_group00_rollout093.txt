import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to sum = target_sum
    h = h / np.sum(h) * target_sum
    
    # Transform to logit space for optimization
    w = np.log(h / (1 - h + 1e-8))
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current C5 and gradients
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient
        g = np.zeros(n_points)
        # Derivative of sigmoid is sigmoid * (1 - sigmoid)
        s = sigmoid(w)
        ds = s * (1 - s)
        
        # For the maximum overlap term, compute contributions from all shift positions
        # This is simplified - in practice, we compute the gradient with respect to the 
        # maximum overlap term
        if max_k < len(h):
            g += ds * (1 - 2 * h)
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        
        # Update weights
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Transform back to h
        h = sigmoid(w)
        
        # Normalize to correct sum
        h = h / np.sum(h) * target_sum
        
        # Ensure feasibility
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Transform back to logit
        w = np.log(h / (1 - h + 1e-8))
        
        # Evaluate and check for improvement
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
