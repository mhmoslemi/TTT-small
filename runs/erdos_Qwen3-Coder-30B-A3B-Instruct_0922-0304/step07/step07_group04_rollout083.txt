import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm Parameters
    population_size = 100
    max_generations = 1000
    tournament_size = population_size // 10
    mutation_rate = 0.1
    blending_min, blending_max = 0.2, 0.8
    
    # Create initial population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand_sum = np.sum(h_rand)
        if h_rand_sum > 0:
            h_rand = h_rand * (target_sum / h_rand_sum)
        population.append(h_rand)
    
    def fitness(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    # Main GA loop
    last_improvement_gen = 0
    current_gen = 0
    
    while time.time() - start_time < budget_s - 5 and current_gen < max_generations:
        current_gen += 1
        
        # Evaluate population
        fitness_scores = [fitness(h) for h in population]
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_c5 = fitness_scores[best_idx]
            best_h = population[best_idx].copy()
            last_improvement_gen = current_gen
            
        # Early termination if no improvement in last 200 gens
        if current_gen - last_improvement_gen > 200:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            # Crossover
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Blending crossover
            alpha = np.random.uniform(blending_min, blending_max)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                mask = np.random.binomial(1, 0.1, n_points)
                mutation_strength = 0.05 * (1 - current_gen / max_generations)
                child += np.random.normal(0, mutation_strength, n_points) * mask
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (target_sum / child_sum)
            
            new_population.append(child)
        
        population = new_population
        # Adaptive mutation rate
        if current_gen % 50 == 0:
            mutation_rate = max(0.01, mutation_rate * 0.95)
    
    # Local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
