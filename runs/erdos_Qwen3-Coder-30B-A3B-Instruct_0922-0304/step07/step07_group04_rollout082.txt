import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 1000
    tournament_size = population_size // 10
    mutation_rate = 0.1
    blend_min, blend_max = 0.2, 0.8
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        # Random perturbation of the initial solution
        mut_h = h + np.random.normal(0, 0.05, n_points)
        mut_h = np.clip(mut_h, 0, 1)
        mut_sum = np.sum(mut_h)
        if mut_sum > 0:
            mut_h = mut_h * (target_sum / mut_sum)
        population.append(mut_h)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fitness.append(evaluate_overlap(ind, dx))
    
    # Keep track of best solution
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # GA loop
    stagnation_count = 0
    stagnation_limit = 50
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        new_population = []
        for _ in range(population_size // 2):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        for i in range(0, len(new_population), 2):
            if i + 1 < len(new_population):
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                
                # Blend crossover with adjustable factor
                alpha = np.random.uniform(blend_min, blend_max)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                if np.random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child1 += noise
                    child1 = np.clip(child1, 0, 1)
                    child_sum = np.sum(child1)
                    if child_sum > 0:
                        child1 = child1 * (target_sum / child_sum)
                
                if np.random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child2 += noise
                    child2 = np.clip(child2, 0, 1)
                    child_sum = np.sum(child2)
                    if child_sum > 0:
                        child2 = child2 * (target_sum / child_sum)
                
                new_population.append(child1)
                new_population.append(child2)
                
        # Keep population size
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = []
        for ind in population:
            fit = evaluate_overlap(ind, dx)
            fitness.append(fit)
            
        # Update best
        best_idx_new = np.argmin(fitness)
        if fitness[best_idx_new] < best_c5:
            best_h = population[best_idx_new].copy()
            best_c5 = fitness[best_idx_new]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive mutation rate
        if stagnation_count > 20:
            mutation_rate = min(0.5, mutation_rate * 1.1)
            
        # Early stopping if no improvement
        if stagnation_count > stagnation_limit:
            break
    
    # Local refinement
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 200, 'ftol': 1e-8})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 0:
            refined_h = refined_h * (target_sum / refined_sum)
        final_c5 = evaluate_overlap(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h
            best_c5 = final_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
