import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategies parameters
    population_size = 20
    elite_size = population_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(np.array(initial_h_values))
        population[0] = np.clip(population[0], 0, 1)
        population[0] = project_to_constraints(population[0], target_sum)
    else:
        population.append(h.copy())
    
    # Add diverse initial individuals
    for _ in range(population_size - 1):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = project_to_constraints(individual, target_sum)
        population.append(individual)
    
    # Evaluate initial population
    fitnesses = []
    for ind in population:
        fitness = compute_c5(ind, dx)
        fitnesses.append(fitness)
    
    best_idx = np.argmin(fitnesses)
    best_individual = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Evolution loop
    max_iter = 20000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        fitnesses = [fitnesses[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring via mutation
        offspring = []
        for ind in elite:
            # Gaussian mutation
            noise = np.random.normal(0, mutation_rate, n_points)
            mutated = ind + noise
            mutated = project_to_constraints(mutated, target_sum)
            offspring.append(mutated)
        
        # Replace worst individuals with offspring
        for i in range(len(offspring)):
            if i < len(population) - elite_size:
                population[i + elite_size] = offspring[i]
                fitnesses[i + elite_size] = compute_c5(offspring[i], dx)
        
        # Update best
        for i in range(len(population)):
            if fitnesses[i] < best_fitness:
                best_fitness = fitnesses[i]
                best_individual = population[i].copy()
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
