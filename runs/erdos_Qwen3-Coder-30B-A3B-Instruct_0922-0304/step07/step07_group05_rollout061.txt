import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Generate 40% of particles as small random variations of initial
        for _ in range(int(0.4 * population_size)):
            variation = h + np.random.normal(0, 0.01, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
        # Fill remaining with random feasible solutions
        for _ in range(population_size - len(particles)):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles.append(h)
    else:
        # Initialize completely random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles.append(h)
    
    # Initialize personal bests and global best
    personal_bests = [p.copy() for p in particles]
    personal_fitnesses = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_fitnesses)
    global_best = personal_bests[global_best_idx].copy()
    global_fitness = personal_fitnesses[global_best_idx]
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1, c2 = 1.5, 1.5
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    max_iters = 60
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** (iteration // 10))
        
        for i in range(population_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_fitnesses[i]:
                personal_fitnesses[i] = fitness
                personal_bests[i] = particles[i].copy()
                
                # Update global best if necessary
                if fitness < global_fitness:
                    global_fitness = fitness
                    global_best = particles[i].copy()
        
        # Early termination check
        if global_fitness <= 0.3807:
            break
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    # Take the global best and refine it
    best_h = global_best.copy()
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
