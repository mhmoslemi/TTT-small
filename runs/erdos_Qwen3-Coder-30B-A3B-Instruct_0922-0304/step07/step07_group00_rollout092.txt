import numpy as np
from scipy.signal import correlate
import time

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    # Normalize to have the required sum
    h = h / np.sum(h) * target_sum
    
    # Transform to unconstrained space using logit
    w = np.log(h / (1 - h + 1e-12))
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Forward pass
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        
        # Update w for numerical stability
        w = np.log(h / (1 - h + 1e-12))
        
        # Compute gradient (simplified version)
        # Overlap computation
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Simple gradient approximation
        g = np.zeros(n_points)
        # The gradient is approximated by finite differences,
        # using the fact that we want to minimize max overlap
        # For now just use a simple gradient that pushes toward
        # a more even distribution to decrease maximum overlap
        
        # Populating the gradient to direct towards reducing maximum overlap
        # This is a simplification but works reasonably in practice
        # We estimate the contribution to the Jacobian w.r.t overlaps
        for i in range(n_points):
            contrib = 0.0
            # Look at how changing h[i] affects the maximum overlap
            # This is a simplified approximation
            if i + k_max < n_points:
                contrib = 1.0 - 2 * h[i]
            g[i] = contrib
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        
        # Adam update step
        w -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Re-project back to feasible region using sigmoid
        h = sigmoid(w)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 1e-8, 1 - 1e-8)
        w = np.log(h / (1 - h + 1e-12))
        
        # Evaluate C5 value
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
