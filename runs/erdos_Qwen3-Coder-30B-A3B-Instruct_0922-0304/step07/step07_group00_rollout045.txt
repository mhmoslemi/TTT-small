import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max):
    """Compute the gradient of overlap_k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For overlap_k_max = sum h[i] * (1 - h[i + k_max])
    # The gradient is (1 - 2 * h[i + k_max]) w.r.t. h[i] 
    # but only for i such that i + k_max < n
    for i in range(n - k_max):
        grad[i] += (1 - 2 * h[i + k_max])
    return grad

def solve_lp_direction(grad, n):
    """Solve the linear program to find direction s that minimizes grad^T s s.t. s in feasible set."""
    # We want to minimize grad^T s subject to:
    # s[i] in [0,1], sum(s) = 0, and we should make it feasible later
    # since s = h + alpha * (s - h), and sum(s) = sum(h) = target_sum
    # So we want sum(s) = target_sum, not 0.
    # Let's adjust to: sum(s) = target_sum, s[i] in [0,1]
    # The linear program (simplified) would seek direction to move
    # within constraints using `grad^T s` goal.
    #
    # To get s that minimizes grad^T s
    
    # Without explicit LP solver, implement a greedy approach infeasible but hinting:
    # Sort indices by gradient and pick minimally contributing ones
    idxs = np.argsort(grad)
    s = np.zeros(n)
    # All s[i] = 0 for now, we'll make our direction properly later
    # A proper way would be to do a Simplex step if using CVXPY or scipy.optimize
    # But here, we'll just give this simple approach
    # Force move toward decreasing the gradient value
    
    # In full/simplex form, minimization with constraints: 
    # Argmin s: grad^T s, s[i] in [0,1], sum(s) = target_sum
    # This is an LP
    # Solving this exactly takes a while; so we make a heuristic
    
    # Heuristic: we want s in [0,1] that minimizes grad.T s. 
    # A simple solution that's slightly good is to just set s = clip(h - lambda*grad, 0, 1)
    # where we let lambda be such that sum(s) = target_sum.
    # This is just a rough approximation to LP direction find.
    
    # We'll try a projection method to find direction s maximizing negative grad.T s
    # s = h - t*grad such that sum(s) = target_sum 
    # So: sum(h) - t*sum(grad) = target_sum
    # t = (sum(h) - target_sum) / sum(grad) 
    # But if sum(grad) == 0, we need another approach
    
    t = 0.5  # Just a test direction  
    s = h - t * grad
    s = np.clip(s, 0, 1)
    s = normalize_h(s, len(s) / 2)
    return s

def backtracking_line_search(h, direction, dx, initial_step=0.5, decrease_factor=0.5):
    """Perform simple backtracking line search."""
    alpha = initial_step
    current_c5 = compute_c5(h, dx)
    while alpha >= 1e-6:
        trial = h + alpha * (direction - h)
        trial = np.clip(trial, 0, 1)
        trial = normalize_h(trial, len(trial) / 2)
        trial_c5 = compute_c5(trial, dx)
        if trial_c5 <= current_c5:
            return alpha
        alpha *= decrease_factor
    return 0.0

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Iteration counter for early stopping
    no_improvement_count = 0
    last_best_c5 = best_c5 + 1.0
    
    # Iteration loop
    for iteration in range(5000):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute overlaps and find the most constraining shift
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient of the overlap at k_max
        grad = compute_gradients(h, k_max)
        
        # Find direction s by simple gradient-based step
        # Standard Frank-Wolfe heuristic:
        s = h.copy() 
        s = h - 0.05 * grad  # A simple gradient-based move
        s = np.clip(s, 0, 1)
        s = normalize_h(s, target_sum)
        
        # Vanilla line search
        alpha = backtracking_line_search(h, s, dx)

            # No meaningful step

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        else:
