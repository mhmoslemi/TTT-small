import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create random feasible starting point
        h = np.random.rand(n_points)
        h = project_to_feasible(h, target_sum)
    
    # Create initial population
    for _ in range(population_size):
        if np.random.rand() < 0.2:
            # Use slight variation of initial solution
            individual = h + np.random.normal(0, 0.02, n_points)
        else:
            # Generate completely random individual
            individual = np.random.rand(n_points)
        individual = project_to_feasible(individual, target_sum)
        population.append(individual)
    
    # PSO parameters
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iterations = 60
    
    # Initialize PSO
    particles = np.array(population)
    velocities = np.random.uniform(-0.1, 0.1, (population_size, n_points))
    
    # Personal best
    pbest = particles.copy()
    pbest_fitness = [compute_c5(p, dx) for p in pbest]
    
    # Global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # Track best solution
    best_h = gbest.copy()
    best_c5 = gbest_fitness
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_coeff * r1 * (pbest[i] - particles[i]) +
                           social_coeff * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = particles[i].copy()
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    stagnation_count = 0
                    if gbest_fitness < best_c5:
                        best_c5 = gbest_fitness
                        best_h = gbest.copy()
            else:
                stagnation_count += 1
                
        # Update inertia weight
        inertia_weight *= 0.98
        
        # Early termination for stagnation
        if stagnation_count > 20:
            break
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    refined_h = best_h.copy()
    best_h_after_refinement = refined_h.copy()
    best_c5_after_refinement = compute_c5(refined_h, dx)
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, refined_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            new_c5 = compute_c5(refined_h, dx)
            if new_c5 < best_c5_after_refinement:
                best_c5_after_refinement = new_c5
                best_h_after_refinement = refined_h.copy()
        else:
            break

    # Final validation and return
    final_h = best_h_after_refinement
    final_c5 = best_c5_after_refinement
    
    return (final_h.tolist(), final_c5, n_points)
