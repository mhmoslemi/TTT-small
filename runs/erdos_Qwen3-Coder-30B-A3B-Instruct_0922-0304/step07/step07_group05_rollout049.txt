import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with provided starting point if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create an initial guess with triangular shape
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / (n_points/4))
        h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    patience = 0
    max_patience = 10
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        if iter_count > 60:
            break
            
        # Compute current C5 and its gradient approximation
        current_c5 = compute_c5(h, dx)
        
        # Use simpler local search approach
        # Try small random perturbations and accept improvements
        perturbation_magnitude = 0.02
        candidate_h = h.copy()
        
        # Make a small random perturbation
        delta = np.random.normal(0, perturbation_magnitude, n_points)
        candidate_h += delta
        candidate_h = project_to_feasible(candidate_h, target_sum)
        
        candidate_c5 = compute_c5(candidate_h, dx)
        
        if candidate_c5 < current_c5:
            h = candidate_h
            patience = 0
        else:
            patience += 1
            
        # Track best solution
        if compute_c5(h, dx) < best_c5:
            best_h = h.copy()
            best_c5 = compute_c5(best_h, dx)
            
        iter_count += 1
            
        if best_c5 <= 0.3807:
            break
    
    # Final refinement using simple local optimization
    # Take a few steps of gradient ascent using numerical gradients
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        c5_before = compute_c5(h, dx)
        # Compute finite differences approximating gradient
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
        
        # Try to move in negative gradient direction
        step_size = 0.01
        trial_h = h - step_size * grad
        trial_h = project_to_feasible(trial_h, target_sum)
        
        if compute_c5(trial_h, dx) < compute_c5(h, dx):
            h = trial_h
            if compute_c5(h, dx) < best_c5:
                best_h = h.copy()
                best_c5 = compute_c5(best_h, dx)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
