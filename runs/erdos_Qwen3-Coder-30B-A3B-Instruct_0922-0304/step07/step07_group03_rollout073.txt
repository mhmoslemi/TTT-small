import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values using sin pattern
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # Alternating minimization parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    decay_tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        c = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(c)
        
        # Compute gradient based on max overlap
        # gradient approximated by shifting (1-h) to match max overlap
        shift = max_k - (len(h) - 1)  # Determine required shift
        gradient = np.zeros_like(h)
        if shift >= 0:
            gradient[shift:] = -(1 - h[:-shift])
        else:
            gradient[:shift] = -(1 - h[-shift:])
            
        # Update h
        h_candidate = h - alpha * gradient
        
        # Add random perturbation
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to feasibility
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        new_c5 = compute_c5(h_candidate, dx)
        
        if new_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = new_c5
            alpha *= 0.99
        else:
            alpha *= 0.999
        
        # Apply perturbation
        h = h_candidate
        
        # Learning rate decay
        alpha = max(0.001, alpha * (1 - 1/decay_tau))
        epsilon = max(0.001, epsilon * (1 - 1/decay_tau))
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
