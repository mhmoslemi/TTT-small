import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Track steady state
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        overlap = correlate(h, 1 - h, mode="full") * dx
        
        # Find maximum overlap
        max_overlap_idx = np.argmax(overlap)
        current_max = overlap[max_overlap_idx]
        
        # Update only top 10% coordinates based on gradient estimation
        # Gradient component approximated as overlap[i] * (1 - 2*h[i])
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # We compute for the first overlap term (diagonal) 
            # that is related most directly to h[i] in the overlap
            if max_overlap_idx < n_points:  # Within first overlap region
                # Approximate gradient contribution to maximum overlap
                gradients[i] = overlap[max_overlap_idx] * (1 - 2 * h[i]) 
            else:
                # Central overlap contribution
                gradients[i] = overlap[max_overlap_idx] * (1 - 2 * h[i])  

        # Select top 10% by absolute gradient 
        top_k = max(1, n_points // 10)
        top_indices = np.argsort(np.abs(gradients))[-top_k:]
        
        # Apply updates to best indices
        updated = False
        for i in top_indices:
            # Estimate step size for decreasing overlap
            # Very rough approximation with fixed step size for learning
            dt = 0.01
            h_new = h.copy()
            h_new[i] = np.clip(h[i] + dt * (1 - 2 * h[i]), 0, 1)
            # Renormalize new vector
            h_new = normalize_h(h_new, target_sum)
            h_new = np.clip(h_new, 0, 1)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                h = h_new.copy()
                updated = True
                
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Slight perturbation to avoid local optima  
        if iteration % 100 == 0 and not updated:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        if iteration - last_improvement > 1000:
            break
            
        # Prevent infinite recursion due to floating point issues
        if np.isnan(best_c5) or best_c5 < 1e-10:
            break
            
    return (best_h.tolist(), best_c5, n_points)
