import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def evaluate_feasibility(h, max_overlap, dx):
    # Check if the current h satisfies the constraint
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx <= max_overlap

def gradient_descent_step(h, dx, max_overlap, step_size=1e-3, max_iter=20):
    # Simple gradient descent for smoothing
    h = h.copy()
    for _ in range(max_iter):
        # Approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        current_overlap = compute_c5(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, len(h)/2)
            overlap_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_constraints(h_minus, len(h)/2)
            overlap_minus = compute_c5(h_minus, dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Update step
        h_new = h - step_size * grad
        h_new = project_to_constraints(h_new, len(h)/2)
        
        # Accept step if it reduces overlap
        new_overlap = compute_c5(h_new, dx)
        if new_overlap <= current_overlap:
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-6:
                break
    
    return h
