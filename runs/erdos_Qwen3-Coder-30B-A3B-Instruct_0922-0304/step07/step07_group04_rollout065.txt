import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient-based optimization with constraints
    def objective(h_vector):
        return evaluate_overlap(h_vector, dx)
    
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Start with a simple gradient descent approach
    max_iter = 500
    lr = 0.01
    prev_c5 = best_c5 + 1
    patience = 0
    patience_limit = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            grad[j] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        
        # Update step
        h = h - lr * grad
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (target_sum / h_sum)
        
        # Evaluate and update best
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if abs(prev_c5 - current_c5) < 1e-8:
            lr *= 0.9
        prev_c5 = current_c5
        
        if patience > patience_limit:
            lr *= 0.9
            patience = 0
            
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s - 10:
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
