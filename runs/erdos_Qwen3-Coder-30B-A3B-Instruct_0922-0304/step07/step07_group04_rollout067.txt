import numpy as np
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Monte Carlo style optimization with local search
    num_iterations = 0
    max_iterations = 5000
    
    # Optimization parameters
    learning_rate = 0.005
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 5 and num_iterations < max_iterations:
        num_iterations += 1
        
        # Evaluate current overlap
        current_overlap = evaluate_overlap(h, dx)
        
        # Find the maximizing shift
        full_overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(full_overlap)
        max_shift = max_k - (n_points - 1)
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute the derivative w.r.t h[i]
            # d/dh_i [sum h_j (1 - h_{(j-k)%n})] = (1 - h_{(i-k)%n}) (when k is max shift)
            j = (i - max_shift) % n_points
            grad[i] = 1 - h[j]
        
        # Add gradient descent step with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Apply bounds and normalization
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        # Evaluate new overlap
        new_c5 = evaluate_overlap(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
            # Reset velocity if we improve
            velocity = np.zeros(n_points)
    
    # Final gradient-based refinement
    if time.time() - start_time < budget_s - 5:
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
