```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create a reasonable starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a triangular pulse as initial guess
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Gradient approximation function
    def approx_grad(h, dx, epsilon=1e-5):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            f_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            f_minus = compute_c5(h_minus, dx)
            
            grad[i] = (f_plus - f_minus) / (2 * epsilon)
        return grad
    
    # Project h to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h
    
    patience = 0
    max_patience = 10
    max_iter = 60
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation
        grad = approx_grad(best_h, dx)
        
        # Formulate and solve the QP
        # We want to minimize: 0.5 * d^T * H * d + grad^T * d
        # Subject to: sum(d) = 0, 0 <= h + d <= 1
        
        # Use a simple diagonal Hessian for simplicity
        H = np.eye(len(best_h)) * 1e-3
        
        # Set up QP
        d = cp.Variable(len(best_h))
        
        # Objective: 0.5 * d^T * H * d + grad^T * d
        objective = 0.5 * cp.quad_form(d, H) + grad.T @ d
        
        # Constraints
        constraints = [
            cp.sum(d) == 0,
            d >= -best_h,
            d <= 1 - best_h
        ]
        
        # Solve QP
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                step_direction = d.value
            else:
                # Fallback to gradient descent if QP fails
                step_direction = -grad
        except:
            # Fallback to gradient descent
            step_direction = -grad
        
        # Line search with backtracking
        alpha = 1.0
        max_alpha = 10.0
        step_size = 0.1
        
        while alpha > 0.01:
            candidate = best_h + alpha * step_direction
            candidate = project(candidate)
            candidate_c5 = compute_c5(candidate, dx)
            
            if candidate_c5 < best_c5:  # Accept the step
                best_h = candidate
                best_c5 = candidate_c5
                patience = 0
                break
            
            alpha *= 0.5
        
        # If no improvement, try a simple gradient descent step
        if alpha <= 0.01 and best_c5 > 0.3807:
            # Simple gradient descent step
            simple_step = -grad * step_size
            simple_candidate = best_h + simple_step
            simple_candidate = project(simple_candidate)
            simple_c5 = compute_c5(simple_candidate, dx)
            
            if simple_c5 < best_c5:
                best_h = simple_candidate
                best_c5 = simple_c5
                patience = 0
            else:
                patience += 1
                
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if patience >= max_patience:
            break
    
    # Final refinement with scipy.optimize if possible
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Try a few rounds of scipy optimization
    try:
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(objective, best_h.copy(), method='SLSQP', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 20})
            if result.success:
                new_h = result.x
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (target_sum / np.sum(new_h))
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
            else:
                break
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
```
