import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def local_optimize(h, dx, target_sum):
    """Use L-BFGS-B local optimization to improve solution."""
    def objective(h_flat):
        h = h_flat.reshape(-1)
        h = project_to_constraints(h, target_sum)
        return compute_c5(h, dx)
    
    def gradient(h_flat):
        # Custom gradient approximation
        epsilon = 1e-6
        grad = np.zeros_like(h_flat)
        for i in range(len(h_flat)):
            h_plus = h_flat.copy()
            h_plus[i] += epsilon
            h_minus = h_flat.copy()
            h_minus[i] -= epsilon
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        return grad
    
    h_initial = h.copy()
    bounds = [(0, 1) for _ in range(len(h_initial))]
    
    try:
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500}
        )
        h_opt = result.x.reshape(-1)
        h_opt = project_to_constraints(h_opt, target_sum)
        return h_opt
    except:
        return h_initial

    # Evolutionary Strategy parameters
    pop_size = 20
    sigma_init = 0.1
    sigma_final = 0.01
    max_generations = 2000
    
    # Create initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        noise = np.random.normal(0, sigma_init, n_points)
        temp_h = h + noise
        temp_h = project_to_constraints(temp_h, target_sum)
        population.append(temp_h)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

        if time.time() - start_time > budget_s - 1:

        # Evaluate population
        fitness = []

            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]

        if fitness[0] < best_c5:
