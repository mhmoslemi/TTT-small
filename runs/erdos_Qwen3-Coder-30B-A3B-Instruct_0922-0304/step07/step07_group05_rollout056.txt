import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Create diverse starting solutions
        particles.append(h.copy())
        for _ in range(population_size // 5):
            variation = h + np.random.normal(0, 0.02, n_points)
            variation = np.clip(variation, 0, 1)
            variation = normalize_h(variation, target_sum)
            particles.append(variation)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
    
    # Fill remaining population
    for _ in range(population_size - len(particles)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
    
    # PSO parameters
    w = 0.8  # Inertia weight
    c1 = 1.5  # Cognitive coefficient
    c2 = 1.5  # Social coefficient
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_fitnesses = [compute_c5(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(personal_best_fitnesses)
    global_best = personal_bests[best_idx].copy()
    global_best_fitness = personal_best_fitnesses[best_idx]
    
    # PSO iterations
    max_iter = 60
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate and update personal best
            fitness = compute_c5(particles[i], dx)
            if fitness < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fitness
                personal_bests[i] = particles[i].copy()
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best = particles[i].copy()
            
            # Early stopping
            if global_best_fitness <= 0.3807:
                break
                
        # Update inertia weight
        w *= 0.98
        
        if global_best_fitness <= 0.3807:
            break
    
    # Refinement with gradient descent
    best_h = global_best.copy()
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Run gradient descent several times with different strategies
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 10})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
