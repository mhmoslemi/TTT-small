import numpy as np
from scipy.signal import correlate
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = h * (target_sum / np.sum(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)
        
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    # Add one component around the initial solution
    means.append(h.copy())
    covariances.append(np.eye(n_points) * initial_sigma)
    weights.append(1.0)
    
    # Add random components for diversity
    for i in range(n_components - 1):
        init_h = np.random.rand(n_points)
        init_h = np.clip(init_h, 0, 1)
        init_h = project_to_constraint(init_h, target_sum)
        means.append(init_h)
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
        
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 0
    max_iterations = 6000  # To ensure we don't exceed budget
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        # Sample population from GMM
        samples = []
        sample_weights = []
        
        for i in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            # Sample from component
            try:
                sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            except np.linalg.LinAlgError:
                # Fallback if covariance matrix is not positive definite
                sample = means[comp_idx] + np.random.normal(0, initial_sigma, n_points)
            
            # Apply constraints
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            samples.append(sample)
        
        # Evaluate all samples
        values = []
        for sample in samples:
            c5 = compute_c5(sample, dx)
            values.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(values)
        top_samples = [samples[i] for i in sorted_indices[:keep_top]]
        top_values = [values[i] for i in sorted_indices[:keep_top]]
        
        # Update best
        if top_values[0] < best_c5:
            best_c5 = top_values[0]
            best_h = top_samples[0].copy()
            
        # Update GMM parameters
        # Update means
        new_means = []
        for i in range(n_components):
            if i < len(top_samples):
                new_means.append(top_samples[i])
            else:
                # Use a random set of previous means as fallback
                idx = np.random.randint(len(means))
                new_means.append(means[idx].copy())

        # Update covariances
        new_covariances = []
        for i in range(n_components):
            if i < len(top_samples):
                # Calculate covariance of top samples for this component
                if i == 0:
                    # Use all top samples for first component
                    sample_subspace = np.array(top_samples)
                else:
                    # For other components, just use top 10 samples
                    sample_subspace = np.array(top_samples[:10])
                
                if sample_subspace.shape[0] > 1:
                    # Use the samples' covariance
                    diff = sample_subspace - means[i]
                    cov = np.cov(diff.T)
                    # Ensure positive definiteness
                    try:
                        np.linalg.cholesky(cov + 1e-8 * np.eye(n_points))
                    except np.linalg.LinAlgError:
                        cov = np.eye(n_points) * initial_sigma
                else:
                    cov = np.eye(n_points) * initial_sigma
                new_covariances.append(cov)
            else:
                cov = covariances[i] * covariance_decay
                # Make sure it remains positive definite
                try:
                    np.linalg.cholesky(cov + 1e-8 * np.eye(n_points))
                except np.linalg.LinAlgError:
                    cov = np.eye(n_points) * initial_sigma
                new_covariances.append(cov)
                
        # Update weights (uniform distribution)
        weights = np.ones(n_components) / n_components
        
        # Update means and covariances
        means = new_means
        covariances = new_covariances
        
        # Decay covariances
        for i in range(n_components):
            covariances[i] = covariances[i] * covariance_decay
            
        iterations += 1
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
