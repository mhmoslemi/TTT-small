import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points).astype(np.float64)
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iterations = 0
    last_improvement = 0
    max_iter_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        current_max = np.max(overlap)
        
        # Find coordinates that contribute most to current_max
        max_indices = np.where(np.abs(overlap - current_max) < 1e-10)[0]
        # Select top contributing indices
        indices = []
        for idx in max_indices:
            # Check each h[i] that is involved in overlap at this index
            start_idx = max(0, idx - (n_points - 1))
            end_idx = min(n_points, idx + 1)
            for i in range(start_idx, end_idx):
                indices.append(i)
        indices = list(set(indices))
        
        # Sort indices by gradient impact
        gradients = np.zeros(n_points)
        for i in range(n_points):
            # Compute approximate gradient
            if overlap[i] == current_max:
                gradients[i] = np.abs(overlap[i] * (1 - 2 * h[i]))
            else:
                gradients[i] = 0
        
        # Select top 10% of indices for update
        top_k = max(1, n_points // 10)
        sorted_indices = np.argsort(-gradients)[:top_k]
        
        # Update selected coordinates
        updated_h = h.copy()
        improved = False
        
        for i in sorted_indices:
            old_val = h[i]
            # Determine step size for descent
            # For coordinate-wise descent, we can move more aggressively
            alpha = 0.05
            h_new = h.copy()
            h_new[i] = np.clip(old_val + alpha * (1 - 2 * old_val), 0, 1)
            # Renormalize
            h_new = normalize_h(h_new, target_sum)
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5 - 1e-9:
                best_c5 = new_c5
                best_h = h_new.copy()
                improved = True
                
        # Update if improvement
        if improved:
            h = best_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Steady state check
        if last_improvement > max_iter_no_improvement:
            break
            
    # Final computation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
