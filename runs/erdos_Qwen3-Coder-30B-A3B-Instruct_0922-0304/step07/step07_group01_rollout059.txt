import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Read initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Simple initial construction
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize to sum constraint
    h = project_to_constraints(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Convex relaxation approach
    # Let's again try a simple gradient-based refinement
    max_iter = 1000
    learning_rate = 0.01
    decay_factor = 0.999
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation via finite differences for each point
        c5_current = compute_c5(h, dx)
        
        # Simple gradient descent with projected updates
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update step
        h_new = h - learning_rate * grad
        h_new = project_to_constraints(h_new, target_sum)
        
        # Update if improvement
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        learning_rate *= decay_factor
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
