import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of C5 with respect to h using finite differences."""
    # Compute correlation of h with (1-h)
    overlap = correlate(h, 1 - h, mode="full")
    
    # Compute gradient for each point
    grad = np.zeros(n_points)
    
    # For each point h[i], compute the gradient - the sum of influence from all shifts
    for i in range(n_points):
        for k in range(n_points):
            if 0 <= i + k < n_points:
                # Contribution to overlap at shift k is h[i] * (1 - h[i+k])
                # So gradient of overlap at shift k w.r.t. h[i] is (1 - h[i+k])
                grad[i] += 2 * overlap[n_points - 1 + k] * (1 - h[i + k]) / dx
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the best known solution or randomly
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = normalize_h(h, target_sum)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Initialize Adam parameters
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradient
        grad = compute_gradient(h, dx, n_points)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (iter_count + 1))
        v_hat = v / (1 - beta2 ** (iter_count + 1))
        
        # Update parameters
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Compute new C5
        current_c5 = compute_c5(h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate decay
        if iter_count % 1000 == 0 and iter_count > 0:
            lr *= 0.999
            
        iter_count += 1
        
        # Check for stagnation
        if no_improvement_count > max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
