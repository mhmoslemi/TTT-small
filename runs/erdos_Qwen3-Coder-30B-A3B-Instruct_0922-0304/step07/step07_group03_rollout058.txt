import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def evolve_population(population, fitnesses, mutation_sigma, target_sum):
    """Evolve population using ES with Gaussian mutation."""
    # Selection: keep top 50%
    idx = np.argsort(fitnesses)
    selected = [population[i] for i in idx[:len(population)//2]]
    
    # Create new population
    new_population = []
    for i in range(len(population)):
        # Select parent
        parent = selected[np.random.randint(0, len(selected))]
        # Gaussian mutation
        mutant = parent + np.random.normal(0, mutation_sigma, len(parent))
        # Apply constraints
        mutant = project_to_constraints(mutant, target_sum)
        new_population.append(mutant)
    
    return new_population

def local_optimize(h, target_sum, dx, max_iter=1000):
    """Use L-BFGS-B to locally refine h."""
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_fun(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': constraint_fun}
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      constraints=constraints, options={'maxiter': max_iter})
    
    if result.success:
        return project_to_constraints(result.x, target_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with diverse candidates
    population_size = 20
    population = []

        base_h = np.clip(base_h, 0, 1)
        base_h = project_to_constraints(base_h, target_sum)
        population.append(base_h)
        # Add perturbed versions
        for _ in range(population_size - 1):
            perturbed = base_h + np.random.normal(0, 0.1, n_points)
            population.append(project_to_constraints(perturbed, target_sum))
    else:
        # Create diverse initial population
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, target_sum)
            population.append(h)
    
    # Evolution parameters
    mutation_sigma_start = 0.1
    mutation_sigma_end = 0.01
    max_generations = 1000
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(h, dx) for h in population]
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Check if we should apply local optimization
        if best_c5 < 0.381:
            # Refine with local optimizer
            local_h = local_optimize(best_h, target_sum, dx)
            local_c5 = compute_c5(local_h, dx)
            if local_c5 < best_c5:
                best_c5 = local_c5
                best_h = local_h.copy()
        
        # Update mutation sigma
        t = generation / max_generations
        mutation_sigma = mutation_sigma_start - t * (mutation_sigma_start - mutation_sigma_end)
        
        # Evolve population
        population = evolve_population(population, fitnesses, mutation_sigma, target_sum)
    
    # Final local optimization on best solution
    final_h = local_optimize(best_h, target_sum, dx)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
