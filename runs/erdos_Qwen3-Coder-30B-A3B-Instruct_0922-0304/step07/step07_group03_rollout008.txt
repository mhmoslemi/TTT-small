import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

        h = normalize_h(h, target_sum)
    
    # ES parameters
    population_size = 20
    sigma_start = 0.1
    sigma_end = 0.01
    max_generations = 10000
    
    # Population
    population = []
    fitness = []
    
    # Initialize population
    if 'initial_h_values' in globals():
        population.append(h.copy())
        fitness.append(compute_c5(h, dx))
        for _ in range(population_size - 1):
            perturbation = np.random.normal(0, sigma_start, n_points)
            h_pert = h + perturbation
            h_pert = project_to_constraints(h_pert, target_sum)
            population.append(h_pert)
            fitness.append(compute_c5(h_pert, dx))
    else:
        for _ in range(population_size):
            h_init = np.random.uniform(0, 1, n_points)
            h_init = normalize_h(h_init, target_sum)
            population.append(h_init)
            fitness.append(compute_c5(h_init, dx))
    
    # Best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolutionary strategy loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update sigma
        sigma = sigma_start - (sigma_start - sigma_end) * gen / max_generations
        
        # Selection (keep top 50%)
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:population_size//2]
        elite_population = [population[i] for i in elite_indices]
        elite_fitness = [fitness[i] for i in elite_indices]
        
        # Generate new population through mutation
        new_population = []
        new_fitness = []
        
        # Add elites unchanged
        for i in range(len(elite_population)):
            new_population.append(elite_population[i].copy())
            new_fitness.append(elite_fitness[i])
        
        # Create offspring
        for _ in range(population_size - len(elite_population)):
            # Select parent randomly from elites
            parent_idx = np.random.choice(len(elite_population))
            parent = elite_population[parent_idx]
            
            # Mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            
            # Evaluate child
            child_c5 = compute_c5(child, dx)

        fitness = new_fitness
        
        # Update best

            best_c5 = fitness[current_best_idx]
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
