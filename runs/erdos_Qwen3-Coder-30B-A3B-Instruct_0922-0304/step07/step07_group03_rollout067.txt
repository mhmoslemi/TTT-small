import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values with sawtooth pattern
        h = np.sin(np.pi * np.arange(n_points) / n_points) ** 2
        h = project_to_constraints(h, target_sum)
    
    # Greedy optimization with adaptive perturbations
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Mutation strength decay parameters
    epsilon = 0.2
    epsilon_decay = 0.99
    min_epsilon = 0.001
    decay_interval = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate random perturbation
        direction = np.random.randn(n_points)
        direction = direction / np.linalg.norm(direction)
        perturbation = epsilon * direction
        
        # Apply perturbation
        candidate_h = best_h + perturbation
        
        # Project to feasible region
        candidate_h = project_to_constraints(candidate_h, target_sum)
        
        # Compute C5
        candidate_c5 = compute_c5(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            # Reduce epsilon more aggressively when accepting
            epsilon = max(min_epsilon, epsilon * epsilon_decay ** 2)
        else:
            # Reduce epsilon less aggressively
            epsilon = max(min_epsilon, epsilon * epsilon_decay)
        
        # Periodic epsilon adjustment
        if iteration % decay_interval == 0:
            epsilon = max(min_epsilon, epsilon * epsilon_decay)
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
