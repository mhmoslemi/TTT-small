import numpy as np
from scipy.signal import correlate
import time
from scipy.stats import multivariate_normal
from scipy.linalg import sqrtm

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # AGMA parameters
    n_components = 5
    population_size = 100
    keep_top = 20
    decay_factor = 0.99
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    # Add one component around the initial solution
    means.append(h.copy())
    covariances.append(np.eye(n_points) * 0.1)
    weights.append(1.0)
    
    # Add random components
    for _ in range(n_components - 1):
        rand_h = np.random.rand(n_points)
        rand_h = np.clip(rand_h, 0, 1)
        rand_h = normalize_h(rand_h, target_sum)
        means.append(rand_h)
        covariances.append(np.eye(n_points) * 0.1)
        weights.append(1.0)
    
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    # Main loop
    iteration = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    while time.time() - start_time < budget_s - 1 and iteration < 6000:
        # Sample from GMM
        candidates = []
        for _ in range(population_size):
            # Select component
            comp_idx = np.random.choice(n_components, p=weights)
            mean = means[comp_idx]
            cov = covariances[comp_idx]
            
            # Sample from multivariate normal
            sample = np.random.multivariate_normal(mean, cov)
            
            # Apply bounds and constraint
            sample = np.clip(sample, 0, 1)
            sample = project_to_constraint(sample, target_sum)
            
            candidates.append(sample)
        
        # Evaluate all candidates
        c5_values = []
        for candidate in candidates:
            c5 = compute_c5(candidate, dx)
            c5_values.append(c5)
        
        # Sort by c5 values
        sorted_indices = np.argsort(c5_values)
        top_candidates = [candidates[i] for i in sorted_indices[:keep_top]]
        top_c5 = [c5_values[i] for i in sorted_indices[:keep_top]]
        
        # Update best
        min_c5 = min(top_c5)
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = top_candidates[0].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Update GMM parameters
        # Update means
        new_means = []
        for i in range(n_components):
            if i < len(top_candidates):
                new_means.append(top_candidates[i])
            else:
                # Reuse existing means for missing components
                new_means.append(means[i])
        
        # Update covariances
        new_covariances = []
        for i in range(n_components):
            if i < len(top_candidates):
                # Fit covariance to top candidates from this component
                # Find candidates with the same component
                pairs = [j for j in range(len(top_candidates)) if np.random.choice(n_components, p=weights) == i]
                if len(pairs) > 0:
                    selected = [top_candidates[j] for j in pairs[:10]]  # Use top 10 from this component
                    # Compute sample covariance
                    if len(selected) > 1:
                        centered = np.array(selected) - np.mean(selected, axis=0)
                        cov = np.cov(centered.T)
                        new_covariances.append(cov + np.eye(n_points) * 1e-6)  # Add regularization
                    else:
                        # Fallback to current covariance
                        new_covariances.append(covariances[i])
                else:
                    # Fallback to current covariance
                    new_covariances.append(covariances[i])
            else:
                # Keep existing covariance
                new_covariances.append(covariances[i])
        
        # Update weights using softmax
        if len(top_c5) > 0:
            # Weight based on inverse of C5 values
            weights = np.array(top_c5)
            weights = 1.0 / (weights + 1e-10)  # Avoid division by zero
            weights = weights / np.sum(weights)
        
        # Apply covariance decay
        new_covariances = [cov * decay_factor for cov in new_covariances]
        
        # Update all parameters
        means = new_means
        covariances = new_covariances
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
