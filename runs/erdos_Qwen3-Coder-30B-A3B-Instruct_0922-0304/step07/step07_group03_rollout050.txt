import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (target_sum / np.sum(base_h))
    else:
        base_h = np.random.uniform(0, 1, n_points)
        base_h = base_h * (target_sum / np.sum(base_h))
    
    # Generate initial population
    for i in range(population_size):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = project_to_constraints(h, target_sum)
        population.append(h)
    
    # Evolution strategy parameters
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_iter = 20000
    
    best_h = None
    best_c5 = float('inf')
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute fitness for all individuals
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        fitnesses = [fitnesses[i] for i in sorted_indices]
        population = [population[i] for i in sorted_indices]
        
        # Update best
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adapt mutation rate
        t = iter_num / max_iter
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * t
        
        # Selection: keep top half
        selected = population[:population_size//2]
        
        # Reproduction: mutate selected individuals
        new_population = selected.copy()
        while len(new_population) < population_size:
            parent = selected[np.random.randint(0, len(selected))]
            # Add Gaussian noise for mutation
            mutated = parent + np.random.normal(0, mutation_rate, n_points)
            mutated = project_to_constraints(mutated, target_sum)
            new_population.append(mutated)
        
        population = new_population
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
