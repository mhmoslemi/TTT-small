import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def gradient_descent_optimization(h, dx, max_iter=10000, lr=0.01, tol=1e-8, decay_rate=0.999):

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    def loss_and_grad(h_vec):
        h_vec = h_vec.copy()
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = normalize_h(h_vec, n_points / 2)
        
        # Compute overlap using correlate
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(overlap)
        c5 = max_overlap * dx
        c5 = float(c5)
        
        # Compute gradient
        grad = np.zeros_like(h_vec)

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        return c5, grad
    
    # Run optimization with adaptive learning rate and tracking
    for i in range(max_iter):
        if i % 1000 == 0:
            lr *= decay_rate
            if lr < 1e-6:
                lr = 1e-6
        
        c5, grad = loss_and_grad(best_h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Update step
        best_h -= lr * grad

            break
            
        if last_improvement > 1000:
            break
    
    return best_h, best_c5
