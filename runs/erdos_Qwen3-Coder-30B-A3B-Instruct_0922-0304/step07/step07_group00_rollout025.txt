import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the objective function."""
    # Compute all overlaps (full convolution)
    h_rev = h[::-1]
    # Using full mode, we get 2*n_points - 1 overlaps
    overlaps = correlate(h, 1 - h, mode="full")
    
    # Compute gradient of the max overlap with respect to h
    # This is an approximation since max operation is not differentiable,
    # but we differentiate w.r.t. the max element
    max_idx = np.argmax(overlaps)
    
    # For each position i, compute contribution to gradient 
    grad = np.zeros_like(h)
    
    # The derivative of overlap_k = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) when k is fixed
    # Only affects terms where i+k is valid
    
    for i in range(n_points):
        # k ranges from -n+1 to n-1 (indices in full correlation)
        # But we map to 0 to 2*n-2 for easier indexing
        for k in range(max(0, i-n+1), min(i+1, n_points)):
            idx = i + n_points - 1 - k
            if idx < len(overlaps):
                grad[i] += (1.0 - 2.0 * h[k]) * (1.0 if i >= k else 0.0)
    
    # Normalize to prevent gradient explosion
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 0:
        grad = grad / grad_norm
        
    return grad * dx

def adam_optimizer(h, dx, n_points, max_iter=10000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, tol=1e-8):
    """Adam optimizer for minimizing the overlap."""
    # Initialize Adam variables
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for t in range(1, max_iter + 1):
        # Compute gradient
        grad = compute_gradient(best_h, dx, n_points)
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update parameters
        h_new = best_h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2)
        
        # Evaluate new objective
        new_c5 = compute_c5(h_new, dx)
        
        # Check improvement
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        # Check if converged
        if np.abs(best_c5 - new_c5) < tol:
            break
            
        # Decay learning rate
        if t % 1000 == 0:
            lr *= 0.999
    
    return best_h, best_c5
