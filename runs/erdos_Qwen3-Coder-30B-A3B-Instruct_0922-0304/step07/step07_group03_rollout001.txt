import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
    
    # Normalize to target sum
    h = project_to_constraints(h, target_sum)
    
    # Evolutionary Strategy parameters
    population_size = 20
    elite_fraction = 0.5
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_generations = 20000
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Generate initial population
    for i in range(population_size):
        if 'initial_h_values' in globals() and i == 0:
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = project_to_constraints(individual, target_sum)
        population.append(individual)
        fitnesses.append(compute_c5(individual, dx))
    
    # Sort by fitness
    sorted_indices = np.argsort(fitnesses)
    population = [population[i] for i in sorted_indices]
    fitnesses = [fitnesses[i] for i in sorted_indices]
    
    best_individual = population[0].copy()
    best_fitness = fitnesses[0]
    
    # ES iteration
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * generation / max_generations
        
        # Select elite
        elite_size = int(population_size * elite_fraction)
        elite = population[:elite_size]
        
        # Generate offspring through mutation
        offspring = []
        for i in range(elite_size):
            parent = elite[i]
            # Mutate with Gaussian noise
            noise = np.random.normal(0, mutation_rate, n_points)
            mutated = parent + noise
            mutated = project_to_constraints(mutated, target_sum)
            offspring.append(mutated)
        
        # Replace worst individuals with offspring
        population = elite + offspring
        
        # Evaluate fitness
        fitnesses = [compute_c5(ind, dx) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update global best
        if fitnesses[0] < best_fitness:
            best_fitness = fitnesses[0]
            best_individual = population[0].copy()
        
        # Early termination
        if best_fitness <= 0.3807:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
