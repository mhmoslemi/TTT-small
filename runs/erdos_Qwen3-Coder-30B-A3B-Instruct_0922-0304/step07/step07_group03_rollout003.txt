import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, target_sum)
    
    # Evolution Strategy parameters
    pop_size = 20
    elite_size = pop_size // 2
    mutation_rate_start = 0.1
    mutation_rate_end = 0.01
    max_iter = 20000
    
    population = [h.copy()]
    for _ in range(pop_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        new_h = h + noise
        new_h = project_to_constraints(new_h, target_sum)
        population.append(new_h)
    
    # Evaluate initial population
    fitness = []
    for p in population:
        fitness.append(compute_c5(p, dx))
    
    # Best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update mutation rate
        mutation_rate = mutation_rate_start - (mutation_rate_start - mutation_rate_end) * iter_num / max_iter
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        elite_fitness = [fitness[i] for i in sorted_indices[:elite_size]]
        
        # Replace worst half with mutated elites
        for i in range(elite_size, pop_size):
            parent = elite[np.random.randint(0, elite_size)]
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = project_to_constraints(child, target_sum)
            population[i] = child
            fitness[i] = compute_c5(child, dx)
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
