import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

    start_time = time.time()

        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = project_to_constraints(h, target_sum)
    
    # Set up dynamic constraint tightening
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    alpha = 0.1
    max_alpha = 0.5
    
    # Bisection bounds
    c_low = 0.0
    c_high = best_c5
    
    # Gradient descent with trust region
    max_iter = 1000
    step_size = 0.01
    trust_radius = 0.1
    min_trust_radius = 1e-6
    max_trust_radius = 1.0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Bisection step
        c_mid = (c_low + c_high) / 2
        
        # Optimize subject to constraint: overlap <= c_mid * (1 + alpha)
        # Use a simple gradient-based approach with projections
        for _ in range(20):  # Inner optimization
            # Compute current overlap
            overlap = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap) * dx
            
            if max_overlap <= c_mid * (1 + alpha):
                break
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, target_sum)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_constraints(h_minus, target_sum)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps * dx)
            
            # Update h_tilde = h - step_size * grad
            h_tilde = h - step_size * grad
            
            # Trust region constraints
            diff = h_tilde - h
            if np.linalg.norm(diff) > trust_radius:
                diff = diff * (trust_radius / np.linalg.norm(diff))
                h_tilde = h + diff
            
            # Project to valid region
            h_tilde = project_to_constraints(h_tilde, target_sum)
            
            # Check convergence
            overlap_new = correlate(h_tilde, 1 - h_tilde, mode="full")
            if np.max(overlap_new) * dx < max_overlap:
                h = h_tilde.copy()
            else:
                # Reduce trust radius
                trust_radius = max(min_trust_radius, trust_radius / 2)

                    break

            trust_radius = min(max_trust_radius, trust_radius * 1.1)

                best_h = h.copy()
        else:
            c_low = c_mid
            # Relax constraints
            alpha = min(max_alpha, alpha * 1.05)
        
        iteration += 1
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
