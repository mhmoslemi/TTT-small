import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    population_size = 100
    particles = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        # Generate diverse particles
        particles.append(h.copy())
        for _ in range(int(0.4 * population_size) - 1):
            noise = np.random.normal(0, 0.03, n_points)
            h_var = h + noise
            h_var = np.clip(h_var, 0, 1)
            h_var = normalize_h(h_var, target_sum)
            particles.append(h_var.copy())
        # Fill the rest with random feasible solutions
        for _ in range(population_size - len(particles)):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            particles.append(h_rand.copy())
    else:
        # Start with random solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            particles.append(h.copy())
    
    # PSO parameters
    w_start = 0.8
    w_decay = 0.98
    c1 = c2 = 1.5
    max_iter = 60
    
    # Initialize personal bests and swarm best
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_c5(p, dx) for p in particles]
    swarm_best_idx = np.argmin(personal_best_fitness)
    swarm_best = personal_best[swarm_best_idx].copy()
    swarm_best_fitness = personal_best_fitness[swarm_best_idx]
    
    # Velocities
    velocities = [np.zeros(n_points) for _ in range(population_size)]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start * (w_decay ** iteration)
        
        for i in range(population_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (swarm_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate new fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update swarm best if needed
                if fitness < swarm_best_fitness:
                    swarm_best = particles[i].copy()
                    swarm_best_fitness = fitness
        
        # Early termination
        if swarm_best_fitness <= 0.3807:
            break
    
    # Gradient descent refinement on the best solution
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Refine the swarm best with gradient descent
    ref_h = swarm_best.copy()
    for _ in range(20):  # Limited iterations for refinement
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, ref_h, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 10})
        if result.success:
            ref_h = result.x
            ref_h = np.clip(ref_h, 0, 1)
            ref_h = normalize_h(ref_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(ref_h, dx)
    
    return (ref_h.tolist(), final_c5, n_points)
