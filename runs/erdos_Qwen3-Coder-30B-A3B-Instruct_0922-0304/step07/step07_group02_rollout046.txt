import numpy as np
from scipy.signal import correlate
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        h = np.clip(h + diff / len(h), 0, 1)
    else:
        # Need to decrease values
        h = np.clip(h + diff / len(h), 0, 1)
    # Scale to fix the sum exactly
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add randomness
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Adaptive Gaussian Mixture Algorithm (AGMA)
    n_components = 5
    population_size = 100
    keep_top = 20
    covariance_decay = 0.99
    initial_sigma = 0.1
    
    # Initialize GMM
    means = []
    covariances = []
    weights = []
    
    # One component from initial solution
    if 'initial_h_values' in globals():
        means.append(np.array(initial_h_values).copy())
    else:
        means.append(h.copy())
    covariances.append(np.eye(n_points) * initial_sigma)
    weights.append(1.0)
    
    # Add random components for diversity
    for _ in range(n_components - 1):
        mean = np.random.rand(n_points) * 0.5
        mean = normalize_h(mean, target_sum)
        means.append(mean.copy())
        covariances.append(np.eye(n_points) * initial_sigma)
        weights.append(1.0)
        
    weights = np.array(weights)
    weights = weights / np.sum(weights)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    max_iter = 6000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample from GMM
        samples = []
        sample_weights = []
        
        for i in range(population_size):
            # Sample component
            comp_idx = np.random.choice(n_components, p=weights)
            # Sample from component
            sample = np.random.multivariate_normal(means[comp_idx], covariances[comp_idx])
            samples.append(sample)
            
        # Process samples
        processed_samples = []
        sample_c5_values = []
        
        for sample in samples:
            # Clip to bounds
            sample = np.clip(sample, 0, 1)
            # Fix sum
            sample = project_to_constraint(sample, target_sum)
            # Evaluate
            c5 = compute_c5(sample, dx)
            processed_samples.append(sample)
            sample_c5_values.append(c5)
            
        # Select best samples
        sorted_indices = np.argsort(sample_c5_values)[:keep_top]
        top_samples = [processed_samples[i] for i in sorted_indices]
        top_c5 = [sample_c5_values[i] for i in sorted_indices]
        
        # Update GMM parameters
        # Update means
        new_means = []
        for i in range(n_components):
            # Find samples from component i (using weights)
            if i < len(top_samples):
                new_means.append(top_samples[i].copy())
            else:
                # Generate new random sample
                new_mean = np.random.rand(n_points) * 0.5
                new_mean = normalize_h(new_mean, target_sum)
                new_means.append(new_mean)
        
        # Update covariances
        new_covariances = []
        for i in range(n_components):
            if i < len(top_samples):
                # Update based on top samples
                if len(top_samples) > 1:
                    # Use centered samples
                    center = new_means[i]
                    centered = [s - center for s in top_samples]
                    # Simple variance estimate (assuming small samples)
                    X = np.array(centered)
                    cov = np.cov(X.T) + 1e-8 * np.eye(n_points)
                    # Clip to prevent matrix becoming too large
                    cov = np.clip(cov, 0, 10)
                else:
                    cov = covariances[i]
            else:
                cov = covariances[i]
            new_covariances.append(cov)
            
        # Covariance adaptation
        for i in range(n_components):
            new_covariances[i] = new_covariances[i] * covariance_decay
            
        # Update weights
        new_weights = np.ones(n_components)
        new_weights = new_weights / np.sum(new_weights)
        
        # Update parameters
        means = new_means
        covariances = new_covariances
        weights = new_weights
        
        # Track best
        for i in range(len(top_samples)):
            if top_c5[i] < best_c5:
                best_c5 = top_c5[i]
                best_h = top_samples[i].copy()
                
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
