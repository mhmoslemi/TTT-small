import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute deficit proportionally
    deficit = target_sum - current_sum
    if deficit > 0:
        # Need to increase values
        h += deficit * h / current_sum
        h = np.clip(h, 0, 1)
    else:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Start with a structured pattern
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = project_to_constraints(h, target_sum)
    
    # AM parameters
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        c = correlate(h, 1 - h, mode="full")
        max_val = np.max(c)
        
        # Identify regions contributing most to overlap
        indices = np.where(np.abs(c - max_val) < 1e-8)[0]
        if len(indices) == 0:
            indices = [np.argmax(c)]
            
        # Gradient-like update: try to reduce the maximum overlap
        # We assume that we're primarily interested in reducing the current max overlap
        h_candidate = h.copy()
        
        # Simple core update rule
        for i in range(n_points):
            contrib_to_max = 0.0
            for k in indices:
                if 0 <= i + k - n_points + 1 < n_points:
                    contrib_to_max += (1 - h[i])
            h_candidate[i] = h[i] - alpha * contrib_to_max
        
        # Add random perturbation
        h_candidate += epsilon * (2 * np.random.rand(n_points) - 1)
        
        # Project to constraints
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate new candidate
        new_c5 = compute_c5(h_candidate, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        # Adaptive learning rates
        alpha = 0.2 / (1 + it / tau)
        epsilon = 0.1 / (1 + it / tau)
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
