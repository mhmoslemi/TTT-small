import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

        # Add small randomness
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    # Try to improve with CVXPY
    try:
        # Define the optimization variables and constraints
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # The objective is to minimize the maximum overlap
        # We introduce auxiliary variables and constraints for this
        # But since this is not straightforward to express in CVXPY,
        # We'll use gradient descent on C5 with constraints instead
        
        # Instead, let's reuse the best solution from a local search
        # We'll try a local search with gradient-based optimization
        max_iter = 1000
        learning_rate = 0.01
        momentum = 0.9
        
        v = np.zeros_like(h)
        best_h = h.copy()
        best_c5 = compute_c5(best_h, dx)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient with finite differences
            c5_current = compute_c5(h, dx)
            eps = 1e-5
            
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)

                c5_plus = compute_c5(h_plus, dx)
                grad[j] = (c5_plus - c5_current) / eps
            
            # Update with momentum
            v = momentum * v - learning_rate * grad
            h = h + v
            
            # Project and enforce constraints
            h = np.clip(h, 0, 1)

            # Evaluate current
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
            if best_c5 <= 0.3807:

        # Return best found
        return (best_h.tolist(), best_c5, n_points)

        # Fall back to simpler optimization

            best_c5 = compute_c5(best_h, dx)

                if time.time() - start_time > budget_s - 1:

        except:
            # Return the initial state if both approaches fail
