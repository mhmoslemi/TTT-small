import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute difference to maintain sum
        diff = target_sum - current_sum
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        # Re-adjust to get exact sum
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, target_sum)
    
    # Simulated Annealing parameters
    T0 = 0.1 * compute_c5(h, dx)
    alpha = 0.99
    T = T0
    max_iter = 10000
    
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb solution
        h_new = h + np.random.uniform(-T, T, n_points)
        h_new = project_to_constraints(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < current_c5 or np.random.random() < np.exp((current_c5 - c5_new) / T):
            h = h_new
            current_c5 = c5_new
            
        # Update best if improved
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Cooling
        T *= alpha
        
        # Check for target
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
