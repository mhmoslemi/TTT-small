import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a smooth initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Remez-like optimization with phase inversion
    max_iterations = 20
    alpha = 0.1
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(overlap)
        
        # Phase inversion in frequency domain to reduce max overlap
        # Convert to frequency domain
        H = np.fft.fft(h)
        # Compute spectrum of (1-h)
        H_conj = np.fft.fft(1 - h)
        # Cross-correlation in frequency domain
        cross_spectrum = H * np.conj(H_conj)
        # Take inverse FFT
        cross_time = np.fft.ifft(cross_spectrum)
        # Find dominant frequency component
        freq_domain = np.fft.fft(h)
        # Find the frequency with highest energy (excluding DC)
        freq_idx = np.argmax(np.abs(freq_domain[1:])) + 1
        # Create phase modulation
        k = freq_idx / n_points  # normalized frequency
        phase_mod = np.sin(2 * np.pi * k * np.arange(n_points))
        # Update h
        h_new = h + alpha * phase_mod
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        h = h_new
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Reduce learning rate
        alpha *= 0.95
        
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with gradient descent
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 20})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target_sum / np.sum(best_h))
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
