import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

        h = project_to_constraints(h, target_sum)
    
    # AM parameters and tracking
    max_iter = 3000
    alpha = 0.2
    epsilon = 0.1
    decay_tau = 1000
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    convergence_threshold = 0.3807
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        
        # Gradient-like update in the direction of lowest overlap
        # Adjust h by minimizing the maximum overlap
        # This is a simplified gradient descent update
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Compute effect of perturbing h[i]
            h_temp = h.copy()
            h_temp[i] += 1e-4  # Small perturbation
            h_temp = project_to_constraints(h_temp, target_sum)
            new_overlap = correlate(h_temp, 1 - h_temp, mode="full")
            gradient[i] = (overlap[max_idx] - new_overlap[max_idx]) / 1e-4
                
        # Flatten the velocity direction (this is a simplified version)
        h_candidate = h - alpha * gradient
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Add random perturbation to escape local minima
        noise = epsilon * (2 * np.random.rand(n_points) - 1)
        h_candidate += noise
        h_candidate = project_to_constraints(h_candidate, target_sum)
        
        # Evaluate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        # Update if better
        if candidate_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = candidate_c5
            # Reduce learning rate after improvement
            alpha = max(0.01, alpha * 0.995)
            epsilon = max(0.01, epsilon * 0.995)

            # Keep current h, but reduce learning rate
