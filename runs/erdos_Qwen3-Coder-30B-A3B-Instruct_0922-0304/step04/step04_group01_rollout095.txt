```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential Convex Programming approach
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Linearization approach using CVXPY
        try:
            # Start fresh each iteration with the current h
            h_current = h.copy()
            h_vars = cp.Variable(n_points)
            s = cp.Variable()
            
            # Constraints for each shift k
            constraints = []
            # Integral constraint
            constraints.append(cp.sum(h_vars) * dx == 1)
            
            # Bound constraints
            constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
            constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
            
            # Linearized overlap constraints for each shift k  
            for k in range(-n_points+1, n_points):
                # Compute shifted version of h_current
                if k >= 0:
                    shifted_h = np.concatenate([h_current[k:], np.zeros(k)])
                else:
                    shifted_h = np.concatenate([np.zeros(-k), h_current[:k]])
                
                # Linear approximation:
                # h_i * (1 - h_{i+k}) ≈ h_current[i] * (1 - h_current[i+k]) + 
                #                        (h_i - h_current[i]) - (h_{i+k} * h_current[i])
                linear_term = 0
                for i in range(n_points):
                    if i + k >= 0 and i + k < n_points:
                        linear_term += (h_current[i] * (1 - h_current[i+k]) + 
                                       (h_vars[i] - h_current[i]) - 
                                       (h_current[i+k] * h_current[i])) * dx
                    else:
                        # Handle boundary conditions
                        linear_term += (h_current[i] * (1 - h_current[i+k]) + 
                                       (h_vars[i] - h_current[i])) * dx
                
                constraints.append(s >= linear_term)
            
            # Objective: minimize s
            objective = cp.Minimize(s)
            prob = cp.Problem(objective, constraints)
            
            # Try to solve with different solvers
            try:
                prob.solve(solver=cp.ECOS, max_iters=500, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    h_new = np.array(h_vars.value)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
                else:
                    # Fallback to gradient descent
                    h = gradient_descent_step(h, dx, start_time, budget_s)
            except:
                # Fallback to gradient descent
                h = gradient_descent_step(h, dx, start_time, budget_s)
                
        except:
            # Fallback to gradient descent
            h = gradient_descent_step(h, dx, start_time, budget_s)
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Occasionally switch to a different starting point
        if iteration % 100 == 0 and iteration > 0:
            # Add some random noise 
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def gradient_descent_step(h, dx, start_time, budget_s):
    """Gradient descent fallback"""
    if time.time() - start_time > budget_s - 1:
        return h
        
    step = 0.005
    iterations = 20
    h_temp = h.copy()
    
    for _ in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        corr = correlate(h_temp, 1 - h_temp, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        grad = np.zeros_like(h_temp)
        eps = 1e-4
        for i in range(len(h_temp)):
            h_plus = h_temp.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        h_temp -= step * grad
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (len(h_temp) / 2) / np.sum(h_temp)
    
    return h_temp
```
