import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 2000
    elite_size = population_size // 4
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population with variations
    for i in range(1, population_size):
        h = population[0].copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # ES parameters
    sigma = 0.1
    adapt_rate = 0.1
    success_count = 0
    success_threshold = population_size // 4
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(population):
        fit = evaluate_c5(h, dx)
        fitness.append(fit)
    
    # Find best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main ES loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        elite_population = [population[i] for i in elite_indices]
        elite_fitness = [fitness[i] for i in elite_indices]
        
        # Track success
        if iteration > 0 and min(elite_fitness) < best_c5:
            success_count += 1
        else:
            success_count = 0
        
        # Adapt sigma
        if success_count > success_threshold:
            sigma *= 1.2
        elif success_count < success_threshold // 2:
            sigma *= 0.8
        
        # Generate offspring
        offspring = []
        for i in range(population_size):
            # Select parent from elite
            parent_idx = np.random.choice(elite_indices)
            parent_h = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, sigma, n_points)
            mutated = parent_h + noise
            mutated = project_to_feasible(mutated, n_points)
            offspring.append(mutated)
        
        # Evaluate offspring
        offspring_fitness = []
        for i, h in enumerate(offspring):
            fit = evaluate_c5(h, dx)
            offspring_fitness.append(fit)
            
            # Update best
            if fit < best_c5:
                best_c5 = fit
                best_h = h.copy()
                if best_c5 <= 0.3807:
                    break
        
        # Replace population
        population = offspring
        fitness = offspring_fitness
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
