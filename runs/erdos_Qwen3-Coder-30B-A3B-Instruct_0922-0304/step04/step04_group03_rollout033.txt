import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5  # Uniform initialization
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    patience_counter = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        current_c5 = evaluate_c5(h, dx)
        
        # Compute gradient and update
        # For each k, we need to compute d(O_k) / dh_i
        # O_k = sum h_i * (1 - h_{i+k}) 
        # dO_k / dh_i = (1 - h_{i+k}) - h_{i-k}
        
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Compute O_k and its gradient
            # We treat each k separately, find which k gives max, then update 
            offset = k
            for i in range(n_points):
                grad[i] += (1 - h[(i + offset) % n_points]) - h[(i - offset) % n_points]
        
        # Apply gradient step
        h = h - learning_rate * grad
        
        # Project to feasible region [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize to meet constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Check improvement
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            # Reduce learning rate
            learning_rate *= 0.9
            patience_counter = 0
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
