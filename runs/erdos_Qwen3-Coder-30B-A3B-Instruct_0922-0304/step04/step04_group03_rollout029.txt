import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h_base = np.zeros(n_points)
        for i in range(n_points):
            h_base[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h_base = np.clip(h_base, 0, 1)
    
    # Normalize to satisfy integral constraint
    h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    # Define optimization parameters
    # We'll optimize a single continuous parameter that defines a two-step function
    # We use: h(x) = 1 for x in [a, a+w] U [b, b+w] with w=0.5, b=a+d
    # So we optimize a and d subject to constraints
    def construct_h(a, d):
        # a: starting point of first interval
        # d: distance between intervals
        w = 0.5  # width of intervals
        h = np.zeros(n_points)
        
        # First interval
        start1 = int(max(0, a / dx))
        end1 = int(min(n_points, (a + w) / dx))
        if end1 > start1:
            h[start1:end1] = 1.0
            
        # Second interval
        start2 = int(max(0, (a + d) / dx))
        end2 = int(min(n_points, (a + d + w) / dx))
        if end2 > start2:
            h[start2:end2] = 1.0
            
        # Normalize to unit integral
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def objective(params):
        a, d = params
        # Constraint checking
        if a < 0 or a > 1.5 or d < 0 or a + d + 0.5 > 2.0:
            return 1e10  # Large penalty for infeasible points
            
        h = construct_h(a, d)
        return evaluate_c5(h, dx)
    
    # Optimize using Nelder-Mead with several random starts
    best_val = float('inf')
    best_params = None
    best_h = None
    
    # Try multiple random initializations
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random start point for a and d
        a_init = np.random.uniform(0, 1)
        d_init = np.random.uniform(0, 1 - a_init - 0.5)
        
        try:
            result = minimize(objective, [a_init, d_init], method='Nelder-Mead', 
                            options={'maxiter': 500, 'disp': False})
            if result.success:
                val = objective(result.x)
                if val < best_val:
                    best_val = val
                    best_params = result.x.copy()
                    best_h = construct_h(best_params[0], best_params[1])
        except:
            continue
    
    # Final refinement - if we found a good solution, do a more focused search
    if best_h is not None and best_val < 0.38085:
        # Try a local grid search near best solution
        a_start, d_start = best_params
        for a in np.linspace(max(0, a_start - 0.2), min(1.5, a_start + 0.2), 20):
            for d in np.linspace(max(0, d_start - 0.2), min(1 - a - 0.5, d_start + 0.2), 20):
                if a + d + 0.5 <= 2.0:
                    try:
                        h = construct_h(a, d)
                        val = evaluate_c5(h, dx)
                        if val < best_val:
                            best_val = val
                            best_h = h
                    except:
                        continue
    
    # If no good solution found, return the original h_base
    if best_h is None:
        best_h = h_base.copy()
        best_val = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_val, n_points)
