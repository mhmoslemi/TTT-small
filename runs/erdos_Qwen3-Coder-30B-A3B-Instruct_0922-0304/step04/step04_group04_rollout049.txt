import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis construction
    def haar_basis(x, n):
        """Compute Haar wavelet basis functions up to scale 5"""
        basis = []
        # Add constant function
        basis.append(np.ones_like(x))
        # Generate Haar wavelets (simplified version for n=30)
        for i in range(1, min(n, 30)):
            # Simple piecewise constant approximation
            if i == 1:
                # Simple step
                wavelet = np.zeros_like(x)
                wavelet[x < 0.5] = 1.0
                wavelet[x >= 0.5] = -1.0
            else:
                # Approximate wavelet with 2^j intervals
                j = int(np.log2(i))
                if j < 5:  # up to 5 scales
                    wavelet = np.zeros_like(x)
                    interval = 1.0 / (2**j)
                    for k in range(2**j):
                        start = k * interval
                        end = (k+1) * interval
                        if k % 2 == 0:
                            wavelet[(x >= start) & (x < end)] = 1.0
                        else:
                            wavelet[(x >= start) & (x < end)] = -1.0
            basis.append(wavelet)
        return np.array(basis)
    
    # Time setup
    time_limit = start_time + budget_s - 2
    
    # Initialize with parent solution if available, otherwise random
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Normalize
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.random.rand(n_points)
        h0 = h0 * (n_points / 2) / np.sum(h0)

    # Create basis functions
    x = np.linspace(0, 2, n_points)
    # Simplified Haar basis (using 30 coefficients max)
    M = 30
    basis = haar_basis(x, M)
    basis = basis[:M]
    basis = basis.T  # Shape (n_points, M)
    
    # Initialize with a simplified approach from parent or random
    h_last = h0.copy()
    c_last = np.linalg.lstsq(basis, h_last, rcond=None)[0]  # coefficients
    best_c = c_last.copy()
    best_h = h_last.copy()
    
    # Point evaluation helper
    def eval_h(c):
        h = basis @ c
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Evaluate overlap
    def eval_c5(h_values):
        corr = correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    best_c5 = eval_c5(h_last)
    
    # Try optimization
    last_improvement = 0
    iteration = 0
    for iteration in range(15000):
        if time.time() > time_limit:
            break
            
        # Generate new candidate via random perturbation
        c_new = c_last + np.random.normal(0, 0.01, M)
        # Ensure c values are in [0,1]
        c_new = np.clip(c_new, 0, 1)
        # Renormalize
        h_new = eval_h(c_new)
        # Constraint
        integral = np.sum(h_new) * dx
        h_new = h_new * (n_points / 2) / integral
        
        c5_val = eval_c5(h_new)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_c = c_new.copy()
            best_h = h_new.copy()
            last_improvement = iteration
        
        c_last = c_new.copy()
        
        # Early stopping heuristic
        if iteration - last_improvement > 500 and best_c5 < 0.3807:
            break
    
    h_result = eval_h(best_c)
    c5_result = eval_c5(h_result)
    
    # Final validation
    integral_check = np.sum(h_result) * dx
    assert abs(integral_check - 1.0) < 1e-4, "Integral constraint violated"
    assert all(0 <= x <= 1 for x in h_result), "Support constraint violated"
    
    return (h_result.tolist(), c5_result, n_points)
