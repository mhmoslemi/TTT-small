import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap_full)
        max_shift = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at shift k_max, gradient[i] = (1 - h[(i+k_max)%n_points])
            # But we only need to consider the contribution from one shift
            shifted_idx = (i + max_shift) % n_points
            gradient[i] = 1.0 - h[shifted_idx]
            
            # Also subtract the contribution from the other side
            # This requires a bit more care - explicitly compute the gradient
            # For k_max, each element i contributes (1 - h[(i + k_max) % n_points])
            # The gradient w.r.t h[i] is 1 if h[(i + k_max) % n_points] appears, 
            # 0 otherwise - but we're taking a derivative of sum h[i] * (1 - h[i+k]) 
            # The contribution of h[i] to overlap at shift k is h[i] * (1 - h[i+k])
            # So dg/dh[i] = 1*(-1)*h[(i+k)%n_points] if k is relevant for "i"
            # But 0 otherwise; actually, for shift k_max it's 1 - h[(i + k_max) % n_points]  
            # Need to compute this properly now
            
        # Let's correctly compute gradient: 
        # o_k = sum_i h[i] * (1 - h[(i+k) % n_points]) = sum_i h[i] - sum_i h[i] * h[(i+k) % n_points]
        # For shift k_max, given that overlap_k_max = max, then taking derivative wrt h[i]: 
        # d(o_k_max)/dh[i] = 1 - h[(i+k_max)%n_points] if k_max is the shift maximizing overlap
        # We need to make sure the gradient is exact: o_k = sum_{i} h[i] * (1 - h[i+k]) 
        # d(o_k)/dh[i] = 1 - h[i+k]
        
        # So we compute gradient correctly: for the shift k_max that achieves maximum overlap 
        # note that 0 <= shift <= 2*n_points - 2 and k_max = shift - (n_points - 1)
        # So shift goes from 0 to 2*n_points - 2, k_max from -n_points + 1 to n_points - 1
        # The correct way to get gradient for shift k_max: 
        # d(overlap_k_max)/dh[i] = sum_{j: such that j+k_max = i mod n} (1 - h[j])  
        # Which means if k_max >= 0 then d.o./dh[i] = 1 - h[i - k_max]; 
        # if k_max < 0 then d.o./dh[i] = 1 - h[i + abs(k_max)]
        # However we work with the full correlation result which makes indexing trickier.
        
        # Just resample.
        # Direct calculation: for overlap with shift k, 
        # w.r.t h_i, we get contribution if term h_i * (1 - h_i+k) appears
        # So partial(i, k)  = 1 - h_i+k  if i+k is valid 
        # 
        # Let's define: rank(i) = i, rank(i+k) = i
        # We have overlap_k = sum_i h[i] * (1 - h[i-k]) 
        # d(overlap_k)/dh[i] = 1 - h[i-k]  when k_i = i-k \ Thus suffice to store 1 - h[i-k] for valid shift k  
        # If complement flag=1, then give 1-h else just 0
        
        # Let's do it completely carefully:
        # The expand principle for overlap with shift k is:
        # O_k = sum_i   h[i] * (1 - h[(i+k) % n])    // Add shift k to index i
        # dO_k / dh[i] = (1 - h[(i+k) % n])  
        
        # Compute directly the gradient for the shift achieving max:
        max_overlap_idx = np.argmax(overlap_full)
        shift_for_max = max_overlap_idx - (n_points - 1)
        
        # Now compute gradient with correct shift:
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + shift_for_max) % n_points  # Equivalent to adding k shift
            gradient[i] = 1.0 - h[j]
            
        # Apply gradient update
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive learning rate
        if iter_num > 0 and iter_num % 100 == 0:
            if current_c5 >= best_c5:
                learning_rate *= 0.9
                
        if current_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
