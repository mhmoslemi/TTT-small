import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness_scores = []
    
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness_scores.append(evaluate_c5(individual, dx))
    
    # GA iterations
    for generation in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness_scores = [evaluate_c5(ind, dx) for ind in population]
        
        # Track best
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_c5 = fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Tournament selection
        new_population = []
        
        for _ in range(population_size):
            # Tournament
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone winner
            child = population[winner_idx].copy()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                partner_idx = np.random.choice(population_size)
                if partner_idx != winner_idx:
                    crossover_point = np.random.randint(1, n_points - 1)
                    child[:crossover_point] = population[winner_idx][:crossover_point]
                    child[crossover_point:] = population[partner_idx][crossover_point:]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_point = np.random.randint(n_points)
                child[mutation_point] += np.random.normal(0, 0.05)
                child[mutation_point] = np.clip(child[mutation_point], 0, 1)
                
            # Rescale to maintain constraint
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
