import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def evaluate_population(population, dx):
    """Evaluate all individuals in the population."""
    fitness = []
    for h in population:
        c5 = compute_c5(h, dx)
        fitness.append(c5)
    return np.array(fitness)

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection."""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Perform single-point crossover."""
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.01):
    """Mutate individual with Gaussian noise."""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, sigma, len(h))
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, len(h) / 2)
    return h

def local_search(h, dx, max_iter=50):
    """Refine solution using local search (SLSQP)."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - len(h_vec) / 2
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    
    if result.success:
        h_new = result.x
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, len(h_new) / 2)
        return h_new
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, n_points / 2)
    else:
        init_h = np.ones(n_points)
        init_h[n_points//2:] = 0
        init_h += np.random.normal(0, 0.05, n_points)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, n_points / 2)

    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = init_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        population.append(h)

    # GA parameters
    best_c5 = float('inf')
    best_h = init_h.copy()

    # Main loop
    generation = 0
    while time.time() - start_time < budget_s - 1:
        if generation % 10 == 0:
            # Evaluate population
            fitness = evaluate_population(population, dx)
            
            # Update best solution
            current_best_idx = np.argmin(fitness)
            current_best_c5 = fitness[current_best_idx]
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = population[current_best_idx].copy()

            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break

        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Local Search on top performers
        top_indices = np.argsort(fitness)[:10]
        for idx in top_indices:
            new_population[idx] = local_search(population[idx], dx)
        
        population = new_population[:population_size]
        
        generation += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
