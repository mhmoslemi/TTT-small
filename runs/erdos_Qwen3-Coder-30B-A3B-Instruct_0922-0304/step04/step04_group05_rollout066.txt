import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population
    for i in range(1, pop_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best
    fitness = [evaluate_c5(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolutionary strategy
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = max(1, int(elite_ratio * pop_size))
        elite_indices = sorted_indices[:elite_size]
        
        # Select parents
        parents = [population[i] for i in elite_indices]
        parent_fitness = [fitness[i] for i in elite_indices]
        
        # Create offspring
        offspring = []
        for _ in range(pop_size - 1):
            # Select two random parents
            p1_idx = np.random.choice(elite_size)
            p2_idx = np.random.choice(elite_size)
            parent1 = parents[p1_idx]
            parent2 = parents[p2_idx]
            
            # Crossover (arithmetic)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            noise = np.random.normal(0, mutation_scale, n_points)
            child += noise
            
            # Project to feasible region
            child = project_feasible(child, n_points)
            offspring.append(child)
        
        # Add best parent as offspring
        offspring.append(parents[0].copy())
        
        # Evaluate offspring
        offspring_fitness = [evaluate_c5(h, dx) for h in offspring]
        
        # Selection: keep best pop_size individuals
        all_individuals = population + offspring
        all_fitness = fitness + offspring_fitness
        
        sorted_indices = np.argsort(all_fitness)
        selected_indices = sorted_indices[:pop_size]
        
        population = [all_individuals[i] for i in selected_indices]
        fitness = [all_fitness[i] for i in selected_indices]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Adaptive mutation scaling
        if iter_num % 10 == 0:
            improvement = np.mean(parent_fitness) - np.mean(fitness[:elite_size])
            if improvement > 0:
                mutation_scale *= 1.05
            else:
                mutation_scale *= 0.95
            mutation_scale = np.clip(mutation_scale, 0.001, 0.5)
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
