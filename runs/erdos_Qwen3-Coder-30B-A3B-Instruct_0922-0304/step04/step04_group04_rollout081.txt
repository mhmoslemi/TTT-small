import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Precompute basis functions
    x = np.linspace(0, 2, n_points)
    phi_basis = np.zeros((n_points, basis_size + 1))
    phi_basis[:, 0] = 1.0  # constant term
    
    for k in range(1, basis_size + 1):
        if k % 2 == 1:  # odd k -> sin
            phi_basis[:, k] = np.sin(k * np.pi * x / 2)
        else:  # even k -> cos
            phi_basis[:, k] = np.cos(k * np.pi * x / 2)

    # Compute integrals of basis functions
    int_phi = np.zeros(basis_size + 1)
    for k in range(basis_size + 1):
        int_phi[k] = np.trapz(phi_basis[:, k], x)

    # Initialize h with the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        c = (phi_basis.T @ h_init) * dx
        # Normalize to satisfy constraint
        c0 = c[0]
        if np.abs(int_phi[0]) > 1e-10:
            c0 = (1 - np.sum(c[1:] * int_phi[1:])) / int_phi[0]
            c[0] = c0
    else:
        # Use a simple initial guess
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        c = (phi_basis.T @ h_init) * dx
        # Normalize to satisfy constraint
        c0 = c[0]
        if np.abs(int_phi[0]) > 1e-10:
            c0 = (1 - np.sum(c[1:] * int_phi[1:])) / int_phi[0]
            c[0] = c0
    h = phi_basis @ c
    h = np.clip(h, 0, 1)
    
    # Construct kernel matrix
    K = np.zeros((basis_size + 1, basis_size + 1, 2 * n_points - 1))
    
    # Function to compute correlation coefficients efficiently using FFT
    def compute_kernel_coefficients():
        # Calculate integral: ∫ φ_i(x) φ_j(x + lag) dx
        for i in range(basis_size + 1):
            for j in range(basis_size + 1):
                for lag in range(2 * n_points - 1):
                    idx_lag = lag - (n_points - 1)
                    if -n_points + 1 <= idx_lag <= n_points - 1:
                        if idx_lag >= 0:
                            # φ_i(x) * φ_j(x + lag)
                            prod = phi_basis[:, i] * phi_basis[idx_lag:, j]
                        else:
                            # φ_i(x) * φ_j(x + lag)
                            prod = phi_basis[:, i] * phi_basis[:idx_lag + n_points, j]
                        K[i, j, lag] = np.trapz(prod, x[:len(prod)])
                    else:
                        K[i, j, lag] = 0.0

    compute_kernel_coefficients()
    
    # Convex optimization formulation
    c_vars = Variable(basis_size + 1)
    z_vars = Variable((basis_size + 1, basis_size + 1))
    
    # Constraints
    constraints = []

    # Integral constraint
    integral_con = int_phi[0] * c_vars[0] + sum(int_phi[i] * c_vars[i] for i in range(1, basis_size + 1)) == 1
    constraints.append(integral_con)
    
    # Variable bounds for c (non-negative constraint)
    for i in range(basis_size + 1):
        constraints.append(c_vars[i] >= 0)
        
    # McCormick envelopes for z_ij = c_i * c_j
    # z_ij >= c_i + c_j - 1
    # z_ij <= c_i
    # z_ij <= c_j
    
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            constraints.append(z_vars[i, j] >= c_vars[i] + c_vars[j] - 1)
            constraints.append(z_vars[i, j] <= c_vars[i])
            constraints.append(z_vars[i, j] <= c_vars[j])
            constraints.append(z_vars[i, j] >= 0)
            
    # Convex Objective
    max_overlap = 0
    max_val = 0
    retries = 0
    
    while retries < 3:
        # Construct overlap cost
        costs = []
        for lag in range(2 * n_points - 1):
            cost = 0
            for i in range(basis_size + 1):
                for j in range(basis_size + 1):
                    cost += z_vars[i, j] * K[i, j, lag]
            costs.append(cost)
            
        # Maximize absolute value of the sum
        max_overlap = Maximize(Maximize(costs) - Minimize(costs))
        
        # Create problem
        prob = Problem(max_overlap, constraints)
        
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=3000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                break
            else:
                retries += 1
        except:
            retries += 1
    
    # Extract best solution after optimization
    if retries < 3:
        c_opt = c_vars.value
    else:
        c_opt = c

    # Reconstruct h and ensure bounds
    h_opt = phi_basis @ c_opt
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Final check
    overlap = correlate(h_opt, 1 - h_opt, mode="full")
    max_overlap_val = np.max(overlap) * dx
    
    return (h_opt.tolist(), max_overlap_val, n_points)
