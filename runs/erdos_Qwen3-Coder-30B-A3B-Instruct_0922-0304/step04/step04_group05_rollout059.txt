import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using CVXPY"""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective: (1/2) * delta^T * H * delta + c^T * delta
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [A @ delta == b]
    
    # Bound constraints
    for i in range(n):
        constraints.append(delta[i] >= delta_low[i])
        constraints.append(delta[i] <= delta_high[i])
    
    # Form and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Precompute shift averages and correlation structures
    n_shifts = 2 * n_points - 1
    shift_matrix = np.zeros((n_points, n_points))
    for i in range(n_points):
        for j in range(n_points):
            shift = abs(i - j) if abs(i - j) <= n_points - 1 else n_points - 1 - abs(i - j)
            shift_matrix[i, j] = 1 if shift == 0 else 0
    
    # SCP loop
    max_iter = 500
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        # For consistency with overlap: h(x) * (1 - h(x+k))
        # Gradient: df/dh = dx * (1 - 2*h - shift_average)
        shift_avg = np.zeros(n_points)
        shift_avg[0] = 0
        for i in range(1, n_points):
            shift_avg[i] = np.mean(h[:-i] * (1 - h[i:])) * dx
        
        # Object is to minimize max_k C_k
        # Approximate gradient: constant across k is dx * (1 - 2*h)  
        gradient = dx * (1 - 2 * h)
        
        # Approximate Hessian: -2 * dx * I_matrix + some shift correlations
        # Use regularized version to maintain positive definiteness
        hessian = -2 * dx * np.eye(n_points) - dx * shift_matrix
        
        # Regularize diagonal to ensure PD
        reg = 1e-5
        for i in range(n_points):
            hessian[i, i] += reg
            
        # Lagrangian multiplier for equality constraint that sum(h) = n_points/2
        # Constraints: sum(delta) = 0
        A_eq = np.ones((1, n_points))
        b_eq = 0.0
        
        # Bound constraints: -1 <= delta <= 1 (relative to h)
        delta_low = np.maximum(-h, -1)
        delta_high = np.minimum(1 - h, 1)
        
        # Solve QP
        delta = _solve_qp(hessian, gradient, A_eq, b_eq, delta_low, delta_high)
        
        if delta is None:
            # Ad-hoc fallback: gradient descent with small step
            step = 0.01 * gradient
            delta = step
        
        #Line search
        alpha = 1.0
        max_alpha = 2.0
        improved = False
        while alpha > 1e-6:
            h_new = h + alpha * delta
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break 
            alpha *= 0.5

        if not improved and alpha <= 1e-6:
            # Resort to simple gradient descent if QP fails
            step = 0.01 * gradient
            h_new = h + step
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
