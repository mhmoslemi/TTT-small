import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to symmetric triangular window
        h = np.zeros(n_points)
        mid = n_points // 2
        half_width = n_points // 4
        h[mid - half_width:mid + half_width] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.05
    
    # Initialize particles
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            particles[i] = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        else:
            particles[i] = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # Evaluate initial particles
    fitness = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    
    # Initialize global best
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    best_c5 = gbest_fitness
    best_h = gbest.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness
            
            # Update global best
            if current_fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = current_fitness
                
                if gbest_fitness < best_c5:
                    best_c5 = gbest_fitness
                    best_h = gbest.copy()
        
        # Periodic reinitialization to avoid stagnation
        if iteration % 100 == 0 and iteration > 0:
            reinit_count = n_particles // 10
            reinit_indices = np.random.choice(n_particles, reinit_count, replace=False)
            for idx in reinit_indices:
                if 'initial_h_values' in globals():
                    particles[idx] = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                else:
                    particles[idx] = np.clip(np.random.random(n_points), 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
                fitness[idx] = evaluate_c5(particles[idx], dx)
                pbest[idx] = particles[idx].copy()
                pbest_fitness[idx] = fitness[idx]
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
