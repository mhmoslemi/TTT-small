import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the given construction if available, otherwise a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of the best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization with projected updates
    max_iter = 10000
    learning_rate = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)  # Convert to shift index
        
        # Compute gradient for the worst-case overlap
        grad = np.zeros(n_points)
        for i in range(n_points):
            # There are two terms in the gradient:
            # 1. For the term h[i] * (1 - h[(i + max_shift) % n_points])
            #    The derivative w.r.t. h[i] is (1 - h[(i + max_shift) % n_points])
            # 2. For the term h[(i - max_shift) % n_points] * (1 - h[i])
            #    The derivative w.r.t. h[i] is -h[(i - max_shift) % n_points]
            grad[i] = (1 - h[(i + max_shift) % n_points]) - h[(i - max_shift) % n_points]
        
        # Update step
        h_new = h - learning_rate * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate the new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept the new solution if it's better
        if new_c5 < best_c5:
            h = h_new
            best_h = h.copy()
            best_c5 = new_c5
            
        # Adaptive learning rate
        if iteration > 0 and iteration % 100 == 0:
            learning_rate *= 0.99
    
    return (best_h.tolist(), best_c5, n_points)
