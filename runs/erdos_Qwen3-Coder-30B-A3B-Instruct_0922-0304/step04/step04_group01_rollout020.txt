```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential Convex Programming approach
    max_iter = 1500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try to optimize using sequential convex programming (SCP)
        try:
            # Create variables
            h_var = cp.Variable(n_points)
            s = cp.Variable()
            
            # Constraints
            constraints = []
            
            # Integral constraint
            constraints.append(cp.sum(h_var) * dx == 1.0)
            
            # Bound constraints
            constraints.append(h_var >= 0)
            constraints.append(h_var <= 1)
            
            # Linearized overlap constraints
            h_current = h.copy()
            support = 2 * n_points - 1
            for k in range(-n_points + 1, n_points):
                # Compute linearization of correlation term
                # h[i] * (1 - h[i+k]) around current h
                linear_term = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    linear_term += (
                        h_current[i] * (1 - h_current[j]) +
                        (h_var[i] - h_current[i]) - 
                        (h_current[j] * (h_var[i] - h_current[i]))
                    )
                constraints.append(s >= linear_term * dx)
            
            # Objective
            objective = cp.Minimize(s)
            
            # Solve the convex problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=100)
            
            if prob.status == 'optimal':
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # Fallback to gradient descent
                step = 0.002
                h_temp = h.copy()
                for _ in range(20):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
            
        except:
            # Fallback to a variant of gradient descent
            step = 0.002
            h_temp = h.copy()
            for _ in range(20):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 50 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
