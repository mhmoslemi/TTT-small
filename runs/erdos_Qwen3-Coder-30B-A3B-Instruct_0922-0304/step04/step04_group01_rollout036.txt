import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to symmetric construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use adaptive gradient descent approach
    max_iter = 20000
    step_size = 0.005
    patience = 0
    best_improvement = 0.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_improvement = 0.0
            patience = 0
        else:
            patience += 1
            if patience > 100:
                step_size *= 0.95
                patience = 0
        
        # Compute gradients using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Store original to avoid modification during gradient calculation
        h_orig = h.copy()
        
        # For each point compute finite difference
        for i in range(n_points):
            h_plus = h_orig.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            
            # Re-normalize to maintain constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with gradient descent
        h = h - step_size * grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        
        # Ensure integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally add noise to escape local optima
        if iteration % 100 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early termination when target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final verification
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
