import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start gradient descent with SLSQP
    n_starts = 40
    max_iter = 300
    
    # Include the initial solution in our starts
    starts = [h.copy()]
    
    # Add random starts
    for _ in range(n_starts - 1):
        h_random = np.random.rand(n_points)
        h_random = np.clip(h_random, 0, 1)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        starts.append(h_random)
    
    # For each starting point, perform gradient descent
    for i, start_h in enumerate(starts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Define constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_arr):
            return np.sum(h_arr) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h_arr):
            return evaluate_c5(h_arr, dx)
        
        # Optimizer
        try:
            result = minimize(objective, start_h, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9})
            
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
                    
                if best_c5 <= 0.3807:
                    break
                    
        except Exception:
            pass
    
    # Final refinement of best solution
    if time.time() - start_time < budget_s * 0.98:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_arr):
            return np.sum(h_arr) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_arr):
            return evaluate_c5(h_arr, dx)
        
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': 100, 'ftol': 1e-10})
            
            if result.success:
                final_h = result.x
                final_c5 = evaluate_c5(final_h, dx)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h.copy()
        except Exception:
            pass
    
    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
