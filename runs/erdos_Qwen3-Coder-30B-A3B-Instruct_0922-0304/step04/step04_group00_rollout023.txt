import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.1
    elite_size = 5
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Main GA loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            f = compute_c5(individual)
            fitness.append(f)
            
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection: tournament selection
        selected = []
        tournament_size = 5
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])[0]
            selected.append(winner.copy())
            
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Create offspring
        while len(new_population) < pop_size:
            # Select two parents
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            
            # Crossover (single point)
            crossover_point = random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:
                if random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.01, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    # Ensure sum constraint
                    child = child * (n_points / 2) / np.sum(child)
                    
                new_population.append(child)
                
        # Local search on top individuals
        top_indices = np.argsort(fitness)[:10]
        for i in top_indices:
            temp_h = population[i].copy()
            try:
                result = minimize(
                    compute_c5, 
                    temp_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                if result.success:
                    temp_h = result.x
                    temp_h = np.clip(temp_h, 0, 1)
                    temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
                    if compute_c5(temp_h) < fitness[i]:
                        new_population[i] = temp_h
            except:
                pass
                
        population = new_population[:pop_size]
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
