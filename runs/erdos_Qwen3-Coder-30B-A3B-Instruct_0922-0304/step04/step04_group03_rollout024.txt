import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric piecewise constant function
        h = np.zeros(n_points)
        # Two intervals of width 0.5 each
        w = 0.5
        i0 = 0
        i1 = int(w / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        
        # Position second interval
        d = 0.5  # Distance between intervals
        i2 = int((w + d) / dx)
        i3 = int((2*w + d) / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
        
        if i2 <= i1:
            # Need to adjust d so intervals don't overlap
            d = w + dx
            i2 = int((w + d) / dx)
            i3 = int((2*w + d) / dx) - 1
            if i3 >= n_points:
                i3 = n_points - 1
        
        # Set intervals
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search for d parameter (distance between intervals)
    d_candidates = np.arange(0.0, 1.0 + 0.005, 0.005)  # Grid from 0 to 1 with step 0.005
    num_candidates = len(d_candidates)
    
    for i, d in enumerate(d_candidates):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create piecewise constant function with two intervals
        h_temp = np.zeros(n_points)
        
        w = 0.5  # Fixed width of each interval
        i0 = 0
        i1 = int(w / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
            
        i2 = int((w + d) / dx)
        i3 = int((2*w + d) / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
            
        if i2 <= i1:
            continue  # Prevent overlap
            
        h_temp[i0:i1+1] = 1
        h_temp[i2:i3+1] = 1
        
        # Normalize to satisfy constraint
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        # Evaluate
        c5 = evaluate_c5(h_temp, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Fine tune the best solution found by gradient descent
    h_tuned = best_h.copy()
    alpha = 0.01  # Learning rate
    max_iter = 500
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (approximate)
        c5_base = evaluate_c5(h_tuned, dx)
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_tuned.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_base) / eps
            
        # Update step (clip to feasible range and maintain constraint)
        h_tuned = h_tuned - alpha * grad
        h_tuned = np.clip(h_tuned, 0, 1)
        h_tuned = h_tuned * (n_points / 2) / np.sum(h_tuned)
        
        # Check improvement
        new_c5 = evaluate_c5(h_tuned, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_tuned.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
