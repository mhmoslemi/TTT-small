import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """
    Solve quadratic program using CVXPY
    min 0.5 * x' * H * x + c' * x
    s.t. A * x = b
         delta_low <= x <= delta_high
    """
    n = len(c)
    x = cp.Variable(n)
    
    # Objective
    obj = cp.Minimize(0.5 * cp.quad_form(x, H) + c @ x)
    
    # Constraints
    constr = [A @ x == b]
    constr += [x >= delta_low]
    constr += [x <= delta_high]
    
    # Solve
    prob = cp.Problem(obj, constr)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # For line search
    alpha_max = 1.0
    alpha_min = 0.01
    
    # Main iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradients and Hessian for the objective
        # We'll linearize around current h
        # Shift counts for auto-correlation
        n_shifts = 2 * n_points - 1
        C = np.zeros((n_shifts, n_points))
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    C[k, i] = h[i] * (1 - h[i + k])
                else:
                    C[k, i] = 0.0
        
        # Compute gradient and Hessian
        grad = np.zeros(n_points)
        hess = np.zeros((n_points, n_points))
        
        # Simple gradient estimation using finite differences
        for i in range(n_points):
            grad[i] = dx * (1 - 2*h[i] - np.sum(h[max(0,i-n_points+1):i]))
            
        # Hessian approximation from image error
        for i in range(n_points):
            for j in range(n_points):
                if i == j:
                    hess[i, j] = -2*dx
                else:
                    hess[i, j] = 0
                
        # Regularize
        reg = 1e-5
        hess += reg * np.eye(n_points)
        
        # Form constraint matrix for sum=1 (or rather, sum=1*2/70)
        A_eq = np.ones((1, n_points))
        b_eq = [n_points / 2]
        
        # delta bounds
        delta_low = -h
        delta_high = 1.0 - h
        
        # Solve QP
        delta = _solve_qp(hess, grad, A_eq, b_eq, delta_low, delta_high)
        if delta is None:
            # fallback: skipping update
            # Try to reduce stepsize
            alpha = 0.5 * alpha_max
        else:
            # Backtracking line search
            alpha = alpha_max
            failed = True
            while failed and alpha >= alpha_min:
                new_h = h + alpha * delta
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_c5(new_h, dx)
                if new_c5 < best_c5:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    failed = False
                else:
                    alpha *= 0.5
            if alpha < alpha_min:
                alpha = 0.5 * alpha_max
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
