import time

    start_time = time.time()

    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute cross-correlation of h with (1-h)
        conv = np.correlate(h, 1 - h, mode="full")
        # The maximum value is at the center of the correlation
        max_val = np.max(conv)
        return max_val * dx
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    T = T0
    max_iter = 10000
    
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor: perturb one element
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        step_size = 0.05 * T / T0  # Decreasing step size
        
        # Perturb one element
        h_new[idx] += np.random.normal(0, step_size)
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5 for new configuration
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        
        # Accept or reject
        if c5_new < c5_old:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improvement_count = 0
        else:
            # Accept worse solution with probability
            if np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                h = h_new
            no_improvement_count += 1
            
        # Cooling schedule
        T = T * alpha
        
        # Restart if no improvement for a while
        if no_improvement_count > max_no_improvement:
            h = best_h.copy()
            T = T0  # Reset temperature
            no_improvement_count = 0
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
