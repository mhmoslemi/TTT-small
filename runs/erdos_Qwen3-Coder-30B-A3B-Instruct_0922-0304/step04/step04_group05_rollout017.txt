import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale_init = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill rest of population with diverse candidates
    for i in range(1, pop_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Evolutionary strategy parameters
    mutation_scale = mutation_scale_init
    elite_size = max(1, int(elite_ratio * pop_size))
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Main ES loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Select elite
        elite = sorted_population[:elite_size]
        
        # Generate offspring with adaptive mutation
        offspring = []
        for _ in range(pop_size - len(elite)):
            # Select parent from elite
            parent = elite[np.random.randint(0, len(elite))]
            
            # Mutation with adaptive scale
            noise = np.random.normal(0, mutation_scale, n_points)
            mutated = parent + noise
            
            # Project to feasible region
            mutated = project_to_feasible(mutated, n_points)
            
            offspring.append(mutated)
        
        # Combine elite and offspring
        population = elite + offspring
        
        # Adapt mutation scale based on success
        if iter_num > 0 and iter_num % 10 == 0:
            if best_c5 < 0.381:
                mutation_scale *= 0.95
            else:
                mutation_scale *= 1.02
                
            # Keep mutation scale reasonable
            mutation_scale = np.clip(mutation_scale, 0.001, 0.3)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
