import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimizer parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Gradient descent with momentum
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient
        # For computational efficiency, compute gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[i] = (c5_plus - best_c5) / epsilon
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project onto feasible set: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
