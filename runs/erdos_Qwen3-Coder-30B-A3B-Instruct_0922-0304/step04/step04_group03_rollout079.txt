```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute correlation of h with (1-h) using full mode
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise use a symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric construction that satisfies the integral constraint
        h = np.zeros(n_points)
        # Two intervals of width 0.5 each, summing to 1
        # Let's make h 1 in [0.25, 0.75] and 0 elsewhere initially
        start1 = int(0.25 / dx)
        end1 = int(0.75 / dx)
        h[start1:end1] = 1.0
        # Normalize to satisfy integral = 1
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Define objective: maximize overlap, but since we minimize C5, 
    # we are interested in the minimum of max overlap
    def objective(params):
        if time.time() - start_time > budget_s - 1:
            return float('inf')
        a, d = params
        # Ensure a, d are within bounds
        if a < 0 or a > 1 - d - 0.5 or d < 0 or d > 1 - 1:
            return float('inf')
        # Create h based on parameters
        h = np.zeros(n_points)
        w = 0.5
        start1 = int(a / dx)
        end1 = int((a + w) / dx)
        start2 = int((a + w + d) / dx)
        end2 = int((a + w + d + w) / dx)
        # Clamp indices to valid range
        start1 = max(0, min(start1, n_points-1))
        end1 = max(0, min(end1, n_points))
        start2 = max(0, min(start2, n_points-1))
        end2 = max(0, min(end2, n_points))
        if end1 > start1:
            h[start1:end1] = 1.0
        if end2 > start2:
            h[start2:end2] = 1.0
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Grid search improvement
    best_a, best_d = 0.25, 0.25
    best_val = objective([best_a, best_d])
    
    # Try different starting values and parameter combinations
    a_test = np.linspace(0, 0.5, 10)
    d_test = np.linspace(0, 0.5, 10)
    
    for a in a_test:
        for d in d_test:
            if time.time() - start_time > budget_s - 1:
                break
            val = objective([a, d])
            if val < best_val:
                best_val = val
                best_a, best_d = a, d
        if time.time() - start_time > budget_s - 1:
            break
            
    # Refine around the best grid point using an optimizer
    # Initialize with the grid minimum, try to refine
    try:
        result = minimize(objective, [best_a, best_d], method='Nelder-Mead', 
                          options={'maxiter': 200, 'xatol': 1e-4, 'fatol': 1e-6})
        if not result.success:
            # Failed to improve, use grid result
            params = [best_a, best_d]
        else:
            params = result.x
            if params[0] < 0 or params[0] > 1 - params[1] - 0.5:
                params = [best_a, best_d]
            
        # Reconstruct h with optimized parameters
        h_opt = np.zeros(n_points)
        w = 0.5
        start1 = int(params[0] / dx)
        end1 = int((params[0] + w) / dx)
        start2 = int((params[0] + w + params[1]) / dx)
        end2 = int((params[0] + w + params[1] + w) / dx)
        
        start1 = max(0, min(start1, n_points-1))
        end1 = max(0, min(end1, n_points))
        start2 = max(0, min(start2, n_points-1))
        end2 = max(0, min(end2, n_points))
        if end1 > start1:
            h_opt[start1:end1] = 1.0
        if end2 > start2:
            h_opt[start2:end2] = 1.0
            
        # Normalize to integral = 1
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5_opt = evaluate_c5(h_opt, dx)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except:
        # If optimization fails, use the best from grid search
        best_val = objective([best_a, best_d])
        h_opt = np.zeros(n_points)
        w = 0.5
        start1 = int(best_a / dx)
        end1 = int((best_a + w) / dx)
        start2 = int((best_a + w + best_d) / dx)
        end2 = int((best_a + w + best_d + w) / dx)
        start1 = max(0, min(start1, n_points-1))
        end1 = max(0, min(end1, n_points))
        start2 = max(0, min(start2, n_points-1))
        end2 = max(0, min(end2, n_points))
        if end1 > start1:
            h_opt[start1:end1] = 1.0
        if end2 > start2:
            h_opt[start2:end2] = 1.0
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5_opt = evaluate_c5(h_opt, dx)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    return (best_h.tolist(), best_c5, n_points)
```
