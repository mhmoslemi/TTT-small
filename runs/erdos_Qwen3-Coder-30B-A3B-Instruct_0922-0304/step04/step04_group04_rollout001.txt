import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    max_iter = 2000
    tolerance = 1e-4
    alpha = 0.01
    
    # Create Fourier basis functions
    def fourier_basis(x, k):
        if k % 2 == 0:
            return np.cos(np.pi * k * x)
        else:
            return np.sin(np.pi * k * x)
    
    # Precompute basis functions and inner products
    x_vals = np.linspace(0, 2, n_points)
    basis = np.array([[fourier_basis(x, k) for x in x_vals] for k in range(n_basis)])
    
    # Compute integrals of basis functions
    basis_integrals = np.array([np.sum(basis[k]) * dx for k in range(n_basis)])
    
    # Set up optimization problem
    c = cp.Variable(n_basis)
    d = cp.Variable(n_basis)  # d_k = c_k * (1 - c_k)
    
    # McCormick envelopes for d_k = c_k * (1 - c_k)
    theta = cp.Variable(n_basis)
    constraints = [
        cp.norm(c, 1) <= 1.0,  # constraint on coefficients
        c >= 0,
        c <= 1,
        d >= c - theta,
        d >= theta - c + alpha,
        d <= 1 - theta,
    ]
    
    # Integral constraint
    constraints.append(cp.sum(cp.multiply(c, basis_integrals)) == 1)
    
    # Objective: minimize max overlap
    # This will be handled with a scalar variable
    max_overlap = cp.Variable()
    
    # The overlap integral in Fourier space is a complex quadratic form
    # We'll solve this by iteratively improving the solution from PSO
    h_initial = np.ones(n_points)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Project initial solution onto fourier bases
    proj_coeffs = np.linalg.lstsq(basis.T, h_initial, rcond=None)[0]
    proj_coeffs = np.clip(proj_coeffs, 0, 1)
    
    # Optimizing with CVXPY
    current_h = np.clip(h_initial, 0, 1)
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h, dx)
    
    # Main optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Project current h to fourier basis
        try:
            coeffs = np.linalg.lstsq(basis.T, best_h, rcond=None)[0]
            coeffs = np.clip(coeffs, 0, 1)
            
            # Solve with CVXPY for optimal coefficients
            c.value = coeffs
            prob = cp.Problem(cp.Minimize(max_overlap), constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                best_coeffs = c.value
                # Reconstruct h
                h_reconstructed = basis.T @ best_coeffs
                h_reconstructed = np.clip(h_reconstructed, 0, 1)
                
                # If this is better, keep it
                new_c5 = evaluate_c5(h_reconstructed, dx)
                if new_c5 < best_c5:
                    best_h = h_reconstructed.copy()
                    best_c5 = new_c5
                    
            # Regularization step: soft threshold and renormalize
            coeffs = np.clip(coeffs, 0, 1)
            coeffs = np.clip(coeffs, 0, 1)  # Keep in [0,1]
            coeffs = coeffs * (n_points / 2) / np.sum(coeffs * basis_integrals)
            
            h_new = basis.T @ coeffs
            h_new = np.clip(h_new, 0, 1)
            
            # Check if this improves our solution
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
        except:
            pass  # If there's a failure, continue iteration
            
        # Occasionally re-project from original h
        if iter_count % 50 == 0:
            h_reprojected = basis.T @ proj_coeffs
            h_reprojected = np.clip(h_reprojected, 0, 1)
            h_reprojected = h_reprojected * (n_points / 2) / np.sum(h_reprojected)
            test_c5 = evaluate_c5(h_reprojected, dx)
            if test_c5 < best_c5:
                best_h = h_reprojected.copy()
                best_c5 = test_c5
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
