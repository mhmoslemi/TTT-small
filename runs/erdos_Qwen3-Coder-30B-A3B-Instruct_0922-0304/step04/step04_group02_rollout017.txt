import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start gradient descent parameters
    n_random_initializations = 50
    n_gradient_steps = 200
    
    # Generate random initializations
    random_h = []
    for _ in range(n_random_initializations - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        random_h.append(h_rand)
    
    # Include the initial construct as starting point
    if 'initial_h_values' in globals():
        random_h.insert(0, h)
    else:
        random_h.insert(0, np.zeros(n_points))
        mid = n_points // 2
        random_h[0][:mid] = 1.0
        random_h[0] = random_h[0] * (n_points / 2) / np.sum(random_h[0])
    
    # MSGD loop
    for i, h_start in enumerate(random_h):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        try:
            result = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                            constraints=cons, options={'maxiter': n_gradient_steps, 'ftol': 1e-9})
            
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h
                    
                if best_c5 <= 0.3807:
                    break
        except:
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
