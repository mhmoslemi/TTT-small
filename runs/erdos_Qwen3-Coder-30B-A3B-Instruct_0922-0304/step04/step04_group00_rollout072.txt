import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    return h * (target_sum / current_sum)

def evaluate_population(population, dx):
    """Evaluate all individuals in population"""
    return [compute_c5(ind, dx) for ind in population]

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Single-point crossover between two parents"""
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.01):
    """Mutate an individual with Gaussian noise"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual += noise
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, len(individual) / 2)
    return individual

def local_search(h, dx, max_iter=50):
    """Refine h using local search (SLSQP)"""
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        return compute_c5(h, dx)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-8}
        )
        if result.success:
            h_new = result.x.reshape(n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, n_points / 2)
            return h_new
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    local_search_top = 10
    
    # Initialize population
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
    
    population = []
    for _ in range(pop_size):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Main GA loop
    max_generations = 1000
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = evaluate_population(population, dx)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            p1, p2 = selected[i], selected[min(i+1, len(selected)-1)]
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1, mutation_rate)
            c2 = mutate(c2, mutation_rate)
            new_population.extend([c1, c2])
        
        # Local search on top individuals
        top_indices = np.argsort(fitness)[:local_search_top]
        for idx in top_indices:
            if np.random.random() < 0.5:  # 50% chance to apply LS
                new_population.append(local_search(population[idx], dx))
        
        # Replacepopulation
        population = new_population[:pop_size]
    
    # Final evaluation and return best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
    else:
        final_c5 = compute_c5(population[0], dx)
        best_h = population[0]
        
    return (best_h.tolist(), final_c5, n_points)
