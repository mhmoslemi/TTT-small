import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using CVXPY"""
    n = len(c)
    delta = cp.Variable(n)
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    constraints = [
        A @ delta == b,
        delta >= delta_low,
        delta <= delta_high
    ]
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Set up constraints matrix
    A_eq = np.ones((1, n_points))
    b_eq = np.array([n_points / 2])
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Check if we've already achieved target
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Main SCP loop
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        # C_k = sum(h[i] * (1 - h[i+k]))
        # Gradient ∇C_k = dx * (1 - 2*h - shift_avg)
        # Hessian ∇²C_k = dx * ( -2*I + shift_matrix )
        
        # Compute for all shifts to get worst-case Hessian
        h_shifts = []
        for k in range(n_points):
            h_shift = np.zeros(n_points)
            if k == 0:
                h_shift = h
            elif k < n_points:
                h_shift[k:] = h[:-k]
            else:
                h_shift = np.zeros(n_points)
            h_shifts.append(h_shift)
        
        # Use one representative for gradient/Hessian (recompute final)
        # Faster to just directly compute 1 shift for the gradient
        C_grad = 1 - 2 * h
        for k in range(n_points):
            if k > 0:
                h_shift_k = np.zeros(n_points)
                h_shift_k[k:] = h[:-k]
                C_grad -= h_shift_k
        C_grad *= dx
        
        # Hessian approximation: -2*I + Symmetric shift matrix
        Hessian_diag = np.full(n_points, -2 * dx)
        Hessian = np.diag(Hessian_diag)
        
        # Add shift matrix components
        for k in range(1, n_points):
            if k < n_points:
                # Off-diagonal entries for shift k
                for i in range(n_points - k):
                    Hessian[i + k, i] += dx
                    Hessian[i, i + k] += dx
                    
        # Regularize
        regularization = 1e-5
        Hessian += regularization * np.eye(n_points)
        
        # Adjust regularization if necessary
        try:
            # Try solving QP with this Hessian
            A = A_eq
            b = b_eq
            delta_low = np.full(n_points, -1)
            delta_high = np.full(n_points, 1)
            
            # Enforce constraint Σh = n_points/2
            delta = _solve_qp(Hessian, C_grad, A, b[0], delta_low, delta_high)
            
            if delta is not None:
                # Line search for step size
                alpha_max = 1.0
                for j in range(n_points):
                    if delta[j] > 0:
                        alpha_max = min(alpha_max, (1 - h[j]) / delta[j])
                    elif delta[j] < 0:
                        alpha_max = min(alpha_max, h[j] / (-delta[j]))
                alpha = min(0.5, alpha_max)
                
                h_new = h + alpha * delta
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_c5(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
                    
                # Early termination
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
        except Exception as e:
            # Fall back to some perturbation
            pass
            
        # Occasionally resample h with slightly different values
        if iteration % 50 == 0 and iteration > 0:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.zeros(n_points)
                for i in range(n_points):
                    h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
                h += np.random.normal(0, 0.1, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
