import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    generations = 2000
    elite_size = population_size // 4
    mutation_rate = 0.1
    initial_sigma = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill rest of population
    for i in range(1, population_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # ES parameters
    sigma = initial_sigma
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Sort by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        for i in range(population_size - len(elite)):
            parent_idx = np.random.choice(elite_size)
            parent = elite[parent_idx]
            
            # Mutation with adaptive sigma
            noise = np.random.normal(0, sigma, n_points)
            child = parent + noise
            
            # Project to feasible region
            child = project_feasible(child, n_points)
            
            offspring.append(child)
            
        # Replace population with elite + offspring
        population = elite + offspring
        
        # Adapt sigma based on success
        if gen % 10 == 0:
            if gen > 0:
                if np.mean(fitness[:elite_size]) < np.mean(fitness[elite_size:]):
                    sigma *= 1.1  # Increase if successful
                else:
                    sigma *= 0.9  # Decrease if not successful
            sigma = max(0.001, min(0.5, sigma))
    
    return (best_h.tolist(), best_c5, n_points)
