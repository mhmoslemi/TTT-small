import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential Convex Programming approach
    max_iter = 1000
    alpha = 0.1
    damping = 0.5
    barrier_mu = 1e-3
    regularization = 1e-8
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient and Hessian approximations
        # For simplicity, approximate the gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            grad[i] = (evaluate_c5(h_plus, dx) - evaluate_c5(h_minus, dx)) / (2 * eps)
        
        # Formulate and solve convex subproblem using CVXOPT
        # Create decision variable
        delta = Variable(n_points)
        
        # Objective: gradient * delta + (1/2) * delta^T * H * delta
        # We'll use a quadratic approximation with diagonal Hessian for simplicity
        hessian_diag = np.ones(n_points) * regularization
        
        # Linear approximation to objective (quadratic in the current point) 
        objective = Minimize(grad.T @ delta + 0.5 * sum_squares(delta))
        
        # Constraints
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            sum(delta) == 0
        ]
        
        # Problem formulation
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=OSQP, verbose=False)
            delta_val = delta.value
            
            if delta_val is not None:
                # Update h
                h_new = h + alpha * delta_val
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Check convergence & adjust step size
                if iter_num > 0 and iter_num % 10 == 0:
                    if evaluate_c5(h, dx) > current_c5:
                        alpha *= damping
        except:
            # In case of failure, proceed with small random adjustment
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
