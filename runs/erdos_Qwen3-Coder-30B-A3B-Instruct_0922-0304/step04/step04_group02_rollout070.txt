import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Dual variables
    u = np.zeros(n_points)
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update h: minimize augmented Lagrangian
        # This is a simplified approximation of the optimization step
        # In practice, you'd solve the subproblem more carefully
        h_prev = h.copy()
        
        # Simple gradient descent step for h
        # Compute gradient of the overlap term (approximated)
        # and gradient of the constraints
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        grad = np.zeros_like(h)
        
        # Simple gradient approximation
        # We approximate the gradient of the overlap w.r.t h
        # by computing finite differences
        epsilon_val = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon_val
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_plus = np.max(overlap_plus)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon_val
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            max_minus = np.max(overlap_minus)
            
            grad[i] = (max_plus - max_minus) / (2 * epsilon_val)
        
        # Add Lagrange multiplier term
        grad += u
        
        # Update h
        h = h - 0.001 * grad
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Enforce constraint ∫h dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Update dual variables
        u = u + rho * (h - h_prev)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
