import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple initial guess
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    max_iter = 10000
    step_size = 0.005
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            # Compute O_k = sum h_i * (1 - h_{i+k})
            o = 0
            for i in range(n_points):
                j = (i + k) % n_points
                o += h[i] * (1 - h[j])
            overlaps[k] = o * dx
        
        # Find worst overlap
        k_max = np.argmax(overlaps)
        
        # Compute gradient of O_{k_max}
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = (1 - h[j])
            # Add conjugate term from shift -k_max
            j = (i - k_max) % n_points
            grad[i] -= h[j]
            
        # Update h
        h_new = h - step_size * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if this is better
        c5 = evaluate_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            
        h = h_new
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
