import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.1
    damping = 0.5
    mu = 1e-3
    beta = 10
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Convex approximation parameters
    h_prev = h.copy()
    grad = np.zeros(n_points)
    
    # Line search parameters
    c1 = 1e-4
    backtrack_factor = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current gradient
        # Compute gradient of C_k (simplified for implementation)
        # Using finite differences for approximate gradient
        epsilon = 1e-6
        grad = np.zeros(n_points)
        
        # Numerical gradient with respect to h
        current_c5 = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += epsilon
            h_eps = np.clip(h_eps, 0, 1)
            h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
            
            c5_eps = evaluate_c5(h_eps, dx)
            grad[i] = (c5_eps - current_c5) / epsilon
            
        # Formulate convex subproblem
        # We want to minimize: grad^T * delta + 0.5 * delta^T * H * delta
        # subject to:
        #   0 <= h + delta <= 1
        #   sum(h + delta) = n_points / 2
        
        # For simplicity, we consider a quadratic approximation using gradient operations
        
        # Create variables
        delta = cp.Variable(n_points)
        
        # Hessian approximation (simplified - diagonal with small regularization)
        reg = 1e-8
        H = np.diag(np.full(n_points, reg))
        
        # Objective
        obj = cp.Minimize(cp.quad_form(delta, H) + grad @ delta)
        
        # Constraints
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0  # since we want sum(h + delta) = sum(h) = n_points/2, delta sum should be 0
        ]
        
        # Create and solve the problem
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            delta_val = delta.value
            if delta_val is None:
                alpha *= damping
                continue
        except:
            # fallback to simple update
            delta_val = -alpha * grad
            delta_val = np.clip(delta_val, -h, 1 - h)
        
        # Line search
        try:
            while True:
                new_h = h + delta_val
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                new_c5 = evaluate_c5(new_h, dx)
                if new_c5 <= current_c5 + c1 * alpha * grad @ delta_val:
                    break
                else:
                    alpha *= backtrack_factor
                    if alpha < 1e-10:
                        break
        except:
            alpha *= damping
            continue
        
        # Update
        h = new_h
        current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Update parameters
        alpha *= damping
        if alpha < 1e-8:
            alpha = 1e-2
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
