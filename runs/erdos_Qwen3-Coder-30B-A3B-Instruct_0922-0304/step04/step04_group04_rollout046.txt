def build_haar_basis(n):
    """Build Haar wavelet basis functions up to scale 5."""
    basis = []
    # First level: 2 functions
    basis.append(lambda x: 1 if 0 <= x < 0.5 else (-1 if 0.5 <= x < 1 else 0))
    basis.append(lambda x: 1 if 0.5 <= x < 1 else (-1 if 1 <= x < 1.5 else 0))
    
    # Subsequent levels
    for scale in range(1, 5):
        for i in range(2**scale):
            def wavelet(x, i=i, scale=scale):
                if i % 2 == 0:
                    return 1 if (i/2**scale <= x < (i+1)/2**scale) else 0
                else:
                    return -1 if ((i-1)/2**scale <= x < i/2**scale) else 0
            basis.append(wavelet)
    
    return basis[:n]

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
