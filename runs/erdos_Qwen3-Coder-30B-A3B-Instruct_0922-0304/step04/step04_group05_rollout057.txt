import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def solve_qp(H, c, A, b, delta_low, delta_high, solver='CVXOPT'):
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective: minimize (1/2) * delta^T * H * delta + c^T * delta
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [
        A @ delta == b,
        delta >= delta_low,
        delta <= delta_high
    ]
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Ensure sum(h) = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian approximations
        # Gradient of overlap w.r.t. h
        g = 1 - 2 * h - np.convolve(h, np.ones(n_points), mode='full')[n_points-1:2*n_points-1]
        g = g * dx
        
        # Hessian of overlap (approximate)
        H = np.eye(n_points) * (-2 * dx)  # Diagonal terms
        for i in range(n_points):
            for j in range(n_points):
                if abs(i - j) < n_points:
                    H[i, j] += dx if i != j else dx  # Off-diagonal terms from second derivative composition
        
        # Add regularization to ensure positive semi-definiteness
        reg = 1e-5
        H += np.eye(n_points) * reg
        
        # Constraint A: sum of delta should be 0 to maintain constraint Σh = n_points/2
        A_eq = np.ones(n_points)
        b_eq = 0.0
        
        # Bounds for delta
        delta_low = -h
        delta_high = 1 - h
        
        # Solve QP
        delta = solve_qp(H, g, A_eq.reshape(1, -1), b_eq, delta_low, delta_high, 'CVXOPT')
        
        # If QP failed or no valid delta, fallback to simple gradient descent with small step
        if delta is None:
            step_size = 0.001
            delta = -step_size * g
            delta = np.clip(delta, delta_low, delta_high)
        
        # Line search - try different step sizes
        step_sizes = [0.1, 0.05, 0.01, 0.005, 0.001, 0.0005]
        best_step = 0.0
        best_h_trial = h.copy()
        for s in step_sizes:
            h_trial = h + s * delta
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h_trial = h_trial.copy()
                best_step = s
                
        # Apply update
        if best_step > 0:
            h = best_h_trial.copy()
            best_h = h.copy()
        else:
            # If no improvement, slightly reduce step size
            step_size = 0.00001
            delta = -step_size * g
            delta = np.clip(delta, delta_low, delta_high)
    
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
