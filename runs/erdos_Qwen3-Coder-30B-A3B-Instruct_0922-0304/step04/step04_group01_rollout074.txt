```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizing using Sequential Convex Programming (SCP)
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Linearization approach: form LP around current h
        try:
            # Define variables for the LP
            h_vars = cp.Variable(n_points)
            s = cp.Variable()
            
            # Constraints
            constraints = []
            # Integral constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Bounding constraints
            constraints.append(h_vars >= 0)
            constraints.append(h_vars <= 1)
            
            # Linearized overlap constraints
            for k in range(-n_points + 1, n_points):
                # Compute indices for circular shifts
                shifted_indices = np.arange(n_points) - k
                shifted_indices = shifted_indices % n_points
                
                # Linearization of h_i * (1 - h_{i+k})
                # Using h^{(n)}_i * (1 - h^{(n)}_{i+k}) + (h_i - h^{(n)}_i) - h_{i+k} * h^{(n)}_i
                # where h^{(n)} is the current estimate
                linearized_terms = []
                for i in range(n_points):
                    h_i_val = h[i]
                    h_k_val = h[shifted_indices[i]]
                    term = h_i_val * (1 - h_k_val) + (h_vars[i] - h_i_val) - h_k_val * h_i_val
                    linearized_terms.append(term)
                
                total = cp.sum(linearized_terms)
                constraints.append(s >= total * dx)
            
            # Objective: minimize s
            objective = cp.Minimize(s)
            prob = cp.Problem(objective, constraints)
            
            # Solve the LP
            prob.solve(solver='ECOS', verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to gradient descent
                step = 0.002
                h_temp = h.copy()
                for _ in range(10):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
                
        except:
            # Fallback to a variant of gradient descent
            step = 0.002
            h_temp = h.copy()
            for _ in range(10):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
