import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise create a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        d = 0.5  # distance between intervals
        i0 = 0
        i1 = int((w) / dx) - 1
        i2 = int((w + d) / dx)
        i3 = int((w + d + w) / dx) - 1
        if i1 >= 0:
            h[i0:i1+1] = 1
        if i3 >= i2:
            h[i2:i3+1] = 1
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search over possible distances between intervals
    d_candidates = np.arange(0.0, 1.01, 0.01)
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create step function with two intervals
        h_new = np.zeros(n_points)
        w = 0.5  # width of each interval
        i0 = 0
        i1 = int((w) / dx) - 1
        i2 = int((w + d) / dx)
        i3 = int((w + d + w) / dx) - 1
        
        if i1 >= 0:
            h_new[i0:i1+1] = 1
        if i3 >= i2:
            h_new[i2:i3+1] = 1
            
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5 = evaluate_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
