import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    w = 0.5  # width of each interval
    a = 0    # starting position (due to translation invariance)
    
    # Grid search over d (distance between intervals)
    d_candidates = np.arange(0, 1.001, 0.01)
    best_c5 = float('inf')
    best_h = None
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute interval indices
        i0 = 0
        i1 = int(np.floor((a + w) / dx)) - 1
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx)) - 1
        
        # Construct h
        h = np.zeros(n_points)
        if i1 >= i0:
            h[i0:i1+1] = 1
        if i3 >= i2:
            h[i2:i3+1] = 1
            
        # Normalize to satisfy constraint ∫h(x)dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute C5
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # If we're still close to the target, do a finer search around best d
    if best_c5 > 0.3807:
        d_candidates_fine = np.arange(max(0, best_h[1]-0.05), 
                                      min(1, best_h[1]+0.05), 0.001)
        for d in d_candidates_fine:
            if time.time() - start_time > budget_s - 1:
                break
                
            i0 = 0
            i1 = int(np.floor((a + w) / dx)) - 1
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx)) - 1
            
            h = np.zeros(n_points)
            if i1 >= i0:
                h[i0:i1+1] = 1
            if i3 >= i2:
                h[i2:i3+1] = 1
                
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
