import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles and velocities
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    pbest = np.zeros((n_particles, n_points))
    pbest_values = np.full(n_particles, float('inf'))
    
    # Initialize particles
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            particles[i] = h + np.random.normal(0, 0.01, n_points)
        else:
            # Create a triangular window-based initialization
            particles[i] = np.clip(np.array([max(0, 1 - abs(2*(j - n_points/2)/(n_points))) 
                                           for j in range(n_points)]) + 
                                 np.random.normal(0, 0.01, n_points), 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # Initialize personal bests
    for i in range(n_particles):
        pbest[i] = particles[i].copy()
        pbest_values[i] = evaluate_c5(particles[i], dx)
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_values)
    gbest = pbest[global_best_idx].copy()
    gbest_value = pbest_values[global_best_idx]
    
    best_c5 = gbest_value
    best_h = gbest.copy()
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest[i] = particles[i].copy()
            
            # Update global best
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest = particles[i].copy()
        
        # Update best solution seen
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = gbest.copy()
        
        # Occasionally reinitialize some particles
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                if 'initial_h_values' in globals():
                    particles[idx] = h + np.random.normal(0, 0.02, n_points)
                else:
                    particles[idx] = np.clip(np.array([max(0, 1 - abs(2*(j - n_points/2)/(n_points))) 
                                                     for j in range(n_points)]) + 
                                           np.random.normal(0, 0.02, n_points), 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
                pbest_values[idx] = evaluate_c5(particles[idx], dx)
                pbest[idx] = particles[idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
