def compute_haar_basis(n_basis, x_vals):
    """Compute Haar wavelet basis functions for given x values."""
    basis = np.zeros((n_basis, len(x_vals)))
    basis[0] = 1.0
    
    # Recursive construction of Haar wavelets
    level = 0
    while 2**level < n_basis:
        for i in range(2**level):
            start = i * 2**(1-level)
            end = start + 2**(1-level)
            if start < len(x_vals) and end <= len(x_vals):
                mid = (start + end) // 2
                basis[2**level + i, start:mid] = 1.0
                basis[2**level + i, mid:end] = -1.0
        level += 1
    
    return basis
