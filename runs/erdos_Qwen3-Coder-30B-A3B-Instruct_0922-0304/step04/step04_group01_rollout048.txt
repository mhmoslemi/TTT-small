```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use sequential convex programming approach
    max_iter = 2000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sequential convex programming approach
        try:
            # Build problem with linearized constraints
            h_prev = h.copy()
            s = cp.Variable()
            
            # Linearized constraints
            constraints = []
            
            # Compute overlap for current solution
            overlap_vals = correlate(h_prev, 1 - h_prev, mode="full")
            max_idx = np.argmax(overlap_vals)
            
            # Add main constraint
            h_vars = cp.Variable(n_points)
            
            # Compute linearized constraints for each shift
            for k in range(-n_points + 1, n_points):
                # Create linearization point for this shift
                linearized_overlap = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    # Linear approximation: h_i * (1 - h_j) + (h_i - h_i_prev) * (1 - h_j_prev) + ...
                    linearized_overlap += (h_prev[i] * (1 - h_prev[j]) + 
                                        (h_vars[i] - h_prev[i]) * (1 - h_prev[j]) + 
                                        h_prev[i] * (h_prev[j] - h_vars[j])) * dx
                
                constraints.append(s >= linearized_overlap)
            
            constraints.append(cp.sum(h_vars) * dx == 1.0)
            constraints.append(h_vars >= 0)
            constraints.append(h_vars <= 1)
            
            # Objective
            objective = cp.Minimize(s)
            prob = cp.Problem(objective, constraints)
            
            # Solve the convex relaxation
            try:
                prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_new = np.array(h_vars.value).reshape(-1)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
                else:
                    # Fall back to gradient-like method
                    step = 0.002
                    h_temp = h.copy()
                    for _ in range(20):
                        corr = correlate(h_temp, 1 - h_temp, mode="full")
                        max_idx = np.argmax(corr)
                        max_overlap = corr[max_idx] * dx
                        
                        grad = np.zeros_like(h_temp)
                        eps = 1e-4
                        for i in range(n_points):
                            h_plus = h_temp.copy()
                            h_plus[i] += eps
                            h_plus = np.clip(h_plus, 0, 1)
                            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                            c5_plus = np.max(corr_plus) * dx
                            grad[i] = (c5_plus - max_overlap) / eps
                        
                        h_temp -= step * grad
                        h_temp = np.clip(h_temp, 0, 1)
                        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                    h = h_temp
            except:
                # Fall back to gradient descent if convex solver fails
                step = 0.002
                h_temp = h.copy()
                for _ in range(20):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp

        except:
            # Fallback to simple gradient descent
            step = 0.002
            h_temp = h.copy()
            for _ in range(20):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 50 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
