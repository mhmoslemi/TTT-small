import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Symmetric triangular window construction
        h0 = np.array([min(i, n_points - i - 1) for i in range(n_points)])
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    for i in range(n_particles):
        particles[i] = h0 + np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    w = 0.9
    phi_p = 2.0
    phi_g = 2.0
    inertia_damp = 0.99
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Find global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_score
    
    # PSO iterations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                           phi_p * r_p * (pbest[i] - particles[i]) + 
                           phi_g * r_g * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest = particles[i].copy()
                    
                    if gbest_score < best_c5:
                        best_c5 = gbest_score
                        best_h = gbest.copy()
        
        # Occasionally reset stagnated particles
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                particles[idx] = h0 + np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
        
        # Apply inertia damping
        if iteration % 100 == 0 and iteration > 0:
            w *= inertia_damp
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
