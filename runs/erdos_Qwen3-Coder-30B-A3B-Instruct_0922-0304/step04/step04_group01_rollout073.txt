import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use sequential convex programming approach
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sequential convex programming approach
        try:
            # Start with a new LP problem each iteration
            h_prev = h.copy()
            h_vars = Variable(n_points)
            s = Variable()
            
            # Constraints
            constraints = [0 <= h_vars, h_vars <= 1, sum(h_vars) == n_points / 2]
            
            # Linearized constraints for all shifts
            for k in range(-n_points + 1, n_points):
                # Compute shift index
                shifted_indices = np.arange(n_points) - k
                shifted_indices = np.mod(shifted_indices, n_points)
                
                # Linearize: h[i]*(1 - h[i+k]) ≈ h_prev[i]*(1 - h_prev[i+k]) + (h[i] - h_prev[i]) - h[i+k]*h_prev[i]
                # But we need to be more careful with the linearization
                # Using the standard approximation
                linear_constraint = s >= 0
                for i in range(n_points):
                    if k >= 0:
                        j = i + k
                        if j < n_points:
                            linear_constraint = linear_constraint + h_vars[i] * (1 - h_vars[j]) >= h_prev[i] * (1 - h_prev[j]) + (h_vars[i] - h_prev[i]) - h_vars[j] * h_prev[i]
                    else:
                        j = i + k
                        if j >= 0:
                            linear_constraint = linear_constraint + h_vars[i] * (1 - h_vars[j]) >= h_prev[i] * (1 - h_prev[j]) + (h_vars[i] - h_prev[i]) - h_vars[j] * h_prev[i]
                
                constraints.append(linear_constraint)
            
            objective = Minimize(s)
            prob = Problem(objective, constraints)
            
            # Solve
            try:
                prob.solve(solver=SCIP, verbose=False, max_iters=100)
                h_new = np.array(h_vars.value).flatten()
                if not np.isnan(h_new).any() and np.isfinite(h_new).all():
                    h = h_new
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except:
                # Fall back to gradient descent if LP solver fails
                step = 0.002
                h_temp = h.copy()
                for _ in range(20):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
                
        except Exception as e:
            # Fallback to gradient descent method
            step = 0.002
            h_temp = h.copy()
            for _ in range(20):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 100 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
