import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_population(population, n_points, dx):
    return [compute_c5(h, dx) for h in population]

def selection(population, fitnesses, num_select):
    indices = np.argsort(fitnesses)[:num_select]
    return [population[i] for i in indices]

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = n // 2
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.01):
    mask = np.random.random(len(h)) < mutation_rate
    h = h + np.random.normal(0, sigma, len(h)) * mask
    return h

def local_search(h, n_points, dx, max_iter=50):
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    
    if result.success:
        return normalize_h(result.x, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    population.append(base_h.copy())
    
    for _ in range(population_size - 1):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = normalize_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitnesses = evaluate_population(population, n_points, dx)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            # Restart with new random population
            population = []
            for _ in range(population_size):
                h = np.random.uniform(0, 1, n_points)
                h = normalize_h(h, n_points)
                population.append(h)
            no_improvement_count = 0
            continue
            
        # Selection (top 50%)
        num_select = population_size // 2
        selected = selection(population, fitnesses, num_select)
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover
        while len(new_population) < population_size:
            if len(selected) >= 2:
                idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
                child1, child2 = crossover(selected[idx1], selected[idx2])
                child1 = normalize_h(child1, n_points)
                child2 = normalize_h(child2, n_points)
                new_population.extend([child1, child2])
            else:
                # Fallback to mutation
                h = selected[0] + np.random.normal(0, 0.02, n_points)
                h = normalize_h(h, n_points)
                new_population.append(h)
        
        new_population = new_population[:population_size]
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < 0.1:
                new_population[i] = mutate(new_population[i])
                new_population[i] = normalize_h(new_population[i], n_points)
                
        # Local search on top solutions
        top_k = min(5, len(new_population))
        top_indices = np.argsort([compute_c5(h, dx) for h in new_population])[:top_k]
        for i in top_indices:
            new_population[i] = local_search(new_population[i], n_points, dx)
            new_population[i] = normalize_h(new_population[i], n_points)
            
        population = new_population
        
    # Final best solution check
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to base solution
        final_h = normalize_h(base_h, n_points)
        final_c5 = compute_c5(final_h, dx)
        return (final_h.tolist(), final_c5, n_points)
