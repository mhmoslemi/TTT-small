import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale = 0.1
    adaptive_factor = 0.95
    success_threshold = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population
    for i in range(1, pop_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Track global best
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Evolutionary Strategy loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        order = np.argsort(fitness)
        fitness = [fitness[i] for i in order]
        population = [population[i] for i in order]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite_size = max(1, int(pop_size * elite_ratio))
        elite = population[:elite_size]
        
        # Mutate elite to form new population
        new_population = elite.copy()
        success_count = 0
        
        for i in range(pop_size - elite_size):
            # Select parent from elite
            parent = elite[np.random.randint(0, len(elite))]
            
            # Generate offspring with mutation
            noise = np.random.normal(0, mutation_scale, n_points)
            offspring = parent + noise
            
            # Project to feasible region
            offspring = project_to_feasible(offspring, n_points)
            
            # Track improvement
            offspring_c5 = evaluate_c5(offspring, dx)
            if offspring_c5 <= fitness[0]:
                success_count += 1
                
            new_population.append(offspring)
        
        # Adaptive mutation scaling
        if success_count / (pop_size - elite_size) > success_threshold:
            mutation_scale *= 1.1
        else:
            mutation_scale *= adaptive_factor
            
        population = new_population
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
