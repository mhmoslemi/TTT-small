import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def gradient_step(h, dx, target_sum):
    # Compute gradients for each shift k
    n = len(h)
    grad = np.zeros(n)
    
    # For overlap integral: C_k = dx * sum h[i] * (1 - h[i+k])
    # Gradient w.r.t. h[i]: dx * (1 - 2*h[i] - sum_{k: i+k < n} h[i+k])
    
    # Efficiently compute the gradient
    for i in range(n):
        # Sum h[i+k] for valid k values (i+k < n)
        grad[i] = 1.0 - 2.0 * h[i]
        for k in range(1, n - i):
            grad[i] -= h[i + k]
    
    # Apply dx
    grad *= dx
    
    # Update with negative gradient direction, but project it back
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    alpha = 0.1
    beta = 0.5
    tolerance = 1e-6
    
    # Initialize with available construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1.0 - 2.0 * abs(i - mid) / n_points)
    
    # Normalize to meet sum constraint
    h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and update
        grad = gradient_step(h, dx, target_sum)
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, target_sum)
        
        # Evaluate the new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Line search
        while new_c5 >= best_c5 and alpha > 1e-8:
            alpha *= beta
            h_new = h - alpha * grad
            h_new = project_to_feasible(h_new, target_sum)
            new_c5 = evaluate_c5(h_new, dx)
        
        # Update if better
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
