import numpy as np
from scipy.signal import correlate
import time
import math

def compute_c5(h, dx):
    """Compute the C5 overlap value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    T = T0
    max_iter = 10000
    step_size = 0.05
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Save current state
        current_h = h.copy()
        current_c5 = compute_c5(h, dx)
        
        # Generate neighbor solution
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        perturbation = np.random.normal(0, step_size)
        h_new[idx] += perturbation
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            c5 = new_c5
        else:
            # Accept with probability
            delta = new_c5 - current_c5
            if np.random.rand() < math.exp(-delta / T):

        # Early exit if target achieved

        # Update parameters
        T *= alpha
        step_size *= alpha
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
