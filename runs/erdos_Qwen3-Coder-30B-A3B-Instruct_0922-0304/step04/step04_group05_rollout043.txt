import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 2000
    elite_fraction = 0.25
    initial_sigma = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
    else:
        # Create triangular function
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
    
    population.append(h0.copy())
    
    # Fill remaining with random feasible solutions
    for _ in range(population_size - 1):
        h = np.random.random(n_points) * 0.5
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Evolutionary strategy variables
    sigma = initial_sigma
    adaptive_factor = 0.95
    success_count = 0
    success_threshold = population_size * elite_fraction
    
    # Evaluate initial population
    fitness = []
    for h in population:
        f = evaluate_c5(h, dx)
        fitness.append(f)
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:int(population_size * elite_fraction)]
        
        # Create offspring
        offspring = []
        offspring_fitness = []
        
        for _ in range(population_size):
            # Select parent from elite
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, sigma, n_points)
            child = parent + noise
            child = project_to_feasible(child, n_points)
            
            # Evaluate child
            child_fitness = evaluate_c5(child, dx)
            offspring.append(child)
            offspring_fitness.append(child_fitness)
            
            # Update best if needed
            if child_fitness < best_c5:
                best_h = child.copy()
                best_c5 = child_fitness
                if best_c5 <= 0.3807:
                    break
        
        # Update population
        population = offspring
        fitness = offspring_fitness
        
        # Adapt sigma based on success
        success_count += 1
        if success_count >= success_threshold:
            # Increase sigma if beneficial
            success_count = 0
            sigma *= 1.1
        else:
            # Decrease sigma
            sigma *= adaptive_factor
            
        # Clamp sigma
        sigma = max(0.001, min(0.5, sigma))
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
