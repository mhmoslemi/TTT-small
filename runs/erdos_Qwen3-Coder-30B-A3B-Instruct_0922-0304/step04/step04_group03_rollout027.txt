import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute full correlation of h and (1-h)
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = np.clip(h_start, 0, 1)
    else:
        # Initialize with a simple symmetric construction
        h_start = np.zeros(n_points)
        mid = n_points // 2
        # Set first half to step up, second to step down
        for i in range(mid):
            h_start[i] = i / (mid - 1)
        for i in range(mid, n_points):
            h_start[i] = (n_points - 1 - i) / (n_points - mid - 1) if n_points - mid - 1 > 0 else 0
        # Smooth and clip
        h_start = np.clip(h_start, 0, 1)
        
    # Normalize to satisfy ∫h(x)dx = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Define function to be minimized: max overlap
    def objective(params):
        # params: [a, d] where h is 1 on [a, a+w] and [a+w+d, a+w+d+w]
        a = params[0]
        d = params[1] 
        w = 0.5  # Since 2*w = 1
        
        if a < 0 or a > 1 - d - w or d < 0 or d > 1 - 2*w:
            return float('inf')
            
        # Create step function with two intervals
        h = np.zeros(n_points)
        start1 = int(round(a / dx))
        end1 = int(round((a + w) / dx))
        start2 = int(round((a + w + d) / dx))
        end2 = int(round((a + w + d + w) / dx))
        
        if end1 < n_points:
            h[start1:end1] = 1.0
        if end2 < n_points:
            h[start2:end2] = 1.0
            
        # Normalize to integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        return evaluate_c5(h, dx)
    
    # Search grid approach to avoid local minima
    best_val = float('inf')
    best_h = h_start.copy()
    
    # Grid search with refinements
    a_vals = np.linspace(0.01, 0.99, 20)
    d_vals = np.linspace(0.01, 0.99, 20)
    
    for a in a_vals:
        for d in d_vals:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                val = objective([a, d])
                if val < best_val:
                    best_val = val
                    # Recompute h for precise values at the best point
                    w = 0.5
                    h = np.zeros(n_points)
                    start1 = int(round(a / dx))
                    end1 = int(round((a + w) / dx))
                    start2 = int(round((a + w + d) / dx))
                    end2 = int(round((a + w + d + w) / dx))
                    if end1 < n_points:
                        h[start1:end1] = 1.0
                    if end2 < n_points:
                        h[start2:end2] = 1.0
                    h = h * (n_points / 2) / np.sum(h)
                    best_h = h.copy()
            except:
                continue
    
    # Final local refinement with scipy
    def refine_objective(params):
        a, d = params
        if a < 0 or a > 0.99 or d < 0 or d > 0.99:
            return float('inf')
        w = 0.5
        h = np.zeros(n_points)
        start1 = int(round(max(0, a) / dx))
        end1 = int(round(min(n_points, a + w) / dx))
        if end1 > start1:
            h[start1:end1] = 1.0
        start2 = int(round(max(0, a + w + d) / dx))
        end2 = int(round(min(n_points, a + w + d + w) / dx)) 
        if end2 > start2:
            h[start2:end2] = 1.0
            
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Use Nelder-Mead for local refinement
    x0 = [0.3, 0.3]  # initial guess
    try:
        res = minimize(refine_objective, x0, method='Nelder-Mead', options={'maxiter': 500})
        if res.success and res.fun < best_val:
            a, d = res.x
            w = 0.5
            h = np.zeros(n_points)
            start1 = int(round(max(0, a) / dx))
            end1 = int(round(min(n_points, a + w) / dx))
            if end1 > start1:
                h[start1:end1] = 1.0
            start2 = int(round(max(0, a + w + d) / dx))
            end2 = int(round(min(n_points, a + w + d + w) / dx))
            if end2 > start2:
                h[start2:end2] = 1.0
            h = h * (n_points / 2) / np.sum(h)
            best_h = h.copy()
            best_val = res.fun
    except:
        pass
    
    return (best_h.tolist(), best_val, n_points)
