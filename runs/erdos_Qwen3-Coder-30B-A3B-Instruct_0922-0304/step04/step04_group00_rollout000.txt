import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=2.0/100):
    """Compute the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        pos = h.copy() + np.random.normal(0, 0.02, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        pbest.append(pos.copy())
        pbest_vals.append(compute_c5(pos, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_val = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    
                    if gbest_val <= 0.3807:
                        break
                        
        iter_count += 1
        
        if gbest_val <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
