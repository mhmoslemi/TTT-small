import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points, target_sum):
    """Project h to satisfy sum constraint and bound constraints"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / n_points)
    return h

def compute_gradient(h, dx, n_points):
    """Compute the gradient of C₅ with respect to h"""
    # Compute all overlaps for different shifts
    # We want gradient of max_k int h(x)(1-h(x+k)) dx
    # This is a complex derivative; we'll use numerical approx of first order
    # For the objective, we approximate the gradient of the max-overlap term
    # A more detailed approach follows:
    # Let's compute a good approximation
    
    # Make a worst-case estimate of the gradient based on current max overlap
    
    # Compute all shifts of h
    h_shifted = np.zeros((n_points, n_points))
    for k in range(n_points):
        h_shifted[:, k] = np.roll(h, k)
    
    # Compute overlaps for each shift
    overlap_values = np.zeros(n_points)
    for k in range(n_points):
        # Compute int h(x) * (1 - h(x+k)) dx
        if k == 0:
            overlap_values[k] = np.sum(h * (1 - h)) * dx
        else:
            overlap_values[k] = np.sum(h * (1 - h_shifted[:,k])) * dx
    
    # Identify the maximal overlap and its shift
    max_idx = np.argmax(overlap_values)
    max_val = overlap_values[max_idx]

    # d/dh ( sum h[i] * (1-h[i+k]) ) = (1 - 2 * h[i] - h_shifted[i, k]) 
    # Note: this assumes no wraparound in shifts, and h is treated as periodic?

    # Let's take gradient that considers the overlap with shift k we are maximizing over
    # Gradient of h[i] in the overlap for shift k:
    # d/dh[i] sum h[j] * (1 - h[j+k]) = (1 - 2*h[i]) if j==i and j+k=i (i.e j=i-k) 
    # And for each j, if j=i+(shift), then it's -h[i]
    # More directly: for shift k, d/dh[i] sum h[j] * (1-h[j+k]) = 1*h[i] - h[i+k] 

    #this: (1 - 2*h[i]) shared with position i-k if i-k is valid, and -h[i] at position i-k also
    #edit:  let's be very direct for now:
    grad = np.zeros(n_points)
    for i in range(n_points):
        # consider the shift that produces the largest overlap
        j = (i - max_idx) % n_points  # this is misleading, let us compute more appropriately
        # for each k, compute derivative of overlap k with respect to h[i]
        # The overlap is: sum_{j=0}^{n-1} h[j] (1 - h[j+k])  
        # d/dh[i] this is: (1 - h[i+k] - h[i]) = 1 - h[i+k] - 1 (if j==i)
        # But should be at correct index of h[i-k] (if valid)
        # Be careful of index wrapping in sliding window.
        # It's the sum of h[j] * (1 - h[j+k]) over all j, so for fixed i, only h[i] contributes:
        # d/dh[i] h[i] * (h[i+k]) = 1 - h[i+k]
        # Overall d/dh[i] sum_{j=0}^{n-1} h[j] (1 - h[j+k]) 
        # = (1 - h[i+k]) if the shift leads to an i
        # = 0 if i is not shifted to i from any j
        # = 1 - h[i+k] if j = (i-k) mod n (but here we consider effect of h[i] on h[j] at some shift k)
        # We're thinking about the i-th h[i], to modify overlap where h[j] contributes where j=i
        # i.e., the term h[j] * (1 - h[j+k]) involves h[i] when j = i
        
        # Special case for overlap k, derivative w.r.t h[i]
        # d/dh[i] sum h[j] * (1 - h[j+k]) = 
        # if j = i, we get (1 - h[i+k])  
        # but j runs over all indices from 0 to n-1, so...
        
        # Actually, the general form:
        # d/dh[i] (sum_{j=0}^{n-1} h[j] (1 - h[j+k])) 
        # = d/dh[i] (h[i] (1 - h[i+k])) if j = i
        # = 1 - h[i+k] in general
        # But this is the derivative at k=0, because h[i] is in integral over all j.

        # gradient for k where max overlap occurs: for terms where h[i] is involved
        # this is 1 - h[i+k] where h[i](1 - h[i+k]) is the term in contribution to overlap with shift k
        # As we want derivatives of sums h[i](1 - h[i+k]) over i, for fixed k:
        # For fixed i, d/dh[i] sum_{j} h[j](1 - h[j+k]) = if j == i, then 1 - h[i+k]

        # For k = 1, h[i] contributes to overlap at location h[i] where we have term h[i-1](1-h[i])  
        # so d/dh[i] this becomes: d/dh[i] h[i-1](1-h[i])

        # Within overlap k, contribution of h[i] to sum = h[i] * (1 - h[i-k])

        # but all terms in the give range are of the form:
        # h[i] * (1 - h[i+k]) = 1 - h[i+k]  at position i
        # That's it. 
        # Not the gradient of what's under the maximum!
        
        # najlepsze aproksymacja: LETS SIMPLY ESTIMATE GRADIENT FOR OVERLAP AT TH MAXIMAL K
        # In fact:
        # DERIVATIVE w.r.t. h[i] of overlap_k = sum_j h[j] * (1 - h[j+k])
        # = 1 - h[i-k] at i
        
        # So gradient[i] = (1 - h[i-k_max])  (which is 1 - h[i-k]) at each i
        # But if k_max maximises overlap, k is fixed. This is a very rough estimate. 
        # For now, let's try a simple direct approximation 

        # The better approach for optimization, to see which direction to change:
        # Compute overlap matrix 
        # We'll take that one overlap, k_max, which is the largest and is most important
        # and compute gradient for this one.

        # compute gradient of the overlap with shift max_idx
        grad[i] = 1 - h[(i - max_idx) % n_points]

    # Normalize or use as is
    return grad * dx  # dx is factor

    
def line_search_alpha(h, grad, step_size_factor=1.0):
    """Do Armijo line search for alpha"""
    # Let's try Armijo line search
    alpha = 1.0 * step_size_factor
    c1 = 1e-4
    max_alpha = 1.0
    min_alpha = 1e-15
    while alpha >= min_alpha:
        # try h_proposed = h - alpha * grad
        h_proposed = h - alpha * grad
        h_proposed = project_to_feasible(h_proposed, len(h), len(h) / 2)
        
        # max_overlap_new = compute_max_overlap(h_proposed)
        max_overlap_new = evaluate_c5(h_proposed, 1.0/len(h))
        # at current h:
        max_overlap_old = evaluate_c5(h, 1.0/len(h))
        
        # Armijo condition:
        # f(h - alpha * grad) <= f(h) - c1 * alpha * grad^T * grad
        # But here we're trying to minimize, so actually we want:
        # f(h - alpha * grad) <= f(h) - c1 * alpha * ||grad||^2
        # Compute left and right
        lhs = max_overlap_new
        rhs = max_overlap_old - c1 * alpha * np.sum(grad**2)
        
        if lhs <= rhs or np.isnan(lhs) or np.isnan(rhs):
            return alpha
        else:
            alpha *= 0.5
    return alpha

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize solution
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        # triangular function from 1 to 0 over half the domain, then 0
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 1.0 - (i / mid)
        h = project_to_feasible(h, n_points, target_sum)
    
    # Optimizer state
    max_iter = 1000
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute gradient of C₅ w.r.t h
        grad = compute_gradient(h, dx, n_points)
        
        # Line search for optimal alpha
        alpha = line_search_alpha(h, grad, 1.0)
        
        # Update h
        h = h - alpha * grad
        h = project_to_feasible(h, n_points, target_sum)
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Early stopping
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
