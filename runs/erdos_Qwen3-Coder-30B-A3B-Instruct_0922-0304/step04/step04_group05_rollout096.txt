import numpy as np
from scipy.signal import correlate
import time

def compute_overlap(h, dx):
    """Compute the maximum overlap integral"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular function as starting point
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    
    # Gradient-based improvement with alternating steps
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Step 1: Local magnitude adjustment
        # Move towards lower overlap by slightly perturbing h locally
        h_new = h.copy()
        for i in range(n_points):
            # Try small adjustments to improve overlap
            candidates = [h[i] * 0.95, h[i] * 1.05, h[i]]
            best_h_i = h[i]
            best_overlap_i = best_c5
            
            for val in candidates:
                if 0 <= val <= 1:
                    h_test = h_new.copy()
                    h_test[i] = val
                    # Renormalize
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    overlap = compute_overlap(h_test, dx)
                    if overlap < best_overlap_i:
                        best_overlap_i = overlap
                        best_h_i = val
                        
            h_new[i] = best_h_i
            
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        overlap_new = compute_overlap(h_new, dx)
        
        # Step 2: Support shifting
        if overlap_new < best_c5:
            h = h_new
            best_c5 = overlap_new
            best_h = h.copy()
        else:
            # Try shifting support regions
            h_shifted = h.copy()
            # Randomly shift some regions
            for _ in range(5):
                start_idx = np.random.randint(0, n_points - 5)
                end_idx = min(start_idx + 5, n_points)
                # Shift the region by 1 position
                if start_idx + 1 < n_points:
                    h_shifted[start_idx:end_idx] = h[start_idx+1:end_idx+1]
                    # Handle wraparound
                    if end_idx == n_points:
                        h_shifted[n_points-1] = h[0]
                    # Normalize
                    h_shifted = h_shifted * (n_points / 2) / np.sum(h_shifted)
                    new_overlap = compute_overlap(h_shifted, dx)
                    if new_overlap < best_c5:
                        h = h_shifted.copy()
                        best_c5 = new_overlap
                        best_h = h.copy()
                        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
