import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral for a given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum, clip values to [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_population(population, dx):
    """Evaluate all individuals in population."""
    return [compute_c5(individual, dx) for individual in population]

def tournament_selection(population, fitness, k=5):
    """Select individuals using tournament selection."""
    selected = []
    for _ in range(len(population) // 2):
        indices = np.random.choice(len(population), k, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Single-point crossover of two parents."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.01):
    """Apply Gaussian mutation to an individual."""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual = individual + noise
    return individual

def local_search(h, dx, max_iter=50):
    """Refine solution using gradient-based optimization."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - 1.0  # N/2 = 1.0 in normalized form

    bounds = [(0, 1) for _ in range(len(h))]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    
    if result.success:
        return result.x
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = 1.0  # Normalized sum (since sum(h) * dx = 1 implies sum(h) = N/2 = 1.0)

    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = normalize_h(base_h, target_sum)
    
    # Create initial population
    for _ in range(population_size):
        h = base_h + np.random.normal(0, 0.01, n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # GA loop
    max_generations = 1000
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = evaluate_population(population, dx)
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                child1 = normalize_h(mutate(child1), target_sum)
                child2 = normalize_h(mutate(child2), target_sum)
                new_population.extend([child1, child2])
            else:
                # Handle odd population size
                mutated = normalize_h(mutate(selected[i]), target_sum)
                new_population.append(mutated)
        
        # Local search
        top_k = min(10, len(new_population))
        sorted_indices = np.argsort([compute_c5(ind, dx) for ind in new_population])[:top_k]
        for i in sorted_indices:
            refined = local_search(new_population[i], dx)
            new_population[i] = normalize_h(refined, target_sum)
        
        # Replace population
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
