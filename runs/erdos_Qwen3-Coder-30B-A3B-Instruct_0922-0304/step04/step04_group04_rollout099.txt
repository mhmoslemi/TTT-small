import numpy as np

def run(seed=42, budget_s=600, **kwargs):

    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis functions
    def haar_wavelets(n_basis):
        # Generate Haar wavelet basis functions up to n_basis
        basis = []
        # Constant function
        basis.append(np.ones(n_points))
        # Haar wavelets
        for i in range(1, n_basis):
            wavelet = np.zeros(n_points)
            # Determine level and position
            level = int(np.log2(i+1))
            pos = i - 2**level
            # Calculate support
            start = pos * 2**(1-level)
            end = (pos+1) * 2**(1-level)
            # Create wavelet
            mid = (start + end) / 2
            for j in range(n_points):
                x = j * dx
                if start <= x < mid:
                    wavelet[j] = 1
                elif mid <= x < end:
                    wavelet[j] = -1
            basis.append(wavelet)
        return np.array(basis)
    
    # Make sure we have an initial solution

        initial_h += np.random.normal(0, 0.05, n_points)
        initial_h = np.clip(initial_h, 0, 1)
        
    # Normalize the initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Project initial_h to Haar wavelet coefficients
    wavelets = haar_wavelets(30)
    # Compute Gram matrix of wavelets
    G = np.dot(wavelets, wavelets.T) * dx
    # Compute inner products with initial_h
    b = np.dot(wavelets, initial_h) * dx

    coeffs = np.linalg.solve(G, b)

    c = Variable(30)
    z = Variable(30 * 30)  # McCormick products

    h_coeffs = np.array([0.0] * 30)
    for i in range(30):
        h_coeffs[i] = coeffs[i]
    
    # Compute kernel matrix for integral: ∫ h(x)(1 - h(x + k)) dx
    # This will be more direct with coefficients
    # For now, minimize maximum of overlap function over all possible k
    
    # Define McCormick envelopes
    # z_ij >= c_i + c_j - 1
    # z_ij <= c_i
    # z_ij <= c_j
    
    # Update constraints to match problem formulation
    constraints = []
    # Integral constraint
    integral_constraint = sum(c) * dx == 1.0
    constraints.append(integral_constraint)

    # Degree of freedom - we will solve by gradient descent for more effective search
    
    # Use scipy minimize for more structured numerical optimization
    def obj_function(coeffs):
        # Reconstruct h(x) in function space
        h = np.dot(wavelets.T, coeffs) * (dx * n_points / 2) / np.sum(coeffs)
        # Clamp between [0,1] as in feasibility check
        h = np.clip(h, 0, 1)
        
        # Compute overlap  
        overlap = correlate(h, 1 - h, mode='full')
        # Scale appropriately
        max_overlap = np.max(overlap) * dx
        return max_overlap
        
    def optimize_step():
        # Use gradient-based method
        def objective(c):
            return obj_function(c)
        
        # Initial guess from our wavelet projection
        c_init = coeffs.copy()
        
        # Solve using L-BFGS-B method
        from scipy.optimize import minimize
        result = minimize(objective, c_init, method='L-BFGS-B',

                          options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6})

            final_coeffs = result.x

            final_coeffs = c_init

        h = np.dot(wavelets.T, final_coeffs) * (dx * n_points / 2) / np.sum(final_coeffs)
        h = np.clip(h, 0, 1)
        
        # Renormalize to meet integral constraint exactly
        h = h * (n_points / 2) / np.sum(h)
        
        overlap = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(overlap) * dx
        
        return h, max_overlap, final_coeffs

    best_h, best_c5, best_coeffs = optimize_step()

    best_c5 = float('inf')

        overlap = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(overlap) * dx
        
    return (best_h.tolist(), float(best_c5), n_points)
