import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimized heuristic with pattern-based gradient descent
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing inspired approach
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    
    max_iter = 3000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor solution
        h_new = h.copy()
        
        # Random perturbation on some indices
        perturb_indices = np.random.choice(n_points, size=n_points//10, replace=False)
        for i in perturb_indices:
            h_new[i] += np.random.normal(0, 0.03)
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute overlap
        current_c5 = evaluate_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        # Accept with probability based on simulated annealing
        delta = current_c5 - evaluate_c5(h, dx)
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new.copy()
        
        # Update temperature
        temp = max(min_temp, temp * cooling_rate)
        
        # Occasionally apply gradient descent
        if iteration % 50 == 0 and iteration > 0:
            step = 0.005
            h_temp = h.copy()
            for _ in range(10):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
