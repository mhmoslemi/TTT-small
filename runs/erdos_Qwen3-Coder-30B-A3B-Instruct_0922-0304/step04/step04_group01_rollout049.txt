import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using sequential convex programming
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sequential Convex Programming approach
        try:
            # Solve a relaxed LP problem
            h_prev = h.copy()
            
            # Build linearized constraints
            s = cp.Variable()
            h_var = cp.Variable(n_points)
            
            # Constraint: sum h * dx = 1
            constraints = [cp.sum(h_var) == n_points / 2]
            
            # Constraint: 0 <= h_var <= 1
            constraints += [h_var >= 0, h_var <= 1]
            
            # Linearized overlap constraints
            for k in range(-n_points + 1, n_points):
                if k >= 0:
                    # h[i] * (1 - h[i+k]) ~ h_prev[i] * (1 - h_prev[i+k]) + (h[i] - h_prev[i]) - h_prev[i+k] * (h[i] - h_prev[i])
                    # For simplicity, use:
                    # sum_{i} h[i] * (1 - h[i+k]) <= s
                    # Linear approximation:
                    # sum_{i} [h_prev[i] * (1 - h_prev[i+k]) + (h[i] - h_prev[i]) - h_prev[i+k] * (h[i] - h_prev[i])] <= s
                    # Simplifies to:
                    # sum_{i} h_prev[i] * (1 - h_prev[i+k]) + sum_{i} (h[i] - h_prev[i]) - sum_{i} h_prev[i+k] * (h[i] - h_prev[i]) <= s
                    # sum_{i} h_prev[i] * (1 - h_prev[i+k]) + sum_{i} h[i] - sum_{i} h_prev[i] - sum_{i} h_prev[i+k] * h[i] + sum_{i} h_prev[i+k] * h_prev[i] <= s
                    # ... but even simpler, just sum_{i} h[i] * (1 - h[i+k]) <= s as before, but we linearize it:
                    
                    # Linearized version:
                    # h[i] * (1 - h[i+k]) <= h_prev[i]* (1 - h_prev[i+k]) + (h[i] - h_prev[i]) - h_prev[i+k] * (h[i] - h_prev[i])
                    # Rearranging:
                    # h[i] - h[i+k]*h[i] <= h_prev[i] - h_prev[i+k] + (h[i] - h_prev[i]) - h_prev[i+k] * (h[i] - h_prev[i])
                    # h[i] - h[i+k]*h[i] <= h_prev[i] - h_prev[i+k] + h[i] - h_prev[i] - h_prev[i+k]*h[i] + h_prev[i+k]*h_prev[i]
                    # - h[i+k]*h[i] <= - h_prev[i+k] - h_prev[i+k]*h[i] + h_prev[i+k]*h_prev[i]
                    # - h[i+k]*h[i] + h_prev[i+k]*h[i] <= - h_prev[i+k] + h_prev[i+k]*h_prev[i]
                    # h[i] * (h_prev[i+k] - h[i+k]) <= h_prev[i+k] * (h_prev[i] - 1)
                    
                    # Actually building the constraint as vectorized linear form:
                    # a_k * h <= b_k where sum_{i} h[i] * (1 - h[i+k]) <= s is approximated as:
                    # sum_{i} h_prev[i] * (1 - h_prev[i+k]) + sum_{i} (h[i] - h_prev[i]) - sum_{i} h_prev[i+k] * (h[i] - h_prev[i]) <= s
                    # This gives us a linear constraint:
                    # sum_{i} (h_prev[i] - h_prev[i+k]) * h[i] <= sum_{i} h_prev[i] * (1 - h_prev[i+k]) + sum_{i} h_prev[i] - sum_{i} h_prev[i+k] * h_prev[i] - s
                    
                    # But also using `h[i]*h[i+k]` linearization:
                    # Let's manually sum for weights (just evaluating):
                    
                    # We can just evaluate using simple approximations
                    ind1 = np.arange(n_points)
                    ind2 = (ind1 + k) % n_points
                    
                    # Avoid too many compute-intensive operations during optimization
                    # Approximated overlap term (constraint): h[i]*(1-h[i+k]) <= s
                    
                    # Optimize just the max feature and ignore fine linearization
                    # For now, let's just do gradient-based optimization refinement:
                    pass
                    
            # We'll just use the computed overlap constraints directly:
            # Evaluate overlap matrix and use Lagrangian-style approx.
            # Fast approximation with just gradient descent on gradient step:
            step = 0.002
            h_temp = h.copy()
            # Compute overlap values in current iteration
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
                
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
            
        except Exception as e:
            # Fallback to simple gradient descent
            step = 0.003
            h_temp = h.copy()
            for _ in range(15):  # Reduced iterations
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic or restart
        if iteration % 30 == 0 and iteration > 0:
            # Add some variance to prevent local minimum
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
