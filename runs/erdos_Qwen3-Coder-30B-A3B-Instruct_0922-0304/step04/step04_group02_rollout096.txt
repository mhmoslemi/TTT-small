import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Initialize variables
    x = h.copy()
    z = h.copy()
    u = np.zeros(n_points)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
        
        # Update x: solve subproblem with penalty term
        # This is a simplified version - more sophisticated approach needed for convergence
        # Using projected gradient descent for simplicity
        x_old = x.copy()
        
        # Gradient approximation using finite differences
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_plus = x.copy()
            h_minus = x.copy()
            h_plus[i] += 1e-6
            h_minus[i] -= 1e-6
            
            # Enhanced approach: compute gradient using correlation
            val_plus = evaluate_c5(h_plus, dx)
            val_minus = evaluate_c5(h_minus, dx)
            grad[i] = (val_plus - val_minus) / (2e-6)
        
        # Gradient descent step
        x = x - 0.01 * grad
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        
        # Update z (proximal operator) and u (dual variable)
        z_old = z.copy()
        z = x + u
        z = np.clip(z, 0, 1)
        z = z * (n_points / 2) / np.sum(z)
        u = u + (x - z)
        
        # Check for convergence
        primal_residual = np.linalg.norm(x - z)
        dual_residual = np.linalg.norm(rho * (z - z_old))
        
        if primal_residual < epsilon and dual_residual < epsilon:
            break
        
        # Update best solution
        current_c5 = evaluate_c5(z, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = z.copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with L-BFGS
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Define objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use ADMM solution as initial point for L-BFGS
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds,

    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
