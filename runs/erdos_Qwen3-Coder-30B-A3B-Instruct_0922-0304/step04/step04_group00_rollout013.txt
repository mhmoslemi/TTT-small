import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5(h, dx):
    """Compute the C5 overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    T = T0
    max_iter = 5000
    step_size = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor solution
        h_new = h.copy()
        idx = random.randint(0, n_points - 1)
        delta = random.uniform(-step_size, step_size)
        h_new[idx] += delta
        
        # Ensure element-wise bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the new solution
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            # Accept with probability based on temperature
            if random.random() < np.exp(-(new_c5 - best_c5) / T):
                h = h_new
                
        # Update temperature and step size
        T *= alpha
        step_size *= alpha
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
