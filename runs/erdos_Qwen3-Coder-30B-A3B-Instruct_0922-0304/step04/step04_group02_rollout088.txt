import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolutionary Strategy parameters
    pop_size = 100
    elite_size = int(0.2 * pop_size)
    mutation_strength = 0.1
    max_iter = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # ES iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            fit = evaluate_c5(ind, dx)
            fitness.append(fit)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Selection (elite + tournament)
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        offspring.extend(elite)
        
        # Generate new individuals through mutation
        while len(offspring) < pop_size:
            # Select parent
            parent = population[np.random.randint(0, pop_size//2)]
            
            # Mutate
            mutant = parent.copy()
            # Mutate 30-50% of elements
            n_mutate = np.random.randint(int(0.3 * n_points), int(0.5 * n_points))
            indices = np.random.choice(n_points, n_mutate, replace=False)
            for i in indices:
                mutant[i] += np.random.normal(0, mutation_strength)
            
            # Repair constraints
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            offspring.append(mutant)
        
        # Update population
        population = offspring
        
        # Adaptive mutation strength
        if iter_count % 50 == 0 and iter_count > 0:
            improvement = np.mean(fitness[:elite_size]) - np.mean(fitness[elite_size:2*elite_size])
            if improvement < 0.0001:
                mutation_strength = min(0.5, mutation_strength * 1.1)
            else:
                mutation_strength = max(0.001, mutation_strength * 0.9)
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
