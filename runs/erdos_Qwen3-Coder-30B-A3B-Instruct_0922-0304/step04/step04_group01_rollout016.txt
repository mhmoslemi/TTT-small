import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback construction
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles around the starting solution
    for i in range(n_particles):
        particles[i] = h_init + np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    w = 0.9
    phi_p = 2.0
    phi_g = 2.0
    inertia_damping = 0.99
    reinitialization_freq = 100
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_values = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_value
    
    # PSO optimization loop
    max_iter = int(1e4)
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           phi_p * r_p * (pbest[i] - particles[i]) + 
                           phi_g * r_g * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate current particle
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_c5
                    if gbest_value < best_c5:
                        best_c5 = gbest_value
                        best_h = gbest.copy()
        
        # Reinitialize some particles periodically
        if iteration % reinitialization_freq == 0 and iteration > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                particles[idx] = h_init + np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
                pbest[idx] = particles[idx].copy()
                pbest_values[idx] = evaluate_c5(particles[idx], dx)
        
        # Apply inertia damping
        if iteration % 100 == 0 and iteration > 0:
            w *= inertia_damping
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
