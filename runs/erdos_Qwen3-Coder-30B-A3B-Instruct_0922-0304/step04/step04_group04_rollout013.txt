import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Try to use the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a smooth initial function that's close to optimal
        h_init = np.zeros(n_points)
        mid = n_points // 2
        # Gentle ramp up and down
        for i in range(n_points):
            if i < mid:
                h_init[i] = 2 * i / mid
            else:
                h_init[i] = 2 * (n_points - i) / (n_points - mid)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Optimization with gradient-based method on smoothed version
    max_iter = 1000
    lr = 0.001
    decay = 0.995
    eps = 1e-8
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a smooth version with some noise
        h_smooth = best_h.copy()
        if iter_count > 0:
            # Add some gradient-based update
            grad = np.zeros(n_points)
            # Compute gradient of overlap w.r.t. h 
            # (simplified approximation)
            for i in range(n_points):
                # Simple finite difference approximation
                eps_val = 1e-4
                h_plus = h_smooth.copy()
                h_minus = h_smooth.copy()
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                # Ensure feasible region
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                # Renormalize
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                c5_plus = evaluate_c5(h_plus, dx)
                c5_minus = evaluate_c5(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * eps_val)
            
            # Update with gradient and momentum (simple version)
            # We'll do gradient descent on the max overlap
            h_smooth = h_smooth - lr * grad
            h_smooth = np.clip(h_smooth, 0, 1)
            h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
            
        # Sample discrete points from smooth function
        h_new = np.copy(h_smooth)
        
        # Ensure constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Reduce learning rate
        lr *= decay
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
    return (best_h.tolist(), best_c5, n_points)
