import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    rho = 1.0
    max_iter = 2000
    epsilon = 1e-4
    
    # Initialize variables
    x = h.copy()
    z = np.zeros(2 * n_points - 1)  # For dual variables
    u = np.zeros(2 * n_points - 1)
    
    # Main ADMM loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update x (minimize augmented Lagrangian)
        # This involves solving a convex optimization subproblem
        # We will use projected gradient descent for this
        x_old = x.copy()
        
        # Compute gradient: This is a simplified version
        # We estimate that h needs to be adjusted to minimize overlap
        # For this we will perform a few gradient steps
        grad = np.zeros_like(x)
        
        # Compute current overlap for gradient estimation
        overlap_current = correlate(x, 1 - x, mode="full")
        
        # Approximate gradient using finite differences for simplicity
        eps = 1e-4
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = np.clip(x_plus, 0, 1)
            x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
            
            overlap_plus = correlate(x_plus, 1 - x_plus, mode="full") 
            grad[i] = (np.max(overlap_plus) - np.max(overlap_current)) / eps
        
        # Gradient descent step with projection
        lr = 0.01
        x = x - lr * grad
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        
        # Update z (consensus variable)
        z_new = x + u
        # Soft thresholding on z - but we will just keep values within bounds
        z_new = np.clip(z_new, 0, 1)
        z_new = z_new * (n_points / 2) / np.sum(z_new)
        z = z_new
        
        # Update dual variable u
        u = u + (x - z)
        
        # Evaluate current solution
        if iteration % 5 == 0:
            current_c5 = evaluate_c5(x, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x.copy()
                
            # Early stopping
            if best_c5 <= 0.3807:
                break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
