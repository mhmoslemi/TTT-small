import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Create initial population
    population = []
    for _ in range(pop_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # GA iterations
    for generation in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness (inverse of C5)
        fitness_scores = []
        for individual in population:
            c5 = evaluate_c5(individual, dx)
            fitness_scores.append(1.0 / (1e-10 + c5))
            
        # Tournament selection
        new_population = []
        for _ in range(pop_size):
            # Tournament
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            
            # Clone and possibly mutate
            child = population[winner_idx].copy()
            
            # Crossover with another random individual
            if np.random.rand() < crossover_rate:
                other_idx = np.random.choice(pop_size)
                crossover_point = np.random.randint(1, n_points - 1)
                child[:crossover_point] = population[other_idx][:crossover_point]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_strength = 0.1
                mutation = np.random.normal(0, mutation_strength, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
            new_population.append(child)
            
        population = new_population
        # Keep track of best individual in current generation
        for individual in population:
            c5 = evaluate_c5(individual, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
                
        if best_c5 <= 0.3807:
            break
            
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.98:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
