import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using CVXPY"""
    try:
        n = len(c)
        delta = cp.Variable(n)
        objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
        constraints = [
            A @ delta == b,
            delta >= delta_low,
            delta <= delta_high
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Constraints for QP: sum(delta) = 0
    A_eq = np.ones(n_points).reshape(1, -1)
    b_eq = 0.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Precompute shift averaging for efficiency
    def compute_shift_averages(h):
        """Computes average of h over all possible shifts"""
        shifts = []
        for k in range(n_points):
            shifted_h = np.roll(h, k)
            shifts.append(shifted_h)
        return np.array(shifts).mean(axis=0)
    
    # Main SCP loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient and hessian approximations
        # C_k = dx * sum h[i] * (1 - h[i+k])
        # For each k, C_k is quadratic:
        # C_k = dx * ( sum h[i] - sum h[i] * h[i+k] )
        # ∇C_k = dx * ( 1 - 2*h - shift_averages )
        # Hessian: ∇²C_k = dx * ( -2*I + shift_matrix )
        
        # Use approximate gradient and Hessian at current h
        # Compute shift averages of h
        shift_avg = compute_shift_averages(h) if it > 0 else np.zeros(n_points)
        
        # Gradient: df/dh = dx * (1 - 2*h - shift_avg)
        grad = dx * (1 - 2 * h - shift_avg)
        
        # Hessian: second derivative of C_k w.r.t h[i]
        # This is a complicated tensor, but we use a regularized local approximation
        # Generic hessian for a sum of quadratic functions
        H = np.zeros((n_points, n_points))
        # Off-diagonal terms: -dx
        H += -dx * np.eye(n_points) # For -a * h[i]*h[j] where a=2dx when j=i, 0 otherwise
        # Diagonal terms: add dx * 2 (from -2 h[i])
        H.flat[::n_points+1] += 2 * dx
        
        # Regularize Hessian to ensure positive definiteness
        H += 1e-5 * np.eye(n_points)
        
        # QP with constraint sum(delta) = 0
        delta = _solve_qp(H, grad, A_eq, b_eq, -1.0, 1.0)
        
        if delta is None:
            # If QP fails, try a smaller step
            step = 0.1
        else:
            # Line search for step size
            step = 1.0
            while step > 1e-6:
                h_new = h + step * delta
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    break
                step *= 0.5
            if step <= 1e-6:
                h = h + 0.01 * delta
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                best_c5 = evaluate_c5(h, dx)
                best_h = h.copy()
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
