import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    max_iter = 2000
    tolerance = 1e-4
    alpha = 0.01
    
    # Fourier basis functions
    def fourier_basis(x, k):
        if k % 2 == 0:
            return np.cos(np.pi * k * x)
        else:
            return np.sin(np.pi * k * x)
    
    # Compute inner products
    A = np.zeros((n_basis, n_basis))
    b = np.zeros(n_basis)
    
    x_vals = np.linspace(0, 2, n_points)
    
    # Compute basis integrals
    for i in range(n_basis):
        for j in range(n_basis):
            val = 0
            for k in range(n_points):
                val += fourier_basis(x_vals[k], i) * fourier_basis(x_vals[k], j) * dx
            A[i, j] = val
        b[i] = np.sum(fourier_basis(x_vals, i) * dx)
    
    # Initialize coefficients
    c = cp.Variable(n_basis)
    
    # Constraints for feasibility and integral
    integral_constraint = cp.sum(cp.multiply(c, b)) == 1
    
    # McCormick envelopes for h(x) = sum(c_k * phi_k(x))
    # Create variables for products
    d_vars = []
    vars_list = []
    
    # Build coefficient matrix to compute overlap
    # This is the convolution integral in Fourier space
    # We'll compute the convolution coefficients for each k
    conv_coef = np.zeros((n_points, n_basis, n_basis))
    
    # Precompute coefficients
    for k in range(n_points):
        for i in range(n_basis):
            for j in range(n_basis):
                val = 0
                for l in range(n_points):
                    val += fourier_basis(x_vals[l], i) * fourier_basis(x_vals[l] + k*dx, j) * dx
                conv_coef[k, i, j] = val
    
    # Initialize with any available construction
    h_values = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Simple initialization
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Project to Fourier coefficients
    c_init = np.linalg.lstsq(np.array([[fourier_basis(x_vals[k], i) 
                                       for i in range(n_basis)] 
                                     for k in range(n_points)]).T, h_init, rcond=None)[0]
    c_init = np.clip(c_init, 0, 1)
    
    # Try different initializations	
    best_score = float('inf')
    best_coeffs = c_init.copy()
    
    # Optimization using alternating direction method
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Project h_init to Fourier domain
        c.value = c_init
        
        # Create the optimization problem for the overlap integral
        # As we're working in the frequency domain, this becomes:
        max_overlap = cp.Variable()
        
        # Objective: minimize max overlap (C5)
        objective = cp.Minimize(max_overlap)
        
        # Constraints
        constraints = [integral_constraint]
        
        # Add metric constraint
        # Overlap requirements
        for k in range(n_points):  # overlap over full range
            c_k_sum = 0
            for i in range(n_basis):
                for j in range(n_basis):
                    # Use precomputed conv_coef[k, i, j] for integral of phi_i(x)(1 - phi_j(x+k))
                    c_k_sum += c[i] * c[j] * conv_coef[k, i, j]
            constraints.append(max_overlap >= c_k_sum)
            
        # Add bounds on coefficients
        for i in range(n_basis):
            constraints.append(c[i] <= 1)
            constraints.append(c[i] >= 0)
        
        # Solve the problem
        try:
            prob = cp.Problem(objective, constraints)
            
            # Use a solver that can handle this structure
            prob.solve(solver=SCS, max_iters=1000, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                c_vals = c.value
                reconstructed_h = np.zeros(n_points)
                for i in range(n_points):
                    for j in range(n_basis):
                        reconstructed_h[i] += c_vals[j] * fourier_basis(x_vals[i], j)
                
                # Ensure it's in [0,1]
                reconstructed_h = np.clip(reconstructed_h, 0, 1)
                
                # Ensure integral constraint
                reconstructed_h = reconstructed_h * (n_points / 2) / np.sum(reconstructed_h)
                
                # Evaluate C5
                current_c5 = evaluate_c5(reconstructed_h, dx)
                
                if current_c5 < best_score:
                    best_score = current_c5
                    best_coeffs = c_vals.copy()
                
                if current_c5 <= 0.3807:
                    break
                    
        except:
            pass
            
        # Simple boundary update step
        c_init = np.clip(c_init + np.random.normal(0, 0.01, n_basis), 0, 1)
        c_init = c_init * (n_points / 2) / np.sum(c_init)
        
    # Construct final h(x) from the best coefficients found
    h_final = np.zeros(n_points)
    for i in range(n_points):
        for j in range(n_basis):
            h_final[i] += best_coeffs[j] * fourier_basis(x_vals[i], j)
    
    # Ensure it meets constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    final_score = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_score, n_points)
