import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=0.02):
    """Compute the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 2000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.02
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle += np.random.normal(0, 0.01, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle += np.random.normal(0, 0.02, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle, dx))

    # Initialize global best
    global_best_idx = np.argmin(p_best_values)
    g_best = p_best[global_best_idx].copy()
    g_best_value = p_best_values[global_best_idx]

    best_h = g_best.copy()
    best_c5 = g_best_value

    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            p_best_particle = p_best[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            velocity = (w * velocity + 
                       c1 * r1 * (p_best_particle - particle) + 
                       c2 * r2 * (g_best - particle))
            
            # Clamp velocity
            velocity = np.clip(velocity, -v_max, v_max)
            
            # Update position
            particle += velocity
            
            # Enforce bounds and normalization
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Update particle
            particles[i] = particle
            
            # Compute fitness
            current_c5 = compute_c5(particle, dx)
            
            # Update personal best
            if current_c5 < p_best_values[i]:
                p_best[i] = particle.copy()
                p_best_values[i] = current_c5
                
                # Update global best if needed
                if current_c5 < g_best_value:
                    g_best = particle.copy()
                    g_best_value = current_c5
                    
                    if g_best_value <= 0.3807:
                        break

        # Update best solution found so far
        if g_best_value < best_c5:
            best_c5 = g_best_value
            best_h = g_best.copy()
            
        if g_best_value <= 0.3807:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
