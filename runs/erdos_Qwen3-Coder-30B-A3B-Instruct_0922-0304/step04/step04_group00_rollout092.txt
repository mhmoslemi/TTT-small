import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_correlations(h):
        # Compute all correlations for all shifts
        corr = correlate(h, 1 - h, mode="full")
        return corr
    
    def project_to_feasible(h):
        # Ensure h is in [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    c5_history = []
    
    # Cutting plane approach with CVXPY
    # First, get initial solution
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Track iterations for convergence
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Main optimization loop
    max_iter = 10000
    step_size = 0.001
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute maximum overlap shift
        max_shift = np.argmax(compute_correlations(h))
        max_corr = correlate(h, 1 - h, mode="full")[max_shift]
        
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Objective: minimize C5
        # For current iteration, we want to add constraint that
        # This shift's overlap <= max_corr - step_size
        
        # Generate the shifted correlation constraint
        # We'll linearize: h · (1 - h_k) <= max_corr - step_size
        shifted_h = np.roll(1 - h, max_shift - n_points + 1)
        linear_constraint = cp.sum(cp.multiply(h_var, shifted_h)) <= max_corr - step_size
        
        # Add constraint
        constraints.append(linear_constraint)
        
        # Optimizing
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var))) 
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Project back to feasible set
                new_h = project_to_feasible(new_h)
                h = new_h
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                c5_history.append(current_c5)
                
                # Reduce step size over time
                if len(c5_history) > 10:
                    if abs(c5_history[-1] - c5_history[-10]) / c5_history[-10] < 1e-4:
                        step_size *= 0.5
            else:
                # If optimization fails, perturb
                h = project_to_feasible(h + np.random.normal(0, 0.01, n_points))
                no_improvement_count += 1
        except:
            # Fallback to perturbation if CVXPY fails
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points))
            no_improvement_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size if no improvement
        if no_improvement_count > 50:
            step_size *= 0.5
            no_improvement_count = 0
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
