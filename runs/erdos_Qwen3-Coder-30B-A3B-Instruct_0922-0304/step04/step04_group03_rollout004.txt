import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute full convolution of h(x) and (1-h(x+k))
    overlap = np.correlate(h, 1 - h, mode="full")
    # Return maximum overlap scaled by dx
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Start from initial construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
    else:
        # Fallback: create a simple initial guess
        h_base = np.zeros(n_points)
        for i in range(n_points):
            h_base[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h_base = np.clip(h_base, 0, 1)
    
    # Normalize to satisfy integral constraint
    h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    # Optimization variables: construct h as a piecewise constant function
    # We will optimize the positions and shapes based on two intervals
    
    def construct_h(a, d):
        # a: start of first interval
        # d: distance between intervals 
        w = 0.5  # Each interval has width of 0.5
        h = np.zeros(n_points)
        
        # Positions of the intervals
        start1 = int(np.round(a / dx))
        end1 = int(np.round((a + w) / dx))
        start2 = int(np.round((a + w + d) / dx))
        end2 = int(np.round((a + w + d + w) / dx))
        
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h

    def objective(params):
        a, d = params
        # Bound validity of parameters
        if a < 0 or a > 1 - d - 0.5 or d < 0 or d > 1 - 1.0:
            return 1e10  # Penalize invalid
        h = construct_h(a, d)
        return evaluate_c5(h, dx)
    
    # Grid search + refinement
    best_c5 = float('inf')
    best_h = h_base.copy()
    best_params = (0.0, 0.0)
    
    # Coarse grid for a, d
    a_grid = np.linspace(0, 0.5, 20)
    d_grid = np.linspace(0, 0.5, 20)
    
    # Optimization to refine
    for a in a_grid:
        for d in d_grid:
            if a > 1 - d - 0.5:
                continue
            try:
                res = minimize(objective, [a, d], method='Nelder-Mead', bounds=[(0, 0.5), (0, 0.5)])
                if res.success:
                    val = res.fun
                    if val < best_c5:
                        best_c5 = val
                        a_sol, d_sol = res.x
                        best_h = construct_h(a_sol, d_sol)
                        best_params = (a_sol, d_sol)
            except:
                continue
                
            if time.time() - start_time > budget_s - 2:
                break
        if time.time() - start_time > budget_s - 2:
            break
    
    return (best_h.tolist(), best_c5, n_points)
