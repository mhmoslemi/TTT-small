import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-8
    alpha = 0.98
    iterations_per_T = 200
    mutation_sigma = 0.05
    
    # SA loop
    while T > T_min and time.time() - start_time < budget_s * 0.9:
        for _ in range(iterations_per_T):
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # Mutation: perturb some points
            h_new = h.copy()
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for i in indices:
                h_new[i] += np.random.normal(0, mutation_sigma)
            
            # Clip and rescale to maintain constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_c5(h_new, dx)
            
            # Accept or reject
            delta = c5_new - evaluate_c5(h, dx)
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
        
        T *= alpha
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
