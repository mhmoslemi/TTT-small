import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    d_candidates = np.arange(0, 1.01, 0.01)
    
    best_c5 = float('inf')

        i3 = int(np.floor((a + w + d + w) / dx))

        i1 = min(i1, n_points - 1)
        i3 = min(i3, n_points - 1)
        
        h = np.zeros(n_points)

        if i1 >= i0:
            h[i0:i1+1] = 1
        if i3 >= i2:
            h[i2:i3+1] = 1
            
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)

        c5 = evaluate_c5(h, dx)

                h[i0:i1+1] = 1
            if i3 >= i2:
