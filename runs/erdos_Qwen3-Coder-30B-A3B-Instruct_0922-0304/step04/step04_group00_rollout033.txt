import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points) * 0.5

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Coordinate-wise updates
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Save current value
            old_h_i = h[i]
            
            # Try different values for h[i]
            best_h_i = old_h_i
            best_local_c5 = float('inf')
            
            # Evaluate several candidates for h[i]
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
            if old_h_i not in candidates:
                candidates.append(old_h_i)
                
            # For continuous search around current value
            for val in np.linspace(max(0, old_h_i - 0.1), min(1, old_h_i + 0.1), 5):
                if val not in candidates:
                    candidates.append(val)
                    
            # Evaluate candidates
            for val in candidates:
                h[i] = val
                # Normalize to maintain sum = n_points/2
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                c5 = compute_c5(h, dx)
                if c5 < best_local_c5:
                    best_local_c5 = c5
                    best_h_i = val
                    
            # Update with best value
            h[i] = best_h_i
            
            # Normalize again
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
