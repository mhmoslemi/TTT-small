def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_haar_basis(n_points):
    """
    Generate Haar wavelet basis functions for the interval [0,2]
    """
    # We generate first 30 Haar wavelets
    basis = []
    dx = 2.0 / n_points
    
    # Compute the Haar functions
    for i in range(30):
        phi = np.zeros(n_points)
        if i == 0:
            # Constant function
            phi[:] = 1.0
        else:
            # Haar wavelet at level j, position k
            # We simply place it at appropriate scales
            scale = i // 2 + 1  # Dynamic scale
            pos = i % 2  # Position within scale
            if scale <= 5:
                # Simple rectangular wavelet
                start = int(pos * n_points / 2**(scale-1))
                end = int((pos+1) * n_points / 2**(scale-1))
                if end > n_points: end = n_points
                if start < n_points:
                    phi[start:end] = 1.0 if pos % 2 == 0 else -1.0
            else:
                # Approximation by simple piecewise constant
                segments = 2**(scale-5)
                if segments >= n_points:
                    phi[:] = 1.0
                else:
                    width = n_points // segments
                    for j in range(segments):
                        start = j * width
                        end = min((j+1) * width, n_points)
                        if pos % 2 == 0:
                            phi[start:end] = 1.0
                        else:
                            phi[start:end] = -1.0
        basis.append(phi)
    
    return np.array(basis)

def get_basis_inner_products(basis, n_points):
    """
    Precompute inner products for all basis pairs
    """
    n_basis = basis.shape[0]
    K = np.zeros((n_basis, n_basis))
    
    for i in range(n_basis):
        for j in range(n_basis):
            K[i,j] = np.dot(basis[i], basis[j])
    
    return K
