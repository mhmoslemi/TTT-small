import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Generate basis functions
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((basis_size, n_points))
    for i in range(basis_size):
        if i % 2 == 0:
            phi[i] = np.cos(np.pi * (i // 2 + 1) * x)
        else:
            phi[i] = np.sin(np.pi * ((i + 1) // 2) * x)
    
    # Compute integral of basis functions
    int_phi = np.zeros(basis_size)
    for i in range(basis_size):
        int_phi[i] = np.trapz(phi[i], x)
    
    # Initialize with the provided solution or a simple step function
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
    
    h0 = np.clip(h0, 0, 1)
    
    # Compute initial Fourier coefficients
    c0 = np.linalg.lstsq(phi.T, h0, rcond=None)[0]
    
    # Set up variables
    c = Variable(basis_size)
    
    # McCormick envelope for products
    z = Variable((basis_size, basis_size))
    
    # Constraints
    constraints = []
    
    # Non-negativity
    constraints.append(c >= 0)
    
    # Convexity for z variables
    for i in range(basis_size):
        for j in range(basis_size):
            constraints.append(z[i, j] >= c[i] + c[j] - 1)
            constraints.append(z[i, j] <= c[i])
            constraints.append(z[i, j] <= c[j])
    
    # Constraint: integral = 1
    constraints.append(2 * c[0] + sum(int_phi[i] * c[i] for i in range(1, basis_size)) == 1)
    
    # Build overlap matrix K
    K = np.zeros((2 * n_points - 1, basis_size * basis_size))
    for lag in range(2 * n_points - 1):
        for i in range(basis_size):
            for j in range(basis_size):
                # Compute integral of φ_i(x) * φ_j(x + lag)
                if lag < n_points:
                    shifted = np.roll(phi[j], lag)
                else:
                    shifted = np.roll(phi[j], lag - (2 * n_points - 2))
                K[lag, i * basis_size + j] = np.trapz(phi[i] * shifted, x)
    
    # Objective: minimize max of overlaps
    M = Variable()
    for lag in range(2 * n_points - 1):
        constraints.append(M >= abs(sum(K[lag, i * basis_size + j] * z[i, j] for i in range(basis_size) for j in range(basis_size))))
        constraints.append(M >= -abs(sum(K[lag, i * basis_size + j] * z[i, j] for i in range(basis_size) for j in range(basis_size))))

    objective = Minimize(M)
    prob = Problem(objective, constraints)
    
    try:
        # Solve with CVXPY
        prob.solve(solver=SCS, max_iters=10000, verbose=False)
        c_val = c.value
        h = phi.T @ c_val
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_c5(h, dx)
    except Exception as e:
        # Fallback to a heuristic approach if optimization fails
        h = h0.copy()
        c5 = evaluate_c5(h, dx)
    
    # Try optimization refinement if we have time
    if time.time() - start_time < budget_s - 2:
        # Combine with PSO approach for further improvement
        n_particles = 15
        max_iter = 1000
        
        # Initialize swarm
        swarm = [h.copy()]
        for i in range(1, n_particles):
            h_p = np.random.rand(n_points)
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (n_points / 2) / np.sum(h_p)
            swarm.append(h_p)
        
        velocities = [np.zeros(n_points) for _ in range(n_particles)]
        pbest = [swarm[i].copy() for i in range(n_particles)]
        pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
        
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        w_start = 0.9
        w_end = 0.4
        c_p = 1.5
        c_g = 1.5
        v_max = 0.1
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            w = w_start - (w_start - w_end) * (iter_count / max_iter)
            
            for i in range(n_particles):
                r_p = np.random.rand(n_points)
                r_g = np.random.rand(n_points)
                velocities[i] = (w * velocities[i] 
                               + c_p * r_p * (pbest[i] - swarm[i])
                               + c_g * r_g * (gbest - swarm[i]))
                
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                swarm[i] = swarm[i] + velocities[i]
                swarm[i] = np.clip(swarm[i], 0, 1)
                swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
                
                score = evaluate_c5(swarm[i], dx)
                
                if score < pbest_scores[i]:
                    pbest[i] = swarm[i].copy()
                    pbest_scores[i] = score
                    
                    if score < gbest_score:
                        gbest = swarm[i].copy()
                        gbest_score = score
                        
            if gbest_score <= 0.3807:
                break
        
        h = gbest
        c5 = gbest_score
    
    return (h.tolist(), c5, n_points)
