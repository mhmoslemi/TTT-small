import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Use initial construct if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Gradient-based optimization with restarts
    n_restarts = 5
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation of initial solution
        h = h_init.copy() + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Gradient descent with projected updates
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate gradient estimate using finite differences
            c5_current = evaluate_c5(h, dx)
            
            # Use finite differences to estimate gradient
            eps = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update step with momentum
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            c5_new = evaluate_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
                if best_c5 <= 0.3807:
                    break

    # Try one more optimization using convex optimization
    if time.time() - start_time < budget_s - 5:
        try:
            # Formulate as convex relaxation
            h_vars = Variable(n_points)
            
            # Constraint: integral = 1
            integral_constraint = cp.sum(h_vars) == n_points / 2
            
            # Constraint: bounds
            bound_constraints = [0 <= h_vars, h_vars <= 1]
            
            # Create overlap constraints
            # We want to minimize max_k \int h(x)(1 - h(x+k)) dx
            # which is max_k (h * (1-h))_k
            
            # Direct constraint on the overlap - compute a few sample overlaps
            # Simplified version: bound the key overlaps directly
            overlaps = []
            for k in range(n_points):
                # Compute overlap for some specific shifts
                if k == 0:  # Shift 0, just h*(1-h)
                    # This is tricky for exact calculation, but we have simpler bounds
                    continue
                # Just focus on small k shifts for practical constraints
            
            # Try to construct more useful optimization
            # Simplified approach: direct constraint-based optimization
            with cp.SolverStats() as stats:
                prob = cp.Problem(cp.Minimize(cp.sum(h_vars) * dx), 
                                  bound_constraints + [integral_constraint])
                
                try:
                    prob.solve(solver=CPLEX, eps_abs=1e-6, eps_rel=1e-6)  # Use best available solver
                    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                        h_opt = h_vars.value
                        if h_opt is not None:
                            h_opt = np.clip(h_opt, 0, 1)
                            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                            c5_opt = evaluate_c5(h_opt, dx)
                            if c5_opt < best_c5:
                                best_h = h_opt.copy()
                                best_c5 = c5_opt
                except:
                    pass

    # Refine final solution with local search
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb and check
        h_test = best_h + np.random.normal(0, 0.005, n_points)
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        c5_test = evaluate_c5(h_test, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            
            if best_c5 <= 0.3807:
                break

    return (best_h.tolist(), best_c5, n_points)
