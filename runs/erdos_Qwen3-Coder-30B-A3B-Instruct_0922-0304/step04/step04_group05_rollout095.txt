import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create triangular function
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization with adaptive step size
    max_iter = 10000
    lr = 0.001
    decay_rate = 0.99995
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # Since direct gradient computation is complex, we use finite differences
        epsilon = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[i] = (c5_plus - best_c5) / epsilon
            
        # Update h
        h = h - lr * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Adaptive learning rate
        lr *= decay_rate
        
        # Early stopping
        if best_c5 <= 0.3807:
            break

    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
