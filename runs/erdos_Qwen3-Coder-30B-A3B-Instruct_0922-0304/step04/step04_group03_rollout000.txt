import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute correlation of h with (1-h) using full mode
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise use a symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a piecewise linear construction
        h = np.zeros(n_points)
        # Define two intervals with a gap
        mid = n_points // 2
        width = n_points // 4  # Strongly constrained widths
        h[mid - width:mid] = 1.0
        h[mid:mid + width] = 1.0
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)

    # Convert to optimization variables (just need to map parameters to h)
    def to_h(params):
        # params: [a1, d]
        a1, d = params
        a1 = max(0.0, min(1.0 - d - 0.5, a1))  # Ensure bounds
        d = max(0.0, min(1.0 - 0.5, d))  # Ensure d is valid
        h_new = np.zeros(n_points)
        w = 0.5  # Width is fixed because total area = 1
        # Set intervals
        start1 = int(round(a1 * n_points / 2))
        end1 = int(round((a1 + w) * n_points / 2))
        start2 = int(round((a1 + w + d) * n_points / 2))
        end2 = int(round((a1 + w + d + w) * n_points / 2))
        
        # Clamp indices
        start1 = max(0, min(n_points-1, start1))
        end1 = max(0, min(n_points, end1))
        start2 = max(0, min(n_points-1, start2))
        end2 = max(0, min(n_points, end2))
        
        h_new[start1:end1] = 1.0
        h_new[start2:end2] = 1.0
        return h_new

    def objective(params):
        h_check = to_h(params)
        # Clamp to [0,1]
        h_check = np.clip(h_check, 0, 1)
        # Normalize
        h_check = h_check * (n_points / 2) / np.sum(h_check)
        return evaluate_c5(h_check, dx)

    # Try multiple starting points
    best_result = float('inf')
    best_h = h.copy()
    best_params = [0.2, 0.1]  # initial guess

    # Tune candidate parameters
    for a1_init in [0.1, 0.2, 0.3, 0.4]:
        for d_init in [0.05, 0.1, 0.15, 0.2]:
            try:
                res = minimize(objective, [a1_init, d_init], method='Nelder-Mead', 
                               options={'maxiter': 100, 'xatol': 1e-5, 'fatol': 1e-5})
                if res.success and res.fun < best_result:
                    best_result = res.fun
                    best_params = res.x
            except Exception:
                continue

    # Get the final h from best parameters
    final_h = to_h(best_params)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)

    # Constrained by time budget
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 1:
        # Perform a grid refinement if time permits
        for a1_ref in np.linspace(0.1, 0.9, 5):
            for d_ref in np.linspace(0.01, 0.2, 5):
                try:
                    res = minimize(objective, [a1_ref, d_ref], method='Nelder-Mead')
                    if res.success and res.fun < best_result:
                        best_result = res.fun
                        best_params = res.x
                except Exception:
                    continue

        # Final evaluation with refined parameters
        final_h = to_h(best_params)
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = evaluate_c5(final_h, dx)

    return (final_h.tolist(), final_c5, n_points)
