import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Precompute basis functions
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size + 1))
    phi[:, 0] = 1.0
    for i in range(1, basis_size + 1):
        if i % 2 == 1:
            phi[:, i] = np.sin(np.pi * (i + 1) // 2 * x)
        else:
            phi[:, i] = np.cos(np.pi * i // 2 * x)
    
    # Compute integral of basis functions
    int_phi = np.zeros(basis_size + 1)
    for i in range(basis_size + 1):
        int_phi[i] = np.trapz(phi[:, i], x)
    
    # Initialize coefficients
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        # Compute Fourier coefficients using least squares
        c = np.linalg.lstsq(phi, h_init, rcond=None)[0]
        # Adjust to satisfy integral constraint
        c[0] = (1.0 - np.sum(c[1:] * int_phi[1:])) / int_phi[0]
    else:
        # Initialize with a simple smooth shape
        c = np.random.rand(basis_size + 1) * 0.1
        c[0] = 0.5  # Set DC component
    
    # Create optimization variables
    c_vars = cp.Variable(basis_size + 1)
    
    # Set up McCormick relaxation for z = c * c
    z_vars = cp.Variable((basis_size + 1, basis_size + 1), symmetric=True)
    
    # Precompute kernel matrix (for convolution in frequency domain)
    K = np.zeros((basis_size + 1, basis_size + 1, 2 * n_points - 1))
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            # For each Lag
            for lag in range(2 * n_points - 1):
                # Compute integral of phi_i(x) * phi_j(x + lag)
                lag_idx = lag - (n_points - 1)
                int_val = 0.0
                for k in range(n_points):
                    x_k = x[k]
                    x_k_lag = x_k + lag_idx * dx
                    if 0 <= x_k_lag <= 2:
                        idx = np.argmin(np.abs(x - x_k_lag))
                        int_val += phi[k, i] * phi[idx, j] * dx
                K[i, j, lag] = int_val
    
    # Create problem formulation
    prob = None
    
    def solve_with_c(c_start):
        c_vars.value = c_start
        objective = cp.Minimize(cp.max(cp.abs(cp.sum(cp.multiply(z_vars, K), axis=(0, 1)))))
        
        # McCormick constraints
        constraints = []
        for i in range(basis_size + 1):
            for j in range(i, basis_size + 1):
                z_vars[i, j].value = c_start[i] * c_start[j]
                z_vars[j, i].value = c_start[i] * c_start[j]
                # z >= c[i] + c[j] - 1
                constraints.append(z_vars[i, j] >= c_vars[i] + c_vars[j] - 1)
                # z <= c[i], z <= c[j]
                constraints.append(z_vars[i, j] <= c_vars[i])
                constraints.append(z_vars[i, j] <= c_vars[j])
        
        # Non-negativity of coefficients
        constraints.append(c_vars >= 0)
        
        # Integral constraint
        # sum c[i] * int_phi[i] = 1
        constraints.append(cp.sum(cp.multiply(c_vars, int_phi)) == 1)
        
        # Create problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return c_vars.value, prob.value
            else:
                return c_start, float('inf')
        except:
            return c_start, float('inf')
            
    # Main optimization loop
    best_c = c.copy()
    best_score = float('inf')
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different starting points to escape local optima
        c_new = best_c + np.random.randn(basis_size + 1) * 0.01
        c_new = np.clip(c_new, 0, 1)
        c_new[0] = (1.0 - np.sum(c_new[1:] * int_phi[1:])) / int_phi[0]
        c_new = np.clip(c_new, 0, 1)
        
        current_c, score = solve_with_c(c_new)
        
        if score < best_score:
            best_score = score
            best_c = current_c.copy()
            
        # Slightly perturb for diversity
        if np.random.rand() < 0.2:
            current_c = current_c + np.random.randn(basis_size + 1) * 0.005
            current_c = np.clip(current_c, 0, 1)
            current_c[0] = (1.0 - np.sum(current_c[1:] * int_phi[1:])) / int_phi[0]
            current_c = np.clip(current_c, 0, 1)
            new_c, new_score = solve_with_c(current_c)
            if new_score < best_score:
                best_score = new_score
                best_c = new_c.copy()
                
        # Check early stopping
        if best_score <= 0.3807:
            break
    
    # Extract final solution
    final_h = phi @ best_c
    # Make sure it's in [0,1]
    final_h = np.clip(final_h, 0, 1)
    # Normalize to sum to 1
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Final evaluation
    overlap = correlate(final_h, 1 - final_h, mode="full")
    c5_final = np.max(overlap) * dx
    
    return (final_h.tolist(), c5_final, n_points)
