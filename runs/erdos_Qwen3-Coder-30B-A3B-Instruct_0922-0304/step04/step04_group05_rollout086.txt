import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_onto_feasible(h, target_sum):
    # Clip to [0,1] and scale to meet sum constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        # Create triangular function
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe with line search
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate gradient for quadratic approximation
        # Overlap term: h[i] * (1 - h[i+k]) 
        # Gradient: -dx * (1 - 2*h - shift_average)
        h_plus = np.concatenate([h, np.zeros(n_points-1)])
        h_minus = np.concatenate([np.zeros(n_points-1), h])
        shift_avg = correlate(h, np.ones(n_points-1), mode='valid')
        shift_avg = shift_avg / (n_points - 1)
        shift_avg = np.pad(shift_avg, (0, n_points-1), 'constant')
        
        # Gradient of overlap function
        grad = -dx * (1 - 2*h - shift_avg)
        
        # Direction is negative gradient
        step_size = 0.1
        delta = -step_size * grad
        
        # Project onto constraint set
        h_new = h + delta
        h_new = project_onto_feasible(h_new, target_sum)
        
        # Line search on step size
        alpha = 1.0
        max_alpha = 1.0
        while alpha > 1e-6:
            h_test = h + alpha * delta
            h_test = project_onto_feasible(h_test, target_sum)
            test_c5 = evaluate_c5(h_test, dx)
            if test_c5 < best_c5:
                h = h_test
                best_c5 = test_c5
                best_h = h.copy()
                break
            else:
                alpha /= 2.0
        
        if best_c5 <= 0.3807:
            break
            
        # Restart if no significant progress
        if iter_num > 0 and iter_num % 50 == 0:
            h = best_h.copy()
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
