import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_generations = 2000
    elite_size = population_size // 4
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill the rest with varied random perturbations
    for i in range(1, population_size):
        h = h0.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best
    best_h = None
    best_c5 = float('inf')
    
    # Evolutionary Strategy loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        
        # Create new generation
        new_population = []
        
        # Keep elite
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        remaining = population_size - elite_size
        for _ in range(remaining):
            # Select parent from elite
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_scale, n_points)
            child = parent + noise
            child = project_to_feasible(child, n_points)
            
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation scale based on success
        if generation % 5 == 0 and generation > 0:
            # If recent generations improved significantly, reduce mutation
            recent_fitness = [fitness[i] for i in sorted_indices[:elite_size]]
            avg_fitness = np.mean(recent_fitness)
            if avg_fitness < best_c5 * 0.99:
                mutation_scale *= 0.95
            else:
                mutation_scale *= 1.05
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        final_c5 = float('inf')
        # Re-evaluate original best if needed
        for h in population:
            c5 = evaluate_c5(h, dx)
            if c5 < final_c5:
                final_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), final_c5, n_points)
