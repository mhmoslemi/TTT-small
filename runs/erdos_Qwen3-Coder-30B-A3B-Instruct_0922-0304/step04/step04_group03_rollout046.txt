import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise create symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create simple symmetric two-interval construction
        h = np.zeros(n_points)
        # Set first interval from 0 to 0.5
        i0 = 0
        i1 = int(0.5 / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        h[i0:i1+1] = 1.0
        
        # Set second interval starting from 1.0 to 1.5
        i2 = int(1.0 / dx)
        i3 = int(1.5 / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
        if i2 < n_points:
            h[i2:i3+1] = 1.0
            
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search on interval separation distance
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create piecewise constant h with two non-overlapping intervals
        h_test = np.zeros(n_points)
        
        # First interval from 0 to 0.5
        i0 = 0
        i1 = int(0.5 / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        h_test[i0:i1+1] = 1.0
        
        # Second interval from (0.5 + d) to (1.0 + d)
        i2 = int((0.5 + d) / dx)
        i3 = int((1.0 + d) / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
        
        # Ensure non-overlapping and valid indices
        if i2 < n_points and i3 >= i2:
            h_test[i2:i3+1] = 1.0
            
        # Normalize to satisfy integral constraint
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Compute overlap
        c5 = evaluate_c5(h_test, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
    
    # Refine with local search around best point
    best_d = 0.0
    for d in np.arange(0.0, 1.0 + 1e-6, 0.001):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_test = np.zeros(n_points)
        i0 = 0
        i1 = int(0.5 / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        h_test[i0:i1+1] = 1.0
        
        i2 = int((0.5 + d) / dx)
        i3 = int((1.0 + d) / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
        
        if i2 < n_points and i3 >= i2:
            h_test[i2:i3+1] = 1.0
            
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5 = evaluate_c5(h_test, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_d = d
            
    # Final refinement with best d
    h_final = np.zeros(n_points)
    i0 = 0
    i1 = int(0.5 / dx) - 1
    if i1 >= n_points:
        i1 = n_points - 1
    h_final[i0:i1+1] = 1.0
    
    i2 = int((0.5 + best_d) / dx)
    i3 = int((1.0 + best_d) / dx) - 1
    if i3 >= n_points:
        i3 = n_points - 1
    
    if i2 < n_points and i3 >= i2:
        h_final[i2:i3+1] = 1.0
        
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
