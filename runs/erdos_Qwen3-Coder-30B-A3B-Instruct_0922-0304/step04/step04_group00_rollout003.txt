import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        # Initialize particle
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            # Add small noise
            particle += np.random.normal(0, 0.01, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle += np.random.normal(0, 0.05, n_points)
            
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle))
    
    # Global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = compute_c5(particles[i])
            
            # Update personal best
            if current_c5 < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_c5
                
                # Update global best
                if current_c5 < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_c5
                    if g_best_value <= 0.3807:
                        break
        
        if g_best_value <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(g_best)
    
    return (g_best.tolist(), final_c5, n_points)
