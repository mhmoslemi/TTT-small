import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Coordinate-wise optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try updating each coordinate
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Save original value
            original_h_i = h[i]
            
            # Try small changes in both directions
            for delta in [-0.01, 0.01]:
                h[i] = original_h_i + delta
                h[i] = np.clip(h[i], 0, 1)
                
                # Normalize
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                # Evaluate
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    # Early exit if target achieved
                    if best_c5 <= 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
                
                # Revert
                h[i] = original_h_i
                
        # Normalize to sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
