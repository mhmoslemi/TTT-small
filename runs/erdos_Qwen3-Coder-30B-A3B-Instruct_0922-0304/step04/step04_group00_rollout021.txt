import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA parameters
    population_size = 50
    generations = 1000
    tournament_size = 5
    mutation_rate = 0.1
    elite_count = 5
    local_search_count = 10
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = h + np.random.normal(0, 0.02, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    def evaluate_population(population):
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append((c5, ind))
        return sorted(fitness, key=lambda x: x[0])
    
    # GA main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        ranked = evaluate_population(population)
        best_in_gen = ranked[0]
        if best_in_gen[0] < best_c5:
            best_c5 = best_in_gen[0]
            best_h = best_in_gen[1].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite = [ind for _, ind in ranked[:elite_count]]
        
        # Create new population
        new_population = elite.copy()
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            tournament = random.sample(ranked, tournament_size)
            parent1 = min(tournament, key=lambda x: x[0])[1]
            tournament = random.sample(ranked, tournament_size)
            parent2 = min(tournament, key=lambda x: x[0])[1]
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.01, n_points)
                child1 += noise
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.01, n_points)
                child2 += noise
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Local search on top individuals
        local_search_inds = [ind for _, ind in ranked[:local_search_count]]
        for ind in local_search_inds:
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(
                compute_c5, 
                ind, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 20, 'ftol': 1e-8}
            )
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = refined * (n_points / 2) / np.sum(refined)
                # Check if this refined solution is better
                refined_c5 = compute_c5(refined)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined.copy()
        
        population = new_population
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
