import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.zeros(n_points)
        h0[n_points//4:3*n_points//4] = 1.0
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize particles around h0
    for i in range(n_particles):
        perturbation = np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(h0 + perturbation, 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    omega = 0.9
    phi_p = 2.0
    phi_g = 2.0
    max_iter = 2000
    
    # Track best solutions
    pbest = particles.copy()
    pbest_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           phi_p * r1 * (pbest[i] - particles[i]) + 
                           phi_g * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
        
        # Periodic reinitialization
        if iteration % 100 == 0 and iteration > 0:
            reinit_count = max(1, n_particles // 10)
            reinit_indices = np.random.choice(n_particles, reinit_count, replace=False)
            for i in reinit_indices:
                particles[i] = np.clip(h0 + np.random.normal(0, 0.01, n_points), 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
        
        # Early exit if target achieved
        if gbest_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
