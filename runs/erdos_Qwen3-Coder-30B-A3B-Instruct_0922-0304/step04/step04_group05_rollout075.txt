import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a triangular-like initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential convex programming approach
    max_iter = 1000
    alpha = 0.1
    mu = 1e-3
    damping_factor = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current overlap values for each shift
        overlaps = []
        for k in range(n_points):
            # Compute the overlap for shift k
            overlap_k = 0.0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    overlap_k += h[i] * (1 - h[j])
            overlaps.append(overlap_k * dx)
        
        # Find the maximum overlap
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_value = overlaps[max_overlap_idx]
        
        # If we've reached the target, stop early
        if max_overlap_value <= 0.3807:
            break
            
        # Compute gradient and Hessian of the objective (quadratic approx)
        # For a single overlap term at shift k, the gradient is
        # grad = [h[i] * (1 - h[i+k]) + (1-h[i+k]) * h[i] for i]
        # Simplified: grad = h[i] - h[i]*h[i+k] + h[i] - h[i]*h[i+k] = 2*h[i] - 2*h[i]*h[i+k]
        # But this gives the Hessian in terms of all constraints.
        # Let's be more direct and compute an approximate gradient:
        
        # Gradient of max_overlaps with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_overlap_idx
            if j < n_points:
                grad[i] += 1 - 2 * h[j]
        
        # Approximate Hessian matrix (only diagonal for simplicity)
        hessian_diag = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_overlap_idx
            if j < n_points:
                hessian_diag[i] += -2 * h[j]  # Second derivative approximation
        
        # Add regularization to Hessian
        hessian_diag += 1e-8
        
        # Set up convex subproblem
        delta = cp.Variable(n_points)
        # Objective: grad^T * delta + (1/2) * delta^T * H * delta
        objective = cp.Minimize(grad @ delta + 0.5 * cp.quad_form(delta, np.diag(hessian_diag)))
        # Constraints: 0 <= h + delta <= 1, sum(h + delta) = n_points/2
        constraints = [
            delta >= -h, 
            delta <= 1 - h,
            cp.sum(delta) == 0  # since sum(h) is fixed, delta must sum to 0
        ]
        
        # Create and solve convex problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-9, reltol=1e-9)
            if prob.status != cp.OPTIMAL:
                # If CVXOPT failed, try without constraints as fallback
                delta = alpha * grad
            else:
                delta = delta.value
                # Step size adjustment
                if np.all(delta >= -h) and np.all(delta <= 1-h):
                    if alpha > 0.001:
                        alpha *= damping_factor
        except:
            # Fallback to gradient descent if optimization fails
            delta = alpha * grad
        
        # Update h
        h_new = h + delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new.copy()
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
