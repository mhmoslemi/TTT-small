import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with adaptive step size and multiple strategies
    max_iter = 5000
    step_size = 0.003
    stagnation_count = 0
    stagnation_limit = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and gradient
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        max_idx = np.argmax(corr)
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update h with momentum
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step sizing and stagnation detection
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
            step_size = min(0.01, step_size * 1.05)  # Increase slightly if improving
        else:
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                step_size *= 0.8  # Reduce step if stuck
                stagnation_count = 0
                # Perturb solution occasionally
                if iteration % 100 == 0:
                    h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally apply random perturbation to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
