import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Variational approach with basis functions
    n_basis = 20
    # Create basis functions: quadratic "hat" functions
    x_basis = np.linspace(0, 2, n_points)
    basis = np.zeros((n_basis, n_points))
    x_centers = np.linspace(0, 2, n_basis)
    dx_basis = 2.0 / n_basis
    
    for i in range(n_basis):
        x_center = x_centers[i]
        # Quadratic basis function
        basis[i] = np.maximum(0, 1 - ((x_basis - x_center) / (dx_basis/2))**2)
        # Normalize
        basis[i] = basis[i] / np.sum(basis[i]) * dx_basis
    
    # Initialize weights
    w = np.ones(n_basis) * 0.5
    # Set up triangular-like initialization for better overlap balancing
    w = np.abs(np.linspace(-1, 1, n_basis)) * 0.5 + 0.25
    w = w / np.sum(w) * n_points / 2
    
    # Optimizer parameters
    learning_rate = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Adam variables
    m = np.zeros(n_basis)
    v = np.zeros(n_basis)
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Reconstruct h from basis weights
        h = basis.T @ w
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        # Apply constraint: integral = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Stop if we achieve target
        if best_c5 <= 0.3807:
            break
            
        # Compute gradients using finite differences
        eps = 1e-5
        grad = np.zeros(n_basis)
        for i in range(n_basis):
            w_plus = w.copy()
            w_plus[i] += eps
            h_plus = basis.T @ w_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(overlap_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        w = w - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply constraints
        w = np.clip(w, 0, 1)
        
        # Renormalize weights
        w = w / np.sum(w) * n_points / 2
        
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_basis)
            w += noise
            w = np.clip(w, 0, 1)
            w = w / np.sum(w) * n_points / 2
    
    return (best_h.tolist(), best_c5, n_points)
