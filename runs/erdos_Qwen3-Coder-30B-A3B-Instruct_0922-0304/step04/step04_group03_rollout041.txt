import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # Width of each interval
        a = 0    # Starting point
        d = 1.0  # Distance between intervals (we'll optimize this)
        
        # Use first interval [a, a+w] and second interval [a+w+d, a+w+d+w] 
        # For 0.5 width, this spans [0, 0.5] and [1.5, 2.0] when d=1
        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        if i1 < n_points:
            h[i0:i1+1] = 1.0
        if i3 < n_points:
            h[i2:i3+1] = 1.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Grid search over d
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Test a range of d values with fine resolution
    d_candidates = np.arange(0.0, 1.01, 0.01)
    
    for d in d_candidates:
        h_test = np.zeros(n_points)
        w = 0.5  # Width of each interval
        
        # Compute indices for intervals
        a = 0
        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        # Ensure bounds
        i1 = min(i1, n_points - 1)
        i3 = min(i3, n_points - 1)
        
        # Set intervals
        h_test[i0:i1+1] = 1.0
        if i2 <= i3:
            h_test[i2:i3+1] = 1.0
        
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate the overlap
        c5 = evaluate_c5(h_test, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
            
        if time.time() - start_time > budget_s - 1:
            break
            
        if best_c5 <= 0.3807:
            break
    
    # If the grid search didn't find a better solution, do a small number
    # of perturbations around the best known solution
    if best_c5 > 0.3807:
        h_pert = best_h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Make a small random perturbation
            delta = np.random.normal(0, 0.01, n_points)
            h_pert = h_pert + delta
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            c5 = evaluate_c5(h_pert, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pert.copy()
                
            if best_c5 <= 0.3807:
                break

    return (best_h.tolist(), best_c5, n_points)
