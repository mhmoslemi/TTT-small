import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high, solver='CVXOPT'):
    """
    Solve quadratic program: minimize (1/2)*delta^T*H*delta + c^T*delta
    subject to A*delta = b and delta_low <= delta <= delta_high
    """
    n = H.shape[0]
    delta = cp.Variable(n)
    
    # QP objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [A @ delta == b]
    constraints += [delta >= delta_low]
    constraints += [delta <= delta_high]
    
    # Problem
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create triangular-like shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Constraint matrix for sum = n_points/2
    A_eq = np.ones((1, n_points))
    b_eq = np.array([n_points / 2])
    
    # QP solver parameters
    max_iter = 500
    reg = 1e-5
    alpha = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        # ∇C_k = dx * (1 - 2*h - shift_average_k)
        # For all shifts k, shift_average_k is the average of h over shifted positions.
        # So for each k, ∇C_k = dx * (1 - 2*h - shift_average_k)
        
        # Gradient vector:
        # G[i] = sum over k of (1 - 2*h[i] - shift_average_k[i])
        # But this is more efficiently computed via convolution:
        # G = dx * (1 - 2*h - shift_average) where shift_average is not h itself,
        # rather it's a reverse-shifted version of h applied across all time lags

        # Gradient with respect to C_k (for the max of all C_k’s)
        # This requires computing derivatives for each shift
        # One approach is to compute the shift average in one go:
        
        # compute shift average for all shifts
        # h_shifted = [h, h_shifted_1, ... h_shifted_npoints] 
        # then sum_i h[i] * (1 - h[i+k]) = sum_k h[i] * (1 - h[i+k]) _over all i_
        # For a fixed k, we sum over i from 0 to npoints-k-1:
        # sum_{i=0}^{npoints-k-1} h[i] * (1 - h[i+k])
        # The gradient with respect to h[i] = dx * (1 - 2*h[i] - shift_average[i])
        # We compute shift average below in a vectorized fashion.
        
        # Compute analytical gradients
        # Shift averages
        h1 = np.concatenate([h, np.zeros(n_points)])
        shift_avg = np.zeros(n_points)
        for k in range(n_points):
            shift_avg[k] = np.sum(h1[k:k+n_points] * h[:n_points-k]) / (n_points - k)
        shift_avg[0] = np.sum(h * h) / n_points  # case k=0, self correlation
        
        grad = dx * (1 - 2 * h - shift_avg)
        
        # Hessian matrix (detailed approximation)
        # For second-order, H[i,j] = d^2/dh_i dh_j (C_k) = 2*dx*delta[i,j] - dx*shift_matrix[i,j]
        H = np.zeros((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):
                if i == j:
                    H[i,j] = 2 * dx  # First term
                else:
                    H[i,j] = -dx  # Cross terms from shifts
        
        # Add regularization
        H += reg * np.eye(n_points)
        
        # Prepare constraints
        delta_low = np.zeros(n_points)  # Have to ensure h[i] + δ[i] >= 0
        delta_high = np.ones(n_points)  # And h[i] + δ[i] <= 1
        
        # Applied continuously & tolerantly
        for i in range(n_points):
            delta_low[i] = -h[i]
            delta_high[i] = 1 - h[i]
        
        # Solve QP subproblem
        delta = _solve_qp(H, grad, A_eq, b_eq, delta_low, delta_high)
        
        if delta is None:
            # fallback; try increasing regularization
            reg *= 10
            H = np.zeros((n_points, n_points))
            for i in range(n_points):
                for j in range(n_points):
                    if i == j:
                        H[i,j] = 2 * dx
                    else:
                        H[i,j] = -dx
            H += reg * np.eye(n_points)
            delta = _solve_qp(H, grad, A_eq, b_eq, delta_low, delta_high)
            if delta is None:
                break  # Give up
        
        # Line search
        step_size = 1.0
        improved = False
        
        while step_size > 1e-6:
            h_new = h + step_size * delta
            h_new = np.clip(h_new, 0, 1)
            try:
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            except:
                break
            c5_new = evaluate_c5(h_new, dx)
            if c5_new <= best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
            else:
                step_size *= alpha
        
        if not improved:
            h = h + 0.1 * delta  # small update to continue some progress
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
