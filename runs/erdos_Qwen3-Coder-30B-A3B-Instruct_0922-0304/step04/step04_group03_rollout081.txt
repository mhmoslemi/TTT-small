import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, dx, target_sum):
    """Project h to satisfy the sum constraint"""
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    return h * scale

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Initialize with initial_h_values if available, otherwise create a smooth construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth bell-shaped initial solution
        x = np.linspace(0, 2, n_points)
        h = np.exp(-((x - 1) / 0.2) ** 2)
        h = h / np.sum(h) * target_sum / dx
    
    h = np.clip(h, 0, 1)
    
    # Gradient descent parameters
    max_iter = 10000
    learning_rate = 0.01
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Compute gradient for the worst case shift
        # For overlap_k = sum h[i] * (1 - h[i+k]), gradient = 1 - h[i+k] - h[i-k]
        grad = np.zeros(n_points)
        for i in range(n_points):
            # The gradient dO_k/dh_i = (1 - h[(i+k)%n_points]) - h[(i-k)%n_points]
            k = max_k - n_points + 1  # Unpack the shift from full correlation index
            if k >= 0:
                grad[i] = (1 - h[(i + k) % n_points]) - h[(i - k) % n_points]
            else:
                grad[i] = (1 - h[(i - k) % n_points]) - h[(i + k) % n_points]
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to satisfy sum constraint
        h_new = project_h(h_new, dx, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Update if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # Check for convergence
        if abs(new_c5 - max_overlap) < tolerance:
            break
            
        h = h_new.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
