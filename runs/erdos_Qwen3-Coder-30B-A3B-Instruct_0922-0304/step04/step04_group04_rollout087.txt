import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Create initial guess from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier basis functions
    def fourier_basis(x, k):
        if k % 2 == 1:
            return np.sin((k+1) * np.pi * x / 2)
        else:
            return np.cos((k//2) * np.pi * x / 2)
    
    # Precompute basis matrix
    x_vals = np.linspace(0, 2, n_points)
    basis_matrix = np.array([[fourier_basis(x, k) for k in range(30)] for x in x_vals])
    
    # Compute inner products for all lags
    K = np.zeros((30, 30, 2*n_points-1))
    for k_idx in range(2*n_points-1):
        lag = (k_idx - (n_points-1)) * dx
        for i in range(30):
            for j in range(30):
                K[i,j,k_idx] = np.sum(basis_matrix[:,i] * basis_matrix[:,j] * np.exp(-1j * (i+1) * np.pi * lag / 2).real) * dx
    
    # Solve optimization using CVXPY
    # Variables
    a = cp.Variable(30)
    z = cp.Variable((30, 30))
    
    # Constraints
    constraints = []
    
    # Integral constraint
    integral_constraint = cp.sum(a @ basis_matrix.T @ np.ones(n_points) * dx) == 1
    constraints.append(integral_constraint)
    
    # Bounds on basis coefficients
    constraints.append(a >= -1)
    constraints.append(a <= 1)
    
    # McCormick envelopes for z_ij = a_i * a_j
    for i in range(30):
        for j in range(30):
            constraints.append(z[i,j] >= a[i] + a[j] - 1)
            constraints.append(z[i,j] <= a[i])
            constraints.append(z[i,j] <= a[j])
    
    # Objective
    max_overlap = cp.Variable()
    for k_idx in range(2*n_points-1):
        constraint = cp.sum(z * K[:,:,k_idx]) <= max_overlap
        constraints.append(constraint)
    
    # Minimize maximum overlap
    objective = cp.Minimize(max_overlap)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            a_opt = a.value
            # Reconstruct h(x)
            h_reconstructed = basis_matrix @ a_opt
            h_reconstructed = np.clip(h_reconstructed, 0, 1)
            h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
            
            # Evaluate final result
            c5_val = np.max(correlate(h_reconstructed, 1 - h_reconstructed, mode="full")) * dx
            return (h_reconstructed.tolist(), c5_val, n_points)
        else:
            # Fallback to initial guess if optimization fails
            c5_val = np.max(correlate(h_init, 1 - h_init, mode="full")) * dx
            return (h_init.tolist(), c5_val, n_points)
    except Exception:
        # Fallback to initial guess
        c5_val = np.max(correlate(h_init, 1 - h_init, mode="full")) * dx
        return (h_init.tolist(), c5_val, n_points)
