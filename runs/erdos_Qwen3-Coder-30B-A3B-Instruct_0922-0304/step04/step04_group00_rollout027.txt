import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    # Compute the overlap integral for the given h
    overlap = correlate(h, 1 - h, mode="full")
    dx = 2.0 / len(h)
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            # Start near the initial_h_values
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            # Random start
            particle = np.random.rand(n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
        pbest.append(particle.copy())
        pbest_scores.append(compute_c5(particle))
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO main loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            pbest_i = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocity = (inertia * velocity + 
                       cognitive * r1 * (pbest_i - particle) + 
                       social * r2 * (gbest - particle))
            
            # Constrain velocity
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            
            # Update particle
            particle += velocity
            
            # Enforce bounds
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Store updated particle
            particles[i] = particle
            velocities[i] = velocity
            
            # Evaluate new score
            score = compute_c5(particle)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particle.copy()
                
                # Update global best
                if score < gbest_score:
                    gbest = particle.copy()
                    gbest_score = score

        # Early exit if we've found a good enough solution
        if gbest_score <= 0.3807:
            break

    return (gbest.tolist(), gbest_score, n_points)
