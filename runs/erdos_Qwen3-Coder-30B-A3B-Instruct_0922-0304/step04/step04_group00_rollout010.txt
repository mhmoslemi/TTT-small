import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    T = T0
    max_iter = 10000
    
    # Track consecutive failures
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current solution evaluation
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate neighbor solution
        h_new = h.copy()
        idx = random.randint(0, n_points - 1)
        
        # Perturb one element
        step_size = 0.05 * T / T0
        h_new[idx] += np.random.normal(0, step_size)
        
        # Clip to [0, 1]
        h_new[idx] = max(0, min(1, h_new[idx]))
        
        # Renormalize to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept or reject based on simulated annealing
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new.copy()
        
        # Update temperature
        T *= alpha
        
        # If no improvement for a while, restart with random perturbation
        if no_improvement_count > max_no_improvement:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
