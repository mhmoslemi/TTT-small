import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ES parameters
    pop_size = 100
    num_generations = 1000
    elite_size = pop_size // 5  # 20%
    mutation_rate = 0.3  # 30% of elements mutated
    initial_sigma = 0.1
    sigma = initial_sigma
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # ES iterations
    for generation in range(num_generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            fitness.append(evaluate_c5(individual, dx))
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Elitism
        next_gen = population[:elite_size]
        
        # Generate offspring
        while len(next_gen) < pop_size:
            # Select parent
            parent_idx = np.random.randint(0, elite_size)
            parent = population[parent_idx]
            
            # Create offspring with mutation
            offspring = parent.copy()
            
            # Mutate a random subset of elements
            indices_to_mutate = np.random.choice(n_points, 
                                               size=int(mutation_rate * n_points), 
                                               replace=False)
            for idx in indices_to_mutate:
                offspring[idx] += np.random.normal(0, sigma)
            
            # Repair constraints
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * (n_points / 2) / np.sum(offspring)
            
            next_gen.append(offspring)
        
        population = next_gen
        
        # Adaptive sigma adjustment
        if generation % 20 == 0:
            improvement = np.mean(fitness[:elite_size]) - np.mean(fitness[-elite_size:])
            if improvement > 0:
                sigma = max(0.001, sigma * 0.95)  # Decrease sigma
            else:
                sigma = min(0.5, sigma * 1.1)  # Increase sigma
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
