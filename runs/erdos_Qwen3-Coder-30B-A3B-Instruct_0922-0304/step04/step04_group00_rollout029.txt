import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=0.02):
    """Compute the maximum overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = np.random.rand(n_particles, n_points) * 0.01 - 0.005
    
    # Personal best
    p_best = particles.copy()
    p_best_c5 = np.array([compute_c5(p) for p in particles])
    
    # Global best
    best_idx = np.argmin(p_best_c5)
    g_best = p_best[best_idx].copy()
    g_best_c5 = p_best_c5[best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (p_best[i] - particles[i]) + 
                            social * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_c5 = compute_c5(particles[i])
            
            # Update personal best
            if current_c5 < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = current_c5
            
            # Update global best
            if current_c5 < g_best_c5:
                g_best = particles[i].copy()
                g_best_c5 = current_c5
                
        # Early termination
        if g_best_c5 <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_c5, n_points)
