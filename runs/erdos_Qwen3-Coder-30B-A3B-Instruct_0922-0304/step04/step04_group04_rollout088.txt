import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback to step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Optimization loop with improved search strategy
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Small perturbation of current best solution
        h_pert = best_h.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.02, n_points)
        h_pert += noise
        
        # Ensure bounds
        h_pert = np.clip(h_pert, 0, 1)
        
        # Renormalize
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        # Evaluate
        c5_pert = evaluate_c5(h_pert, dx)
        
        # Accept if better
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
            
        # Occasionally perturb to escape local minima
        if iteration % 50 == 0:
            # Generate a completely new solution
            h_new = np.random.rand(n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    # Final validation
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
