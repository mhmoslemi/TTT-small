import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Fourier basis functions
    def fourier_basis(x, k):
        if k % 2 == 1:
            return np.sin((k+1)//2 * np.pi * x)
        else:
            return np.cos(k//2 * np.pi * x)
    
    # Optimization variables
    a = cp.Variable(30)  # Reduced basis size for efficiency
    z = cp.Variable((30, 30), symmetric=True)
    
    # Precompute basis inner products
    K = np.zeros((30, 30, 2*n_points))
    x_points = np.linspace(0, 2, n_points)
    
    for i in range(30):
        for j in range(30):
            for k_idx in range(2*n_points):
                # Compute integral of phi_i(x)*phi_j(x+k) over [0,2]
                k = 2.0 * k_idx / n_points - 1.0  # Normalized lag
                integrand = np.zeros(n_points)
                for x_idx in range(n_points):
                    phi_i = fourier_basis(x_points[x_idx], i)
                    phi_j = fourier_basis(x_points[x_idx] + k, j)
                    integrand[x_idx] = phi_i * phi_j
                K[i, j, k_idx] = np.trapz(integrand, x_points)
    
    # Constraints for integral constraint
    integral_constraint = cp.sum(a) * dx == 1
    
    # McCormick envelopes
    constraints = [cp.constraints.NonPos(z[i, j] - a[i] - a[j] + 1) for i in range(30) for j in range(30)]
    constraints += [cp.constraints.NonPos(a[i]) for i in range(30)]
    constraints += [cp.constraints.NonPos(-a[i] + 1) for i in range(30)]
    constraints += [cp.constraints.NonPos(z[i, j] - a[i]) for i in range(30) for j in range(30)]
    constraints += [cp.constraints.NonPos(z[i, j] - a[j]) for i in range(30) for j in range(30)]
    
    # Objective: minimize max overlap
    max_overlap = cp.Variable(1)
    
    # Catch max overlap
    overlap_constructed = [cp.sum(cp.multiply(z, K[:, :, k])) for k in range(2*n_points)]
    objective = cp.Minimize(max_overlap)
    
    # Add max constraint:
    for k in range(2*n_points):
        constraints.append(cp.constraints.NonPos(max_overlap - overlap_constructed[k]))
        constraints.append(cp.constraints.NonPos(overlap_constructed[k] - max_overlap))
    
    # Add constraints
    constraints.append(integral_constraint)
    
    # Formulate and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status == cp.OPTIMAL:
            a_sol = a.value
            # Reconstruct h function
            h_reconstructed = np.zeros(n_points)
            for i in range(n_points):
                for k in range(30):
                    h_reconstructed[i] += a_sol[k] * fourier_basis(x_points[i], k)
            
            # Ensure h values are in [0,1]
            h_reconstructed = np.clip(h_reconstructed, 0, 1)
            # Normalize to satisfy integral constraint
            h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
            
            c5_bound = evaluate_c5(h_reconstructed, dx)
            return (h_reconstructed.tolist(), c5_bound, n_points)
    except:
        pass

    # Fallback to just returning the best greedy construction
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based refinement
    max_iter = 1000
    learning_rate = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient ascent on overlap function
        current_c5 = evaluate_c5(best_h, dx)
        
        # Compute gradient (approximate)
        eps = 1e-4
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[j] = (c5_plus - current_c5) / eps
        
        best_h = best_h - learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        new_c5 = evaluate_c5(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
