import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1e-4  # Initial temperature
    alpha = 0.99  # Cooling rate
    min_T = 1e-10  # Minimum temperature
    
    # Track improvements
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        # Perturb solution
        h_new = h.copy()
        idx = random.randint(0, n_points - 1)
        step_size = 0.05 * T  # Adaptive step size
        
        # Perturb one element
        h_new[idx] += np.random.normal(0, step_size)
        
        # Ensure within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new)
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5 or np.random.random() < np.exp(-(new_c5 - best_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Cooling schedule
        T = max(min_T, T * alpha)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Reset if no improvement for too long
        if no_improvement_count > max_no_improvement:
            # Reinitialize with a better starting point
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            T = 1e-4  # Reset temperature

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
