import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    max_iter = 2000
    tolerance = 1e-4
    alpha = 0.01
    
    # Create Fourier basis functions
    x = np.linspace(0, 2, n_points, endpoint=False)
    basis = []
    for k in range(1, n_basis + 1):
        if k % 2 == 1:
            basis.append(np.sin(np.pi * k * x))
        else:
            basis.append(np.cos(np.pi * k * x))
    basis = np.array(basis)
    
    # Compute inner products between basis functions
    inner_products = np.dot(basis, basis.T) * dx
    
    # Compute integral of each basis function
    basis_integrals = np.sum(basis, axis=1) * dx
    
    # Initialize coefficients
    c = Variable(n_basis)
    
    # Project initial solution onto basis if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Project h onto basis
        coefficients = np.dot(basis.T, h_init) * dx
        c.value = coefficients
    else:
        # Fallback: use an initial smooth approximation
        c.value = np.random.rand(n_basis) * 0.5
    
    # Add constraint for integral
    constraint_sum = Equal(basis_integrals @ c, 1)
    
    # Declaration of variables
    d = Variable((n_basis, n_basis))
    theta = Variable(n_basis)
    
    # McCormick Envelope constraints
    constraints = [0 <= c, c <= 1]
    for i in range(n_basis):
        for j in range(n_basis):
            # d[i,j] = c[i] * (1 - c[j]) 
            constraints.append(d[i,j] >= c[i] - theta[i])
            constraints.append(d[i,j] >= theta[i] - c[j] + alpha)
            constraints.append(d[i,j] <= 1 - c[j])
    
    # Objective: minimize the maximum overlap
    # Manually compute overlaps in frequency domain 
    # For computational simplicity, optimize a weighted sum
    coeffs = np.ones(n_basis)
    weights = np.ones((n_basis, n_basis))
    max_overlap = Variable()
    
    # Define objective
    objective = Minimize(max_overlap)
    
    # Initialize model and solve
    prob = Problem(objective, constraints + [max_overlap >= sum(d[i,j] for i in range(n_basis) for j in range(n_basis))])
    
    best_bound = float('inf')
    best_h = None
    
    # Iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=100)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                # Recompute h from coefficients
                h_values = basis.T @ c.value * dx
                h_values = np.clip(h_values, 0, 1)
                h_values = h_values * (n_points / 2) / np.sum(h_values)
                
                # Evaluate overlap
                overlap = np.correlate(h_values, 1 - h_values, mode="full") * dx
                current_bound = np.max(overlap)
                
                if current_bound < best_bound:
                    best_bound = current_bound
                    best_h = h_values.copy()
                
                if best_bound <= 0.3807:
                    break
                        
        except Exception:
            # Fallback to optimization with reduced complexity
            try:
                # Simplified approach: minimize maximum over all lags
                overlap = np.zeros(2 * n_points - 1)
                for i in range(2 * n_points - 1):
                    overlap[i] = np.sum(h_values * np.roll(1 - h_values, i - n_points + 1))
                current_bound = np.max(overlap) * dx

            except:
                break

        # Generate new coefficients with small perturbations
        c.value += np.random.randn(n_basis) * 0.01
        c.value = np.clip(c.value, 0, 1)
    
    # Return best result found
    if best_h is None:
        # Fallback to initial solution if no better found
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            # Simple identity solution 
            best_h = np.ones(n_points)
            best_h[n_points//2:] = 0
            best_h = best_h * (n_points / 2) / np.sum(best_h)

    # Final evaluation
    overlap_final = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_bound = np.max(overlap_final)
    
    return (best_h.tolist(), final_bound, n_points)
