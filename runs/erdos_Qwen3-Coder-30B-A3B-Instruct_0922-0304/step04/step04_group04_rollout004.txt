import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with the initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    else:
        # Fallback: simple step function
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Initialize variables for optimization
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current, dx)
    
    # Fourier representation approach
    # Using cosine and sine basis functions
    n_basis = 20  # Number of basis functions
    
    # Precompute basis functions and integrals
    x = np.linspace(0, 2, n_points, endpoint=False)
    basis = []
    basis_integrals = []
    
    for k in range(n_basis):
        if k % 2 == 0:
            # Cosine basis
            phi = np.cos(np.pi * k * x / 2)
        else:
            # Sine basis
            phi = np.sin(np.pi * k * x / 2)
        basis.append(phi)
        basis_integrals.append(np.sum(phi) * dx)
    
    basis_integrals = np.array(basis_integrals)
    
    # Optimize using convex relaxation
    max_iter = 1000
    alpha = 0.01  # McCormick relaxation parameter
    tolerance = 1e-4
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Project current h onto Fourier basis
        fourier_coeffs = []
        for i in range(n_basis):
            coeff = np.sum(h_current * basis[i]) * dx
            fourier_coeffs.append(coeff)
        fourier_coeffs = np.array(fourier_coeffs)
        
        # CVXPY optimization setup
        try:
            # Variables
            c = cp.Variable(n_basis)
            d = cp.Variable(n_basis * n_basis)
            
            # Constraints
            constraints = []
            
            # Integral constraint
            integral_constraint = cp.sum(cp.multiply(c, basis_integrals)) == 1
            constraints.append(integral_constraint)
            
            # Non-negativity and bounds
            for i in range(n_basis):
                constraints.append(c[i] >= 0)
                constraints.append(c[i] <= 1)
                
            # McCormick envelopes for product terms
            # For each pair of basis functions we want d[i*n_basis + j] = c[i]*c[j]
            # This can be relaxed using McCormick envelopes
            cnt = 0
            for i in range(n_basis):
                for j in range(n_basis):
                    # d[i*n_basis + j] = c[i] * c[j]
                    # We model it as two variables, one holds the value of this product
                    d[cnt] = cp.Variable()
                    constraints.append(d[cnt] >= c[i] + c[j] - 1)
                    constraints.append(d[cnt] >= 0)
                    constraints.append(d[cnt] <= c[i])
                    constraints.append(d[cnt] <= c[j])
                    cnt += 1
            
            # Objective: minimize maximum overlap
            # Note: In the original approach we evaluated all correlations, not just max_k
            # for now assuming that we approximate something like C_5 = sum(r[i] * (1 - h[i]))
            
            # To do this more accurately, we'll compute overlaps in the Fourier domain
            # But let's optimize the full objective Function using the already computed data
            h_reconstructed = np.sum([c[i] * basis[i] for i in range(n_basis)], axis=0)
            if np.sum(h_reconstructed) > 0:
                normalized_h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
            else:
                normalized_h_reconstructed = h_reconstructed
            
            # This is the most important step: reconstruct h(x) and compute overlap
            h_eval = normalized_h_reconstructed
            overlap = correlate(h_eval, 1 - h_eval, mode="full")
            max_overlap = np.max(overlap) * dx
            
            # Direct loss function using CVXPY
            # Maximize coefficient over the max overlap term
            obj = cp.Minimize(max_overlap)
            
            # Solve and update
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS, verbose=False, max_iters=100)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_coeffs = c.value
                h_norm = np.sum([new_coeffs[i] * basis[i] for i in range(n_basis)], axis=0)
                h_norm = h_norm * (n_points / 2) / np.sum(h_norm)
                new_c5 = evaluate_c5(h_norm, dx)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_norm.copy()
                    
        except Exception:
            pass  # Graphical solvers might fail occasionally, so proceed
            
        # Simple local neighborhood search
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
            neighbour = h_current + np.random.normal(0, 0.01, n_points)
            neighbour = np.clip(neighbour, 0, 1)
            neighbour = neighbour * (n_points / 2) / np.sum(neighbour)
            new_c5 = evaluate_c5(neighbour, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = neighbour.copy()
        
        h_current = best_h.copy()
        
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
