import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 bound from a step function h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use a geometric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        # Create a smooth transition
        for i in range(n_points):
            if i < mid:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Add some smoothness to reduce sharp discontinuities
        h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Coordinate-wise optimization with learning rate
    lr = 0.01
    max_iters = 5000
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Shuffle coordinates for better exploration
        indices = np.random.permutation(n_points)
        
        improved = False
        for i in indices:
            if time.time() - start_time > budget_s - 1:
                break
                
            # Save original value
            original_h_i = h[i]
            
            # Try small perturbations around current value
            perturbations = [0.0, -lr, lr]
            
            current_c5 = evaluate_c5(h, dx)
            best_h_i = original_h_i
            best_c5_i = current_c5
            
            for delta in perturbations:
                new_h_i = original_h_i + delta
                new_h_i = np.clip(new_h_i, 0, 1)
                
                # Temporarily update
                old_val = h[i]
                h[i] = new_h_i
                
                # Normalize
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                h = np.clip(h, 0, 1)
                
                new_c5 = evaluate_c5(h, dx)
                
                if new_c5 < best_c5_i:
                    best_c5_i = new_c5
                    best_h_i = new_h_i
                
                # Restore
                h[i] = old_val
                
            # Apply best update
            if best_h_i != original_h_i:
                h[i] = best_h_i
                improved = True
                
                # Re-normalize
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                h = np.clip(h, 0, 1)
                
        # Check for improvement
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if current_c5 <= 0.3807:
            break
            
        if no_improvement > patience:
            # Minor random perturbation
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement = 0
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
