import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:

        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Dual decomposition approach
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Step 1: Optimize magnitudes with fixed support
        # Create support mask from current h
        support = (h > 1e-6).astype(int)
        
        # Optimize magnitudes with constraints
        # Using a simple gradient-based approach with projection
        h_old = h.copy()
        
        # Simple gradient descent for magnitudes
        learning_rate = 0.01
        for _ in range(10):  # Inner loop for magnitude optimization
            if time.time() - start_time > budget_s - 1:
                break
            # Compute gradient (simplified)
            # This is a crude approximation of gradient descent
            h_new = h_old + learning_rate * (1 - 2 * h_old)  # Rough gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                h = h_new.copy()
            else:
                h = h_old.copy()
            
            h_old = h.copy()
            
        # Step 2: Optimize support location
        # Rough heuristic: shift support to reduce overlaps
        support = (h > 1e-6).astype(int)
        if np.sum(support) > 0:
            # Calculate where to move support to reduce overlap
            # This is a local search approach
            current_c5 = evaluate_c5(h, dx)
            best_shift = 0
            best_shift_c5 = current_c5
            
            # Try small shifts
            for shift in range(-5, 6):
                if shift == 0:
                    continue
                # Shift support
                shifted_support = np.zeros(n_points)
                if shift > 0:
                    shifted_support[shift:] = support[:-shift]
                else:
                    shifted_support[:shift] = support[-shift:]
                
                # Create new h from shifted support
                h_shifted = np.zeros(n_points)
                h_shifted[shifted_support == 1] = h[shifted_support == 1]
                h_shifted = h_shifted * (n_points / 2) / np.sum(h_shifted)
                
                shifted_c5 = evaluate_c5(h_shifted, dx)
                if shifted_c5 < best_shift_c5:
                    best_shift_c5 = shifted_c5
                    best_shift = shift
                    
            # Apply best shift if it reduces C5
            if best_shift != 0:
                shifted_support = np.zeros(n_points)
                if best_shift > 0:
                    shifted_support[best_shift:] = support[:-best_shift]

                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
                    h = h_new.copy()
                    
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
