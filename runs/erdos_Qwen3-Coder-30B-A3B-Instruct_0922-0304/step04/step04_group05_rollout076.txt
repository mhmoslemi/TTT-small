import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential Convex Programming approach
    max_iter = 1000
    alpha = 0.1
    mu = 1e-3
    damping = 0.5
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian approximation for quadratic model
        # We'll linearize the overlap at current h
        h1 = 1 - h
        
        # Compute gradient of max overlap with respect to h
        # This is a complex operation - we'll use finite differences for simplicity
        grad = np.zeros(n_points)
        epsilon = 1e-5
        
        # Approximate gradient via finite differences of overlap terms
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = evaluate_c5(h_plus, dx)
            overlap_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        
        # Form qudratic approximation
        # Manually simplify for efficiency: steep descent step
        # Compute h1 again for correlation
        h1 = 1 - h
        
        # Create a quadratic model to solve
        # min 0.5 * delta^T * H * delta + grad^T * delta
        # subject to bounds and sum constraint
        # For convex optimization, we solve convex subproblem with barrier terms
        
        # Use CVXPY approach
        delta = cp.Variable(n_points)
        
        # Objective, quadratic approximation:
        # We use a simplified model: just linearized terms, not fully second order
        objective = cp.Minimize(cp.sum(cp.multiply(grad, delta)))
        
        # Constraints
        constraints = [
            cp.sum(delta) == 0,  # Maintain sum
            delta >= -h,  # Lower bound
            delta <= 1 - h,  # Upper bound
        ]
        
        # Create the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            delta_vals = delta.value
        except:
            # If optimization fails, fallback to gradient descent
            delta_vals = -alpha * grad
        
        # Update h
        if delta_vals is not None and not np.isnan(delta_vals).any():
            h_new = h + alpha * delta_vals
            h_new = np.clip(h_new, 0, 1)
            # Enforce integral constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
                if best_c5 <= 0.3807:
                    break
            elif new_c5 >= best_c5:
                alpha *= damping  # Reduce step size
            else:
                h = h_new.copy()
        else:
            # Fallback: simple gradient descent with small step
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, dx)
            
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
                if best_c5 <= 0.3807:
                    break
            elif new_c5 >= best_c5:
                alpha *= damping  # Reduce step size
            else:
                h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
