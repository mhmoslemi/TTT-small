import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute C5 for a given h"""
    # h is assumed to be normalized to sum to 1/dx = n_points/2
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    max_vel = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    
    for i in range(n_particles):
        if i == 0:
            # Start with initial solution
            particle_h = h.copy()
        else:
            # Perturb the initial solution slightly
            particle_h = h + np.random.normal(0, 0.02, n_points)
            particle_h = np.clip(particle_h, 0, 1)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            
        particles.append(particle_h)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [compute_c5(p) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (omega * velocities[i] + 
                 phi_p * r1 * (personal_bests[i] - particles[i]) + 
                 phi_g * r2 * (global_best - particles[i]))
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            particles[i] += v
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Ensure sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_score = compute_c5(particles[i])
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_bests[i] = particles[i].copy()
                
                # Update global best if needed
                if current_score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = current_score
                    
        # Early stopping condition
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
