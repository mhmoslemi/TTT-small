import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n_points = len(h)
    max_overlap = 0
    for k in range(2 * n_points - 1):
        overlap = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Initial construction: piecewise constant function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimize using scipy
    def objective(x):
        # x contains [a, d] parameters
        a, d = x
        # Ensure they are within bounds
        a = max(0, min(1 - d - 0.5, a))
        d = max(0, min(1 - 1.0, d))
        
        # Construct step function
        h_new = np.zeros(n_points)
        w = 0.5  # Each step has width 0.5 to satisfy integral constraint
        start1 = int(a * n_points / 2.0)  # Relative start position
        end1 = start1 + int(w * n_points / 2.0)
        start2 = end1 + int(d * n_points / 2.0)
        end2 = start2 + int(w * n_points / 2.0)
        
        # Ensure bounds
        start1 = max(0, min(n_points - int(w * n_points / 2.0), start1))
        end1 = min(n_points, start1 + int(w * n_points / 2.0))
        start2 = max(0, min(n_points - int(w * n_points / 2.0), start2))
        end2 = min(n_points, start2 + int(w * n_points / 2.0))
        
        h_new[start1:end1] = 1.0
        h_new[start2:end2] = 1.0
        
        # Normalize to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        return evaluate_c5(h_new, dx)
    
    # Grid search for initial parameters
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Try different approaches
    for i in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random initialization
        a = np.random.uniform(0, 1 - 0.5)
        d = np.random.uniform(0, 1 - 1.0)
        x0 = [a, d]
        
        try:
            res = minimize(objective, x0, method='Nelder-Mead', options={'maxiter': 100})
            if res.success:
                a_opt, d_opt = res.x
                # Reconstruct h with optimal parameters
                h_new = np.zeros(n_points)
                w = 0.5
                start1 = max(0, min(n_points - int(w * n_points / 2.0), int(a_opt * n_points / 2.0)))
                end1 = start1 + int(w * n_points / 2.0)
                start2 = end1 + int(d_opt * n_points / 2.0)
                end2 = start2 + int(w * n_points / 2.0)
                
                # Ensure bounds
                start1 = max(0, min(n_points - int(w * n_points / 2.0), start1))
                end1 = min(n_points, start1 + int(w * n_points / 2.0))
                start2 = max(0, min(n_points - int(w * n_points / 2.0), start2))
                end2 = min(n_points, start2 + int(w * n_points / 2.0))
                
                h_new[start1:end1] = 1.0
                h_new[start2:end2] = 1.0
                
                # Normalize to satisfy integral constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except:
            continue
    
    # Try refining with a simpler search over a few parameters
    for i in range(50):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb existing solution
        h_change = np.copy(best_h)
        idx = np.random.randint(0, n_points)
        h_change[idx] += np.random.normal(0, 0.05)
        h_change = np.clip(h_change, 0, 1)
        h_change = h_change * (n_points / 2) / np.sum(h_change)
        
        c5_new = evaluate_c5(h_change, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_change.copy()
    
    # Final refinement using the best known h
    return (best_h.tolist(), best_c5, n_points)
