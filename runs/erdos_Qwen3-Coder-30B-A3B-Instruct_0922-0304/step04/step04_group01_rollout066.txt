import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to symmetric triangular window
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, h[::-1]])
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.05
    
    # Initialize particles and velocities
    particles = np.array([h.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    
    # Normalize particles
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Evaluate initial particles
    fitness = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Find best particles
    pbest = particles.copy()
    pbest fitness = fitness.copy()
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary conditions
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new fitness
            new_fitness = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
        
        # Periodic reinitialization to avoid stagnation
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(n_particles)
                particles[idx] = np.random.uniform(0, 1, n_points)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
