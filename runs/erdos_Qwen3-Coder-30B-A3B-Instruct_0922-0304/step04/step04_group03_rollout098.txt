import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise use a symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with a piecewise constant construction
        h = np.zeros(n_points)
        # Two intervals of width 0.5 each, separated by distance d
        # We'll search over possible d values to find best
        w = 0.5  # width of each interval
        d_candidates = np.arange(0, 1.001, 0.01)  # step size 0.01
        best_c5 = float('inf')
        best_h = None
        
        for d in d_candidates:
            # Calculate indices for the two intervals
            # First interval [0, w]
            i0 = 0
            i1 = int(np.floor(w / dx))
            # Second interval [w+d, w+d+w]
            i2 = int(np.ceil((w + d) / dx))
            i3 = int(np.floor((w + d + w) / dx))
            
            # Make sure intervals don't overlap
            if i2 <= i1:
                # Create a valid h
                h_temp = np.zeros(n_points)
                h_temp[i0:i1+1] = 1
                h_temp[i2:i3+1] = 1
                
                # Normalize
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                # Evaluate
                c5 = evaluate_c5(h_temp, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_temp.copy()
        
        h = best_h if best_h is not None else h
    
    # Local search around the best known solution
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Add a few more local search steps
    for iter_num in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make small random changes
        h_new = h.copy()
        # Randomly perturb some indices
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.01)
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
