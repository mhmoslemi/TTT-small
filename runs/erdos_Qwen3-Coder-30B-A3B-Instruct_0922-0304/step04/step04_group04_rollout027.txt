import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    max_iter = 2000
    tolerance = 1e-4
    alpha = 0.01  # McCormick relaxation tolerance
    
    # Create Fourier basis functions
    def fourier_basis(x, k):
        if k % 2 == 0:
            return np.cos(np.pi * k * x)
        else:
            return np.sin(np.pi * k * x)
    
    # Define x points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Compute basis integrals
    basis_integrals = np.zeros(n_basis)
    for k in range(n_basis):
        basis_integrals[k] = np.trapz(fourier_basis(x, k), x)
    
    # Project initial values onto Fourier basis if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Convert to Fourier coefficients
        coeffs_init = np.zeros(n_basis)
        for k in range(n_basis):
            coeffs_init[k] = np.trapz(h_init * fourier_basis(x, k), x)
        coeffs = coeffs_init.copy()
    else:
        # Start with a simple step function
        coeffs = np.zeros(n_basis)
        coeffs[0] = 1.0  # DC component
        
    # Initialize the optimization variables
    c = cp.Variable(n_basis)
    d = cp.Variable(n_basis * n_basis)  # Compressed variable for the square terms (d_k = c_k * (1 - c_k))
    
    # Create McCormick envelopes for d_k = c_k * (1 - c_k)
    # 0 <= c_k <= 1, 0 <= d_k <= 1
    # d_k >= c_k - 1, d_k >= -c_k, d_k <= c_k, d_k <= 1 - c_k
    
    constraints = []
    
    # Integral constraint
    constraints.append(cp.sum(cp.multiply(c, basis_integrals)) == 1)
    
    # Bounds on coefficients
    for i in range(n_basis):
        constraints.append(c[i] >= 0)
        constraints.append(c[i] <= 1)
    
    # McCormick envelopes for d_k = c_k * (1 - c_k)  
    for i in range(n_basis):
        for j in range(n_basis):
            idx = i * n_basis + j
            # d_kj <= c_k
            constraints.append(d[idx] <= c[i])
            # d_kj <= 1 - c_j  
            constraints.append(d[idx] <= 1 - c[j])
            # d_kj >= c_k - 1
            constraints.append(d[idx] >= c[i] - 1)
            # d_kj >= -c_j
            constraints.append(d[idx] >= -c[j])
            
    # Objective function: minimize max overlap across all shifts
    # This will be a hot fix to minimize the maximum overlap term
    # But let's just minimize it using a method similar to what we had before
    
    # Simplified approach: optimize for a fixed number of iterations
    best_solution = coeffs.copy()
    best_score = evaluate_c5(np.clip(np.array([0.5] * n_points), 0, 1), dx)
    
    # Create a neighbor search/reinit approach
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Create a new candidate solution based on current coeffs
        new_h = np.zeros(n_points)
        for k in range(n_basis):
            new_h += coeffs[k] * fourier_basis(x, k)
        # Clip and normalize
        new_h = np.clip(new_h, 0, 1) 
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        score = evaluate_c5(new_h, dx)
        
        if score < best_score:
            best_solution = coeffs.copy()
            best_score = score
            
        # Small random perturbation to coefficient vector 
        perturb = np.random.normal(0, 0.05, n_basis)
        coeffs = np.clip(coeffs + perturb, 0, 1)
        
        iter_count += 1
        
    # Now compute final approximation using best solution
    final_h = np.zeros(n_points)
    for k in range(n_basis):
        final_h += best_solution[k] * fourier_basis(x, k)
    
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_score = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_score, n_points)
