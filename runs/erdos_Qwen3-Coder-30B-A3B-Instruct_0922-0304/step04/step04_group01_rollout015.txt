import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize particles around the starting solution
    for i in range(n_particles):
        pert = np.random.normal(0, 0.05, n_points)
        particles[i] = np.clip(h0 + pert, 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    
    # Parameters
    omega = 0.9
    phi_p = 2.0
    phi_g = 2.0
    inertia_damping = 0.99
    
    # Track personal bests
    pbest = particles.copy()
    pbest_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Find global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_c5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % 100 == 0 and iteration > 0:
            # Reinitialize some particles to avoid stagnation
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                particles[idx] = np.clip(h0 + np.random.normal(0, 0.1, n_points), 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            velocities[i] = (omega * velocities[i] + 
                            phi_p * r_p * (pbest[i] - particles[i]) + 
                            phi_g * r_g * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
                    
                    if gbest_c5 < best_c5:
                        best_c5 = gbest_c5
                        best_h = gbest.copy()
        
        # Apply inertia damping periodically
        if iteration % 100 == 0:
            omega *= inertia_damping
        
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
