import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        # Initialize with triangular function
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population with random feasible samples
    for i in range(1, population_size):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Evolutionary Strategy
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        elite_size = max(1, int(population_size * elite_ratio))
        
        # Update best solution
        if fitness[sorted_indices[0]] < best_c5:
            best_c5 = fitness[sorted_indices[0]]
            best_h = population[sorted_indices[0]].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite_indices = sorted_indices[:elite_size]
        elite_population = [population[i] for i in elite_indices]
        elite_fitness = [fitness[i] for i in elite_indices]
        
        # Generate offspring viamutation
        offspring = []
        for i in range(population_size - len(elite_population)):
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx].copy()
            mutation = np.random.normal(0, mutation_scale, n_points)
            child = parent + mutation
            child = project_feasible(child, n_points)
            offspring.append(child)
            
        # Replace population with elite + offspring
        population = elite_population + offspring
        
        # Adaptive mutation scaling
        if iteration % 20 == 0 and iteration > 0:
            if np.std(elite_fitness) < 1e-5:
                mutation_scale *= 0.9
            else:
                mutation_scale *= 1.05
    
    # Final evaluation for best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to first individual
        return (population[0].tolist(), evaluate_c5(population[0], dx), n_points)
