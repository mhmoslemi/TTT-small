import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting-plane approach with CVXPY
    # Define variables
    h_var = cp.Variable(n_points)
    c_var = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Cutting plane iterations
    max_iter = 2000
    step_size = 0.001
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps using correlate
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Add constraint: h · (1 - h_k) <= max_overlap - step_size
        # We approximate 1 - h_k by creating a shifted version of 1-h
        shifted_h = np.roll(1 - h, k_max - n_points + 1)
        constraint_expr = cp.sum(cp.multiply(h_var, shifted_h)) <= max_overlap - step_size
        
        # Add to constraints
        constraints.append(constraint_expr)
        
        # Formulate and solve the optimization problem
        objective = cp.Minimize(c_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-6, reltol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
                    # Evaluate new solution
                    new_c5 = compute_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        
                    # Reduce step size to encourage convergence
                    step_size = max(1e-6, step_size * 0.95)
                    
        except:
            # Restore previous solution if optimization fails
            pass
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Clean up constraints to avoid memory issues
        if len(constraints) > 30:
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
