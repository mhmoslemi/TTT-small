import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create initial two-interval step function
        h_init = np.zeros(n_points)
        # Two intervals of width 0.5 each
        w = 0.5
        # Place them at the start and end to spread out
        start1 = 0.0
        start2 = 2.0 - w
        
        i1_start = int(start1 / dx)
        i1_end = int((start1 + w) / dx)
        i2_start = int(start2 / dx)
        i2_end = int((start2 + w) / dx)
        
        h_init[i1_start:i1_end] = 1.0
        h_init[i2_start:i2_end] = 1.0
        
        # Normalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Optimization approach based on continuous relaxation
    # Minimize maximum overlap by adjusting positions
    # Here we use a grid-search approach over a few key parameters
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Grid of interval placements around [0,1] and [1,2] 
    # to maximize spread and reduce overlap
    n_search = 20
    a_vals = np.linspace(0.0, 1.0, n_search)
    b_vals = np.linspace(0.0, 1.0, n_search)
    
    # Try different interval widths
    widths = [0.3, 0.4, 0.5, 0.6, 0.7]
    
    # Define function to create h from two intervals
    def make_h(a_val, b_val, w):
        h = np.zeros(n_points)
        i1_start = int(a_val / dx)
        i1_end = int((a_val + w) / dx)
        i2_start = int(b_val / dx)
        i2_end = int((b_val + w) / dx)
        
        # Handle wrapping properly
        if i1_end > n_points:
            h[i1_start:] = 1.0
            h[:i1_end - n_points] = 1.0
        else:
            h[i1_start:i1_end] = 1.0
            
        if i2_end > n_points:
            h[i2_start:] = 1.0
            h[:i2_end - n_points] = 1.0
        else:
            h[i2_start:i2_end] = 1.0
            
        # Normalize to area 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    for w in widths:
        for a in a_vals:
            for b in b_vals:
                # Ensure intervals are within bounds
                if a + w > 2.0 or b + w > 2.0:
                    continue
                if a + w > b:
                    continue
                if b + w > 2.0:
                    continue
                h_new = make_h(a, b, w)
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    
                if time.time() - start_time > budget_s - 1:
                    return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
