import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    return h * (target_sum / current_sum)

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Population-based GA approach
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 1.0
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        new_h = h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, target_sum)
        population.append(new_h)
    
    # Function to select parents via tournament selection
    def select_parents(population, fitnesses):
        selected = []
        for _ in range(len(population)):
            tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected
    
    # Function to perform crossover
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Function to mutate
    def mutate(individual, mutation_rate):
        mask = np.random.random(n_points) < mutation_rate
        individual += np.random.normal(0, 0.01, n_points) * mask
        individual = np.clip(individual, 0, 1)
        return normalize_h(individual, target_sum)
    
    # Evaluate initial population
    fitnesses = [evaluate_individual(p, dx) for p in population]
    
    best_fitness = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    # Main GA loop
    max_generations = 1000
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best solution
        current_best_fitness = sorted_fitnesses[0]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = sorted_population[0].copy()
        
        if best_fitness <= 0.3807:
            break
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Elitism
        new_population.extend(sorted_population[:elite_size])
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Local search on top individuals
        for i in range(min(elite_size, len(new_population))):
            individual = new_population[i]
            
            # Local refinement with SLSQP
            def objective(h):
                return evaluate_individual(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            result = minimize(
                objective,
                individual,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                improved_individual = result.x
                improved_individual = np.clip(improved_individual, 0, 1)
                improved_individual = normalize_h(improved_individual, target_sum)
                new_population[i] = improved_individual
        
        population = new_population
        fitnesses = [evaluate_individual(p, dx) for p in population]
    
    # Final evaluation
    final_fitness = evaluate_individual(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
