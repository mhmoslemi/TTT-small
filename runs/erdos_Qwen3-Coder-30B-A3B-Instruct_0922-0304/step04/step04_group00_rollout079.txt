import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=2.0/100):
    """Compute the C5 bound for given h values."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Initialize particle with slight perturbation around h
        particle_h = h + np.random.normal(0, 0.02, n_points)
        particle_h = np.clip(particle_h, 0, 1)
        particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
        particles.append(particle_h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        p_best.append(particle_h.copy())
        p_best_vals.append(compute_c5(particle_h, dx))
    
    # Global best
    global_best_idx = np.argmin(p_best_vals)
    g_best = p_best[global_best_idx].copy()
    g_best_val = p_best_vals[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_val = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
                    
                    # Early exit if target achieved
                    if g_best_val <= 0.3807:
                        break
                        
        # Early exit if target achieved
        if g_best_val <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_val, n_points)
