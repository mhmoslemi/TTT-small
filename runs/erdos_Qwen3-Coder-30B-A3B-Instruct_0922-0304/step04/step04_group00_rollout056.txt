from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with smooth step
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute indices for efficient correlation
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordiante-wise optimization with dual normalization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Backup current value
            old_h_i = h[i]
            
            # Try several candidate values for h[i]
            candidates = []
            current_sum = np.sum(h)
            new_sum = current_sum - old_h_i
            
            # Generate candidates around old_h_i
            for candidate_h_i in [0.0, 0.25, 0.5, 0.75, 1.0]:
                new_h = h.copy()
                new_h[i] = candidate_h_i
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                try:
                    c5_candidate = compute_c5(new_h)
                    candidates.append((c5_candidate, new_h[i]))
                except:
                    continue
                    
            if candidates:
                min_c5, best_h_i = min(candidates, key=lambda x: x[0])
                if min_c5 < current_c5:
                    h[i] = best_h_i
                    h = h * (n_points / 2) / np.sum(h)
        
        # Periodic normalization
        if iteration % 50 == 0:
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
