import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    max_overlap = 0
    for k in range(1, 2*n - 1):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx

    start_time = time.time()

        start1 = 0.25
        start2 = 1.25
        i1_start = int(start1 / dx)
        i1_end = int((start1 + w) / dx)
        i2_start = int(start2 / dx)
        i2_end = int((start2 + w) / dx)
        h[i1_start:i1_end] = 1.0
        h[i2_start:i2_end] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    # Refine using gradient-free optimization
    def objective(params):
        # params: [a1, a2, d] where intervals [a1, a1+w] and [a2, a2+w] 
        a1, a2, d = params
        w = 0.25  # Fixed width
        h = np.zeros(n_points)
        i1_start = int(max(0, a1) / dx)
        i1_end = int(min(2, a1 + w) / dx)
        i2_start = int(max(0, a2) / dx)
        i2_end = int(min(2, a2 + w) / dx)
        if i1_end > i1_start:
            h[i1_start:i1_end] = 1.0
        if i2_end > i2_start:
            h[i2_start:i2_end] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Sample strategies from a smaller grid
    best_h = h.copy()

    ])
    
    # Evaluate each point
    for param in search_space:
