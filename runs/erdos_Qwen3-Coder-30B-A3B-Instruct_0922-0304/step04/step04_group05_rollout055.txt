import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradients_hessians(h, dx, n_points):
    # Compute gradient and Hessian of overlap function at h
    # Gradient: ∇C_k = dx * (1 - 2*h - shift_avg) where shift_avg is sum of h shifted by k
    # Hessian: ∇²C_k = dx * (-2*I + shift_matrix)
    
    # Precompute shift averages
    shift_averages = []
    for k in range(2 * n_points - 1):
        # Compute average of h shifted by k positions
        shift_avg = np.zeros(2 * n_points - 1)
        if k < n_points:
            # Left shift
            shift_avg[k:k+n_points] = h
        else:
            # Right shift
            shift_avg[:n_points-(k-n_points+1)] = h[(k-n_points+1):]
        # Take average
        shift_avg = shift_avg / n_points
        shift_averages.append(shift_avg)
    
    # For simplicity, estimate gradient and Hessian using first-order approximation
    # This is a heuristic simplification and may not capture the full complexity
    grad = dx * (1 - 2 * h - np.mean(shift_averages, axis=0)[:n_points])
    hess = dx * (-2 * np.eye(n_points) + np.eye(n_points, k=0) * 2) # Incorrect Hessian for simplicity
    return grad, hess

def solve_qp_cvxpy(H, c, A, b, delta_low, delta_high, n_points):
    # Solve QP: minimize (1/2)*x^T*H*x + c^T*x subject to A*x = b, delta_low <= x <= delta_high
    delta = cp.Variable(n_points)
    
    # Objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [A @ delta == b]
    for i in range(n_points):
        constraints.append(delta[i] >= delta_low[i])
        constraints.append(delta[i] <= delta_high[i])
    
    # Problem
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def project_h(h, n_points, dx):
    # Ensure h is within bounds and has correct integral
    h = np.clip(h, 0, 1)
    # Adjust to have correct integral
    integral = np.sum(h) * dx
    h = h * (n_points / 2) / integral 
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Normalize
        integral = np.sum(h) * dx
        h = h * (n_points / 2) / integral
        h = np.clip(h, 0, 1)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        integral = np.sum(h) * dx
        h = h * (n_points / 2) / integral
        h = np.clip(h, 0, 1)
    
    # Identity matrix for constraints
    A_eq = np.ones((1, n_points))
    b_eq = np.array([n_points / 2])
    
    # Initialize best solutions
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main Iterative Block
    max_iter = 500  # Same as old eval budget
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Estimate gradient and Hessian (approximate)
        grad = np.zeros(n_points)
        hess = np.eye(n_points) * (-2 * dx)
        # Add some approximation of shift matrix contribution
        for j in range(1, n_points):
            hess += np.eye(n_points, k=j) * dx
            hess += np.eye(n_points, k=-j) * dx
            
        # Regularize the Hessian
        hess += 1e-5 * np.eye(n_points)
        
        # Solve QP with cvxpy
        delta = solve_qp_cvxpy(
            hess,
            grad,
            A_eq,
            b_eq,
            np.full(n_points, -1.0),  # delta_lower_bound
            np.full(n_points, 1.0),   # delta_upper_bound
            n_points
        )
        
        if delta is None:
            # If solve fails, try with increased regularization
            hess += 1e-3 * np.eye(n_points)
            delta = solve_qp_cvxpy(
                hess,
                grad,
                A_eq,
                b_eq,
                np.full(n_points, -1.0),
                np.full(n_points, 1.0),
                n_points
            )
        
        if delta is not None:
            # Line search
            alpha = 1.0
            improved = False
            
            while alpha > 1e-6:
                h_new = h + alpha * delta
                h_new = project_h(h_new, n_points, dx)
                c5_new = evaluate_c5(h_new, dx)
                
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    alpha *= 0.5
            
        else:
            # Replace with linear search heuristic if no improvement
            # This is a fallback for numerical instability or infeasible QP
            # Take a small step in random direction
            step = np.random.normal(0, 0.01, n_points)
            alpha = 0.1  # Step size
            h_new = h + alpha * step
            h_new = project_h(h_new, n_points, dx)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
