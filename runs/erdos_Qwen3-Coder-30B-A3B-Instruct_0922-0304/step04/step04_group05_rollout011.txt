import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to match sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def frank_wolfe_step(h, grad, dx, n_points, step_size=0.01):
    # Frank-Wolfe approach
    # Move in direction of -grad to update h
    # Maintain constraints
    # We compute delta = -step_size * grad
    # Then clip to [0,1] and then scale to make sum correct
    delta = - step_size * grad
    
    # Project to non-negative
    new_h = h + delta
    new_h = np.clip(new_h, 0, 1)
    
    # Project sum constraint
    current_sum = np.sum(new_h)
    if current_sum > 0:
        new_h = new_h * (n_points / 2) / current_sum
    
    # Return step
    return new_h
