import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a piecewise constant function with two intervals
        h = np.zeros(n_points)
        # Set first interval from 0 to 0.5
        i0 = 0
        i1 = int(0.5 / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        h[i0:i1+1] = 1.0
        
        # Set second interval from 1.0 to 1.5 
        i2 = int(1.0 / dx)
        i3 = int(1.5 / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
        if i2 < n_points:
            h[i2:i3+1] = 1.0
            
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # 1D grid search over the distance between intervals
    d_candidates = np.arange(0.0, 1.0 + 0.001, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create piecewise constant function with two intervals
        h_test = np.zeros(n_points)
        
        # First interval [0, 0.5]
        i0 = 0
        i1 = int(0.5 / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        h_test[i0:i1+1] = 1.0
        
        # Second interval [0.5+d, 1.0+d]
        i2 = int((0.5 + d) / dx)
        i3 = int((1.0 + d) / dx) - 1
        if i3 >= n_points:
            i3 = n_points - 1
        if i2 < n_points and i3 >= i2:
            h_test[i2:i3+1] = 1.0
            
        # Normalize to satisfy constraint ∫h dx = 1
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Compute C5
        c5 = evaluate_c5(h_test, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
            
    return (best_h.tolist(), best_c5, n_points)
