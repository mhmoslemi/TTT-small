import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 overlap integral for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum, clipping values to [0, 1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def evaluate_population(population, dx):
    """Evaluate all individuals in population."""
    scores = []
    for h in population:
        scores.append(compute_c5(h, dx))
    return np.array(scores)

def tournament_selection(population, scores, k=5):
    """Select individuals using tournament selection."""
    selected = []
    for _ in range(len(population) // 2):
        tournament_indices = np.random.choice(len(population), k, replace=False)
        tournament_scores = scores[tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_scores)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    """Single-point crossover."""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.01):
    """Mutate h with Gaussian noise."""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(h))
        h = h + noise
    return h

def local_search(h, dx, max_iter=50):
    """Local search using SLSQP."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - 70 / 2  # n_points = 70
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(len(h))]
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-8})
    if result.success:
        h = result.x
    return normalize_h(h, 70 / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values or create base
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
    
    # Create initial population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = base_h + np.random.normal(0, 0.02, n_points)
        population.append(normalize_h(h, target_sum))
    
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    generations = 0
    max_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 1.0
    
    while time.time() - start_time < budget_s - 1 and generations < max_generations:
        # Evaluate population
        scores = evaluate_population(population, dx)
        
        # Track best
        min_idx = np.argmin(scores)
        if scores[min_idx] < best_c5:
            best_c5 = scores[min_idx]
            best_h = population[min_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, scores)
        
        # Crossover
        new_population = []
        for i in range(0, len(selected), 2):
            p1 = selected[i]
            p2 = selected[(i + 1) % len(selected)] if i + 1 < len(selected) else selected[0]
            if np.random.random() < crossover_rate:
                c1, c2 = crossover(p1, p2)
                new_population.extend([normalize_h(c1, target_sum), normalize_h(c2, target_sum)])
            else:
                new_population.extend([p1.copy(), p2.copy()])
        
        # Mutation
        for i in range(len(new_population)):
            new_population[i] = mutate(new_population[i], mutation_rate)
            new_population[i] = normalize_h(new_population[i], target_sum)
        
        # Local search on top individuals
        top_indices = np.argsort(scores)[:10]
        for i in top_indices:
            new_population.append(local_search(population[i].copy(), dx))
        
        # Replace population
        population = new_population[:population_size]
        
        generations += 1
    
    # Final refinement
    if best_h is not None:
        final_h = local_search(best_h, dx)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    if best_h is None:
        # Fallback to base
        best_h = base_h
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
