import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis functions (up to scale 5 for n=30)
    def haar_wavelet(x, j, k):
        # j: scale, k: translation
        if j == 0 and k == 0:
            return 1.0 if 0 <= x < 1 else 0.0
        elif j == 0 and k == 1:
            return 1.0 if 1 <= x < 2 else 0.0
        else:
            # Compute Haar basis
            scale = 2**j
            translation = k / scale
            if 0 <= x < translation or translation + 1/scale <= x < 2:
                return 0.0
            elif translation <= x < translation + 0.5/scale:
                return 2**(j/2)
            elif translation + 0.5/scale <= x < translation + 1/scale:
                return -2**(j/2)
            else:
                return 0.0

    # Generate Haar basis matrix
    n_basis = 30
    basis = np.zeros((n_points, n_basis))
    
    # Simplified Haar basis generation (first few are simple)
    for i in range(n_basis):
        for j in range(n_points):
            x = j * dx
            if i == 0:
                # First basis: constant
                basis[j, i] = 1.0
            elif i == 1:
                # Step function
                basis[j, i] = 1.0 if x < 1 else 0.0
            elif i == 2:
                basis[j, i] = 1.0 if x >= 1 else 0.0
            elif i == 3:
                basis[j, i] = 1.0 if x < 0.5 else 0.0
            else:
                # For higher scales, use a simplified approach
                if x < 0.5:
                    basis[j, i] = 1.0 if i % 2 == 0 else -1.0
                else:
                    basis[j, i] = 1.0 if i % 2 == 1 else -1.0

    # Normalize basis
    for i in range(n_basis):
        if np.linalg.norm(basis[:, i]) > 0:
            basis[:, i] /= np.linalg.norm(basis[:, i])

    # Initialize coefficients
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Random initialization
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        
    # Express h_start in Haar basis
    coeffs = np.linalg.lstsq(basis, h_start, rcond=None)[0]
    # Ensure that integral is correct
    integral_constraint = np.sum(basis[:, 0]) * coeffs[0] * dx
    if integral_constraint != 0:
        coeffs[0] = (1.0 / dx) / np.sum(basis[:, 0])
    else:
        coeffs[0] = 1.0 / dx

    # Convex optimization approach
    # Variables
    c = Variable(n_basis)
    z = Variable((n_basis, n_basis))  # McCormick envelopes for c_i * c_j

    # Objective 
    # In practice, this becomes a problem to solve for maximum overlap, not directly dependent on coefficients
    # We are minimizing max_k int_0^2 h(x)(1-h(x+k)) dx
    
    # Avoid complexity of full McCormick envelopes.
    # Let's instead make a direct optimization:
    
    # Solve the problem using a clean alternating optimization approach
    
    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full") 
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    h_values = np.array(h_start)    
    best_h = h_values.copy()
    best_c5 = evaluate_c5(h_values)
    
    # Simple gradient-based approach with clipping
    max_iter = 5000
    lr = 0.001
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        h = h * (1.0 / h_sum)  # Renormalize
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
            
        # Gradient of overlap approximately by finite difference
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_pert = h.copy()
            h_pert[j] += eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert_sum = np.sum(h_pert) * dx
            h_pert = h_pert * (1.0 / h_pert_sum)
            
            overlap_pert = correlate(h_pert, 1 - h_pert, mode="full")
            max_overlap_pert = np.max(overlap_pert) * dx
            
            grad[j] = (max_overlap_pert - max_overlap) / eps
            
        # Update
        h_values = h_values - lr * grad
        h_values = np.clip(h_values, 0, 1) 
        h_sum = np.sum(h_values) * dx
        h_values = h_values * (1.0 / h_sum)

    # Final evaluation
    best_h = np.clip(best_h, 0, 1) 
    best_sum = np.sum(best_h) * dx
    best_h = best_h * (1.0 / best_sum)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
