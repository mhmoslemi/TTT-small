import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    max_overlap = 0
    for k in range(2*n - 1):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Simple triangular shape
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h_init = np.clip(h_init, 0, 1)
    
    # Normalize the initial solution
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Define objective function for optimization
    def objective(x):
        # x contains [a, d] where h = 1 on [a, a+w] and [a+w+d, a+w+d+w]
        # w = 0.5 (since we want total area of 1)
        a, d = x
        if a < 0 or a > 1 - d - 0.5 or d < 0 or d > 1 - 1:
            return float('inf')
        
        h = np.zeros(n_points)
        # First interval
        start1 = int(a / dx)
        end1 = int((a + 0.5) / dx)
        if end1 > n_points:
            end1 = n_points
        if start1 < n_points:
            h[start1:end1] = 1
        
        # Second interval
        start2 = int((a + 0.5 + d) / dx)
        end2 = int((a + 0.5 + d + 0.5) / dx)
        if end2 > n_points:
            end2 = n_points
        if start2 < n_points:
            h[start2:end2] = 1
        
        # Normalize to area = 1
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Grid search with refinement
    best_x = [0.2, 0.2]
    best_val = float('inf')
    
    # Coarse grid to start
    grid_points = 50
    a_vals = np.linspace(0.01, 0.99, grid_points)
    d_vals = np.linspace(0.01, 0.99, grid_points)
    
    for a in a_vals:
        for d in d_vals:
            if a + d + 1 > 2:
                continue
            val = objective([a, d])
            if val < best_val:
                best_val = val
                best_x = [a, d]
    
    # Fine optimization using Nelder-Mead
    try:
        result = minimize(objective, best_x, method='Nelder-Mead', options={'maxiter': 500})
        if result.success:
            final_a, final_d = result.x
            h = np.zeros(n_points)
            # First interval
            start1 = int(final_a / dx)
            end1 = int((final_a + 0.5) / dx)
            if end1 > n_points:
                end1 = n_points
            if start1 < n_points:
                h[start1:end1] = 1
            
            # Second interval
            start2 = int((final_a + 0.5 + final_d) / dx)
            end2 = int((final_a + 0.5 + final_d + 0.5) / dx)
            if end2 > n_points:
                end2 = n_points
            if start2 < n_points:
                h[start2:end2] = 1
            
            # Normalize to area = 1
            h = h * (n_points / 2) / np.sum(h)
            final_c5 = evaluate_c5(h, dx)
            if final_c5 < best_c5:
                best_h = h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
