def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Flatten for convex programming
    h_flat = h.copy()
    
    # Sequential convex programming
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current value
        current_c5 = evaluate_c5(h_flat, dx)
        if current_c5 < best_c5:
            best_h = h_flat.copy()
            best_c5 = current_c5
            
        if current_c5 <= 0.3807:
            break
            
        # Compute approximate gradient and Hessian

        for i in range(0, n_points, 3):  # sparse computation

            h_minus[i] -= eps

            h_plus = np.clip(h_plus, 0, 1)

            c_minus = evaluate_c5(h_minus, dx)
            gradient[i] = (c_plus - c_minus) / (2.0 * eps)
        
        # Eliminate noise and direction infeasible updates
        # Add some smoothing
        smoothed_grad = np.convolve(gradient, np.ones(5)/5, mode='same')
        update_direction = -smoothed_grad

        # We'll use a line search approach
        alpha = 0.1

    # Final evaluation

    return (best_h.tolist(), final_c5, n_points)
