import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential convex programming
    max_iter = 1000
    alpha = 0.1
    tolerance = 1e-6
    last_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian approximations
        # This uses finite differences since we don't have explicit formulas
        eps = 1e-5
        grad = np.zeros(n_points)
        h_backup = h.copy()
        
        for i in range(n_points):
            # Forward difference approximation
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1: h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            if h_minus[i] < 0: h_minus[i] = 0
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c_plus = evaluate_c5(h_plus, dx)
            c_minus = evaluate_c5(h_minus, dx)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Quadratic approximation: min (grad^T * delta + 0.5 * delta^T * H * delta)
        # Hessian is approximated by identity for simplicity (diagonal approximation)
        hessian = np.eye(n_points) * 1e-4  # Regularization
        hessian += np.diag(np.full(n_points, 1e-4))  # Small diagonal
        
        # Add small regularization to prevent ill-conditioning
        reg = 1e-6 * np.eye(n_points)
        hessian += reg
        
        # Try to solve the linear system to get delta
        try:
            delta = np.linalg.solve(hessian, -grad)
        except np.linalg.LinAlgError:
            # fallback to pseudo-inverse if singular
            delta = np.linalg.pinv(hessian) @ (-grad)
        
        # Update h with step
        new_h = h + alpha * delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept or reject step
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            best_h = new_h.copy()
            last_improvement = iter_num
            
            # Decrease step size
            alpha = max(0.01, alpha * 0.95)
        else:
            # If no improvement, reduce step size
            alpha *= 0.5
            
            # Reset if stuck for a while
            if iter_num - last_improvement > 50:
                h = best_h.copy()
                alpha = 0.1
                
        # Early exit if goal achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
