import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise constant with some noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Compute all correlations
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting plane method
    constraints_list = []
    epsilon = 0.001
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration > 0 and iteration % 10 == 0:
            epsilon *= 0.5
            
        # Compute current overlaps and find worst case
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)
        
        # Create the constraint: sum(h[i] * (1 - h[i + k])) <= C - epsilon
        # For shift k, we compute the correlation at that shift
        shifted_h = np.roll(h, max_k)
        if max_k >= 0:
            # h[i] * (1 - h[i + k]) for i = 0..n_points-k-1
            constraint_expr = np.sum(h[:n_points-max_k] * (1 - shifted[max_k:]))
        else:
            # h[i] * (1 - h[i + k]) for i = 0..n_points+k-1 (note: k is negative)
            constraint_expr = np.sum(h[-max_k:] * (1 - shifted[:n_points+max_k]))
        
        # Create constraint matrix and vector
        constraint_matrix = np.zeros((n_points, n_points))
        for i in range(n_points):
            if max_k >= 0:
                if i < n_points - max_k:
                    constraint_matrix[i, i + max_k] = 1
            else:
                if i + max_k >= 0:
                    constraint_matrix[i, i + max_k] = 1
        
        # For simplicity, we'll optimize without building a constraint matrix
        # and just solve with all variables and additional linear constraints
        # instead, we build a convex optimization model directly
        # Implemented properly in a way that avoids arrays big than needed
        
        # Start new convex optimization
        h_var = cp.Variable(n_points)
        
        # Define the constraint to minimize max correlation: slight hack to model the problem
        # Since objective is not linear in original formulation  
        # But we're approximating the minimum of a max overlap by perturbing the approach
        
        # Our objective is equivalent to minimizing max_k int h(x)(1-h(x+k)) dx
        # We repeatedly add linear constraints to tighten the problem
        objective = cp.Minimize(cp.sum(h_var))  # dummy for now
        
        constraints = [cp.sum(h_var) == n_points / 2,
                       h_var >= 0,
                       h_var <= 1]
        
        # Add new constraint based on last computed shift
        expr = 0
        for i in range(n_points):
            j = i + max_k
            if 0 <= j < n_points:
                # Sum over valid pairs for this shift
                expr += h_var[i] * (1 - h_var[j])
                
        constraints.append(expr <= compute_c5(h) - epsilon)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                h = h_var.value
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient step on current function
            # Just adjust by a gradient step on one of the correlations
            shift = max_k
            if shift >= 0:
                diff = h[:-shift] - h[shift:]
                grad = 1 - 2*h[:-shift]
            else:
                diff = h[-shift:] - h[:shift]
                grad = 1 - 2*h[-shift:]
                
            # Take gradient step in direction that reduces overlap
            step_size = 0.01
            # But we also need to maintain constraint sum
            adjustment = step_size * grad
            
            # Project ontofeasible set
            h = h - adjustment
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
