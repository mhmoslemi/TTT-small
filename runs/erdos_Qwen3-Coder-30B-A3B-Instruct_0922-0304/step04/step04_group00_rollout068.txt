from scipy.signal import correlate
import cvxpy as cp
import time

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting-plane algorithm
    constraints_list = []
    epsilon = 0.001
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Save best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Constraint: h · (1 - h_k) <= max_overlap - epsilon
        # Use the full correlation array to compute h · (1 - h_k)

        # where a_i = (1 - h[i - k]) for positive k, etc.
        
        # efficiently extract the shifted version
        h_shifted = np.roll(1 - h, -max_k + n_points - 1)
        # The overlap is sum(h_i * (1 - h_{i+k}))
        # which equals sum(h_i * h_shifted_i) -- this may need more care
        
        # More accurately we want to extract the indices where convolution is computed for k
        # For mode='full': result length = 2*n - 1
        #     shift k = 0..2n-2 in full mode  
        # k = max_k - (n_points - 1) due to indexing in full conv
        
        # Let's recompute properly
        shift = max_k - (n_points - 1)
        h_shifted = np.roll(h, -shift)

        # which is same as sum(h[i] * 1) - sum(h[i] * h_shifted[i]) = sum(h) - sum(h * h_shifted)
        
        # But better to be more careful with alignment and indices
        # Full convolution: for k in -n+1 ... n-1 where convolution index is k + n-1
        # We'll just choose to make this linear constraint:
        # sum(h[i] * (1 - h_shifted[i]) for i) <= max_overlap - epsilon
        
        # k = max_k, shift=k-n+1 (left shift by abs(k-n+1))
        shift = max_k - (n_points - 1)
        shifted = np.roll(1 - h, -shift)
        # constraint: sum(h * shifted) <= max_overlap - epsilon
        constraint_coeffs = shifted
        constraint_rhs = max_overlap - epsilon
        constraints_list.append((constraint_coeffs, constraint_rhs))
        
        # Convex optimization to solve the constrained minimization
        # Minimize max overlap, subject to constraints
        h_var = cp.Variable(n_points)

        # but rather solve a sequence of problems where we iteratively add constraints.

        # But CVXPY needs all constraints to be linear in the variable being optimized.
        # So we will use a heuristic to solve:
        # solve h such that  h_ij >= 0, sum(h) = n_points/2, 
        # and h·(1 - h_k) <= C - epsilon for each k added to constraints.

        h_vars = cp.Variable(n_points)

        # Let’s just solve:
        # Minimize sum of some function subject to:
        # - h >= 0
        # - h <= 1
        # - sum(h) = n_points / 2
        # - for each constraint defined by pair (a_k, b_k): sum(a_k * h) <= b_k
        
        # Max overlap retention: we'll optimize the function directly in the loop
        # So we start with a relaxed problem: minimize h*0 + sum(h) + sum(1-h) subject to constraints
        
        # Maybe we should reconsider and solve a proper sequence:
        # At each iteration we minimize some strictly increasing function of max overlap.
        # Just use SLSQP based approach with lifting update constraints...
        
        # Re-do UX: Use Cvxpy for the optimization step
        # Vars and Data
        h_v = cp.Variable(n_points)

        for a, b in constraints_list:
            constraints.append(cp.sum(cp.multiply(h_v, a)) <= b)
            
        # Objective: might use a proxy for this
        # Directly optimize via hand-coded hand-simplified approach:
        # Try to improve by solving a constrained optimization problem
                
        # Rely on current h and minimize the _existing_ constraints' violations on current h
        # Use linear approximation: add constraint at current state

            h_var = cp.Variable(n_points)
            obj = cp.Minimize(1e5 * cp.sum(cp.multiply(h_var, 1-h_var)))  # hack to approximate overlap minimization

            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]

                constraints.append(cp.sum(cp.multiply(h_var, a)) <= b)

            if prob.status == cp.OPTIMAL:
                h = h_var.value.copy()
                h = np.clip(h, 0, 1)
                # Re-normalize to sum = n_points/2
                h = h * (n_points / 2) / np.sum(h)
            else:
                # fallback to just greedy
                h = h * (n_points / 2) / np.sum(h)
                
        except Exception:
            #兜底方案
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally update epsilon to allow convergence, and restart if improvement stalls
        if iter_count % 10 == 0:
            epsilon *= 0.5
        if iter_count % 50 == 0:
            if best_c5 <= 0.3807:
                break

    # Final re-eval
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
