import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise initialize a symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        # Place two intervals of length 0.5 (i.e. 35 points) symmetrically
        left_start = n_points // 4
        right_start = 3 * n_points // 4
        h[left_start:left_start + n_points//4] = 1.0
        h[right_start:right_start + n_points//4] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    def objective(params):
        # params = [a, d] where a is position of left interval, d is distance between intervals
        a, d = params
        # Normalize a to valid range [0, 1-d] where d is between 0 and 1
        if a < 0 or a > 1 - d:
            return float('inf')
        if d < 0 or d > 1:
            return float('inf')
        
        # Create step function
        h_test = np.zeros(n_points)
        w = 0.5  # width of each interval (area = 1)
        left_start = int(a * n_points)
        right_start = int((a + w + d) * n_points)
        left_end = left_start + int(w * n_points)
        right_end = right_start + int(w * n_points)
        
        h_test[left_start:left_end] = 1.0
        h_test[right_start:right_end] = 1.0
        
        # Normalize to area = 1
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Calculate C5
        c5 = evaluate_c5(h_test, dx)
        return c5
    
    # Try different optimization approaches
    
    # 1. Grid search for rough exploration
    best_params = [0.25, 0.25]
    best_val = objective(best_params)
    
    # Try grid search
    fine_grid = np.linspace(0.05, 0.95, 20)
    for a in fine_grid:
        for d in fine_grid:
            if a + 0.5 + d <= 1.0:  # ensure valid
                val = objective([a, d])
                if val < best_val:
                    best_val = val
                    best_params = [a, d]
    
    # 2. Use scipy optimizer
    try:
        # Optimize from best grid point
        result = minimize(objective, best_params, method='Nelder-Mead', 
                         options={'maxiter': 500, 'xatol': 1e-5, 'fatol': 1e-6})
        if result.success:
            final_val = objective(result.x)
            if final_val < best_val:
                best_val = final_val
                best_params = result.x
    except:
        pass

    # Create final h from best parameters
    a, d = best_params
    h_final = np.zeros(n_points)
    w = 0.5
    left_start = int(a * n_points)
    right_start = int((a + w + d) * n_points)
    left_end = left_start + int(w * n_points)
    right_end = right_start + int(w * n_points)
    
    h_final[left_start:left_end] = 1.0
    h_final[right_start:right_end] = 1.0
    
    # Normalize
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
