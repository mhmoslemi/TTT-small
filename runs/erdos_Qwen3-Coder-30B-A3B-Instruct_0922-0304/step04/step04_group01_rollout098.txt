import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try different optimization approaches
    for iteration in range(1000):  # Roughly 1000 iterations
        if time.time() - start_time > budget_s - 1:
            break
            
        # Alternative: Simulated Annealing inspired approach
        h_candidate = h.copy()
        
        # Small random perturbation
        perturbation = np.random.normal(0, 0.01, n_points)
        h_candidate += perturbation
        
        # Clip and normalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate this candidate
        candidate_c5 = evaluate_c5(h_candidate, dx)
        
        # Accept if better or with some probability
        if candidate_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = candidate_c5
            best_h = h.copy()
        else:
            # Sometimes accept worse solutions to escape local optima
            if np.random.random() < 0.01:
                h = h_candidate.copy()
        
        # Occasionally try more aggressive shifts
        if iteration % 30 == 0:
            # Try to shoot for better configuration with pattern
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
