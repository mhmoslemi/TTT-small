import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 2000
    mutation_rate = 0.1
    elite_size = population_size // 4
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
    else:
        # Initialize with triangular function
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
    
    # Generate initial population
    population.append(h0.copy())
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # ES parameters
    sigma = 0.1
    adaptation_rate = 0.1
    success_count = 0
    success_threshold = population_size // 2
    
    # Main ES loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection: keep top 25%
        elite = population[:elite_size]
        
        # Create offspring through mutation
        offspring = []
        for h in elite:
            # Adaptive Gaussian mutation
            noise = np.random.normal(0, sigma, n_points)
            mutated_h = h + noise
            mutated_h = project_feasible(mutated_h, n_points)
            offspring.append(mutated_h)
            
            # Add some random exploration
            if np.random.rand() < 0.3:
                random_h = np.random.rand(n_points)
                random_h = project_feasible(random_h, n_points)
                offspring.append(random_h)
        
        # Fill up population
        while len(offspring) < population_size:
            # Generate random feasible solution
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            offspring.append(h)
            
        population = offspring[:population_size]
        
        # Adapt sigma based on success
        if iter_num % 10 == 0:
            success_count = 0
            for i in range(elite_size):
                if fitness[i] < fitness[elite_size]:
                    success_count += 1
            
            if success_count > success_threshold:
                sigma *= 1.1
            else:
                sigma /= 1.1
                
            # Clip sigma to reasonable range
            sigma = np.clip(sigma, 0.001, 0.5)
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
