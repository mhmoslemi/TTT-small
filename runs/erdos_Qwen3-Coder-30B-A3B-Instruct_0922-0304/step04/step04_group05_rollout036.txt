import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum, dx):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to match sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # GD with line search
    gamma = 0.1
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation (linearized)
        # C_k = \int h(x)(1 - h(x+k)) dx  
        # Gradient wrt h[i] = dx * (1 - 2*h[i] - average of h[i+k])
        # But since we are looking for max, we want -grad
        grad = np.zeros(n_points)
        
        # Compute h[i] * (1 - h[i+k]) for all shifts k
        # This op is couple with the overlap computation
        # We compute all overlaps for maximum
        corr = correlate(h, 1-h, mode="full")
        c5_value = np.max(corr) * dx
        # Find which shift gives max overlap, approximate gradient
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)  # convert full mode to actual shift
        
        # Absolute shift for overlap computation
        abs_shift = max_k if max_k >= 0 else -max_k
        shifted_indices = np.arange(n_points) + max_k
        valid_indices = (shifted_indices >= 0) & (shifted_indices < n_points)
        
        # Gradient estimate
        # We compute gradient for one primary shift k* where max overlap occurs
        # F(h) = \int h(x)(1 - h(x+k*)) dx = \sum h[i] * (1 - h[i+k*])
        # So dF/dh[i] = (1 - h[i+k*] - h[i]) 
        # restricted to valid k* shifts only
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + max_k >= 0 and i + max_k < n_points:
                grad[i] = 1 - h[i + max_k] - h[i]
            elif i + max_k < 0 and max_k < 0:
                shift = i + max_k
                grad[i] = 1 - h[shift] - h[i]
            elif i + max_k >= n_points and max_k > 0:
                shift = i + max_k - n_points
                grad[i] = 1 - h[shift] - h[i]
            else:
                grad[i] = 1 - h[i] - h[i]  # fallback (some issue with boundary)
                
        # Avoid reliance on numerical errors for gradient
        grad = -dx * grad  # negative for minimization
        
        # Frank-Wolfe line search or Armijo backtracking
        step_size = gamma
        while step_size > 1e-10:
            h_new = h - step_size * grad
            h_new = project_to_feasible(h_new, target_sum, dx)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h.copy()
                stagnation_count = 0
                break
            else:
                step_size *= 0.5
                stagnation_count += 1
                
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        if stagnation_count > max_stagnation:
            gamma *= 0.5
            stagnation_count = 0
            
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
