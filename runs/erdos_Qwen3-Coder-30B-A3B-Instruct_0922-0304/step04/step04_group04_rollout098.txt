import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    else:
        # Fallback initialization
        h_current = np.ones(n_points)
        h_current[n_points//2:] = 0
        h_current += np.random.normal(0, 0.05, n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Use a simple local optimization approach: gradient descent with projection
    max_iter = 10000
    learning_rate = 0.01
    tolerance = 1e-6
    
    best_h = h_current.copy()
    best_score = evaluate_c5(best_h, dx)
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h_current.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            score_plus = evaluate_c5(h_plus, dx)
            score_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (score_plus - score_minus) / (2 * eps)
        
        # Update step
        h_new = h_current - learning_rate * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        new_score = evaluate_c5(h_new, dx)
        
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
            
        if abs(new_score - best_score) < tolerance:
            break
            
        h_current = h_new
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_score = evaluate_c5(final_h, dx)
    
    # Ensure we return a valid result within time budget
    if final_score <= 0.3807:
        return (final_h.tolist(), final_score, n_points)
    
    return (final_h.tolist(), final_score, n_points)
