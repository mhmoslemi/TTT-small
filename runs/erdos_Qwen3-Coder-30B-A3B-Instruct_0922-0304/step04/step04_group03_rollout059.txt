import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraint(h, dx, target_sum):
    """Project h to satisfy sum(h) = target_sum"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    return h * target_sum / current_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with uniform distribution
        h = np.ones(n_points) * 0.5
    
    h = project_to_constraint(h, dx, target_sum)
    
    # Gradient descent parameters
    lr = 0.01
    max_iter = 10000
    tol = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Initialize gradient tracking
    prev_c5 = best_c5 + 1
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k] * dx
        
        # Compute gradient for worst-case shift
        # For shift k, O_k = sum(h_i * (1 - h_{i+k}))
        # Gradient w.r.t. h_i: dO_k/dh_i = 1 - h_{i+k} - h_{i-k}  
        # But we must be careful about indices
        
        # Create a larger array to avoid modular arithmetic
        h_extended = np.concatenate([h, h])
        gradient = np.zeros_like(h)
        
        # For each position i, compute contribution to gradient
        for j in range(n_points):
            k = max_k - n_points + j  # offset for the maximal overlap position
            if k >= 0:
                gradient[j] = 1 - h_extended[j + k] - h_extended[j - k]
            else:
                gradient[j] = 1 - h_extended[j - k] - h_extended[j - k]
                
        # Update step
        h_new = h - lr * gradient
        
        # Clip to [0,1] and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraint(h_new, dx, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        # Check for convergence
        if abs(prev_c5 - new_c5) < tol:
            break
        prev_c5 = new_c5
        h = h_new
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Ensure final h is valid
    h_final = np.clip(best_h, 0, 1)
    h_final = project_to_constraint(h_final, dx, target_sum)
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
