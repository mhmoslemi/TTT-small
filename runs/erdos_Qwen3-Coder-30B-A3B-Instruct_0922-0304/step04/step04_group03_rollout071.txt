```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Try different interval configurations
    # We'll use a grid search on the distance between intervals
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Interval width
        w = 0.5
        # First interval is [0, w]
        i0 = 0
        i1 = int(np.floor(w / dx))
        
        # Second interval is [w+d, w+d+w] = [w+d, w+d+w]
        i2 = int(np.ceil((w + d) / dx))
        i3 = int(np.floor((w + d + w) / dx))
        
        # Ensure intervals don't overlap
        if i2 <= i1:
            continue
            
        # Create step function
        h = np.zeros(n_points)
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        
        # Normalize to satisfy the constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # For good measure, do some local search around the best found
    if best_h is not None:
        h = best_h.copy()
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx)
        
        # Try small perturbations
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            # Small random mutation
            idx = np.random.randint(0, n_points)
            h_pert = h.copy()
            h_pert[idx] += np.random.normal(0, 0.01)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            c5_pert = evaluate_c5(h_pert, dx)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
```
