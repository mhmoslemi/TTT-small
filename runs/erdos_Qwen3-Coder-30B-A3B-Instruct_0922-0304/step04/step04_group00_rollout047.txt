import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_constraint(h, target_sum):
    """Evaluate constraint violation"""
    return abs(np.sum(h) - target_sum)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

def mutate_h(h, mutation_rate=0.1, noise_std=0.01):
    """Apply mutation to h"""
    mask = np.random.random(len(h)) < mutation_rate
    h = h.copy()
    h[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return normalize_h(h, len(h) / 2)

def crossover_h(h1, h2):
    """Perform single-point crossover between two h vectors"""
    point = random.randint(1, len(h1) - 1)
    child1 = np.concatenate([h1[:point], h2[point:]])
    child2 = np.concatenate([h2[:point], h1[point:]])
    return normalize_h(child1, len(h1) / 2), normalize_h(child2, len(h1) / 2)

def tournament_selection(population, fitnesses, k=5):
    """Perform tournament selection"""
    selected = []
    for _ in range(len(population) // 2):
        tournament_indices = np.random.choice(len(population), k, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

def local_search(h, dx, max_iter=50):
    """Perform local search using SLSQP"""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - len(h_vec) / 2
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    
    if result.success:
        return np.clip(result.x, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        
    # Create initial population
    for _ in range(population_size):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Begin GA loop
    max_generations = 1000
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected_parents = tournament_selection(population, fitnesses, k=5)
        
        # Crossover and mutation
        new_population = []
        
        # Keep top 20% (elitism)
        elite_indices = np.argsort(fitnesses)[:population_size // 5]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            if len(selected_parents) < 2:
                # If not enough selected, re-randomize
                p1 = np.random.choice(population)
                p2 = np.random.choice(population)
            else:
                p1, p2 = np.random.choice(selected_parents, 2, replace=False)
            
            c1, c2 = crossover_h(p1, p2)
            c1 = mutate_h(c1)
            c2 = mutate_h(c2)
            
            new_population.extend([c1, c2])
            
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Local search on top performers
        sorted_indices = np.argsort(fitnesses)[:10]
        for idx in sorted_indices:
            population[idx] = local_search(population[idx], dx, max_iter=50)
    
    # Final evaluation
    final_evaluation = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_evaluation, n_points)
