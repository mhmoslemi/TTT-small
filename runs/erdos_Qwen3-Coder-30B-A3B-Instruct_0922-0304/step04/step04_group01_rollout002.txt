import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 75
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to triangular construction
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, np.linspace(1, 0, n_points - n_points//2)])
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian optimization parameters
    lambda_lag = 0.0
    mu = 1.0
    nu = 1.0
    alpha = 0.002
    beta = 0.99
    max_iter = 2000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate current overlap and gradient
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Approximate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update h using gradient descent with adaptive step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce constraint sum(h) = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = h_new.copy()
        
        # Update Lagrange multipliers
        constraint_violation = np.sum(h) - n_points / 2
        lambda_lag += mu * constraint_violation
        
        # Apply adaptive step size
        if iteration % 10 == 0:
            alpha *= beta
            if alpha < 1e-6:
                alpha = 1e-6
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Occasional perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
