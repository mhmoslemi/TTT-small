import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points, dx):
    """Project h to satisfy the constraint sum(h) * dx = 1"""
    h_sum = np.sum(h)
    if abs(h_sum * dx - 1.0) > 1e-10:
        h = h * (1.0 / h_sum) * (n_points / 2)
    return np.clip(h, 0, 1)

def compute_gradients(h, k_max, n_points, dx):
    """Compute gradient of overlap at k_max"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For overlap at shift k_max, dO_k_max/dh_i = (1 - h[(i+k_max)%n]) - h[(i-k_max)%n]
        grad[i] = (1 - h[(i + k_max) % n_points]) - h[(i - k_max) % n_points]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 30000
    
    # Initialize with a better starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a more sophisticated shape
        h = np.zeros(n_points)
        # Create a piecewise linear function with smooth transitions
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h[i] = 2 * x
            elif x < 1.5:
                h[i] = 1
            else:
                h[i] = 2 - 2 * x
        # Add a small Gaussian bump in the center
        center = n_points // 2
        sigma = n_points / 10
        for i in range(n_points):
            h[i] += 0.1 * np.exp(-0.5 * ((i - center) / sigma) ** 2)
        
    h = np.clip(h, 0, 1)
    h = project_h(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization with adaptive step size
    step_size = 0.01
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        all_overlaps = overlap * dx
        
        # Find maximum overlap and its shift
        k_max = np.argmax(all_overlaps)
        max_overlap = all_overlaps[k_max]
        
        # Check convergence
        if max_overlap <= 0.3807:
            break
            
        # Compute gradient at maximum overlap
        grad = compute_gradients(h, k_max, n_points, dx)
        
        # Update h
        h = h - step_size * grad
        
        # Project to feasible region
        h = project_h(h, n_points, dx)
        
        # Check for new best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive step size
        if iter_num % 100 == 0 and iter_num > 0:
            step_size *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
