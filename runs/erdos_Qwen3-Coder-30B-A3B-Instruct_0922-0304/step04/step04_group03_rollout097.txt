import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
    else:
        # Create a simple triangular shape as base
        h_base = np.zeros(n_points)
        for i in range(n_points):
            h_base[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h_base = np.clip(h_base, 0, 1)
    
    # Normalize to satisfy constraint
    h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    # Grid search on optimized piecewise structure
    best_h = h_base.copy()
    best_c5 = evaluate_c5(h_base, dx)
    
    # Try symmetric piecewise constant functions
    # For an optimal solution, we expect two intervals of width w and distance d
    # Total width of 1, so w = 0.5, d in [0, 1]
    d_values = np.linspace(0.0, 1.0, 101)  # step size of 0.01
    
    for d in d_values:
        # Create piecewise constant function with two intervals
        h = np.zeros(n_points)
        
        # Interval 1: [0, 0.5]
        i0 = 0
        i1 = int(0.5 / dx)  # inclusive
        
        # Interval 2: [0.5 + d, 1 + d]
        i2 = int((0.5 + d) / dx)  # inclusive start
        i3 = int((1.0 + d) / dx)  # inclusive end
        
        # Ensure indices are valid
        i1 = min(i1, n_points - 1)
        i2 = max(i2, 0)
        i3 = min(i3, n_points - 1)
        
        # Check that intervals don't overlap
        if i2 > i1:
            h[i0:i1+1] = 1
            h[i2:i3+1] = 1
            
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate
            c5 = evaluate_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final optimization with local search around the best found
    h_opt = best_h.copy()
    c5_opt = best_c5
    
    # Small perturbation search in vicinity of d
    d_opt = 0.5  # rough estimate from the search
    d_range = np.linspace(max(0.0, d_opt - 0.1), min(1.0, d_opt + 0.1), 21)
    
    for d in d_range:
        # Create piecewise constant function with two intervals
        h = np.zeros(n_points)
        
        # Interval 1: [0, 0.5]
        i0 = 0
        i1 = int(0.5 / dx)  # inclusive
        
        # Interval 2: [0.5 + d, 1 + d]
        i2 = int((0.5 + d) / dx)  # inclusive start
        i3 = int((1.0 + d) / dx)  # inclusive end
        
        # Ensure indices are valid
        i1 = min(i1, n_points - 1)
        i2 = max(i2, 0)
        i3 = min(i3, n_points - 1)
        
        # Check that intervals don't overlap
        if i2 > i1:
            h[i0:i1+1] = 1
            h[i2:i3+1] = 1
            
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate
            c5 = evaluate_c5(h, dx)
            
            if c5 < c5_opt:
                c5_opt = c5
                h_opt = h.copy()
                
        if time.time() - start_time > budget_s - 1:
            break
    
    return (h_opt.tolist(), c5_opt, n_points)
