import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current state
        c5 = evaluate_c5(h, dx)
        
        # Find the most constraining shift
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Compute gradient: d/dh[i] of overlap at k_max
        # O_k = sum h[i] * (1 - h[(i+k) % n_points])
        # dO_k/dh[i] = (1 - h[(i+k) % n_points]) - h[(i-k) % n_points]
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = 1 - h[j]
            j = (i - k_max) % n_points
            gradient[i] -= h[j]
        gradient *= dx
        
        # Update h using gradient descent
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Check improvement
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 50  # reset patience
        else:
            patience -= 1
            if patience <= 0:
                # Reduce learning rate if no improvement
                learning_rate *= 0.9
                patience = 50
                if learning_rate < 1e-6:
                    break
    
    return (best_h.tolist(), best_c5, n_points)
