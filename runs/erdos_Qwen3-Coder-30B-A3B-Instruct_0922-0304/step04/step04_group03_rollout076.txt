import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    max_overlap = 0
    # Compute overlap for all shifts
    for k in range(2*n - 1):
        overlap = 0
        for i in range(n):
            j = i + k - (n - 1)
            if 0 <= j < n:
                overlap += h[i] * (1 - h[j]) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
    else:
        # Create a more informed initial guess
        h_init = np.zeros(n_points)
        # Place two intervals
        w = 0.5  # width of each interval
        d = 0.0  # distance between intervals
        a = 0.25  # start position
        for i in range(n_points):
            x = i * dx
            if (a <= x < a + w) or (a + w + d <= x < a + w + d + w):
                h_init[i] = 1.0
        # Normalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    # Optimize using scipy minimize
    def objective(x):
        # x = [a, d] where a is first interval start, d is distance between intervals
        a, d = x
        if a < 0 or a > 1 - d - 0.5 or d < 0 or d > 1 - 1.0:  # bounds check
            return 1e10
        h = np.zeros(n_points)
        # Create h based on a and d
        for i in range(n_points):
            x_pos = i * dx
            if (a <= x_pos < a + 0.5) or (a + 0.5 + d <= x_pos < a + 0.5 + d + 0.5):
                h[i] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Initial guess
    x0 = [0.25, 0.0]  # a = 0.25, d = 0.0
    bounds = [(0, 0.5), (0, 0.5)]
    
    # Use L-BFGS-B method with bounds
    try:
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 1000}, callback=lambda x: None)
        if res.success:
            a_opt, d_opt = res.x
            h_opt = np.zeros(n_points)
            for i in range(n_points):
                x_pos = i * dx
                if (a_opt <= x_pos < a_opt + 0.5) or (a_opt + 0.5 + d_opt <= x_pos < a_opt + 0.5 + d_opt + 0.5):
                    h_opt[i] = 1.0
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_bound = evaluate_c5(h_opt, dx)
        else:
            # Fall back to original h_init
            c5_bound = evaluate_c5(h_init, dx)
            h_opt = h_init
    except Exception:
        # Fall back to original h_init
        c5_bound = evaluate_c5(h_init, dx)
        h_opt = h_init
    
    # Use simulated annealing approach to fine-tune
    max_iter = 1000 + int(500 * (budget_s - (time.time() - start_time)) / budget_s)
    current_h = h_opt.copy()
    current_c5 = c5_bound
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Perturbations
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Make small change to h
        h_new = current_h.copy()
        idx = np.random.randint(0, n_points)
        # Small perturbation
        perturbation = np.random.normal(0, 0.01)
        h_new[idx] += perturbation
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        # Re-normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n_points)
