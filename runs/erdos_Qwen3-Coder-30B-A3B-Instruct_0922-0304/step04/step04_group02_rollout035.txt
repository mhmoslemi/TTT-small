import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ES parameters
    pop_size = 100
    generations = 1000
    elite_size = int(0.2 * pop_size)
    mutation_rate = 0.3
    initial_sigma = 0.1
    sigma = initial_sigma
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        noise = np.random.normal(0, sigma, n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # ES iterations
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness = []
        for indiv in population:
            fit = evaluate_c5(indiv, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep elite
        selected = population[:elite_size]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elite_size):
            # Select parent
            parent = selected[np.random.randint(0, elite_size)]
            
            # Mutate
            mutated = parent.copy()
            indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
            for i in indices:
                mutated[i] += np.random.normal(0, sigma)
            
            # Repair constraints
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * (n_points / 2) / np.sum(mutated)
            
            offspring.append(mutated)
        
        # Create new population
        population = selected + offspring
        
        # Adapt mutation strength
        if gen % 10 == 0:
            if gen > 0:
                improvement = np.mean(fitness[:elite_size]) / np.mean(fitness[elite_size:])
                if improvement > 1.05:
                    sigma = min(sigma * 1.1, 0.5)
                else:
                    sigma = max(sigma * 0.9, 0.01)
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.98:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
