import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp_init = 5.0
    temp_min = 1e-6
    cooling_rate = 0.995
    iter_per_temp = 200
    
    # Initial temperature
    temp = temp_init
    
    # Simulated Annealing loop
    while temp > temp_min and time.time() - start_time < budget_s * 0.9:
        # Generate new candidate through mutation
        h_new = h.copy()
        
        # Randomly select indices to modify
        n_mutate = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, n_mutate, replace=False)
        
        # Apply Gaussian noise
        noise = np.random.normal(0, 0.05, n_mutate)
        h_new[indices] += noise
        
        # Clip values to [0,1]
        h_new = np.clip(h_new, 0, 1)

        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)

        delta = c5_new - evaluate_c5(h, dx)

        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]

            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}

            return evaluate_c5(h, dx)
