import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    max_overlap = 0
    for k in range(2*n - 1):
        overlap = 0
        for i in range(n):
            j = i + k - (n - 1)
            if 0 <= j < n:
                overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Initialize with a simple construction
        h = np.zeros(n_points)
        # Create a piecewise constant function
        mid = n_points // 2
        h[mid-5:mid+5] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Define objective function for optimization
    def objective(params):
        a, d = params
        # Ensure valid bounds
        if a < 0 or a > 1 - d - 0.5 or d < 0 or d > 1:
            return float('inf')
        
        # Create the step function
        h_temp = np.zeros(n_points)
        w = 0.5  # Each interval width
        
        # First interval
        i_start1 = int(a / dx)
        i_end1 = int((a + w) / dx)
        if i_start1 < n_points:
            h_temp[i_start1:i_end1] = 1.0
        
        # Second interval
        i_start2 = int((a + w + d) / dx)
        i_end2 = int((a + w + d + w) / dx)
        if i_start2 < n_points:
            h_temp[i_start2:i_end2] = 1.0
        
        # Normalize to constraint
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        return evaluate_c5(h_temp, dx)
    
    # Grid search in promising regions
    best_params = None
    best_val = float('inf')
    
    # Try various configurations
    a_range = np.linspace(0, 0.5, 20)
    d_range = np.linspace(0, 0.5, 20)
    
    for a in a_range:
        for d in d_range:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                val = objective([a, d])
                if val < best_val:
                    best_val = val
                    best_params = [a, d]
            except Exception:
                continue
    
    # If we found a good candidate, refine with optimization
    if best_params is not None:
        result = minimize(objective, best_params, method='Nelder-Mead', 
                          options={'maxiter': 100, 'xatol': 1e-5, 'fatol': 1e-5})
        if result.success:
            a, d = result.x
            # Recompute with final parameters
            h_final = np.zeros(n_points)
            w = 0.5
            
            i_start1 = int(a / dx)
            i_end1 = int((a + w) / dx)
            if i_start1 < n_points:
                h_final[i_start1:i_end1] = 1.0
            
            i_start2 = int((a + w + d) / dx)
            i_end2 = int((a + w + d + w) / dx)
            if i_start2 < n_points:
                h_final[i_start2:i_end2] = 1.0
                
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_c5(h_final, dx)
            
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    
    # Final validation and return
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
