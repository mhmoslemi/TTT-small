import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer with hybrid approach: variable learning rate and gradient estimation
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Adaptive gradient estimation
        step = 0.005 * (1 - iteration / max_iter) + 0.001
        h_temp = h.copy()
        
        # Compute current gradient using finite differences
        corr = correlate(h_temp, 1 - h_temp, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx

        grad = np.zeros_like(h_temp)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_temp.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        # Gradient descent step
        h_temp -= step * grad
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        # Update h
        h = h_temp

        # Compute new value
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Occasionally perturb to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
