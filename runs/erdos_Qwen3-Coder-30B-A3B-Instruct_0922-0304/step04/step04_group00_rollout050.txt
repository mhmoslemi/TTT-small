import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute the maximum overlap integral for given h."""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles around initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    v_max = 0.01  # Maximum velocity
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.1 + h_init
    velocities = np.random.rand(n_particles, n_points) * 0.01 - 0.005
    
    # Normalize particles to satisfy constraint
    for i in range(n_particles):
        particles[i] *= (n_points / 2) / np.sum(particles[i])
        
    # Track personal and global bests
    personal_best = particles.copy()
    personal_best_scores = np.array([compute_c5(p) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best[i] - particles[i]) + 
                             c2 * r2 * (global_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Compute new score
            new_score = compute_c5(particles[i])
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_best[i] = particles[i].copy()
                
                # Update global best if needed
                if new_score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = new_score
                    
                    # Early exit if target achieved
                    if global_best_score <= 0.3807:
                        break
                        
    # Final evaluation of global best
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
