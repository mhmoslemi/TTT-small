import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > 50:
            learning_rate *= 0.9
            patience = 0
            
        # Compute gradient
        # For each shift k, compute d/dh[i] of overlap_k
        # This is approximately: (1 - h[i+k]) - h[i-k] for valid indices
        gradient = np.zeros(n_points)
        
        # Get the shift that maximizes the overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # The actual shifts are from -n_points+1 to n_points-1
        actual_k = max_k - (n_points - 1)
        
        # Compute gradient based on that shift
        for i in range(n_points):
            left_idx = (i - actual_k) % n_points
            right_idx = (i + actual_k) % n_points
            gradient[i] = (1 - h[right_idx]) - h[left_idx]
            
        gradient *= dx
        
        # Update h
        h += -learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    return (best_h.tolist(), best_c5, n_points)
