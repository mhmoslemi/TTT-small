import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraints(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * target_sum / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe style optimization with line search
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # This is the gradient of the overlap term with respect to h_i
        # We compute h * (1 - h) = h - h^2
        # The gradient of overlap at shift k with respect to h_i is the sum of contributions
        # from all j such that (j+k) is valid and h[j] contributes
        # ∇C = dx * (1 - 2*h - shift_average)
        
        # Instead, we compute this more directly
        # For each shift k, we have:
        # C_k = ∫ h(x)(1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx
        # Let's compute the actual overlap integral
        
        # Compute overlap integral contributions for each shift
        # Using convolution: C_k = ∫ h(x)(1 - h(x+k)) dx
        # The actual overlap integral we want is: max_{k} ∫ h(x)(1 - h(x+k)) dx
        
        # More directly, for C_k = ∫ h(x)(1 - h(x+k)) dx
        # Consider: C_k = ∫ h(x)dx - ∫ h(x)h(x+k)dx
        # So dC_k / dh_i = 1 - h_i - terms from sum over shifts
        
        # We compute the full overlap matrix D[i,k] = ∫ h(x) h(x+k+shift) dx 
        # But we want ∫ h(x)(1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x) h(x+k) dx
        # = sum(h) - overlap. We can compute this straightforwardly.
        
        # For gradient of total overlap:
        # ∇ C = ∇ (h (1-h)) = ∇ (h - h^2) = 1 - 2h - ∇(shift)
        
        # The non-convexity makes this non-straightforward
        # Let's try with an easier approach.
        # Most effective is to use Frank-Wolfe methods or projected gradient descent
        # but focus on the most influential shift in the moment
        
        # Compute overlap for all shifts and find the maximum
        current_c5 = evaluate_c5(h, dx)
        
        # Let's try direct approximation using the -∇C = 1 - 2*h - average of 
        # cross terms for shifts (this is approximation)
        # Let F = ∫ h(x)(1 - h(x+k)) dx = sum(h) - ∫h(x)h(x+k) dx = 1 - overlap
        # ∇F_i = - (1 - h_i) - ∑(h_i+k) over all valid k
        
        # For maximum of these F_k functions, we guess that it's a given function F_k0
        # and compute gradient for that k0
        
        # For speed, we'll optimize for the largest shift term regardless of shape
        # Compute the full overlap matrix
        conv = correlate(h, 1 - h, mode="full")
        # Only look at non-zero shifts, that is from -n_points+1 to n_points-1 (i.e., 2*n_points-2 shifts)
        # For each shift k (from -n_points to n_points-1), we compute ∫ h(x)(1-h(x+k)) dx
        max_val = -np.inf
        best_shift = 0
        
        # Just consider the shifts that actually produce non-zero integrals
        # The overlap for shift k is at correlation index n_points-1+k (k negative means before the end)
        # We compute all to know maximum
        for k in range(1 - n_points, n_points):
            idx = n_points - 1 + k  # Maps shift k to idx in full correlation output
            if 0 <= idx < len(conv):
                val = conv[idx] * dx
                if val > max_val:
                    max_val = val
                    best_shift = k
        
        # We now optimize based on this single shift's gradient
        # Gradient at each point i with respect to shift best_shift is:
        # The contribution of h[i](1-h[i+best_shift]) making up the shift integral
        
        # More robust solution: Direct gradient descent with Frank-Wolfe updates
        # Handle: minimize max_k ∫ h(x) (1 - h(x+k)) dx
        # Take gradient for max_k shift integral
        # ~ Frank-Wolfe style: 
        # 1. compute gradient 
        # 2. line search to maintain linearization approximation
        # 3. project solution
        
    # Let's run the Frank-Wolfe directly with some stabilization
    # We will try Frank-Wolfe new step (without conjugate gradients or complex line search)
    # We use a modification where we choose the gradient from the worst shift
    # and update using Frank-Wolfe descent direction, but project after
        
    # Optimization loop: run 500 steps graphically
    max_iter = 500
    step_size = 0.05
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap values
        overlap = correlate(h, 1 - h, mode="full")
        max_val = -np.inf
        max_idx = 0
        for k in range(1 - n_points, n_points):
            idx = n_points - 1 + k
            if 0 <= idx < len(overlap):
                val = overlap[idx] * dx
                if val > max_val:
                    max_val = val
                    max_idx = k
        
        # Compute gradient for the shift that gives max overlap
        # The shift_index = k => index = n_points - 1 + k
        # We want the gradient direction for maximum shift term to minimize it
        # The gradient ∇C_{max_idx} should be computed as:
        # For shift k, we compute ∫ h(x)(1-h(x+k)) dx
        # For a point i, that contributes via h[i](1-h[i+k]) 
        # This is 1 - h[i+k]  at gradient with respect to h[i] 
        
        # Shorthand:
        # If the integral for k is ∫ h(x)(1 - h(x+k)) dx, then gradient wrt h[i]:
        # = (1 - h[i+k]) if i+k valid and h[i+k] exists
        # = -h[i] if we count h[i] * 1 and h[i] * -h[i]
        
        # Let's compute the actual discrete approximation for gradient of 
        # ∫ h(x)(1 - h(x+k)) dx with k = max_idx
        
        # Compute num - 1 cross terms: 
        # What does this mean exactly? Looking at the overlap term:
        # ∫ h(x)(1 - h(x+k)) dx  
        # since h(x)(1-h(x+k)) = h(x) - h(x)h(x+k)
        # so the derivative w.r.t. h[i] is:
        # -1 * (1 if i+k valid) - (1 if i valid and i+k valid) * h(currentposition) = -1*delta[i+k] - h[i+k]
        # unless i has k=0... Let me recalculate
        # real dx: h_i * (1 - h_{i+k}) = h_i - h_i * h_{i+k} 
        # ∂/∂h_i = 1 - h_{i+k} 
        # ∂/∂h_{i+k} = -h_i
        # What we really want is to minimize max_k ∫ h(x)(1 - h(x+k)) dx
        # So, the f_k = ∫ h(x)(1 - h(x+k)) dx
        # We want to minimize max_k f_k by adjusting h_i
        # So gradient of max_k f_k in some sense
        # Let's just compute the gradient in the direction of the shift k that gives maximum
        
        # We want ∫ h(x)(1 - h(x+k)) dx = sum(h[i] * (1 - h[i+k])) over valid i/k pairs
        # Then ∂/∂h_i = 1 - h_{i+k} (where h is allowed to be 0 for indices out of bounds)
        # Let's compute matrix D[i,j] = h[i] * h[j]
        # But for every shift k: term contribution is h[i] * h[i+k] from i to i+k
        # which we compute as sum(h[i] - h[i]*h[i+k]) = 1 - sum(h[i] * h[i+k]) 
        # as long as sum(h) = 1 
        # Actually let's simplify. 
        # For shift k, for each i, if i+k is valid, then gradient is:
        #      1 - h[i+k]
        # Only over i such that i+k is valid. Similar computation.
        
        full_grad = np.zeros(n_points)
        for k in range(1 - n_points, n_points):
            if k >= 0:
                # h[i](1 - h[i+k]) gradient wrt h[i] is 1 - h[i+k] 
                if k < n_points:
                    full_grad[:n_points-k] += 1 - h[k:]
            else:
                # h[i](1 - h[i+k]) gradient wrt h[i] is 1 - h[i+k]
                if (n_points + k) > 0:
                    full_grad[-k:] += 1 - h[:n_points+k]
        
        # Take negative to minimize, as gradient ascent direction
        direction = -full_grad / (n_points)  # Normalize for step size
        # Or just: 
        # direction = -full_grad
        
        # Since h_i is bounded, we move along this direction and intersect with [0,1]
        # We can approximate for greedy full update:
        
        # New estimate: compute s = h + α * direction, project, compute c5
        # Hard to do this directly, instead do alternating projection:
        # 1) Set s = h + step * direction
        # 2) Clip s to [0,1]
        # 3) Scale s to sum to target
        # 4) evaluate new c5
        
        new_h = h + step_size * direction
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        
        new_c5 = evaluate_c5(new_h, dx) 
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        h = new_h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
