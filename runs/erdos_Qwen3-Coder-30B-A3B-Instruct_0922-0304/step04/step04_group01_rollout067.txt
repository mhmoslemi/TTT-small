import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Simple symmetric construction
        h_init = np.ones(n_points)
        mid = n_points // 2
        h_init[mid:] = 0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles around h_init
    for i in range(n_particles):
        particles[i] = h_init + np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    omega = 0.9
    phi_p = 2.0
    phi_g = 2.0
    inertia_damp = 0.99
    
    # Track personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO iterations
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            velocities[i] = (omega * velocities[i] 
                           + phi_p * r_p * (pbest[i] - particles[i])
                           + phi_g * r_g * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest = particles[i].copy()
        
        # Periodic reinitialization to prevent stagnation
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                particles[idx] = h_init + np.random.normal(0, 0.02, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
                pbest_scores[idx] = evaluate_c5(particles[idx], dx)
                if pbest_scores[idx] < gbest_score:
                    gbest_score = pbest_scores[idx]
                    gbest = particles[idx].copy()
        
        # Damp inertia periodically
        if iteration % 200 == 0 and iteration > 0:
            omega *= inertia_damp
            
        # Early exit if target achieved
        if gbest_score <= 0.3807:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
