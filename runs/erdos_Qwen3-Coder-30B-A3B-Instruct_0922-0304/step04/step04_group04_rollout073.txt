def compute_haar_basis(n_basis):
    """Compute Haar wavelet basis functions on [0,2]"""
    basis = []
    
    # Level 0: Constant
    basis.append(lambda x: 1.0)
    
    # Levels 1 to 4: Haar functions
    for level in range(1, 5):
        n_func = 2**(level-1)
        for i in range(n_func):
            def wavelet(x, j=i, k=level):
                if k == 0: return 1.0
                scale = 2**(k-1)
                pos = j / scale
                if pos <= x < pos + 1/scale:
                    return 1.0 if j % 2 == 0 else -1.0 
                return 0.0
            basis.append(wavelet)
    
    # Keep only n_basis functions (first 30)
    return basis[:n_basis]

def evaluate_c5(h, dx):
    """Evaluate the overlap integral"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_onto_basis(h_vals, basis):

    n_basis = len(basis)
    n_points = len(h_vals)

    int_basis = []

        integral = 0.0
        for j in range(n_points):
            x = j * dx
            integral += basis[i](x) * dx
        int_basis.append(integral)
    
    # Compute inner products
    A = np.zeros((n_basis, n_basis))
    for i in range(n_basis):
        for j in range(n_basis):
            s = 0.0
            for k in range(n_points):
                x = k * dx
                s += basis[i](x) * basis[j](x) * dx
            A[i,j] = s
    
    # Project h into basis (A * c = b)
    b = np.zeros(n_basis)
    for i in range(n_basis):
        s = 0.0
        for j in range(n_points):
            x = j * dx
            s += h_vals[j] * basis[i](x) * dx
        b[i] = s
        
    # Solve Ac = b for coefficients
    try:
        c = np.linalg.solve(A, b)
    except:
        c = np.linalg.lstsq(A, b, rcond=None)[0]
    
    return np.array(c)
