import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    else:
        # Initial triangular function
        h_current = np.zeros(n_points)
        for i in range(n_points):
            h_current[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current, dx)
    
    # Gradient-based local search with adaptive steps
    max_iter = 1000
    lr = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (approximated with finite differences)
        eps = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h_current.copy()
            h_minus[i] = max(0.0, h_minus[i] - eps)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient descent step
        h_new = h_current - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        h_current = h_new
        lr *= 0.9995  # Adaptive learning rate
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
