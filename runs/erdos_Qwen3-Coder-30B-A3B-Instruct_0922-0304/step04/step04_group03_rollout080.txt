import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to meet integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    step_size = 0.01
    min_step_size = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for the worst case
        grad = np.zeros(n_points)
        for i in range(n_points):
            # dO_k/dh_i = (1 - h[(i+k) % n]) - h[(i-k) % n]
            forward_idx = (i + max_k) % n_points
            backward_idx = (i - max_k) % n_points
            grad[i] = (1 - h[forward_idx]) - h[backward_idx]
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept the new solution if it's better
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        # Adapt step size
        if abs(new_c5 - best_c5) < 1e-8:
            step_size = max(min_step_size, step_size * 0.9)
    
    return (best_h.tolist(), best_c5, n_points)
