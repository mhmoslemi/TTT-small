import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def build_haar_basis(n):
    """Build Haar wavelet basis functions up to scale 5"""
    basis = []
    
    # Base function ψ₁(x)
    psi1 = np.zeros(n)
    mid = n // 2
    psi1[:mid] = 1
    psi1[mid:] = -1
    basis.append(psi1)
    
    # Generate higher scale functions
    for scale in range(1, 6):
        if len(basis) >= n:
            break
        for i in range(len(basis)):
            if len(basis) >= n:
                break
            if i == 0:
                # ψ₂ₖ(x) = ψₖ(2x)
                psi = np.zeros(n)
                for j in range(n // 2):
                    psi[2 * j] = basis[i][j]
                basis.append(psi)
            else:
                # ψ₂ₖ₊₁(x) = ψₖ(2x - 1)
                psi = np.zeros(n)
                for j in range(n // 2):
                    psi[2 * j + 1] = basis[i][j]
                basis.append(psi)
    
    return basis[:n]

def compute_inner_products(basis, n_points):
    """Compute all inner products K_ij(k) = ∫₀² ψᵢ(x)ψⱼ(x+k) dx"""
    K = {}
    for i, psi_i in enumerate(basis):
        for j, psi_j in enumerate(basis):
            K[(i, j)] = []
            for k in range(n_points):
                # Compute integral with shifted basis
                shifted_psi_j = np.zeros(n_points)
                if k < n_points:
                    shifted_psi_j[k:] = psi_j[:-k]
                else:
                    shifted_psi_j[:] = 0
                integral = np.sum(psi_i * shifted_psi_j)
                K[(i, j)].append(integral)
    return K

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 30  # Reduced for faster computation
    
    # Build Haar basis
    basis = build_haar_basis(basis_size)
    
    # Compute inner products
    K = compute_inner_products(basis, n_points)
    
    # Initialize coefficients
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Project to Haar basis
        coeffs = np.zeros(basis_size)
        for i in range(basis_size):
            coeffs[i] = np.sum(h_init * basis[i]) / n_points
        coeffs = np.clip(coeffs, 0, 1)
        coeffs = coeffs * (n_points / 2) / np.sum(coeffs)
    else:
        coeffs = np.random.rand(basis_size)
        coeffs = coeffs * (n_points / 2) / np.sum(coeffs)
    
    # Formulate and solve optimization problem
    c = cp.Variable(basis_size)
    z = cp.Variable((basis_size, basis_size), symmetric=True)
    
    # Objective: minimize maximum overlap
    max_overlap = cp.Variable()
    
    # Add constraints
    constraints = []
    
    # Integral constraint
    integral_constraint = cp.sum(c) == n_points / 2
    constraints.append(integral_constraint)
    
    # McCormick bounds
    for i in range(basis_size):
        for j in range(basis_size):
            # z_ij >= c_i + c_j - 1
            constraints.append(z[i, j] >= c[i] + c[j] - 1)
            # z_ij <= c_i
            constraints.append(z[i, j] <= c[i])
            # z_ij <= c_j
            constraints.append(z[i, j] <= c[j])
    
    # Add overlap expression with McCormick linearization
    # The overlap integral can be expressed using the inner products
    overlap_terms = []
    for k in range(n_points):
        term = 0
        for i in range(basis_size):
            for j in range(basis_size):
                term += c[i] * c[j] * K[(i, j)][k]
        overlap_terms.append(term)
    
    # Bound the maximum overlap
    for k in range(n_points):
        constraints.append(max_overlap >= overlap_terms[k])
        constraints.append(max_overlap >= -overlap_terms[k])
    
    # Solve optimization problem
    prob = cp.Problem(cp.Minimize(max_overlap), constraints)
    
    # Set solver and solve with timeout
    timeout_remaining = budget_s - (time.time() - start_time)
    if timeout_remaining < 5:
        return (h_init.tolist(), evaluate_c5(h_init, dx), n_points)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=timeout_remaining)
    except:
        # Fall back to less strict solver if MOSEK is not available
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, time_limit=timeout_remaining)
        except:
            # Fall back to standard optimization
            try:
                prob.solve(solver=cp.GLPK, verbose=False, time_limit=timeout_remaining)
            except:
                # Return initial solution if optimization fails
                return (h_init.tolist(), evaluate_c5(h_init, dx), n_points)
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        coeffs_opt = c.value
    else:
        coeffs_opt = coeffs
    
    # Reconstruct h(x) and verify
    h_opt = np.zeros(n_points)
    for i in range(basis_size):
        h_opt += coeffs_opt[i] * basis[i]
    
    # Ensure h(x) is in [0,1]
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Verify constraints
    if np.abs(np.sum(h_opt) * dx - 1) > 1e-4:
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    c5_bound = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), c5_bound, n_points)
