import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def mutate_individual(h, noise_level=0.01):
    """Apply small random mutation to an individual."""
    h = h + np.random.normal(0, noise_level, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(p1, p2):
    """Perform single-point crossover."""
    cutoff = np.random.randint(1, len(p1) - 1)
    child1 = np.concatenate([p1[:cutoff], p2[cutoff:]])
    child2 = np.concatenate([p2[:cutoff], p1[cutoff:]])
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Generate initial population
    for _ in range(population_size):
        h = mutate_individual(base_h.copy(), 0.02)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate entire population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Track best individual so far
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament selection)
        selected_indices = []
        tournament_size = 5
        for _ in range(population_size):
            tournament = np.random.choice(len(population), tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(best_individual.copy())
        
        # Crossover and mutation
        selected_pop = [population[i] for i in selected_indices]
        num_pairs = (population_size - 1) // 2
        
        for _ in range(num_pairs):
            parent1_idx = np.random.randint(len(selected_pop))
            parent2_idx = np.random.randint(len(selected_pop))
            
            if parent1_idx != parent2_idx:
                child1, child2 = crossover_parents(selected_pop[parent1_idx], selected_pop[parent2_idx])
                child1 = mutate_individual(child1, 0.01)
                child2 = mutate_individual(child2, 0.01)
                new_population.append(normalize_h(child1, target_sum))
                new_population.append(normalize_h(child2, target_sum))
        
        # Local search on top individuals
        sorted_indices = np.argsort(fitness)[:10]
        for i in sorted_indices:
            h = population[i].copy()
            
            # Constraint: must sum to target_sum
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            # Define bounds
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                new_population.append(h_refined)
                
        # Fill up to population_size
        while len(new_population) < population_size:
            new_population.append(mutate_individual(base_h.copy(), 0.02))
            new_population[-1] = normalize_h(new_population[-1], target_sum)
        
        # Update population
        population = new_population[:population_size]
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
