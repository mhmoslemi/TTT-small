import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial solution as warm start if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.random.rand(n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Prepare basis functions (Fourier)
    n_basis = 30
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    def basis_func(k, x):
        if k % 2 == 1:
            return np.sin((k+1) * np.pi * x / 2)
        else:
            return np.cos((k//2) * np.pi * x / 2)
            
    # Build basis matrix
    Phi = np.zeros((n_points, n_basis))
    for k in range(n_basis):
        Phi[:, k] = basis_func(k, x)
    
    # Compute inner products for all lags
    K = np.zeros((n_basis, n_basis, 2*n_points-1))
    for k in range(2*n_points-1):
        lag = k - n_points + 1
        for i in range(n_basis):
            for j in range(n_basis):
                Phi_ij = basis_func(i, x) * basis_func(j, x + lag)
                K[i, j, k] = np.trapz(Phi_ij, x) * dx
    
    # Optimization
    # Variables
    a = cp.Variable(n_basis)
    z = cp.Variable((n_basis, n_basis), symmetric=True)
    
    # Constraints
    constraints = []
    
    # Integral constraint
    constraints.append(cp.sum(a) * dx == 1.0)
    
    # McCormick envelopes
    for i in range(n_basis):
        for j in range(n_basis):
            z[i, j] >= a[i] + a[j] - 1
            z[i, j] <= a[i]
            z[i, j] <= a[j]
            
    # Find bounds on a_k
    a_lower = np.zeros(n_basis)
    a_upper = np.ones(n_basis)
    
    for i in range(n_basis):
        constraints.append(a[i] >= 0)
        constraints.append(a[i] <= 1)
    
    # Objective
    objective = cp.Minimize(cp.max([cp.sum(cp.multiply(z, K[:, :, k])) for k in range(2*n_points-1)]))
    
    # Problem formulation
    problem = cp.Problem(objective, constraints)
    
    # Initial guess from the warm start
    a0 = Phi.T @ h_init
    a.value = a0
    
    best_score = float('inf')
    best_h = h_init
    
    # Try optimization
    try:
        problem.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6, reltol=1e-6)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            a_opt = a.value
            h_opt = Phi @ a_opt
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt) 
            score = evaluate_c5(h_opt, dx)
            if score < best_score:
                best_score = score
                best_h = h_opt
            
    except Exception as e:
        pass  # Fall back to basic PSO
    
    # Fallback PSO if needed
    if best_score >= 0.3807:
        # Reinitialize using a simpler approach
        n_particles = 15
        max_iter = 2000
        
        # Initialize swarm using scaled initial_h_values
        swarm = []
        h_init = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())

        for i in range(1, n_particles):
            h = h_init.copy()
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            swarm.append(h)
        
        # Initialize personal bests
        pbest = [swarm[i].copy() for i in range(n_particles)]
        pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
        
        # Initialize global best
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        # Main PSO loop
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            
            for i in range(n_particles):
                # Simple update with some randomization
                r_p = np.random.rand(n_points)
                r_g = np.random.rand(n_points)
                # Random perturbation
                swarm[i] = swarm[i] + 0.02 * np.random.randn(n_points)
                swarm[i] = np.clip(swarm[i], 0, 1)
                swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
                
                # Evaluate
                score = evaluate_c5(swarm[i], dx)
                
                # Update personal best
                if score < pbest_scores[i]:
                    pbest[i] = swarm[i].copy()
                    pbest_scores[i] = score
                    
                    # Update global best
                    if score < gbest_score:
                        gbest = swarm[i].copy()
                        gbest_score = score
                        
        if gbest_score < best_score:
            best_h = gbest
            best_score = gbest_score
    
    return (best_h.tolist(), best_score, n_points)
