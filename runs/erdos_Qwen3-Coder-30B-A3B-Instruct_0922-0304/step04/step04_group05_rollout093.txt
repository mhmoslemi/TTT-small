import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 2000
    elite_size = population_size // 4
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population with random feasible individuals
    for i in range(1, population_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Main evolutionary loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        offspring = []
        for i in range(population_size - len(elite)):
            # Tournament selection
            parent_idx = sorted_indices[np.random.randint(elite_size)]
            parent = population[parent_idx]
            
            # Mutation
            noise = np.random.normal(0, mutation_scale, n_points)
            child = parent + noise
            child = project_to_feasible(child, n_points)
            offspring.append(child)
        
        # Form new population
        population = elite + offspring
        
        # Adaptive mutation scaling
        if iteration % 10 == 0:
            if iteration > 0:
                recent_improvements = 0
                for i in range(1, 11):
                    if fitness[-i] < best_c5:
                        recent_improvements += 1
                if recent_improvements > 7:
                    mutation_scale *= 1.05
                elif recent_improvements < 3:
                    mutation_scale *= 0.95
                mutation_scale = np.clip(mutation_scale, 0.001, 0.5)
    
    # Final evaluation of best
    if best_h is not None:
        final_c5 = evaluate_c5(best_h, dx)
    else:
        final_c5 = evaluate_c5(population[0], dx)
    
    return (best_h.tolist() if best_h is not None else population[0].tolist(), 
            final_c5, n_points)
