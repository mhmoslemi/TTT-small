import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    max_overlap = 0
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The overlap for each shift k
    for k in range(2 * n - 1):
        overlap = conv[k] * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise construct a simple initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # A reasonably good initial trial solution
        h = np.zeros(n_points)
        # Place two intervals: one from 0.2 to 0.7 and another from 1.2 to 1.7
        # Normalize to integrate to 1
        for i in range(n_points):
            x = i * dx
            if 0.2 <= x <= 0.7:
                h[i] = 1.0
            elif 1.2 <= x <= 1.7:
                h[i] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimizer: directly optimize h with constraints using scipy
    def objective(h_vec):
        # Reshape back into h
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        # Evaluate C5
        return evaluate_c5(h, dx)
    
    # Define bounds for each h_i
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Linear constraint for integral = 1
    # A * h = b where A is the sum vector and b = 1
    A_eq = np.ones(n_points)
    b_eq = 1.0
    
    # Use L-BFGS-B for bound constrained optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000})
    
    # Extract best solution
    final_h = np.clip(result.x, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_c5(final_h, dx)
    
    # Refine with local search (grid search in region around current)
    best_h = final_h.copy()
    best_c5 = final_c5
    
    # Take 20x20 grid of nearby solutions for local improvement
    steps_a = np.linspace(-0.1, 0.1, 20)
    steps_b = np.linspace(-0.1, 0.1, 20)
    
    for da in steps_a:
        for db in steps_b:
            h_trial = final_h + da * np.random.rand(n_points) + db * np.random.rand(n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                if time.time() - start_time > budget_s - 1:
                    break
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
