import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start gradient descent
    n_random_initializations = 50
    n_gradient_steps = 200
    
    # Main MSGD loop
    for restart in range(n_random_initializations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Initialize with either parent or random
        if restart == 0 and 'initial_h_values' in globals():
            current_h = h.copy()
        else:
            # Random initialization
            current_h = np.random.rand(n_points)
            # Scale to satisfy constraint
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            # Clip to feasible region
            current_h = np.clip(current_h, 0, 1)
        
        # Local optimization with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vec):
            return evaluate_c5(h_vec, dx)
        
        try:
            result = minimize(objective, current_h, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': n_gradient_steps, 'ftol': 1e-9})
            
            if result.success:
                final_h = result.x
                final_c5 = evaluate_c5(final_h, dx)
                
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h.copy()
                    
        except:
            pass
            
        # Early stopping check
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with more gradient steps
    if time.time() - start_time < budget_s * 0.95:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vec):
            return evaluate_c5(h_vec, dx)
        
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': 1000, 'ftol': 1e-12})
            
            if result.success:
                final_h = result.x
                final_c5 = evaluate_c5(final_h, dx)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h
        except:
            pass
    
    # Final validation and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
