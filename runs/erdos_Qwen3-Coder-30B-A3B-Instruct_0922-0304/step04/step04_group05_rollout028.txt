```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Sequential convex programming
    max_iter = 1000
    alpha = 0.1
    beta = 0.5
    min_alpha = 1e-6
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # For each iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute finite differences for gradients
        # This gives us gradient in terms of h values
        # We approximate ∇C_k based on current h
        # But since we're trying to minimize max C_k, we look at the violating k to focus descent
        
        # Compute current C values for all shifts
        C_vals = np.zeros(n_points * 2 - 1)
        for k in range(-(n_points-1), n_points):
            if k >= 0:
                C_vals[k + n_points - 1] = np.sum(h[:n_points-k] * (1 - h[k:]))
            else:
                C_vals[k + n_points - 1] = np.sum(h[-k:] * (1 - h[:n_points+k]))
        C_vals = C_vals * dx
        
        # Find the index of largest C_k
        max_k_idx = np.argmax(C_vals)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute the gradient of C_max_k w.r.t. h
        # This is going to be a bit tricky - we approximate it
        
        # For each component of h, get its contribution to the maximum correlation
        grad = np.zeros(n_points)
        if max_k >= 0:
            # We're looking at correlation h[i] * (1-h[i+max_k])
            # So gradient is for h[i] -> (1 - h[i+max_k]) and for h[i+max_k] -> -h[i]  
            # But we're adding h[i]*1 - h[i]*h[i+max_k]
            for i in range(n_points - max_k):
                grad[i] += 1 - h[i + max_k]
                if (i + max_k) < n_points:
                    grad[i + max_k] -= h[i]
        else:
            # max_k < 0, means h contrib to h[i+|max_k|]
            max_k_abs = -max_k
            for i in range(n_points - max_k_abs):
                grad[i] += 1 - h[i + max_k_abs]
                if i + max_k_abs < n_points:
                    grad[i + max_k_abs] -= h[i]
        
        # Approximate Hessian (second-order adjoint) - simplify to identity + regularization
        # Add regularization to ensure positive definiteness
        reg = 1e-8
        H = np.eye(n_points) * reg
    
        # Now we set up a quadratic objective to locally minimize the maximum correlation
        # minimize (1/2)*||δ||^2 + grad^T * δ subject to linear constraints
        
        # We solve:
        # minimize (1/2)*|δ|^2 + grad^T * δ (total change) 
        # s.t. 0 <= h + δ <= 1
        #      sum(h + δ) = n_points / 2
        # Can write as minimize (1/2)*||δ||^2 + grad^T * δ subject to Ax = b, l <= x <= u
        
        # Due to constraints, this MPC's Hessian with augmented Lagrangian
        
        try:
            # Actually, use CVXPY to get precise convex subproblem solution
            delta = cp.Variable(n_points)
            # Objective: ∇C^T δ + ½ δ^T ∇²C δ
            objective = cp.Minimize(cp.scalar_product(grad, delta) + 0.5 * cp.quad_form(delta, H))
            
            # Constraints
            constraints = [
                delta >= -h,  # h + delta >= 0 => delta >= -h
                delta <= 1-h, # h + delta <= 1 => delta <= 1-h
                cp.sum(delta) == 0 # sum(h + delta) = sum(h) => delta sum = 0 but that's wrong
            ]
            
            # Actually if we're a convex problem, reuse simpler bounds
            # However, we must ensure sum(delta) == 0 (meaning total mass preservation)
            # So we constrain:
            # sum(delta) == 0 (since we're enforcing constraint sum(h + delta) = sum(h))
            # This is what the original constraint checks
            constraints = [
                delta >= -h,  
                delta <= 1-h, 
                cp.sum(delta) == 0
            ]
            
            # For performance and for facility with processes, use modified subproblems
            # Step 1: remove any offset brought by h being slightly off from sum = n_points/2
            corrected_sum = np.sum(h)
            offset = corrected_sum - n_points/2
            # Add to one damping slider to make normalized
            delta = cp.Variable(n_points)
            objective = cp.Minimize(cp.scalar_product(grad, delta) + 0.5 * cp.quad_form(delta, H))
            constraints = [
                delta >= -h,
                delta <= 1-h, 
                cp.sum(delta) == 0
            ]
            prob = cp.Problem(objective, constraints)
            
            try:
                # Try primal-dual interior-point method or Ecos solver
                prob.solve(solver=cp.ECOS, verbose=False)
                step = delta.value.astype(np.float64)  # or choose another solver
    
                if step is None:
                    # fallback step size
                    step = alpha * grad
                else:
                    step = -step  # because we want to decrease the objective directly
                
            except:
                # fallback to gradient steps if CVX fails
                step = alpha * grad
                
        except Exception as e:
            # fallback
            step = alpha * grad
            
        # To stay feasible, we project back into bounds 
        h_new = h + step
        # Clip soft bounds (0 and 1) 
        h_new = np.clip(h_new, 0.0, 1.0)
        # Then adjust to enforce exact sum(n_points/2)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply the update
        h = h_new.copy()
        
        # Get new C5 value
        current_c5 = evaluate_c5(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Adaptive stepsize
        if best_c5 <= 0.3807:
            break
            
        # Slight damping after few iterations to avoid oscillation
        if iter_num > 20:
            alpha *= 0.99
        
        if alpha < min_alpha:
            alpha = min_alpha

    # Final check
    final_c5 = evaluate_c5(best_h, dx)

    return (best_h.tolist(), final_c5, n_points)
```
