import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        # Initialize with triangular function
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill the rest with random feasible solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = [evaluate_c5(h, dx) for h in population]
    
    # Track best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Buffers to store elite candidates
    elite_size = int(elite_ratio * population_size)
    
    # Evolutionary strategy main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate offspring using mutation
        offspring = []
        for i in range(population_size):
            parent = population[i]
            # Mutate with Gaussian noise
            noise = np.random.normal(0, mutation_scale, n_points)
            child = parent + noise
            child = project_feasible(child, n_points)
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = [evaluate_c5(h, dx) for h in offspring]
        
        # Combine population and offspring
        combined_population = population + offspring
        combined_fitness = fitness + offspring_fitness
        
        # Select the best individuals
        sorted_indices = np.argsort(combined_fitness)
        selected_indices = sorted_indices[:population_size]
        
        # Update population and fitness
        population = [combined_population[i] for i in selected_indices]
        fitness = [combined_fitness[i] for i in selected_indices]
        
        # Update best if improved
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Adaptive mutation: adjust based on success
        if iteration % 50 == 0:
            # Calculate success rate over last 50 generations
            if iteration > 0:
                mutation_scale *= 0.95  # Decrease scale gradually
            else:
                # If we're doing well, reduce noise
                if best_c5 < 0.381:
                    mutation_scale = max(0.01, mutation_scale * 0.9)
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
