import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # GA iterations
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            fit = evaluate_c5(individual, dx)
            fitness.append(fit)
        
        # Find best in generation
        gen_best_idx = np.argmin(fitness)
        if fitness[gen_best_idx] < best_c5:
            best_c5 = fitness[gen_best_idx]
            best_h = population[gen_best_idx].copy()
            
        # Check if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child1 += noise
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child2 += noise
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP from the best GA result
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
