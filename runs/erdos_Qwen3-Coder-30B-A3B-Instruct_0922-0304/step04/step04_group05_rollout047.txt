import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Alternating optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Estimate gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1, h[i] + eps)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[i] = max(0, h[i] - eps)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
