import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Grid search for optimal interval separation
    d_candidates = np.arange(0, 1.01, 0.01)
    
    for d in d_candidates:
        # Create step function with two intervals
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        a = 0    # starting position
        
        # First interval [a, a+w]
        i0 = 0
        i1 = int((a + w) / dx) - 1
        
        # Second interval [a+w+d, a+w+d+w]
        i2 = int((a + w + d) / dx)
        i3 = int((a + w + d + w) / dx) - 1
        
        if i1 >= 0 and i0 < n_points:
            h[i0:i1+1] = 1
        if i3 >= 0 and i2 < n_points:
            h[i2:i3+1] = 1
            
        # Normalize to ensure ∫h(x)dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate overlap
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # If no improvement found, return initial solution
    if best_h is None:
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            best_h = h.copy()
            best_c5 = evaluate_c5(best_h, dx)
        else:
            # Fallback: triangular shape
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            best_h = h.copy()
            best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
