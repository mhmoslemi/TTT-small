import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    # Generate initial population
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            # Start from initial_h_values with small perturbation
            ind = h + np.random.normal(0, 0.05, n_points)
        else:
            # Random initialization
            ind = np.random.rand(n_points)
        ind = np.clip(ind, 0, 1)
        ind = ind * (n_points / 2) / np.sum(ind)
        population.append(ind)
        fitness.append(evaluate_c5(ind, dx))
    
    # GA iterations
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Tournament selection
        new_population = []
        for _ in range(population_size):
            # Select tournament participants
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone winner
            new_ind = population[winner_idx].copy()
            
            # Apply crossover with another random individual
            if np.random.rand() < crossover_rate:
                other_idx = np.random.randint(0, population_size)
                if other_idx != winner_idx:
                    crossover_point = np.random.randint(1, n_points - 1)
                    new_ind[:crossover_point] = population[other_idx][:crossover_point]
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                new_ind[mutation_idx] += np.random.normal(0, 0.1)
                new_ind[mutation_idx] = np.clip(new_ind[mutation_idx], 0, 1)
                new_ind = new_ind * (n_points / 2) / np.sum(new_ind)
            
            new_population.append(new_ind)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_c5(ind, dx) for ind in population]
        
        # Track best
        best_in_gen = np.min(fitness)
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[np.argmin(fitness)].copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
