import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for i in range(n_particles):
            particles[i] = h0 + np.random.normal(0, 0.01, n_points)
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    else:
        # Fallback to a symmetric construction
        h0 = np.linspace(0, 1, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for i in range(n_particles):
            particles[i] = h0 + np.random.normal(0, 0.01, n_points)
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    
    # Set PSO parameters
    w = 0.9
    phi_p = 2.0
    phi_g = 2.0
    
    # Track personal and global bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_c5(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            phi_p * r_p * (pbest[i] - particles[i]) + 
                            phi_g * r_g * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
        
        # Periodic reinitialization to avoid stagnation
        if iteration % 100 == 0 and iteration > 0:
            n_reinit = n_particles // 10
            indices = np.random.choice(n_particles, n_reinit, replace=False)
            for i in indices:
                if 'initial_h_values' in globals():
                    h0 = np.array(initial_h_values)
                    h0 = np.clip(h0, 0, 1)
                    h0 = h0 * (n_points / 2) / np.sum(h0)
                    particles[i] = h0 + np.random.normal(0, 0.01, n_points)
                else:
                    particles[i] = np.linspace(0, 1, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
                # Re-evaluate
                score = evaluate_c5(particles[i], dx)
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
        
        # Early exit
        if gbest_score <= 0.3807:
            break
            
        # Apply velocity damping occasionally
        if iteration % 100 == 0 and iteration > 0:
            velocities *= 0.99
    
    return (gbest.tolist(), gbest_score, n_points)
