import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a triangular shape as a starting point
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint ∫h dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search on interval configuration
    # Try different positions of two non-overlapping intervals of width 0.5
    d_candidates = np.arange(0.0, 1.0 + 0.001, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create piecewise constant function with two intervals
        # Interval 1: [0, 0.5], Interval 2: [0.5 + d, 1.0 + d]
        # Each interval has width 0.5, so total width = 1.0
        
        # Compute indices
        i0 = 0
        i1 = int(0.5 / dx) - 1
        i2 = int((0.5 + d) / dx)
        i3 = int((1.0 + d) / dx) - 1
        
        # Check for valid indices
        if i1 < i0 or i3 < i2 or i2 <= i1:
            continue
            
        # Create h
        h_new = np.zeros(n_points)
        h_new[i0:i1+1] = 1.0
        h_new[i2:i3+1] = 1.0
        
        # Normalize to satisfy ∫h dx = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5 = evaluate_c5(h_new, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Refine using gradient descent
    # Only refine a bit if we're near target
    if best_c5 <= 0.381:
        # Use fewer evaluations for optimization to stay within time limit
        max_iter = 500
        learning_rate = 0.1
        
        h_opt = best_h.copy()
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient over all shifts (approximated)
            # For simplicity, we are directly computing with the current h
            c5 = evaluate_c5(h_opt, dx)
            
            # Compute a simple gradient approximation here by small perturbations
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] = min(1.0, h_plus[i] + eps)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps
            
            # Update h
            h_opt = h_opt - learning_rate * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Update best
            c5_new = evaluate_c5(h_opt, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_opt.copy()
    
    return (best_h.tolist(), best_c5, n_points)
