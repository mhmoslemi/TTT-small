import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize using the parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint ∫h(x)dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 value
        current_c5 = evaluate_c5(h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                # Reduce learning rate and reset patience
                learning_rate *= 0.9
                patience_counter = 0
                
        # Find the shift k that gives maximum overlap
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            # Forward shift
            forward_idx = (j + k_max) % n_points
            # Backward shift
            backward_idx = (j - k_max) % n_points
            gradient[j] = (1 - h[forward_idx]) - h[backward_idx]
            
        # Apply gradient step
        h = h - learning_rate * gradient * dx
        
        # Clip values to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if we're close to target
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
