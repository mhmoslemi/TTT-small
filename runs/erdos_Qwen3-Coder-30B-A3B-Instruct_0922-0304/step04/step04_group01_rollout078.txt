import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def compute_gradient(h, dx):
    """Compute gradient of C5 w.r.t. h using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution
    n_points = 75
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Alternative initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 10000
    step_size = 0.002
    decay_factor = 0.995
    perturbation_freq = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and update
        grad = compute_gradient(h, dx)
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        current_c5 = evaluate_c5(h_new, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
        # Adaptive step size
        if iteration > 0 and iteration % 10 == 0:
            step_size *= decay_factor
            
        h = h_new
        
        # Periodic perturbation to escape local minima
        if iteration % perturbation_freq == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
