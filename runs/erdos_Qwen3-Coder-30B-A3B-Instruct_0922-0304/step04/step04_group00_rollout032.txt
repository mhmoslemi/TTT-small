import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Save original value
            original_h_i = h[i]
            
            # Try different values for h[i]
            best_h_i = original_h_i
            best_local_c5 = current_c5
            
            # Sample a few values around current value
            test_values = np.linspace(max(0, original_h_i - 0.1), 
                                    min(1, original_h_i + 0.1), 10)
            
            for test_h_i in test_values:
                h[i] = test_h_i
                # Normalize to maintain sum = n_points / 2
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                h = np.clip(h, 0, 1)
                
                # Compute new C5
                new_c5 = compute_c5(h)
                if new_c5 < best_local_c5:
                    best_local_c5 = new_c5
                    best_h_i = test_h_i
                    
            # Update h[i] to best value found
            h[i] = best_h_i
            # Normalize to maintain sum = n_points / 2
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            h = np.clip(h, 0, 1)

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
