import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > 50:
            # Restart with a perturbed version
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            
        # Compute gradient
        # For each position i, we need to determine how changing h[i] affects the maximum overlap
        # This is a complex derivative, so we approximate it
        # We compute the gradient using finite differences
        eps = 1e-5
        h_plus = h.copy()
        h_minus = h.copy()
        
        max_overlap_idx = np.argmax(correlate(h, 1-h, mode="full"))
        max_k = max_overlap_idx - (n_points - 1)
        
        # Gradient estimation for maximum overlap term
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            
            if max_k > 0:
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_overlap_idx]
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_overlap_idx]
            else:
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_overlap_idx]
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_overlap_idx]
                
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if iter_num % 100 == 0 and iter_num > 0:
            learning_rate *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)
