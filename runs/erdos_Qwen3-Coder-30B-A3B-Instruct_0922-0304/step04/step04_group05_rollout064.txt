import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    h = project_to_feasible(h, n_points, dx)
    
    # Gradient-based optimization
    max_iter = 2000
    alpha = 0.1
    beta = 0.5
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # C_k = dx * sum h[i] * (1 - h[i+k])
        # dC_k/dh[j] = dx * (1 - h[j+k] - h[j]) if k = j-j0 
        # We use exact gradient computation for one valid k
        
        # More direct approach: approximate gradients using finite differences
        # but for now let's compute using correlation
        overlap = correlate(h, 1 - h, mode="full")
        c5_value = np.max(overlap) * dx
        overlap_idx = np.argmax(overlap)
        # Compute gradient with respect to shifting the values
        # This is computationally expensive; use approximate direction
        # We do gradient descent using an approximation
        
        # Try gradient decent using a linear approximation of the objective
        # as in Frank-Wolfe: we want to minimize a linear approximation
        
        # Gradients of overlap wrt h, computed approximately:
        # d/dh[i] (sum_{j,k} h[j](1-h[j+k])) = (1 - 2*h[i] - avg(h[i+k]))
        gradient = np.zeros(n_points)
        
        # We approximate gradient as (1 - 2*h - average of shifted)
        shift_plus = np.zeros(n_points)
        shift_minus = np.zeros(n_points)
        for k in range(1, n_points):
            shift_plus[k] = np.sum(h[:-k] * (1 - h[k:])) if k < n_points else 0
            shift_minus[k] = np.sum(h[k:] * (1 - h[:-k])) if k < n_points else 0
        # Use actual correlation as approximation for gradient
        full_corr = correlate(h, 1-h, mode='full')
        max_idx = np.argmax(full_corr)
        target_corr = full_corr[max_idx]
        # Direction of descent should be - gradient direction
        # Approximate gradient at each point
        shifted = np.zeros(n_points)
        for i in range(n_points):
            shifted[i] = np.sum(h * np.roll(1-h, i))
        
        shifted2 = np.zeros(n_points)
        for i in range(n_points):
            shifted2[i] = np.sum(np.roll(h, i) * (1 - h))
        
        # Compute finite difference gradient
        eps = 1e-4
        grad_approx = np.zeros(n_points)
        for i in range(n_points):
            h_new = h.copy()
            h_new[i] += eps
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_plus = evaluate_c5(h_new, dx)
            
            h_new = h.copy()
            h_new[i] -= eps
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_minus = evaluate_c5(h_new, dx)
            grad_approx[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Now try Frank-Wolfe like update: find direction of descent 
        # by solving constrained optimization of linearized objective
        # we make small gradient step
        descent = - grad_approx
        
        # Find step size via Armijo backtracking
        t = alpha
        while True:
            h_new = h + t * descent
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new <= c5_value + 1e-6 * t * np.dot(grad_approx, descent):
                break
            t *= beta
            
        h = h + t * descent
        h = project_to_feasible(h, n_points, dx)
        
        # Check convergence
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        if abs(c5 - c5_value) < tolerance:
            break
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
