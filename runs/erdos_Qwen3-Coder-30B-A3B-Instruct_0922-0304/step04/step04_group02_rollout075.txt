import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    max_generations = 2000
    
    # Initialize population
    population = []
    fitness_values = []
    
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness_values.append(evaluate_c5(individual, dx))
    
    # Check if we've already found a good solution
    for i in range(population_size):
        if fitness_values[i] < best_c5:
            best_c5 = fitness_values[i]
            best_h = population[i].copy()
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Genetic Algorithm iterations
    for generation in range(max_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Tournament selection
        new_population = []
        new_fitness = []
        
        for _ in range(population_size):
            # Tournament
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_values[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            
            new_population.append(population[winner_index].copy())
            new_fitness.append(fitness_values[winner_index])
        
        # Crossover and mutation
        offspring = []
        offspring_fitness = []
        
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[(i + 1) % population_size]
            
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.randint(0, n_points, size=int(n_points * 0.1))
                for idx in mutation_indices:
                    child1[idx] += np.random.normal(0, 0.05)
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.randint(0, n_points, size=int(n_points * 0.1))
                for idx in mutation_indices:
                    child2[idx] += np.random.normal(0, 0.05)
            
            # Rescale to meet constraint
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            offspring.append(child1)
            offspring.append(child2)
            
            offspring_fitness.append(evaluate_c5(child1, dx))
            offspring_fitness.append(evaluate_c5(child2, dx))
        
        # Replace population with offspring
        population = offspring
        fitness_values = offspring_fitness
        
        # Update best solution
        for i in range(population_size):
            if fitness_values[i] < best_c5:
                best_c5 = fitness_values[i]
                best_h = population[i].copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
