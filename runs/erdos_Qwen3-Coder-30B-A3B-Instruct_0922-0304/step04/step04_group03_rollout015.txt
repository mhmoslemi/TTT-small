import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # Width of each interval
        # Place intervals at symmetric positions
        a = 0.0
        i0 = 0
        i1 = int((a + w) / dx) - 1
        i2 = int((a + w + 0.5) / dx)
        i3 = int((a + w + 0.5 + w) / dx) - 1
        
        # Ensure indices are valid
        i1 = min(i1, n_points - 1)
        i3 = min(i3, n_points - 1)
        
        if i1 >= 0 and i2 < n_points:
            h[i0:i1+1] = 1.0
            h[i2:i3+1] = 1.0
        
        # Normalize to integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search over interval spacing
    d_candidates = np.arange(0.0, 1.01, 0.01)
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        h_candidate = np.zeros(n_points)
        w = 0.5  # Each interval has width 0.5
        
        # Left interval
        a = 0.0
        i0 = 0
        i1 = int((a + w) / dx) - 1
        
        # Right interval
        i2 = int((a + w + d) / dx)
        i3 = int((a + w + d + w) / dx) - 1
        
        # Clamp indices to valid range
        i1 = min(i1, n_points - 1)
        i3 = min(i3, n_points - 1)
        
        # Ensure valid ranges and non-overlapping
        if i1 >= 0 and i2 < n_points:
            h_candidate[i0:i1+1] = 1.0
            h_candidate[i2:i3+1] = 1.0
            
        # Normalize to satisfy ∫h(x)dx = 1
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        c5_val = evaluate_c5(h_candidate, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
