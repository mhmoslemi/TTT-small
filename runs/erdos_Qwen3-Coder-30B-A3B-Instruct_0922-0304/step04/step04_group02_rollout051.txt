import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(pop_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness.append(evaluate_c5(individual, dx))
    
    # Sort initial population
    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]
    fitness = [fitness[i] for i in sorted_indices]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Tournament selection
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone winner to population
            new_individual = population[winner_idx].copy()
            
            # Apply crossover with probability
            if np.random.rand() < crossover_rate:
                # Select another parent
                parent2_idx = np.random.choice(pop_size, tournament_size)
                parent2_fitness = [fitness[i] for i in parent2_idx]
                parent2_winner = parent2_idx[np.argmin(parent2_fitness)]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points-1)
                new_individual[:crossover_point] = population[winner_idx][:crossover_point]
                new_individual[crossover_point:] = population[parent2_winner][crossover_point:]
            
            # Apply mutation with probability
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.1, n_points)
                new_individual += noise
                new_individual = np.clip(new_individual, 0, 1)
                new_individual = new_individual * (n_points / 2) / np.sum(new_individual)
                
                # Uniform random mutation (alternative)
                # if np.random.rand() < 0.5:
                #     mutation_indices = np.random.choice(n_points, size=n_points//10)
                #     new_individual[mutation_indices] = np.random.rand(n_points//10)
                #     new_individual = new_individual * (n_points / 2) / np.sum(new_individual)
            
            new_population.append(new_individual)
        
        # Evaluate new population
        new_fitness = []
        for individual in new_population:
            new_fitness.append(evaluate_c5(individual, dx))
        
        # Combine and sort populations
        population.extend(new_population)
        fitness.extend(new_fitness)
        
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices[:pop_size]]
        fitness = [fitness[i] for i in sorted_indices[:pop_size]]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
