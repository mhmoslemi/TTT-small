import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 75
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to triangular construction
        h = np.linspace(0, 1, n_points)
        h = np.minimum(h, 1 - h)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using gradient-based method with adaptive step size
    max_iter = 10000
    current_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and gradient
        corr = correlate(current_h, 1 - current_h, mode="full")
        max_idx = np.argmax(corr)
        current_c5 = corr[max_idx] * dx
        
        # Compute gradient approximation using finite differences
        grad = np.zeros_like(current_h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adaptive step size based on improvement
        step_size = 0.002
        h_new = current_h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = correlate(h_new, 1 - h_new, mode="full")[max_idx] * dx
        
        if new_c5 < current_c5:
            current_h = h_new
        else:
            # Try reduced step size
            step_size *= 0.5
            h_new = current_h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = correlate(h_new, 1 - h_new, mode="full")[max_idx] * dx
            
            if new_c5 < current_c5:
                current_h = h_new
        
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            current_h += noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Update best solution
        final_corr = correlate(current_h, 1 - current_h, mode="full")
        final_c5 = np.max(final_corr) * dx
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = current_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
