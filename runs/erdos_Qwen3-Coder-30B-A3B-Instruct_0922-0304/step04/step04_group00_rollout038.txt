import numpy as np
from scipy.signal import correlate
import time
import math

def compute_c5(h, dx):
    """Compute the C5 overlap integral"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    n_points = 100
    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Simulated annealing parameters
    T0 = 1e-4
    alpha = 0.99
    T = T0
    max_iter = 10000
    
    # Main simulated annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb a random point
        i = np.random.randint(0, n_points)
        old_h = h[i]
        
        # Generate neighbor by perturbing one point
        delta = np.random.normal(0, 0.05)
        h[i] += delta
        
        # Clip to [0, 1]
        h[i] = np.clip(h[i], 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_c5 = new_c5

            delta_c5 = new_c5 - best_c5

            else:
                # Reject, revert h

        # Cool temperature
        T *= alpha
