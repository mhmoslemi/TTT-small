import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import linprog

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Dual decomposition approach
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Update magnitudes while keeping support fixed
        # Solve optimization problem for current support
        support = (h > 1e-6).astype(int)
        active_indices = np.where(support > 0)[0]
        
        # Create constraint matrix
        A_eq = np.array([np.ones(n_points)])
        b_eq = np.array([n_points / 2])
        
        # Bounds: 0 <= h[i] <= 1
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Project to feasible region
        proj_h = h.copy()
        proj_h = np.clip(proj_h, 0, 1)
        proj_h = proj_h * (n_points / 2) / np.sum(proj_h)
        
        # Check if we should update current solution
        c5_val = evaluate_c5(proj_h, dx)
        if c5_val < best_c5:
            best_h = proj_h.copy()
            best_c5 = c5_val
            if best_c5 <= 0.3807:
                break
        
        # Small random perturbation for exploration
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
