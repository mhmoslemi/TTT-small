import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * 0.02  # dx = 2 / n_points = 2 / 100 = 0.02

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iters = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle += np.random.normal(0, 0.02, n_points)
            particle = np.clip(particle, 0, 1)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle += np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle))
    
    # Global best
    global_best_idx = np.argmin(p_best_values)
    global_best = p_best[global_best_idx].copy()
    global_best_value = p_best_values[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            current_value = compute_c5(particles[i])
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                
            # Update global best
            if current_value < global_best_value:
                global_best = particles[i].copy()
                global_best_value = current_value
                
        # Early termination
        if global_best_value <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_value, n_points)
