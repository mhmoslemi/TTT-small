import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: simple smoothed step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization: Project onto Haar wavelet basis
    # Construct Haar wavelets up to level 5 (32 basis functions)
    # We'll compute c_k by projecting h onto these modes
    n_basis = 30  # Less than full wavelet basis for efficiency
    basis_functions = []
    
    # Generate Haar wavelet basis functions
    for j in range(6):  # up to level 5
        for k in range(2**j):
            if j == 0:
                # Scaling function at level 0
                wavelet = np.ones(n_points)
            else:
                # Haar wavelet at level j, position k
                wavelet = np.zeros(n_points)
                start_idx = k * (n_points // (2**j))
                end_idx = start_idx + (n_points // (2**j))
                mid = (start_idx + end_idx) // 2
                wavelet[start_idx:mid] = 1.0
                wavelet[mid:end_idx] = -1.0
            basis_functions.append(wavelet)
    
    # Limit to n_basis basis functions (we'll use first 30)
    basis_functions = basis_functions[:n_basis]
    
    # Convert to matrix for projection
    Phi = np.array(basis_functions).T  # Shape: (n_points, n_basis)
    
    # Project h onto basis to get coefficients c
    c = np.linalg.lstsq(Phi, h, rcond=None)[0]
    
    # Use convex optimization to refine c
    c_vars = cp.Variable(n_basis)
    
    # Objective: minimize max overlap
    # We'll use an iterative approach - constrain c to be close to previous
    # to preserve good properties but allow continuous optimization
    
    # Constraints
    integral_constraint = cp.sum(cp.multiply(c_vars, Phi.T @ np.ones(n_points))) == 1
    bounds_constraints = [0 <= c_vars[i] for i in range(n_basis)]
    bounds_constraints += [c_vars[i] <= 1 for i in range(n_basis)]
    
    # Convex program
    prob = cp.Problem(cp.Minimize(cp.sum_squares(c_vars - c)), 
                      bounds_constraints + [integral_constraint])
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        refined_c = c_vars.value
    except:
        refined_c = c.copy()
    
    # Compute refined h from coefficients
    refined_h = Phi @ refined_c
    
    # Ensure constraints
    refined_h = np.clip(refined_h, 0, 1)
    refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
    
    # Final evaluation
    c5_final = evaluate_c5(refined_h, dx)
    
    # We do a local search refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Small perturbation and reoptimize
        delta = np.random.normal(0, 0.01, n_points)
        new_h = refined_h + delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = evaluate_c5(new_h, dx)
        if new_c5 < c5_final:
            c5_final = new_c5
            refined_h = new_h
        elif _ % 100 == 0:  # Occasionally check
            if c5_final <= 0.3807:
                break

    return (refined_h.tolist(), c5_final, n_points)
