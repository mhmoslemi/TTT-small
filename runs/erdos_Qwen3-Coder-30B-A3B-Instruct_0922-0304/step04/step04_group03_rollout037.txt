import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    
    # Initialize h - use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5 and identify the worst overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Find the shift k that gives maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(overlap)
        max_k = max_k_idx - (n_points - 1)  # Convert from full correlation indexing to actual shift
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if max_k > 0:
                gradient[i] = -h[(i + max_k) % n_points]
            else:
                gradient[i] = -h[(i - max_k) % n_points]
            if max_k < 0:
                gradient[i] += h[(i - max_k) % n_points]
            else:
                gradient[i] += h[(i + max_k) % n_points]
        
        # Update h
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping
        if iter_num % 50 == 0 and iter_num > 0:
            if abs(current_c5 - best_c5) < 1e-6:
                patience -= 1
                if patience <= 0:
                    break
            else:
                patience = 50
    
    # Final refinement
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
