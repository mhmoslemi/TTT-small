import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute C5 value for given h"""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_iter = 10000
    
    # Initialize population using initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    else:
        # Create a simple initialization
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.01
    
    # Initialize population
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(pop_size):
        # Create perturbed version of base solution
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        p_best.append(h.copy())
        p_best_vals.append(compute_c5(h))
    
    # Initialize global best
    global_best_idx = np.argmin(p_best_vals)
    g_best = p_best[global_best_idx].copy()
    g_best_val = p_best_vals[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(pop_size):
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            current_val = compute_c5(particles[i])
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
            
            # Update global best
            if current_val < g_best_val:
                g_best = particles[i].copy()
                g_best_val = current_val
        
        # Early exit if target achieved
        if g_best_val <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(g_best)
    
    return (g_best.tolist(), final_c5, n_points)
