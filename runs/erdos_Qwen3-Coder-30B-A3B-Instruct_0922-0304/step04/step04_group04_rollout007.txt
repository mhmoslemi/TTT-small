import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    max_iter = 100
    
    # Generate basis functions: sin(kπx) and cos(kπx) for k=1..basis_size
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size))
    for i in range(1, basis_size+1):
        if i % 2 == 1:
            # Odd index: sine
            phi[:, (i-1)//2] = np.sin(i * np.pi * x / 2)
        else:
            # Even index: cosine
            phi[:, i//2 - 1] = np.cos(i * np.pi * x / 2)
    
    # Precompute integrals of basis functions
    int_phi = np.zeros(basis_size)
    for i in range(basis_size):
        int_phi[i] = np.trapz(phi[:, i], x)
    
    # Precompute kernel matrix K for convolution
    K = np.zeros((basis_size, basis_size, 2*n_points - 1))
    for i in range(basis_size):
        for j in range(basis_size):
            # This is ∫ φ_i(x) * φ_j(x+t) dx over x ∈ [0,2]
            # We'll do this efficiently in fourier domain
            for lag in range(2*n_points - 1):
                lag_idx = lag - (n_points - 1)  # shift lag to be -n+1 to n-1
                if lag_idx < 0:
                    # x in [0,2], so x+lag in [0,2+lag], so [0,2] is [0,2] intersected with [0,2+lag]
                    # In the integral: x: [0, 2-lag], integrand is [0,2] intersection [lag, 2]
                    # but that's getting complicated. Let's simplify: Use extension.
                    # Compute full convolution.
                    corr = np.correlate(phi[:, i], phi[:, j], mode='full')
                    K[i, j, lag] = corr[n_points - 1 + lag_idx]
                else:
                    corr = np.correlate(phi[:, i], phi[:, j], mode='full')
                    K[i, j, lag] = corr[n_points - 1 + lag_idx]
    
    # Initialize coefficients c based on initial construction if available
    c = Variable(basis_size)
    if 'initial_h_values' in globals():
        h_samples = np.array(initial_h_values)
        # Compute Fourier coefficients to approximate h_samples
        c0 = np.mean(h_samples)
        # Normalize the hypothesis to have sum = 1
        h_samples = h_samples * (n_points / 2) / np.sum(h_samples)
        # Solve for coefficients: h_samples = phi @ c
        # We have: h_samples = phi @ c + c0 * ones -> adjust c
        c_approx = np.linalg.lstsq(phi, h_samples, rcond=None)[0]
        # Initialize with these better coefficients
        c.value = c_approx
    else:
        # Initialize randomly
        c.value = np.random.rand(basis_size)
        c.value = c.value * 0.5  # Scale down
        
    # The objective is to minimize max_lag |sum_{i,j} z_ij * K_ij(lag)|
    # But first, let's integrate out the constraints with optimized approach
    # Use a convex model to optimize the support

    # Define z variables for McCormick relaxation
    z = []
    for i in range(basis_size):
        row = []
        for j in range(basis_size):
            zij = Variable()
            row.append(zij)
        z.append(row)

    # Symmetry: z_{i,j} = z_{j,i}, but we'll not enforce that directly.
    # We already know that basis is typically symmetric with regard to parameter.

    # Set up the full optimization problem
    # Objective: minimize max(overlap)
    # Constraint: ∫ h dx = 1, so ∫ (c0 + polynomial terms) dx = 1
    # c0 * 2 + sum(c_i * int_phi_i) = 1 => c0 = (1 - sum(c_i * int_phi_i)) / 2

    # Build two optimization problems:
    # 1. Given c, compute overlap kerformed functions
    # 2. Minimize that

    # Now we'll set up for a sequence of tries with structured search
    best_score = float('inf')
    best_coeffs = None
    best_h = None

    # Try various combinations, adjust basis size for problem
    # Better approach: Adam optimizer with exact gradient, but not possible
    # Let's try pattern search and coordinate descent through this convoluted space
    
    # Let's just use predefined randomized strategy on that domain.
    # Relax to optimization over reals using simple gradient-based approach

    # Since we know its a structured convex problem, use first solve directly

    coeffs_more = []
    
    # Systematically evaluate across various grid of coefficient settings
    steps = np.linspace(0.5, 2, 20)
    for i in range(20):
        c_trial = np.random.normal(scale=0.5, size=basis_size)
        c_trial = np.absolute(c_trial)
        c_trial = c_trial * 0.8 + 0.2  # make all > 0.2
        c_trial = c_trial / np.sum(c_trial) * (n_points / 2)  # Renormalize
        # Try using the best candidate as starting point
        v = np.random.rand(basis_size)
        v = v / np.sum(v)
        c_trial = v * 0.8 + 0.1
        coeffs_more.append(c_trial)
    
    # Direct optimization over what we have
    # The objective is a little tricky to exact match, 
    # Let's run 50 trials of structured and simple local searches in grid
    for iter_try in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate a new solution with randomization
        # But let’s try a coordinate descent-like method below:
        # Step 1: start with a random candidate
        if iter_try == 0 and 'initial_h_values' in globals():
            h_trial = np.array(initial_h_values)
            # Re-normalize
            h_normalized = h_trial * (n_points / 2) / np.sum(h_trial)
            # Convert to coefficients:
            c_coeffs = np.linalg.lstsq(phi, h_normalized, rcond=None)[0]
            c.value = c_coeffs
        else:
            # Random init
            c.value = np.random.rand(basis_size) * 0.2 + 0.1
            
        # Try iterative improvement - simple descent step with projection
        for step in range(10):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute h from c
            h = np.array([phi[i, :].dot(c.value) for i in range(n_points)])
            h = np.clip(h, 0, 1)
            # Renormalize
            h = h * (n_points / 2) / np.sum(h)
            
            # Simple gradient estimation: just finite differences
            # For proximal step, let’s update gradient for c
            # partition the problem using ad hoc closed form
            c_current = c.value.copy()
            
            # Evaluate C5
            overlap = correlate(h, 1 - h, mode="full")
            score = np.max(overlap) * dx
            
            # Not fully correct but let's use a coarse step
            # Let's use SGD style step
            # Proximal step: Take random gradient direction
            grad = np.random.normal(0, 0.05, basis_size)
            
            new_c = c_current + grad
            new_c = np.clip(new_c, 0, 10)
            
            c.value = new_c
        
        # Maybe after a few grow phase, return solution
        h = np.array([phi[i, :].dot(c.value) for i in range(n_points)])
        h = np.clip(h, 0, 1)
        h_norm = h * (n_points / 2) / np.sum(h)
        overlap = correlate(h_norm, 1 - h_norm, mode="full")
        score = np.max(overlap) * dx
        
        if score < best_score:
            best_score = score
            best_coeffs = c.value.copy()
            best_h = h_norm.copy()
            
        # Early exit if a good enough value is found.
        if best_score <= 0.3807:
            break
            
    if best_coeffs is None:
        # Final fallback: try evaluating the original heuristic solution directly
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.zeros(n_points)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        
        overlap = correlate(h, 1 - h, mode="full")
        best_score = np.max(overlap) * dx
    
    # Final check and validation
    if best_h is not None:
        eval_check = evaluate_c5(best_h, dx)
        assert abs(eval_check - best_score) < 1e-3, "Mismatch in check"
    
    return (best_h.tolist(), best_score, n_points)
