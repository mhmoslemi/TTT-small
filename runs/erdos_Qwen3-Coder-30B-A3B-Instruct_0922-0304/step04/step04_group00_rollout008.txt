import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Adjust to meet the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Track best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Save original value
            original_h_i = h[i]
            
            # Compute current contribution to C5
            old_overlap = correlate(h, 1 - h, mode="full")
            old_max = np.max(old_overlap)
            
            # Try small perturbations
            best_h_i = original_h_i
            best_local_c5 = old_max
            
            # Test a few candidate values
            candidates = []
            if original_h_i > 0.01:
                candidates.append(original_h_i - 0.01)
            if original_h_i < 0.99:
                candidates.append(original_h_i + 0.01)
            candidates.append(max(0, min(1, original_h_i)))
            
            # Try different step sizes
            for step in [0.001, 0.005, 0.01, 0.02]:
                if original_h_i + step <= 1:
                    candidates.append(original_h_i + step)
                if original_h_i - step >= 0:
                    candidates.append(original_h_i - step)
            
            for candidate in candidates:
                h[i] = candidate
                # Ensure the sum constraint
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                h = np.clip(h, 0, 1)
                
                # Check if we improved the C5 value
                new_c5 = compute_c5(h, dx)
                if new_c5 < best_local_c5:
                    best_local_c5 = new_c5
                    best_h_i = candidate
                    
            # Final update with best value
            h[i] = best_h_i
            # Re-normalize
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            h = np.clip(h, 0, 1)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
