import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using gradient-based method with adaptive step size
    max_iter = 20000
    adaptive_step = 0.003
    step_reduction_factor = 0.9
    step_reduction_every = 200
    perturbation_every = 100
    max_perturbation = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using finite differences
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient estimate
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # For each point, approximate gradient
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step size adaptively
        if iteration % step_reduction_every == 0 and iteration > 0:
            adaptive_step *= step_reduction_factor
            
        # Update h
        h = h - adaptive_step * grad
        
        # Ensure h remains in bounds
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Random perturbation to escape local minima
        if iteration % perturbation_every == 0 and iteration > 0:
            noise = np.random.normal(0, max_perturbation, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
