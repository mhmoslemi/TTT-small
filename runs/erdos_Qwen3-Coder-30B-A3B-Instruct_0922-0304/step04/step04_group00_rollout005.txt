import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
   overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points) * 0.5
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise optimization with normalization
    step_size = 0.01
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise updates
        for i in range(n_points):
            # Compute gradient estimate by finite differences
            h_plus = h.copy()
            h_minus = h.copy()
            
            # Small perturbations
            perturb = step_size
            if h[i] + perturb <= 1:
                h_plus[i] = h[i] + perturb
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            else:
                h_plus[i] = 1.0
                
            if h[i] - perturb >= 0:
                h_minus[i] = h[i] - perturb
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            else:
                h_minus[i] = 0.0
            
            # Evaluate the objective at nearby points
            c5_plus = compute_c5(h_plus, dx) if h[i] + perturb <= 1 else float('inf')
            c5_minus = compute_c5(h_minus, dx) if h[i] - perturb >= 0 else float('inf')
            
            # Update if better
            if c5_plus < c5_minus and c5_plus < current_c5:
                h[i] = min(1.0, h[i] + step_size)
            elif c5_minus < c5_plus and c5_minus < current_c5:
                h[i] = max(0.0, h[i] - step_size)
            
            # Re-normalize to maintain integral constraint
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
    
    # Ensure final solution is valid
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
