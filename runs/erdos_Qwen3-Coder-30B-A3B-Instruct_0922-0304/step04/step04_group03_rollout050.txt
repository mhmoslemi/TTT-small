import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    max_overlap = 0
    for k in range(2*n - 1):
        overlap = 0
        for i in range(n):
            j = i + k
            if j < n:
                overlap += h[i] * (1 - h[j]) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = np.clip(h_start, 0, 1)
    else:
        # Create a smooth initial guess
        x = np.linspace(0, 2, n_points)
        h_start = np.exp(-10 * (x - 1)**2)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to satisfy integral constraint
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Define objective function for optimization
    def objective(params):
        if time.time() - start_time > budget_s - 1:
            return float('inf')
            
        # params: [a, b, c, d] for interval positions
        a, b, c, d = params
        h = np.zeros(n_points)
        
        # Define intervals in terms of indices
        i1_start = max(0, int(a * n_points / 2))
        i1_end = min(n_points, int((a + b) * n_points / 2))
        i2_start = max(0, int((a + b + c) * n_points / 2))
        i2_end = min(n_points, int((a + b + c + d) * n_points / 2))
        
        if i1_end > i1_start:
            h[i1_start:i1_end] = 1.0
        if i2_end > i2_start:
            h[i2_start:i2_end] = 1.0
            
        # Normalize to unit integral
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraint function to ensure valid intervals
    def constraint_func(params):
        a, b, c, d = params
        if a < 0 or a > 1 or b < 0 or b > 1 or c < 0 or c > 1 or d < 0 or d > 1:
            return -1e-6
        if a + b + c + d > 2:
            return 2 - (a + b + c + d)
        return 1e-6
    
    # Initial guess
    a, b, c, d = 0.2, 0.3, 0.2, 0.3  # start with 4 segments summing to 1
    
    # Optimize with bounds and constraints
    bounds = [(0, 1), (0, 1), (0, 1), (0, 1)]
    constraints = {'type': 'ineq', 'fun': constraint_func}
    
    # Use Nelder-Mead method which handles non-smooth objectives well
    result = minimize(objective, [a, b, c, d], method='Nelder-Mead', 
                      bounds=bounds, constraints=constraints, 
                      options={'maxiter': 5000, 'adaptive': True})
    
    # Reconstruct final h
    a, b, c, d = result.x
    h = np.zeros(n_points)
    
    i1_start = max(0, int(a * n_points / 2))
    i1_end = min(n_points, int((a + b) * n_points / 2))
    i2_start = max(0, int((a + b + c) * n_points / 2))
    i2_end = min(n_points, int((a + b + c + d) * n_points / 2))
    
    if i1_end > i1_start:
        h[i1_start:i1_end] = 1.0
    if i2_end > i2_start:
        h[i2_start:i2_end] = 1.0
        
    h = h * (n_points / 2) / np.sum(h)
    c5_bound = evaluate_c5(h, dx)
    
    # If we're close to the target and trying for a better solution, do another round
    if c5_bound > 0.3807:
        # Iteratively refine with more elaborate search for a few seconds
        best_h = h.copy()
        best_c5 = c5_bound
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Small perturbation
            delta = np.random.normal(0, 0.01, 4)
            new_params = np.clip(result.x + delta, 0, 1)
            new_params[3] = 1.0 - new_params[0] - new_params[1] - new_params[2]
            
            c5 = objective(new_params)
            if c5 < best_c5:
                best_c5 = c5
                a = new_params[0]
                b = new_params[1]
                c = new_params[2]
                d = new_params[3]
                best_h = np.zeros(n_points)
                i1_start = max(0, int(a * n_points / 2))
                i1_end = min(n_points, int((a + b) * n_points / 2))
                i2_start = max(0, int((a + b + c) * n_points / 2))
                i2_end = min(n_points, int((a + b + c + d) * n_points / 2))
                
                if i1_end > i1_start:
                    best_h[i1_start:i1_end] = 1.0
                if i2_end > i2_start:
                    best_h[i2_start:i2_end] = 1.0
                best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
