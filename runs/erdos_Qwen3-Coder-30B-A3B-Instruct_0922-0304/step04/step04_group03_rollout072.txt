import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        i0 = 0
        i1 = int(np.floor(w / dx))
        # We want to try to find optimal d by grid search
        best_c5 = float('inf')
        best_h = None
        
        # Try different placements of the second interval
        d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
        for d in d_candidates:
            if d + w > 2.0:
                continue
            h_temp = np.zeros(n_points)
            i2 = int(np.ceil((w + d) / dx))
            i3 = int(np.floor((w + d + w) / dx))
            if i2 > i1:  # no overlap between intervals
                h_temp[i0:i1+1] = 1
                h_temp[i2:i3+1] = 1
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                # Compute overlap
                overlap = correlate(h_temp, 1 - h_temp, mode="full")
                max_overlap = np.max(overlap) * dx
                if max_overlap < best_c5:
                    best_c5 = max_overlap
                    best_h = h_temp.copy()
        
        h = best_h if best_h is not None else h
    
    # Try local optimization around the best interval construction
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Grid search with finer resolution around best so far
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.001)
    for d in d_candidates:
        if d + 0.5 > 2.0:
            continue
        h_temp = np.zeros(n_points)
        i0 = 0
        i1 = int(np.floor(0.5 / dx))
        i2 = int(np.ceil((0.5 + d) / dx))
        i3 = int(np.floor((1.0 + d) / dx))
        if i2 > i1:  # no overlap between intervals
            h_temp[i0:i1+1] = 1
            h_temp[i2:i3+1] = 1
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            # Compute overlap
            overlap = correlate(h_temp, 1 - h_temp, mode="full")
            max_overlap = np.max(overlap) * dx
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h_temp.copy()
    
    # Coarse PSO to further improve solution
    n_particles = 20
    max_iter = 200
    w_start = 0.9
    w_end = 0.4
    c_p = 1.5
    c_g = 1.5
    v_max = 0.1
    
    # Initialize particles with variations of the best solution
    particles = []
    velocities = []
    
    for i in range(n_particles):
        if i == 0:
            h_p = best_h.copy()
        else:
            # Add small noise
            h_p = best_h + np.random.normal(0, 0.05, n_points)
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (n_points / 2) / np.sum(h_p)
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Evaluate initial particles
    pbest = [p.copy() for p in particles]
    pbest_vals = [float('inf')] * n_particles
    
    for i in range(n_particles):
        overlap = correlate(particles[i], 1 - particles[i], mode="full")
        pbest_vals[i] = np.max(overlap) * dx
    
    # Track global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                            c_p * r_p * (pbest[i] - particles[i]) + 
                            c_g * r_g * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update particle
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to satisfy constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new solution
            overlap = correlate(particles[i], 1 - particles[i], mode="full")
            current_val = np.max(overlap) * dx
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best if needed
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    
        # Early exit if target achieved
        if gbest_val <= 0.3807:
            break
            
    # Final validation
    overlap = correlate(gbest, 1 - gbest, mode="full")
    final_c5 = np.max(overlap) * dx
    
    return (gbest.tolist(), final_c5, n_points)
