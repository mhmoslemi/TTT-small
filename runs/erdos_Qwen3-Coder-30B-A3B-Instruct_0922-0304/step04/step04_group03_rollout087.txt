import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available, otherwise create a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization
    max_iter = 1000
    learning_rate = 0.01
    patience = 20
    best_val = evaluate_c5(h, dx)
    best_h = h.copy()
    no_improvement_count = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_full)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for current maximum overlap
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Add contribution from h[i] * (1 - h[(i+k) % n_points])
            j = (i + max_k) % n_points
            grad[i] += (1 - h[j]) * dx
            
            # Subtract contribution from h[(i-k) % n_points] * (1 - h[i])
            j = (i - max_k) % n_points
            grad[i] -= h[j] * dx
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        current_val = evaluate_c5(h, dx)
        
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
            no_improvement_count = 0
            if best_val <= 0.3807:
                break
        else:
            no_improvement_count += 1
            
        # Adjust learning rate if no improvement for a while
        if no_improvement_count > patience:
            learning_rate *= 0.9
            no_improvement_count = 0
            
        # Early exit if no improvement seen for too long
        if no_improvement_count > 50:
            break
    
    return (best_h.tolist(), best_val, n_points)
