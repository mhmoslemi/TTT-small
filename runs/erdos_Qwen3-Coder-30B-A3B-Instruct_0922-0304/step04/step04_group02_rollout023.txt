import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM Parameters
    rho = 1.0
    max_iter = 2000
    epsilon = 1e-4
    
    # Initialize variables
    x = h.copy()
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM Iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update x (minimize augmented Lagrangian)
        # This is a simplified version that does not exactly solve the subproblem
        # but provides a reasonable update
        x_old = x.copy()
        # Minimize: ||x - z + u||^2 + rho * (sum of constraints)
        # For simplicity: take gradient step on the overlap term
        
        # Compute gradient approximation (simplified)
        overlap = correlate(x, 1 - x, mode="full")
        max_idx = np.argmax(overlap)
        grad = np.zeros(n_points)
        
        # Approximate gradient using finite differences
        eps = 1e-4
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = np.clip(x_plus, 0, 1)
            x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
            
            overlap_plus = correlate(x_plus, 1 - x_plus, mode="full")
            grad[i] = (overlap_plus[max_idx] - overlap[max_idx]) / eps if eps > 0 else 0
            
        # Update x using gradient descent step
        step_size = 1e-3
        x = x - step_size * grad
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        
        # Update z (proximal operator with constraint on sum)
        z = x + u
        z = np.clip(z, 0, 1)
        z = z * (n_points / 2) / np.sum(z)
        
        # Update u (dual variable)
        u = u + x - z
        
        # Check convergence (simple version)
        if np.linalg.norm(x - x_old) < epsilon:
            break
            
        # Check and store best result
        if iteration % 10 == 0:
            current_c5 = evaluate_c5(x, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
