import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=0.02):
    """Compute the maximum overlap integral for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize population
    n_particles = 50
    particles = []
    velocities = []
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (n_points / 2) / np.sum(base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    # Create initial population
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track best solution
    global_best_h = None
    global_best_c5 = float('inf')
    
    # PSO parameters
    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.1
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        particle_c5s = [compute_c5(particle, dx) for particle in particles]
        
        # Update personal bests and global best
        for i, (particle, c5) in enumerate(zip(particles, particle_c5s)):
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = particle.copy()
                
        # Early exit if target achieved
        if global_best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            pbest = particles[i]  # For current particle
            if particle_c5s[i] < compute_c5(pbest, dx):
                pbest = particle.copy()
                
            velocity = (omega * velocity + 
                       c1 * r1 * (pbest - particle) + 
                       c2 * r2 * (global_best_h - particle))
            
            # Clamp velocity
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            
            # Update position
            particle += velocity
            
            # Enforce bounds
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Store updated values
            particles[i] = particle
            velocities[i] = velocity
    
    # Final validation
    final_c5 = compute_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
