```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Sequential convex programming approach
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Linearize the problem using current h estimate
        # Create a convex problem with linearized constraints
        h_current = h.copy()
        
        # Define the optimization variables
        h_vars = cp.Variable(n_points)
        s = cp.Variable()
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Linearized constraints for overlapping terms
        for k in range(-n_points+1, n_points):
            # Compute the linearization at h_current
            if k >= 0:
                # For positive shifts: h[i] * (1 - h[i+k]) 
                # Linearized as: h_current[i] * (1 - h_current[i+k]) + (h[i] - h_current[i]) - h[i+k] * h_current[i]
                linear_term = 0
                for i in range(n_points):
                    if i + k < n_points:
                        linear_term += h_current[i] * (1 - h_current[i+k]) * dx
                        linear_term += (h_vars[i] - h_current[i]) * dx
                        linear_term -= h_current[i] * h_vars[i+k] * dx
                    else:
                        linear_term += h_current[i] * (1 - h_current[i+k-n_points]) * dx
                        linear_term += (h_vars[i] - h_current[i]) * dx
                        linear_term -= h_current[i] * h_vars[i+k-n_points] * dx
            else:
                # For negative shifts: h[i] * (1 - h[i+k]) 
                k_abs = -k
                linear_term = 0
                for i in range(n_points):
                    if i - k_abs >= 0:
                        linear_term += h_current[i] * (1 - h_current[i-k_abs]) * dx
                        linear_term += (h_vars[i] - h_current[i]) * dx
                        linear_term -= h_current[i] * h_vars[i-k_abs] * dx
                    else:
                        linear_term += h_current[i] * (1 - h_current[i-k_abs+n_points]) * dx
                        linear_term += (h_vars[i] - h_current[i]) * dx
                        linear_term -= h_current[i] * h_vars[i-k_abs+n_points] * dx
                        
            constraints.append(s >= linear_term)
            
        # Bounds on h
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Objective
        objective = cp.Minimize(s)
        
        # Form the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = np.array(h_vars.value).flatten()
                h = np.clip(h_solution, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient descent if LP fails
            step = 0.002
            h_temp = h.copy()
            for _ in range(20):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 100 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
