import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    max_iter = 2000
    tolerance = 1e-4

    x = np.linspace(0, 2, n_points, endpoint=False)
    fourier_basis = []
    basis_integrals = []
    
    for k in range(n_basis):
        if k % 2 == 0:
            # Cosine
            phi = np.cos(np.pi * k * x / 2)
        else:
            # Sine
            phi = np.sin(np.pi * k * x / 2)
        fourier_basis.append(phi)
        basis_integrals.append(np.sum(phi) * dx)
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback: simple step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Project onto Fourier basis
    coeffs = []
    for i in range(n_basis):
        coeffs.append(np.sum(h_init * fourier_basis[i]) * dx)
    
    # Newton-Raphson loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Set up convex optimization problem
        c = cp.Variable(n_basis)
        d = cp.Variable(n_basis * n_basis)
        
        # McCormick envelopes
        # d[i*n_basis + j] represents c[i]*c[j] where i,j = 0..n_basis-1
        constraints = []
        
        # Add integral constraint
        constraints.append(cp.sum(cp.multiply(c, basis_integrals)) == 1)
        
        # Add bounds on coefficients
        for i in range(n_basis):
            constraints.append(c[i] >= 0)
            constraints.append(c[i] <= 1)
        
        # Add McCormick envelopes for d[i*n_basis + j] = c[i] * c[j]
        for i in range(n_basis):
            for j in range(n_basis):
                idx = i * n_basis + j
                # y = x1 * x2
                # McCormick envelopes:
                # y >= x1 + x2 - 1
                # y <= x1
                # y <= x2
                # y >= 0
                constraints.append(d[idx] >= c[i] + c[j] - 1)
                constraints.append(d[idx] <= c[i])
                constraints.append(d[idx] <= c[j])
                constraints.append(d[idx] >= 0)
        
        # Compute overlap terms using Fourier properties
        # We approximate:
        # sum_k integral h(x) * (1 - h(x+k)) dx
        # = sum_k sum_i sum_j c_i c_j integral phi_i(x) * (1 - phi_j(x+k)) dx
        # = sum_k sum_i sum_j c_i c_j [ integral phi_i(x)*1 dx - integral phi_i(x)*phi_j(x+k) dx ]
        
        # Let's make it simple: assuming we can get the full correlation pattern
        # But it's easier to compute the overlap directly and minimize.
        # Let's just set up the main optimization
        
        # Value to minimize: max_k over overlap from full convolution-like operation
        
        # Use Lagrangian approach for solving to prevent overfitting
        
        # We'll just solve using CVXPY with quadratic objective
        # and add penalties for feasibility + integral
        
        # Influence kernel coefficients
        # In the discretization: 
        # For computing h(x) (which is sum c_k phi_k(x)), and its shifted version,
        # overlap is integral [ h(x)(1 - h(x+k)) ] from 0 to 2.
        # Let's calculate form the discrete overlap instead:

        h_current = np.zeros(n_points)

        h_current = np.clip(h_current, 0, 1)

        c5 = evaluate_c5(h_current, dx)

        try:
            # Solver variables
            c = cp.Variable(n_basis)

                    ct.append(d[idx] <= c[j])
                    ct.append(d[idx] >= 0)

            # Try to approximate min max penalty from computing all overlaps
            # Instead, form the vector of overlap terms and minimize max in a roundabout way
            # We don't solve directly the C5 integral in terms of coeffs because that would be very complex
            # Let's try to reframe: our current approach is optimized to directly get a better h(x) based on initial value
            
            # Try to recompute from previous coeffs but let the problem be in a better shape
            h_old = np.zeros(n_points)

                h_old += coeffs[k] * fourier_basis[k]
            h_old = np.clip(h_old, 0, 1)
            h_old = h_old * (n_points / 2) / np.sum(h_old)

            # Normalize to integral constraint
            integral_sum = sum(new_coeffs[i] * basis_integrals[i] for i in range(n_basis))
            new_coeffs = [new_coeffs[i] / integral_sum * 1.0 for i in range(n_basis)]
            
            # Create new h vector
            h_new = np.zeros(n_points)
            for i in range(n_basis):
                h_new += new_coeffs[i] * fourier_basis[i]
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate overlap
            c5_new = evaluate_c5(h_new, dx)
            
            # If improvement, accept
            if c5_new < c5:
                coeffs = new_coeffs.copy()
                c5 = c5_new
                
        except Exception as e:
            # Fall back to previous solution
            pass
        
        # Every few iterations, update and reevaluate
        if iter_count % 50 == 0:

            for i in range(n_basis):

            if c5 <= 0.3807 and c5 < 0.381277:

    # Final computation
    h_final = np.zeros(n_points)
    for i in range(n_basis):
        h_final += coeffs[i] * fourier_basis[i]
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), c5, n_points)
