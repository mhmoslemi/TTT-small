import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    # Compute the overlap integral with proper scaling
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    n_particles = 50
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Initialize particle position
        if 'initial_h_values' in globals():
            pos = np.array(initial_h_values)
            # Add slight randomization
            pos += np.random.normal(0, 0.01, n_points)

        particles.append(pos.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        pbest.append(pos.copy())
        pbest_vals.append(compute_c5(pos))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

    omega = 0.7

            particles[i] += velocities[i]

            particles[i] = np.clip(particles[i], 0, 1)

            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
