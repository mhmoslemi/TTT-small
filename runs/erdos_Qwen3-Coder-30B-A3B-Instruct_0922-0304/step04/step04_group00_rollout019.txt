import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_overlap_shifts(h):
        """Return overlaps for all possible shifts"""
        overlap = correlate(h, 1 - h, mode="full")
        # Only consider valid shifts (0 to 2*dx, 2*dx to 4*dx, ..., up to 2*dx*(n_points-1))
        # In the full correlation, we actually have n_points*2 - 1 overlaps
        # 0 shift corresponds to n_points - 1 in the full array
        # Which overlap represents shift k? shift k corresponds to overlap at index n_points - 1 + k
        # For shift -k, it's at n_points - 1 - k
        
        # NumPy correlate returns correlation (h, 1-h) for all time lags
        # The shift is k where k goes from -(n_points-1) to (n_points-1)
        # Index of shift k is n_points-1 + k
        # We only want k = 0 to (n_points-1)
        # We want maximal k to be some positive range of overlaps
        # Since we want overlaps over [0,2] with dx = 2/n_points
        # Takes shift k = 0, 1, ..., n_points-1 as corresponding to the shift of 0*dx, 1*dx, ..., (n_points-1)*dx
        # We start at the full width of 2*dx which means total discretization of N points.
        # So, for overlap corresponding to (x+k) where k=0 to n_points-1, 
        # The actual shift vector goes from 0*k*dx to (n_points-1)*k*dx
        # Total shift span corresponds to 0*dx * (n_points) to (n_points-1)*dx * (n_points) = 0 to 2*dx
        # The shift indices correspond to k=0 to n_points-1 and span 1 to n_points values
        # But they differ in actual computation, so we take around indices from n_points-1 down
        # Let me re-think
        
        # corr at index i corresponds to shift = (i - (n_points-1)) * dx
        # We want shifts in [0, 2*dx]
        # shift = 0 corresponds to i = n_points-1
        # shift=1 corresponds to i = n_points
        # shift=n_points-1 corresponds to i = 2*n_points - 2
        # Within a full overlap array, if there are 2*N-1 elements:
        # Index 0 to N-2 corresponds to shifts -N+1 to -1
        # Index N-1 corresponds to shift 0
        # Index N to 2*N-2 corresponds to shifts 1 to N-1
        
        shifts = np.arange(n_points)
        return overlap[n_points-1:n_points-1+shifts.shape[0]] * dx

    def solve_convex_subproblem(h_current):
        # Use CVXPY for the convex subproblem
        # Variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)  # sum = N/2 = 35
        
        # Add the convergence constraint: for current max overlap, add a constraint that it can't exceed current value
        # We'll find the shift with the highest overlap (but it's approximate with the current h)
        # Actually according to cutting plane: we want to fix the issue and stop convergence
        
        # We have some comparison point for now (h_via_overlap = overlap)
        # Heuristic: maximize linear relaxation: we assume current_difference (from max overlap) somehow
        # Let's just carry using the direction: use current overlap to get inequalities from fixed shifts
        
        # Build linear relaxations based on extreme shifts
        
        # Linearize: approximate max overlap with a set of constraints
        # But since we already computed the overlaps, just get the maximum shift (we don't need to solve another optimization yet)
        # Simplified approach - try to improve by cutting planes using all valid shifts at the moment, 
        # but not add *every* constraint - just some critical shifts
        
        # But we need to be very efficient. Since we're adding constraints and solving the optimization 
        # This is inefficient when we have too many constraints in the convex solver. Need to be smart.
        # As ε gets small, we want a focused local optimization, so we can reduce to linear optimization
        
        # Instead of a naïve cutting plane with all constraints, we do this:
        # 1. At iteration t, we have a set of constraints from past iterations  
        # 2. The current objective is to minimize max_overs in the overlap vector
        # Let’s instead keep tracking best overlap and at biggest shift not quiet yet satisfiable
        
        return h, constraints

    def minimize_with_cuts(h_init, max_iters=100):
        # Build up constraints gradually 
        alloc_h = h_init.copy()
        
        # Use a duality approach to reduce cognitive loading.
        # We try cutting planes based on a greedy principle
        # 1. Start with good initial h (already normalized, valid)
        # 2. At each iteration, find worst case shift in the current h
        # 3. Add constraint on that shift
        # 4. Solve convex subproblem (minimize for C subject to existing constraints + new constraint)   
        # 5. Project back to feasible set
        
        # Start basic/simple
        # Replace the .copy() references to be safe
        
        # At each step we compute the max overlap
        # The more constraint we add, the more constrained it becomes.
        
        # Normalize once at start
        normalizer = np.sum(alloc_h)
        if abs(normalizer - n_points / 2) > 1e-10:
            alloc_h = alloc_h * (n_points / 2) / normalizer
        alloc_h = np.clip(alloc_h, 0, 1)
        
        best_h = alloc_h.copy()
        best_c5 = compute_c5(alloc_h)
        
        # Track how we're doing
        last_improvement = 0  # iteration of last improvement (from best_c5)
        max_no_improvement = 20
        
        # Multiplier impose 1e-3 minimum for constraints to be active
        base_epsilon = 1e-4
        
        for it in range(max_iters):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute the current overlaps 
            overlaps = compute_overlap_shifts(alloc_h)
            
            # Find the current maximum shift
            max_shift_idx = np.argmax(overlaps)  # The k index in [0, n_points-1] such that overlap[k] is max
            max_overlap = overlaps[max_shift_idx]
            
            # Adjust this to suit our linear constraint
            # We want to make sure max_overlap doesn't get larger than a value
            # This is somewhat akin to a cutting plane: current shift k has overlap max_overlap_value.
            # We want to constrain future h values to not allow any shift >= max_overlap - epsilon
            # Thus we can add a single constraint:
            # Destroy 1 consistent term and add constraint from that shift_k
            
            # Most natural for CVXPY to do projected subgradient descent or using bounds, but we want to leverage convex programming
            # But that can be complex - instead, we do many small update steps based on descent approximations
            
            # Start simple: replace the current method entirely by linear programming approach with convex-safe cutting plane
            
            # Use `cvxpy` to form one convex problem:
            # min C
            # s.t.  h element-wise ∈ [0, 1]
            #       sum(h) = n_points / 2
            #       ∀s ∈ Shifts_Used: h * (1-h_shifted[s]) <= C - base_epsilon  [or add the specific constraint]
            
            # Add only *one* constraint per iteration initially at max hardest shift
            
            # Now formulate constraints efficiently
            # Want to solve: minimize C such that:
            # for each shift k:
            #   ∫ h(x) * (1 - h(x+k)) dx <= C
            
            # We have the discretized version of the integral: h_T * (1 - h_shifted)
            
            # YIELD the method: Use cvxpy with constraints
            
            # Reinitialize problem to solve
            alloc_h_var = cp.Variable(n_points)
            constraints = [alloc_h_var >= 0, alloc_h_var <= 1]
            constraints.append(cp.sum(alloc_h_var) == n_points / 2)
            
            # At least 1 new constraint:
            # Extract from current h the overlap vector for shift k = max_shift_idx
            # shift k = max_shift_idx corresponds to index k = n_points - 1 + k in np.correlate output
            # The shift we're interested in specifically is the left shift vector.
            # So shift k is cast obtain:
            # In overlap outputs: index = max_shift_idx + n_points - 1 for shift k
            # But others always be evaluated with full convolution
            
            # Build the shifted version of h
            shift_k = max_shift_idx
            shifted_h = np.zeros(n_points)
            if shift_k == 0:
                shifted_h = alloc_h.copy()
            else:
                shifted_h[shift_k:] = alloc_h[:-shift_k]
            
            # Linear term in half of dot product: h^T * (1-shifted_h) = sum(h* (1-shifted_h)) over 0..N-1
            
            constraint = cp.sum(cp.multiply(alloc_h_var, 1 - shifted_h)) <= max_overlap - 1e-4
            # Note: fixing epsilon to be stable
            
            constraints.append(constraint)
            
            # Objective
            objective = cp.Minimize(1)  # Simplification later, we'll take the max later
            
            prob = cp.Problem(objective, constraints)
            
            try:
                # Use ECOS or SCS as fast mehod
                prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
                # Prob solution is a feasibility check
                # Variable allocation : alloc_h_var.value gives us solution
                
                sol_h = alloc_h_var.value
                if sol_h is not None:
                    sol_h = np.array(sol_h)
                    # Re-normalize and clip
                    sol_h = np.clip(sol_h, 0, 1)
                    s_norm = np.sum(sol_h)
                    if abs(s_norm - n_points / 2) > 1e-10:
                        sol_h = sol_h * (n_points / 2) / s_norm
                    
                    h_test = sol_h
                    test_c5 = compute_c5(h_test)
                    if test_c5 < best_c5:
                        best_c5 = test_c5
                        best_h = h_test.copy()
                        last_improvement = it
                        
                    alloc_h = h_test
                    
                    # Early return check
                    if best_c5 <= 0.3807:
                        break
                        
                    # Good check
                    if it - last_improvement > max_no_improvement:
                        # We're in an unconverged or stabilized state, try a restart
                        # This restart should try more aggressive discretization or initialization variant
                        # Since it might be non-smooth
                        # Random perturb + rebalance  
                        alloc_h = np.copy(best_h)
                        # Add small noise
                        noise = np.random.normal(0, 0.01, n_points)
                        alloc_h += noise
                        alloc_h = np.clip(alloc_h, 0, 1)
                        s_norm = np.sum(alloc_h)
                        alloc_h = alloc_h * (n_points / 2) / s_norm
                        
            except Exception as e:
                # If CVXPY fails, reset with current h + noise
                alloc_h = np.copy(best_h)
                noise = np.random.normal(0, 0.05, n_points)
                alloc_h += noise
                alloc_h = np.clip(alloc_h, 0, 1)
                s_norm = np.sum(alloc_h)
                alloc_h = alloc_h * (n_points / 2) / s_norm

        return best_h, best_c5
    
    # Start the iterative approach
    final_h, final_c5 = minimize_with_cuts(h, max_iters=300)
    
    return (final_h.tolist(), final_c5, n_points)
