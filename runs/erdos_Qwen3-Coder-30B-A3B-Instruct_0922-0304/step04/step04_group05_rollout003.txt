import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Line search parameters
    alpha = 0.1
    decay = 0.5
    min_alpha = 1e-6
    
    # Iterative improvement using sequential convex programming
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradients and Hessian approximation
        # We'll simplify by approximating gradient of overlap at current h
        h_current = h.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Evaluate overlaps
        overlap = correlate(h_current, 1 - h_current, mode="full")
        c5 = np.max(overlap) * dx
        
        # Simple gradient approximation (simplified version)
        # For this approach, we will compute linear approximation 
        # by finite differences in a few directions
        
        grad_h = np.zeros(n_points)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            
            # Ensure constraints
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad_h[i] = (np.max(overlap_plus) - np.max(overlap_minus)) * dx / (2 * epsilon)
            
        # Compute a descent direction via convex quadratic programming
        # Minimize g^T * delta + 0.5 * delta^T * H * delta
        # subject to linear constraints
        
        # Approximate Hessian: diag(1e-6) to regularize
        H = np.eye(n_points) * 1e-6
        
        # Construct the convex optimization problem
        delta = cp.Variable(n_points)
        
        # Objective: quadratic approximation
        obj = cp.Minimize(cp.matmul(grad_h.T, delta) + 0.5 * cp.quad_form(delta, H))
        
        # Constraints
        constraints = [
            h_current + delta >= 0,
            h_current + delta <= 1,
            cp.sum(delta) == 0  # Ensures sum remains constant
        ]
        
        # Form and solve the convex problem
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                delta_val = delta.value
                if np.any(np.isnan(delta_val)) or np.any(np.isinf(delta_val)):
                    # fallback for numerical issues
                    delta_val = -grad_h * alpha
                else:
                    delta_val = delta.value
            else:
                # Fallback to gradient descent step
                delta_val = -grad_h * alpha
        except:
            # Fallback when convex solve fails
            delta_val = -grad_h * alpha

        # Update h
        h_new = h_current + alpha * delta_val
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new function value
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept the update if it improves the objective
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            
            # Reset step size
            alpha = 0.1
        else:
            # Reduce step size if no improvement
            alpha *= decay
            if alpha < min_alpha:
                alpha = min_alpha

        # Early termination if target reached
        if best_c5 <= 0.3807:
            break

    # Final validation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
