import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a symmetric sawtooth pattern
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using gradient-based method with adaptive step size
    max_iter = 20000
    step_size = 0.005
    min_step_size = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        # Get current maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx

        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        # Update step size and h
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            step = step_size / grad_norm
            step = max(min_step_size, step)
        else:
            step = step_size
            
        h -= step * grad
        h = np.clip(h, 0, 1)
        
        # Maintain sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Decrease step size over time
        if iteration % 200 == 0 and step_size > min_step_size:
            step_size *= 0.9

        # Occasionally add noise to escape local optima
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
