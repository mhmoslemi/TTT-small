import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential convex programming approach
    max_iter = 1000
    alpha = 0.1
    mu = 1e-3
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation terms and gradients
        # We'll approximate the Hessian by using a simple quadratic model
        # A simple approach to handle nonlinearity is to linearize
        # Compute current overlaps
        C = correlate(h, 1 - h, mode="full") * dx
        max_k = np.argmax(C)
        max_overlap = C[max_k]
        
        # Compute gradient of max overlap with respect to h
        # Gradient approximated as finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            C_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (C_plus[max_k] - max_overlap) / eps
        
        # Formulate convex subproblem
        # Objective: minimize grad^T * delta + (1/2) * delta^T * H * delta
        # We use a diagonal approximation of Hessian (second order finite diff)
        H = np.eye(n_points) * 1e-6  # Regularization
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            C_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            C_minus = correlate(h_minus, 1 - h_minus, mode="full") * dx
            
            H[i, i] = (C_plus[max_k] - 2*max_overlap + C_minus[max_k]) / (eps*eps)
        
        # Add regularization
        H += np.eye(n_points)*1e-5
        
        # Solve the convex subproblem to find delta
        delta = np.zeros(n_points)
        try:
            # Use convex optimization to find delta
            delta_var = cp.Variable(n_points)
            objective = cp.Minimize(grad @ delta_var + 0.5 * cp.quad_form(delta_var, H))
            
            # Constraints
            constraints = [
                delta_var >= -h,
                delta_var <= 1 - h,
                cp.sum(delta_var) == 0
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == 'optimal':
                delta = delta_var.value
            else:
                # Fallback if solve fails, use first-order update
                delta = -alpha * grad
                delta = np.clip(delta, -h, 1-h)
        except:
            # Fallback to gradient descent
            delta = -alpha * grad
            delta = np.clip(delta, -h, 1-h)
        
        # Update h
        h_new = h + delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Adaptive step size damping if no improvement
        if new_c5 >= max_overlap:
            alpha *= 0.5
        else:
            alpha = min(alpha * 1.1, 0.5)
            
        h = h_new
        
        # Early termination check
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
