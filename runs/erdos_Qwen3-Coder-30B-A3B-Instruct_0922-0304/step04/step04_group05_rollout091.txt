import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_generations = 2000
    elite_size = population_size // 4
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        # Start with triangular function
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill rest with random feasible solutions
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Evolutionary strategy main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            fit = evaluate_c5(h, dx)
            fitness.append(fit)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Create offspring using mutation
        offspring = []
        
        # Elite individuals (best 25%) are preserved
        for i in range(elite_size):
            offspring.append(population[i].copy())
        
        # Generate new individuals
        while len(offspring) < population_size:
            # Select parent from top 50%
            parent_idx = np.random.randint(0, population_size // 2)
            parent = population[parent_idx]
            
            # Mutate
            mutation = np.random.normal(0, mutation_scale, n_points)
            child = parent + mutation
            child = project_to_feasible(child, n_points)
            
            offspring.append(child)
        
        # Update population
        population = offspring
        
        # Adaptive mutation scale
        if generation % 10 == 0:
            # Adjust mutation scale based on recent progress
            recent_improvement = fitness[0] - fitness[-1] if len(fitness) > 1 else 0
            if recent_improvement < 1e-6:
                mutation_scale *= 0.95  # Reduce if no improvement
            else:
                mutation_scale *= 1.01  # Increase if progress
            mutation_scale = np.clip(mutation_scale, 0.001, 0.5)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
