import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Initialize variables
    h_prev = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update h using proximal gradient step
        # Objective: minimize sum_k (h * (1-h_k)) over all shifts k
        # This is equivalent to solving: h = argmin_h (h^T * (1-h) + rho/2 * ||h-h_prev+u/rho||^2)
        # We approximate this with gradient descent
        
        # Compute overlap
        overlap = correlate(h, 1-h, mode="full")
        grad_h = np.zeros(n_points)
        
        # Approximate gradient from correlation
        for i in range(n_points):
            # Approximate derivative at each point
            grad_h[i] = overlap[n_points-1+i] - overlap[n_points-1-i] if i < n_points-1 else 0
            if i > 0:
                grad_h[i] += overlap[n_points-1+i] - overlap[n_points-1-i]
        
        # Update h (simple gradient descent)
        h_new = h - 0.01 * grad_h
        
        # Apply constraints: clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new) 
        
        # Update dual variable u
        u += rho * (h_new - h_prev)
        
        # Update h_prev
        h_prev = h_new
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h_new, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
