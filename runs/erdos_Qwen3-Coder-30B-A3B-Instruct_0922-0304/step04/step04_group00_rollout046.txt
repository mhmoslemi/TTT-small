import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_and_normalize(h, n_points, dx):
    """Ensure h is feasible and return C5"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Population-based optimization approach
    population_size = 50
    population = []
    
    # Initialize population
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h_candidate = np.array(initial_h_values)
            # Add small noise
            h_candidate += np.random.normal(0, 0.01, n_points)
        else:
            h_candidate = np.random.uniform(0, 1, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        population.append(h_candidate.copy())
    
    # GA parameters
    mutation_rate = 0.1
    crossover_rate = 1.0
    elite_count = 5
    
    # For tracking consecutive failures
    no_improvement_count = 0
    max_no_improvement = 50

    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate entire population
        fitness = []
        for individual in population:
            c5 = evaluate_and_normalize(individual, n_points, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_current_c5 = fitness[sorted_indices[0]]
        
        if best_current_c5 < best_c5:
            best_c5 = best_current_c5
            best_h = population[sorted_indices[0]].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected_indices = []
        tournament_size = 5
        for _ in range(population_size):
            tournament = random.sample(range(population_size), tournament_size)
            winner = min(tournament, key=lambda i: fitness[i])
            selected_indices.append(winner)
        
        # Create new population
        new_population = []
        
        # Elite individuals
        elite_indices = sorted_indices[:elite_count]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Parents selection
            p1_idx, p2_idx = random.sample(selected_indices, 2)
            p1, p2 = population[p1_idx], population[p2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.01)
                if random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.01)
            
            # Normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Local refinement for top individuals
        top_individuals = sorted_indices[:min(10, len(sorted_indices))]
        for idx in top_individuals:
            h_refined = population[idx].copy()
            def obj_func(h):
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                return evaluate_and_normalize(h, n_points, dx)
            
            # Local optimization
            try:
                result = minimize(
                    obj_func, 
                    h_refined, 
                    method='SLSQP', 
                    bounds=bounds,
                    options={'maxiter': 20, 'ftol': 1e-6}
                )
                if result.success:
                    h_refined = result.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            except:
                pass  # Continue if local search fails
            
            # Replace the current individual if better
            current_fitness = fitness[idx]
            refined_fitness = evaluate_and_normalize(h_refined, n_points, dx)
            if refined_fitness < current_fitness:
                population[idx] = h_refined.copy()
        
        # Update population
        population = new_population[:]
        
        # Perturbation if no improvement
        if no_improvement_count > max_no_improvement:
            for i in range(population_size):
                population[i] += np.random.normal(0, 0.02, n_points)
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] * (n_points / 2) / np.sum(population[i])
            no_improvement_count = 0

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
