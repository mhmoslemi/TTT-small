import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population_scores = []
    
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        population_scores.append(evaluate_c5(individual, dx))
    
    # Store best from initial population
    min_idx = np.argmin(population_scores)
    if population_scores[min_idx] < best_c5:
        best_c5 = population_scores[min_idx]
        best_h = population[min_idx].copy()
    
    # Helper for tournament selection
    def tournament_selection(population, scores, k=tournament_size):
        indices = np.random.choice(len(population), k)
        best_idx = indices[np.argmin([scores[i] for i in indices])]
        return population[best_idx].copy()
    
    # GA iterations
    for generation in range(num_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(population_scores)
        population = [population[i] for i in sorted_indices]
        population_scores = [population_scores[i] for i in sorted_indices]
        
        # Keep best and create new population
        new_population = [population[0].copy()]
        new_population_scores = [population_scores[0]]
        
        # Elitism: keep top 20%
        elite_size = population_size // 5
        for i in range(1, elite_size):
            new_population.append(population[i].copy())
            new_population_scores.append(population_scores[i])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, population_scores)
            parent2 = tournament_selection(population, population_scores)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mut_idx = np.random.randint(0, n_points)
                    child[mut_idx] = np.clip(child[mut_idx] + np.random.normal(0, 0.1), 0, 1)
                    
                # Renormalize
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                new_population_scores.append(evaluate_c5(child, dx))
        
        # Trim to right size if needed
        population = new_population[:population_size]
        population_scores = new_population_scores[:population_size]
        
        # Update best solution
        min_idx = np.argmin(population_scores)
        if population_scores[min_idx] < best_c5:
            best_c5 = population_scores[min_idx]
            best_h = population[min_idx].copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
