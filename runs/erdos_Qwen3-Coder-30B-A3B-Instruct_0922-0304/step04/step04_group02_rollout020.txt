import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Best solution tracking
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Initialize ADMM variables
    x = h.copy()
    z = np.zeros_like(x)
    u = np.zeros_like(x)
    
    # Constraint: sum(h) = n_points/2
    constraint_val = n_points / 2
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Store previous solution
        x_prev = x.copy()
        
        # Update x: minimize augmented Lagrangian
        # This is a simplified update - in practice solving this
        # requires more sophisticated methods, but we'll use a gradient-based
        # approach to find better points
        
        # Simple gradient ascent approach with constraints
        # Compute gradient approximation
        def grad(h):
            # This is not exactly the gradient but approximates the effect
            # of small changes in h on the overlap
            overlap = correlate(h, 1 - h, mode="full")
            # Approximate gradient direction
            grad_h = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                grad_h[i] = (overlap_plus.max() - overlap.max()) / 1e-5
            return grad_h
        
        # Since analytical gradients for this complex overlapping function are hard,
        # we're trying a coordinate descent approach with projection onto feasible set
        
        # Coordinate descent
        for i in range(n_points):
            # Quick local search for coordinate i 
            current_val = x[i]
            best_val = current_val
            best_score = evaluate_c5(x, dx)
            
            # Test several nearby values
            test_vals = np.linspace(max(0, current_val - 0.1), min(1, current_val + 0.1), 10)
            for test_val in test_vals:
                x_test = x.copy()
                x_test[i] = test_val
                # Maintain constraint
                diff = np.sum(x_test) - constraint_val
                if abs(diff) > 1e-6:
                    # Adjust other values to maintain constraint
                    # Simple average adjustment for others
                    adjust = -diff / (n_points - 1)
                    x_test = np.clip(x_test + adjust, 0, 1)
                    x_test[i] = test_val
                
                score = evaluate_c5(x_test, dx)
                if score < best_score:
                    best_score = score
                    best_val = test_val
                    
            # Update coordinate
            x[i] = best_val
            
        # Project onto feasible set (ensure sum constraint)
        sum_x = np.sum(x)
        if abs(sum_x - constraint_val) > 1e-6:
            # Rescale to meet constraint
            x = x * (constraint_val / sum_x)
        
        # Ensure bounds

            return np.sum(h) - n_points / 2

    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
