import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization using sinusoidal
        h = 0.5 + 0.5 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simple local search with gradient-like updates
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try perturbing the function around current optimum
        # Add small random noise and reproject
        h_new = h + np.random.normal(0, 0.01, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
            
        # Occasionally use larger perturbations
        if iter_count % 50 == 0:
            h_new = h + np.random.normal(0, 0.05, n_points) * np.random.choice([1, -1], n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
                
        # Occasionally reinitialize to parent solution
        if iter_count % 1000 == 0 and 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation and refinement
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Additional refinements if time allows
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try discrete optimization steps
        for i in range(n_points):
            # Try shifting i-th sample up or down
            step = 0.02 * dx
            original = final_h[i]
            test_h = final_h.copy()
            
            # Try upward shift
            test_h[i] = min(1.0, original + step)
            test_h = test_h * (n_points / 2) / np.sum(test_h)
            c5_up = evaluate_c5(test_h, dx)
            
            # Try downward shift
            test_h[i] = max(0.0, original - step)
            test_h = test_h * (n_points / 2) / np.sum(test_h)
            c5_down = evaluate_c5(test_h, dx)
            
            # Update if better
            better = c5_up < final_c5 or c5_down < final_c5
            if better and (c5_up < c5_down or c5_down >= final_c5):
                final_h[i] = min(1.0, original + step)
                final_c5 = c5_up
            elif better and (c5_down < c5_up or c5_up >= final_c5):
                final_h[i] = max(0.0, original - step)
                final_c5 = c5_down
                
            final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
