import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Generate basis functions
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size + 1))
    phi[:, 0] = 1
    for i in range(1, basis_size + 1):
        if i % 2 == 1:
            phi[:, i] = np.sin(np.pi * (i + 1) / 2 * x)
        else:
            phi[:, i] = np.cos(np.pi * i / 2 * x)
    
    # Compute int phi_i dx
    int_phi = np.zeros(basis_size + 1)
    for i in range(basis_size + 1):
        int_phi[i] = np.trapz(phi[:, i], x)
    
    # Initialize coefficients
    c = Variable(basis_size + 1)
    
    # Compute kernel matrix K
    K = np.zeros((basis_size + 1, basis_size + 1, 2 * n_points - 1))
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            for lag in range(2 * n_points - 1):
                idx = min(lag, 2 * n_points - 2 - lag)
                K[i, j, lag] = np.trapz(phi[:, i] * phi[:, j][idx:], x)
    
    # Initial guess
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        c_init = phi.T @ h_init * dx 
        c_init = c_init / np.sum(c_init) * 2
        c.value = c_init
    else:
        # Fallback to a simple step function
        c.value = np.zeros(basis_size + 1)
        c.value[0] = 0.5
        c.value[1] = 0.3
        c.value[2] = -0.1
    
    # Add convex constraints for non-negativity of h
    constraints = [
        c[0] * 2 + np.sum(c[1:] * int_phi[1:]) == 1,  # Integral constraint
    ]
    
    # Add bounds for c to enforce h in [0,1] approximately
    # Since h = phi @ c, we require bounds on c to keep h in [0,1]
    for i in range(basis_size + 1):
        constraints.append(c[i] >= 0)
    
    # Add McCormick relaxation for products
    z = Variable((basis_size + 1, basis_size + 1))
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            constraints.append(z[i, j] >= c[i] + c[j] - 1)
            constraints.append(z[i, j] <= c[i])
            constraints.append(z[i, j] <= c[j])
    
    # Objective
    max_val = Variable()
    obj = Minimize(max_val)
    
    # Add constraints for maximum overlap
    for lag in range(2 * n_points - 1):
        expr = 0
        for i in range(basis_size + 1):
            for j in range(basis_size + 1):
                expr += z[i, j] * K[i, j, lag]
        constraints.append(expr <= max_val)
        constraints.append(-expr <= max_val)
    
    # Construct and solve the problem
    prob = Problem(obj, constraints)
    
    # Use SCS solver for better performance
    try:
        prob.solve(solver=SCS, max_iters=10000, verbose=False)
    except:
        pass
    
    # If optimization did not converge, fallback to something reasonable
    if prob.status not in [OPTIMAL, OPTIMAL_INACCURATE]:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    else:
        h = phi @ c.value
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return (h.tolist(), max_overlap, n_points)
