import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high, solver='CVXOPT'):
    """Solve quadratic program using cvxpy"""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective: minimize (1/2)*delta^T * H * delta + c^T * delta
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints:
    # A * delta = b (equality constraint)
    constraints = [A @ delta == b]
    
    # delta_low <= delta <= delta_high (bound constraints)
    constraints.append(delta >= delta_low)
    constraints.append(delta <= delta_high)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize to satisfy constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Initialize for SCP procedure
    h_prev = h.copy()
    reg = 1e-5
    
    A = np.ones((1, n_points))
    b = np.array([0.0])  # Σδ = 0 as constraint
    
    delta_low = np.full(n_points, -1.0)
    delta_high = np.full(n_points, 1.0)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate gradient and Hessian
        # Analytical gradient: ∇C = dx * (1 - 2*h - shift_average)
        # Analytical Hessian from second-order terms is more complex
        
        # Instead, we'll use a simpler approach based on finite differences 
        # or approximations. We use the overlap Jacobian for the first-order term.
        
        # Simpler and more stable approach with finite differences for gradient:
        # For each shift k, overlap = sum h[i] (1 - h[i+k])
        # J[i] = 1 - 2 * h[i] - average of h[i+k] over all k
        # But for now, directly evaluate overlap and take approximate gradient.
        
        # Let's reuse the direct approach similar to before but with refined technique
        # Compute the overlap more efficiently
        overlap = correlate(h, 1 - h, mode="full")
        max_ind = np.argmax(overlap)
        max_val = overlap[max_ind]
        
        # Simplified gradient calculation: assuming max is at shift k,  
        # for k = max_ind - n_points + 1
        shift = max_ind - n_points + 1
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            other_idx = (i + shift) % n_points
            grad[i] = dx * (1 - 2 * h[i] - h[other_idx])
        
        # Hessian approximates as second derivative
        H = np.zeros((n_points, n_points))
        # Main diagonal approximates second derivatives
        for i in range(n_points):
            H[i, i] = -2 * dx  # Approximate second derivative of h[i]^2 term
            
        # Add regularization
        H += reg * np.eye(n_points)
        
        # Solve QP to get delta
        delta = _solve_qp(H, grad, A, b, delta_low, delta_high)
        
        if delta is None:
            # If infeasible or error, fallback to small steps with line search
            # Try multiple α values
            alpha_best = 0.0
            h_best = h.copy()
            best_c5_iter = best_c5
            
            for alpha in [0.5, 0.25, 0.1]:
                h_trial = h + alpha * (h - h_prev)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                c5_trial = evaluate_c5(h_trial, dx)
                if c5_trial < best_c5_iter:
                    best_c5_iter = c5_trial
                    h_best = h_trial.copy()
                    alpha_best = alpha
                    
            if alpha_best > 0:
                h = h_best.copy()
                best_c5 = best_c5_iter
                if best_c5 < best_c5:
                    best_h = h.copy()
            else:
                reg *= 10  # Increase regularization
            continue
        
        # Perform line search for best step size
        alpha_max = 1.0
        step_sizes = [0.5, 0.25, 0.1, 0.05]
        h_test = h + alpha_max * delta
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5_test = evaluate_c5(h_test, dx)
        
        for alpha in step_sizes:
            if alpha <= alpha_max:
                h_trial = h + alpha * delta
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                c5_trial = evaluate_c5(h_trial, dx)
                if c5_trial < c5_test:
                    c5_test = c5_trial
                    alpha_max = alpha
        
        # Apply the optimal step
        h = h + alpha_max * delta
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best value
        c5_val = evaluate_c5(h, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Update for next iteration
        h_prev = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
