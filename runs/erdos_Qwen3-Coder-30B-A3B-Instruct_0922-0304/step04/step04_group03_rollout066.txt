```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    w = 0.5  # width of each interval
    a = 0    # starting position of first interval
    
    # Search over d (distance between intervals)
    d_candidates = np.arange(0, 1.01, 0.01)
    
    best_c5 = float('inf')
    best_h = None
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute indices for two intervals
        i0 = 0
        i1 = int(np.floor((a + w) / dx))  # First interval end (inclusive)
        i2 = int(np.ceil((a + w + d) / dx))  # Second interval start
        i3 = int(np.floor((a + w + d + w) / dx))  # Second interval end (inclusive)
        
        # Create h
        h = np.zeros(n_points)
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        
        # Ensure sum equals 1 (normalize if needed)
        h = h * (n_points / 2) / np.sum(h)
        
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Refine search around best d if we have time
    if best_h is not None and time.time() - start_time < budget_s - 5:
        # Find the best d from coarse search
        d_candidates_fine = []
        best_d = None
        for i, d in enumerate(d_candidates):
            if time.time() - start_time > budget_s - 5:
                break
            i0 = 0
            i1 = int(np.floor((a + w) / dx))
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx))
            
            h = np.zeros(n_points)
            h[i0:i1+1] = 1
            h[i2:i3+1] = 1
            h = h * (n_points / 2) / np.sum(h)
            
            c5 = evaluate_c5(h, dx)
            
            if i == 0 or c5 < best_c5:
                best_c5 = c5
                best_d = d
                best_h = h.copy()

        if best_d is not None:
            d_min = max(0, best_d - 0.02)
            d_max = min(1, best_d + 0.02)
            d_candidates_fine = np.arange(d_min, d_max + 0.001, 0.001)
            
            for d in d_candidates_fine:
                if time.time() - start_time > budget_s - 5:
                    break
                    
                i0 = 0
                i1 = int(np.floor((a + w) / dx))
                i2 = int(np.ceil((a + w + d) / dx))
                i3 = int(np.floor((a + w + d + w) / dx))
                
                h = np.zeros(n_points)
                h[i0:i1+1] = 1
                h[i2:i3+1] = 1
                h = h * (n_points / 2) / np.sum(h)
                
                c5 = evaluate_c5(h, dx)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    # Return final solution
    return (best_h.tolist(), best_c5, n_points)
```
