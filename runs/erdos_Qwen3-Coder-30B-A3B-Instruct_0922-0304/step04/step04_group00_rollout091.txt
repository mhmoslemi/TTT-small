import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting-plane approach
    # Collect constraints
    constraints = []
    
    # Add initial constraints
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    # Run optimization loop
    iter_count = 0
    max_no_improvement = 30
    no_improvement = 0
    epsilon = 0.001
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iter_count += 1
        # Evaluate and identify worst overlap shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)  # Convert to shift index
        max_overlap = overlap[max_idx] * dx
        
        # Collect all constraints
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_var))
        
        # Add all constraints
        constrs = []
        # Add constraint for max overlap
        for i in range(len(constraints)):
            constrs.append(constraints[i])
        # Add new constraint
        if max_overlap < best_c5 - 1e-8:
            # The maximum shift gives us a constraint: h * (1 - h_shift) <= max_overlap
            # So we add: sum(h[i] * (1 - h[i + shift])) <= max_overlap
            shift = max_shift
            if shift >= 0:
                # h[i] * (1 - h[i + shift]) terms
                expr = 0
                for i in range(n_points - shift):
                    expr += h_var[i] * (1 - h_var[i + shift])
                # Also add terms where the shift wraps around
                for i in range(shift, n_points):
                    expr += h_var[i] * (1 - h_var[i - shift])
            else:
                shift = -shift
                # h[i] * (1 - h[i - shift]) terms
                expr = 0
                for i in range(shift, n_points):
                    expr += h_var[i] * (1 - h_var[i - shift])
                for i in range(shift):
                    expr += h_var[i] * (1 - h_var[i + n_points - shift])
            
            constrs.append(expr <= max_overlap)
        
        # Add simple bounds
        for i in range(n_points):
            constrs.append(h_var[i] >= 0)
            constrs.append(h_var[i] <= 1)
            
        # Add sum constraint
        constrs.append(cp.sum(h_var) == n_points / 2)
        
        # Solve the problem
        prob = cp.Problem(objective, constrs)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                new_h = h_var.value
                # Ensure new_h satisfies the constraints
                new_h = np.clip(new_h, 0, 1)
                new_h *= (n_points / 2) / np.sum(new_h)
                h = new_h
            else:
                # fallback to gradient-based method if problems
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Evaluate the new solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Periodically do some random perturbations
        if iter_count % 5 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
