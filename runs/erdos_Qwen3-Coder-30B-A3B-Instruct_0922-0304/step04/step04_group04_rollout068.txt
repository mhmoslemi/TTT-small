import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Try to use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_best = h_init.copy()
        c5_best = evaluate_c5(h_init, dx)
    else:
        # Initialize with simple construction
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_best = h_init.copy()
        c5_best = evaluate_c5(h_init, dx)
        
    # Benched optimization approach using convex optimization with Haar wavelets
        
    # Setup Haar wavelet basis
    def haar_wavelet_basis(n):
        # Generate Haar wavelets on [0,1] scaled to [0,2]
        basis = []
        # Scaling function (constant)
        phi = np.ones(n)
        basis.append(phi)
        
        # Generate wavelets at different scales
        j = 0
        while 2**j <= n:
            # At level j, 2^j wavelets
            for k in range(2**j):
                psi = np.zeros(n)
                start = k * 2**(1-j)
                end = start + 2**(1-j)
                if end > n:
                    end = n
                if start < n:
                    mid = (start + end) // 2
                    psi[start:mid] = 1
                    psi[mid:end] = -1
                basis.append(psi)
            j += 1
        return basis
    
    # small basis for efficiency
    basis_functions = haar_wavelet_basis(32)
    
    # Convex optimization approach
    # Use a small number of basis functions for manageable optimization
    num_basis = min(len(basis_functions), 16)  # just first 16
    
    # Check for feasibility of the current result
    if c5_best <= 0.3807:
        return (h_best.tolist(), c5_best, n_points)
    
    # Solver approach with optimization
    # Try to optimize using full basis
    h = Variable(n_points)
    
    # Existing optimization is much slower, so we will be using a hybrid:
    # 1. Start with initial construction
    # 2. For some time, try small perturbations
    # 3. Terminate early if within bounds
    
    # Perturbation search approach
    perturb_count = 100
    for i in range(perturb_count):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adjust current best
        h_perturb = h_best.copy()
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points) 
        # Simple manipulation
        if abs(h_perturb[idx1] - h_perturb[idx2]) > 0.1:
            # Swap values with clamping
            h_perturb[idx1], h_perturb[idx2] = h_perturb[idx2], h_perturb[idx1]
            # Renormalize
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            # Clip
            h_perturb = np.clip(h_perturb, 0, 1)
            
            c5_pert = evaluate_c5(h_perturb, dx)
            if c5_pert < c5_best:
                h_best = h_perturb.copy()
                c5_best = c5_pert
                
                # Early termination if we have good enough result
                if c5_best <= 0.3807:
                    break
    
    # Final adjustment: smooth and renormalize to get final form
    h_final = h_best.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
