import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        a = 0    # starting point of first interval
        d = 0.5  # distance between intervals
        
        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        
        # Normalize to constrain integral to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Grid search for the best spacing d within [0, 1]
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Try a fine-grained grid around the initial configuration
    d_candidates = np.arange(0.0, 1.01, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new h with spacing d
        h_new = np.zeros(n_points)
        w = 0.5
        a = 0
        
        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        # Ensure we don't exceed array bounds
        i1 = min(i1, n_points - 1)
        i3 = min(i3, n_points - 1)
        i2 = max(i2, 0)
        
        h_new[i0:i1+1] = 1
        h_new[i2:i3+1] = 1
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5 = evaluate_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
    
    # If we're still above target, try refined search around best
    if best_c5 > 0.3807:
        d_best = np.arange(max(0, np.argmin([evaluate_c5(
            np.concatenate([np.zeros(int(np.floor(i/100/dx)) + 1), 
                           np.ones(int(np.floor((i/100 + 0.5) / dx) - 
                                   int(np.floor(i/100/dx)) - 1))), 
                     np.zeros(n_points - int(np.floor((i/100 + 0.5) / dx) - 1) - 1)]), 
                     dx) for i in np.arange(0.0, 1.01, 0.01)])), 1.0, 0.001)
        
        for d in d_best:
            if time.time() - start_time > budget_s - 1:
                break

            h_temp = np.zeros(n_points)
            w = 0.5
            a = 0
            i0 = int(np.floor(a / dx))
            i1 = int(np.floor((a + w) / dx))
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx))
            
            h_temp[i0:i1+1] = 1
            h_temp[i2:i3+1] = 1
            
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            c5 = evaluate_c5(h_temp, dx)
            if c5 < best_c5:
                best_h = h_temp.copy()
                best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
