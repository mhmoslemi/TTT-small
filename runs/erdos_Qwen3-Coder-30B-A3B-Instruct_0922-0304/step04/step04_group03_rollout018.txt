import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Grid search over interval separation d
    d_candidates = np.arange(0.0, 1.001, 0.01)
    best_c5 = float('inf')

    w = 0.5  # width of each interval
    a = 0.0  # start of first interval

        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        # Ensure indices are within bounds
        i0 = max(0, i0)
        i1 = min(n_points - 1, i1)
        i2 = max(0, i2)
        i3 = min(n_points - 1, i3)
        
        # Create h vector
        h = np.zeros(n_points)
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        
        # Normalize to satisfy constraint ∫h(x)dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate C5
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

            c5 = evaluate_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        for d in fine_d_candidates:
            i0 = int(np.floor(a / dx))
            i1 = int(np.floor((a + w) / dx))
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx))
            
            i0 = max(0, i0)
            i1 = min(n_points - 1, i1)
            i2 = max(0, i2)
            i3 = min(n_points - 1, i3)
            
            h = np.zeros(n_points)
            h[i0:i1+1] = 1
            h[i2:i3+1] = 1
            
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5

            best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
