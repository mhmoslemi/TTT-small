import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, dx, target_sum):
    """Project h to satisfy the sum constraint."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def project_bounds(h):
    """Ensure h stays within [0, 1]."""
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    h = project_h(h, dx, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent parameters
    max_iter = 10000
    step_size = 0.01
    decay_factor = 0.999
    tolerance = 1e-6
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find worst case
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # dO_k/dh_i = (1 - h[(i + k) % n]) - h[(i - k) % n]
            # But we want the gradient for the worst case k
            # k = max_idx - n_points (when max_idx >= n_points)
            # k = max_idx (when max_idx < n_points)
            k = max_idx - n_points // 2
            if k < 0:
                grad[i] = (1 - h[(i + k) % n_points]) - h[(i - k) % n_points]
            else:
                grad[i] = (1 - h[(i + k) % n_points]) - h[(i - k) % n_points]
                
        # Update h
        h_new = h - step_size * grad
        h_new = project_bounds(h_new)
        h_new = project_h(h_new, dx, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept if improvement
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            step_size = min(step_size * 1.05, 0.1)
        else:
            # Try a smaller step
            step_size *= decay_factor
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
