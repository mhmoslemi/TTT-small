import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting plane loop
    max_iter = 10000
    epsilon = 0.001
    iteration = 0
    
    # Collect constraints
    constraints_list = []
    h_vars = cp.Variable(n_points)
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Compute current overlaps and find max
        current_overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(current_overlap)
        max_k = max_idx - (n_points - 1)
        max_overlap_val = current_overlap[max_idx] * dx
        
        # Check if we should stop
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Add constraint that overlaps at shift k are bounded
        # We encode the constraint manually similar to correlation
        # This is a challenging part, so we'll do a different approach:
        # Approximate with convolutions and LP relaxation
        # But in practice, much simpler approach:
        # Use a feasible descent method based on gradient estimation
        
        # Heuristic descent: perturb toward reducing maximum overlap
        # Compute gradient-like information (what changes would reduce overlap)
        max_k_shifted = max_idx - (n_points - 1)
        
        # Create a simple direction of descent
        if max_k_shifted >= 0:
            # Shift to reduce this overlap
            h_new = h.copy()
            # Decrease h at positions where we shift appears in the correlation
            for i in range(max_k_shifted, n_points):
                if i - max_k_shifted >= 0:
                    h_new[i] = max(0, h_new[i] - 0.001)
            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        else:
            # Negative shift
            h_new = h.copy()
            for i in range(n_points + max_k_shifted):
                h_new[i] = max(0, h_new[i] - 0.001)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # When in the first 50 iterations, also use a random perturbation
        if iteration % 10 == 0 and iteration < 50:
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        iteration += 1
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
