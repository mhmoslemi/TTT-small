import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization
    learning_rate = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 20 or c5 <= 0.3807:
            break
            
        # Compute gradients
        # For each k, we have overlap_k = sum h[i] * (1 - h[i+k])
        # Gradient wrt h[i] = sum over k of (1 - h[i+k]) where k is such that (i+k) mod n_points corresponds to position with respect to i
        # Simplified approach: compute full correlation for all shifts
        max_overlap_idx = np.argmax(correlate(h, 1 - h, mode="full"))
        k_max = max_overlap_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from shift k_max
            j = (i + k_max) % n_points
            gradient[i] += 1 - h[j]
            
            # Contribution from negative shift (shift -k_max)
            j = (i - k_max) % n_points
            gradient[i] -= h[j]
        
        # Update h
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        
        if iteration % 100 == 0:
            learning_rate *= 0.9
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
