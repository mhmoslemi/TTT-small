import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Start with simple construction
        h = np.zeros(n_points)
        h[20:50] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search for interval spacing
    d_candidates = np.arange(0.0, 1.01, 0.01)
    best_d = 0.0
    best_d_c5 = float('inf')
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create two intervals
        w = 0.5  # width of each interval
        a = 0.0  # start position
        
        # Calculate interval indices
        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        # Construct h
        h_test = np.zeros(n_points)
        if i1 >= i0:
            h_test[i0:i1+1] = 1.0
        if i3 >= i2:
            h_test[i2:i3+1] = 1.0
        
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5 = evaluate_c5(h_test, dx)
        
        if c5 < best_d_c5:
            best_d_c5 = c5
            best_d = d
            
        # Early exit if target achieved
        if c5 <= 0.3807:
            break
    
    # Refine around best found d
    if best_d_c5 < best_c5:
        d_range = np.arange(max(0.0, best_d-0.05), min(1.0, best_d+0.05), 0.005)
        for d in d_range:
            if time.time() - start_time > budget_s - 1:
                break
                
            w = 0.5
            a = 0.0
            
            i0 = int(np.floor(a / dx))
            i1 = int(np.floor((a + w) / dx))
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx))
            
            h_test = np.zeros(n_points)
            if i1 >= i0:
                h_test[i0:i1+1] = 1.0
            if i3 >= i2:
                h_test[i2:i3+1] = 1.0
                
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            c5 = evaluate_c5(h_test, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_test.copy()
    
    # Try different heuristics for remaining time
    if time.time() - start_time < budget_s - 10:
        # Try local refinement around current best
        for i in range(30):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb best h
            h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            c5 = evaluate_c5(h_perturbed, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturbed.copy()
    
    return (best_h.tolist(), best_c5, n_points)
