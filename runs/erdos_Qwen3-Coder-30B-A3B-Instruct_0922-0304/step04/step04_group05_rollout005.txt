import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def solve_qp(H, c, A, b, delta_low, delta_high, solver='CVXOPT'):
    """Solve quadratic program with cvxpy"""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [A @ delta == b, 
                   delta >= delta_low,
                   delta <= delta_high]
    
    # Solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500

    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Line search parameters
    alpha_max = 1.0
    alpha_decay = 0.5
    line_search_max = 10
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps to build gradient and Hessian
        # For simplicity and speed, we'll compute numerical gradients
        # For small perturbations
        epsilon = 1e-4
        n_shifts = 2 * n_points - 1  # Full cross-correlation
        shift_indices = np.arange(-n_points + 1, n_points)
        
        # Use method of Lagrange multipliers for gradients
        # Currently simplifying: use finite difference approach
        # But since the objective is not smooth, we'll build
        # simple linear approximations in line-search
                
        # Gradient and Hessian approximation
        # Approximate ∇C_k =  dx * (1 - 2 * h - shift_avg)
        # Approximate ∇²C_k = dx *(- 2*I + shift_matrix)
        
        # For simplicity, compute full overlap matrix, then approximate
        # Just estimate gradient of the maximum overlap
        
        # We compute approximate gradient via finite difference
        # with small random perturbations
        grad = np.zeros(n_points)
        h_base = h.copy()
        
        # Get gradient via finite difference
        grad_est = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            # Renormalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c_plus = evaluate_c5(h_plus, dx)
            c_minus = evaluate_c5(h_minus, dx)
            grad_est[i] = (c_plus - c_minus) / (2 * epsilon)
            
        grad = grad_est.copy()
        
        # Hessian: To simplify, we approximate as diagonal Hessian
        # using a central difference of the gradient
        # Regularization for numerical stability:
        reg = 1e-5
        
        # Create quadratic approximation problem
        # Hessian is not available in closed form, so we'll use a simple approach
        
        # Instead of computing Hessian, we'll project and perform QP
        # to find descent step
        
        # Approximate Hessian based on curvature (as 2nd order approximation from gradient)
        # We make it diagonal with uniform regularization
        
        H = np.eye(n_points) * reg  # Regularized Hessian
        
        # Add regularization matrix with positive values to ensure convexity
        # Proper Hessian structure would be complex, so we use sensitivity approximation
        # Specifically, keep H diagonal minimal and slightly positive
        
        # Set up linear constraints for sum(h) = n_points/2, and bounds 0 <= h <= 1
        # Inequality constraints must be change in h within [0,1]
        # We'll project back after update
        
        # Construct equality constraint matrix A eq b
        A_eq = np.ones((1, n_points))
        b_eq = np.array([n_points/2])
        
        # We solve a projected QP for the update:
        # min δ (c^T δ + (1/2) δ^T H δ)
        # s.t. A_eq δ = b_eq - A_eq h
        #      0 <= h + δ <= 1 (bound constraints)
        
        # Solve QP subproblem
        A_total = A_eq
        b_total = b_eq - A_eq @ h
        
        # Compute bounds
        delta_lower = -h
        delta_upper = 1 - h

        current_delta = None
        best_delta = None

            delta = solve_qp(H, grad, A_total, b_total, delta_lower, delta_upper)

                step_alpha = alpha_max
                found_step = False

                        c_trial = evaluate_c5(h_new, dx)
                        if c_trial < best_delta_val:
                            best_delta_val = c_trial

                        if c_trial <= best_c5:
                            # Acceptable instead of exploring further
                            found_step = True
                            break

                        pass
                    step_alpha *= alpha_decay
                if not found_step and best_delta is not None:
                    # Accept best found update
                    current_delta = best_delta
                else:
                    current_delta = delta
        except:
            current_delta = None
            
        # Update h
        if current_delta is not None:
            h_new = h + current_delta
        else:
            h_new = h.copy()
            
        # Enforce bounds and sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        h = h_new.copy()
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    final_h = best_h.tolist()
    
    return (final_h, final_c5, n_points)
