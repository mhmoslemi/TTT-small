```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, norm, sum, multiply, abs as cabs

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Fourier basis functions on [0,2]
    n_basis = 30
    t = np.linspace(0, 2, n_points, endpoint=False)
    phi = np.zeros((n_points, n_basis))
    for i in range(n_basis):
        if i % 2 == 0:
            phi[:, i] = np.sin((i//2 + 1) * np.pi * t)
        else:
            phi[:, i] = np.cos(((i+1)//2) * np.pi * t)
    
    # Compute overlap kernel for all lags
    K = np.zeros((n_basis, n_basis, 2*n_points - 1))
    for k in range(2*n_points - 1):
        lag = k - n_points + 1
        k_shifted = np.zeros(n_points)
        if lag >= 0:
            k_shifted[lag:] = 1
        else:
            k_shifted[:lag] = 1
        K[:,:,k] = np.dot(phi.T, np.diag(1 - k_shifted)) @ phi
    
    # Try to solve with CVXPY using MOSEK
    a = Variable(n_basis)
    # Rayleigh quotient constraint on integral
    integral_constraint = sum(a * (phi.T @ np.ones(n_points)) * dx) == 1
    
    # McCormick envelopes for product terms
    z = Variable((n_basis, n_basis))
    # Define McCormick inequalities
    constraints = []
    for i in range(n_basis):
        for j in range(n_basis):
            # z[i,j] >= a[i] + a[j] - 1
            constraints.append(z[i,j] >= a[i] + a[j] - 1)
            # z[i,j] <= a[i]
            constraints.append(z[i,j] <= a[i])
            # z[i,j] <= a[j]
            constraints.append(z[i,j] <= a[j])
            
    # Add integral and boundary constraints
    constraints.append(sum(a) >= 0)
    constraints.append(sum(a) <= 10)
    
    # Set up the optimization
    objective = Minimize(cp.max([cp.abs(cp.sum(z * K[:,:,k])) for k in range(2*n_points-1)]))
    prob = Problem(objective, constraints + [integral_constraint])
    
    # Try with MOSEK
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-7, reltol=1e-7, abstol=1e-7)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            a_vals = a.value
            h = phi @ a_vals
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Fallback to direct optimization with old method
            h = h0.copy()
            
    except:
        h = h0.copy()
    
    # Local refinement with gradient-based method
    def obj(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        overlap = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(overlap) * dx
    
    def grad(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        # Approximate gradient of the max overlap using finite differences
        eps = 1e-6
        grad_vec = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            grad_vec[i] = (obj(h_plus) - obj(h_minus)) / (2 * eps)
        return grad_vec
    
    # Gradient descent with restarts
    h_best = h.copy()
    best_val = obj(h_best)
    max_iter = 3000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Gradient descent
        g = grad(h_best)
        alpha = 0.01
        h_temp = h_best - alpha * g
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        new_val = obj(h_temp)
        if new_val < best_val:
            h_best = h_temp.copy()
            best_val = new_val
        # Occasionally restart from a better point
        if i % 300 == 299:
            h_init = h0.copy()
            h_init += np.random.normal(0, 0.02, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            h_temp = h_init.copy()
            val = obj(h_temp)
            if val < best_val:
                h_best = h_temp.copy()
                best_val = val
    
    # Final validation and correction
    h_final = np.clip(h_best, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final objective computation
    overlap_final = correlate(h_final, 1 - h_final, mode="full")
    c5_final = np.max(overlap_final) * dx
    
    return (h_final.tolist(), c5_final, n_points)
```
