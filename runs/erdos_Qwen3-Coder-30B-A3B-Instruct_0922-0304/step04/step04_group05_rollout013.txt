import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradients(h, dx, n_points):
    """Compute gradient of overlap integral"""
    # Compute all overlaps for shifts
    h_shifted = np.concatenate([h, np.zeros(n_points)])
    gradients = np.zeros(n_points)
    
    for k in range(n_points):
        # This is effectively the gradient of the k-th overlap
        # We compute: d/dh_i (sum j h[j] * (1 - h[j+k])) = (1 - h[j+k]) - h[j+k]
        # which gives ∇C_k = dx * (1 - 2*h - shift_average)
        h_k = h_shifted[k:k+n_points]
        gradients += (1 - 2 * h) * h_k
    
    return dx * gradients

def line_search(h, grad, dx, n_points, max_alpha=1.0):
    """Find optimal step size using Armijo rule"""
    alpha = max_alpha
    c1 = 1e-4
    beta = 0.5
    
    # Try different alphas until good enough
    for _ in range(20):  # Max iterations
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n_points, dx)
        new_c5 = evaluate_c5(h_new, dx)
        old_c5 = evaluate_c5(h, dx)
        
        if new_c5 <= old_c5 - c1 * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= beta
    
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    min_improvement = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a triangular function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - 2 * abs(i - mid) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Current best
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradients(h, dx, n_points)
        
        # Line search for step size
        alpha, h_new = line_search(h, grad, dx, n_points)
        
        # Update h
        h = h_new
        
        # Evaluate
        current_c5 = evaluate_c5(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Stability check - prevent oscillation
        if i > 10 and abs(best_c5 - evaluate_c5(best_h, dx)) < min_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
