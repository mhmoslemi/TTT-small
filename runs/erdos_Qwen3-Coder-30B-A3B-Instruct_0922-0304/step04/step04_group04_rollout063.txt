import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier basis functions
    def fourier_basis(x, n_basis):
        basis = []
        for i in range(1, n_basis+1):
            if i % 2 == 1:
                basis.append(np.sin(i * np.pi * x))
            else:
                basis.append(np.cos((i//2) * np.pi * x))
        return np.array(basis)
    
    # Optimization with relaxed constraints using CVXPY
    n_basis = 30  # Number of Fourier basis functions
    
    # Create x grid
    x = np.linspace(0, 2, n_points)
    
    # Create basis matrix
    Phi = fourier_basis(x, n_basis).T  # (n_points, n_basis)
    
    # Compute the kernel matrix K_k for all shifts k
    K = np.zeros((n_points, n_basis, n_basis))
    for k in range(n_points):
        x_shifted = x + dx * k
        # Make sure x_shifted stays within [0,2]
        x_shifted = np.where(x_shifted > 2, x_shifted - 2, x_shifted)
        x_shifted = np.where(x_shifted < 0, x_shifted + 2, x_shifted)
        Phi_shifted = fourier_basis(x_shifted, n_basis).T
        for i in range(n_basis):
            for j in range(n_basis):
                K[k, i, j] = np.dot(Phi[:, i], Phi_shifted[:, j])
    
    # Choice of solver
    solver = cp.MOSEK
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Create as many Fourier coefficients as needed
    a = cp.Variable(n_basis)
    
    # To avoid considering all coefficients in the solution that
    # can be tuned independently, add an at least heuristic penalty 
    # using reconstruction error, promoting sparsity in the basis
    # We iterate with different versions of the optimization.
    
    max_iterations = 500
    
    for iter_count in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute integral constraint
        integral_constraint = cp.sum(cp.multiply(a, Phi @ np.ones(n_basis) * dx)) == 1
        
        # Add additional approximating constraints based on homogeneity of variance
        # and relative smoothness assumption
        
        # Construct quadratic programming model over coefficients using exact matrix
        # representation of kernel terms for different lags
        
        # Make full incompatible Reed-Solomon like approximation)
        # seems overly complex. Instead, project on def init copy of basis
        # and construct the penn matrix efficiently
        
        # No new method used here - just sense-check via local perturbation
        # and convolution references
        
        fixed_constraint = cp.sum(cp.multiply(a, Phi @ np.ones(n_basis) * dx)) == 1
        
        # Subject to bounds: h(x) = \sum a_k phi_k(x) sum(a_k) = 1 initial constraint
        
        # Objective: minimize maximum overlap over full correlation
        # Instead of explicitly modeling over all lags, we constrain the 
        # the moments or integral maximum terms which reduces problem size.
        
        # Approximate optimization: Solve for minimal-max overlap not completely exact,
        # using projected gradients near initial.
        trial_h = np.clip(Phi @ a.value, 0, 1)
        trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
        
        trial_c5 = evaluate_c5(trial_h, dx)
        
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h.copy()
            
        # Random perturbation search
        if iter_count % 5 == 0 and iter_count > 10:
            perturbation = np.random.random(n_basis) * 0.01 - 0.005
            a.value = a.value + perturbation
            # Restart with new basis coefficients initialized near best
            a.value = np.clip(a.value, -10, 10)
            
    # Final optimization with clamping to ensure validity
    h_opt = best_h
    
    return (h_opt.tolist(), best_c5, n_points)
