import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolutionary Strategy parameters
    population_size = 100
    elite_size = int(0.2 * population_size)
    mutation_strength = 0.1
    max_iter = 1000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        new_h = h + np.random.normal(0, mutation_strength, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # ES iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for ind in population:
            fitness = evaluate_c5(ind, dx)
            fitness_scores.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Selection: keep elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Mutation: for each offspring, perturb 30% of elements
        while len(new_population) < population_size:
            parent = elite[np.random.randint(0, elite_size)]
            child = parent.copy()
            
            # Perturb 30% of the indices
            num_to_mutate = max(1, int(0.3 * n_points))
            indices_to_mutate = np.random.choice(n_points, num_to_mutate, replace=False)
            
            # Apply Gaussian noise
            for i in indices_to_mutate:
                child[i] += np.random.normal(0, mutation_strength)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation strength
        if iter_count % 10 == 0:
            improvement = np.mean(fitness_scores[:elite_size]) - np.mean(fitness_scores[elite_size:elite_size+10])
            if improvement > 0.0001:
                mutation_strength *= 1.1
            else:
                mutation_strength *= 0.95
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.98:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
