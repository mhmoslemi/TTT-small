import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to ensure sum(h) = n_points / 2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradients(h, dx, n_points):
    """Compute exact gradient for the overlap objective"""
    # Compute 1 - h - shift average
    cross_corr = correlate(h, 1 - h, mode="full")
    shift_avg = np.zeros(n_points)
    for k in range(n_points):
        shift_avg[k] = np.sum(cross_corr[n_points-1+k:]) / (n_points - k) if (n_points - k) > 0 else 0
    
    grad = dx * (1 - 2 * h - shift_avg)
    return grad

def line_search(h, grad, dx, n_points, alpha_max=1.0):
    """Backtracking line search to find suitable step size"""
    alpha = alpha_max
    current_c5 = evaluate_c5(h, dx)
    c1 = 1e-4
    max_iter = 20
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 <= current_c5 + c1 * alpha * np.dot(grad, -grad):
            break
        alpha *= 0.5
    return h_new, alpha

def frank_wolfe_step(h, grad, dx, n_points):
    """Frank-Wolfe style step to maintain feasibility"""
    # Find direction that minimizes gradient descent while respecting constraints
    # This is simplified to just a step with projection
    step_size = 0.1
    h_new = h - step_size * grad
    h_new = project_h(h_new, n_points)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    h = project_h(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adaptive step sizes
    alpha = 0.1
    improvement_count = 0
    stagnation_count = 0
    stagnation_threshold = 50
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradients(h, dx, n_points)
        
        # Perform line search for step size
        h_new, alpha = line_search(h, grad, dx, n_points, alpha)
        
        # Update
        h = h_new
        c5 = evaluate_c5(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            improvement_count = 0
            stagnation_count = 0
        else:
            improvement_count += 1
            stagnation_count += 1
            
        # Adjust step size if stagnating
        if stagnation_count > stagnation_threshold:
            alpha *= 0.5
            stagnation_count = 0
            if alpha < 1e-6:
                break
    
    return (best_h.tolist(), best_c5, n_points)
