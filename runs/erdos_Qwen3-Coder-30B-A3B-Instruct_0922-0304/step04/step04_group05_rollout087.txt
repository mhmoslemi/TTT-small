import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to meet sum constraint
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h.fill(target_sum / len(h))
    return np.clip(h, 0, 1)

def compute_gradient(h, dx, n_points):
    """Compute gradient of overlap with respect to h"""
    # Compute overlap for each shift
    # The gradient is approximately: dx * (1 - 2*h - shift_average)
    # But we need to be more precise
    # For each point i, we look at all shifts where h[i] contributes
    grad = np.zeros(n_points)
    
    # For each shift k, compute contribution to gradient
    # We can compute these more directly:
    # Overlap at shift k is sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k])
    # So gradient with respect to h[i] is sum over all shifts where i appears:
    # sum_{k, i+k < n_points} (1 - h[i+k])  - sum_{k, i+k < n_points} h[i+k] = 
    # sum_{k, i+k < n_points} (1 - 2*h[i+k])
    
    # Build all needed stacked convolutions
    ones = np.ones(n_points)
    gradient_contribution = np.zeros(n_points)
    
    # Each h[i] appears in overlap with h[i+k] for all shifts k
    # When we compute the gradient of overlap with respect to h[i]
    # it's sum of (1 - 2 * h[i+k]) for valid shifts k
    
    # More efficiently:
    # Create matrix of contributions using convolution
    ones_h = np.zeros(n_points * 2 - 1)
    ones_h[:n_points] = 1
    ones_h[n_points:] = 0
    h_aug = np.zeros(n_points * 2 - 1)
    h_aug[:n_points] = h
    h_aug[n_points:] = 0
    
    # Gradient contribution from the 1 term:
    grad_ones = correlate(ones_h, np.ones(2*n_points - 1), mode='full')[:n_points] 
    # Gradient contribution from the 2*h term:
    grad_h = correlate(h_aug, np.ones(2*n_points - 1), mode='full')[:n_points] 
    grad = dx * (grad_ones - 2 * grad_h)
    
    return grad

def armijo_line_search(h, grad, c5_current, dx, n_points, alpha_max=1.0):
    """Perform Armijo backtracking line search"""
    c = 0.1
    rho = 0.5
    alpha = alpha_max
    c5_old = c5_current
    max_iter = 20
    for iter in range(max_iter):
        # Try step
        h_new = h - alpha * grad
        h_new = project_to_constraint(h_new, n_points / 2)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new <= c5_old - c * alpha * np.sum(grad**2):
            return h_new, alpha, c5_new
        else:
            alpha *= rho
    return h, alpha, c5_old

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create initialized function: a triangular starting point
        h = np.zeros(n_points)
        midpoint = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - 2 * abs(i - midpoint) / n_points)
        h = project_to_constraint(h, n_points / 2)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h, dx, n_points)
        
        # Perform line search
        h_new, step_size, c5_new = armijo_line_search(h, grad, best_c5, dx, n_points)
        
        if step_size < 1e-12:
            break
            
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Termination check
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
