import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise create a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Grid search on interval spacing
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Test intervals with fixed width of 0.5
    interval_width = 0.5
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute indices for the two intervals
        i0 = 0
        i1 = int((interval_width) / dx) - 1  # First interval [0, 0.5]
        i2 = int((interval_width + d) / dx)  # Second interval starts at d
        i3 = int((2 * interval_width + d) / dx) - 1  # Second interval ends at d + 0.5
        
        # Avoid overlaps
        if i2 <= i1:
            continue
            
        # Create the step function
        h_test = np.zeros(n_points)
        h_test[i0:i1+1] = 1.0
        h_test[i2:i3+1] = 1.0
        
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5_test = evaluate_c5(h_test, dx)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # If no improvement found, perform a random perturbation of the best result
    if best_c5 > 0.3807:
        h_perturbed = best_h.copy()
        for i in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            idx = np.random.randint(0, n_points)
            h_perturbed[idx] = np.clip(h_perturbed[idx] + np.random.normal(0, 0.01), 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = evaluate_c5(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            
        # Fine-tune using simple local search
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            idx = np.random.randint(0, n_points)
            step = np.random.normal(0, 0.005)
            new_h = best_h.copy()
            new_h[idx] = np.clip(new_h[idx] + step, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
