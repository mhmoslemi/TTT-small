import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill rest with random feasible samples
    for i in range(1, pop_size):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best
    fitness = [evaluate_c5(p, dx) for p in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # ES parameters
    sigma = 0.1
    c_adapt = 0.3
    adapt_count = 0
    success_count = 0
    success_threshold = pop_size // 4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(p, dx) for p in population]
        
        # Update best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_h = population[new_best_idx].copy()
            best_c5 = fitness[new_best_idx]
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection - keep top 25%
        selected_indices = np.argsort(fitness)[:pop_size//4]
        selected = [population[i] for i in selected_indices]
        selected_fitness = [fitness[i] for i in selected_indices]
        
        # Prepare next generation
        new_population = []
        
        # Keep best individual
        new_population.append(selected[0])
        
        # Generate offspring with adaptive mutation
        for _ in range(pop_size - 1):
            # Select parent randomly from top 25%
            parent_idx = np.random.choice(len(selected))
            parent = selected[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, sigma, n_points)
            offspring = parent + noise
            
            # Project to feasible region
            offspring = project_feasible(offspring, n_points)
            
            new_population.append(offspring)
            
        population = new_population
        
        # Adapt mutation rate based on success
        adapt_count += 1
        if adapt_count >= 10:
            if success_count > success_threshold:
                sigma = min(sigma * 1.2, 0.5)
            else:
                sigma = max(sigma * 0.8, 0.01)
            success_count = 0
            adapt_count = 0
        
    return (best_h.tolist(), best_c5, n_points)
