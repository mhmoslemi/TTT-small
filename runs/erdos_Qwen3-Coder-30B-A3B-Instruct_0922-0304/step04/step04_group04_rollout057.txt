import numpy as np
from scipy.signal import correlate
import time
from cvxpy import*

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Precompute basis functions
    x = np.linspace(0, 2, n_points)
    basis_functions = []
    basis_functions.append(np.ones(n_points))  # c0 term
    
    # Sine and cosine terms for Fourier basis
    for k in range(1, basis_size):
        if k % 2 == 1:  # Odd index -> sine
            basis_functions.append(np.sin(np.pi * k * x / 2))
        else:  # Even index -> cosine
            basis_functions.append(np.cos(np.pi * k * x / 2))
    
    basis_matrix = np.array(basis_functions).T
    
    # Compute integral of each basis function
    integral_basis = np.array([np.trapz(phi, x) for phi in basis_functions])
    
    # Initialize variables
    c = Variable(basis_size)
    
    # Define constraints
    constraints = []
    
    # Constraint 1: integral equals 1
    constraints.append(sum(c[i] * integral_basis[i] for i in range(basis_size)) == 1)
    
    # Constraint 2: bounds on coefficients (will ensure h(x) in [0,1])
    # We'll refine this in the solution since it's complex to directly model
    
    # Build matrix for convolution
    K = np.zeros((basis_size, basis_size, 2*n_points - 1))
    for i in range(basis_size):
        for j in range(basis_size):
            # Compute correlation coefficient for basis functions
            corr = correlate(basis_functions[i], basis_functions[j], mode='full')
            # Correlation over [0,2] with step dx vs. continuous integral
            K[i,j,:] = corr * dx
    
    # Create McCormick relaxation nodes
    z = Variable((basis_size, basis_size))
    
    # Add McCormick constraints for z[i,j] = c[i]*c[j]
    for i in range(basis_size):
        for j in range(basis_size):
            constraints.append(z[i,j] >= c[i] + c[j] - 1)
            constraints.append(z[i,j] <= c[i])
            constraints.append(z[i,j] <= c[j])
    
    # The objective is to minimize the maximum overlap integral
    # Compute the overlap integral for various lags
    max_overlap_expr = 0
    lag_range = range(2*n_points - 1)
    for lag in lag_range:
        expr = sum(z[i,j] * K[i,j,lag] for i in range(basis_size) for j in range(basis_size))
        max_overlap_expr = max(max_overlap_expr, abs(expr))
    
    # Objective
    objective = Minimize(max_overlap_expr)
    
    # Create problem
    problem = Problem(objective, constraints)
    
    # Get initial guess
    h_guess = np.ones(n_points) * 0.5
    if 'initial_h_values' in globals():
        h_guess = np.array(initial_h_values)
    
    # Convert initial guess to coefficients
    c_guess = np.linalg.lstsq(basis_matrix, h_guess, rcond=None)[0]
    
    # Early termination
    if time.time() - start_time > budget_s - 1:
        return (h_guess.tolist(), 0.381, n_points)
    
    # Solve
    try:
        problem.solve(solver=SCS, max_it=5000, verbose=False)
        if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            c_sol = c.value
            h_sol = basis_matrix @ c_sol
            h_sol = np.clip(h_sol, 0, 1)
            # Renormalize
            h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
            overlap = correlate(h_sol, 1 - h_sol, mode="full")
            max_overlap = np.max(overlap) * dx
            return (h_sol.tolist(), max_overlap, n_points)
        else:
            # Fallback method
            h_fallback = h_guess
            h_fallback = np.clip(h_fallback, 0, 1)
            h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
            overlap = correlate(h_fallback, 1 - h_fallback, mode="full")
            max_overlap = np.max(overlap) * dx
            return (h_fallback.tolist(), max_overlap, n_points)
    except:
        # Fallback if optimization fails
        h_fallback = h_guess
        h_fallback = np.clip(h_fallback, 0, 1)
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        overlap = correlate(h_fallback, 1 - h_fallback, mode="full")
        max_overlap = np.max(overlap) * dx
        return (h_fallback.tolist(), max_overlap, n_points)
