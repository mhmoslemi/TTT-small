import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:

        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)

    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1e-4  # Initial temperature
    alpha = 0.99  # Cooling rate
    max_iter = 10000
    step_size = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor: perturb one random element
        h_new = h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, step_size)
        h_new[idx] += delta
        
        # Ensure element-wise constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Acceptance criterion
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        # Update temperature and step size
        T *= alpha
        step_size *= alpha
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
