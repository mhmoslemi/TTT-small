```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Helper function
    def evaluate_c5(h_vals):
        overlap = correlate(h_vals, 1 - h_vals, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    # Use gradient descent with constraint
    learning_rate = 0.01
    decay = 0.995
    patience = 10
    best_h = h.copy()
    best_score = evaluate_c5(h)
    patience_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # Approximate gradient with finite difference
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] = min(1, h_plus[j] + eps)
            h_minus = h.copy()
            h_minus[j] = max(0, h_minus[j] - eps)
            
            score_plus = evaluate_c5(h_plus)
            score_minus = evaluate_c5(h_minus)
            grad[j] = (score_plus - score_minus) / (2 * eps)
            
        # Update h
        h = h - learning_rate * grad
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Enforce constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        score = evaluate_c5(h)
        
        if score < best_score:
            best_h = h.copy()
            best_score = score
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            learning_rate *= decay
            patience_count = 0
            
        if best_score <= 0.3807:
            break

    return (best_h.tolist(), best_score, n_points)
```
