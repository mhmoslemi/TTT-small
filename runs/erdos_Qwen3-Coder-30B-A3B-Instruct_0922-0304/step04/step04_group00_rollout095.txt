import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def evaluate_population(population, dx):
    """Evaluate all individuals in population"""
    values = []
    for h in population:
        values.append(compute_c5(h, dx))
    return np.array(values)

def selection(population, fitness, num_select):
    """Tournament selection"""
    selected = []
    for _ in range(num_select):
        # Tournament of 3
        indices = np.random.choice(len(population), 3, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Single point crossover"""
    n = len(parent1)
    point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.01):
    """Add Gaussian noise to h"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(h))
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, np.sum(h))  # Keep sum intact, but renormalize if needed
    return h

def local_search(h, dx, max_iter=50):
    """Local refinement using SLSQP"""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - np.sum(h)  # Target sum is fixed
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-8}
        )
        if result.success:
            h = result.x
        h = np.clip(h, 0, 1)
        h = normalize_h(h, np.sum(h))
    except:
        pass  # If optimization fails, return original
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h)
    
    # Generate diverse initial population
    for _ in range(population_size - 1):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # GA loop
    generation = 0
    max_generations = 1000
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate current population
        fitness = evaluate_population(population, dx)
        
        # Track best
        current_best_idx = np.argmin(fitness)
        current_best_h = population[current_best_idx]
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check termination
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Selection
        num_select = population_size // 2
        selected = selection(population, fitness, num_select)
        
        # Create new population through crossover and mutation
        new_population = [current_best_h.copy()]
        
        # Crossover for pairs
        for i in range(0, len(selected) - 1, 2):
            child1, child2 = crossover(selected[i], selected[i+1])
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            new_population.append(child1)
            new_population.append(child2)
            
        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < 0.1:  # 10% mutation rate
                new_population[i] = mutate(new_population[i])
                
        # Local search for top candidates
        fitness_new = evaluate_population(new_population, dx)
        top_indices = np.argsort(fitness_new)[:10]  # Top 10
        for i in top_indices:
            if i < len(new_population):
                new_population[i] = local_search(new_population[i], dx)
                
        # Replace worst with new generation
        worst_indices = np.argsort(fitness)[-20:]  # Replace worst 20%
        for i, worst_idx in enumerate(worst_indices):
            if i < len(new_population):
                population[worst_idx] = new_population[i]
    
    # Final evaluation
    if best_h is None:
        # Fallback to best from initial population
        fitness = evaluate_population(population, dx)
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
