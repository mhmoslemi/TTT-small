import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute overlap integral for all shifts
    n = len(h)
    max_overlap = 0
    # Using manual correlation to avoid boundary issues
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
    else:
        # Create a symmetric initial value
        h_init = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 4
        h_init[mid - width:mid + width] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    def objective(x):
        # x contains [a, d] where h is 1 on [a, a+w] and [a+w+d, a+w+d+w]
        # but we map it to a 70-point step function
        a, d = x
        w = 0.5  # Fixed width to satisfy integral constraint
        h = np.zeros(n_points)
        a_idx = int(round(a * n_points / 2))
        d_idx = int(round(d * n_points / 2))
        
        # First interval [a, a+w] => [a_idx, a_idx + w*n_points/2]
        h_start1 = a_idx
        h_end1 = min(a_idx + int(w * n_points / 2), n_points)
        if h_end1 > h_start1:
            h[h_start1:h_end1] = 1.0
        
        # Second interval [a+w+d, a+w+d+w] => [a_idx+w*n_points/2+d*n_points/2, a_idx+w*n_points/2+d*n_points/2+w*n_points/2]
        h_start2 = h_end1 + d_idx
        h_end2 = min(h_start2 + int(w * n_points / 2), n_points)
        if h_end2 > h_start2:
            h[h_start2:h_end2] = 1.0
        
        # Normalize to integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        return evaluate_c5(h, dx)
    
    # Bounded optimization
    bounds = [(0.0, 1.0), (0.0, 0.5)]  # a in [0,1], d in [0, 0.5]
    x0 = [0.25, 0.25]  # Initial guess
    
    # Try a few starting points and best of multiple local searches
    best_result = None
    best_c5 = float('inf')
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        x0 = [np.random.uniform(0.0, 1.0), np.random.uniform(0.0, 0.5)]
        try:
            res = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 50})
            if res.success:
                c5 = res.fun
                if c5 < best_c5:
                    best_c5 = c5
                    best_result = res
        except:
            continue
    
    # If no improvement, return initial h
    if best_result is None:
        h_final = h_init
        c5_final = evaluate_c5(h_init, dx)
    else:
        # Reconstruct h from best result
        a, d = best_result.x
        w = 0.5
        h_final = np.zeros(n_points)
        a_idx = int(round(a * n_points / 2))
        d_idx = int(round(d * n_points / 2))
        
        # First interval [a, a+w] => [a_idx, a_idx + w*n_points/2]
        h_start1 = a_idx
        h_end1 = min(a_idx + int(w * n_points / 2), n_points)
        if h_end1 > h_start1:
            h_final[h_start1:h_end1] = 1.0
        
        # Second interval [a+w+d, a+w+d+w]
        h_start2 = h_end1 + d_idx
        h_end2 = min(h_start2 + int(w * n_points / 2), n_points)
        if h_end2 > h_start2:
            h_final[h_start2:h_end2] = 1.0
        
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5_final = evaluate_c5(h_final, dx)
    
    # Final evaluation
    return (h_final.tolist(), c5_final, n_points)
