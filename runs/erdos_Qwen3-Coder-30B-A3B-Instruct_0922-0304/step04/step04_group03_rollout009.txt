import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Default initialization: triangle shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization
    max_iter = 5000
    learning_rate = 0.01
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap_val = overlap[max_overlap_idx] * dx
        
        # For a shift k, gradient is 1 - h[i+k] - h[i-k]
        # We need to consider the full convolution
        # But for practical optimization, minimize the worst-case overlap
        
        # Compute gradients for each point
        grad = np.zeros(n_points)
        # We need to compute the gradient of max overlap with respect to h
        # This requires computing the gradient for each shift k in the overlap
        
        # For shift k:
        # We compute grad[i] = (1 - h[(i+k)%n_points]) - h[(i-k)%n_points]
        # But we need to compute this for the shift that maximizes overlap
        max_k = max_overlap_idx - n_points + 1  # convert full convolution index to shift
        for i in range(n_points):
            # Add contributions for the current max overlap shift
            k = max_k
            if i + k >= 0 and i + k < n_points:
                grad[i] += -h[(i + k) % n_points]
            if i - k >= 0 and i - k < n_points:
                grad[i] += -h[(i - k) % n_points]
            # Add the 1 from 1 - h[i-k]
            if i - k >= 0 and i - k < n_points:
                grad[i] += 1
            
        # Update h in direction of steepest descent
        h_new = h - learning_rate * grad
        
        # Clamp to unit interval
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Update if better
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new.copy()
        else:
            # Reduce learning rate if not improving
            learning_rate *= 0.995
            
    return (best_h.tolist(), best_c5, n_points)
