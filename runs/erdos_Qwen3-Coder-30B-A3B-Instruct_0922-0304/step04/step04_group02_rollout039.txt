import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # ES parameters
    pop_size = 100
    num_generations = 1000
    mutation_strength = 0.1
    elitism_rate = 0.2
    elite_count = int(pop_size * elitism_rate)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        mutated = h + np.random.normal(0, mutation_strength, n_points)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        population.append(mutated)
    
    # ES iterations
    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            fit = evaluate_c5(individual, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Create new generation
        new_pop = population[:elite_count]
        
        # Generate offspring
        for _ in range(pop_size - elite_count):
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_count, 2, replace=False)
            parent = population[idx1] if fitness[idx1] < fitness[idx2] else population[idx2]
            
            # Mutation
            child = parent + np.random.normal(0, mutation_strength, n_points)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_pop.append(child)
        
        population = new_pop
        
        # Adjust mutation strength
        if gen % 50 == 0 and gen > 0:
            improvement = (fitness[0] - best_c5) if gen > 50 else 0
            if improvement < 0:
                mutation_strength *= 1.1
            else:
                mutation_strength *= 0.9
                
            mutation_strength = max(0.001, min(0.5, mutation_strength))
    
    # SLSQP refinement of best solution
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
