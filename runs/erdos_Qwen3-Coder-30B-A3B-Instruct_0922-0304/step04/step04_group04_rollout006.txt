import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Generate basis functions
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size))
    for i in range(basis_size):
        if i % 2 == 0:
            phi[:, i] = np.cos(np.pi * (i // 2 + 1) * x)
        else:
            phi[:, i] = np.sin(np.pi * ((i + 1) // 2) * x)
    
    # Compute integral of basis functions
    int_phi = np.zeros(basis_size)
    for i in range(basis_size):
        int_phi[i] = np.sum(phi[:, i]) * dx
    
    # Initialize coefficients
    c = Variable(basis_size)
    
    # Precompute the kernel matrix K
    K = np.zeros((basis_size, basis_size, 2 * n_points - 1))
    for i in range(basis_size):
        for j in range(basis_size):
            # Computing integral of phi_i(x)*phi_j(x+k)
            for lag in range(2 * n_points - 1):
                k = lag - n_points + 1
                if k < 0:
                    val = np.sum(phi[:n_points+k, i] * phi[-k:, j]) * dx
                elif k >= 0:
                    val = np.sum(phi[k:, i] * phi[:n_points-k, j]) * dx
                K[i, j, lag] = val
    
    # Define variables for McCormick relaxation
    z = Variable((basis_size, basis_size))
    
    # Objective
    max_lag = 2 * n_points - 1
    objective_list = []
    for lag in range(max_lag):
        obj = 0
        for i in range(basis_size):
            for j in range(basis_size):
                obj += z[i,j] * K[i,j,lag]
        objective_list.append(obj)
    
    # Constraint that the integral equals 1
    integral_constraint = 2 * c[0] + sum(int_phi[i] * c[i] for i in range(1, basis_size)) == 1
    
    # Define constraints for McCormick relaxation
    constraints = []
    for i in range(basis_size):
        for j in range(basis_size):
            constraints.append(z[i,j] >= c[i] + c[j] - 1)
            constraints.append(z[i,j] <= c[i])
            constraints.append(z[i,j] <= c[j])
    
    # Add bounds on c
    for i in range(basis_size):
        constraints.append(c[i] >= 0)
    
    # Add integral constraint
    constraints.append(integral_constraint)
    
    # Add McCormick domain
    for i in range(basis_size):
        for j in range(basis_size):
            constraints.append(z[i,j] >= 0)
            
    # Create problem
    max_overlap = Variable()
    objective = Minimize(max_overlap)
    
    # Add constraints for max overlap
    for i, obj in enumerate(objective_list):
        constraints.append(obj <= max_overlap)
        constraints.append(-obj <= max_overlap)
    
    # Solve with CVXPY
    prob = Problem(objective, constraints)
    
    if 'initial_h_values' in globals():
        # Use the initial values to initialize the solution
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Compute Fourier coefficients
        c_init = (phi.T @ h_init) * dx
        
        # Set initial value
        c.value = c_init.copy()
    else:
        c.value = np.zeros(basis_size)  # Default initialization
        
    try:
        prob.solve(solver=SCS, max_iters=10000, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            c_opt = c.value
        else:
            c_opt = c.value
    except Exception:
        pass
    
    # Generate final h
    h = phi @ c_opt
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    c5_bound = evaluate_c5(h, dx)
    
    # Cleanup
    h = h.tolist()
    return (h, c5_bound, n_points)
