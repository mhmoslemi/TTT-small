import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, dx):
    """Project h to satisfy constraint sum(h) * dx = 1"""
    h_sum = np.sum(h)
    if abs(h_sum * dx - 1.0) > 1e-10:
        h = h * (1.0 / (h_sum * dx))
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Start with triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    h = project_h(h, dx)
    
    # Gradient-based optimization
    max_iter = 20000
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps for all shifts
        overlaps = correlate(h, 1 - h, mode="full")
        
        # Find the worst-case overlap (maximum)
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        # Compute gradient of the worst overlap with respect to h
        # The gradient dO_k_max/dh_i 
        # = (1 - h_{(i + k) mod n_points}) - h_{(i - k) mod n_points}  
        n_full = 2 * n_points - 1
        k_shift = max_idx - n_points + 1  # the shift k that corresponds to max_idx
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            # For shift k, O_k = sum_{j} h_i * (1 - h_{i+k})
            # dO_k/dh_i = 1 - h_{i+k} if we're looking at positive i
            # But we have to do this for right shift (the way correlate works)
            # i+shift position
            if i + k_shift >= 0 and i + k_shift < n_points:
                grad[i] = 1.0 - h[i + k_shift]
            else:
                grad[i] = 0.0
            
            # And negative term h_{i-k}  
            # which corresponds to index i-k where k is the shift
            if i - k_shift >= 0 and i - k_shift < n_points:
                grad[i] -= h[i - k_shift]
            else:
                grad[i] -= 0.0
                
        # Update velocity (momentum)
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project to feasible region
        h = project_h(h, dx)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Adaptive learning rate decay
        if iter_num > 0 and iter_num % 1000 == 0:
            learning_rate *= 0.95
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
