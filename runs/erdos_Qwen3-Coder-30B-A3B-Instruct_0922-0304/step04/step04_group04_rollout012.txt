import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier basis functions for a smooth representation
    n_basis = 30
    x = np.linspace(0, 2, n_points)
    
    # Compute Fourier basis functions (sine/cosine)
    def fourier_basis(x, n):
        basis = []
        for i in range(1, n+1):
            if i % 2 == 1:
                basis.append(np.sin(i * np.pi * x / 2))
            else:
                basis.append(np.cos(i * np.pi * x / 2))
        return np.array(basis).T
    
    # Create basis matrix
    Phi = fourier_basis(x, n_basis)
    
    # Compute the integral constraint matrix
    integral_constraint = np.mean(Phi, axis=0)
    
    # Optimization variables
    a = cp.Variable(n_basis)
    z = cp.Variable((n_basis, n_basis), symmetric=True)
    
    # McCormick envelopes for each product term
    # z[i,j] >= a[i] + a[j] - 1, z[i,j] <= a[i], z[i,j] <= a[j]
    McCormick = []
    for i in range(n_basis):
        for j in range(i, n_basis):
            # Create McCormick inequalities
            z_ij = z[i, j]
            McCormick.append(z_ij >= a[i] + a[j] - 1)
            McCormick.append(z_ij <= a[i])
            McCormick.append(z_ij <= a[j])
    
    # Setup the integral constraint
    integral_con = cp.Equality(integral_constraint @ a, 1.0)
    
    # Define maximum overlap over all lags
    # We'll approximate it using a grid of lag values
    overlap_coeffs = []
    for k in range(2 * n_points):
        # Compute the kernel for shift k
        shifted_phi = fourier_basis(x + k * dx, n_basis)
        coeffs = np.zeros((n_basis, n_basis))
        for i in range(n_basis):
            for j in range(n_basis):
                coeffs[i, j] = np.dot(Phi[:, i], shifted_phi[:, j])
        overlap_coeffs.append(coeffs)
    
    # Objective constraints: capture max overlap
    max_overlap = cp.Variable()
    objective_exprs = []
    for k in range(2 * n_points):
        # For each lag, constraint the sum of products with z
        expr = cp.trace(overlap_coeffs[k] @ z)
        objective_exprs.append(expr)
    
    # The maximum overlap is upper bounded by max_overlap
    # We want to minimize max_overlap, so add constraint
    constraints = [cp.max(objective_exprs) <= max_overlap] + McCormick + [integral_con]
    
    # Objective is minimizing max_overlap
    objective = cp.Minimize(max_overlap)
    
    # Solve with CVXPY
    prob = cp.Problem(objective, constraints)
    best_score = float('inf')
    best_h = h_init.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Warm start with current h
        try:
            # Project h to Fourier coefficients
            h_coeffs = np.array([np.sum(h_init * Phi[:, i]) * dx for i in range(n_basis)])
            a.value = h_coeffs
            prob.solve(solver=cp.MOSEK, verbose=False, warm_start=True)
            
            if prob.status == cp.OPTIMAL:
                a_val = a.value
                # Reconstruct h(x)
                h_reconstructed = Phi @ a_val
                h_reconstructed = np.clip(h_reconstructed, 0, 1)
                h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
                
                # Evaluate the actual overlap
                score = evaluate_c5(h_reconstructed, dx)
                if score < best_score:
                    best_score = score
                    best_h = h_reconstructed.copy()
                    
                # Early termination if good enough
                if best_score <= 0.3807:
                    break
                    
        except Exception as e:
            pass  # Skip iteration if solver fails
            
        # Fallback optimization using gradient-based method from previous solution
        # Direct search with small perturbations on basis
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        score = evaluate_c5(h_perturbed, dx)
        if score < best_score:
            best_score = score
            best_h = h_perturbed.copy()
    
    return (best_h.tolist(), best_score, n_points)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
