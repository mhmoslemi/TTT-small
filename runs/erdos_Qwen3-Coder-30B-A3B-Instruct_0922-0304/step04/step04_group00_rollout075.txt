import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=0.02):
    """Compute the maximum overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    
    # Initialize population around initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        # Fallback initialization
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h += np.random.normal(0, 0.05, n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(population_size):
        # Start near the initial solution with some noise
        h = initial_h.copy() + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
        p_best.append(h.copy())
        p_best_scores.append(compute_c5(h, dx))
    
    # Global best
    global_best_idx = np.argmin(p_best_scores)
    g_best = p_best[global_best_idx].copy()
    g_best_score = p_best_scores[global_best_idx]
    
    # PSO iterations
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = compute_c5(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
                    
                    # Early exit condition
                    if g_best_score <= 0.3807:
                        break
        
        # Early exit
        if g_best_score <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_score, n_points)
