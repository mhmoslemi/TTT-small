import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """
    Solve quadratic program: minimize 0.5 * delta^T * H * delta + c^T * delta
    subject to A * delta = b and delta_low <= delta <= delta_high
    """
    n = len(c)
    delta = cp.Variable(n)
    
    # Create quadratic objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [
        A @ delta == b,
        delta >= delta_low,
        delta <= delta_high
    ]
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Iteration parameters
    max_iter = 500
    reg = 1e-5
    alpha = 0.5
    
    # Consistency checks for constraints
    A_eq = np.ones((1, n_points))
    b_eq = np.array([n_points / 2])
    delta_bounds = np.ones(n_points) * 1e10
    delta_low = -h
    delta_high = 1.0 - h
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate overlaps using full correlation
        # But we compute only the maximum overlap value for current h
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap_val = overlap[max_overlap_idx]
        
        # Reconstruct gradient & Hessian efficiently for the current h
        # Gradient: dx * (1 - 2*h - shift_average)
        # Hessian: dx * ( -2*I + shift_matrix ) 
        # Use finite difference approach to approximate gradient/Hessian
        
        # Gradient approx(Vec) - background: real gradient for current h is 
        # c_i = dx * (1 - 2*h[i] - A_i) for each i, where A_i is avg of h[i:] and h[:i]
        # Since overlap involves shifts:
        
        # Get all shifted versions and sum
        shifts = []
        for k in range(2 * n_points - 1):
            shift = np.zeros(n_points)
            if k < n_points:
                shift[k:] = h[:n_points-k]
            else:
                shift[:n_points-(k-n_points+1)] = h[k-n_points+1:]
            shifts.append(shift)

        # Construct average over all shifts
        shift_avg = np.array(shifts).mean(axis=0)
        grad = dx * (1 - 2 * h - shift_avg)
        
        # Hessian approximation via second order finite difference about h 
        # Approximate Hess = dx * (-2*I + shift_matrix) where shift_matrix(i,j) represents how much shift contributes
        # Quick matrix building (slightly inaccurate but more efficient in small scales)
        # Similar to approx. with finite differences
        hess = np.zeros((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):  
                hess[i,j] = dx * (-2 if i==j else 0)
                # Add contribution from shifts
                # For cross terms from creating h(i)(1-h(i+k))
                if j >= i and j-i < n_points:
                    hess[i,j] += dx
                elif j < i and i-j < n_points:
                    hess[i,j] += dx
                
        # Regulate Hessian
        hess += reg * np.eye(n_points)
        
        # Solve QP with cvxpy
        delta = _solve_qp(hess, grad, A_eq, b_eq, delta_low, delta_high)
        
        # If infeasible, increase reg and retry once
        if delta is None:
            reg *= 10.0
            hess = np.zeros((n_points, n_points))
            for i in range(n_points):
                for j in range(n_points):
                    hess[i,j] = dx * (-2 if i==j else 0)
                    if j >= i and j-i < n_points:
                        hess[i,j] += dx
                    elif j < i and i-j < n_points:
                        hess[i,j] += dx
            hess += reg * np.eye(n_points)
            delta = _solve_qp(hess, grad, A_eq, b_eq, delta_low, delta_high)
        
        if delta is not None:
            # Line search magnitude update
            # Try different steps until convergence or 10 tries
            step_size = 1.0
            temp_h = h + step_size * delta
            # Enforce clipping
            temp_h = np.clip(temp_h, 0, 1)
            # Normalize again for sum
            temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
            # Evaluate
            temp_c5 = evaluate_c5(temp_h, dx)
            
            # Backtracking line search for alpha
            tries = 10
            while tries > 0 and temp_c5 >= best_c5:
                step_size *= alpha
                temp_h = h + step_size * delta
                temp_h = np.clip(temp_h, 0, 1)
                temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
                temp_c5 = evaluate_c5(temp_h, dx)
                tries -= 1
                
            if temp_c5 < best_c5:
                h = temp_h.copy()
                best_c5 = temp_c5
                best_h = h.copy()
                
                if best_c5 <= 0.3807:
                    break
        else:
            # No solution found: maybe too strict constraints
            if reg < 1e-1:
                reg *= 10.0
        
        # Periodic improvement
        if iteration % 20 == 0:
            # Minor random walk for escape
            perturbs = np.random.normal(0, 0.01, n_points)
            h += perturbs
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
