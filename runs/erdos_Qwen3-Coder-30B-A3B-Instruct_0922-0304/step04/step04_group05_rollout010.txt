import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    return h

def compute_gradient(h, dx, n_points):
    """Compute gradient of the overlap integral"""
    # For each position, compute the gradient contribution
    # This is a simplified version: ideally would compute all shifts
    # We focus on main contribution to make it efficient
    grad = np.zeros(n_points)
    
    # Compute approximate gradient based on shifted overlaps
    # We do this approximately by evaluating the main overlap structure
    # and using the fact that for shift k, gradient of h[i]*h[i+k] 
    # with respect to h[i] is h[i+k], and h[i] is 1-h[i+k] etc.
    # Final gradient approximation for overlap integral:
    # d/dh[i] sum_j h[j](1-h[j+k]) = (1 - 2h[i]) - average of h constrained by shift
    
    # More compact approximation:
    # gradient should be computed in a way that:
    # d/dh[i] sum_j h[j](1-h[j+k]) for shift k = 1-h[i+k] - h[i] 
    # But we should look at overlapping contributions
    
    # Take a more practical approach - 
    # Gradient ∇C_k = dx * (1 - 2*h - shift_average) where shift_average is  
    # for shifts that include the point i
    # We'll compute this numerically with correction
    # Make sure it handles circular nature
    
    # Approximate using the main nonlinearity: ∇C = dx*(N - 2*h - shift_terms)  
    # Shift terms being weighted average of h shifted by k over all k for h[i]
    
    # Two direct approaches: 
    # Approach 1: Compute overlaps for all shifts (EXACT) 
    # Approach 2: Compute finite differences in order to bail 
    # We'll go with full gradient via convolution as that's more stable.
    
    # We compute the actual gradient as the sum of partials
    # Each term in actual gradient of objective (max over shifts) is quite involved
    
    # Given complexity of exact gradient, we use a decent approximation here:
    # ∇C ≈ dx * (1 - 2*h - average_shifted_h) 
    # The approximation: for the sum over all shifts k,
    # derivative w.r.t. h[i] is the number of time h[i] appears in sum 
    # minus twice h[i] (due to -h[i]*h[i+k] terms) 
    # So we estimate this as turning it into a mechanistic solver over overlaps
    
    # Simpler workaround:
    # Get all shifts as needed and build the gradient approximately as:
    # 1 - 2*h with correction (approximate) 
    # (We're taking shortcut due to energy budget)
    
    # But the true canyon is that in C_k = sum h[i](1 - h[i+k]), we want 
    # ∂C_k/∂h[i] = 1 - h[i+k] - h[i] (only in possibly active contributions)
    # The maximum is over k, the change in max is tricky. 
    # We're doing a simple gradient ascent on some estimated max.
    
    # Let's compute the best shift that's currently contributing:
    overlaps = np.zeros(n_points * 2 - 1)  # full correlation result
    for k in range(n_points):
        if k == 0:
            # k=0: shift=0, constant term is h[i] + (-1)*h[i]*h[i] term
            term = np.array([1] * n_points)
        else:
            term = np.zeros(n_points) 
            term[k:] = 1  # shift k to the right

    # Instead: skip to optimization with line search with feasible versions

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 1000

    # Initialize h using initial_h_values if available; otherwise triangular
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / mid)
        h = h * target_sum / np.sum(h)
    
    # Initial constraints check and normalization
    h = project_to_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Test decrease from start
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)

    # Simple quasi-Newton scheme we'll perform line search 
    # Focus on fast adaptive learning and safety
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute approximate gradient  
        # A simpler model that just samples overlap values along shifted h
        # Much faster...
        gradient_approx = np.zeros(n_points)
        
        # Get all correlation values for a series of shifts
        # We sample few main shifts and license getting approximate gradient
        shifts = np.arange(1, min(5, n_points))
        moments = np.zeros(n_points)
        for s in shifts:
            # Gain insight from shifts: the overlap h[i](1-h[i+s]) becomes 
            # gradient term is -h[i+s] when i transits forward by s
            # and 1-h[i] for positive shift:
            # ∂h[j]/∂h[i] = 1 if j==i, 0 otherwise  
            # ∂(h[j](1-h[j+s]))/∂h[i] = (1-h[i+s]) if j==i, -h[i] if i+s==j
            # Overall ∂C_k/∂h[i] = 1 - h[i+s] - h[i]
            # then max_k sums over all instantiations which is approximate
            # We use current h to estimate an average based on shifted reflection:
            reflected = np.zeros(n_points)
            reflected[s:] = h[:-s]
            moments += (1 - reflected - h) * dx
            
        moments = np.nan_to_num(moments, nan=0)
        
        # Feature-based ascent step
        # Use small maximum descent step
        step = 0.01
        delta = -step * moments
        
        # Project update into feasible bound  
        h_new = h + delta
        
        # Ensure bounds hold (project to [0,1])
        h_new = np.clip(h_new, 0, 1)
        # And sum constraint
        h_new = h_new * target_sum / np.sum(h_new) 

        # Evaluate
        current_c5 = evaluate_c5(h_new, dx)
        
        # Accept sift if better
        if current_c5 < best_c5:
            h = h_new.copy()
            best_c5 = current_c5
            best_h = h.copy() 
            
            if best_c5 <= 0.3807:
                break
        else:
            # Try to improve by line search step
            # One way is to decay learning rate
            step *= 0.95
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
