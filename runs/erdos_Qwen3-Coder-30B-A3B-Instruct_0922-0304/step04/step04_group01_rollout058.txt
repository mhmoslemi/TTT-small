import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using modified gradient descent with adaptive steps
    max_iter = 1500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        
        # Compute gradient
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Adaptive step size
        step = 0.002 * (1.0 + 0.01 * iteration)
        
        # Update h
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Check if solution improved
        current_c5 = max_overlap
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Occasionally add noise to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = best_c5
    return (best_h.tolist(), final_c5, n_points)
