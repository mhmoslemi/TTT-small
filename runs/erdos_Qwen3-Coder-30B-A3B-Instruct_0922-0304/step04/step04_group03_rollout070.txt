import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric piecewise constant function
        h = np.zeros(n_points)
        # Set two intervals of width 0.5 each
        interval_width = 0.5
        idx1_start = 0
        idx1_end = int(interval_width / dx)
        h[idx1_start:idx1_end] = 1.0
        
        # Find appropriate position for second interval
        d = 0.5  # Distance between intervals
        idx2_start = int((0.5 + d) / dx)
        idx2_end = int((1.0 + d) / dx)
        if idx2_start < n_points:
            h[idx2_start:min(idx2_end, n_points)] = 1.0
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    # Try to improve using grid search of interval positions
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search over interval positions
    d_candidates = np.arange(0.0, 1.0 + 0.001, 0.01)
    
    # For each candidate interval separation
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create h with two intervals
        h_test = np.zeros(n_points)
        
        # First interval [0, 0.5]
        idx1_start = 0
        idx1_end = int(0.5 / dx)
        
        # Second interval [0.5+d, 1.0+d] 
        idx2_start = int((0.5 + d) / dx)
        idx2_end = int((1.0 + d) / dx)
        
        # Make sure indices don't exceed boundary
        if idx1_end >= n_points:
            idx1_end = n_points - 1
        if idx2_start < n_points:
            if idx2_end >= n_points:
                idx2_end = n_points - 1
            h_test[idx1_start:idx1_end] = 1.0
            h_test[idx2_start:idx2_end] = 1.0
        else:
            # Handle case where second interval goes over the boundary
            h_test[idx1_start:idx1_end] = 1.0
            # Add to the beginning
            h_test[0:idx2_end - n_points] = 1.0
            
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5 = evaluate_c5(h_test, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
    
    # Additional local refinement
    # Try small perturbations around best solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb interval separation
        d_pert = best_h[0] + np.random.normal(0, 0.01)  # This is a very approx hack
        d_pert = np.clip(d_pert, 0.0, 1.0)
        h_test = np.zeros(n_points)
        idx1_start = 0
        idx1_end = int(0.5 / dx)
        idx2_start = int((0.5 + d_pert) / dx)
        idx2_end = int((1.0 + d_pert) / dx)
        
        if idx2_start < n_points:
            h_test[idx1_start:idx1_end] = 1.0
            h_test[idx2_start:min(idx2_end, n_points)] = 1.0
            
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5 = evaluate_c5(h_test, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
            
        # Also try more random modifications if close to target
        if best_c5 < 0.381:
            for i in range(100):
                if time.time() - start_time > budget_s - 1:
                    break
                h_temp = best_h.copy()
                # Randomly modify some values
                idx = np.random.randint(0, n_points)
                h_temp[idx] = np.clip(h_temp[idx] + np.random.normal(0, 0.05), 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                c5 = evaluate_c5(h_temp, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_temp.copy()

    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
