import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the parent construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (70 / 2) / np.sum(h_current)
    else:
        # Initialize with a simple step function
        h_current = np.zeros(70)
        h_current[:35] = 1.0
        h_current = h_current * (70 / 2) / np.sum(h_current)
    
    # Use a simpler optimization approach: gradient descent with constraints
    # Start with simple alternating optimization
    
    n_points = 70
    dx = 2.0 / n_points
    learning_rate = 0.001
    max_iter = 10000
    
    # Compute initial value
    current_c5 = evaluate_c5(h_current, dx)
    best_c5 = current_c5
    best_h = h_current.copy()
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus[i] = max(0.0, h_minus[i] - eps)
            
            # Normalize 
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient descent
        h_new = h_current - learning_rate * grad
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce constraint sum(h) * dx = 1, so sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_current = h_new
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h_current, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
