import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    # Compute the overlap integral
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * 0.02  # dx = 2 / 100 = 0.02

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Particle Swarm Optimization parameters
    n_particles = 50
    max_iter = 10000
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy() + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.rand(n_points)
            particle[:n_points//2] = 1
            particle[n_points//2:] = 0
            particle += np.random.normal(0, 0.01, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        p_best.append(particle.copy())
        p_best_vals.append(compute_c5(particle))
    
    # Find global best
    global_best_idx = np.argmin(p_best_vals)
    g_best = p_best[global_best_idx].copy()
    g_best_val = p_best_vals[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -0.01, 0.01)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize particle to satisfy sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            current_val = compute_c5(particles[i])
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best if needed
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
                    
        # Early exit if target achieved
        if g_best_val <= 0.3807:
            break
            
    return (g_best.tolist(), g_best_val, n_points)
