import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolutionary Strategy parameters
    pop_size = 100
    elite_size = int(0.2 * pop_size)
    mutation_strength = 0.1
    max_iter = 1000
    
    # Generate initial population
    population = []
    for _ in range(pop_size):
        individual = h + np.random.normal(0, mutation_strength, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Main ES loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate all individuals
        fitness = []
        for individual in population:
            fit = evaluate_c5(individual, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        fitness = [fitness[i] for i in sorted_indices]
        population = [population[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Selection: keep elite
        selected = population[:elite_size]
        
        # Generate offspring via mutation
        offspring = []
        for i in range(pop_size - elite_size):
            # Select parent from top 50%
            parent_idx = np.random.randint(0, pop_size // 2)
            parent = population[parent_idx]
            
            # Mutate
            mut_strength = mutation_strength
            mutated = parent.copy()
            
            # Mutate about 40% of elements
            num_mut = max(1, int(0.4 * n_points))
            indices = np.random.choice(n_points, num_mut, replace=False)
            for idx in indices:
                mutated[idx] += np.random.normal(0, mut_strength)
            
            # Repair
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * (n_points / 2) / np.sum(mutated)
            
            offspring.append(mutated)
        
        population = selected + offspring
        
        # Adaptive mutation strength
        if iter_count % 10 == 0:
            improvement = max(0.01, 1.0 - (fitness[0] / best_c5))
            if improvement > 0.001:
                mutation_strength *= 0.95
            else:
                mutation_strength = min(0.5, mutation_strength * 1.1)
    
    # SLSQP refinement with best solution
    if time.time() - start_time < budget_s * 0.98:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
