import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    tolerance = 1e-6
    last_improvement = 0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from shift k
            j = (i + max_k) % n_points
            gradient[i] += (1 - h[j])
            j = (i - max_k) % n_points
            gradient[i] -= h[j]
        gradient *= dx
            
        # Update h
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iteration
            h = h_new.copy()
        else:
            # Reduce learning rate if no improvement for a while
            if iteration - last_improvement > 50:
                learning_rate *= 0.9
                last_improvement = iteration
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
