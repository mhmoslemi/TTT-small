import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Initialize particles around the initial solution
        for i in range(n_particles):
            particles[i] = np.clip(h_init + np.random.normal(0, 0.01, n_points), 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    else:
        # Use symmetric triangular window as base
        base_h = np.array([min(i, n_points - i) for i in range(1, n_points + 1)])
        base_h = base_h / np.sum(base_h) * (n_points / 2)
        for i in range(n_particles):
            particles[i] = np.clip(base_h + np.random.normal(0, 0.01, n_points), 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    
    # PSO parameters
    w = 0.9
    phi_p = 2.0
    phi_g = 2.0
    inertia_damping = 0.99
    
    # Initialize personal bests
    pbest_particles = particles.copy()
    pbest_values = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_values)
    gbest = particles[global_best_idx].copy()
    gbest_value = pbest_values[global_best_idx]
    
    best_c5 = gbest_value
    best_h = gbest.copy()
    
    # Optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            phi_p * r1 * (pbest_particles[i] - particles[i]) + 
                            phi_g * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_particles[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest = particles[i].copy()
                    if gbest_value < best_c5:
                        best_c5 = gbest_value
                        best_h = gbest.copy()
        
        # Periodic reinitialization to avoid stagnation
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):
                j = np.random.randint(0, n_particles)
                # Reinitialize with slight perturbation
                if 'initial_h_values' in globals():
                    particles[j] = np.clip(h_init + np.random.normal(0, 0.02, n_points), 0, 1)
                else:
                    base_h = np.array([min(k, n_points - k) for k in range(1, n_points + 1)])
                    base_h = base_h / np.sum(base_h) * (n_points / 2)
                    particles[j] = np.clip(base_h + np.random.normal(0, 0.02, n_points), 0, 1)
                particles[j] = particles[j] * (n_points / 2) / np.sum(particles[j])
                velocities[j] = np.random.uniform(-0.01, 0.01, n_points)
        
        # Damp inertia
        if iteration % 100 == 0 and iteration > 0:
            w *= inertia_damping
    
    return (best_h.tolist(), best_c5, n_points)
