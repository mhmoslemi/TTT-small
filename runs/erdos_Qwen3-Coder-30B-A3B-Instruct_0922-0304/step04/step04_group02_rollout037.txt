import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolutionary Strategy parameters
    pop_size = 100
    elite_size = int(0.2 * pop_size)
    mutation_strength = 0.1
    max_iter = 1000
    adaptive_threshold = 0.01
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        temp_h = h + np.random.normal(0, mutation_strength, n_points)
        temp_h = np.clip(temp_h, 0, 1)
        temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
        population.append(temp_h)
    
    # ES iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            fit = evaluate_c5(ind, dx)
            fitness.append(fit)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection and elitism
        elite = population[:elite_size]
        
        # Reproduction with mutation
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Select parent
            parent_idx = np.random.randint(0, elite_size)
            parent = population[parent_idx]
            
            # Mutate
            child = parent + np.random.normal(0, mutation_strength, n_points)
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        # Update population
        population = new_population
        
        # Adapt mutation strength based on improvement
        if iter_count > 0 and iter_count % 10 == 0:
            if abs(fitness[0] - fitness[-1]) < adaptive_threshold:
                mutation_strength *= 1.1  # Increase mutation if no improvement
            else:
                mutation_strength = max(0.01, mutation_strength * 0.9)  # Decrease mutation if improving
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
