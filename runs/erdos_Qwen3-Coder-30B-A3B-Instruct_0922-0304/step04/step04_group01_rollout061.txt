import numpy as np
from scipy.signal import correlate
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h with the initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a sawtooth pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i < n_points//2 else 0.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    eta = 0.003
    stagnation_count = 0
    max_stagnation = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current max overlap
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early termination if we've hit our target
        if best_c5 <= 0.3807:
            break
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
            
        # Update h using gradient descent
        h -= eta * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust step size
        if iteration % 200 == 0 and iteration > 0:
            eta *= 0.9
            
        # Occasionally add random perturbations to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Stagnation handling
        if iteration % 10 == 0:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
    
    final_c5 = float('inf')
    try:
        corr = correlate(best_h, 1 - best_h, mode="full")
        final_c5 = np.max(corr) * dx
    except:
        final_c5 = best_c5
    
    return (best_h.tolist(), final_c5, n_points)
