import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent optimization
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlap
        current_c5 = evaluate_c5(h, dx)
        
        # Find the most constraining shift k
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient for the most constraining shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # For shift k_max, the gradient term is (1 - h[i+k_max])
            # But we need to handle circular shift properly
            j = (i + k_max) % n_points
            gradient[i] = (1 - h[j]) / dx
        
        # Update
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Decrease learning rate
        if iter_num % 100 == 0 and iter_num > 0:
            learning_rate *= 0.9
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
