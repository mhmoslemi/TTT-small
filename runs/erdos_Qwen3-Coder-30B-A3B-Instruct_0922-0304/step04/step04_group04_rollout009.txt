import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Precompute basis functions
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size+1))
    phi[:, 0] = 1.0  # constant term
    
    for i in range(1, basis_size+1):
        if i % 2 == 1:  # sine terms
            phi[:, i] = np.sin(np.pi * (i+1)//2 * x)
        else:  # cosine terms
            phi[:, i] = np.cos(np.pi * i//2 * x)
    
    # Compute integral of basis functions
    int_phi = np.zeros(basis_size+1)
    for i in range(basis_size+1):
        int_phi[i] = np.sum(phi[:, i]) * dx
    
    # Setup initial guess
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        c_init = np.linalg.lstsq(phi, h_init, rcond=None)[0]
        # Adjust to meet constraint ∫h = 1
        c_init[0] = (1 - np.sum(c_init[1:] * int_phi[1:])) / int_phi[0]
        c_init = np.maximum(0, c_init)  # enforce non-negativity
    else:
        # Default: simple step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        c_init = np.linalg.lstsq(phi, h_init, rcond=None)[0]
        c_init[0] = (1 - np.sum(c_init[1:] * int_phi[1:])) / int_phi[0]
    
    # Solve using convex optimization with McCormick relaxations
    # Variables
    c = Variable(basis_size+1)
    
    # Constraints
    constraints = [
        c[0] >= 0,
        c[1:] >= 0,
        c[0] * int_phi[0] + sum(c[i] * int_phi[i] for i in range(1, basis_size+1)) == 1.0
    ]
    
    # Fit each term to h(x) in L2 sense as guide
    # Interpolate h(x) from initial guess to smooth function
    h_smooth = phi @ c_init  # Initial smooth approximation
    h_smooth = np.clip(h_smooth, 0, 1)  # Keep in [0,1]
    h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)  # normalize
    
    # Update c_init based on better smooth representation
    try:
        c_sol = np.linalg.lstsq(phi, h_smooth, rcond=None)[0]
        c_sol[0] = (1 - np.sum(c_sol[1:] * int_phi[1:])) / int_phi[0]
    except:
        c_sol = c_init
    
    c.value = c_sol
    
    # McCormick envelopes: z[i,j] = c[i]*c[j] 
    # Add convex relaxations via log-barrier or iterative tightening
    z = np.zeros((basis_size+1, basis_size+1))
    
    # Try to construct K matrix using local shifts
    # Instead of expanding K, just do one optimization trial with the constraint
    # on max overlap, modified by convex relaxation
    
    # Manual optimization with bounded convex relaxation constraints
    def compute_overlap(c_vec):
        h_func = phi @ c_vec
        h_func = np.clip(h_func, 0, 1)
        overlap = correlate(h_func, 1 - h_func, mode="full")
        return np.max(overlap) * dx
    
    # Start with simple gradient descent to refine solution
    max_iter = 10000
    lr = 0.01
    current_c = c_sol.copy()
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current score
        try:
            current_score = compute_overlap(current_c)
        except:
            continue

        # Compute gradients using small finite differences 
        grad = np.zeros_like(current_c)
        eps = 1e-5
        for i in range(len(current_c)):
            c_plus = current_c.copy()
            c_plus[i] += eps
            c_minus = current_c.copy()
            c_minus[i] -= eps
            try:
                score_plus = compute_overlap(c_plus)
                score_minus = compute_overlap(c_minus)
                grad[i] = (score_plus - score_minus) / (2*eps)
            except:
                continue
                
        # Update variables with gradient descent and projection
        current_c = current_c - lr * grad
        
        # Projection to feasible set (non-negative coefficients)
        current_c = np.maximum(0, current_c)
        
        # Ensure constraint ∫h(x)dx = 1
        h_func = phi @ current_c
        h_func = np.clip(h_func, 0, 1)
        h_func = h_func * (n_points / 2) / np.sum(h_func)
        # Update c to match h_func again (iteration via L2 projection)
        current_c = np.linalg.lstsq(phi, h_func, rcond=None)[0]
        # Adjust constant term to meet integral constraint
        current_c[0] = (1 - np.sum(current_c[1:] * int_phi[1:])) / int_phi[0]
         
        # Early termination
        if current_score <= 0.3807:
            break
    
    # Final refinement for best result
    final_c = current_c.copy()
    final_h = phi @ final_c
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c = np.linalg.lstsq(phi, final_h, rcond=None)[0]
    final_c[0] = (1 - np.sum(final_c[1:] * int_phi[1:])) / int_phi[0]
    
    # Compute final solution
    best_h = phi @ final_c
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_overlap = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), best_overlap, n_points)

def evaluate_overlap(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
