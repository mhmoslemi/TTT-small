import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent with adaptive step
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Use conjugate gradient-like updates
    grad = np.zeros(n_points)
    prev_grad = np.zeros(n_points)
    prev_h = h.copy()
    step_size = 0.01
    max_iter = 10000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current gradient approximation
        # We approximate the gradient of the overlap integral with respect to h
        # by finite differences of the overlap function
        
        # Compute overlap with current h
        overlap_current = correlate(h, 1 - h, mode="full") * dx
        
        # Estimate gradient by finite differences in each dimension
        eps = 1e-5
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Create perturbed h
            h_temp = h.copy()
            h_temp[j] += eps
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            overlap_pert = correlate(h_temp, 1 - h_temp, mode="full") * dx
            grad[j] = (np.max(overlap_pert) - np.max(overlap_current)) / eps
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
        else:
            # Reduce step size if didn't improve
            step_size *= 0.95
            
        # Early termination
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
