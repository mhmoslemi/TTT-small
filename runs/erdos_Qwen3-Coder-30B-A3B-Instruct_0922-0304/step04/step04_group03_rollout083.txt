import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy the constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent with projected updates
    max_iter = 20000
    lr = 0.01
    decay_factor = 0.999
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        
        # Find the worst-case shift that gives maximum overlap
        # The overlap array has length 2*n_points - 1, so the actual shift k
        # corresponds to k = max_idx - (n_points - 1) when max_idx >= n_points - 1
        # and k = max_idx - (n_points - 1) when max_idx < n_points - 1  
        # Actually, it's easier to work with the full overlap array
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient of overlap w.r.t h
        # This is the convolution of h with flipped (1-h) at shift k_max
        # But this simplifies to: for each i, dO_k_max/dh_i = (1-h[i+k_max]) + (1-h[i-k_max])
        # But let's be more precise: 
        # The overlap computation O_k = sum_i h[i] * (1 - h[i+k])
        # So dO_k/dh_i = 1 - h[i+k] when i is a valid index for h[i+k]
        # and similarly dO_k/dh_i = h[i-k] when valid index for h[i-k]
        # We'll compute the gradient at all points with a convolution approach
        # The gradient of O_k with respect to h_i is:
        # dO_k/dh_i = (1 - h[(i+k) % n_points]) + (1 - h[(i-k) % n_points])
        # But since we have O_k = sum_h[i] * (1 - h[i+k]), so 
        # dO_k/dh_i = 1 - h[(i+k) % n_points] - h[(i-k) % n_points] + (ha-ha) + ... 
        # If k = 0 (center alignment): dO_k/dh_i = 1 - h[i] - h[i] = 1 - 2*h[i]
        # Actually better to compute it as:
        # When you look at Overlap_k = sum_i h[i] * (1 - h[i + k]), for any i:
        # gradient of this term w.r.t h_i is (1 - h[i+k])
        # For the term h[i+k] * (1 - h[i]), the gradient w.r.t h_i is - h[i+k]
        # So we're making a convolution an approximation of this: 
        # Let's compute it directly for the maximum overlap
        # Gradient w.r.t h[i] for最大overlap: 
        # Compute O_k_max = sum h[i] * (1 - h[i+k_max])
        # dO_k_max/dh_i = (1 - h[i+k_max]) + (1 - h[i-k_max])  
        # We're essentially taking the discrete convolution with 1, -1, 1 pattern shifted
        # More accurately, dO_k_max/dh_i = 1 - h[i+k_max] - h[i-k_max]
        # After fixing the indexing:
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            k = (i - k_max) % n_points
            grad[i] = 1 - h[j] - h[k]
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project onto feasible region: [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint: sum(h) = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Evaluate the new value  
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive learning rate decay
        lr *= decay_factor
        
        # Early exit if target is achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
