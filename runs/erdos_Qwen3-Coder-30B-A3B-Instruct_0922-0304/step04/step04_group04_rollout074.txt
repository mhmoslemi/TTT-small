import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize with parent solution if available, otherwise create a reasonable starting point
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a reasonable initial function
        h_init = np.zeros(n_points)
        # Create a smooth step-like function
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 1.0
            elif x < 1.5:
                h_init[i] = 1.0 - (x - 0.5) / 1.0
            else:
                h_init[i] = 0.0
        # Normalize it to integral = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Use projected gradient descent with simple line search
    h_current = h_init.copy()
    best_h = h_current.copy()
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = np.max(correlate(h_current, 1 - h_current, mode="full")) * dx
        
        # Compute gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h_current)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            
            # Ensure bounds before correlation
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            c5_plus = np.max(correlate(h_plus, 1 - h_plus, mode="full")) * dx
            c5_minus = np.max(correlate(h_minus, 1 - h_minus, mode="full")) * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Simple gradient descent step
        step_size = 0.01
        h_new = h_current - step_size * grad
        
        # Project to [0,1] and then renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # If new is better, accept it
        new_c5 = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
        if new_c5 < current_c5:
            h_current = h_new
            if new_c5 < np.max(correlate(best_h, 1 - best_h, mode="full")) * dx:
                best_h = h_new.copy()
        else:
            # If not better, try small random perturbations for escape from local minimum
            h_perturb = h_current.copy()
            idx = np.random.randint(n_points)
            h_perturb[idx] = np.clip(h_perturb[idx] + np.random.normal(0, 0.02), 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            perturb_c5 = np.max(correlate(h_perturb, 1 - h_perturb, mode="full")) * dx
            if perturb_c5 < np.max(correlate(best_h, 1 - best_h, mode="full")) * dx:
                best_h = h_perturb.copy()
                h_current = h_perturb.copy()
    
    # Final validation of best solution
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
