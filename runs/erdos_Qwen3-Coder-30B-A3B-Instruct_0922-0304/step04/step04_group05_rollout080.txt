import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using cvxpy."""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective: (1/2) * delta^T * H * delta + c^T * delta
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [
        A @ delta == b,
        delta >= delta_low,
        delta <= delta_high
    ]
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    tol = 1e-6
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure sum constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Termination criteria
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        # Gradient components: 
        # ∂C_k/∂h_i = dx * (1 - 2*h_i - shift_average[i])
        # Hessian is more complex, simplified as linear approximation
        # ∇C = dx * (1 - 2*h - average_shifts)
        
        # Shift averages for each index
        shift_averages = np.zeros_like(h)
        for i in range(n_points):
            shift_sum = 0
            count = 0
            for k in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    shift_sum += h[j]
                    count += 1
            shift_averages[i] = shift_sum / count if count > 0 else 0
        
        # Approximate gradient (this is a crude estimate)
        # For a better approach, compute full gradient directly
        grad = dx * (1 - 2 * h - shift_averages)
        
        # Hessian approximation (real gradient would be more precise)
        # This is a rough approximation only
        H = np.zeros((n_points, n_points))
        np.fill_diagonal(H, -2 * dx)
        # Add small regularization
        H += 1e-5 * np.eye(n_points)
        
        # Equality constraint: sum(delta) = 0
        A_eq = np.ones((1, n_points))
        b_eq = np.array([0.0])
        
        # Boundary constraints
        delta_low = -h  # delta must be at least -h so that h + delta >= 0
        delta_high = 1 - h  # delta must be at most 1 - h so that h + delta <= 1
        
        # Solve QP
        delta = _solve_qp(H, grad, A_eq, b_eq, delta_low, delta_high)
        
        if delta is None:
            # If QP failed try with higher regularization
            H += 1e-3 * np.eye(n_points)
            delta = _solve_qp(H, grad, A_eq, b_eq, delta_low, delta_high)
            if delta is None:
                # Update h anyway with small step
                alpha = 0.1
                delta = alpha * grad / np.linalg.norm(grad) * 0.01
                if np.isnan(delta).any():
                    delta = np.zeros_like(h)
                
        # Line search
        alpha = 1.0
        step_size = 0.1
        new_h = h + alpha * delta
        
        # Project back to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        # Update h
        h = new_h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
