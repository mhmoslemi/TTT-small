import numpy as np
from scipy.fft import fft, ifft
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    """Evaluate the overlap integral."""
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

    basis_functions = []

        if k % 2 == 1:
            phi_k = lambda x, k=k: np.sin(np.pi * k * x)

        basis_functions.append(phi_k)

    inner_products = np.zeros((len(basis_functions), len(basis_functions)))
    x_vals = np.linspace(0, 2, n_points, endpoint=False)

    h_vals = h_init
    coeffs = np.linalg.lstsq(basis_evals.T, h_vals, rcond=None)[0]
