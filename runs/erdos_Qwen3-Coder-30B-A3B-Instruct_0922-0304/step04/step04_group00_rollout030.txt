import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise optimization
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            # Save current value
            old_h_i = h[i]
            
            # Try small perturbations
            perturbations = [-0.01, -0.005, 0.005, 0.01]
            
            best_h_i = old_h_i
            best_local_c5 = current_c5
            
            for delta in perturbations:
                new_h_i = old_h_i + delta
                new_h_i = np.clip(new_h_i, 0, 1)
                
                # Create candidate h
                candidate_h = h.copy()
                candidate_h[i] = new_h_i
                
                # Re-normalize to maintain constraint
                sum_h = np.sum(candidate_h)
                candidate_h = candidate_h * (n_points / 2) / sum_h
                
                # Compute C5
                candidate_c5 = compute_c5(candidate_h, dx)
                
                if candidate_c5 < best_local_c5:
                    best_local_c5 = candidate_c5
                    best_h_i = new_h_i

            # Re-normalize again after clipping
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
