import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    n = len(h)
    max_overlap = 0
    # Compute overlap for all shifts
    for k in range(2 * n - 1):
        overlap = 0
        for i in range(n):
            j = i + k
            if j < n:
                overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or create a piecewise constant function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a two-interval construction
        h = np.zeros(n_points)
        # Place two intervals of width 0.5
        w = 0.5
        # Start at position a = 0.1, end at position a + w = 0.6
        a = 0.1
        start1 = int(a / dx)
        end1 = int((a + w) / dx)
        # Place second interval after some gap
        d = 0.5
        start2 = int((a + w + d) / dx)
        end2 = int((a + w + d + w) / dx)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        
        # Adjust to satisfy integral constraint
        total_area = np.sum(h) * dx
        h = h * (1.0 / total_area)
        
    # Try to improve using optimization
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    def objective(params):
        # params: [a1, w1, a2, w2, d]
        a1, w1, a2, w2, d = params
        # Clip to enforce constraints
        a1 = max(0, min(2 - w1 - d - w2, a1))
        a2 = max(a1 + w1 + d, min(2 - w2, a2))
        w1 = max(0.001, min(1, w1))
        w2 = max(0.001, min(1, w2))
        d = max(0, min(1, d))
        
        h_new = np.zeros(n_points)
        start1 = int(a1 / dx)
        end1 = int((a1 + w1) / dx)
        start2 = int((a1 + w1 + d) / dx)
        end2 = int((a1 + w1 + d + w2) / dx)
        h_new[start1:end1] = 1.0
        h_new[start2:end2] = 1.0
        h_new = h_new * (1.0 / np.sum(h_new) * dx)  # normalize
        
        return evaluate_c5(h_new, dx)
    
    # Grid search over some parameters
    best_params = [0.1, 0.5, 0.6, 0.5, 0.2]
    n_grid = 20
    best_val = float('inf')
    for a1 in np.linspace(0, 1, n_grid):
        for w1 in np.linspace(0.2, 0.8, n_grid//2):
            for d in np.linspace(0, 1 - w1, n_grid//2):
                a2 = a1 + w1 + d
                w2 = 1 - (a2 - a1 - w1)
                if w2 > 0:
                    params = [a1, w1, a2, w2, d]
                    try:
                        val = objective(params)
                        if val < best_val:
                            best_val = val
                            best_params = params
                    except:
                        continue
        if time.time() - start_time > budget_s - 1:
            break
    
    # Optimization using scipy minimize
    # Initialize parameters
    a1, w1, a2, w2, d = best_params
    # Refine with optimization
    bounds = [(0, 2 - w1 - d - w2), (0.001, 1), (0, 2 - w2), (0.001, 1), (0, 1)]
    x0 = [a1, w1, a2, w2, d]
    
    # Factor optimization solving
    def opt_objective(params):
        a1, w1, a2, w2, d = params
        a1 = max(0, min(2 - w1 - d - w2, a1))
        a2 = max(a1 + w1 + d, min(2 - w2, a2))
        w1 = max(0.001, min(1, w1))
        w2 = max(0.001, min(1, w2))
        d = max(0, min(1, d))
        
        h_new = np.zeros(n_points)
        start1 = int(max(0, a1) / dx)
        end1 = int(min(2, a1 + w1) / dx)
        start2 = int(max(0, a1 + w1 + d) / dx)
        end2 = int(min(2, a1 + w1 + d + w2) / dx)
        if end1 > start1:
            h_new[start1:end1] = 1.0
        if end2 > start2:
            h_new[start2:end2] = 1.0
        if np.sum(h_new) > 0:
            h_new = h_new * (1.0 / np.sum(h_new) * dx)
        else:
            h_new = np.ones_like(h_new) * (1.0 / (n_points * dx))
        return evaluate_c5(h_new, dx)
    
    try:
        result = minimize(opt_objective, x0, bounds=bounds, method='L-BFGS-B', 
                         options={'maxiter': 100})
        if result.success:
            a1, w1, a2, w2, d = result.x
            h_opt = np.zeros(n_points)
            start1 = int(max(0, a1) / dx)
            end1 = int(min(2, a1 + w1) / dx)
            start2 = int(max(0, a1 + w1 + d) / dx)
            end2 = int(min(2, a1 + w1 + d + w2) / dx)
            if end1 > start1:
                h_opt[start1:end1] = 1.0
            if end2 > start2:
                h_opt[start2:end2] = 1.0
            if np.sum(h_opt) > 0:
                h_opt = h_opt * (1.0 / np.sum(h_opt) * dx)
            else:
                h_opt = np.ones_like(h_opt) * (1.0 / (n_points * dx))
            val_opt = evaluate_c5(h_opt, dx)
            if val_opt < best_c5:
                best_h = h_opt
                best_c5 = val_opt
    except Exception:
        pass
        
    # Ensure normalization 
    total_area = np.sum(best_h) * dx
    if total_area != 1.0:
        best_h = best_h * (1.0 / total_area)
    
    return (best_h.tolist(), best_c5, n_points)
