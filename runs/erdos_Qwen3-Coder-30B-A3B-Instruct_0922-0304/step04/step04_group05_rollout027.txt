import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.1
    damping = 0.5
    mu = 1e-3
    reg = 1e-8
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Early stopping criterion
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Iterative improvement using sequential convex programming
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Forward pass: compute ∇C and ∇²C
        # Approximate second-order derivatives for quadratic model
        # Use finite differences for computing gradients
        grad_C = np.zeros(n_points)
        h_conv = np.convolve(h, np.ones(2), mode='full')[:n_points]
        
        # For all k, ∫ h(x)(1 - h(x + k)) dx
        # Compute C_k = ∑ h[i] * (1 - h[i+k]) * dx for all valid k
        C = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points - i):
                C[i] += h[i] * (1 - h[i+k]) * dx
        
        # Gradient approximated from difference in overlaps
        # Use finite differences for gradient computation
        epsilon = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Normalize again
            C_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)  # Normalize again
            C_minus = evaluate_c5(h_minus, dx)
            
            grad_C[i] = (C_plus - C_minus) / (2 * epsilon)
        
        # Create Hessian approximation using identity regularization
        H = np.eye(n_points) * reg
        
        # For a better Hessian approximation, consider finite difference in gradients
        # But for simplicity, use identity added to a regularization term
        
        # Convex subproblem: minimize ∇C^T δ + (1/2)δ^T H δ
        # subject to: 0 ≤ h+δ ≤ 1, sum(h+δ) = n_points/2
        # Use cvxpy for solving this subproblem
        
        # Define variables
        delta = cp.Variable(n_points)
        h_var = h.copy()
        
        # Objective
        obj = cp.Minimize(cp.matmul(grad_C.T, delta) + 0.5 * cp.quad_form(delta, H))
        # Constraints
        constraints = [
            delta >= -h_var,
            delta <= 1 - h_var,
            cp.sum(delta) == 0  # Since we ensure ∑h = n_points/2
        ]
        
        # Create and solve the problem
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-8)
            delta_solution = delta.value
            
            # Update h: h = h + alpha * delta_solution
            h_candidate = h + alpha * delta_solution
            
            # Ensure bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Re-normalize to meet constraint exactly
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate new value
            c5_candidate = evaluate_c5(h_candidate, dx)
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
            
            # Line search - adjust step size
            if c5_candidate > best_c5:
                alpha *= damping
            else:
                alpha = min(1.0, alpha * 1.2)
                
            h = h_candidate
            
        except Exception:
            # Fall back on small fixed step if optimization fails
            h = h + alpha * np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5_new = evaluate_c5(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)

    return (best_h.tolist(), best_c5, n_points)
