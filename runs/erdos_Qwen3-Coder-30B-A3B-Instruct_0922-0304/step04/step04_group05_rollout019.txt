import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to meet constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population
    for i in range(1, pop_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Evolutionary strategy parameters
    sigma = 0.1
    success_count = 0
    success_threshold = pop_size // 4
    adaptation_rate = 0.2
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_c5(individual, dx) for individual in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        fitness = [fitness[i] for i in sorted_indices]
        population = [population[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Select top performing individuals
        elite_size = pop_size // 4
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size):
            # Select parent from elite
            parent = elite[np.random.randint(0, elite_size)]
            
            # Mutate
            noise = np.random.normal(0, sigma, n_points)
            mutated = parent + noise
            mutated = project_to_feasible(mutated, n_points)
            offspring.append(mutated)
        
        # Replace population
        population = offspring.copy()
        
        # Adaptive sigma
        success_count += 1
        if success_count >= success_threshold:
            # Increase mutation rate if improvement
            sigma *= 1.2
            success_count = 0
        else:
            # Decrease mutation rate if no improvement
            sigma *= 0.95
        
        # Clamp sigma to reasonable range
        sigma = np.clip(sigma, 0.001, 0.5)
    
    return (best_h.tolist(), best_c5, n_points)
