import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    step_size = 0.05
    T = T0
    iterations = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Record current state
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate neighbor
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-step_size, step_size)
        h_new[idx] += delta
        
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability
            prob = math.exp(-(new_c5 - current_c5) / T)
            if np.random.rand() < prob:
                h = h_new
                
        # Update temperature and step size
        T *= alpha
        step_size *= alpha
        
        # Occasionally perturb if we're stuck
        if iterations % 100 == 0 and T < 1e-6:
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            T = T0
            step_size = 0.05

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
