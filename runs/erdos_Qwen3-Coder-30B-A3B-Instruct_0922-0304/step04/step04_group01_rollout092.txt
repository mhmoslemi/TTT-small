import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for i in range(n_particles):
            particles[i] = h0 + np.random.normal(0, 0.01, n_points)
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    else:
        # Fallback to symmetric triangular window
        h0 = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i <= mid:
                h0[i] = i / mid
            else:
                h0[i] = 1 - (i - mid) / (n_points - mid)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        for i in range(n_particles):
            particles[i] = h0 + np.random.normal(0, 0.01, n_points)
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    
    # Track personal bests and global best
    pbest = particles.copy()
    pbest_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO parameters
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
        
        # Occasionally reinitialize particles to avoid stagnation
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                particles[idx] = np.random.random(n_points)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
        
        # Early exit if target achieved
        if gbest_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
