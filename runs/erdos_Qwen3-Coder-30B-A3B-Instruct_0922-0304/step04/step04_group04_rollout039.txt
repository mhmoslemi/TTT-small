import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h_current = h_init.copy()
    current_score = evaluate_c5(h_current, dx)
    
    # Try different optimization strategies
    best_score = current_score
    best_h = h_current.copy()
    
    # Strategy 1: Local optimization with gradient descent steps
    max_iter_gd = 1000
    learning_rate = 0.01
    
    for i in range(max_iter_gd):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradients for each point
        grad = np.zeros(n_points)
        h_forward = np.roll(h_current, -1)
        h_backward = np.roll(h_current, 1)
        
        # Approximate gradient by finite differences
        for j in range(n_points):
            if j == 0:
                h_diff = h_forward[j] - 1.0  # h[1] - 1
            elif j == n_points - 1:
                h_diff = 1.0 - h_backward[j]  # 1 - h[n-2]
            else:
                h_diff = h_forward[j] - h_backward[j]  # h[j+1] - h[j-1]
            grad[j] = h_diff
            
        # Update step
        new_h = h_current - learning_rate * grad
        
        # Ensure bounds and constraint
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_score = evaluate_c5(new_h, dx)
        
        if new_score < best_score:
            best_score = new_score
            best_h = new_h.copy()
            
        if new_score < current_score:
            h_current = new_h
            current_score = new_score
        else:
            learning_rate *= 0.99  # Decrease learning rate
            
        if best_score <= 0.3807:
            break
    
    # Strategy 2: Simple local perturbation search
    max_perturbations = 5000
    for i in range(max_perturbations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate perturbated solution
        h_perturbed = h_current.copy()
        idx = np.random.randint(n_points)
        perturbation = np.random.uniform(-0.05, 0.05)
        h_perturbed[idx] += perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        new_score = evaluate_c5(h_perturbed, dx)
        if new_score < best_score:
            best_score = new_score
            best_h = h_perturbed.copy()
            
        if best_score <= 0.3807:
            break
    
    # Final validation of best result
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_score = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_score, n_points)
