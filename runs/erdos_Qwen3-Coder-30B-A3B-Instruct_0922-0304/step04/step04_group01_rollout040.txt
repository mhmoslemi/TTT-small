import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available

        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Start with triangular window
        h0 = np.clip(np.linspace(0, 1, n_points//2), 0, 1)
        h0 = np.concatenate([h0, h0[::-1]])
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize particles around h0
    for i in range(n_particles):
        perturbation = np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(h0 + perturbation, 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    w = 0.9
    phi_p = 2.0
    phi_g = 2.0
    c1 = 0.5
    c2 = 0.5
    
    # Track personal bests
    personal_best = particles.copy()
    personal_best_c5 = np.full(n_particles, float('inf'))
    
    # Track global best
    global_best_idx = 0
    global_best_c5 = float('inf')
    
    # Evaluate initial population
    for i in range(n_particles):
        c5 = evaluate_c5(particles[i], dx)
        personal_best_c5[i] = c5
        if c5 < global_best_c5:
            global_best_c5 = c5
            global_best_idx = i
    
    # Optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_idx = i
                
        # Occasionally reinitialize some particles to avoid local optima
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                perturbation = np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(h0 + perturbation, 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
        
        # Early termination if target achieved
        if global_best_c5 <= 0.3807:
            break
    
    return (personal_best[global_best_idx].tolist(), global_best_c5, n_points)
