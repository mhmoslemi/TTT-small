import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using cvxpy"""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [
        A @ delta == b,
        delta >= delta_low,
        delta <= delta_high
    ]
    
    # Solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 500
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        # C_k = dx * sum h[i] * (1 - h[i+k])
        # Gradient: dC/dh = dx * (1 - 2*h - shift_average)
        # Hessian: d²C/dh² = dx * (-2*I + shift_matrix)
        
        # Precompute shift averages for efficiency
        shift_avg = np.zeros(n_points)
        for k in range(n_points):
            shift_avg[k] = np.sum(h[:-k] if k > 0 else h) / (n_points - k)
            
        # Gradient
        grad = dx * (1.0 - 2.0 * h - shift_avg)
        
        # Hessian approximation (simplified)
        # Add a small regularization to ensure positive definiteness
        reg = 1e-5
        H = np.zeros((n_points, n_points))
        for i in range(n_points):
            # Diagonal elements
            H[i, i] = -2 * dx
            # Off-diagonal elements (shifted sums)
            for j in range(n_points):
                if i-j >= 0 and i-j < n_points:
                    H[i, j] += dx
        
        # Add regularization
        H += reg * np.eye(n_points)
            
        # Constraint matrix for sum = n_points/2
        A_eq = np.ones(n_points).reshape(1, -1)
        b_eq = np.array([0.0])  # We will adjust later: h + delta must sum to n_points/2
        
        # Ensure constraints
        delta_lb = -h  # h + delta >= 0 means delta >= -h
        delta_ub = 1 - h  # h + delta <= 1 means delta <= 1-h
        
        # Solve QP
        delta = solve_qp(H, grad, A_eq, b_eq, delta_lb, delta_ub)
        
        if delta is None:
            # Fallback to simple gradient descent if QP fails
            delta = -0.01 * grad
            delta = np.clip(delta, delta_lb, delta_ub)
            
        # Line search
        step_size = 1.0
        new_h = h + step_size * delta
        new_h = np.clip(new_h, 0, 1)
        # Adjust to maintain constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = evaluate_c5(new_h, dx)
        
        # Backtrack if needed
        max_backtrack = 10
        for _ in range(max_backtrack):
            if new_c5 <= best_c5:
                # Accept step
                h = new_h
                best_c5 = new_c5
                if best_c5 <= 0.3807:
                    break
                break
            else:
                step_size *= 0.5
                new_h = h + step_size * delta
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_c5(new_h, dx)
        else:
            # Accept anyway if no improvement found but set blocking
            h = new_h
            best_c5 = new_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
