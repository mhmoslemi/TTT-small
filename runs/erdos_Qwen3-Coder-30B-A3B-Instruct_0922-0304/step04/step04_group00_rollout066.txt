import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting plane method
    max_iter = 10000
    epsilon = 0.001
    constraint_list = []
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find maximal overlap shift and add constraint
        overlap = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(overlap)
        max_overlap = overlap[max_shift] * dx
        
        # Add constraint to the model
        constraint_list.append(max_overlap <= best_c5 - epsilon)
        
        # Solve using CVXPY with ECOS
        try:
            # Create optimization variables
            h_var = cp.Variable(n_points)
            
            # Objective
            obj = cp.Minimize(cp.sum(h_var))
            
            # Constraints
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Add our constraints from cutting plane
            for c in constraint_list:
                constraints.append(c)
                
            # Solve problem
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h = np.array(h_var.value).flatten()
                # normalize to sum = n_points/2
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
            else:
                # If solver fails, perturb randomly
                h += np.random.normal(0, 0.005, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # If solver fails, perturb randomly
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
