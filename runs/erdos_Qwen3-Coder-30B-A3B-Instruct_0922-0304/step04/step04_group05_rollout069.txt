import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: clamped [0,1] and scaled to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
    else:
        # Triangular initial function
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
    
    population.append(h0.copy())
    
    # Fill remaining population with diverse variations
    for i in range(1, pop_size):
        h = h0.copy()
        # Add random perturbations
        noise = np.random.normal(0, mutation_scale * 0.5, n_points)
        h += noise
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Evolutionary Strategy
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = max(1, int(pop_size * elite_ratio))
        elite_indices = sorted_indices[:elite_size]
        
        # Update global best
        current_best_idx = sorted_indices[0]
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create next generation
        new_population = []
        
        # Elites
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent_idx = elite_indices[np.random.randint(elite_size)]
            parent = population[parent_idx]
            
            # Mutation
            mutation = np.random.normal(0, mutation_scale, n_points)
            offspring = parent + mutation
            offspring = project_feasible(offspring, n_points)
            new_population.append(offspring)
            
        population = new_population
        
        # Adapt mutation scale based on success
        if iteration % 50 == 0 and iteration > 0:
            recent_fitness = [evaluate_c5(h, dx) for h in population[:elite_size]]
            avg_fitness = np.mean(recent_fitness)
            if avg_fitness < 0.4:  # If progress is good
                mutation_scale = max(0.001, mutation_scale * 1.1)
            else:  # If stagnation
                mutation_scale = max(0.001, mutation_scale * 0.9)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
