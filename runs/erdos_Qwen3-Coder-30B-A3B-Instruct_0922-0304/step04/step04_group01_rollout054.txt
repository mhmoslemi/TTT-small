import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 75
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian optimization
    lambda_mult = 0.0
    mu = 1.0
    nu = 1.0
    step_size = 0.002
    
    max_iter = 3000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and gradient
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
            
        # Update h using augmented Lagrangian
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        
        # Constraint handling
        h_sum = np.sum(h)
        constraint_error = h_sum - n_points / 2
        lambda_mult += mu * constraint_error
        
        # Apply constraints more directly
        h = h + lambda_mult / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step sizing
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.95
            
        # Exploratory perturbations
        if iteration % 50 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
