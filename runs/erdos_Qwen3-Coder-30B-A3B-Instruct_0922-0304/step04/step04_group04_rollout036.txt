import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize using the parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Try a different approach: gradient-based optimization
    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap

    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use projected gradient descent with adaptive step size
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Construct basis for improving search
    # Sine and cosine basis for smooth functions
    def fourier_basis(x_vals, n_basis=30):
        basis = []
        for i in range(1, n_basis+1):
            if i % 2 == 1:
                basis.append(np.sin(i * np.pi * x_vals))
            else:
                basis.append(np.cos((i//2) * np.pi * x_vals))
        return np.array(basis).T
    
    x_vals = np.linspace(0, 2, n_points)
    basis = fourier_basis(x_vals, 20)
    basis /= np.linalg.norm(basis, axis=0)  # Normalize
    
    # Newton-like updates (projected gradient descent with line search)
    max_iter = 5000
    lr = 0.01
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute gradient approximation (numerical)
        eps = 1e-4
        grad = np.zeros(n_points)
        current_val = evaluate_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            val_plus = evaluate_c5(h_plus)
            grad[i] = (val_plus - current_val) / eps

        # Update with gradient descent and projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        if evaluate_c5(h_new) < best_c5:
            best_c5 = evaluate_c5(h_new)
            best_h = h_new.copy()
            lr = max(lr * 1.05, 0.001)  # Increase learning rate for progress
        else:
            lr *= 0.9  # Decrease learning rate
            
        if iter_count % 100 == 0 and lr < 1e-6:
            lr = 0.01  # Reset learning rate if it gets too small
            
        h = h_new.copy()
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
