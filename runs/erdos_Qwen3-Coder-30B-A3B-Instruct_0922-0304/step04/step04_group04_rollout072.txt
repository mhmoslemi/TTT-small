import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: 3-step piecewise constant
        h = np.zeros(n_points)
        h[:n_points//3] = 0.5
        h[n_points//3:2*n_points//3] = 1.0
        h[2*n_points//3:] = 0.2
        h = h * (n_points / 2) / np.sum(h)
    
    # Store the best solution
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    # Gradient-based local search with momentum
    velocity = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.05
    max_velocity = 0.1
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_overlap = correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(current_overlap)
        
        # Compute gradient approximation (finite differences)
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            # Perturb h[j] to estimate gradient
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[j] = (np.max(overlap_plus) - current_c5) / eps
        
        # Update velocity and position with momentum
        velocity = momentum * velocity - learning_rate * grad
        velocity = np.clip(velocity, -max_velocity, max_velocity)
        h = h + velocity
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best solution
        new_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
