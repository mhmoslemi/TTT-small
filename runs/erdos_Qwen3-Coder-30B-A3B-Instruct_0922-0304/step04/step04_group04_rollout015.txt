```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a starting function - simple piecewise constant
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Wavelet-based optimization using convex programming
    # We'll simplify and use a banded optimization approach
    # Start with the given function and apply a local search
    
    best_h = h_init.copy()
    best_score = evaluate_c5(best_h, dx)
    
    # Use a simple gradient-free method: Nelder-Mead with restarts
    # First try a few restarts with random perturbations
    
    # Try multiple initialization strategies
    strategies = [
        lambda: h_init.copy(),  # original
        lambda: np.random.rand(n_points),  # random
        lambda: np.ones(n_points),  # constant 1
        lambda: np.zeros(n_points),  # constant 0
        lambda: np.ones(n_points) * 0.5,  # constant 0.5
    ]
    
    # Optimization iterations
    max_iter = 1000
    for r in range(20):  # restarts
        if time.time() - start_time > budget_s - 1:
            break
            
        # Choose a strategy
        strat = strategies[r % len(strategies)]
        h = strat()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        score = evaluate_c5(h, dx)
        
        # Local search: small perturbations
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Sample a small perturbation
            pert = np.random.normal(0, 0.005, n_points)
            h_new = h + pert
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_score = evaluate_c5(h_new, dx)
            
            if new_score < score:
                h = h_new
                score = new_score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
                    
            # Early stopping
            if score <= 0.3807:
                break
        
        if best_score <= 0.3807:
            break
    
    return (best_h.tolist(), best_score, n_points)
```
