import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is clipped to [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_population(population, dx):
    """Evaluate all individuals in population"""
    return [compute_c5(h, dx) for h in population]

def tournament_selection(population, fitness, k=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population) // 2):
        tournament_indices = random.sample(range(len(population)), k)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Single point crossover"""
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.01):
    """Mutate h with Gaussian noise"""
    if random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(h))
        h = h + noise
    return h

def local_search(h, target_sum, dx, max_iter=50):
    """Apply gradient-based local search to h"""
    def objective(h_flat):
        h_resized = h_flat.reshape(h.shape)
        return compute_c5(h_resized, dx)
    
    def constraint_sum(h_flat):
        h_resized = h_flat.reshape(h.shape)
        return np.sum(h_resized) - target_sum

    bounds = [(0, 1) for _ in range(len(h))]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective, 
        h.flatten(), 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    
    if result.success:
        h = result.x.reshape(h.shape)
    
    # Ensure feasibility after optimization
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # This ensures sum(h) * dx = 1
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values) * (target_sum / np.sum(initial_h_values))
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = normalize_h(base_h, target_sum)
    
    # Create initial population by perturbing the base
    for _ in range(population_size):
        noise = np.random.normal(0, 0.02, n_points)
        h = base_h + noise
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Best solution tracking
    best_c5 = float('inf')
    best_h = None
    
    # GA iterations
    max_generations = 1000
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = evaluate_population(population, dx)
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        
        # Early exit if target is met
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                p1, p2 = selected[i], selected[i+1]
                c1, c2 = crossover(p1, p2)
            else:
                c1, c2 = selected[i].copy(), selected[i].copy()
            
            c1 = mutate(c1)
            c2 = mutate(c2)
            
            new_population.append(normalize_h(c1, target_sum))
            new_population.append(normalize_h(c2, target_sum))
        
        # Local search on top solutions
        sorted_indices = np.argsort(fitness)
        for i in range(min(10, len(sorted_indices))):
            idx = sorted_indices[i]
            if random.random() < 0.8:  # 80% chance to do local search
                population[idx] = local_search(population[idx], target_sum, dx)
        
        # Replace worst individuals
        population = new_population
        
        generation += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx) if best_h is not None else float('inf')
    
    return (best_h.tolist(), final_c5, n_points)
