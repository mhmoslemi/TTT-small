import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    w = 0.5  # width of each interval
    a = 0.0  # starting position
    d_candidates = np.arange(0, 1.01, 0.01)
    
    best_c5 = float('inf')
    best_h = None
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        h = np.zeros(n_points)
        
        # First interval [a, a+w]
        i0 = int(np.floor(a / dx))
        i1 = int(np.floor((a + w) / dx))
        if i1 >= n_points:
            i1 = n_points - 1
        h[i0:i1+1] = 1.0
        
        # Second interval [a+w+d, a+w+d+w] 
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        if i2 < n_points and i3 >= 0:
            if i3 >= n_points:
                i3 = n_points - 1
            h[i2:i3+1] = 1.0
        
        # Normalize so that sum(h)*dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
    
    # Refine around best d if time permits
    if best_h is not None and time.time() - start_time < budget_s - 5:
