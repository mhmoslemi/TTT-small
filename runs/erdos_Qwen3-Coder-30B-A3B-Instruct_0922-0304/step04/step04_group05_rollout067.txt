import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 30
    generations = 2000
    elite_ratio = 0.25
    mutation_scale_init = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, target_sum)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_feasible(h0, target_sum)
        population.append(h0.copy())
    
    # Fill rest of population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h = project_feasible(h, target_sum)
        population.append(h.copy())
    
    # Tracks best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # ES parameters
    mutation_scale = mutation_scale_init
    success_count = 0
    max_success = 5
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = np.array([evaluate_c5(ind, dx) for ind in population])
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = max(1, int(elite_ratio * population_size))
        elite_indices = sorted_indices[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Generate new population
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            # Select parent from elite
            parent = elite[np.random.randint(0, len(elite))]
            
            # Mutate
            noise = np.random.normal(0, mutation_scale, n_points)
            offspring = parent + noise
            
            # Project to feasible region
            offspring = project_feasible(offspring, target_sum)
            
            new_population.append(offspring)
            
        population = new_population[:population_size]
        
        # Adapt mutation scale
        success_count += 1
        if success_count >= max_success:
            mutation_scale *= 1.1
            success_count = 0
        elif len(elite) > 1:
            # Check if improvement was made
            old_best_fitness = fitness[sorted_indices[0]]
            new_best_fitness = evaluate_c5(population[0], dx)
            if new_best_fitness < old_best_fitness * 0.99:
                # Improvement, reduce mutation
                mutation_scale *= 0.99
                success_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
