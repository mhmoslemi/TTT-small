import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = None
    
    # Try different grid spacings for d
    d_candidates = np.linspace(0, 1, 100)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create step function with two intervals
        h = np.zeros(n_points)
        
        # First interval: [0, 0.5]
        i0 = 0
        i1 = int(0.5 / dx) - 1
        
        # Second interval: [0.5 + d, 1 + d]
        i2 = int((0.5 + d) / dx)
        i3 = int((1 + d) / dx) - 1
        
        # Make sure intervals don't overlap
        if i2 > i1:
            # Fill the intervals with 1s
            h[i0:i1+1] = 1
            h[i2:i3+1] = 1
            
            # Normalize to integral = 1
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate
            c5 = evaluate_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # If we didn't find a better solution, use the initial construction
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = np.clip(best_h, 0, 1)
        else:
            best_h = np.zeros(n_points)
            for i in range(n_points):
                best_h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
            best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, dx)
            
    return (best_h.tolist(), best_c5, n_points)
