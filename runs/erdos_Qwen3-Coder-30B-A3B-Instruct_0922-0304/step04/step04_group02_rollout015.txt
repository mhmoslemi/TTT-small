import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start approach with gradient descent
    n_starts = 50
    max_iter = 200
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Function to enforce constraints after each update
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Multi-start loop
    for start in range(n_starts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random initialization
        h_start = np.random.random(n_points) * 0.5
        h_start = project(h_start)
        
        # SLSQP optimization
        try:
            result = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9})
            if result.success:
                optimized_h = result.x
                optimized_h = project(optimized_h)
                optimized_c5 = evaluate_c5(optimized_h, dx)
                if optimized_c5 < best_c5:
                    best_c5 = optimized_c5
                    best_h = optimized_h.copy()
                    
                if best_c5 <= 0.3807:
                    break
        except:
            pass
    
    # Final refinement with SLSQP on the best found solution
    if time.time() - start_time < budget_s * 0.95:
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': 100, 'ftol': 1e-10})
            if result.success:
                refined_h = result.x
                refined_h = project(refined_h)
                refined_c5 = evaluate_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h
        except:
            pass
    
    # Final validation and normalization
    best_h = project(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
