import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, dx, target_sum):
    """Project h to satisfy the sum constraint"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with existing construction or create a new one
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5  # uniform initial guess
    
    # Normalize to satisfy constraint
    h = project_h(h, dx, target_sum)
    
    # Gradient descent parameters
    max_iter = 10000
    learning_rate = 0.01
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all overlaps and find the worst case
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)  # convert to shift index
        
        # Compute gradient for the worst overlap
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient of O_k with respect to h[i]
            idx_plus = (i + max_k) % n_points
            idx_minus = (i - max_k) % n_points
            grad[i] = (1 - h[idx_plus]) - h[idx_minus]
        
        # Update with gradient descent
        h_new = h - learning_rate * grad
        
        # Ensure bounds and sum constraint
        h_new = project_h(h_new, dx, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Convergence check
        if abs(new_c5 - best_c5) < tolerance:
            break
            
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
