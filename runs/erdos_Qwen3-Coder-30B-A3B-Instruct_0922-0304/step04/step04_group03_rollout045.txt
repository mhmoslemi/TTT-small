import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Try piecewise constant construction based on symmetry analysis
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Grid search on interval spacing
    d_candidates = np.arange(0.0, 1.01, 0.01)
    min_c5 = best_c5
    best_d = 0.0
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create piecewise constant function with two intervals
        w = 0.5  # width of each interval
        i0 = 0
        i1 = int(np.floor(w / dx))
        i2 = int(np.ceil((w + d) / dx))
        i3 = int(np.floor((2*w + d) / dx))
        
        # Check if indices are valid
        if i2 <= i1 or i3 < i2:
            continue
            
        h_candidate = np.zeros(n_points)
        h_candidate[i0:i1+1] = 1.0
        h_candidate[i2:i3+1] = 1.0
        
        # Normalize to have integral = 1
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute C5
        c5 = evaluate_c5(h_candidate, dx)
        
        if c5 < min_c5:
            min_c5 = c5
            best_h = h_candidate.copy()
            best_d = d
            
    # Local optimization around best d
    if min_c5 < best_c5:
        d_start = max(0.0, best_d - 0.02)
        d_end = min(1.0, best_d + 0.02)
        d_candidates_local = np.arange(d_start, d_end, 0.001)
        
        for d in d_candidates_local:
            if time.time() - start_time > budget_s - 1:
                break
                
            w = 0.5
            i0 = 0
            i1 = int(np.floor(w / dx))
            i2 = int(np.ceil((w + d) / dx))
            i3 = int(np.floor((2*w + d) / dx))
            
            if i2 <= i1 or i3 < i2:
                continue
                
            h_candidate = np.zeros(n_points)
            h_candidate[i0:i1+1] = 1.0
            h_candidate[i2:i3+1] = 1.0
            
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5 = evaluate_c5(h_candidate, dx)
            
            if c5 < min_c5:
                min_c5 = c5
                best_h = h_candidate.copy()
    
    return (best_h.tolist(), min_c5, n_points)
