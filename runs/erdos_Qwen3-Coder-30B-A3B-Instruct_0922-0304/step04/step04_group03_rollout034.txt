import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple triangular shape to start
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of the best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.00005
    max_iter = 30000
    step_size = 0.0001
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate current overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = np.max(overlaps)
        
        # Compute gradient for the maximum overlap case
        # O_k = sum(h[i] * (1 - h[i+k]))
        # dO_k/dh_i = (1 - h[i+k]) - h[i-k] (considering circular indexing)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            k = max_overlap_idx - n_points + 1  # This is the shift of maximum overlap
            # Account for circular nature
            k_pos = (i + k) % n_points
            k_neg = (i - k) % n_points
            # This calculation gives gradient of O_k w.r.t. h_i
            gradient[i] = (1 - h[k_pos]) - h[k_neg]
        
        # Update h in direction of negative gradient
        h_new = h - learning_rate * gradient
        
        # Project to [0,1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
