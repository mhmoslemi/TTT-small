import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to match target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Frank-Wolfe style update with gradient descent
    gamma = 0.1
    beta = 0.5
    stagnation_count = 0
    stagnation_threshold = 10
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (approximated)
        # Gradient of overlap term is approximately: -2*h + 1 - average of h shifted
        overlap = correlate(h, 1 - h, mode="full")
        # Extract the relevant indices for the overlap we want to minimize
        # The maximum overlap occurs over all possible shifts
        max_idx = np.argmax(overlap)
        # Use the derivative of the overlap with respect to h at this shift
        # This is a simplified approach - a full gradient would require
        # computing the gradient for each shift, weighting by the maximum
        
        # Gradient approximation using a finite difference approach
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            # Compute finite difference gradient approximation
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Frank-Wolfe update direction
        # This is actually projected gradient descent with feasible projection
        # Move in the negative gradient direction
        delta = -gamma * grad
        # Apply the update
        h_new = h + delta
        # Project to feasible set (0 <= h <= 1 and sum = target_sum)
        h_new = project_to_feasible(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update current solution
        h = h_new
        
        # Adjust step size
        if stagnation_count > stagnation_threshold:
            gamma *= beta
            stagnation_count = 0
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
