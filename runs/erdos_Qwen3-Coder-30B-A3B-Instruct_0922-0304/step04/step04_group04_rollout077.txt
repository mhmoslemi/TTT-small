import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    max_iter = 2000
    tolerance = 1e-4
    alpha = 0.01
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a simple initial guess
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create Fourier basis
    x = np.linspace(0, 2, n_points, endpoint=False)
    fourier_basis = []
    for k in range(1, n_basis+1):
        if k % 2 == 1:
            fourier_basis.append(np.sin(np.pi * k * x))
        else:
            fourier_basis.append(np.cos(np.pi * k * x))
    
    # Precompute inner products for basis functions
    A = np.zeros((n_basis, n_basis))
    for i in range(n_basis):
        for j in range(n_basis):
            A[i,j] = np.dot(fourier_basis[i], fourier_basis[j]) * dx
    
    # Compute integral of basis functions
    basis_integrals = np.array([np.trapz(basis, x) for basis in fourier_basis])

    # Optimization loop with convex relaxation
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Try optimization
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Project initial guess to Fourier coefficients
        try:
            # Solve the convex relaxation
            c = Variable(n_basis)
            
            # Method to approximate h given c
            def h_from_c(c_vals):
                return np.array([sum(c_vals[i] * basis[i] for i in range(n_basis)) 
                               for basis in fourier_basis])
            
            # McCormick envelopes for products (simplified version)
            # Minimize max_k of integral terms - need full convolution handling

            # Estimate overlaps using the best approximation
            h_guess = h_from_c(c.value if c.value is not None else np.zeros(n_basis))
            h_guess = np.clip(h_guess, 0, 1)
            h_guess = h_guess * (n_points / 2) / np.sum(h_guess)
            
            current_c5 = evaluate_c5(h_guess, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_guess.copy()

        except Exception as e:
            pass
            
        # Occasionally resample to explore other regions
        if iter_count % 50 == 0:
            h_samp = np.random.rand(n_points)
            h_samp = np.clip(h_samp, 0, 1)
            h_samp = h_samp * (n_points / 2) / np.sum(h_samp)
            c5_samp = evaluate_c5(h_samp, dx)
            if c5_samp < best_c5:
                best_c5 = c5_samp
                best_h = h_samp.copy()
        
        # Early exit if target reached
        if best_c5 < 0.3807:
            break

    # Final verification
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
