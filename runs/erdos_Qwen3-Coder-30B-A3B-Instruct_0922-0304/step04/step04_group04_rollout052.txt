import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 2000
    tolerance = 1e-4
    
    # Define basis functions (Fourier)
    n_basis = 20
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Basis functions: sin and cos
    basis_functions = []
    for i in range(n_basis):
        if i % 2 == 0:
            basis_functions.append(np.cos(np.pi * i * x / 2))
        else:
            basis_functions.append(np.sin(np.pi * i * x / 2))
    
    # Precompute inner products
    inner_products = np.zeros((n_basis, n_basis))
    for i in range(n_basis):
        for j in range(n_basis):
            inner_products[i, j] = np.dot(basis_functions[i], basis_functions[j]) * dx
    
    # Integral constraint coefficients
    integral_coeffs = np.array([np.dot(basis_functions[i], np.ones(n_points)) * dx 
                              for i in range(n_basis)])
    
    # Try to use initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Project onto basis
        coeffs = np.linalg.lstsq(np.array(basis_functions).T, h0, rcond=None)[0]
        coeffs = np.clip(coeffs, 0, 1)
    else:
        # Generate default initialization
        coeffs = np.random.rand(n_basis) * 0.5
        coeffs[0] = 0.5  # Set DC component
    
    # Try convex relaxation approach
    best_coeffs = coeffs.copy()
    best_c5 = evaluate_c5(np.dot(np.array(basis_functions).T, coeffs), dx)
    
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use CVXPY to optimize
        try:
            # Define optimization variables
            c_vars = cp.Variable(n_basis)
            d_vars = cp.Variable(n_basis * n_basis)
            
            # McCormick relaxation for c[i]*c[j] terms
            # Linear constraints
            constraints = []
            
            # Integral constraint
            constraints.append(cp.sum(cp.multiply(c_vars, integral_coeffs)) == 1)
            
            # Bounds on coefficients
            constraints.extend([c_vars[i] >= 0 for i in range(n_basis)])
            constraints.extend([c_vars[i] <= 1 for i in range(n_basis)])
            
            # Additional constraints to support McCormick envelopes
            # We'll compute using a simpler method
            
            # Relax objective
            # That is a complex nonconvex term involving overlaps over k
            
            # Simplify approach using tight projections and gradient descent
            
            # For now, attempt projected gradient descent using basis representation
            
            # Compute current h
            current_h = np.dot(np.array(basis_functions).T, coeffs)
            
            # Flow in function space
            h_proj = np.clip(current_h, 0, 1)
            h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
            
            # Re-project to basis
            new_coeffs = np.linalg.lstsq(np.array(basis_functions).T, h_proj, rcond=None)[0]
            
            # Gradient estimation for update using finite differences
            g = np.zeros_like(new_coeffs)
            eps = 1e-6
            
            # Compute g = ∇f
            for i in range(n_basis):
                coeffs_plus = new_coeffs.copy()
                coeffs_plus[i] += eps
                
                h_plus = np.dot(np.array(basis_functions).T, coeffs_plus)
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c_plus = evaluate_c5(h_plus, dx)
                
                coeffs_minus = new_coeffs.copy()
                coeffs_minus[i] -= eps
                
                h_minus = np.dot(np.array(basis_functions).T, coeffs_minus)
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                c_minus = evaluate_c5(h_minus, dx)
                
                g[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update with gradient descent
            step_size = 0.01
            updated_coeffs = new_coeffs - step_size * g
            
            # Clamp coefficients
            updated_coeffs = np.clip(updated_coeffs, 0, 1)
            
            # Renormalize
            h_proposed = np.dot(np.array(basis_functions).T, updated_coeffs)
            h_proposed = np.clip(h_proposed, 0, 1)
            h_proposed = h_proposed * (n_points / 2) / np.sum(h_proposed)
            
            # Check if improvement
            proposed_c5 = evaluate_c5(h_proposed, dx)
            if proposed_c5 < best_c5:
                best_c5 = proposed_c5
                best_coeffs = updated_coeffs.copy()

            if best_c5 <= 0.3807:

        except Exception:

        iter_count += 1
        
        if iter_count % 50 == 0:
            # Check if we've improved

            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            current_c5 = evaluate_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_coeffs = coeffs.copy()
    
    # Final solution
    best_h = np.dot(np.array(basis_functions).T, best_coeffs)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
