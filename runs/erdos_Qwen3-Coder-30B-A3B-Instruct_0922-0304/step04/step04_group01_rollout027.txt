import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (75 / 2) / np.sum(h)
    else:
        # Fallback: triangular window
        n_points = 75
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (75 / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian optimization with adaptive step sizes
    max_iter = 10000
    lambda_mult = 0.0
    penalty_coeff = 1.0
    step_size = 0.002
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and gradient with finite differences
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Gradient estimation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Augmented Lagrangian step with penalty
        h_new = h - step_size * (grad + lambda_mult * (h - 0.5 * np.ones_like(h)) + 
                                penalty_coeff * np.clip(h, 0, 1) * (h < 0) + 
                                penalty_coeff * np.clip(h, 0, 1) * (h > 1))
        
        # Project back to feasible region and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update Lagrange multipliers based on constraint violation
        constraint_violation = np.sum(h_new) - n_points / 2
        lambda_mult += penalty_coeff * constraint_violation
        
        # Adaptive step size adjustment
        if iteration % 20 == 0:
            prev_c5 = evaluate_c5(h, dx)
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < prev_c5:
                step_size *= 1.1
            else:
                step_size *= 0.9
            
        h = h_new
        
        # Periodic exploration
        if iteration % 50 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
