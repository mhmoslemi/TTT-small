import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create an initial guess that's likely to work
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h_best = h_init.copy()
    c5_best = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    # Try different optimization strategies
    strategies = [
        ("gradient_descent", 100),
        ("coordinate_descent", 200),
        ("local_search", 300),
    ]
    
    for strategy, steps in strategies:
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = h_init.copy()
        c5_current = c5_best
        
        if strategy == "gradient_descent":
            # Simple gradient descent approach
            for step in range(steps):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute overlap derivatives numerically
                eps = 1e-4
                grad = np.zeros(n_points)
                base_overlap = np.correlate(h_current, 1 - h_current, mode="full")
                max_base = np.max(base_overlap) * dx
                
                for i in range(n_points):
                    h_plus = h_current.copy()
                    h_plus[i] += eps 
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    overlap_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
                    max_plus = np.max(overlap_plus) * dx
                    
                    grad[i] = (max_plus - max_base) / eps
                
                # Update step
                lr = 0.01
                h_new = h_current - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate new solution
                overlap_new = np.correlate(h_new, 1 - h_new, mode="full")
                max_new = np.max(overlap_new) * dx
                
                if max_new < c5_current:
                    h_current = h_new
                    c5_current = max_new
                    
        elif strategy == "coordinate_descent":
            # Coordinate descent approach
            for step in range(steps):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                for i in range(n_points):
                    if time.time() - start_time > budget_s - 1:
                        break
                        
                    # Try small changes to each coordinate
                    original = h_current[i]
                    for delta in [-0.05, -0.02, 0.02, 0.05]:
                        h_new = h_current.copy()
                        h_new[i] += delta
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                        
                        overlap_new = np.correlate(h_new, 1 - h_new, mode="full")
                        max_new = np.max(overlap_new) * dx
                        
                        if max_new < c5_current:
                            h_current = h_new
                            c5_current = max_new
                            break
                    # Reset if no improvement
                    if h_current[i] != original and h_current[i] < 0:
                        h_current[i] = 0
                
        elif strategy == "local_search":
            # Simpler local search with random perturbations
            for step in range(steps):
                if time.time() - start_time > budget_s - 1:
                    break
                
                # Create a new candidate by slight perturbations
                h_new = h_current.copy()
                indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                
                for i in indices:
                    h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.03), 0, 1)
                
                # Normalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate
                overlap_new = np.correlate(h_new, 1 - h_new, mode="full")
                max_new = np.max(overlap_new) * dx
                
                if max_new < c5_current:
                    h_current = h_new
                    c5_current = max_new
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
