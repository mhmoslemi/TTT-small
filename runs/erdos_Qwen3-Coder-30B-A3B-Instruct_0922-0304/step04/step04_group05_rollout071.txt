import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local search: modify h at random positions
        h_new = h.copy()
        num_changes = np.random.randint(1, 10)
        for _ in range(num_changes):
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
