import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Initialize ADMM variables
    x = h.copy()
    z = np.zeros_like(x)
    u = np.zeros_like(x)
    
    # For constraint enforcement: ensure sum(h) = n_points / 2
    def project_sum(h_vec):
        return h_vec * (n_points / 2) / np.sum(h_vec)
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update x: minimize augmented Lagrangian with respect to x
        # This is a proximal gradient step for the smooth part
        # We want to minimize: overlap term + rho/2 * ||x - z + u||^2
        # but we don't have a closed form for this. Instead, we use 
        # gradient descent with line search
        
        # Compute gradient of overlap term (approximate)
        # This is not exact but provides a gradient direction
        tmp = np.correlate(x, 1 - x, mode="full") * dx
        grad = np.zeros_like(x)
        # Simple finite difference estimate for gradient (convoluted via correlation)
        for i in range(len(x)):
            h_plus = x.copy()
            h_plus[i] += 1e-6
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_sum(h_plus)
            tmp_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (tmp_plus - tmp) / 1e-6
            
        # Update x using gradient descent
        x_new = x - 0.01 * grad
        x_new = np.clip(x_new, 0, 1)
        x_new = project_sum(x_new)
        x = x_new
        
        # Update z using soft thresholding (simple approach here: just enforce constraints)
        z_old = z.copy()
        z = x + u
        z = np.clip(z, 0, 1)
        z = project_sum(z)
        
        # Update dual variable u
        u += x - z
        
        # Update best solution
        current_c5 = evaluate_c5(x, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds,

        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
