import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using CVXPY"""
    try:
        n = len(c)
        delta = cp.Variable(n)
        
        # Objective: minimize c^T * delta + (1/2) * delta^T * H * delta
        objective = cp.Minimize(cp.matmul(c.T, delta) + 0.5 * cp.matmul(cp.matmul(delta.T, H), delta))
        
        # Constraints
        constraints = [
            A @ delta == b,
            delta >= delta_low,
            delta <= delta_high
        ]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-8, reltol=1e-8)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def generate_initial_h(n_points):
    """Generate initial h using the provided initial_h_values or a triangular function"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    else:
        # Generation of a triangular function for initialization
        h = np.zeros(n_points)
        center = n_points // 2
        for i in range(n_points):
            h[i] = 1 - abs(i - center) / (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Generate initial step function
    h = generate_initial_h(n_points)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Define constraint matrix for sum(h) = n_points/2
    A_eq = np.ones(n_points)
    b_eq = n_points / 2
    
    # Precompute shift averages for efficiency
    shift_averages = np.zeros(n_points)
    for i in range(n_points):
        shift_avg = 0
        count = 0
        for j in range(n_points):
            if 0 <= i + j < n_points:
                shift_avg += h[j]
                count += 1
        if count > 0:
            shift_averages[i] = shift_avg / count
        else:
            shift_averages[i] = 0
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradients and Hessian of overlap values
        c = np.zeros(n_points)
        H = np.zeros((n_points, n_points))
        
        for i in range(n_points):
            # Gradient: c[i] = dx * (1 - 2*h[i] - shift_averages[i])
            c[i] = dx * (1 - 2 * h[i] - shift_averages[i])
            
            # Hessian diagonal elements
            H[i, i] = dx * (-2)
            
            # Off-diagonal elements from shift effects
            for j in range(n_points):
                if 0 <= i + j < n_points:
                    H[i, i + j] += dx * -1
                    H[i + j, i] += dx * -1
        
        # Regularization to ensure positive definiteness
        H = H + 1e-5 * np.eye(n_points)
        
        # Solve QP subproblem
        A = A_eq.reshape(1, -1)
        b = b_eq
        
        delta_low = -h  # minimum delta is -h to maintain h >= 0
        delta_high = 1 - h  # maximum delta is 1 - h to maintain h <= 1
        
        delta = _solve_qp(H, c, A, b, delta_low, delta_high)
        
        # Line search with backtracking
        if delta is not None:
            alpha = 1.0
            step_size = 0.5
            best_new_h = h + alpha * delta
            
            # Keep trying smaller alpha till feasible
            while np.any(best_new_h < 0) or np.any(best_new_h > 1):
                alpha *= step_size
                if alpha < 1e-8:
                    break
                best_new_h = h + alpha * delta
                
            if np.all((best_new_h >= 0) & (best_new_h <= 1)):
                h = best_new_h
                # Ensure constraint again
                h = h * (n_points / 2) / np.sum(h)
                
                # Evaluate new C5
                current_c5 = evaluate_c5(h, dx)
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
                if best_c5 <= 0.3807:
                    break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
