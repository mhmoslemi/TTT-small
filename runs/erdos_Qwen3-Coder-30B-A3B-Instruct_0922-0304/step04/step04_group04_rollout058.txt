import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Basis functions: sine and cosine terms
    def basis_functions(x):
        # phi_0 = 1
        # phi_2k-1 = sin(pi*k*x) for k=1,2,... 
        # phi_2k = cos(pi*k*x) for k=1,2,...
        basis = np.ones((len(x), basis_size + 1))
        basis[:, 0] = 1.0
        for k in range(1, basis_size + 1):
            if k % 2 == 1:  # odd k -> sin
                basis[:, k] = np.sin(np.pi * (k // 2 + 1) * x)
            else:  # even k -> cos
                basis[:, k] = np.cos(np.pi * (k // 2) * x)
        return basis
    
    # Precompute basis functions and integrals over [0,2]
    x_grid = np.linspace(0, 2, n_points)
    basis = basis_functions(x_grid)
    integral_basis = np.array([np.trapz(basis[:, i], x_grid) for i in range(basis_size + 1)])
    
    # Set up optimization variables
    c = cp.Variable(basis_size + 1)  # coefficients for basis functions
    
    # Precompute kernel matrix: K[i,j][lag] = integral of basis[i](x)*basis[j](x+lag)
    K = np.zeros((basis_size + 1, basis_size + 1, 2 * n_points - 1))
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            for lag in range(2 * n_points - 1):
                # This is a discrete approximation of the integral
                # Actual compute for this problem is expensive, so let's precompute a
                # realization that's less computational-intensive.
                # Alternatively, use a faster approximation.
                # For now, keep this simplified to ensure fast completion
                if lag < n_points:
                    indices = np.arange(lag, min(lag + n_points, 2 * n_points))
                    integrand = basis[:, i][0:(len(indices) - lag)] * basis[:, j][lag:(len(indices))]
                    if len(integrand) > 0:
                        K[i, j, lag] = np.trapz(integrand, dx=dx)
                else:
                    # Handle wrap-around
                    lag_offset = lag - (n_points - 1)
                    indices = np.arange(0, lag_offset)
                    integrand = basis[:, i][n_points - lag_offset:] * basis[:, j][0:lag_offset]
                    if len(integrand) > 0:
                        K[i, j, lag] = np.trapz(integrand, dx=dx)

    # Convex relaxation with McCormick envelopes
    z = cp.Variable((basis_size + 1, basis_size + 1), symmetric=True)
    
    # Instantiate a problem for solving
    # Minimize max_lag abs(sum_ij z_ij * K[i,j][lag])
    constraints = []
    
    # Constraints for McCormick relaxation:
    for i in range(basis_size + 1):
        for j in range(i+1, basis_size + 1):  # Only do upper triangle
            # z[i,j] >= c[i] + c[j] - 1
            constraints.append(z[i, j] >= c[i] + c[j] - 1)
            # z[i,j] <= min(c[i], c[j])
            constraints.append(z[i, j] <= c[i])
            constraints.append(z[i, j] <= c[j])
    
    # Additional constraint: integral = 1, with proper scaling
    # ∫ h(x)dx = 1 => c₀*2 + ∫ c_i φ_i(x) dx = 1
    for i in range(basis_size + 1):
        v = c[i] * integral_basis[i]
        if i == 0:
            total_integral = v
        else:
            total_integral += v
    constraints.append(total_integral == 1)
    
    # positiveness (variable support)
    constraints.extend([c[i] >= 0 for i in range(basis_size + 1)])
    
    # approximate constraints: z[i,j] are approximations of c[i]*c[j]
    # now compute the maximum correlation based on the approximated terms
    
    # Do the concave optimization, but eliminate entries that make h not feasible
    # Instead, we'll proceed with gradient descent on a subset of our variables
    # Start from the step function
    
    # Initial solution from the parent
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    # Use coordinate descent
    # Keep h fixed, optimize over c's
    iter = 0
    max_iter = 2000
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Optimization with local search
    # Cycle through a few local refinements as a fallback
    
    while iter < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        iter += 1
        
        # Move from h_index to c coefficients
        c_vals = np.linalg.lstsq(basis.T, h_init, rcond=None)[0]
        
        # Project onto valid bounds
        c_vals = np.clip(c_vals, 0, 1)
        
        # Compute h_back
        h_back = basis @ c_vals
        h_back = np.clip(h_back, 0, 1)
        h_back = h_back * (n_points / 2) / np.sum(h_back)
        
        # Change if it's better
        c5_new = evaluate_c5(h_back, dx)
        if c5_new < best_c5:
            best_h = h_back.copy()
            best_c5 = c5_new
            
        # Try small random perturbation of coefficients as well
        random_pert = np.random.normal(0, 0.001, basis_size + 1)
        c_pert = c_vals + random_pert
        c_pert = np.clip(c_pert, 0, 1)
        
        h_pert = basis @ c_pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        c5_pert = evaluate_c5(h_pert, dx)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
        
    return (best_h.tolist(), best_c5, n_points)
