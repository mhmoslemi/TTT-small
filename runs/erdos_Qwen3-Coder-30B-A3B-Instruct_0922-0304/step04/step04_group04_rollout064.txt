import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        basis = []

                # Sine terms
                basis.append(np.sin(k * np.pi * x))
            else:
                # Cosine terms
                basis.append(np.cos((k // 2) * np.pi * x))
        return np.array(basis).T

    # Project initial solution to Fourier coefficients
    a_init = basis.T @ h_init

        eps = 1e-4
