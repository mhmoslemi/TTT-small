import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    max_iter = 1000
    learning_rate = 0.01
    decay_factor = 0.9
    patience = 50
    patience_counter = 0
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if patience_counter >= patience:
            # Reduce learning rate
            learning_rate *= decay_factor
            patience_counter = 0
            
        # Compute gradient for the worst case overlap
        # Find k that maximizes overlap
        overlap_values = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_values)
        
        # Compute gradient approximating the derivative with respect to h
        gradient = np.zeros(n_points)
        for j in range(n_points):
            # Contribution from h[j] to overlap with shifted h at position (j + k_max) % n_points
            grad_component1 = 1 - h[(j + k_max) % n_points]
            # Contribution from h[(j - k_max) % n_points] to overlap with h[j] 
            grad_component2 = h[(j - k_max) % n_points]
            gradient[j] = grad_component1 - grad_component2
            
        # Update h
        h = h - learning_rate * gradient * dx
        
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize h
        h = h * (n_points / 2) / np.sum(h)
        
    return (best_h.tolist(), best_c5, n_points)
