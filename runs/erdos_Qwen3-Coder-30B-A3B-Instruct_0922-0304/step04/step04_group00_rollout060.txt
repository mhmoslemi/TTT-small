import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Compute correlation
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Temperature schedule
    T0 = 1e-4
    alpha = 0.99
    T = T0
    
    # Perturbation step size
    step_size = 0.05
    
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Perturb a random point
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        
        # Random perturbation
        delta = np.random.uniform(-step_size, step_size)
        h_new[i] += delta
        
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        # Update temperature and step size
        T *= alpha
        step_size *= alpha
        
        iter_count += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
