import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Generate basis functions: sine and cosine terms
    basis_size = 20
    x = np.linspace(0, 2, n_points)
    
    # Create basis matrix
    phi_basis = np.zeros((n_points, basis_size + 1))
    phi_basis[:, 0] = 1.0  # Constant term
    for i in range(1, basis_size + 1):
        if i % 2 == 1:
            # Odd index: sine terms
            phi_basis[:, i] = np.sin(np.pi * (i + 1) / 2 * x)
        else:
            # Even index: cosine terms
            phi_basis[:, i] = np.cos(np.pi * i / 2 * x)
    
    # Compute integral of basis functions
    int_phi = np.zeros(basis_size + 1)
    for i in range(basis_size + 1):
        int_phi[i] = np.trapz(phi_basis[:, i], x)
    
    # Initialize coefficients
    c = Variable(basis_size + 1)
    
    # McCormick relaxation setup for products
    z = Variable((basis_size + 1, basis_size + 1), symmetric=True)
    
    # Build overlap matrix with precomputed integrals
    K = np.zeros((n_points, basis_size + 1, basis_size + 1))
    for lag in range(n_points):
        # Compute kernel for this lag
        lagged_phi = np.zeros((n_points, basis_size + 1))
        for i in range(basis_size + 1):
            if i == 0:
                lagged_phi[:, i] = 1.0
            elif i % 2 == 1:
                lagged_phi[:, i] = np.sin(np.pi * (i + 1) / 2 * (x - lag * dx))
            else:
                lagged_phi[:, i] = np.cos(np.pi * i / 2 * (x - lag * dx))
        K[lag] = phi_basis.T @ lagged_phi
    
    # Setup constraints
    constraints = []
    
    # Integral constraint: c₀ + sum(cᵢ * ∫φᵢ) = 1
    integral_constraint = c[0] + sum(c[i] * int_phi[i] for i in range(1, basis_size + 1)) == 1
    constraints.append(integral_constraint)
    
    # Non-negativity of coefficients
    for i in range(basis_size + 1):
        constraints.append(c[i] >= 0)
    
    # McCormick envelopes: z[i,j] approximates c[i]*c[j]
    for i in range(basis_size + 1):
        for j in range(i, basis_size + 1):
            # z[i,j] >= c[i] + c[j] - 1
            constraints.append(z[i,j] >= c[i] + c[j] - 1)
            # z[i,j] <= c[i]
            constraints.append(z[i,j] <= c[i])
            # z[i,j] <= c[j]
            constraints.append(z[i,j] <= c[j])
            # Ensure symmetry
            if i != j:
                constraints.append(z[j,i] == z[i,j])
    
    # Objective: minimize max_lag |sum_ij z[i,j] * K[lag,i,j]|
    # Approximate by a single maximum term, test several lags
    max_overlap_expr = 0
    for lag in range(0, n_points, 5):  # sample every 5th lag for efficiency
        expr = 0
        for i in range(basis_size + 1):
            for j in range(basis_size + 1):
                expr += z[i,j] * K[lag,i,j]
        max_overlap_expr = max(max_overlap_expr, abs(expr))
    
    # Optimization problem
    objective = Minimize(max_overlap_expr)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, max_iter=10000, verbose=False)
        if prob.status == "optimal":
            c_val = c.value
        else:
            # Fallback to initial solution
            if 'initial_h_values' in globals():
                h_init = np.array(initial_h_values)
                c_val = (phi_basis.T @ h_init) * dx
                c_val[0] = np.clip(c_val[0], 0, 1)
                c_val = np.clip(c_val, 0, 1)
            else:
                # Use default initialization
                c_val = np.ones(basis_size + 1)
                c_val[0] = 0.5
    except:
        # Fallback if optimization fails
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            c_val = (phi_basis.T @ h_init) * dx
        else:
            c_val = np.ones(basis_size + 1)
        
    # Reconstruct h from coefficients
    h_reconstructed = phi_basis @ c_val
    h_reconstructed = np.clip(h_reconstructed, 0, 1)
    h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
    
    # Evaluate final overlap
    overlap = correlate(h_reconstructed, 1 - h_reconstructed, mode="full")
    c5_final = np.max(overlap) * dx
    
    return (h_reconstructed.tolist(), c5_final, n_points)
