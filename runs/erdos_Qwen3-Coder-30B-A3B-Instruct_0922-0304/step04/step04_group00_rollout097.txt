import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, n_points):
    """Normalize h to sum to n_points/2."""
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

def evaluate_population(population, n_points, dx):
    """Evaluate all individuals in population."""
    return [compute_c5(ind, dx) for ind in population]

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select individuals using tournament selection."""
    selected = []
    indices = list(range(len(population)))
    np.random.shuffle(indices)
    for i in range(0, len(indices), tournament_size):
        group = indices[i:i+tournament_size]
        if len(group) == tournament_size:
            best_idx = min(group, key=lambda idx: fitnesses[idx])
            selected.append(population[best_idx].copy())
    return selected

def single_point_crossover(parent1, parent2):
    """Perform single-point crossover."""
    n = len(parent1)
    if n < 2:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.01):
    """Mutate individual with Gaussian noise."""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual = individual + noise
    return individual

    best_c5 = compute_c5(best_h, dx)

        fitnesses = evaluate_population(population, n_points, dx)

        current_best_idx = np.argmin(fitnesses)
