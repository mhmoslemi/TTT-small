import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential Convex Programming
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current gradient and Hessian numerically
        # For simplicity, use a finite difference approximation
        # Compute overlap at current point
        current_overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(current_overlap)
        max_overlap = current_overlap[max_overlap_idx] * dx
        
        # Create feasibility constraints
        # x = h
        # min max_k C_k where C_k = sum(h[i]*(1-h[i+k]))
        # Use a quadratic approximation
        
        # Solve convex subproblem using CVXPY
        # Decision variable
        delta = cp.Variable(n_points)
        
        # Objective: minimize ∇f^T * delta + (1/2)*delta^T * H * delta
        # We're approximating the gradient of the max overlap term
        # For now, just maintain constraints more tightly
        
        # Construct linear constraints
        # h + delta in [0,1], sum(h + delta) = n_points/2
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0  # sum(delta) = 0 ensures sum of h + delta = sum h
        ]
        
        # Minimize linear approximation to max term
        objective = cp.Minimize(cp.sum_squares(delta))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            delta_val = delta.value
        except:
            # Fallback if optimization fails
            delta_val = np.random.normal(0, 0.01, n_points)
        
        # Update h
        alpha = 0.1
        h_new = h + alpha * delta_val
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
