import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # CVXPY approach for optimization
    # Fourier basis functions with odd and even terms
    n_basis = 20
    
    # Create Fourier basis matrix
    x = np.linspace(0, 2, n_points)
    basis_matrix = np.zeros((n_points, n_basis))
    
    for i in range(n_basis):
        if i % 2 == 0:
            basis_matrix[:, i] = np.cos(np.pi * i * x / 2)
        else:
            basis_matrix[:, i] = np.sin(np.pi * i * x / 2)
    
    # Normalize basis
    for i in range(n_basis):
        norm = np.sqrt(np.sum(basis_matrix[:, i]**2) * dx)
        if norm > 0:
            basis_matrix[:, i] /= norm
    
    # Project initial function onto basis
    coeffs_init = np.linalg.lstsq(basis_matrix, h_init, rcond=None)[0]
    
    # Define CVXPY variables
    coeffs = cp.Variable(n_basis)
    h_var = basis_matrix @ coeffs
    
    # Integral constraint
    integral_coeffs = np.array([np.trapz(basis_matrix[:, i], dx=dx) for i in range(n_basis)])
    integral_constraint = cp.sum(cp.multiply(coeffs, integral_coeffs)) == 1
    
    # Bounds on coefficients (via bounds on h)
    bounds = [0 <= h_var, h_var <= 1]
    
    # Objective: minimizing maximum overlap
    # We'll solve an approximate version using McCormick envelopes for products
    
    C = 0
    
    # Full Formulation for direct CPU optimization
    max_iter = 1000
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Setup optimization problem
        objective = cp.Minimize(C)
        constraints = [integral_constraint, *bounds]
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                coeffs_val = coeffs.value
                h_opt = basis_matrix @ coeffs_val
                h_opt = np.clip(h_opt, 0, 1)
                actual_c5 = evaluate_c5(h_opt, dx)
                
                # Update C with improved bound if applicable
                if actual_c5 < C or C == 0:
                    C = actual_c5
                    
                # Break early if we're close to target
                if actual_c5 <= 0.3807:
                    break
                    
                # Update with projection of solution
                coeffs_init = np.linalg.lstsq(basis_matrix, h_opt, rcond=None)[0]
            else:
                break
        except:
            break
    
    # Final optimization using local search with PSO-inspired parameters
    if C >= 0.3807:
        # Re-run optimization with better local search
        max_iter = 1500
        n_particles = 20
        
        # Initialize swarm
        swarm = []
        swarm.append(h_init)
        
        for i in range(1, n_particles):
            h = h_init.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            swarm.append(h)
        
        velocities = [np.zeros(n_points) for _ in range(n_particles)]
        pbest = [swarm[i].copy() for i in range(n_particles)]
        pbest_scores = [evaluate_c5(swarm[i], dx) for i in range(n_particles)]
        
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        w_start, w_end = 0.9, 0.4
        c_p, c_g = 1.5, 1.5
        v_max = 0.1
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            w = w_start - (w_start - w_end) * (iter_count / max_iter)
            
            for i in range(n_particles):
                r_p = np.random.rand(n_points)
                r_g = np.random.rand(n_points)
                velocities[i] = (w * velocities[i] 
                               + c_p * r_p * (pbest[i] - swarm[i])
                               + c_g * r_g * (gbest - swarm[i]))
                
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                swarm[i] = swarm[i] + velocities[i]
                
                swarm[i] = np.clip(swarm[i], 0, 1)
                swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
                
                score = evaluate_c5(swarm[i], dx)
                
                if score < pbest_scores[i]:
                    pbest[i] = swarm[i].copy()
                    pbest_scores[i] = score
                    
                    if score < gbest_score:
                        gbest = swarm[i].copy()
                        gbest_score = score
                        
            if gbest_score <= 0.3807:
                break
                
        return (gbest.tolist(), gbest_score, n_points)
        
    # Return result from CVXPY approach
    return (h_opt.tolist(), C, n_points)
