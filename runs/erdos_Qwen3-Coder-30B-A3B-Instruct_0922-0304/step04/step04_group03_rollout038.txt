import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial shape
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    max_iter = 1000
    learning_rate = 0.01
    patience = 20
    best_c5 = float('inf')
    patience_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            patience_count = 0
        else:
            patience_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if patience_count > patience:
            break
            
        # Compute the gradient
        # For each point, compute how much changing it affects the max overlap
        # We compute overlap for all shifts and find max
        full_correlation = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(full_correlation)
        
        # The gradient is approximately the derivative of the overlap with respect to h
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Contributions to the overlap involve h[i] and h[(i + k_max) % n_points] 
            # Also involve h[(i - k_max) % n_points] in some terms
            if (i + k_max) < n_points:
                gradient[i] += 1 - h[(i + k_max) % n_points]
            else:
                gradient[i] += 1 - h[(i + k_max) % n_points] - h[(i + k_max - n_points) % n_points]
            
            if (i - k_max) >= 0:
                gradient[i] -= h[(i - k_max) % n_points]
            else:
                gradient[i] -= h[(i - k_max) % n_points] + h[(i - k_max + n_points) % n_points]
        
        gradient /= dx
        
        # Update h
        h = h - learning_rate * gradient
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Reduce learning rate if not improving
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
    
    # Refine final result with a bit more optimization
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        current_c5 = evaluate_c5(h, dx)
        full_correlation = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(full_correlation)
        
        # Compute gradient more carefully
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Only look at the term that is maximized
            if (i + k_max) < n_points:
                gradient[i] -= 1 - h[(i + k_max) % n_points]
            else:
                gradient[i] -= 1 - h[(i + k_max) % n_points] - h[(i + k_max - n_points) % n_points]
            
            if (i - k_max) >= 0:
                gradient[i] += h[(i - k_max) % n_points]
            else:
                gradient[i] += h[(i - k_max) % n_points] + h[(i - k_max + n_points) % n_points]
        
        h = h - 0.001 * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = evaluate_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
