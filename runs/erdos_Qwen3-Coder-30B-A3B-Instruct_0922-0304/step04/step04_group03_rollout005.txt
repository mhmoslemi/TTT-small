import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, dx, n_points):
    # Ensure h is within [0,1] bounds
    h = np.clip(h, 0, 1)
    # Scale to meet constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth initial function
        x = np.linspace(0, 2, n_points)
        h = np.exp(-10 * (x - 1)**2)
        h = h / np.sum(h) * (n_points / 2)
    
    h = project_h(h, dx, n_points)
    
    # Gradient-based optimization
    max_iter = 50000
    learning_rate = 0.001
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Compute gradient of max overlap w.r.t. h
        grad = np.zeros(n_points)
        # For a shift k, O_k = sum(h_i * (1 - h_{i+k}))
        # dO_k/dh_i = (1 - h_{i+k}) - h_{i-k}
        # We have max_overlap at index max_overlap_idx
        k = max_overlap_idx - (n_points - 1)
        for j in range(n_points):
            if k >= 0:
                grad[j] = (1 - h[(j + k) % n_points]) - h[(j - k) % n_points]
            else:
                grad[j] = (1 - h[(j - k) % n_points]) - h[(j + k) % n_points]
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, dx, n_points)
        
        # Check if we improved
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
