import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # Each interval width
        a = 0    # Start position
        # Try different separations
        best_c5 = float('inf')
        best_h = None
        
        # Grid search over separation distances
        d_candidates = np.arange(0.0, 1.0 + 0.005, 0.005)
        
        for d in d_candidates:
            if (a + w + d + w) > 2.0:
                continue
            
            # Compute indices for two intervals
            i0 = 0
            i1 = int(np.floor((a + w) / dx))
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx))
            
            # Ensure indices are valid
            i1 = min(i1, n_points - 1)
            i3 = min(i3, n_points - 1)
            i2 = max(i2, 0)
            
            # Create the h function
            h_test = np.zeros(n_points)
            h_test[i0:i1+1] = 1.0
            h_test[i2:i3+1] = 1.0
            
            # Normalize so that sum(h) * dx = 1
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            # Compute C5
            c5 = evaluate_c5(h_test, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_test.copy()
        
        h = best_h
    
    # Refine using gradient-based optimization
    # Set up the optimization problem
    def obj(h_vec):
        # Normalize
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return evaluate_c5(h_vec, dx)
    
    # Perform simple local search around best result
    best_c5 = obj(h)
    best_h = h.copy()
    
    n_iter = 0
    max_iter = 5000
    
    while n_iter < max_iter and time.time() - start_time < budget_s - 1:
        # Try small perturbations
        h_new = h + np.random.normal(0, 0.03, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = obj(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        
        n_iter += 1
        
        # Also try some systematic moves for the two-interval case
        if n_iter % 10 == 0:
            # Vary the separation distance slightly
            w = 0.5
            a = 0
            d = (a + w + a + w + 0.5) / 2.0  # middle
            d_candidates = [d - 0.01, d, d + 0.01]
            
            for d_try in d_candidates:
                if d_try < 0 or d_try > 1.0:
                    continue
                i0 = 0
                i1 = int(np.floor((a + w) / dx))
                i2 = int(np.ceil((a + w + d_try) / dx))
                i3 = int(np.floor((a + w + d_try + w) / dx))
                i1 = min(i1, n_points - 1)
                i3 = min(i3, n_points - 1)
                i2 = max(i2, 0)
                h_test = np.zeros(n_points)
                h_test[i0:i1+1] = 1.0
                h_test[i2:i3+1] = 1.0
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                c5 = evaluate_c5(h_test, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_test.copy()
                    h = h_test.copy()
    
    return (best_h.tolist(), best_c5, n_points)
