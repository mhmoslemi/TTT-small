import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to meet constraint sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        scale = target_sum / current_sum
        h = h * scale
    return h

def armijo_line_search(h, delta, dx, n_points, c1=1e-4, alpha_max=1.0):
    # Try to find good alpha, try alpha/2 if not good enough
    # If not improving, use falling step size
    alpha = alpha_max
    current_val = evaluate_c5(h, dx)
    
    max_iter = 10
    for _ in range(max_iter):
        h_trial = h + alpha * delta
        h_trial = project_to_feasible(h_trial, n_points, dx)
        trial_val = evaluate_c5(h_trial, dx)
        
        # Armijo condition: f(x+alpha*d) <= f(x) + c1*alpha*<grad,d>
        # Use <grad,d> = np.dot(h_trial - h, delta) / alpha for approx.
        grad_approx = delta  
        armijo_cond = trial_val <= current_val + c1 * alpha * np.dot(grad_approx, delta)
        
        if armijo_cond or trial_val <= current_val:
            return alpha, h_trial
        
        alpha *= 0.5
        
    # return fallback
    return alpha, h_trial
