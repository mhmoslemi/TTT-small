import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a triangular shape as initial guess
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    patience_counter = 0
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5 value
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
        if best_c5 <= 0.3807:
            break
        
        # Find the index that maximizes overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient for the worst case overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Forward difference
            if i + k_max < n_points:
                gradient[i] += (1 - h[i + k_max])
            else:
                gradient[i] += (1 - h[(i + k_max) % n_points])
            
            # Backward difference  
            if i - k_max >= 0:
                gradient[i] -= h[i - k_max]
            else:
                gradient[i] -= h[(i - k_max) % n_points]
                
        # Normalize gradient
        gradient = gradient / np.linalg.norm(gradient)
        
        # Update h with gradient descent
        h += -learning_rate * gradient
        
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Reduce learning rate if no improvement
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
            
    return (h.tolist(), best_c5, n_points)
