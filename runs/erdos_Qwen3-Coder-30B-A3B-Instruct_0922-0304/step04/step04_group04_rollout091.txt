import numpy as np
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Check for initial solution
    h_init = None
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Use a hybrid approach: start with parent or a simple construction
    if h_init is not None:
        h = h_init.copy()
    else:
        # Simple initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    
    # Grid search with local refinement
    n_iters = 2000
    step_sizes = [0.05, 0.02, 0.01]
    
    for i in range(n_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current solution quality
        current_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Vary the solution
        step_size = step_sizes[min(i // 500, len(step_sizes)-1)]
        new_h = h.copy()
        
        # Perturb a few random points
        indices = np.random.choice(n_points, size=min(5, n_points//4), replace=False)
        for idx in indices:
            new_h[idx] += np.random.normal(0, step_size)
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Check improvement
        new_c5 = np.max(np.correlate(new_h, 1-new_h, mode='full')) * dx
        if new_c5 < current_c5:
            h = new_h
        elif np.random.rand() < 0.1:  # Occasionally accept worse steps to escape local minima
            h = new_h
    
    # Final validation
    final_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx
    return (best_h.tolist(), final_c5, n_points)
