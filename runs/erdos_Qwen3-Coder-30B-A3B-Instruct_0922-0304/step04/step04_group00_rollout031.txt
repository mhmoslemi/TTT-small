import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 600
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Coordinate-wise updates
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Create a copy to test update
            h_test = h.copy()
            
            # Compute current contribution to C5
            current_c5 = compute_c5(h)
            
            # Try small perturbations
            step_size = 0.01
            best_h_i = h[i]
            best_c5_i = current_c5
            
            # Try positive and negative steps
            for delta in [-step_size, step_size]:
                h_test[i] = h[i] + delta
                h_test = np.clip(h_test, 0, 1)
                # Normalize
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                new_c5 = compute_c5(h_test)
                if new_c5 < best_c5_i:
                    best_c5_i = new_c5
                    best_h_i = h_test[i]
            
            # Apply best update
            h[i] = best_h_i
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
