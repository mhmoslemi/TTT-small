import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Alternating optimization
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize magnitudes while fixing support (simple projection)
        current_c5 = evaluate_c5(h, dx)
        
        # Generate neighbor by small random perturbation
        h_new = h + np.random.normal(0, 0.01, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Early stopping condition
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
