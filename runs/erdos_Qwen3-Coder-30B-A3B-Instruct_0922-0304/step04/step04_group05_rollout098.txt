import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to integral = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Coordinate descent approach
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update magnitudes while keeping support fixed
        # Simple gradient ascent on magnitudes 
        h_new = h.copy()
        step_size = 0.01
        
        # Gradient approximated by finite differences
        for i in range(n_points):
            if h[i] < 1e-6:  # Skip near-zero values
                continue
            h_test = h.copy()
            h_test[i] += step_size
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5_test = evaluate_c5(h_test, dx)
            if c5_test < best_c5:
                h_new = h_test
                best_c5 = c5_test
                best_h = h_test.copy()
                if best_c5 <= 0.3807:
                    break

        h = h_new.copy()
        
        # Simple support shaping: shift around active regions
        # Find active regions and try to slide them
        active_indices = np.where(h > 1e-3)[0]
        if len(active_indices) > 0:
            # Shift all active regions by small amounts
            shifts = np.random.normal(0, 0.1, len(active_indices))
            new_h = np.zeros(n_points)
            for i, idx in enumerate(active_indices):
                # Try shifting: compute new positions
                shifted_idx = int(idx + shifts[i])
                if 0 <= shifted_idx < n_points:
                    new_h[shifted_idx] += h[idx]
            if np.sum(new_h) > 0:
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                c5_test = evaluate_c5(new_h, dx)
                if c5_test < best_c5:
                    h = new_h
                    best_c5 = c5_test
                    best_h = h.copy()
                    if best_c5 <= 0.3807:
                        break
                        
    return (best_h.tolist(), best_c5, n_points)
