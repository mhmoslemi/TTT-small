import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            # Store current value
            old_h_i = h[i]
            
            # Try different values for h[i]
            best_h_i = old_h_i
            best_local_c5 = current_c5
            
            # Try small perturbations
            steps = np.linspace(-0.1, 0.1, 21)
            for step in steps:
                new_h_i = old_h_i + step
                # Ensure it's within bounds
                new_h_i = np.clip(new_h_i, 0, 1)
                if new_h_i == old_h_i:
                    continue
                    
                # Temporarily update
                h[i] = new_h_i
                
                # Renormalize to maintain sum constraint
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                h = np.clip(h, 0, 1)
                
                # Renormalize again to be sure
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum

                new_c5 = evaluate_c5(h, dx)

            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            
            # Early stopping if no improvement
            if best_local_c5 < current_c5:
                current_c5 = best_local_c5
            else:
