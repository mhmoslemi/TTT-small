import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=2.0/100):
    """Compute the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum=1.0):
    """Normalize h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Fallback initialization
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = normalize_h(base_h, target_sum)
    
    # Create initial population
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.1
    
    # Track best solution
    global_best_h = None
    global_best_c5 = float('inf')
    p_best_h = [p.copy() for p in particles]
    p_best_c5 = [compute_c5(p, dx) for p in particles]
    
    # Main PSO loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update global best
        for i in range(n_particles):
            current_c5 = compute_c5(particles[i], dx)
            if current_c5 < p_best_c5[i]:
                p_best_c5[i] = current_c5
                p_best_h[i] = particles[i].copy()
            
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = particles[i].copy()
        
        # Early exit if target achieved
        if global_best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best_h[i] - particles[i]) + 
                           c2 * r2 * (global_best_h - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
    
    # Final evaluation
    final_c5 = compute_c5(global_best_h, dx) if global_best_h is not None else float('inf')
    
    return (global_best_h.tolist() if global_best_h is not None else np.zeros(n_points).tolist(), 
            final_c5, n_points)
