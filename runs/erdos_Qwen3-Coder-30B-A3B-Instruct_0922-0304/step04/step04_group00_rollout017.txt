import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - uniform with small perturbations
        h = np.ones(n_points) * 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Compute all correlations
        overlap = correlate(h, 1 - h, mode="full")
        # Return the maximum value scaled by dx
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting plane algorithm parameters
    epsilon = 0.001
    max_iter = 5000
    constraint_freq = 5
    
    # Constraint storage
    constraints_list = []
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Add new constraint every constraint_freq iterations
        if iteration % constraint_freq == 0:
            # Find maximum overlap shift
            overlap = correlate(h, 1 - h, mode="full")
            max_index = np.argmax(overlap)
            max_shift = max_index - (n_points - 1)
            
            # Create the constraint (signs may need adjustment based on indexing)
            # For h * (1 - h_k) <= C - epsilon
            # This is a linear constraint on h
            # We construct a constraint for the specific shift
            # Constraint: sum(h[i] * (1 - h[i+shift])) <= C - epsilon
            
            # Since we're working with a circular correlation, we must be careful
            # Build the constraint row by row
            constraint_coeffs = np.zeros(n_points)
            if max_shift >= 0:
                # h[i] * (1 - h[i+shift])
                for i in range(n_points - max_shift):
                    constraint_coeffs[i] += 1.0
                    constraint_coeffs[i + max_shift] -= 1.0
            else:
                # shift is negative, so we flip
                for i in range(-max_shift, n_points):
                    constraint_coeffs[i] += 1.0
                    constraint_coeffs[i + max_shift] -= 1.0
            
            # Normalize appropriately or just keep as is for constraint
            # Update constraint if the new one improves the current constraint
            constraints_list.append((constraint_coeffs, current_c5 - epsilon))
            
            # If we're overconstraining, keep last few constraints only
            # This helps stability
            if len(constraints_list) > 10:
                constraints_list = constraints_list[-5:]
    
        # Solve the next optimization
        # Create a new variable for optimization
        h_var = cp.Variable(n_points)
        
        # Objective: minimize the max overlap
        # Use a variable to track an upper bound on C5
        c5_var = cp.Variable(1)
        
        # Constraints
       objective = cp.Minimize(c5_var)
        
        # Sum constraint  
        constraints = [cp.sum(h_var) == n_points / 2.0]
        
        # Box constraints
        constraints.extend([h_var[i] >= 0 for i in range(n_points)])
        constraints.extend([h_var[i] <= 1 for i in range(n_points)])
        
        # Cutting plane constraints
        # Each constraint is: a * h <= b
        for a, b in constraints_list:
            # Compute the constraint as: sum(a[i] * h_var[i]) <= b
            # Standard form: sum(a[i] * h_var[i]) - b <= 0
            # So we use: a @ h_var <= b
            constraints.append(a @ h_var <= b)
        
        constraints.append(c5_var >= cp.max(cp.multiply(h_var, 1 - h_var)))
        # Or we approximate the max with a quadratic constraint?
        # Let's simplify by minimizing over the set of constraints
        
        # We want to minimize h @ (1 - h) over the rotation
        # We will be more explicit and approximate the constraint manually
        # Instead, let's solve using more direct method:
        
        # Short term solution: hinge on convex optimization
        
        problem = cp.Problem(objective, constraints)
        
        try:
            # Try to solve with small tolerance
            problem.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Ensure it's valid
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
            else:
                # Perturb if we can't optimize
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to perturbation
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
