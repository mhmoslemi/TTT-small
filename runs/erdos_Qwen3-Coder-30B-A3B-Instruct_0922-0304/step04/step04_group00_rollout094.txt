import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting-plane method parameters
    epsilon = 0.001
    max_iterations = 10000
    constraint_frequency = 5
    
    # Store constraints
    constraints_list = []
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Add constraint every few iterations
        if iteration % constraint_frequency == 0:
            overlap = correlate(h, 1 - h, mode="full")
            max_shift_idx = np.argmax(overlap)
            max_overlap = overlap[max_shift_idx] * dx
            
            # Add linear constraint based on maximum overlap shift
            # This is a simplification - we're constraining the specific shift found
            new_constraint = max_overlap - epsilon
            constraints_list.append(new_constraint)
            
            # Create CVXPY variables
            h_var = cp.Variable(n_points)
            
            # Objective: minimize C5 (we'll use a linear approximation)
            # Minimize max shift, subject to constraints
            prob = cp.Problem(
                cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var))),
                [
                    cp.sum(h_var) == n_points / 2,
                    h_var >= 0,
                    h_var <= 1
                ]
            )
            
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h = np.array(h_var.value).flatten()
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except:
                # Fall back to simple optimization
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Simple gradient-based update
            h -= 0.001 * (h - 0.5)  # Simple proximity to 0.5
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
