import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using gradient descent with adaptive step size
    max_iter = 10000
    step_size = 0.003
    stagnation_counter = 0
    prev_best = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and gradient
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps

        # Update step with adaptive learning rate
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if improvement
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            stagnation_counter = 0
            # Increase step size slightly for faster convergence
            step_size = min(0.01, step_size * 1.02)
        else:
            stagnation_counter += 1
            # Reduce step size if no progress
            step_size *= 0.95
            
        # Apply update
        h = h_new
        
        # Apply random perturbations occasionally
        if iteration % 100 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size decay
        if stagnation_counter > 50:
            step_size *= 0.9
            stagnation_counter = 0

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
