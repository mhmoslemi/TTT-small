import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Simple symmetric triangular construction
        h0 = np.zeros(n_points)
        mid = n_points // 2
        h0[:mid] = np.linspace(0, 1, mid)
        h0[mid:] = np.linspace(1, 0, n_points - mid)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial particles around the initial solution
    for i in range(n_particles):
        particles[i] = h0 + np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.01, 0.01, (n_particles, n_points))
    
    # PSO parameters
    omega = 0.9
    phi_p = 2.0
    phi_g = 2.0
    
    # Store personal best positions and their C5 values
    personal_best = particles.copy()
    personal_best_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # Main PSO loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                omega * velocities[i] +
                phi_p * r1 * (personal_best[i] - particles[i]) +
                phi_g * r2 * (global_best - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            c5 = evaluate_c5(particles[i], dx)
            
            # Check if this is a new personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = c5
                
                # Check if this is a new global best
                if c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = c5

        # Periodic reinitialization to prevent stagnation
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                particles[idx] = h0 + np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
                
        # Early exit if target achieved
        if global_best_c5 <= 0.3807:
            break
    
    # Return best solution
    return (global_best.tolist(), float(global_best_c5), n_points)
