import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 600  # Allow up to 600 iterations within time budget
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        for i in range(n_points):

            best_local_c5 = float('inf')

                # Normalize to maintain sum constraint
                total = np.sum(h)
                h *= (n_points / 2) / total

                local_c5 = compute_c5(h, dx)

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
