import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to match target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradients(h, dx, n_points):
    # Compute gradient of C with respect to h
    # ∇C = dx * (1 - 2*h - shift average)
    
    # We'll approximate gradient for all shifts by computing overlap values
    # For an efficient gradient estimate, we'll use a sparse approach
    # For large shift indices, we can derive a simple gradient estimate
    
    # Here we compute C_k = ∫ h(x) * (1 - h(x+k)) dx
    # Gradient w.r.t. h[i] is dx * (1 - h[i+k] - h[i]) * delta_il 
    # But this is complex. Simpler: compute approximate gradient by forward difference
    
    # For now, we use the fact that for each shift k,
    # the gradient terms for h[i] is: 1 - h[i+k] - h[i] for k >= 0
    # We'll take an approximation of ∇C w.r.t h
    # The gradient at each point sums contributions over all shifts
    # Exploit the structure that if we shift with k, we have h[i] * (1 - h[i+k])
    # So for first order approx of derivative ∂C/∂h[i] = - h[i+k] + 1 for such shifts
    
    # So for precise gradient computation:
    # ∑_k ∫ h(x) (1 - h(x+k)) dx = ∑_k ∑_i h[i] (1 - h[i+k]) 
    # So ∂C/∂h[i] = ∑_k (1 - h[i+k] - h[i])
    # Due to mode="full", we compute effects for k from -n_points to n_points
    # In context: we care about the maximum k effect (C_k for all k)
    # We run this for one representative example of overlapping terms
    
    # To make this practical, approximate by using overlapping sums
    # ∇C[i] = dx * (1 - 2 * sum(h[i+k]) - h[i]) for k in valid range
    
    # Compute the shift sum for all possible shifts in relevance
    # Efficient linear indexing approach
    
    grad = np.zeros(n_points)
    
    # Compute overlaps across shifts
    # Approximate ∇C[i] ≈ -2 * h[i+k] + 1 for overlapping terms 
    # Since shift operations have symmetric impact on gradients
    
    # For j-th element, gradient computation = dx * (1 - 2*sum(h[j+i+k]) - h[j]) 
    # Therefore we estimate it as ∇C[i] = dx * (1 - h[i] - 2 * sum shifted terms)
    
    # Instead, we will compute gradient by finite differences  
    # For small step size h_step, and force it starts with +1 and -1
        
    # In our case for simplicity and based on the formula structure:
    # ∇C[i] ~ dx * (1 - 2 * h[i] - (sum h[i+k]))
    
    # We can get exact linearized gradient at each evaluation using the full overlaps:
    # Each shift k contributes 1 to ∂C/∂h[i] if i+k is valid and vice versa
    # Direct approach: compute full matrix of overlaps used for gradient estimation
    # For simplicity, approximate gradient with a direct linear algebra approach
    
    # Gradient of C = dx * [ ∫ h(x)(1 - h(x+k)) dx ]_k
    # Approximate ∇C[i] = dx * [ ∑_k (1 - h[i+k] - h[i]) ] 
    # The term h[i+k] accounts for all overlaps in k direction.
    
    # Use direct approximation
    # Let’s compute the overlap directly and assume bias correction for gradient
    
    # Sub-optimal but executable: first an approximate gradient version
    
    # Simple empirical gradient calculation:
    # What is the gradient of max term about h?
    # Let’s compute an approximate gradient as derivative of overlap sum
    
    # Assume gradient structure: for k-th term: 
    # One term in derivative: d/dh_i [ h_j (1 - h_{j+k}) ] = δ_{ij} (1 - h_{i+k}) + (-h_{i+k} δ_{i,j+k})  
    # For max({C_k}), we use approximate Gradients:
    # Let ∇C_k[i] = (1 - h[i+k] - h[i])  
    # Let ∇max C_k = ∇C_k for the k that maximizes overlap
    # Or just consider common shorthand for the gradient for highest shift

    # Given problem complexity, we take the most simple version that works well:
    # Approximate current gradient of total overlap optimistically:
    # We derive ∇C for now as dx*(1 - 2*h), with a fudge for the shift effects from symmetric sums
    
    # Approximate gradient using shift sums:
    grad = np.zeros_like(h)
    for shift_k in range(n_points):
        # Shift right by k steps  
        shifted = np.roll(h, shift_k)
        # Add to gradient
        grad += 1 - shifted - h
        
    grad = grad * dx
    return grad 

def adaptive_gradient_descent(h0, dx, target_sum, max_iter=1000, tol=1e-6):
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    
    prev_c5_value = float('inf')
    gamma = 0.05
    stagnation_count = 0
    
    for iteration in range(max_iter):
        c5_value = evaluate_c5(h, dx)
        if iteration % 100 == 0:
            print(f"Iteration {iteration}, C5: {c5_value:.6f}")
            
        if abs(prev_c5_value - c5_value) < tol:
            stagnation_count += 1
            if stagnation_count > 5:
                break
        else:
            stagnation_count = 0
            
        prev_c5_value = c5_value
        
        if c5_value <= 0.3807:
            break
            
        # Compute gradient
        # We now directly approximate the gradient
        grad = compute_gradients(h, dx, len(h))
        
        # Minimize ∇C with step size adjustment based on momentum fix
        update = -gamma * grad
        
        # Project to feasible set
        h_new = h + update
        h_new = project_to_feasible(h_new, target_sum)
        
        # Simple backtracking line search
        alpha = 1.0
        while alpha > 1e-8:
            h_try = h + alpha * update
            h_try = project_to_feasible(h_try, target_sum)
            try_c5 = evaluate_c5(h_try, dx)
            if try_c5 <= c5_value:
                h = h_try
                break
            alpha *= 0.5
        
        # Adaptive gamma adjustment
        if c5_value < 0.385:
            gamma *= 1.2
        else:
            gamma *= 0.95
            
        gamma = max(gamma, 1e-6)
    
    return h, evaluate_c5(h, dx)
