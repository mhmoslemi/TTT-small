import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation
        c5_old = evaluate_c5(h, dx)
        
        # Finite difference gradient estimate
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[i] = (c5_plus - c5_old) / epsilon
            
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
