import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple triangular distribution
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
        
    # Normalize to satisfy the constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization
    learning_rate = 0.05
    decay_rate = 0.999
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        # The overlap function is periodic we need to compute it correctly over full range
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)  # convert to actual shift value
        
        # Derivative of overlap with respect to h_i for the max shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from h[i] term
            if (i + max_shift) % n_points < n_points and (i + max_shift) >= 0:
                gradient[i] += 1 - h[(i + max_shift) % n_points]
            # Gradient contribution from h[i] term in the reversed correlation
            if (i - max_shift) % n_points < n_points and (i - max_shift) >= 0:
                gradient[i] -= h[(i - max_shift) % n_points]
        
        # Update h
        h_new = h - learning_rate * gradient
        
        # Project to valid range
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize the sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject the update
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Accept with small probability to escape local minima
            if np.random.random() < 0.01:
                h = h_new.copy()
        
        # Adaptive learning rate decay
        learning_rate *= decay_rate
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
