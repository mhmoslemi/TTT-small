import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 200
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness_scores = []
    
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness_scores.append(evaluate_c5(individual, dx))
    
    # Keep track of best solution
    for i in range(population_size):
        if fitness_scores[i] < best_c5:
            best_c5 = fitness_scores[i]
            best_h = population[i].copy()
    
    # GA iterations
    for generation in range(num_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Tournament selection
        new_population = []
        new_fitness = []
        
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            
            # Copy winner
            parent = population[winner_index].copy()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                partner_index = np.random.choice(population_size)
                partner = population[partner_index]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent[:crossover_point], partner[crossover_point:]])
            else:
                child = parent.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for idx in mutation_indices:
                    child[idx] = np.clip(child[idx] + np.random.normal(0, 0.1), 0, 1)
            
            # Rescale to maintain constraint sum(h) = n_points/2
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
            new_fitness.append(evaluate_c5(child, dx))
            
            # Update best solution
            if new_fitness[-1] < best_c5:
                best_c5 = new_fitness[-1]
                best_h = child.copy()
                
            if best_c5 <= 0.3807:
                break
        
        population = new_population
        fitness_scores = new_fitness
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
