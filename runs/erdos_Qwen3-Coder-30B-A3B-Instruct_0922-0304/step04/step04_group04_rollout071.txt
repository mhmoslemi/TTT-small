import numpy as np
from scipy.signal import correlate

def generate_haar_basis(n_basis=30):
    # Generate Haar wavelet basis functions
    basis = []
    x = np.linspace(0, 2, 1000)
    
    # First basis function: step function
    psi1 = np.zeros_like(x)
    psi1[x < 0.5] = 1
    psi1[(x >= 0.5) & (x < 1)] = -1
    basis.append(psi1)
    
    # Generate higher order Haar wavelets
    for i in range(1, n_basis):
        scale = 2**(i // 2)
        shift = (i % 2) * (1.0 / scale)
        psi = np.zeros_like(x)
        x_scaled = (x - shift) * scale
        psi[(x_scaled >= 0) & (x_scaled < 0.5)] = 1
        psi[(x_scaled >= 0.5) & (x_scaled < 1)] = -1
        basis.append(psi)
    
    return np.array(basis)

def compute_inner_products(basis):
    # Compute M[i,j,k] = integral of basis[i](x) * basis[j](x+k) dx
    n_basis = basis.shape[0]
    n_points = basis.shape[1]
    M = np.zeros((n_basis, n_basis, n_points))
    
    for i in range(n_basis):
        for j in range(n_basis):
            for k in range(n_points):
                # Convolve basis[i] with basis[j] shifted by k
                if k == 0:
                    integrand = basis[i] * basis[j]
                else:
                    shifted = np.roll(basis[j], k)
                    integrand = basis[i] * shifted
                M[i, j, k] = np.trapz(integrand, np.linspace(0, 2, n_points))
    return M

def evaluate_c5(h_values, dx):
    # Evaluate overlap integral for h(x)
    if len(h_values) != 70:
        raise ValueError("h_values must have 70 samples")
    overlap = correlate(h_values, 1 - np.array(h_values), mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
