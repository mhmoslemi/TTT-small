import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint
        h = h * (70 / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(70)
        h[35:] = 0
        h += np.random.normal(0, 0.05, 70)
        h = np.clip(h, 0, 1)
        h = h * (70 / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Basis functions approach
    n_basis = 20
    n_points = 70
    dx = 2.0 / n_points
    x_points = np.linspace(0, 2, n_points)
    
    # Create basis functions (quadratic on overlapping intervals)
    x_basis = np.linspace(0, 2, n_basis + 1)
    basis_width = 0.2
    basis_functions = []
    for i in range(n_basis):
        x0 = x_basis[i]
        x1 = x_basis[i+1]
        # Support interval: [x0-basis_width, x1+basis_width]
        x_support = np.linspace(max(0, x0-basis_width), min(2, x1+basis_width), 100)
        phi = np.zeros_like(x_support)
        # Quadratic basis (not really smooth but okay)
        for j, x in enumerate(x_support):
            if x0 <= x <= x1:
                # Quadratic peak at center
                center = 0.5 * (x0 + x1)
                offset = x - center
                # Ensure non-negative and more localized
                phi[j] = max(0, 1 - (offset / (basis_width/2))**2)
        # Account for integration in dx grid
        basis_functions.append(phi)
        
    # Sample basis functions at grid points
    basis_samples = np.zeros((n_basis, n_points))
    for i in range(n_basis):
        basis_samples[i] = np.interp(x_points, 
                                   np.linspace(max(0, x_basis[i]-basis_width), 
                                             min(2, x_basis[i+1]+basis_width), 100),
                                   basis_functions[i])
        # Normalize each basis to have integral of 1 (approximately for this simple case)
        # This can be refined, but for now keep it simple
    
    # Initialize weights
    w = np.ones(n_basis) / n_basis
    w[0] = 0.1  # Initial skew
    w[-1] = 0.1
    
    # Optimization loop with Adam-like scheme
    max_iter = 1000
    learning_rate = 0.01
    beta1 = 0.9
    momentum_w = np.zeros_like(w)
    velocity_w = np.zeros_like(w)
    eps = 1e-8
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute h based on weights
        h = np.dot(w, basis_samples)
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)

        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient w.r.t weights (finite difference)
        grad_w = np.zeros_like(w)
        eps_w = 1e-5
        for i in range(n_basis):
            w_plus = w.copy()
            w_plus[i] += eps_w
            h_plus = np.dot(w_plus, basis_samples)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            grad_w[i] = (c5_plus - current_c5) / eps_w
            
        # Adaptive gradient updates (Adam-like)
        momentum_w = beta1 * momentum_w + (1 - beta1) * grad_w
        velocity_w = beta1 * velocity_w + (1 - beta1) * grad_w**2
        momentum_corrected = momentum_w / (1 - beta1**(iteration+1))
        velocity_corrected = velocity_w / (1 - beta1**(iteration+1))
        w -= learning_rate * momentum_corrected / (np.sqrt(velocity_corrected) + eps)
        
        # Project to valid range
        w = np.clip(w, 0, 1)
        # Renormalize
        w = w / np.sum(w) * n_basis
        
        # Occasionally perturb to avoid local optima
        if iteration % 50 == 0:
            w += np.random.normal(0, 0.001, n_basis)
            w = np.clip(w, 0, 1)
            w = w / np.sum(w) * n_basis
            
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, 70)
