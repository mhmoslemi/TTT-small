import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    except:
        # Fallback initialization
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Wavelet-based optimization approach
    # Generate Haar wavelet basis (simplified)
    basis_size = min(30, n_points)
    
    def haar_wavelets(n):
        # Generate Haar wavelets up to scale log2(n)
        waves = []
        for j in range(6):  # up to 6 levels
            for k in range(2**j):
                wave = np.zeros(n)
                start = k * 2**(5-j)
                end = start + 2**(5-j)
                if end <= n:
                    mid = (start + end) // 2
                    wave[start:mid] = 1
                    wave[mid:end] = -1
                    waves.append(wave)
        return np.array(waves[:basis_size])
    
    # Create basis
    basis = haar_wavelets(n_points)
    basis = np.vstack([np.ones(n_points), basis])
    basis = basis[:basis_size]
    
    # Initial guess based on parent
    if 'initial_h_values' in globals():
        # Project initial_h_values to wavelet coefficients
        try:
            h_parent = np.array(initial_h_values)
            h_parent = np.clip(h_parent, 0, 1)
            h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
            coeffs = np.linalg.lstsq(basis.T, h_parent, rcond=None)[0]
        except:
            coeffs = np.random.rand(basis_size) * 0.1
    else:
        coeffs = np.random.rand(basis_size) * 0.1
    
    # CVXPY optimization
    coeffs_var = cp.Variable(basis_size)
    
    # Reconstruct h from wavelet coefficients
    h_recon = basis.T @ coeffs_var
    h_recon = cp.reshape(h_recon, n_points)
    
    # Constraints
    constraints = [
        cp.sum(h_recon) == n_points / 2,  # integral constraint
        h_recon >= 0,                     # non-negativity
        h_recon <= 1                      # bound constraint
    ]
    
    # Objective: minimize overlap
    # This is a complex non-convex constraint; we'll use a simpler direct approach
    # Define an upper bound as convex approximation
    overlap_expr = cp.Variable()
    
    # Simple optimization via gradient descent on the wavelet coefficients
    learning_rate = 0.001
    max_iter = 5000
    
    h_current = np.clip(basis.T @ coeffs, 0, 1)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    current_score = evaluate_c5(h_current, dx)
    
    best_h = h_current.copy()
    best_score = current_score
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        h_current = np.clip(basis.T @ coeffs, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        current_score = evaluate_c5(h_current, dx)
        
        # Adaptively reduce learning rate to avoid rough oscillations
        if current_score < best_score:
            best_h = h_current.copy()
            best_score = current_score
            learning_rate *= 1.05  # Increase learning rate
        else:
            learning_rate *= 0.95  # Decrease learning rate to stabilize
            
        # Simple gradient-based update
        # Note: this is a very simplified gradient approximation
        if i % 50 == 0:  # Every few iterations
            try:
                # Evaluate the effect of small perturbations
                directions = np.random.randn(basis_size) * 0.1
                perturbed_coeffs = coeffs + directions * learning_rate
                h_perturb = np.clip(basis.T @ perturbed_coeffs, 0, 1)
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                new_score = evaluate_c5(h_perturb, dx)
                if new_score < current_score:
                    coeffs = perturbed_coeffs.copy()
            except:
                pass  # Just proceed with current coeffs
        else:
            # Just perform small updates
            coeffs = coeffs - learning_rate * np.random.randn(basis_size) * 0.001
    
    h_final = best_h.copy()
    
    # Final verification
    final_score = evaluate_c5(h_final, dx)
    return (h_final.tolist(), final_score, n_points)
