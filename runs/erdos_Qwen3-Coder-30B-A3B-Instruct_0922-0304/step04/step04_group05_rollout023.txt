import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1.0, 2.0 * abs(j - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Precompute full correlation for faster evaluation
    def compute_correlation(h):
        corr = correlate(h, 1 - h, mode="full")
        return corr
    
    corr_full = compute_correlation(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimization step: perturb h slightly
        h_new = h.copy()
        perturbation_magnitude = 0.01
        
        # Choose random indices to modify
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            # Add noise
            delta = np.random.normal(0, perturbation_magnitude)
            h_new[idx] += delta
        
        # Apply bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
