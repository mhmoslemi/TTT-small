import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis functions
    def haar_wavelet(x, j, k):
        # Haar wavelet at scale j, translation k
        if j == 0:
            if k == 0:
                return 1 if 0 <= x < 0.5 else (-1 if 0.5 <= x < 1 else 0)
            else:
                return 0
        else:
            # Recursive construction
            x = 2**j * x - k
            if 0 <= x < 0.5:
                return 1
            elif 0.5 <= x < 1:
                return -1
            else:
                return 0

    def compute_haar_basis(n_basis=30):
        # Precompute Haar basis functions for this resolution
        basis = []
        for j in range(5):  # up to level 4 (scale 1/16)
            for k in range(2**j):
                basis.append(np.array([haar_wavelet(x, j, k) for x in np.linspace(0, 2, n_points)]))
        return basis[:n_basis]

    def compute_overlap_matrix(basis):
        n = len(basis)
        overlap = []
        for i in range(n):
            row = []
            for j in range(n):
                # Compute integral of basis[i](x) * basis[j](x+k) over [0,2]
                integrals = []
                for k_shift in range(n_points):
                    integral = 0
                    for l in range(n_points):
                        x = l * dx
                        # Construct shifted basis function at position k_shift
                        shifted_val = basis[j][min(max(0, k_shift - (l - j)), n_points - 1)]
                        integral += basis[i][l] * shifted_val * dx
                    integrals.append(integral)
                # Sum over all shifts
                overlap.append(np.sum(integrals))
            overlap.append(row)
        return np.array(overlap).reshape(n, n)

    def project_to_basis(h_vals, basis, n_basis):
        coeffs = []
        for psi in basis[:n_basis]:
            coeffs.append(np.sum(h_vals * psi) * dx)
        return np.array(coeffs)

    def reconstruct_h(coeffs, basis, n_basis):
        h = np.zeros(n_points)
        for i in range(min(n_basis, len(coeffs))):
            h += coeffs[i] * basis[i]
        return h

    # Generate Haar basis
    basis = compute_haar_basis(30)  # Use 30 basis functions
    n_basis = len(basis)

    # Initialize from parent or create simple starting point
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Trivial init
        h_init = np.ones(n_points)
        h_init[:n_points//2] = 0
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Project to basis
    coeffs = project_to_basis(h_init, basis, n_basis)

    # Initialize optimization variables
    c = cp.Variable(n_basis)
    z = cp.Variable((n_basis, n_basis), symmetric=True)  # For McCormick products
    
    # Optimality conditions for overlap
    # We want to minimize max_k (h(x) * (1 - h(x + k))) = max_k integral of this product over [0,2]

    # Formulate overlap in terms of basis coefficients
    # This will involve positive semi-definite constraints and extra variables for literature Mackinney bounds.

    # Simplified approach: Direct optimization with constraints
    # Create a suitable optimization formulation

    # Pick constraints directly:
    integral_constraint = cp.sum(c) == n_points / 2  # Normalized w.r.t. dx=2/n_points
    
    # McCormick envelope for c_i*c_j
    # Since c_i,c_j are optimized and are assumed non-negative
    # and we are taking a max overall integral (approx. form)
    def objective_from_coeffs(c_coeffs, basis, dx):
        # Recompute h from coefficients
        h = np.zeros(n_points)
        for j in range(len(basis)):
            h += c_coeffs[j] * basis[j]
        return np.max(correlate(h, 1 - h, mode='full')) * dx  # We approximate the C5
    
    # Set up the problem using rough coefficients but optimize across optimized variables
    # This is a dual-reconstruct approach to minimize expensive full overlapping operations
    
    # In our Markdown model, attempt to define a convex dual form.
    # For now, stick to using a direct projection from desired solution space 
    # and bi-level trust-region optimization

    # Implement simpler chained optimization
    # Use basis expansion to represent h(x) in space
    # Apply lazy-repair and refinement approach
    
    best_c = coeffs.copy()
    best_score = objective_from_coeffs(coeffs, basis, dx)
    
    max_iterations = 500
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Various perturbations to current solution for exploration:
        # random changes to coefficients
        # uniform random perturbation of small size
        
        # Try small random move:
        eps = 0.01
        delta_c = np.random.randn(n_basis) * eps
        try_c = best_c + delta_c
        try_c = np.maximum(0, try_c)  # clip to non-negative
        try_c = try_c / np.sum(try_c) * (n_points / 2)  # renormalize to sum 1 (wrt dx)
        try_score = objective_from_coeffs(try_c, basis, dx)

            best_c = try_c

        # Occasionally re-sample pattern
        if iter_num % 20 == 0:
            # At regular intervals, also search locally from midpoint for entropy among pieces
            c_mid = (best_c + np.random.randn(n_basis) * 0.005) * 0.5

            c_mid = c_mid / np.sum(c_mid) * (n_points / 2)
            c_mid_score = objective_from_coeffs(c_mid, basis, dx)
            if c_mid_score < best_score:
                best_c = c_mid
                best_score = c_mid_score
    
    # Convert back to step function
    h_vals = reconstruct_h(best_c, basis, n_basis)
    # Clip and renormalize
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    # Final evaluation
    overlap_eval = np.max(correlate(h_vals, 1 - h_vals, mode='full')) * dx
    
    return (h_vals.tolist(), overlap_eval, n_points)
