import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth triangular shape
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    best_c5 = evaluate_c5(h, dx)
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlap
        current_c5 = evaluate_c5(h, dx)
        
        # Find the shift that gives maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(overlap)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Compute contribution to gradient for shift k_max
            if max_k > 0:
                gradient[i] += (1 - h[(i + max_k) % n_points])
                gradient[i] -= h[(i - max_k) % n_points]
            else:
                gradient[i] += (1 - h[(i - max_k) % n_points])
                gradient[i] -= h[(i + max_k) % n_points]
        
        # Normalize gradient
        gradient = gradient / np.sqrt(np.sum(gradient**2) + 1e-8)
        
        # Update step
        h_new = h - learning_rate * gradient
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept improvement or gradual update
        if new_c5 < current_c5:
            h = h_new
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            # Reduce learning rate if no improvement
            if patience_counter > 10:
                learning_rate *= 0.95
                patience_counter = 0
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (h.tolist(), best_c5, n_points)
