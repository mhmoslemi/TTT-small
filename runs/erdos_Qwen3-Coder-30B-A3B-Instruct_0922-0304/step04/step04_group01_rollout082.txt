import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a more sophisticated initial guess
        n_points = 70
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = np.zeros(n_points)
        # Sigmoid-like shape with concentration near center
        h = 1.0 / (1.0 + np.exp(-10 * (x - 1.0)))
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Variational approach with adaptive steps
    max_iter = 1000
    lr = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Adam optimizer variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_overlap = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = evaluate_c5(h_plus, dx)
            grad[i] = (overlap_plus - base_overlap) / eps
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally perturb the solution
        if iteration % 100 == 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
