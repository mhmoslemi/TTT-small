import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 50
    
    # Initialize with parent or fallback
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Simulated Annealing approach - more focused search
    temp_schedule = np.logspace(0, -3, max_iter)
    
    # Marching to optimization
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Small random perturbation
        h_new = best_h.copy()
        pert_idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.03)
        h_new[pert_idx] += delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject with SA probability
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp_schedule[i]):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
