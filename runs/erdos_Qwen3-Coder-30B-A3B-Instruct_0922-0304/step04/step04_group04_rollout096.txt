import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    else:
        # Fallback initialization: piecewise constant with some randomness
        h_current = np.ones(n_points)
        h_current[n_points//2:] = 0
        h_current += np.random.normal(0, 0.05, n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = np.max(correlate(h_current, 1 - h_current, mode="full")) * dx
    
    # Local search optimization
    max_iter = 5000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try small random perturbations
        h_trial = h_current.copy()
        # Select a few points to modify
        n_modify = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_modify, replace=False)
        for i in indices:
            # Small random change within bounds
            delta = np.random.normal(0, 0.02)
            h_trial[i] += delta
        # Keep within [0,1] and normalize
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Evaluate
        c5_trial = np.max(correlate(h_trial, 1 - h_trial, mode="full")) * dx
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h_current = h_trial.copy()
        else:
            # Occasionally accept worse solutions to escape local optima
            if np.random.rand() < 0.05:
                h_current = h_trial.copy()
    
    # Final validation and refinement
    c5_final = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), c5_final, n_points)
