import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Define basis functions: sine and cosine
    x = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size + 1))
    phi[:, 0] = 1.0  # constant term
    for i in range(1, basis_size + 1):
        if i % 2 == 1:
            # sine terms
            phi[:, i] = np.sin(np.pi * (i + 1) / 2 * x)
        else:
            # cosine terms
            phi[:, i] = np.cos(np.pi * i / 2 * x)
    
    # Compute integral of basis functions over [0,2]
    int_phi = np.zeros(basis_size + 1)
    for i in range(basis_size + 1):
        int_phi[i] = np.trapz(phi[:, i], x)
    
    # Build K matrix: precompute integrals of products
    K = np.zeros((basis_size + 1, basis_size + 1, 2 * n_points - 1))
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            for lag in range(2 * n_points - 1):
                idx = lag - n_points + 1
                if idx < 0:
                    # Correlation: overlap with shifted version
                    shifted = np.concatenate([phi[:idx, j], np.zeros(-idx)])
                else:
                    shifted = np.concatenate([np.zeros(idx), phi[idx:, j]])
                K[i, j, lag] = np.trapz(phi[:, i] * shifted, x)
    
    # Get initial guess from parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        c_guess = np.linalg.lstsq(phi, h_init, rcond=None)[0]
        c_guess = np.maximum(0, c_guess)  # enforce non-negative for now
        # Adjust to satisfy constraint
        c_guess = c_guess * (2.0 / np.sum(c_guess))  # Approximate at least
    else:
        # Default initialization
        c_guess = np.ones(basis_size + 1) * 0.5
        c_guess[0] = 1.0  # set mean to 1
    
    # Solve using convex optimization
    c = Variable(basis_size + 1)
    
    # Build McCormick relaxation
    z = []
    z_vars = []
    for i in range(basis_size + 1):
        z_row = []
        z_row_vars = []
        for j in range(basis_size + 1):
            z_ij = Variable()
            z_row.append(z_ij)
            z_row_vars.append(z_ij)
        z.append(z_row)
        z_vars.append(z_row_vars)
    
    # Convex constraints
    constraints = []
    constraints.append(c >= 0)
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
           constraints.append(z[i][j] >= c[i] + c[j] - 1)
            constraints.append(z[i][j] <= c[i])
            constraints.append(z[i][j] <= c[j])
    
    # Integral constraint: ∫h(x)dx = 1
    constraints.append(sum_entries(c * int_phi) == 1)
    
    # Objective: minimize C5
    C5_vars = []
    for lag in range(2 * n_points - 1):
        temp = 0
        for i in range(basis_size + 1):
            for j in range(basis_size + 1):
                temp += z[i][j] * K[i, j, lag]
        C5_vars.append(temp)
    
    C5 = Variable()
    constraints.append(C5 >= -C5_vars)
    constraints.append(C5 >= C5_vars)
    
    # Solve
    objective = Minimize(C5)
    prob = Problem(objective, constraints)
    
    # Solver config
    prob.solve(solver=SCS, verbose=False, max_iters=10000)
    
    if prob.status != "optimal":
        # fallback to initial guess and gradient descent
        h_final = np.clip(phi @ c_guess, 0, 1) 
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        return (h_final.tolist(), evaluate_c5(h_final, dx), n_points)
        
    # Extract solution
    c_values = c.value
    h_final = np.clip(phi @ c_values, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Validate final h
    # Ensure it satisfies constraints
    h_final = np.clip(h_final, 0, 1)
    if np.abs(np.sum(h_final) * dx - 1.0) > 1e-3:
        h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), evaluate_c5(h_final, dx), n_points)

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    # Find maximum value and scale properly
    max_overlap = np.max(overlap) * dx
    return max_overlap
