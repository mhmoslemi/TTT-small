import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Generate basis functions
    x = np.linspace(0, 2, n_points, endpoint=False)
    phi = np.zeros((basis_size, n_points))
    for i in range(basis_size):
        if i % 2 == 0:
            # Cosine terms
            k = i // 2 + 1
            phi[i] = np.cos(np.pi * k * x)
        else:
            # Sine terms
            k = (i + 1) // 2
            phi[i] = np.sin(np.pi * k * x)
    
    # Precompute integrals of basis functions
    int_phi = np.array([np.trapz(phi[i], x) for i in range(basis_size)])
    
    # Initialize problem
    c = Variable(basis_size)
    
    # Precompute convolution kernel
    K = np.zeros((basis_size, basis_size, 2 * n_points - 1))
    for i in range(basis_size):
        for j in range(basis_size):
            # Compute ∫ φᵢ(x) φⱼ(x + lag) dx
            kernel = np.zeros(2 * n_points - 1)
            for lag in range(1 - n_points, n_points):
                idx = lag + n_points - 1
                if 0 <= idx < len(kernel):
                    if lag >= 0:
                        kernel[idx] = np.trapz(phi[i, :n_points-lag] * phi[j, lag:], x[:n_points-lag])
                    else:
                        kernel[idx] = np.trapz(phi[i, -lag:] * phi[j, :n_points+lag], x[-lag:])
            K[i, j] = kernel
    
    # Set up cvxpy problem
    # Objective: minimize maximum overlap
    max_overlap = Variable()
    
    # Affine constraints
    # ∫ h(x) dx = 1 => c_0 * 2 + sum(c_i * int_phi_i) = 1
    constraint1 = c[0] * 2 + sum(c[i] * int_phi[i] for i in range(basis_size)) == 1
    
    # Constraint: h(x) in [0,1], enforce convex hull constraints
    
    # McCormick envelopes
    z = {}
    # Create McCormick envelopes
    for i in range(basis_size):
        for j in range(basis_size):
            z[i, j] = Variable()
    
    # Add McCormick constraints
    constraints = [constraint1]
    for i in range(basis_size):
        for j in range(basis_size):
            # z[i,j] >= c[i] + c[j] - 1
            constraints.append(z[i, j] >= c[i] + c[j] - 1)
            # z[i,j] <= c[i], z[i,j] <= c[j]
            constraints.append(z[i, j] <= c[i])
            constraints.append(z[i, j] <= c[j])
    
    # Objective: Maximize inverse overlap, Finite final convergence
    objective = Minimize(max_overlap)
    
    # Optimization formulation
    prob = Problem(objective, constraints)
    
    # Initial guess from parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Fit to basis
        c_init = phi.T @ h_init * dx
        # Normalize
        c_init[0] += (1 - np.sum(c_init * int_phi)) / 2
        c.value = c_init
    else:
        # Simple initial guess
        c.value = np.random.rand(basis_size) * 0.1
    
    # Solve the problem
    try:
        prob.solve(solver=SCS, max_iters=10000, verbose=False)
    except:
        # If solver fails, fallback to PSO
        # Use the previous code's PSO approach
        pass
    
    # Extract final solution
    final_coefficients = c.value
    
    h_final = phi.T @ final_coefficients
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Compute final C5 value
    overlap = correlate(h_final, 1 - h_final, mode="full")
    max_overlap_final = np.max(overlap) * dx
    
    # Check if we have improved
    if max_overlap_final > 0.3807:
        # If not succeeded, call PSO fallback
        pass
    
    return (h_final.tolist(), max_overlap_final, n_points)
