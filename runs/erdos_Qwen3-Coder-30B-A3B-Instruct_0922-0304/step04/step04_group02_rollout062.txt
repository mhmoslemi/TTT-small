import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ES parameters
    pop_size = 100
    num_generations = 1000
    elite_rate = 0.2
    mutation_rate = 0.3
    initial_sigma = 0.1
    sigma = initial_sigma
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(pop_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness.append(evaluate_c5(individual, dx))
    
    # ES iterations
    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_count = int(pop_size * elite_rate)
        elite_indices = sorted_indices[:elite_count]
        
        # Update best solution
        if fitness[sorted_indices[0]] < best_c5:
            best_c5 = fitness[sorted_indices[0]]
            best_h = population[sorted_indices[0]].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elitism
        for i in elite_indices:
            new_population.append(population[i].copy())
            new_fitness.append(fitness[i])
        
        # Generate offspring
        for _ in range(pop_size - elite_count):
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            winner_idx = tournament_indices[np.argmin([fitness[i] for i in tournament_indices])]
            
            # Mutate
            offspring = population[winner_idx].copy()
            mutation_mask = np.random.random(n_points) < mutation_rate
            noise = np.random.normal(0, sigma, n_points)
            offspring[mutation_mask] += noise[mutation_mask]
            
            # Repair
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * (n_points / 2) / np.sum(offspring)
            
            new_population.append(offspring)
            new_fitness.append(evaluate_c5(offspring, dx))
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Adaptive sigma
        if gen % 10 == 0:
            improvement = np.mean(fitness[:elite_count]) - np.mean(fitness[elite_count:])
            if improvement > 0:
                sigma = max(0.001, sigma * 0.9)
            else:
                sigma = min(0.5, sigma * 1.1)
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
