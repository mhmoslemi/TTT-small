import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 75
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: triangular window
        h = np.array([min(i, n_points-1-i) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian optimization
    lambda_sum = 0.0
    mu = 1.0
    nu = 1.0
    
    max_iter = 10000
    step_size = 0.002
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and gradient
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Gradient computation
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update h using gradient descent with adaptive step
        h_new = h - step_size * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if new solution is better
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Decide whether to accept new solution
        h = h_new
        
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
