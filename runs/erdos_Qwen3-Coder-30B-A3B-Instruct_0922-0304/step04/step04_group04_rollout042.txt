import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis setup
    def haar_wavelet_basis(n):
        # Generate Haar wavelets up to level 5 for n_points
        basis = []
        # Add scaling function for DC component
        basis.append(np.ones(n))
        
        # Generate wavelets
        j = 0
        while 2**j < n:
            for k in range(2**j):
                psi = np.zeros(n)
                left = k * 2**(1-j)
                right = (k+1) * 2**(1-j)
                if right <= n:
                    psi[left:right] = 1
                    psi[left:right//2] = 1
                    psi[right//2:right] = -1
                    basis.append(psi)
            j += 1
            if j > 5: break
        
        # Trim to exactly n_points
        if len(basis) > n_points:
            basis = basis[:n_points]
        else:
            while len(basis) < n_points:
                basis.append(np.zeros(n_points))
        
        return np.array(basis)

    # Load initial solution or use basic construction
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
    else:
        # Fallback: simple initial construction
        h_parent = np.zeros(n_points)
        h_parent[:n_points//2] = 1
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
    
    # Begin optimization with a variant of the excellent previous construction
    # but constrained to piecewise constant support.
    
    # Create a combinatorial approach to refine the step function
    def create_initial():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            h = h * (n_points / 2) / np.sum(h)
        return h
    
    best_h = create_initial()
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_c5(best_h, dx)
    
    # Generate wavelet basis for approximation
    basis = haar_wavelet_basis(n_points)
    n_basis = basis.shape[0]
    
    # Convex optimization with CVXPY
    C = Variable(n_basis)
    h = basis.T @ C
    h = h / np.sum(h) * (n_points / 2)  # Normalize appropriately
    
    # Convex relaxation of the overlap integral
    # For now, do gradient-based refinement of step functions
    # Using a more radical heuristic approach: 
    # optimize on a discrete change problem with local refinement
    
    # Function to sample step positions
    def sample_steps(n_max=6):
        # Generate all possible step function combinations
        choices = []
        for n_steps in range(1, min(n_max, n_points//2+1)):
            for pos in range(1, n_points):  # step at position
                choices.append(np.array([pos] + [0]*(n_steps-1)))
        return choices
    
    # Try different step size allocations
    def sample_step_functions(n_points=n_points, n_steps=3):
        pad = n_points // (n_steps+1)  # each step region has padding
        steps = np.linspace(pad, n_points-pad, n_steps, dtype=int)
        if len(steps) == 0:
            steps = [n_points//2]
        h = np.zeros(n_points)
        # Set values based on global and local density
        # First replicate the performance of PSO in parts of a modern approach
        for i in range(len(steps)):
            start = steps[i] if i == 0 else steps[i-1]
            end = steps[i+1] if i+1 < len(steps) else n_points
            h[start:end] = 1.0
        
        return np.array(h)
    
    # Try multiple randomized structures
    best_score = evaluate_c5(best_h, dx)
    best_sol = best_h.copy()
    
    # Monte Carlo refinement
    iter_count = 0
    max_iter = 500
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random variation in step locations
        h_new = np.zeros(n_points)
        n_steps = np.random.randint(2, 10)  # between 2 and 10 steps
        capacities = np.random.rand(n_steps)
        capacities = capacities / np.sum(capacities) * (n_points / 2)
        
        pos = 0
        for i in range(n_steps):
            span = max(1, int(capacities[i] / (n_points / 2) * n_points))
            end = min(pos + span, n_points)
            h_new[pos:end] = 1
            pos = end
            
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        score = evaluate_c5(h_new, dx)
        
        if score < best_score:
            best_score = score
            best_sol = h_new.copy()
            
        iter_count += 1
        
        # Early stopping if target achieved
        if best_score <= 0.3807:
            break
    
    # Refinement with local search
    niters = 200
    for i in range(niters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Small perturbations to best solution
        h_trial = best_sol.copy()
        idx = np.random.randint(0, n_points)
        # Perturb the value
        h_trial[idx] = max(0, min(1, h_trial[idx] + np.random.normal(0, 0.05)))
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        score = evaluate_c5(h_trial, dx)
        
        if score < best_score:
            best_score = score
            best_sol = h_trial.copy()
            
        if best_score <= 0.3807:
            break
    
    return (best_sol.tolist(), best_score, n_points)
