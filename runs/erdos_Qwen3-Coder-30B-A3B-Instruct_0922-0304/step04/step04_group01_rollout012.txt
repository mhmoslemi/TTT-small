import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a symmetric sawtooth pattern
        h = np.ones(n_points)
        midpoint = n_points // 2
        h[:midpoint] = 1.0
        h[midpoint:] = 0.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop with adaptive gradient descent
    max_iter = 20000
    step_size = 0.003
    patience = 0
    max_patience = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size and patience logic
        if iteration > 0 and iteration % 200 == 0:
            if patience > 0:
                step_size *= 0.9
            patience = 0
        
        # Gradient computation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        base_overlap = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_overlap = evaluate_c5(h_plus, dx)
            grad[i] = (plus_overlap - base_overlap) / eps
        
        # Update with gradient descent
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally random perturbations for escape
        if iteration % 100 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Check for stagnation
        if iteration > 0 and iteration % 10 == 0:
            current_c5 = evaluate_c5(h, dx)
            if abs(current_c5 - best_c5) < 1e-8:
                patience += 1
                if patience >= max_patience:
                    # Restart with a new random configuration
                    h = np.ones(n_points)
                    midpoint = n_points // 2
                    h[:midpoint] = 1.0
                    h[midpoint:] = 0.0
                    h += np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    patience = 0
            else:
                patience = 0

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
