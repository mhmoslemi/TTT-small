import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        # place intervals at [0, w] and [w+d, w+d+w]
        # We'll sweep d from 0 to 1
        d_candidates = np.arange(0.0, 1.01, 0.01)
        best_c5 = float('inf')
        best_h = None
        
        for d in d_candidates:
            h_test = np.zeros(n_points)
            # First interval [0, w]
            i0 = 0
            i1 = int(np.floor(w / dx))
            h_test[i0:i1+1] = 1
            
            # Second interval [w+d, w+d+w]
            i2 = int(np.ceil((w + d) / dx))
            i3 = int(np.floor((w + d + w) / dx))
            if i2 <= i3:
                h_test[i2:i3+1] = 1
            
            # Normalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            c5 = evaluate_c5(h_test, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_test.copy()
        
        h = best_h
    
    # Fine search around best d
    if time.time() - start_time < budget_s - 1:
        # Find current best d
        w = 0.5
        # Now we search with 0.005 steps 
        d_candidates = np.arange(0.0, 1.01, 0.005)
        best_c5 = float('inf')
        best_h = h
        
        for d in d_candidates:
            h_test = np.zeros(n_points)
            # First interval [0, w]
            i0 = 0
            i1 = int(np.floor(w / dx))
            h_test[i0:i1+1] = 1
            
            # Second interval [w+d, w+d+w]
            i2 = int(np.ceil((w + d) / dx))
            i3 = int(np.floor((w + d + w) / dx))
            if i2 <= i3:
                h_test[i2:i3+1] = 1
            
            # Normalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            c5 = evaluate_c5(h_test, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_test.copy()
        
        h = best_h
        
    return (h.tolist(), evaluate_c5(h, dx), n_points)
