import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using sequential convex programming approach
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Linearized approach: solve a sequence of linear programs
        try:
            # Construct constraints
            h_current = h.copy()
            # Get the maximum overlap index
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Create variables
            h_vars = Variable(n_points)
            s = Variable()
            
            # Add constraints
            constraints = [s >= 0]
            # Add integral constraint
            constraints.append(sum(h_vars) == n_points / 2)
            # Add bounds
            for i in range(n_points):
                constraints.append(h_vars[i] >= 0)
                constraints.append(h_vars[i] <= 1)
                
            # Add overlap constraints
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    terms = []
                    for i in range(n_points):
                        j = (i + k) % n_points
                        terms.append(h_current[i] * (1 - h_current[j]) + 
                                   (h_vars[i] - h_current[i]) - 
                                   (h_current[j] * (h_vars[i] - h_current[i])))
                    overlap_sum = sum(terms) * dx
                else:
                    terms = []
                    for i in range(n_points):
                        j = (i + k) % n_points
                        terms.append(h_current[i] * (1 - h_current[j]) + 
                                   (h_vars[i] - h_current[i]) - 
                                   (h_current[j] * (h_vars[i] - h_current[i])))
                    overlap_sum = sum(terms) * dx
                constraints.append(s >= overlap_sum)
                
            # Formulate and solve the LP
            objective = Minimize(s)
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array([float(h_vars[i].value) for i in range(n_points)])
                h = h_new
                # Ensure integral constraint
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to gradient-based method
                step = 0.001
                h_temp = h.copy()
                for _ in range(20):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
                
        except Exception:
            # Fallback to gradient descent approach
            step = 0.001
            h_temp = h.copy()
            for _ in range(20):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 100 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
