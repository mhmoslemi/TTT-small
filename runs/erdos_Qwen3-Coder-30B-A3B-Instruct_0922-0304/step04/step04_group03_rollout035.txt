import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    lr = 0.01
    
    # Initialize from parent solution or create a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent optimization
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Find shift with maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # h[i] contributes to overlap with (1-h[i+k_max]) for positive shifts
            # and h[i-k_max] for negative shifts
            if k_max > 0:
                # for positive k_max, h[i] pairs with 1-h[i+k_max]
                if i + k_max < n_points:
                    gradient[i] += (1 - h[i + k_max]) * dx
                else:
                    # wrap around
                    gradient[i] += (1 - h[i + k_max - n_points]) * dx
                # Also from contribution of h[i+k_max] to overlap of h[i]
                if i - k_max >= 0:
                    gradient[i] -= h[i - k_max] * dx
                else:
                    # wrap around
                    gradient[i] -= h[i - k_max + n_points] * dx
            elif k_max < 0:
                # for negative k_max, h[i] pairs with 1-h[i+k_max] 
                # where k_max is negative so i+k_max is smaller
                if i + k_max >= 0:
                    gradient[i] += (1 - h[i + k_max]) * dx
                else:
                    # wrap around
                    gradient[i] += (1 - h[i + k_max + n_points]) * dx
                # Also from contribution of h[i-k_max] to overlap of h[i]
                if i - k_max < n_points:
                    gradient[i] -= h[i - k_max] * dx
                else:
                    # wrap around
                    gradient[i] -= h[i - k_max - n_points] * dx
            else:
                # k_max == 0
                gradient[i] = -h[i] * dx  # derivative of h[i]^2
                    
        # Apply gradient descent
        h += -lr * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adapt learning rate
        if iter_num % 100 == 0 and iter_num > 0:
            lr *= 0.9
        
        if current_c5 <= 0.3807:
            break
            
    final_c5 = evaluate_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
