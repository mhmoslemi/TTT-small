import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Use fft for efficient correlation computation
        H = np.fft.fft(h)
        H_conj = np.conj(H)
        conv = np.fft.ifft(H * H_conj).real
        # Take the maximum of the full convolution
        max_conv = np.max(conv)
        return max_conv * dx

    def compute_c5_slow(h):
        # The original approach for comparison
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def perturb(h, temp, step_size):
        # Copy and perturb one element
        h_new = h.copy()
        idx = random.randint(0, n_points - 1)
        delta = random.uniform(-step_size, step_size)
        h_new[idx] += delta
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to sum to n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        return h_new

    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    step_size = 0.05
    
    # Monitoring variables
    best_c5 = float('inf')
    best_h = h.copy()
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Secondary monitoring variable
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main optimization loop
    iteration = 0
    temperature = T0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Create neighbor
        new_h = perturb(current_h, temperature, step_size)
        new_c5 = compute_c5(new_h)
        
        # Accept with simulated annealing probability
        if new_c5 < current_c5 or random.random() < np.exp(-(new_c5 - current_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Exit early if target is met
        if best_c5 <= 0.3807:
            break
            
        # Annealing schedule
        temperature *= alpha
        step_size *= alpha
        
        # Break if we're stuck
        if no_improvement_count >= max_no_improvement:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
