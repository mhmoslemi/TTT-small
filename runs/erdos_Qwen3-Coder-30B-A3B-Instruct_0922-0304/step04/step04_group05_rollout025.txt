import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.1
    mu = 1e-3
    damping = 0.5
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential Convex Programming
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        h_tilde = h * dx
        corr_h = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr_h)
        
        # Gradient and Hessian approximation
        grad = np.zeros(n_points)
        hess = np.zeros((n_points, n_points))
        
        # Analytical gradients and Hessian for the max overlap term
        # We approximate that each element contributes to the max overlap
        for i in range(n_points):
            for j in range(n_points):
                if i + j <= 2 * (n_points - 1):
                    k = i + j
                    if i < n_points and j < n_points:
                        coeff = 1
                        if k >= n_points:
                            k = 2 * n_points - 2 - k
                        grad[i] += coeff * (1 - 2 * h[j])
                        hess[i, i] += 2 * coeff
        
        # Avoid singular matrix
        hess.flat[::n_points + 1] += 1e-6
        
        # Assemble convex subproblem
        delta = cp.Variable(n_points)
        objective = cp.Minimize(cp.matmul(grad.T, delta) + 0.5 * cp.quad_form(delta, hess) )
        constraints = [
            delta >= -h,
            delta <= 1 - h,
            cp.sum(delta) == 0
        ]
        
        # Solve with CVXOPT
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                delta_val = np.array(delta.value).flatten()
                h_new = h + alpha * delta_val
            else:
                h_new = h
        except:
            h_new = h
            
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Line search damping
        if iter_num > 0 and iter_num % 10 == 0:
            alpha *= damping
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
