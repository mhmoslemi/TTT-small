import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    w = 0.5  # width of each interval
    a = 0    # starting position of first interval
    
    best_h = None
    best_c5 = float('inf')
    
    # Grid search over possible distances between intervals
    d_candidates = np.arange(0, 1.001, 0.01)
    
    # To optimize, also try finer grid around promising regions
    # but we'll do this at the end if time permits
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute indices for intervals
        # First interval: [a, a+w]
        i0 = 0
        i1 = int(np.floor((a + w) / dx))
        
        # Second interval: [a+w+d, a+w+d+w]
        i2 = int(np.ceil((a + w + d) / dx))
        i3 = int(np.floor((a + w + d + w) / dx))
        
        # If second interval goes out of bounds, break
        if i2 >= n_points:
            continue
            
        # Construct h
        h = np.zeros(n_points)
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # If we haven't found a good enough solution yet, perform a finer search
    if best_c5 > 0.3808 and time.time() - start_time < budget_s - 1:
        # Refine around the best d value found
        # Find the best d in the coarse search
        d_candidates = np.arange(0, 1.001, 0.001)
        
        d_best_coarse = None
        min_c5_coarse = float('inf')
        
        for d in d_candidates:
            if time.time() - start_time > budget_s - 1:
                break
                
            i0 = 0
            i1 = int(np.floor((a + w) / dx))
            i2 = int(np.ceil((a + w + d) / dx))
            i3 = int(np.floor((a + w + d + w) / dx))
            
            if i2 >= n_points:
                continue
                
            h = np.zeros(n_points)
            h[i0:i1+1] = 1
            h[i2:i3+1] = 1
            
            h = h * (n_points / 2) / np.sum(h)
            
            c5 = evaluate_c5(h, dx)
            
            if c5 < min_c5_coarse:
                min_c5_coarse = c5
                d_best_coarse = d
                
        # Now refine around this d_best_coarse
        if time.time() - start_time < budget_s - 1:
            d_range = 0.01
            if d_best_coarse is not None:
                d_candidates_fine = np.arange(
                    max(0, d_best_coarse - d_range),
                    min(1, d_best_coarse + d_range + 0.001),
                    0.0001
                )
                
                for d in d_candidates_fine:
                    if time.time() - start_time > budget_s - 1:
                        break
                        
                    i0 = 0
                    i1 = int(np.floor((a + w) / dx))
                    i2 = int(np.ceil((a + w + d) / dx))
                    i3 = int(np.floor((a + w + d + w) / dx))
                    
                    if i2 >= n_points:
                        continue
                        
                    h = np.zeros(n_points)
                    h[i0:i1+1] = 1
                    h[i2:i3+1] = 1
                    
                    h = h * (n_points / 2) / np.sum(h)
                    
                    c5 = evaluate_c5(h, dx)
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    # Even if we didn't find a better solution, return the best we have
    if best_h is None:
        # Fallback to a symmetric triangular approximation or something simpler
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = evaluate_c5(h, dx)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
