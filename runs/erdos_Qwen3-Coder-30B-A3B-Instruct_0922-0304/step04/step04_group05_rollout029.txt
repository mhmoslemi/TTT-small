import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # SCP parameters
    max_iter = 1000
    alpha = 0.1
    mu = 1e-3
    damping_factor = 0.5
    tolerance = 1e-6
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate and solve convex subproblem
        # Compute gradient and Hessian approximations
        h_current = h.copy()
        
        # Simple gradient approximation: finite differences
        eps = 1e-5
        grad_h = np.zeros(n_points)
        
        # Evaluate current overlap at multiple shifts
        current_overlap = np.zeros(n_points)
        for i in range(n_points):
            overlap_i = 0
            for j in range(n_points):
                if i + j < n_points:
                    overlap_i += h_current[j] * (1 - h_current[i + j])
                else:
                    overlap_i += h_current[j] * (1 - h_current[i + j - n_points])
            current_overlap[i] = overlap_i * dx

        # Simplified gradient calculation - in practice use more precise methods
        # For now, approximate gradient by perturbing individual entries
        for k in range(n_points):
            h_plus = h_current.copy()
            h_plus[k] += eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = evaluate_c5(h_plus, dx)
            grad_h[k] = (overlap_plus - best_c5) / eps
        
        # Approximate Hessian with identity matrix for simplicity
        hessian = np.eye(n_points) * 1e-6
        # Add regularization to prevent ill-conditioning
        hessian += np.eye(n_points) * 1e-4
        
        # Solve convex subproblem with CVXOPT
        delta = cp.Variable(n_points)
        
        # Objective: linearize around current point
        objective = cp.Minimize(grad_h.T @ delta + 
                               0.5 * cp.quad_form(delta, hessian))
        
        # Constraints
        constraints = [
            delta >= -h_current,
            delta <= 1 - h_current,
            cp.sum(delta) == 0  # Preserve integral constraint
        ]
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                delta_val = delta.value
                # Update current solution
                h_new = h_current + alpha * delta_val
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate objective
                new_c5 = evaluate_c5(h_new, dx)
                
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    alpha *= damping_factor
                    
                if best_c5 <= 0.3807:
                    break
                    
        except:
            # If optimization fails, reduce step size and try again
            alpha *= damping_factor
            
    return (best_h.tolist(), best_c5, n_points)
