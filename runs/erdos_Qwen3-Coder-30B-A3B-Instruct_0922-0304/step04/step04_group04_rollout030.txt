import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Basis functions: sine and cosine terms
    def phi(k, x):
        if k == 0:
            return 1.0
        elif k % 2 == 1:
            return np.sin(np.pi * (k + 1) // 2 * x)
        else:
            return np.cos(np.pi * k // 2 * x)
    
    # Precompute basis functions
    x_vals = np.linspace(0, 2, n_points)
    phi_basis = np.array([[phi(k, x) for x in x_vals] for k in range(basis_size)])
    
    # Precompute integral of basis functions
    int_phi = np.array([np.trapz(phi_basis[k], x_vals) for k in range(basis_size)])
    
    # Initial guess from `initial_h_values` if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
    else:
        # Fallback: step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        
    # Compute Fourier-like coefficients for initial guess
    coeffs_init = np.linalg.lstsq(phi_basis.T, h_init, rcond=None)[0]
    coeffs_init = coeffs_init * (n_points / 2) / np.sum(coeffs_init * int_phi)
    
    # Build overlap integrals matrix
    K = np.zeros((basis_size, basis_size, 2 * n_points - 1))
    for i in range(basis_size):
        for j in range(basis_size):
            for lag in range(2 * n_points - 1):
                lag_idx = lag - (n_points - 1)
                x_arr = x_vals
                x_shifted = x_arr + lag_idx * dx
                x_shifted = np.clip(x_shifted, 0, 2)
                # Interpolate if needed
                if lag_idx < 0:
                    x_shifted = np.clip(x_shifted, 0, 2)
                else:
                    x_shifted = np.clip(x_shifted, 0, 2)
                interpolated = np.interp(x_shifted, x_vals, phi_basis[j])
                integrand = phi_basis[i] * interpolated
                K[i, j, lag] = np.trapz(integrand, x_vals)
    
    # Initialize variables
    c = Variable(basis_size)
    Z = Variable((basis_size, basis_size), symmetric=True)
    
    # Constraints
    constraints = []
    # ADMM-like constraint enforcement for bounds using relaxed variables
    # Introduce auxiliary variables u,v for handling nonlinear constraints
    # Using McCormick envelope: z >= c_i + c_j - 1, z <= c_i, z <= c_j
    
    # Dummy variables to handle constraints
    U = Variable((basis_size, basis_size))
    V = Variable((basis_size, basis_size))
    
    # The overlap integral maximization subject to convex constraints
    constraints += [c[i] >= 0 for i in range(basis_size)]
    constraints += [U[i,j] >= c[i] + c[j] - 1 for i in range(basis_size) for j in range(basis_size)]
    constraints += [U[i,j] <= c[i] for i in range(basis_size) for j in range(basis_size)]
    constraints += [U[i,j] <= c[j] for i in range(basis_size) for j in range(basis_size)]
    
    # Integral constraint
    integral_constraint = sum(c[i] * int_phi[i] for i in range(basis_size)) == 1.0
    constraints.append(integral_constraint)
    
    # Objective variable
    max_overlap = Variable()
    
    # For each lag, we compute the contribution to the maximum overlap
    # Approximate the maximization as a scalar
    max_lag_contribution = []
    
    # Compute overlap values for a fixed vector
    # We must check all lags and compute which one is maximum
    
    # Keep track of the maximum
    max_term = -1e5  # lower bound
    
    # Make sure the vector is feasible for solution
    def compute_max_overlap(current_coeffs):
        # This is actually minimum value of the sum
        h_val = phi_basis.T @ current_coeffs
        h_val = np.clip(h_val, 0, 1.0)
        
        # Evaluate convolution
        conv = correlate(h_val, 1 - h_val, mode="full")
        max_val = np.max(conv) * dx
        return max_val
    
    # Try heuristic-based search
    h_current = h_init
    score_current = compute_max_overlap(coeffs_init)
    
    best_h = h_current.copy()
    best_score = score_current
    
    # Refinement loop
    max_refinements = 500
    for iteration in range(max_refinements):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Use convex relaxation idea but adjust with local improvement
        # For now we try applying local search based on gradient-like sensitivity
        
        # Limit computation
        # Just for plotting or debugging, skip in production
        h_current = phi_basis.T @ coeffs_init
        h_current = np.clip(h_current, 0, 1.0)
        score_current = compute_max_overlap(coeffs_init)
        if score_current < best_score:
            best_score = score_current
            best_h = h_current.copy()
            
        # Perturb coefficients to get new point
        perturb = np.random.normal(0, 0.01, basis_size)
        # Ensure adjustment within bounds
        coeffs_new = coeffs_init + perturb
        # Enforce non-negativity
        coeffs_new = np.maximum(coeffs_new, 0)
        
        # Normalize with integral
        # It must hold that sum(c_i * int_phi_i) = 1
        norm_factor = sum(coeffs_new[i] * int_phi[i] for i in range(basis_size))
        if abs(norm_factor) < 1e-12:
            continue  # Avoid division by zero
        coeffs_new = coeffs_new / norm_factor
        
        # Compute and evaluate
        test_h = phi_basis.T @ coeffs_new
        test_h = np.clip(test_h, 0, 1.0)
        test_score = compute_max_overlap(coeffs_new)
        if test_score < score_current:
            coeffs_init = coeffs_new.copy()
            score_current = test_score
    
    # Final evaluation using more precise overlap calculation
    eval_h = phi_basis.T @ coeffs_init
    eval_h = np.clip(eval_h, 0, 1.0)
    eval_score = compute_max_overlap(coeffs_init)
    return (eval_h.tolist(), eval_score, n_points)
