import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    min_temp = 1e-8
    max_iter = 10000
    step_size = 0.05

    best_c5 = float('inf')
    best_h = h.copy()
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and T > min_temp:
        if iteration >= max_iter:
            break
            
        # Generate neighbor by perturbing one random element
        idx = np.random.randint(0, n_points)
        old_h = current_h[idx]
        new_h = old_h + np.random.normal(0, step_size)
        current_h[idx] = np.clip(new_h, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_sum = np.sum(current_h)
        current_h = current_h * (n_points / 2) / h_sum
        
        # Compute new C5
        new_c5 = compute_c5(current_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < math.exp(-delta / T):
                current_c5 = new_c5
            else:
                # Revert to previous state
                current_h[idx] = old_h
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Cooling schedule
        T *= alpha
        step_size *= alpha
        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
