import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple piecewise constant function
        h = np.zeros(n_points)
        # Set first half to 1
        h[:n_points//2] = 1.0
        # Normalize to integrate to 1
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search for optimal d in piecewise constant construction
    d_candidates = np.arange(0.0, 1.0, 0.01)
    
    # For each d, create corresponding h and evaluate
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define h as two intervals of width 0.5
        # First interval [0, 0.5]
        i0 = 0
        i1 = int(0.5 / dx) - 1
        
        # Second interval [0.5 + d, 1.0 + d]
        i2 = int((0.5 + d) / dx)
        i3 = int((1.0 + d) / dx) - 1
        
        # Skip if intervals overlap
        if i2 <= i1:
            continue
            
        # Create h function
        h_test = np.zeros(n_points)
        h_test[i0:i1+1] = 1.0
        h_test[i2:i3+1] = 1.0
        
        # Normalize to satisfy integral constraint
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate C5
        c5_test = evaluate_c5(h_test, dx)
        
        # Update best if found better
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # If still no improvement found, try some random perturbations
    if best_c5 > 0.3807:
        np.random.seed(seed)
        max_perturb_iter = 1000
        for i in range(max_perturb_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Create small random perturbation
            pert_h = best_h + np.random.normal(0, 0.01, n_points)
            pert_h = np.clip(pert_h, 0, 1)
            # Renormalize
            pert_h = pert_h * (n_points / 2) / np.sum(pert_h)
            
            c5_test = evaluate_c5(pert_h, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = pert_h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
