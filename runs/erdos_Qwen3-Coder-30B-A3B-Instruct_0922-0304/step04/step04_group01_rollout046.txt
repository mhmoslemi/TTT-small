import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use sequential convex programming approach with LP relaxations
    max_iter = 2000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Build constraints for this iteration
        h_current = h.copy()
        
        # Create optimization variables
        h_vars = cp.Variable(n_points)
        s = cp.Variable()
        
        # Create constraints for all shifts k
        constraints = []
        
        # Integral constraint
        constraints.append(cp.sum(h_vars) * dx == 1)
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Linearized overlap constraints
        for k in range(-n_points+1, n_points):
            # Compute linear approximation around current h
            if k >= 0:
                left_term = h_current[:n_points-k]
                right_term = h_current[k:]
            else:
                left_term = h_current[-k:]
                right_term = h_current[:n_points+k]
            
            # Linearization: h_i * (1 - h_{i+k}) \approx h_i^old * (1 - h_{i+k}^old) + 
            # (h_i - h_i^old) - h_{i+k}^old * (h_i - h_i^old) 
            # But actually: h_i * (1 - h_{i+k}) = h_i - h_i * h_{i+k}
            # Linearization around h_current: h_i^old * (1 - h_{i+k}^old) + 
            # (1 - h_{i+k}^old) * (h_i - h_i^old) - h_i^old * (h_{i+k} - h_{i+k}^old)
            
            # Let's be more precise:
            # Approximate h_i * (1 - h_{i+k}) using h_current[i] (i+k) 
            # This is a bilinear constraint, linearize:
            # Let h_i = h_i_current + delta_i and h_{i+k} = h_{i+k}_current + delta_{i+k}
            # So h_i * (1 - h_{i+k}) ~ h_i_current * (1 - h_{i+k}_current) + 
            # (1-h_{i+k}_current)*delta_i - h_i_current*delta_{i+k}
            # Sum over all i with appropriate index shifting
            
            if k >= 0:
                term_1 = h_current[:n_points-k] 
                term_2 = h_current[k:]
                term_1_h = h_vars[:n_points-k]
                term_2_h = h_vars[k:]
                left_part = cp.sum(term_1 * (1 - term_2_h) * dx) + cp.sum((1 - term_2) * (term_1_h - term_1) * dx)
                offset = h_current[:n_points-k] * (1 - h_current[k:]) * dx
            else:
                k = -k
                term_1 = h_current[k:]
                term_2 = h_current[:n_points-k]
                term_1_h = h_vars[k:]
                term_2_h = h_vars[:n_points-k]
                left_part = cp.sum(term_1 * (1 - term_2_h) * dx) + cp.sum((1 - term_2) * (term_1_h - term_1) * dx)
                offset = h_current[k:] * (1 - h_current[:n_points-k]) * dx
                
            # Multiplying by dx and utilizing LHS to match with coefficient term
            # Effectively, we need to ensure: s >= sum(h_i*(1-h_{i+k})dx)
            # The linearization gives:
            # sum = sum_i (h_i_current * (1 - h_{i+k}_current) + (1 - h_{i+k}_current) * (h_i - h_i_current) - h_i_current * (h_{i+k} - h_{i+k}_current)) * dx
            # = sum_i (h_i_current * (1 - h_{i+k}_current) + (1 - h_{i+k}_current) * h_i - (1 - h_{i+k}_current) * h_i_current - h_i_current * h_{i+k} + h_i_current * h_{i+k}_current) * dx
            # = sum_i (h_i_current * (1 - h_{i+k}_current) - (1 - h_{i+k}_current) * h_i_current - h_i_current * h_{i+k} + h_i_current * h_{i+k}_current) dx
            # = sum_i h_i_current * (1 - h_{i+k}_current + h_{i+k}_current - h_{i+k}_current - h_{i+k} + h_{i+k}_current) dx
            # = sum_i h_i_current*(1 - h_{i+k}) dx
            
            # Because all terms except offset satisfy the linearization:
            # Let's compute: sum_i (h_i_current * (1 - h_{i+k}_current)) * dx  
            # and then  sum_i ( (1 - h_{i+k}_current) * (h_i - h_i_current) ) * dx 
            # and sum_i ( -h_i_current * (h_{i+k} - h_{i+k}_current) ) * dx
            
            linearized_sum = cp.sum(offset)
            offset_term = h_current - h_current
            if k >= 0:
                left_term = h_current[:n_points-k]
                right_term = h_current[k:]
                inner_sum = cp.sum(left_term * (1 - h_vars[k:]))
                inner_sum2 = cp.sum((1 - right_term) * (h_vars[:n_points-k] - h_current[:n_points-k]))
            else:
                k_neg = -k
                left_term = h_current[k_neg:]
                right_term = h_current[:n_points-k_neg]
                inner_sum = cp.sum(left_term * (1 - h_vars[:n_points-k_neg]))
                inner_sum2 = cp.sum((1 - right_term) * (h_vars[k_neg:] - h_current[k_neg:]))
                
            # Add linearized term
            linearized_sum = cp.sum(h_current * (1 - np.roll(h_current, k)) * dx) - cp.sum((1 - np.roll(h_current, k)) * (h_vars - h_current) * dx) + cp.sum((h_current * (1 - np.roll(h_current, k))) * dx) 
            # Simplified: 
            linearized_sum = cp.sum( (h_current * (1 - np.roll(h_current, k)) + (1 - np.roll(h_current, k)) * (h_vars - h_current) - h_current * (np.roll(h_vars, -k) - np.roll(h_current, -k))) * dx)
            
            # Let's simplify and create a generic form that minimizes complexity:
            # For each k, strictly write the constraint:
            # s >= sum_i h_i_current * (1 - h_{i+k}_current) * dx + sum_i (1 - h_{i+k}_current) * (h_i - h_i_current) * dx - sum_i h_i_current * (h_{i+k} - h_{i+k}_current) * dx
            # = sum_i h_i_current * (1 - h_{i+k}_current) * dx + sum_i (1-h_{i+k}_current) * h_i * dx - sum_i (1-h_{i+k}_current) * h_i_current * dx - sum_i h_i_current * h_{i+k} * dx + sum_i h_i_current * h_{i+k}_current * dx
            # = sum_i h_i_current * (1 - h_{i+k}_current) * dx + (1-h_{i+k}_current) * (sum h_i) * dx - (1-h_{i+k}_current) * (sum h_i_current) * dx 
            # - sum_i h_i_current * h_{i+k} * dx + sum_i h_i_current * h_{i+k}_current * dx
            
            # Instead, directly do this:
            # Because h_current sum is fixed:
            # s >= sum_i h_i (1 - h_{i+k}) dx 
            # Placeholder solver solution!
            # Easy: just try random sampling approach
            # So go with a simpler way of approximating the gradients for now
            
            # For small search, we can also try finite differences with normal gradient approach on just one step:
            # Do actual margin-based optimization on largest value
            
        # Try a hybrid approach with exact computation for small steps for better precision
        
        # Use earlier gradient approach for direct optimization:
        try:
            # Perform direct optimization with SLSQP (which works better)
            from scipy.optimize import minimize
            
            # Define the objective function as max of overlaps
            def objective(h):
                return evaluate_c5(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 50, 'ftol': 1e-12})
            
            if result.success:
                h = result.x
            else:
                # fallback to simpler gradient descent with delta
                step = 0.002
                iterations = 40
                h_temp = h.copy()
                for _ in range(iterations):
                    # Compute exact overlap
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    # Estimate gradient via finite differences
                    grad = np.zeros_like(h_temp)
                    eps = 1e-5
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback if anything fails: gradient descent version
            step = 0.003
            h_temp = h.copy()
            for _ in range(30):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to random noise
        if iteration % 100 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.03, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    final_c5 = evaluate_c5(best_h, dx) 
    return (best_h.tolist(), final_c5, n_points)
