import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.1
    damping = 0.5
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Sequential convex programming approach
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian approximations using finite differences
        # For computational efficiency, we compute the gradient by finite difference
        eps = 1e-6
        grad = np.zeros(n_points)
        h_current = h.copy()
        
        # Compute gradient components
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[i] -= eps
            # Ensure bounds
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Form and solve convex subproblem
        # Minimize: grad^T * delta + (1/2) * delta^T * H * delta
        # Subject to: 0 <= h + delta <= 1, sum(h + delta) = n_points/2
        
        delta = cp.Variable(n_points)
        
        # Construct quadratic objective with regularization
        # For simplicity, use identity matrix as Hessian approximation
        reg = 1e-8
        obj = cp.Minimize(cp.matmul(grad, delta) + 0.5 * cp.matmul(delta, delta) + reg * cp.sum_squares(delta))
        
        # Constraints
        constraints = [
            cp.sum(delta) == 0,  # Keeps sum constant
            h + delta >= 0,      # Lower bound
            h + delta <= 1       # Upper bound
        ]
        
        # Create and solve problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                delta_val = delta.value
                h_new = h + alpha * delta_val
                
                # Re-project to feasibility
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_c5(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
                h = h_new
                
                # Early termination
                if best_c5 <= 0.3807:
                    break
                    
            else:
                # If optimization fails, use a small step based on gradient
                h_new = h - alpha * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_c5(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                
                h = h_new
        except:
            # Fallback to simple gradient descent step
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            h = h_new
            
        # Reduce step size if improvement stalls
        if iteration > 0 and iteration % 10 == 0:
            if best_c5 >= 0.3807 * 1.02:
                alpha *= damping
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
