import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h: use initial_h_values if available, otherwise create a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # For gradient descent
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    last_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        # If we've hit target, return early
        if current_c5 <= 0.3807:
            break
            
        # Track improvement
        if current_c5 < last_c5:
            last_c5 = current_c5
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                break
        
        # Compute gradients using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step
        h = h - learning_rate * grad
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate if no progress
        if patience > 0 and patience % 10 == 0:
            learning_rate *= 0.9
        
    # Final evaluation
    final_c5 = evaluate_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
