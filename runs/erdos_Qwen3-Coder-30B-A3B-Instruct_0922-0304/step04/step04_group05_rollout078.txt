import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        h = np.clip(h, 0, 1)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    max_iter = 1000
    alpha = 0.1
    damping = 0.5
    mu = 1e-3

        # Practical gradient (approximate for convex minimization)
        # Compute linearized problem - design a convex subproblem

        # Linearize the objective around current h
        # Objective is to minimize max of correlations

        h_new = h - alpha * grad
