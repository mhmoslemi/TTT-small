import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Population size and ES parameters
    pop_size = 30
    elite_size = pop_size // 4
    mutation_sigma = 0.1
    max_iter = 2000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        # Create triangular function
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill the rest with random perturbations
    for _ in range(pop_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Evaluation of initial population
    fitness = [evaluate_c5(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # ES main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        
        # Create new population through mutation
        new_population = [population[i].copy() for i in elite_indices]
        
        # Generate offspring by mutating elite
        for _ in range(pop_size - elite_size):
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_sigma, n_points)
            offspring = parent + noise
            offspring = project_to_feasible(offspring, n_points)
            
            new_population.append(offspring)
        
        # Evaluate new population
        new_fitness = [evaluate_c5(h, dx) for h in new_population]
        
        # Update global best
        min_fitness_idx = np.argmin(new_fitness)
        if new_fitness[min_fitness_idx] < best_c5:
            best_h = new_population[min_fitness_idx].copy()
            best_c5 = new_fitness[min_fitness_idx]
        
        # Update population and fitness
        population = new_population
        fitness = new_fitness
        
        # Adapt mutation sigma based on success
        if iter_num % 10 == 0:
            # Adjust sigma based on proportion of improvements
            recent_fitness = new_fitness[-10:] if len(new_fitness) >= 10 else new_fitness
            if len(recent_fitness) > 0:
                improvement_ratio = sum(1 for f in recent_fitness if f < best_c5) / len(recent_fitness)
                if improvement_ratio > 0.3:
                    mutation_sigma *= 1.1
                elif improvement_ratio < 0.1:
                    mutation_sigma *= 0.9
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
