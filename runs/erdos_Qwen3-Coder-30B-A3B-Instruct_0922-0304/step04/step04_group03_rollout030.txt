import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraints(h, n_points, dx):
    # Ensure h is in [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a more sophisticated initial guess
        h = np.zeros(n_points)
        # Create a two-peak structure
        peak1 = n_points // 3
        peak2 = 2 * n_points // 3
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - peak1) / (n_points // 6))
            h[i] = max(0, h[i] - abs(i - peak2) / (n_points // 6))
        h = np.clip(h, 0, 1)
    
    h = project_to_constraints(h, n_points, dx)
    
    # Gradient-based optimization
    max_iter = 10000
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # Compute correlation for shift k
            shift_h = np.roll(h, k - n_points + 1)
            overlaps[k] = np.sum(h * (1 - shift_h)) * dx
        
        # Find the worst case shift
        k_max = np.argmax(overlaps)
        
        # Compute gradient for this shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Gradient of overlap k_max w.r.t. h[i]
            # This is: (1 - h[(i + k_max) % n_points]) - h[(i - k_max) % n_points]
            idx1 = (i + k_max) % n_points
            idx2 = (i - k_max) % n_points
            gradient[i] = (1 - h[idx1]) - h[idx2]
        
        # Update with momentum
        velocity = momentum * velocity - lr * gradient
        h = h + velocity
        
        # Project to constraints
        h = project_to_constraints(h, n_points, dx)
        
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Check for early exit if target is achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
