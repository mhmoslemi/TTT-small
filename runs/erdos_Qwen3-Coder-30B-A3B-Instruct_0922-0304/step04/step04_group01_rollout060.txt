import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a simple sawtooth-like construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add gentle noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using gradient-based approach with adaptive step size
    max_iter = 5000
    step_size = 0.003
    stagnation_count = 0
    min_step_size = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        base_overlap = evaluate_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = evaluate_c5(h_plus, dx)
            grad[i] = (overlap_plus - base_overlap) / eps
            
        # Update with gradient descent
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        stagnation_count += 1
        if stagnation_count > 50:
            step_size *= 0.9
            stagnation_count = 0
            if step_size < min_step_size:
                step_size = min_step_size
                
        # Occasionally add perturbation to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.005, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
