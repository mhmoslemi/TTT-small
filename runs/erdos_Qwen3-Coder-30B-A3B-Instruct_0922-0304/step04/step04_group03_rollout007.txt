import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_constraint(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 while keeping bounds"""
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    h = project_to_constraint(h, n_points)
    
    # Gradient-based optimization
    max_iter = 50000
    learning_rate = 0.001
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        c5 = evaluate_c5(h, dx)
        
        # Find worst case shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient with respect to max_k
        # dO_k/dh_i = (1 - h_{i+k}) - h_{i-k}
        gradient = np.zeros(n_points)
        for j in range(n_points):
            k = max_k
            # Handle negative indices
            idx_plus = (j + k) % n_points
            idx_minus = (j - k) % n_points
            if k >= 0:
                gradient[j] = (1 - h[idx_plus]) - h[idx_minus]
            else:
                gradient[j] = (1 - h[idx_plus]) - h[idx_minus]
        
        # Update velocity with momentum
        velocity = momentum * velocity - learning_rate * gradient
        
        # Update h
        h = h + velocity
        h = project_to_constraint(h, n_points)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive learning rate
        if i % 1000 == 0 and i > 0:
            learning_rate *= 0.95
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
