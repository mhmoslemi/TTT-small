import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Cutting plane method
    constraints_list = []
    eps = 0.001
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(current_overlap)
        max_overlap = current_overlap[max_k] * dx
        
        # Add constraint: sum(h[i] * (1 - h[i + k])) <= max_overlap - eps
        # This introduces a linear constraint
        constraint_expr = 0
        for i in range(n_points):
            j = (i + max_k - (n_points - 1)) % n_points
            constraint_expr += h[i] * (1 - h[j])
        
        # Add to constraints
        constraints_list.append(constraint_expr <= max_overlap - eps)
        
        # Create CVXPY variables and problem
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(h_vars))  # We will adjust this

        # Add constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add cutting constraints
        for constr in constraints_list[:-1]:  # All previous constraints
            constraints.append(constr)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6, abstol=1e-6)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(h_vars.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Evaluate new solution
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                
                # Reduce epsilon as we improve
                if current_c5 < 0.3807:
                    break
        except:
            # Fallback option
            pass
            
        iteration += 1
        
        # Keep solving until target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
