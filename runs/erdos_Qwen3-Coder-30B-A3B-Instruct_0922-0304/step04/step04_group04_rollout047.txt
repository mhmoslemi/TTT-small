import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis setup
    def haar_wavelet_basis(n_basis=30):
        # Precompute Haar basis functions
        basis = []
        x = np.linspace(0, 2, n_points)
        x = x.reshape(-1, 1)
        
        # Simple Haar basis construction (first few wavelets)
        for i in range(1, n_basis + 1):
            wavelet = np.zeros(n_points)
            # Compute Haar wavelet with different scales and positions
            level = 0
            while (1 << level) <= i:
                level += 1
            level -= 1
            pos = i - (1 << level)
            
            if level == 0:  # Simple wavelet
                wavelet[:n_points//2] = 1
                wavelet[n_points//2:] = -1
            else:
                width = 1 << (level - 1)
                start_idx = pos * width
                if start_idx + width <= n_points:
                    wavelet[start_idx:start_idx + width] = 1
                    if start_idx + width + width <= n_points:
                        wavelet[start_idx + width:start_idx + width + width] = -1
                else:
                    wavelet[start_idx:] = 1 if pos == 0 else -1
            
            basis.append(wavelet)
        
        return np.array(basis).T  # Shape (n_points, n_basis)

    def prepare_haar_basis():
        n_basis = min(30, n_points)
        basis = haar_wavelet_basis(n_basis)
        return basis

    # Try to initialize using previous solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = h_init.copy()
    else:
        # Use simpler construction or random
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Add some randomness to prevent stagnation early
        noise = np.random.normal(0, 0.02, n_points)
        h_init += noise
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Optimize in a simple heuristic loop with multiple random restarts
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Shortening to more practical iteration count
    max_iter = 3000
    n_starts = 8
    for start_iter in range(n_starts):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Pick random restart solution
        if start_iter == 0:
            h = h_init.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Simple gradient-step type search
        tf = 0.1
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Simple optimization by moving points up/down very slightly
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            h_new[idx] += delta
            h_new = np.clip(h_new, 0, 1)
            # Renormalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            if c5_new < best_c5:
                h = h_new.copy()
            else:
                h = h.copy()  # Keep current
            
            # Early stopping
            if best_c5 <= 0.3807:
                break
        
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)


def evaluate_c5(h, dx):
    # Compute overlaps with all shifts
    shifted = np.concatenate([h, np.zeros_like(h)])
    overlap = correlate(shifted, 1 - h, mode='valid')
    max_overlap = np.max(overlap) * dx
    return max_overlap
