import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple initial guess
        h = np.zeros(n_points)
        h[0:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search over interval distance
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
    best_d = 0.0
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create two intervals of width 0.5
        w = 0.5
        i0 = 0
        i1 = int((w) / dx) - 1
        i2 = int((w + d) / dx)
        i3 = int((2*w + d) / dx) - 1
        
        # Ensure indices are valid and non-overlapping
        if i2 > i1 and i3 < n_points:
            h_candidate = np.zeros(n_points)
            h_candidate[i0:i1+1] = 1.0
            h_candidate[i2:i3+1] = 1.0
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5 = evaluate_c5(h_candidate, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
                best_d = d
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Refine with local search around best d
    if best_d > 0 and best_d < 1:
        fine_d_candidates = np.arange(max(0, best_d - 0.02), min(1, best_d + 0.02), 0.001)
        for d in fine_d_candidates:
            if time.time() - start_time > budget_s - 1:
                break
                
            w = 0.5
            i0 = 0
            i1 = int((w) / dx) - 1
            i2 = int((w + d) / dx)
            i3 = int((2*w + d) / dx) - 1
            
            if i2 > i1 and i3 < n_points:
                h_candidate = np.zeros(n_points)
                h_candidate[i0:i1+1] = 1.0
                h_candidate[i2:i3+1] = 1.0
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                
                c5 = evaluate_c5(h_candidate, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate.copy()
                    
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
