import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def solve_qp(H, c, A, b, delta_low, delta_high):
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective: (1/2) * delta^T * H * delta + c^T * delta
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints:
    # A * delta = b
    # delta_low <= delta <= delta_high
    constraints = [A @ delta == b]
    constraints += [delta >= delta_low]
    constraints += [delta <= delta_high]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        return delta.value
    except:
        return None

def project_to_feasible(h, n_points, dx):
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available, otherwise triangular
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    h = project_to_feasible(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Precompute shift averages for efficiency
    # For each k, compute average of h[i] for shifts
    shift_avg = []
    for k in range(2 * n_points - 1):
        if k < n_points:
            avg = np.mean(h[k:])
        else:
            avg = np.mean(h[:2*n_points-k-1])
        shift_avg.append(avg)
    shift_avg = np.array(shift_avg)
    
    # QP iteration loop
    max_iter = 500
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        # ∇C_k = dx * (1 - 2*h - shift_average_k)
        # Hessian approximation (assumes constant Hessian for simplicity)
        # H = -2 * dx * I + some shift terms
        c = np.zeros(n_points)
        H = np.zeros((n_points, n_points))
        
        # For each shift k, compute contribution to c and H
        # This corresponds to linearizing the overlap at current h
        for k in range(n_points):
            # This is a simplified derivative approximation
            # Using first-order Taylor: ∇C_k ≈ dx*(1 - 2*h - shift_avg_k)
            # We want to minimize max_k C_k, so we take gradient of one at a time
            grad = dx * (1.0 - 2.0*h - shift_avg[k])
            c += grad
            # Hessian approximation
            H += -2.0 * dx * np.eye(n_points)
        
        # Add regularization to ensure positive definiteness and numerical stability
        H += 1e-4 * np.eye(n_points)
        
        # For the equality constraint Σδ = 0, we just add it to A
        A = np.ones((1, n_points))  # 1 * δ = 0
        b = np.array([0.0])  # Σδ = 0
        
        # Solve QP
        delta = solve_qp(H, c, A, b, -1.0, 1.0)
        
        if delta is None:
            # Try increasing regularization
            H += 1e-3 * np.eye(n_points)
            delta = solve_qp(H, c, A, b, -1.0, 1.0)
        
        if delta is None:
            # If QP fails, fall back to identity update
            delta = -c / (np.max(np.abs(c)) + 1e-8)
        
        # Line search to find a good step size
        alpha_max = 1.0
        for alpha in [1.0, 0.5, 0.25, 0.125]:
            h_new = h + alpha * delta
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            if c5_new < 0.3807:
                break
            
            if c5_new < best_c5:
                h = h_new
                break
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
