import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ES parameters
    population_size = 100
    generations = 1000
    sigma = 0.1
    elitism_rate = 0.2
    mutation_rate = 0.3
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness.append(evaluate_c5(individual, dx))
    
    # Evolutionary loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Elitism
        elite_count = int(elitism_rate * population_size)
        elite = population[:elite_count]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring through mutation
        while len(new_population) < population_size:
            # Select parent
            parent_idx = np.random.randint(0, elite_count)
            parent = population[parent_idx]
            
            # Mutate
            child = parent.copy()
            indices_to_mutate = np.random.choice(n_points, 
                                               size=int(mutation_rate * n_points), 
                                               replace=False)
            noise = np.random.normal(0, sigma, len(indices_to_mutate))
            child[indices_to_mutate] += noise
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Evaluate new population
        fitness = []
        for individual in population:
            fitness.append(evaluate_c5(individual, dx))
        
        # Adjust mutation rate based on improvement
        if gen % 10 == 0 and gen > 0:
            if np.mean(fitness[:elite_count]) < 0.01:  # If no significant improvement
                sigma = min(0.5, sigma * 1.1)
            else:
                sigma = max(0.001, sigma * 0.9)
    
    # SLSQP refinement
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
