import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using CVXPY"""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [A @ delta == b]
    constraints += [delta >= delta_low]
    constraints += [delta <= delta_high]
    
    # Solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian (approximate)
        # Gradient: ∇C = dx * (1 - 2*h - shift_average)
        # Hessian: ∇²C = dx * (-2*I + shift_matrix)
        
        # Simple finite difference for gradient
        C_base = evaluate_c5(h, dx)
        
        # Gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        h_plus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            C_plus = evaluate_c5(h_plus, dx)
            grad[i] = (C_plus - C_base) / eps
            h_plus[i] = h[i]
        
        # Simple second order approximation for Hessian (diagonal)
        hessian_diag = -2 * dx * np.ones(n_points)
        
        # Formulate and solve QP
        H = np.diag(hessian_diag) + 1e-5 * np.eye(n_points)
        c = grad
        
        # Equality constraint A*delta = b
        A = np.ones((1, n_points))
        b = 0
        
        # Bound constraints (relative to current h)
        delta_low = -h.copy()
        delta_high = 1 - h.copy()
        
        # Solve QP
        delta = _solve_qp(H, c, A, b, delta_low, delta_high)
        
        if delta is None:
            # Try with more regularization
            H += 1e-3 * np.eye(n_points)
            delta = _solve_qp(H, c, A, b, delta_low, delta_high)
            
        if delta is not None:
            # Line search
            alpha_max = 1.0
            for alpha in [1.0, 0.5, 0.25]:
                h_new = h + alpha * delta
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    h = h_new.copy()
                    best_c5 = c5_new
                    best_h = h.copy()
                    alpha_max = alpha
                    break
            else:
                # If no improvement with line search, use max step
                if alpha_max < 1.0:
                    h = h + alpha_max * delta
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    best_c5 = evaluate_c5(h, dx)
                    best_h = h.copy()
                    
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
