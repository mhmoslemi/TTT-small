import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Initialize dual variables
    u = np.zeros(n_points)
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update h by solving the subproblem
        # This is a simplified version focusing on minimizing the overlap
        # We'll perform a gradient-based update with gradient of overlap
        h_old = h.copy()
        
        # Compute gradient of overlap (approximate)
        h_conv = correlate(h, 1-h, mode="full")
        # For gradient approximation, we use finite differences
        grad_h = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_test = h.copy()
            h_test[i] += eps
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            h_conv_test = correlate(h_test, 1 - h_test, mode="full")
            grad_h[i] = (np.max(h_conv_test) - np.max(h_conv)) / eps
        
        # Update h
        h = h - 0.001 * grad_h
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Update dual variable
        u = u + rho * (h - h_old)
        
        # Check convergence
        primal_res = np.linalg.norm(h - h_old)
        dual_res = np.linalg.norm(rho * (h - h_old))
        if primal_res < epsilon and dual_res < epsilon:
            break
            
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
