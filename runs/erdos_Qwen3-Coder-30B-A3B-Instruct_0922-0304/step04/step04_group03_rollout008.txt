import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_update = 0.05
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        # Find the index of the maximum overlap
        k = max_overlap_idx - (n_points - 1)
        
        # Compute gradient of this overlap with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # The gradient component for h[i] when computing overlap with shift k
            if i - k >= 0 and i - k < n_points:
                grad[i] += (1 - h[i - k])  # derivative of h[i]*(1-h[i+k])
            if i + k >= 0 and i + k < n_points:
                grad[i] -= h[i + k]  # derivative of h[i+k]*(1-h[i])
        
        # Update h in negative gradient direction
        h_new = h - learning_rate * grad
        
        # Apply bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Ensure the step is not too large
        delta = h_new - h
        if np.max(np.abs(delta)) > max_update:
            delta = delta / np.max(np.abs(delta)) * max_update
            h_new = h + delta
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept the update if it improves the solution
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.995
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
