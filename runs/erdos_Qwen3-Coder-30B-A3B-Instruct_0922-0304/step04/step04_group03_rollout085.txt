import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    tolerance = 1e-6
    patience = 20
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Find the shift that gives maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j_forward = (i + k_max) % n_points
            j_backward = (i - k_max) % n_points
            gradient[i] = (1 - h[j_forward]) - h[j_backward]
        
        # Apply gradient
        h_new = h - learning_rate * gradient * dx
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Update if better
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        # Decrease learning rate if no improvement for too long
        if patience_counter >= patience:
            learning_rate *= 0.9
            patience_counter = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
