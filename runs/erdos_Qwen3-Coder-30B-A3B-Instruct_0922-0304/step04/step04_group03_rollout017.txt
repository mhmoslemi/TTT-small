import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        # Place intervals at symmetric positions
        i0 = 0
        i1 = int((w) / dx) - 1
        i2 = int((1.0 + w) / dx) 
        i3 = int((1.0 + 2*w) / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        if i2 >= n_points:
            i2 = n_points - 1
        if i3 >= n_points:
            i3 = n_points - 1
        if i2 <= i1:
            i2 = i1 + 1
        if i3 <= i2:
            i3 = i2 + 1
        h[i0:i1+1] = 1.0
        h[i2:i3+1] = 1.0
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Start with current solution
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Perform a grid search on interval separation parameter
    d_candidates = np.arange(0.0, 1.0 + 0.01, 0.01)
    best_d = 0.0
    grid_search_best = best_c5
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
        h_test = np.zeros(n_points)
        # Create two intervals separated by d
        w = 0.5  # width of each interval (set to achieve integral 1)
        # First interval at [0, w]
        i0 = 0
        i1 = int((w) / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        # Second interval at [w + d, w + d + w]
        i2 = int((w + d) / dx)
        i3 = int((w + d + w) / dx) - 1
        if i2 >= n_points:
            continue
        if i3 >= n_points:
            i3 = n_points - 1
        if i2 <= i1:
            continue
        if i3 <= i2:
            continue
        h_test[i0:i1+1] = 1.0
        h_test[i2:i3+1] = 1.0
        # Normalize to ensure integral = 1
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5_val = evaluate_c5(h_test, dx)
        if c5_val < grid_search_best:
            grid_search_best = c5_val
            best_d = d
    
    # Use the best found from grid search for refinement
    if grid_search_best < best_c5:
        best_h = np.zeros(n_points)
        w = 0.5
        d = best_d
        i0 = 0
        i1 = int((w) / dx) - 1
        if i1 >= n_points:
            i1 = n_points - 1
        i2 = int((w + d) / dx)
        i3 = int((w + d + w) / dx) - 1
        if i2 >= n_points:
            i2 = n_points - 1
        if i3 >= n_points:
            i3 = n_points - 1
        if i2 <= i1:
            i2 = i1 + 1
        if i3 <= i2:
            i3 = i2 + 1
        best_h[i0:i1+1] = 1.0
        best_h[i2:i3+1] = 1.0
        # Normalize to integral = 1
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, dx)
    
    # If we're still not satisfied, apply local perturbation method
    if best_c5 > 0.3807:
        # Iterate with small perturbations around best h
        step_size = 0.02
        max_iter = 500
        for iter_num in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Slightly perturb each value
            h_new = best_h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.3:  # Perturb 30% of indices
                    delta = np.random.uniform(-step_size, step_size)
                    h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            # Re-normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            # Reduce step size over time
            step_size *= 0.995
            
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
