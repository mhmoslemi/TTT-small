import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback to a simple step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init, dx)
    
    # Try different local search approaches to escape local minima
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient-based refinement using scipy.optimize
        from scipy.optimize import minimize
        
        def objective(h_vec):
            # We need to compute C5 = max_k int h(x)(1-h(x+k)) dx
            # This is the overlap integral we want to minimize
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_c5(h, dx)
            
        def constraint_eq(h_vec):
            return np.sum(h_vec) * dx - 1.0
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B to refine current solution
        result = minimize(objective, best_h.copy(), method='L-BFGS-B', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 50, 'ftol': 1e-8})
        if result.success:
            refined = np.clip(result.x, 0, 1)
            refined = refined * (n_points / 2) / np.sum(refined)
            new_c5 = evaluate_c5(refined, dx)
            if new_c5 < best_c5:
                best_h = refined
                best_c5 = new_c5
                
        # Random perturbation to escape local minima more effectively
        perturb = np.random.normal(0, 0.02, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        
        new_c5 = evaluate_c5(perturbed, dx)
        if new_c5 < best_c5:
            best_h = perturbed
            best_c5 = new_c5
            
        # Particle swarm optimization step to explore further (10% chance)
        if np.random.rand() < 0.1:
            # Simple perturbation with small step size
            step = 0.01 * np.random.randn(n_points)
            h_swarm = np.clip(best_h + step, 0, 1)
            h_swarm = h_swarm * (n_points / 2) / np.sum(h_swarm)
            new_c5 = evaluate_c5(h_swarm, dx)
            if new_c5 < best_c5:
                best_h = h_swarm
                best_c5 = new_c5
                
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Use the same evaluation in the harness for consistency
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap
