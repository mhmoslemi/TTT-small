import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular initial solution
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Perturbations with different strategies
    strategies = [
        lambda x: x + np.random.normal(0, 0.02, n_points),  # small noise
        lambda x: np.clip(x + np.random.normal(0, 0.05, n_points), 0, 1),  # larger noise
        lambda x: x * (1 + np.random.normal(0, 0.01, n_points)),  # multiplicative
    ]
    
    # Additional refinement: gradient descent based on overlap gradient
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try several local perturbations
        candidates = []
        for strategy in strategies:
            perturbed = strategy(h)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            candidates.append(perturbed)
            
        # Evaluate candidates
        for candidate in candidates:
            c5_candidate = evaluate_c5(candidate, dx)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()
                
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Gradient improvement via small steps
        h_original = h.copy()
        h = h * (n_points / 2) / np.sum(h)
        
        # Simple gradient descent (direct numerical)
        eps = 1e-4
        current_c5 = evaluate_c5(h, dx)
        
        # Sample around current point for gradient estimate
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update using gradient step
        step_size = 0.01
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Live updating of best
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
