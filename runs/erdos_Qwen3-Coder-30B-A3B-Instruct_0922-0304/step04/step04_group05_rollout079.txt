import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a triangular function
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based approach with sequential convex programming
    alpha = 0.1
    damping = 0.5
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the correlation for current h
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Approximate gradient and Hessian
        # For each point, we approximate the contribution to the overlap
        # We build a linear model around current h
        
        # Solve convex subproblem
        delta = cp.Variable(n_points)
        
        # Build the quadratic approximation of the objective
        # Gradient approximation for overlap our current maximum overlap term
        # This is a simple 1st order approximation based on finite differences:
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward difference gradient
            h_forward = h.copy()
            h_forward[i] += 1e-6
            corr_forward = correlate(h_forward, 1 - h_forward, mode="full")
            # Derivative of overlap w.r.t. h[i] at this location
            grad[i] = (corr_forward[max_idx] - max_val) * dx / 1e-6
        
        # Hessian approximation using finite diff
        hess = np.zeros((n_points, n_points))
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            hess[i, i] = (corr_plus[max_idx] - 2 * max_val + corr_minus[max_idx]) / (eps ** 2)
        
        # Add a small regularization to make it numerically stable
        hess += 1e-6 * np.eye(n_points)
        
        # Convex programming problem for update delta
        objective = cp.Minimize(cp.matmul(grad.T, delta) + 0.5 * cp.quad_form(delta, hess))
        
        # Constraints
        constraints = [
            h + delta >= 0,
            h + delta <= 1,
            cp.sum(delta) == 0  # This ensures the sum constraint remains invariant
        ]
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                delta_val = delta.value
            else:
                delta_val = np.zeros(n_points)
        except:
            delta_val = np.zeros(n_points)
        
        # Line search
        step_size = alpha
        new_h = h + step_size * delta_val
        
        # Apply bounds
        new_h = np.clip(new_h, 0, 1)
        # Restore area constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = evaluate_c5(new_h, dx)
        
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
            alpha = min(alpha * 1.2, 1.0)  # Increase step size
        else:
            alpha *= damping  # Decrease step size
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
