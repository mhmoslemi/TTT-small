import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise create a triangular shape
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Grid search on interval separation distance
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Try different interval widths and separation distances
    interval_width = 0.5
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create piecewise constant function with two intervals
        i0 = 0
        i1 = int((interval_width) / dx)
        i2 = int((interval_width + d) / dx)
        i3 = int((2 * interval_width + d) / dx)
        
        # Ensure valid indices
        if i2 <= i1:
            continue
        if i3 > n_points:
            i3 = n_points - 1
            
        h_candidate = np.zeros(n_points)
        h_candidate[i0:i1+1] = 1.0
        h_candidate[i2:i3+1] = 1.0
        
        # Normalize to satisfy ∫h dx = 1
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        c5 = evaluate_c5(h_candidate, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Fine-tune the best solution found
    fine_d_candidates = np.arange(max(0, best_h[1] - 0.05), min(1, best_h[1] + 0.05), 0.001)
    for d in fine_d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        i0 = 0
        i1 = int((interval_width) / dx)
        i2 = int((interval_width + d) / dx)
        i3 = int((2 * interval_width + d) / dx)
        
        if i2 <= i1:
            continue
        if i3 > n_points:
            i3 = n_points - 1
            
        h_candidate = np.zeros(n_points)
        h_candidate[i0:i1+1] = 1.0
        h_candidate[i2:i3+1] = 1.0
        
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        c5 = evaluate_c5(h_candidate, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
