import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute C5 from h values"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum, keeping values in [0,1]"""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h
