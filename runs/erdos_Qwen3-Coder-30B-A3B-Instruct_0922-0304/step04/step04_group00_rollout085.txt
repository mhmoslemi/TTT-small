import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5  # default initialization
        
    def compute_c5(h):
        # Compute the maximum overlap integral
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1e-4  # Initial temperature
    alpha = 0.99  # Cooling rate
    min_T = 1e-10  # Minimum temperature
    step_size = 0.05  # Initial step size for perturbation
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Make a perturbation to h
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        perturbation = np.random.normal(0, step_size)
        h_new[idx] += perturbation
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5 for new solution
        new_c5 = compute_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        delta = new_c5 - compute_c5(h)
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
        
        # Update best solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Cooling schedule
        T = max(min_T, T * alpha)
        step_size = max(1e-6, step_size * alpha)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
