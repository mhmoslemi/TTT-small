import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    max_iter = 2000
    rho = 1.0
    epsilon = 1e-4
    
    # Initialize variables
    x = h.copy()
    z = np.zeros(2 * n_points - 1)  # For correlation
    u = np.zeros(2 * n_points - 1)  # Dual variable
    lambda_ = np.zeros(n_points)    # For sum constraint
    mu = 0.0                        # Dual variable for sum constraint
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update x (primal variable for h)
        # This computes the gradient of the objective w.r.t. h
        # We use a gradient descent step
        new_x = x.copy()
        # Simple gradient step using gradient estimation
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (evaluate_c5(x_plus, dx) - evaluate_c5(x_minus, dx)) / (2 * eps)
        
        new_x -= 0.001 * grad  # Gradient step
        new_x = np.clip(new_x, 0, 1)
        new_x = new_x * (n_points / 2) / np.sum(new_x)  # Rescale for sum constraint
        
        # Update z (primal variable for correlation)
        corr = correlate(new_x, 1 - new_x, mode="full")
        z = corr + u / rho  # Add dual variable and scale
        
        # Update dual variables
        u += rho * (corr - z)
        
        # Update lambda and mu for sum constraint
        sum_violation = np.sum(new_x) - n_points / 2
        lambda_ += 0.01 * sum_violation
        mu += 0.01 * sum_violation
        
        # Update x with the sum constraint
        x = new_x
        x = x - np.mean(x) + 0.5  # Adjust to meet target mean
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)  # Final rescale
        
        # Evaluate current solution
        current_c5 = evaluate_c5(x, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
