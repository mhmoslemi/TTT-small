import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_generations = 2000
    mutation_scale = 0.1
    elite_ratio = 0.25
    success_threshold = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_to_feasible(h0, n_points)
    
    population.append(h0.copy())
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Evolutionary strategy
    generation = 0
    success_count = 0
    last_improvement = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = [evaluate_c5(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            last_improvement = generation
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = max(1, int(elite_ratio * population_size))
        elite_indices = sorted_indices[:elite_size]
        
        # Create offspring
        new_population = []
        for i in range(population_size):
            # Select parent from elite
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_scale, n_points)
            offspring = parent + noise
            
            # Project to feasible region
            offspring = project_to_feasible(offspring, n_points)
            
            new_population.append(offspring)
        
        # Adaptive mutation adjustment
        if generation > 1 and generation - last_improvement > 10:
            mutation_scale = min(0.5, mutation_scale * 1.1)
        elif success_count > 0 and success_count >= elite_size * success_threshold:
            mutation_scale = max(0.001, mutation_scale * 0.9)
            
        population = new_population
        
        # Adaptive behavior
        if generation - last_improvement > 50:
            mutation_scale = min(0.5, mutation_scale * 1.2)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
