import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 75
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple triangular construction
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / (mid + 1))
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian optimization with adaptive step size
    lambda_mult = 0.0  # Lagrange multiplier for sum constraint
    max_iter = 10000
    step_size = 0.002
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap and gradient
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Estimate gradient by finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update h using gradient descent
        h = h - step_size * grad
        
        # Project onto feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update Lagrange multiplier
        constraint_violation = np.sum(h) - n_points / 2
        lambda_mult = lambda_mult + 0.1 * constraint_violation
        
        # Adaptive step sizing
        if iteration % 10 == 0:
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                # Reduce step size if making progress
                step_size = max(1e-6, step_size * 0.95)
            else:
                # Increase step size if stuck
                step_size = min(0.01, step_size * 1.02)
        
        # Occasionally apply perturbations to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
