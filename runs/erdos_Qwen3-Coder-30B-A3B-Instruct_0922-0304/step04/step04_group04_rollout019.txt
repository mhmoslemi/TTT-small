import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis setup - use first 30 basis functions
    n_basis = 30
    
    # Generate Haar wavelet basis
    def haar_wavelet(x, j, k):
        # j = level, k = position
        # Wavelet function psi_{j,k}(x) = 2^{j/2} psi(2^j x - k)
        if j == 0:
            return 1.0 if 0 <= x < 1 else 0.0
        else:
            # Compute 2^j x - k
            arg = 2**j * x - k
            if 0 <= arg < 0.5:
                return 2**(j/2)
            elif 0.5 <= arg < 1:
                return -2**(j/2)
            else:
                return 0.0
    
    # Precompute basis functions at discrete points
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    basis_matrix = np.zeros((n_points, n_basis))
    
    # Embed wavelet basis function
    for j in range(6):  # Level 0 to 5
        for k in range(2**j):
            if j == 0:
                basis_matrix[:, j] = (x_vals >= 0) & (x_vals < 1)
            else:
                for i in range(n_points):
                    basis_matrix[i, j] = haar_wavelet(x_vals[i], j, k)
    
    # Normalize basis functions
    for i in range(n_basis):
        if np.linalg.norm(basis_matrix[:, i]) > 0:
            basis_matrix[:, i] /= np.linalg.norm(basis_matrix[:, i])
    
    # Try to initialize using parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Project to wavelet coefficients
        c_init = basis_matrix.T @ h_init
    else:
        # Random initialization
        c_init = np.random.rand(n_basis) - 0.5
    
    # Try optimization with CVXPY
    try:
        # Define variables
        c = Variable(n_basis)
        # w = Variable(n_points)
        
        # Define constraints
        # Integral constraint: sum(c * integral of basis functions) == 1
        integral_constraint = sum(c[i] * (basis_matrix[:, i] @ np.ones(n_points)) * dx for i in range(n_basis)) == 1
        
        # Bounds constraint
        bounds_constraints = [c[i] >= -10 for i in range(n_basis)]
        bounds_constraints.extend([c[i] <= 10 for i in range(n_basis)])
        
        # Build maximum overlap constraint
        # We'll do this with a comprehensive approach by solving
        # a simpler and direct search if possible
        
        # Direct search in the sampling space
        # Try to make compact search space based on initial_h_values
        
        # Better solution approach: dual approach with refined PSO on valid function space
        # Create a differentiable optimization
        # We start with PSO on the full function space to warm start
        
        # Direct constraint
        def wavelet_to_function(c_vals, dx):
            h = basis_matrix @ c_vals
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return h
        
        def evaluate_c5_wavelet(c_vals):
            h = wavelet_to_function(c_vals, dx)
            # Correlation-based computation
            overlap = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap) * dx
            return max_overlap

        # PSO-style descent on wavelet space
        n_pso_particles = 20
        max_iter_pso = 500
        w_start = 0.9
        w_end = 0.4
        c_p = 1.5
        c_g = 1.5
        v_max = 0.1
        
        swarm = []
        velocities = []
        pbest = []
        pbest_scores = []
        
        # Initialize particles with varying strategies:
        for i in range(n_pso_particles):
            if i == 0 and 'initial_h_values' in globals():
                h_in = np.array(initial_h_values)
                h_in = np.clip(h_in, 0, 1)
                h_in = h_in * (n_points / 2) / np.sum(h_in)
                c = basis_matrix.T @ h_in
            else:
                # Random search with wavelet basis
                c = np.random.rand(n_basis) * 2 - 1  # [-1,1]
                # Clip to [0,1] part for reasonable starting
            swarm.append(c.copy())
            velocities.append(np.zeros(n_basis))
            pbest.append(c.copy())
            score = evaluate_c5_wavelet(c)
            pbest_scores.append(score)
            
        # Global best
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        for iter_count in range(max_iter_pso):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update inertia weight
            w = w_start - (w_start - w_end) * (iter_count / max_iter_pso)
            
            for i in range(n_pso_particles):
                # Update velocity
                r_p = np.random.rand(n_basis)
                r_g = np.random.rand(n_basis)
                
                velocities[i] = (w * velocities[i] 
                               + c_p * r_p * (pbest[i] - swarm[i])
                               + c_g * r_g * (gbest - swarm[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                swarm[i] = swarm[i] + velocities[i]
                
                # Evaluate
                score = evaluate_c5_wavelet(swarm[i])
                
                # Update personal best
                if score < pbest_scores[i]:
                    pbest[i] = swarm[i].copy()
                    pbest_scores[i] = score
                    
                    # Update global best
                    if score < gbest_score:
                        gbest = swarm[i].copy()
                        gbest_score = score
                        
            # Early termination if already good
            if gbest_score <= 0.3807:
                break
        
        # Final reconstruction and evaluation
        final_h = wavelet_to_function(gbest, dx)
        final_c5 = evaluate_c5_wavelet(gbest)
        
        return (final_h.tolist(), final_c5, n_points)
    
    except Exception as e:
        # Fallback to previous PSO method with tuned parameters if everything else fails
        try:
            n_particles = 20
            max_iter = 1000
            
            # Define swarm variables
            swarm = []
            velocities = []
            pbest = []
            pbest_scores = []
            
            # Initialize from parent if available
            if 'initial_h_values' in globals():
                h_init = np.array(initial_h_values)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                swarm.append(h_init.copy())
            else:
                h_init = np.zeros(n_points)
                # Create a more balanced shape
                h_init[:n_points//2] = 0.7
                h_init[n_points//2:] = 0.3
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                swarm.append(h_init.copy())
            
            # Add remaining swarm particles
            for i in range(1, n_particles):
                h = np.random.rand(n_points) * 0.5 + 0.2  # Random function with reasonable range
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                swarm.append(h.copy())
            velocities = [np.random.rand(n_points) * 0.02 for _ in range(n_particles)]
            pbest = [s.copy() for s in swarm]
            pbest_scores = [evaluate_c5(s, dx) for s in swarm]
            
            gbest_idx = np.argmin(pbest_scores)
            gbest = pbest[gbest_idx].copy()
            gbest_score = pbest_scores[gbest_idx]
            
            # PSO parameters
            w_start = 0.9
            w_end = 0.4
            c_p = 1.5
            c_g = 1.5
            v_max = 0.05
            
            for iter_count in range(max_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                
                # Update inertia weight
                w = w_start - (w_start - w_end) * (iter_count / max_iter)
                
                for i in range(n_particles):
                    # Update velocity
                    r_p = np.random.rand(n_points)
                    r_g = np.random.rand(n_points)
                    velocities[i] = (w * velocities[i] 
                                   + c_p * r_p * (pbest[i] - swarm[i])
                                   + c_g * r_g * (gbest - swarm[i]))
                    
                    # Clamp velocity to max
                    velocities[i] = np.clip(velocities[i], -v_max, v_max)
                    
                    # Update position
                    swarm[i] = swarm[i] + velocities[i]
                    
                    # Enforce bounds
                    swarm[i] = np.clip(swarm[i], 0, 1)
                    
                    # Enforce normalization
                    swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
                    
                    # Evaluate
                    score = evaluate_c5(swarm[i], dx)
                    
                    # Update personal best
                    if score < pbest_scores[i]:
                        pbest[i] = swarm[i].copy()
                        pbest_scores[i] = score
                        
                        # Update global best
                        if score < gbest_score:
                            gbest = swarm[i].copy()
                            gbest_score = score
                            
                # Early termination
                if gbest_score <= 0.3807:
                    break
                    
            return (gbest.tolist(), gbest_score, n_points)
        except Exception as e2:
            # Last resort - return initial solution or a simple form if it exists
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                return (h.tolist(), evaluate_c5(h, dx), n_points)
            else:
                # Default step function
                h = np.zeros(n_points)
                h[:n_points//2] = 1
                return (h.tolist(), evaluate_c5(h, dx), n_points)
