import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to match sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def line_search(h, grad, dx, n_points, target_sum, max_alpha=1.0):
    # Simple Armijo line search
    alpha_max = max_alpha
    alpha = alpha_max
    c = 1e-4
    beta = 0.5
    
    while alpha > 1e-8:
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, target_sum)
        new_c5 = evaluate_c5(h_new, dx)
        old_c5 = evaluate_c5(h, dx)
        # Armijo condition: f(x + alpha * d) <= f(x) + c * alpha * <grad, d>
        if new_c5 <= old_c5 - c * alpha * np.dot(grad, -grad):
            return h_new, alpha
        alpha *= beta
    
    return h, alpha
