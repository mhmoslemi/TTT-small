import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Simple initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier approach with McCormick relaxation
    basis_size = 20
    
    # Build basis functions
    x_vals = np.linspace(0, 2, n_points)
    phi = np.zeros((n_points, basis_size + 1))
    phi[:, 0] = 1  # constant term
    
    for i in range(1, basis_size + 1):
        if i % 2 == 1:  # odd: sine terms
            phi[:, i] = np.sin(np.pi * (i + 1) // 2 * x_vals)
        else:  # even: cosine terms
            phi[:, i] = np.cos(np.pi * i // 2 * x_vals)
    
    # Precompute kernel K with full convolution
    conv_len = 2 * n_points - 1
    K = np.zeros((basis_size + 1, basis_size + 1, conv_len))
    
    # Compute convolution kernel for each pair of basis functions
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            kernel = correlate(phi[:, i], 1 - phi[:, j], mode="full")
            K[i, j, :] = kernel * dx  # scale by dx
    
    # Define variables
    c = Variable(basis_size + 1)  # Fourier coefficients
    z = Variable((basis_size + 1, basis_size + 1))  # McCormick relaxation variables
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(c[0] * 2 + sum(c[i] * np.sum(phi[:, i]) * dx for i in range(1, basis_size + 1)) == 1)
    
    # Bound constraints for h(x)
    h_values = phi @ c
    for i in range(n_points):
        constraints.append(h_values[i] >= 0)
        constraints.append(h_values[i] <= 1)
    
    # McCormick relaxation
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            # z[i,j] >= c[i] + c[j] - 1
            constraints.append(z[i, j] >= c[i] + c[j] - 1)
            # z[i,j] <= c[i]
            constraints.append(z[i, j] <= c[i])
            # z[i,j] <= c[j]
            constraints.append(z[i, j] <= c[j])
            # z[i,j] >= 0
            constraints.append(z[i, j] >= 0)
    
    # Objective: minimize maximum overlap
    # We need to approximate the maximum of the correlations
    # For now we'll create a concrete approximate objective
    max_overlap_expr = 0
    for l in range(conv_len):
        expr = sum(z[i, j] * K[i, j, l] for i in range(basis_size + 1) for j in range(basis_size + 1))
        max_overlap_expr = Max(max_overlap_expr, abs(expr))
    
    # We'll instead use a practical approach: optimize by gradient descent with restarts
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Adapted from PSO version: refine using gradient descent
    best_h = h_init.copy()
    best_score = objective(best_h)
    
    # Simple gradient-based refinement
    for iter_count in range(500):  # Increase iterations for refinement
        
        if time.time() - start_time > budget_s - 1:
            break
            
        h_new = best_h.copy()
        
        # Apply small random changes, and accept if better
        for i in range(n_points):
            # Take small step (not too large)
            step = np.random.normal(0, 0.001)
            h_new[i] = h_new[i] + step
            
            # Enforce bounds
            h_new[i] = np.clip(h_new[i], 0, 1)
            
            # Renormalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            score = objective(h_new)
            
            if score < best_score:
                best_h = h_new.copy()
                best_score = score
            else:
                h_new[i] = best_h[i]  # reject
            
        if best_score <= 0.3807:
            break
    
    return (best_h.tolist(), best_score, n_points)
