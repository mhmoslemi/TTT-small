import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # For tracking consecutive failures
    no_improvement_count = 0
    max_no_improvement = 20
    
    # GA parameters
    pop_size = 50
    num_generations = 1000
    tournament_size = 5
    mutation_rate = 0.1
    elite_size = 5
    local_search_fraction = 0.2

    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = h + np.random.normal(0, 0.01, n_points)
        else:
            individual = np.ones(n_points)
            individual[n_points//2:] = 0
            individual += np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual.copy())
    
    # Evaluation function for population
    def evaluate_population(pop):
        fitness = []
        for ind in pop:
            fit = compute_c5(ind)
            fitness.append(fit)
        return np.array(fitness)
    
    # Tournament selection
    def tournament_selection(pop, fitness, tournament_size):
        selected = []
        for _ in range(len(pop)):
            tournament_indices = np.random.choice(len(pop), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(pop[winner_idx].copy())
        return selected
    
    # Crossover function
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Ensure feasibility
        child1 = np.clip(child1, 0, 1)
        child1 = child1 * (n_points / 2) / np.sum(child1)
        
        child2 = np.clip(child2, 0, 1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    # Mutation function
    def mutate(individual):
        mask = np.random.random(n_points) < mutation_rate
        individual = individual + np.random.normal(0, 0.01, n_points) * mask
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual

    # Main GA loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        fitness = evaluate_population(population)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Local search on top individuals
        top_indices = np.argsort(fitness)[:int(pop_size * local_search_fraction)]
        for idx in top_indices:
            current = population[idx].copy()
            result = minimize(
                compute_c5, 
                current, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if result.success:
                new_h = result.x
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                population[idx] = new_h
                
        # Selection
        selected_pop = tournament_selection(population, fitness, tournament_size)
        
        # Create new population
        new_population = []
        
        # Keep elite
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Crossover and mutation for rest
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected_pop, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:pop_size]
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
