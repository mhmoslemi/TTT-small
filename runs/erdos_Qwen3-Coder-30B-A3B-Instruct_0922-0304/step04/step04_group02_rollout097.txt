import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    rho = 1.0
    max_iter = 2000
    epsilon = 1e-4
    
    # Initialize ADMM variables
    x = h.copy()
    z = np.zeros(n_points)
    u = np.zeros(n_points)
    
    # Create correlation matrix for efficiency
    def compute_correlation(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(v):
        # Clip to [0,1] and rescale
        v = np.clip(v, 0, 1)
        v = v * (n_points / 2) / np.sum(v)
        return v
    
    # ADMM iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update x (minimize augmented Lagrangian)
        # x = argmin_{x} (max_k correlation) + (rho/2)||x - z + u||^2
        # We approximate this by solving a quadratic subproblem
        # But for simplicity, we use gradient descent approach or optimized update
        
        # Compute gradient of the correlation with respect to x
        # Since it's not straightforward, we use a simpler approach:
        # Update using gradient descent from current solution
        old_x = x.copy()
        
        # Simple gradient ascent on relatively smooth approximations
        # We will use a randomized approach for fine adjustments
        gradient = np.zeros(n_points)
        
        # Estimate gradient using finite differences around current point
        eps = 1e-4
        for i in range(n_points):
            h_plus = x.copy()
            h_minus = x.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Make sure they're feasible
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            val_plus = compute_correlation(h_plus)
            val_minus = compute_correlation(h_minus)
            
            gradient[i] = (val_plus - val_minus) / (2 * eps)
        
        # Update x similar to gradient descent
        x = old_x - 0.1 * gradient
        x = project_feasible(x)
        
        # Update z (consensus variable)
        z = (x + u) / 2.0  # Simplified for single constraint
        z = project_feasible(z)
        
        # Update dual variables (u)
        u = u + (x - z)
        
        # Evaluate current solution
        current_c5 = compute_correlation(x)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        # Constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}

            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds,

                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation

    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
