import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Simulated annealing with local search
    temperature = 0.1
    cooling_rate = 0.999
    min_temp = 1e-6
    step_size = 0.02
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create candidate solution by perturbing current solution
        candidate_h = h.copy()
        # Random perturbation
        perturb_indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        for idx in perturb_indices:
            candidate_h[idx] += np.random.normal(0, step_size)
        
        # Enforce bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Enforce integral constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate candidate
        c5_candidate = evaluate_c5(candidate_h, dx)
        
        # Accept or reject based on simulated annealing
        if c5_candidate < best_c5 or np.random.random() < np.exp(-(c5_candidate - best_c5) / temperature):
            h = candidate_h
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                
        # Cool down temperature
        temperature *= cooling_rate
        if temperature < min_temp:
            temperature = min_temp
            
    return (best_h.tolist(), best_c5, n_points)
