import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple triangular distribution
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Try piecewise constant construction with two intervals
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search over interval separation parameter
    d_candidates = np.arange(0.0, 1.0 + 1e-6, 0.01)
    
    for d in d_candidates:
        # Define two intervals of width 0.5 each
        w = 0.5
        i0 = 0
        i1 = int((w) / dx) - 1
        i2 = int((w + d) / dx)
        i3 = int((2*w + d) / dx) - 1
        
        # Ensure valid indices
        if i1 >= n_points or i2 < 0:
            continue
            
        # Ensure valid intervals
        if i2 <= i1:
            continue
            
        # Create step function
        h_test = np.zeros(n_points)
        if i0 <= i1:
            h_test[i0:i1+1] = 1.0
        if i2 <= i3:
            h_test[i2:i3+1] = 1.0
            
        # Normalize to satisfy constraint
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5_test = evaluate_c5(h_test, dx)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Fine search around best found so far
    if best_c5 > 0.3807:
        d_candidates_fine = np.arange(max(0, best_c5 - 0.01), 
                                    min(1, best_c5 + 0.01), 
                                    0.001)
        for d in d_candidates_fine:
            w = 0.5
            i0 = 0
            i1 = int((w) / dx) - 1
            i2 = int((w + d) / dx)
            i3 = int((2*w + d) / dx) - 1
            
            if i1 >= n_points or i2 < 0:
                continue
                
            if i2 <= i1:
                continue
                
            h_test = np.zeros(n_points)
            if i0 <= i1:
                h_test[i0:i1+1] = 1.0
            if i2 <= i3:
                h_test[i2:i3+1] = 1.0
                
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            c5_test = evaluate_c5(h_test, dx)
            
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                
            if time.time() - start_time > budget_s - 1:
                break
    
    # If still far from target, do simple ASGD optimization
    if best_c5 > 0.3807 and time.time() - start_time < budget_s - 5:
        # Initialize with best solution and slightly randomize
        h = best_h.copy()
        # Make sure it's normalized
        h = h * (n_points / 2) / np.sum(h)
        
        # Use gradient-based method with carefully controlled perturbations
        for iter_num in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Slight random perturbation
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            perturbation = np.random.uniform(-0.05, 0.05)
            h_new[idx] += perturbation
            
            # Clamp
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Eval
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
