import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    """Project h to stay within [0,1] and match target sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        # Create a triangular starting function
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / (n_points / 2))
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Adaptive gradient descent with line search
    max_iter = 1000
    step_size = 0.1
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # For h[i] (1 - h[i+k]), derivative w.r.t. h[i] is (1 - 2*h[i] - average of shifted terms)
        # We compute the overlaps and gradients approximately
        C_vals = []
        for shift in range(n_points):
            # Compute the overlap for shift k
            shifted = np.roll(h, shift)  # Circular shift to simulate the integral
            if shift == 0:
                overlap = h * (1 - h)
            else:
                overlap = h[shift:] * (1 - shifted[:-shift])
            C_vals.append(np.sum(overlap) * dx)
        
        # Actually compute the full overlap matrix
        h_extended = np.concatenate([h, np.zeros(n_points-1)])
        complement = 1 - h
        complement_extended = np.concatenate([complement, np.zeros(n_points-1)])
        conv = np.convolve(h_extended, complement_extended, mode='full')
        conv = conv[:n_points] * dx
        C5 = np.max(conv)
        
        # Approximate gradient numerically by finite difference
        gradient = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            C_plus = evaluate_c5(h_plus, dx)
            gradient[i] = (C_plus - C5) / eps
        
        # Gradient descent update, projected to feasible region
        h_new = h - step_size * gradient
        h_new = project_to_feasible(h_new, target_sum)
        
        # Line search
        alpha = 1.0
        max_alpha = 1.0
        min_alpha = 1e-5
        while alpha >= min_alpha:
            h_trial = h - alpha * gradient
            h_trial = project_to_feasible(h_trial, target_sum)
            trial_c5 = evaluate_c5(h_trial, dx)
            
            if trial_c5 <= C5:
                h = h_trial
                new_c5 = trial_c5
                break
            else:
                alpha *= 0.5
        else:
            h = h_new
            new_c5 = evaluate_c5(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
            step_size = min(0.1, step_size * 1.1)  # Adaptive step size increase
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                step_size *= 0.5  # Reduce step size if stuck
                stagnation_count = 0
        
        # Early stopping
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
