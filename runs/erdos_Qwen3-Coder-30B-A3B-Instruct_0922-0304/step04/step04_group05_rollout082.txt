import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    """Solve quadratic program using CVXPY"""
    n = len(c)
    delta = cp.Variable(n)
    
    # Objective: minimize (1/2)*delta^T*H*delta + c^T*delta
    objective = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [
        A @ delta == b,
        delta >= delta_low,
        delta <= delta_high
    ]
    
    # Form and solve problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Precompute stuff
    shift_averages = []
    for k in range(n_points):
        avg = np.mean(h[:-k] if k > 0 else h)
        shift_averages.append(avg)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian
        grad = dx * (1 - 2 * h - np.array(shift_averages))
        
        # Hessian approximation (stationary point for quadratic)
        H = np.zeros((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):
                if i == j:
                    H[i, j] = -2 * dx
                else:
                    H[i, j] = dx if abs(i - j) < n_points else 0
        
        # Regularize Hessian
        H += 1e-5 * np.eye(n_points)
        
        # Compute A matrix (constraint that sum of delta = 0)
        A = np.ones((1, n_points))
        b = 0.
        
        # Solve QP
        delta = _solve_qp(H, grad, A, b, -h, 1-h)
        
        if delta is None:
            # Try again with more regularization
            H += 1e-3 * np.eye(n_points)
            delta = _solve_qp(H, grad, A, b, -h, 1-h)
            
        if delta is None:
            # Fallback to small random perturbation
            delta = np.random.normal(0, 0.001, n_points)
            delta = delta - np.mean(delta)  # preserve sum zero
            
        # Line search for optimal step size
        alpha_max = 1.0
        for alpha in [0.5, 0.25, 0.1]:
            h_new = h + alpha * delta
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            if c5_new <= 0.3807:
                break
                
            if c5_new < evaluate_c5(h, dx):
                alpha_max = alpha
                break
                
        # Update h with best alpha found
        alpha = alpha_max
        h = h + alpha * delta
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check if we've improved
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
