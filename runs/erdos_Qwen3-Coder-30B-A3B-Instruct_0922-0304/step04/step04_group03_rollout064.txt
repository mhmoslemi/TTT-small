import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    patience_counter = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        # Find k that maximizes overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient: for each i, gradient[i] = (1 - h[(i+k_max)%n_points]) - h[(i-k_max)%n_points]
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j_plus = (i + k_max) % n_points
            j_minus = (i - k_max) % n_points
            gradient[i] = (1 - h[j_plus]) - h[j_minus]
        
        # Update h using gradient descent
        h = h - learning_rate * gradient
        
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Check for improvement
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience_counter = 0
            # Reduce learning rate if we're making good progress
            if learning_rate > 0.001:
                learning_rate *= 0.99
        else:
            patience_counter += 1
            if patience_counter > patience:
                # Reduce learning rate and reset patience
                learning_rate *= 0.5
                patience_counter = 0
                if learning_rate < 1e-6:
                    break
    
    return (best_h.tolist(), best_c5, n_points)
