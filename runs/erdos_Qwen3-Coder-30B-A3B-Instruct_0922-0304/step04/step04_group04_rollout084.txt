import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Precompute basis functions
    x_vals = np.linspace(0, 2, n_points)
    phi_basis = np.zeros((n_points, basis_size + 1))
    phi_basis[:, 0] = 1.0  # constant term
    for k in range(1, basis_size + 1):
        if k % 2 == 1:
            phi_basis[:, k] = np.sin(np.pi * k * x_vals / 2)
        else:
            phi_basis[:, k] = np.cos(np.pi * k * x_vals / 2)
    
    # Precompute integral of basis functions
    int_phi = np.zeros(basis_size + 1)
    for k in range(basis_size + 1):
        int_phi[k] = np.trapz(phi_basis[:, k], x_vals)
    
    # Initialize coefficients
    c = Variable(basis_size + 1)
    
    # Set up the optimization problem
    # We will try to optimize C5 = max_k integral h(x)*(1-h(x+k)) dx
    # Using a Fourier decomposition and McCormick relaxation
    
    # Initialize with existing solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Convert to Fourier coefficients
        c_init = phi_basis.T @ h_init * dx
        c_init[0] = c_init[0] - (np.sum(c_init * int_phi) - 1) / int_phi[0] if int_phi[0] != 0 else c_init[0]
        c.value = c_init
    else:
        # Simple initial guess
        c.value = np.random.rand(basis_size + 1) * 0.1
        c.value[0] = 0.5  # Mean
    
    # Try to solve with CVXPY
    max_lag = 2 * n_points
    
    # Precompute convolution matrix
    K = np.zeros((basis_size + 1, basis_size + 1, max_lag))
    for i in range(basis_size + 1):
        for j in range(basis_size + 1):
            for lag in range(max_lag):
                shift = (lag - max_lag // 2) * dx
                if i == 0 and j == 0:
                    K[i, j, lag] = 1
                else:
                    # Evaluate shifted basis
                    basis_i = phi_basis[:, i]
                    basis_j_shifted = np.interp(x_vals + shift, x_vals, phi_basis[:, j], left=0, right=0)
                    K[i, j, lag] = np.trapz(basis_i * basis_j_shifted, x_vals)
    
    # Define variables for McCormick relaxation
    Z = Variable((basis_size + 1, basis_size + 1), symmetric=True)
    
    # Constraints
    constraints = [c[0] >= 0, c[0] <= 1]
    for i in range(1, basis_size + 1):
        constraints.append(c[i] >= 0)
        
    # Convex relaxation constraints via McCormick
    for i in range(basis_size + 1):
        for j in range(i, basis_size + 1):
            # McCormick envelope constraints
            if i == 0 and j == 0:
                z = 1
            elif i == 0 or j == 0:
                constraints.append(Z[i, j] >= 0)
                constraints.append(Z[i, j] <= c[i])
                constraints.append(Z[i, j] <= c[j])
            else:
                constraints.append(Z[i, j] >= c[i] + c[j] - 1)
                constraints.append(Z[i, j] <= c[i])
                constraints.append(Z[i, j] <= c[j])
    
    # Normalize first coefficient to match integral constraint
    constraints.append(2 * c[0] + sum(int_phi[k] * c[k] for k in range(1, basis_size + 1)) == 1)
    
    # Objective: minimize maximum overlap
    # We'll evaluate this over all convolution lags
    overlap_terms = []
    for lag in range(max_lag):
        term = sum(Z[i, j] * K[i, j, lag] for i in range(basis_size + 1) for j in range(basis_size + 1) if i <= j)
        overlap_terms.append(term)
        
    # The original problem is to minimize the maximum value among these terms
    # But our relaxation involves summing products rather than doing correlation correctly
    # For simplicity, we'll take a direct and practical approach
    # Reconstruct h via basis and evaluate the integral in a common way
    # So instead of optimizing through something complex, restart with simpler solver and cleaner initialization
    
    # NumPy-based gradient descent on small space
    def compute_c5(h):
        # Compute overlap integral with self shifted
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Start with better initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize using slightly better step-like approach
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = np.abs(h - 0.1 * np.random.rand(n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimize using simple local search around the initial solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Begin local optimization
    for iter_count in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb with random walk
        h_new = h + 0.02 * np.random.randn(n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
            
        # Occasionally, accept a move that saves a bit more
        if iter_count == 0 or np.random.rand() < 0.01:
            h = h_new.copy()
            
    return (best_h.tolist(), best_c5, n_points)
