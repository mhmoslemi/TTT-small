import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: create a simple piecewise function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.01
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.rand(n_particles, n_points) * 0.01 - 0.005
    
    # Normalize particles to satisfy sum constraint
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Track personal bests
    p_best = particles.copy()
    p_best_vals = np.array([compute_c5(p, dx) for p in particles])
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_val
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Enforce sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_c5
                
                # Update global best if needed
                if current_c5 < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_c5
                    
                    # Early exit if target achieved
                    if g_best_val <= 0.3807:
                        break
                        
        iteration += 1
        
        # Update best solution
        if g_best_val < best_c5:
            best_c5 = g_best_val
            best_h = g_best.copy()
    
    return (best_h.tolist(), best_c5, n_points)
