```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sequential convex programming approach
        try:
            # Solve a relaxed linear program at each iteration
            h_current = h.copy()
            
            # Create CVXPY variables
            h_vars = cp.Variable(n_points)
            s = cp.Variable()
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1]
            
            # Integral constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Overlap constraints for each shift
            for k in range(-n_points+1, n_points):
                # Compute shifted indices manually
                shifted_indices = np.arange(n_points) - k
                shifted_indices = np.mod(shifted_indices, n_points)
                if k < 0:
                    # Negative shift
                    expr = cp.sum(cp.multiply(h_vars[:n_points+k], 1 - h_vars[shifted_indices[:n_points+k]]))
                elif k > 0:
                    # Positive shift
                    expr = cp.sum(cp.multiply(h_vars[k:], 1 - h_vars[shifted_indices[k:]]))
                else:
                    # Zero shift
                    expr = cp.sum(cp.multiply(h_vars, 1 - h_vars))
                constraints.append(s >= expr * dx)
            
            # Objective
            objective = cp.Minimize(s)
            
            # Form and solve the problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=100)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to gradient step
                step = 0.001
                h_temp = h.copy()
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
                
        except Exception as e:
            # Fallback to simple gradient descent
            step = 0.001
            h_temp = h.copy()
            
            corr = correlate(h_temp, 1 - h_temp, mode="full")
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            grad = np.zeros_like(h_temp)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            h_temp -= step * grad
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 50 == 0 and iteration > 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
