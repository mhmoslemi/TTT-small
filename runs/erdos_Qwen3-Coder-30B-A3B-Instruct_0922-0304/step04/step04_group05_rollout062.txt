import numpy as np
from scipy.signal import correlate
import time

def compute_overlap(h, dx):
    """Compute the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_onto_constraints(h, n_points, dx):
    """Project h onto the constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx):
    """Compute the gradient for the current h."""
    # Compute shifted sums
    h_shifted = np.concatenate([h, np.zeros(len(h))])
    h1 = np.concatenate([np.zeros(len(h)), 1 - h])
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad[i] = dx * (1 - 2 * h[i] - np.sum(h_shifted[i:i+len(h)] * h1[i:i+len(h)]))
    return grad

def armijo_line_search(h, grad, dx, alpha_max=1.0, c1=1e-4):
    """Perform Armijo line search to find step size alpha."""
    alpha = alpha_max
    for _ in range(20):  # Max iterations
        h_new = h - alpha * grad
        h_new = project_onto_constraints(h_new, len(h), dx)
        new_val = compute_overlap(h_new, dx)
        old_val = compute_overlap(h, dx)
        if new_val <= old_val - c1 * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= 0.5
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a triangular initial function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = 1.0 - 2.0 * i / n_points
        h[mid:] = 0.0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize
    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    patience = 0
    max_patience = 50
    
    # Optimizer loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Line search to determine step size
        alpha, h_new = armijo_line_search(h, grad, dx)
        
        # Update
        h = h_new
        
        # Evaluate
        current_c5 = compute_overlap(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if patience > max_patience:
            # Try a random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_onto_constraints(h, n_points, dx)
            patience = 0
            
    final_c5 = compute_overlap(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
