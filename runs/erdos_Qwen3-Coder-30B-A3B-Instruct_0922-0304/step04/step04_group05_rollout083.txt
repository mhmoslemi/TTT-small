import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def _solve_qp(H, c, A, b, delta_low, delta_high):
    n = H.shape[0]
    delta = cp.Variable(n)
    
    # Quadratic objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Linear constraints
    constraints = [A @ delta == b]
    
    # Bound constraints
    constraints.extend([delta >= delta_low, delta <= delta_high])
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Triangular initialization
        h = np.zeros(n_points)
        for j in range(n_points):
            h[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Precompute shift averages for efficiency
    shift_averages = []
    for k in range(2 * n_points - 1):
        shift_avg = np.zeros(n_points)
        if k < n_points:
            shift_avg[k:] = np.mean(h[:n_points - k])
        else:
            shift_avg[:n_points - k] = np.mean(h[k - n_points + 1:])
        shift_averages.append(shift_avg)
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and Hessian (linearized and quadratic parts)
        # Since overlap = ∫ h(x)(1-h(x+k)) dx, compute ∇C_k = dx * (1 - 2h - shift_average)
        # and ∇²C_k = dx * (-2*I + shift_matrix)
        
        # Compute gradient of max overlap
        c = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            # Compute h_i * (1 - h_{i+k}) contributions
            shift_avg = shift_averages[k]
            c += -2 * dx * h + dx * (1 - shift_avg)
        
        # Compute approximate quadratic part
        Q = np.zeros((n_points, n_points))
        for k in range(2 * n_points - 1):
            # Create shifted indicator matrix (simplified for sparse case)
            for j in range(n_points):
                if k < n_points and j + k < n_points:
                    Q[j, j+k] += -2 * dx
                elif k >= n_points and j >= k - n_points + 1:
                    Q[j, j-(k-n_points+1)] += -2 * dx
                else:
                    Q[j, j] += -2 * dx
        
        # Regularize to ensure positive definiteness
        Q += 1e-5 * np.eye(n_points)
        
        # Constraint matrices for sum constraint sum(h) = n_points/2
        A = np.ones((1, n_points))
        b = np.array([n_points / 2])
        
        # bounds on delta
        delta_low = np.zeros(n_points)
        delta_high = np.ones(n_points)
        
        # Solve QP
        delta = _solve_qp(Q, c, A, b, delta_low, delta_high)
        
        if delta is None:
            # fallback: perturb in direction of gradient
            delta = -c * 0.01
            
        # Line search
        alpha_max = 1.0
        for alpha in [1.0, 0.5, 0.1, 0.01]:
            h_new = h + alpha * delta
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                break
        
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
