import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    dx = 2.0 / n_points

    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Save current value
            old_h_i = h[i]
            
            # Try small adjustments
            step_size = 0.01
            best_h_i = old_h_i
            best_local_c5 = current_c5
            
            # Try increasing
            h[i] = min(1.0, old_h_i + step_size)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h, dx)
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_h_i = h[i]
                
            # Try decreasing
            h[i] = max(0.0, old_h_i - step_size)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h, dx)
            if c5 < best_local_c5:
                best_local_c5 = c5

            h[i] = old_h_i
            c5 = evaluate_c5(h, dx)
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_h_i = h[i]
                
            # Update if better
            if best_h_i != old_h_i:
                h[i] = best_h_i
                h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally perturb to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
