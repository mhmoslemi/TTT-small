```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a sequential convex programming approach with disciplined convex programming
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        try:
            # Create a new convex optimization problem each time
            h_var = cp.Variable(n_points)
            s = cp.Variable()
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Linearize the overlap constraints
            # For each k, we linearize: sum h[i] * (1 - h[(i+k)%n]) 
            # around the current h estimate
            for k in range(-n_points + 1, n_points):
                # We build constraints for shift k
                if k >= 0:
                    left = h_var
                    right = cp.hstack([h_var[k:], h_var[:k]])
                else:
                    left = h_var
                    right = cp.hstack([h_var[k:], h_var[:k]])
                
                # Linearization: relax h[i] * (1 - h[(i+k)%n]) 
                # around current values
                # Take a reasonable approximation using the current h values
                c = cp.sum(cp.multiply(left, 1 - right)) * dx
                # Approximation:
                # sum( h[i] * (1 - h[(i+k)%n]) )  
                # ~= sum( h_old[i] * (1 - h_old[(i+k)%n])) + 
                #     sum( (h[i] - h_old[i]) * (1 - h_old[(i+k)%n]) ) + 
                #     sum( h_old[i] * (h_old[(i+k)%n] - h[(i+k)%n]) )
                # We handle this as: 
                # c = sum( h[i] * (1 - h[(i+k)%n]) ) ~ s
                # We add s >= c' where c' is linearized at current h
                # s >= sum( h_old[i] * (1 - h_old[(i+k)%n]) ) 
                #       + sum( (h[i]-h_old[i]) * (1-h_old[(i+k)%n]) ) 
                #       + sum( h_old[i] * (h_old[(i+k)%n] - h[(i+k)%n]) )
                # -> s >= sum(h_old[i]*(1-h_old[(i+k)%n])) 
                #        + sum(h[i] * (1-h_old[(i+k)%n])) 
                #        - sum(h_old[i]*h_old[(i+k)%n]) 
                #        - sum(h_old[i]*h[(i+k)%n])
                # this simplifies to:
                # s >= sum( h_old[i] * (1 - h_old[(i+k)%n]) )
                #      + sum( h[i] * (1 - h_old[(i+k)%n]) ) 
                #      - sum( h_old[i] * h_old[(i+k)%n] )
                #      - sum( h_old[i] * h[(i+k)%n] )
                # yapf: disable
                constraints.append(s >= 
                                  np.sum(h * (1-h[np.roll(np.arange(n_points), -k)])) +
                                  cp.sum(cp.multiply(h_var, 1-h[np.roll(np.arange(n_points), -k)])) - 
                                  np.sum(h * h[np.roll(np.arange(n_points), -k)]) -
                                  cp.sum(cp.multiply(h, h_var[np.roll(np.arange(n_points), -k)]))
                )

            # Add the constraint for the actual term but with approx linearization
            # Since the full overlap term is more complex we reuse a direct approach
            # For each term, we approximate it well with linear terms
            # Approximate more directly using convex upper bounds  
            
            # Give up on convexification for simplicity and just do plain optimization
            # But with better heuristics and using cvxpy properly
            
            objective = cp.Minimize(s)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to small gradient descent
                step = 0.001
                iterations = 20
                h_temp = h.copy()
                for _ in range(iterations):
                    corr = correlate(h_temp, 1 - h_temp, mode="full")
                    max_idx = np.argmax(corr)
                    max_overlap = corr[max_idx] * dx
                    
                    grad = np.zeros_like(h_temp)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h_temp.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                        c5_plus = np.max(corr_plus) * dx
                        grad[i] = (c5_plus - max_overlap) / eps
                    
                    h_temp -= step * grad
                    h_temp = np.clip(h_temp, 0, 1)
                    h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                h = h_temp
        except:
            # Fallback to a variant of gradient descent if solver fails
            step = 0.001
            h_temp = h.copy()
            for _ in range(15):
                corr = correlate(h_temp, 1 - h_temp, mode="full")
                max_idx = np.argmax(corr)
                max_overlap = corr[max_idx] * dx
                
                grad = np.zeros_like(h_temp)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_temp.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - max_overlap) / eps
                
                h_temp -= step * grad
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h = h_temp
        
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally switch to a different heuristic
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
