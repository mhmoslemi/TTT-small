import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    # Compute the overlap integral C5
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * 0.02

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    n_particles = 50
    particles = []
    velocities = []
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create initial population
    for _ in range(n_particles):
        h = h_init + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Initialize personal bests and global best
    p_best = [p.copy() for p in particles]
    p_best_scores = [compute_c5(p) for p in particles]
    
    g_best = p_best[np.argmin(p_best_scores)].copy()
    g_best_score = min(p_best_scores)
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    best_c5 = g_best_score
    best_h = g_best.copy()
    
    max_iter = 6000  # iterations allowed within time budget
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        for i in range(n_particles):
            if (time.time() - start_time) > budget_s - 1:
                break
                
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c1 * r1 * (p_best[i] - h) + 
                 c2 * r2 * (g_best - h))
            
            # Velocity clamping
            v = np.clip(v, -0.02, 0.02)
            
            # Update position
            h = h + v
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate new position
            score = compute_c5(h)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = h.copy()
                p_best_scores[i] = score
                
            # Update global best
            if score < g_best_score:
                g_best = h.copy()
                g_best_score = score
                
            # Update particles and velocities
            particles[i] = h
            velocities[i] = v
            
        # Update best solution found so far
        if g_best_score < best_c5:
            best_c5 = g_best_score
            best_h = g_best.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
