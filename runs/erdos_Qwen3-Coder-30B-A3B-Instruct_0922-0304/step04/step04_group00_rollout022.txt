import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA parameters
    population_size = 50
    generations = 1000
    elite_size = population_size // 2
    tournament_size = population_size // 5
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h_ind in population:
            c5 = compute_c5(h_ind)
            fitness.append(c5)
            
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Selection (tournament)
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent = population[winner_idx].copy()
            
            # Mutate
            if np.random.rand() < 0.1:  # 10% mutation rate
                parent += np.random.normal(0, 0.01, n_points)
                parent = np.clip(parent, 0, 1)
                parent = parent * (n_points / 2) / np.sum(parent)
            
            new_population.append(parent)
            
        # Local search on top candidates
        sorted_pop = [population[i] for i in sorted_indices]
        for i in range(min(10, len(sorted_pop))):
            h_local = sorted_pop[i]
            # SLSQP local search
            try:
                result = minimize(
                    compute_c5,
                    h_local,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 30, 'ftol': 1e-8}
                )
                if result.success:
                    h_local = result.x
                    h_local = np.clip(h_local, 0, 1)
                    h_local = h_local * (n_points / 2) / np.sum(h_local)
                    # If improvement, replace
                    if compute_c5(h_local) < fitness[sorted_indices[i]]:
                        new_population[i] = h_local
            except:
                pass  # If local search fails, keep original
        
        # Update population
        population = new_population
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
