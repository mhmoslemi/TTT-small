import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple symmetric construction
        h = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 4
        h[mid-width:mid+width] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Try different interval configurations using a grid search
    d_candidates = np.linspace(0.0, 1.0, 100)
    best_d = 0.0
    best_c5 = float('inf')
    
    # For each d, create a piecewise constant function
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create two intervals: [0, 0.5] and [0.5+d, 1+d]
        # Convert to indices
        i0 = 0
        i1 = int(0.5 / dx)  # End index of first interval
        
        i2 = int((0.5 + d) / dx)  # Start index of second interval
        i3 = int((1.0 + d) / dx)  # End index of second interval
        
        # Ensure valid indices
        if i2 >= n_points:
            continue
            
        # Create h function with two intervals
        h_test = np.zeros(n_points)
        h_test[i0:i1+1] = 1.0
        h_test[i2:i3+1] = 1.0
        
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        c5 = evaluate_c5(h_test, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
            best_d = d
    
    # Fine search around best d
    if best_d > 0 and best_d < 1:
        fine_d = np.linspace(max(0, best_d - 0.02), min(1, best_d + 0.02), 50)
        for d in fine_d:
            if time.time() - start_time > budget_s - 1:
                break
                
            i0 = 0
            i1 = int(0.5 / dx)
            i2 = int((0.5 + d) / dx)
            i3 = int((1.0 + d) / dx)
            
            if i2 >= n_points:
                continue
                
            h_test = np.zeros(n_points)
            h_test[i0:i1+1] = 1.0
            h_test[i2:i3+1] = 1.0
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            c5 = evaluate_c5(h_test, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_test.copy()
    
    return (best_h.tolist(), best_c5, n_points)
