import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 300
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        new_h = h + np.random.normal(0, 0.05, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # Evaluation function
    def evaluate_population(pop):
        scores = []
        for ind in pop:
            score = evaluate_c5(ind, dx)
            scores.append(score)
        return scores
    
    # Selection operator (tournament)
    def tournament_selection(pop, scores, k=tournament_size):
        selected = []
        for _ in range(len(pop)):
            tournament_indices = np.random.choice(len(pop), k)
            tournament_scores = [scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_scores)]
            selected.append(pop[winner_idx].copy())
        return selected
    
    # Crossover operator
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        crossover_point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        
        return child1, child2
    
    # Mutation operator
    def mutate(individual):
        if np.random.rand() > mutation_rate:
            return individual.copy()
        
        # Add Gaussian noise
        mutated = individual + np.random.normal(0, 0.05, n_points)
        mutated = np.clip(mutated, 0, 1)
        
        # Rescale to maintain sum constraint
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        scores = evaluate_population(population)
        
        # Update best solution
        min_idx = np.argmin(scores)
        if scores[min_idx] < best_c5:
            best_c5 = scores[min_idx]
            best_h = population[min_idx].copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = tournament_selection(population, scores)
        
        # Create new population
        new_population = [population[min_idx].copy()]  # Elitism
        
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
