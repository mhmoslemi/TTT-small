import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Create a symmetric triangular window
        h0 = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h0[i] = i / mid
            else:
                h0[i] = 1 - (i - mid) / mid
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize particles around the initial solution
    for i in range(n_particles):
        particles[i] = h0 + np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    max_iter = 2000
    
    # Initialize personal bests
    personal_best = particles.copy()
    personal_best_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    
    # Global best
    gbest_idx = np.argmin(personal_best_c5)
    gbest = personal_best[gbest_idx].copy()
    gbest_c5 = personal_best_c5[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_c5
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = current_c5
            
            # Update global best
            if current_c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = current_c5
                
                if gbest_c5 < best_c5:
                    best_c5 = gbest_c5
                    best_h = gbest.copy()
        
        # Periodic reinitialization
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(n_particles)
                particles[idx] = h0 + np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
