import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    rho = 1.0
    max_iter = 2000
    epsilon = 1e-4
    
    # Initialize ADMM variables
    x = h.copy()
    z = np.zeros_like(x)
    u = np.zeros_like(x)

        # Update x (proximal step)
        # Minimize: ||x - z + u||^2 + rho * (overlap term)

        # Compute gradient approximation
        eps = 1e-5
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = np.clip(x_plus, 0, 1)
            x_plus = x_plus * (n_points / 2) / np.sum(x_plus)

        step_size = 0.01
