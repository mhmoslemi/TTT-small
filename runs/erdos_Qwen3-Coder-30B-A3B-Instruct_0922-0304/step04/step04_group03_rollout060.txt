import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5  # Uniform initial guess
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Early exit threshold
    target_c5 = 0.3807
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if current_c5 <= target_c5:
            break
            
        # Find the k that gives maximum overlap
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at shift k_max, gradient is partial derivative
            # d/dh[i] O_k_max = (1 - h[i + k_max]) - h[i - k_max]
            if k_max <= i:
                j1 = i - k_max
                j2 = i + k_max
            else:
                j1 = i + k_max
                j2 = i - k_max
            # Wrap around for indices
            j1 = j1 % n_points
            j2 = j2 % n_points
            gradient[i] = (1 - h[j1]) - h[j2]
        
        # Update h
        h = h - learning_rate * gradient
        
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Reduce learning rate every 100 iterations if no improvement
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
