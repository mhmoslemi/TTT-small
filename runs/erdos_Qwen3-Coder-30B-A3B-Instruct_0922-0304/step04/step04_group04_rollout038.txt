import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        # Renormalize to satisfy ∫h(x)dx = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a step-like construction
        h_init = np.zeros(n_points)
        half = n_points // 2
        h_init[:half] = 1
        # Renormalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # We'll use a variant of gradient-based optimization with restarts
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Gradient ascent on the correlation objective (minimizing overlap)
    # Use a variant that samples gradient directions
    
    # Try different gradient-based approaches  
    max_iter = 2000
    step_size = 0.01
    
    # Create a grid of random perturbations to try
    n_perturb = 20
    perturbations = np.random.rand(n_perturb, n_points) * 0.1
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all perturbations
        new_h = best_h.copy()
        candidate_h = []
        candidate_c5 = []
        
        for i in range(n_perturb):
            # Add noise
            new_h = best_h + step_size * perturbations[i]
            # Project to [0,1]
            new_h = np.clip(new_h, 0, 1)
            # Renormalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5 = evaluate_c5(new_h, dx)
            candidate_h.append(new_h.copy())
            candidate_c5.append(c5)
        
        # Find best among candidates
        min_c5_idx = np.argmin(candidate_c5)
        if candidate_c5[min_c5_idx] < best_c5:
            best_c5 = candidate_c5[min_c5_idx]
            best_h = candidate_h[min_c5_idx]
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
