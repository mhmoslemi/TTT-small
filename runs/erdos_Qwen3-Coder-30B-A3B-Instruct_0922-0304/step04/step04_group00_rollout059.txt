import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Coordinate-wise update
        for i in range(n_points):
            # Store current value
            old_h_i = h[i]
            
            # Compute the effect of changing h[i] 
            # We'll do a simple line search around the current value
            best_h_i = old_h_i
            best_local_c5 = current_c5
            
            # Try a few small adjustments
            step_sizes = np.linspace(-0.05, 0.05, 21)
            for step in step_sizes:
                new_h_i = old_h_i + step
                # Ensure it's within bounds
                new_h_i = np.clip(new_h_i, 0, 1)
                
                # Temporarily update
                h[i] = new_h_i
                
                # Re-normalize to maintain sum constraint
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                
                # Compute new C5
                new_c5 = compute_c5(h, dx)
                
                if new_c5 < best_local_c5:
                    best_local_c5 = new_c5
                    best_h_i = h[i]
                    
            # Set the best found value
            h[i] = best_h_i
            
        # Final normalization
        current_sum = np.sum(h)
        h = h * (n_points / 2) / current_sum
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
