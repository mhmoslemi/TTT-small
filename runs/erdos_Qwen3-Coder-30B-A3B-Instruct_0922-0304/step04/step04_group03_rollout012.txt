import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent solution if available, otherwise use triangular
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h = np.clip(h, 0, 1)
    
    # Normalize to ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent setup
    max_iter = 2000
    learning_rate = 0.01
    patience = 50
    best_c5 = evaluate_c5(h, dx)
    no_improve_count = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the maximum overlap index k
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Compute gradient for the max overlap case
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at k_max, gradient is partial derivative of O_k_max
            # Overlap sum = h[i] * (1 - h[(i+k_max) % n_points]) 
            # so d/dh[i] = (1 - h[(i+k_max) % n_points])
            # d/dh[(i-k_max) % n_points] = -h[i]
            idx1 = (i + k_max) % n_points
            idx2 = (i - k_max) % n_points
            gradient[i] = (1 - h[idx1]) - h[idx2] 
        
        # Update step with gradient
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Adaptive learning rate
        if new_c5 < best_c5:
            best_c5 = new_c5
            h = h_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > patience:
                learning_rate *= 0.9
                no_improve_count = 0
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (h.tolist(), best_c5, n_points)
